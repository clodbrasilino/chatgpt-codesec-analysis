```c
#include <stdio.h>
#include <stdlib.h>

int is_greater_than_all(int number, const int *arr, size_t size) {
    if (arr == NULL && size > 0) {
        return 0;
    }
    for (size_t i = 0; i < size; i++) {
        if (number <= arr[i]) {
            return 0;
        }
    }
    return 1;
}

int main(void) {
    int arr[] = {10, 20, 30, 40, 50};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int number;

    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (is_greater_than_all(number, arr, size)) {
        printf("%d is greater than all elements in the array\n", number);
    } else {
        printf("%d is not greater than all elements in the array\n", number);
    }

    return EXIT_SUCCESS;
}
```
