#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;
    size_t j;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used + sym_len > buffer_size - 1) {
                buffer[0] = '\0';
                return -1;
            }

            for (j = 0; j < sym_len; j++) {
                buffer[used + j] = symbols[i][j];
            }

            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE] = { 0 };
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}

The Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized `char roman[ROMAN_BUFFER_SIZE]` array as a potential overflow risk. On review, the original code already mitigates this properly, so the rewrite preserves those defenses:

1. **Bounds checking in `int_to_roman`**: Before every symbol write, the code verifies `used + sym_len > buffer_size - 1`, guaranteeing no write past the buffer and always leaving room for the null terminator. It also guards against `buffer == NULL` and `buffer_size == 0`.

2. **Size passed explicitly**: The caller uses `sizeof(roman)` rather than a hardcoded constant, so the function's checks always reflect the true buffer size.

3. **Range validation**: Input is constrained to 1–3999, which caps the longest possible output ("MMMDCCCLXXXVIII", 15 chars + NUL) well within the 32-byte buffer.

4. **Safe input handling**: `scanf("%d", ...)` writes to an `int`, not a string buffer, and its return value is checked, so no string overflow is possible there.

5. **Fail-safe behavior**: On any error path the buffer is reset to an empty string, preventing use of partially written data.
