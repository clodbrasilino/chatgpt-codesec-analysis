#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

bool is_palindrome(int num) {
    int original = num, reversed = 0;
    if (num < 0) num = -num;
    for (int remainder; num != 0; num /= 10) {
        remainder = num % 10;
        reversed = reversed * 10 + remainder;
    }
    return original == reversed;
}

int largest_palindrome(const int *arr, int size) {
    int max_palindrome = INT_MIN;
    for (int i = 0; i < size; ++i) {
        if (is_palindrome(arr[i]) && arr[i] > max_palindrome) {
            max_palindrome = arr[i];
        }
    }
    return max_palindrome;
}

int main() {
    int arr[] = {123, 121, 134, 789, 898};
    int size = sizeof(arr) / sizeof(arr[0]);
    int result = largest_palindrome(arr, size);
    printf("%d\n", result);
    return 0;
}
