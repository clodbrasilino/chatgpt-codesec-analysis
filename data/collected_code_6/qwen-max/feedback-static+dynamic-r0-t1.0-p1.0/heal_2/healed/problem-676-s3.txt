#include <stdio.h>
#include <string.h>
#include <regex.h>

void remove_non_alphanumeric(char *str) {
    regex_t regex;
    regmatch_t match[1];
    int offset = 0;

    if (regcomp(&regex, "[^a-zA-Z0-9]", REG_EXTENDED|REG_NEWLINE) != 0) {
        return;
    }

    while (regexec(&regex, str + offset, 1, match, 0) == 0) {
        size_t length = strlen(str + offset + match[0].rm_eo);
        memmove(str + offset + match[0].rm_so, str + offset + match[0].rm_eo, length + 1);
        offset += match[0].rm_eo - match[0].rm_so;
    }

    regfree(&regex);
}

int main() {
    char str[] = "Hello! This is a test. @123";
    remove_non_alphanumeric(str);
    printf("%s\n", str);
    return 0;
}
