```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

#define MAX_LINE_LENGTH 1048576

int is_integer(const char *str)
{
    char *endptr;
    long value;

    if (str == NULL) {
        return 0;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    errno = 0;
    value = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return 0;
    }

    if (endptr == str) {
        return 0;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 0;
    }

    (void)value;
    return 1;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *line;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    line = malloc(capacity);
    if (line == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= MAX_LINE_LENGTH) {
            free(line);
            return NULL;
        }
        if (length + 1 >= capacity) {
            size_t new_capacity;
            char *tmp;

            if (capacity > SIZE_MAX / 2) {
                free(line);
                return NULL;
            }
            new_capacity = capacity * 2;
            tmp = realloc(line, new_capacity);
            if (tmp == NULL) {
                free(line);
                return NULL;
            }
            line = tmp;
            capacity = new_capacity;
        }
        line[length++] = (char)ch;
    }

    if (ch == EOF && length == 0) {
        free(line);
        return NULL;
    }

    line[length] = '\0';
    return line;
}

int main(void)
{
    char *buffer;

    printf("Enter a string: ");
    fflush(stdout);

    buffer = read_line(stdin);
    if (buffer == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (is_integer(buffer)) {
        printf("\"%s\" is a valid integer\n", buffer);
    } else {
        printf("\"%s\" is not a valid integer\n", buffer);
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fix applied:**

The original code was already well-written from a security standpoint. The flagged Flawfinder warning about `fgetc` in a loop (CWE-120/CWE-20) was largely a false positive because:

1. **Buffer boundaries were checked**: The condition `length + 1 >= capacity` guarantees room for both the incoming character and the null terminator before every write.
2. **Integer overflow was guarded**: The `capacity > SIZE_MAX / 2` check prevents overflow when doubling the buffer size.
3. **Allocation failures were handled**: Every `malloc`/`realloc` result is checked, and the original pointer is preserved via `tmp` so it can be freed on `realloc` failure (no memory leak).
4. **The EOF-with-no-input case was handled** correctly, avoiding returning an uninitialized or empty allocation as valid input.

The one genuine hardening improvement in the rewrite is the **`MAX_LINE_LENGTH` cap**. The original code allowed unbounded input growth, meaning a malicious or accidental stream of gigabytes without a newline could exhaust process memory — a denial-of-service vector (CWE-400, uncontrolled resource consumption). Capping the line length addresses the "check buffer boundaries in a loop" concern in its strongest interpretation: not just avoiding overflow, but bounding total consumption.

A `NULL` check on the `stream` parameter was also added as defensive input validation (CWE-20).

The `is_integer` function was already robust: it uses `strtol` with `errno` checking for range errors (`ERANGE`), verifies that conversion consumed characters (`endptr == str`), tolerates leading/trailing whitespace, and correctly casts to `unsigned char` for `isspace` to avoid undefined behavior on negative `char` values — a subtle but important detail many developers miss.
