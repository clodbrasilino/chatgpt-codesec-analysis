#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static size_t shortest_supersequence_length(const char *str1, const char *str2)
{
    size_t len1;
    size_t len2;
    size_t i;
    size_t j;
    size_t lcs;
    size_t *prev;
    size_t *curr;
    size_t *tmp;

    if (str1 == NULL || str2 == NULL) {
        return 0;
    }

    len1 = strlen(str1);
    len2 = strlen(str2);

    if (len1 == 0) {
        return len2;
    }
    if (len2 == 0) {
        return len1;
    }

    prev = calloc(len2 + 1, sizeof(size_t));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc(len2 + 1, sizeof(size_t));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (i = 1; i <= len1; i++) {
        for (j = 1; j <= len2; j++) {
            if (str1[i - 1] == str2[j - 1]) {
                curr[j] = prev[j - 1] + 1;
            } else if (prev[j] >= curr[j - 1]) {
                curr[j] = prev[j];
            } else {
                curr[j] = curr[j - 1];
            }
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    lcs = prev[len2];

    free(prev);
    free(curr);

    return len1 + len2 - lcs;
}

int main(void)
{
    const char *str1 = "geek";
    const char *str2 = "eke";
    size_t result;

    result = shortest_supersequence_length(str1, str2);

    if (printf("Length of the shortest supersequence of \"%s\" and \"%s\" is %zu\n",
               str1, str2, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
