```c
#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

int match_a_followed_by_optional_b(const char *str)
{
    size_t i;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, MAX_INPUT_LEN);

    for (i = 0; i < len; i++) {
        if (str[i] == 'a') {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    const char *tests[] = { "ab", "a", "abc", "ac", "bb", "", "bab" };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        if (match_a_followed_by_optional_b(tests[i])) {
            if (printf("\"%s\" -> match\n", tests[i]) < 0) {
                return 1;
            }
        } else {
            if (printf("\"%s\" -> no match\n", tests[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

Key change: replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, which addresses the CWE-126 (buffer over-read) risk flagged by Flawfinder. If the input string is not NUL-terminated, the scan is capped at a defined maximum rather than reading past the buffer indefinitely. The rest of the logic — NULL check, bounded loop, and checked `printf` return values in `main` — was already defensive and remains unchanged.
