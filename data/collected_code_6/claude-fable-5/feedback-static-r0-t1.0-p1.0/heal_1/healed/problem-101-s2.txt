```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int find_kth_element(const int *arr, size_t n, size_t k, int *result)
{
    int *copy;
    size_t total_size;

    if (arr == NULL || result == NULL || n == 0 || k == 0 || k > n) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*copy)) {
        return -1;
    }

    total_size = n * sizeof(*copy);

    copy = malloc(total_size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, total_size);
    qsort(copy, n, sizeof(*copy), compare_ints);

    *result = copy[k - 1];
    free(copy);

    return 0;
}

int main(void)
{
    int arr[] = { 12, 3, 5, 7, 4, 19, 26 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 3;
    int result = 0;

    if (find_kth_element(arr, n, k, &result) != 0) {
        fprintf(stderr, "Error: invalid input or allocation failure\n");
        return EXIT_FAILURE;
    }

    if (printf("The %zu-th smallest element is %d\n", k, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and changes made:**

The Flawfinder warning on `memcpy` (CWE-120) flags the call because static analysis cannot verify that the destination buffer is large enough for the copied data. In the original code, the destination `copy` was allocated with `malloc(n * sizeof(*copy))` and exactly `n * sizeof(*copy)` bytes were copied, so the sizes matched. However, there was one genuine latent weakness: the multiplication `n * sizeof(*copy)` could overflow `size_t` for a sufficiently large `n`, resulting in a small allocation followed by a large copy — a classic heap buffer overflow.

Fixes applied:

1. **Integer overflow guard** — added a check `n > SIZE_MAX / sizeof(*copy)` before the multiplication, rejecting inputs that would wrap around. This is the key hardening step that makes the `memcpy` provably safe.
2. **Single size computation** — `total_size` is computed once and used for both `malloc` and `memcpy`, guaranteeing the destination capacity and copy length can never diverge.
3. **Added `<stdint.h>`** for `SIZE_MAX`.

The rest of the code was already sound: `qsort` with a correct three-way comparator (avoiding the `x - y` overflow anti-pattern), full parameter validation, `malloc` failure handling, no memory leaks, and checked `printf` return in `main`.
