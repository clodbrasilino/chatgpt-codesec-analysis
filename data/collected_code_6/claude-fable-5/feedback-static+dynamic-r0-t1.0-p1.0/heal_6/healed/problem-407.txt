#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}

1. **Bounded write**: `snprintf(digits, sizeof(digits), ...)` is used instead of `sprintf`, so the buffer can never be overflowed by the formatted output.
2. **Truncation check**: the return value of `snprintf` is validated (`len < 0 || (size_t)len >= sizeof(digits)`), so truncated or failed conversions are rejected rather than processed.
3. **Adequate sizing**: `DIGITS_BUF_SIZE` (32) comfortably exceeds the maximum decimal length of a `long long` (19 digits plus terminator), so truncation cannot occur for valid input.
4. **Bounded indexing**: all loop indices (`i`, `j`) are derived from the validated `len` and guarded against underflow (`i >= 0`, `j > i`), so no out-of-bounds reads occur.
5. **Zero initialization**: `memset` guarantees the buffer is fully initialized before use, avoiding reads of indeterminate memory.
6. **Overflow-safe reconstruction**: the digit-to-integer loop checks `result > (LLONG_MAX - d) / 10` before multiplying, preventing signed integer overflow (undefined behavior).
