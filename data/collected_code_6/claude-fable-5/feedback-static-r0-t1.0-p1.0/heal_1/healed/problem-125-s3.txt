```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096

int find_max_difference(const char *str)
{
    size_t len;
    size_t i;
    int current;
    int best;
    int value;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LENGTH);
    if (len == 0 || len >= MAX_INPUT_LENGTH) {
        return -1;
    }

    current = 0;
    best = -1;

    for (i = 0; i < len; i++) {
        if (str[i] == '0') {
            value = 1;
        } else if (str[i] == '1') {
            value = -1;
        } else {
            return -1;
        }

        current += value;
        if (current < value) {
            current = value;
        }
        if (current > best) {
            best = current;
        }
    }

    return best;
}

int main(void)
{
    const char *tests[] = {
        "11000010001",
        "1111",
        "0000",
        "101010"
    };
    size_t count;
    size_t i;

    count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < count; i++) {
        int result = find_max_difference(tests[i]);
        if (result < 0) {
            if (printf("String: %s -> no valid substring with more 0s than 1s\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("String: %s -> max difference: %d\n", tests[i], result) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Changes made to address the identified weaknesses:**

1. **CWE-126 (buffer over-read via `strlen`)**: Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LENGTH`. If the string is not null-terminated within the bound, `strnlen` returns `MAX_INPUT_LENGTH`, which is rejected as invalid input (`len >= MAX_INPUT_LENGTH`). This guarantees the read never exceeds a known limit, even if the caller passes a non-terminated buffer.

2. **[variableScope] on `result`**: The declaration of `result` was moved from function scope into the `for` loop body, reducing its scope to the smallest block where it is used, which improves readability and prevents accidental reuse of stale values across iterations.

The core algorithm (Kadane's algorithm treating `'0'` as +1 and `'1'` as -1 to find the substring with the maximum excess of zeros over ones) remains unchanged, as does the error-checked `printf` handling in `main`.
