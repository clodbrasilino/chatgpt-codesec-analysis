```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    size_t i;
    size_t first_idx;
    size_t last_idx;
    int found_first;
    int found_last;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    found_first = 0;
    found_last = 0;
    first_idx = 0U;
    last_idx = 0U;

    for (i = 0U; i < len; i++) {
        if (str[i] == ch) {
            if (found_first == 0) {
                first_idx = i;
                found_first = 1;
            }
            last_idx = i;
            found_last = 1;
        }
    }

    if (found_first == 0) {
        return;
    }

    if (found_last == 1 && last_idx != first_idx) {
        memmove(&str[last_idx], &str[last_idx + 1U], len - last_idx);
        len--;
    }

    memmove(&str[first_idx], &str[first_idx + 1U], len - first_idx);
}

static int discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);

    return c;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    if (strchr(buffer, '\n') == NULL) {
        (void)discard_remaining_input();
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        (void)discard_remaining_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already defensively written, and the Flawfinder hits are informational rather than actual defects:

1. **`char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** Mitigated. `fgets` is called with `sizeof(buffer)`, so it can never write past the array. The explicit `buffer[sizeof(buffer) - 1U] = '\0'` guarantees termination, and `remove_first_last` uses `strnlen` with `max_len` plus a re-termination guard, so it never trusts the string to be terminated.

2. **`getchar` in a loop (CWE-120/CWE-20):** Not a real risk here. `discard_remaining_input` never stores the characters — it only drains stdin until newline or `EOF`, so no buffer is involved. The `EOF` check prevents an infinite loop on closed input.

3. **Input hygiene:** Overlong lines are detected (`strchr(buffer, '\n') == NULL`) and the excess is flushed before the next prompt, preventing stale input from being consumed as the character choice. `EOF` from `getchar()` is handled before the narrowing cast to `char`.

4. **`memmove` bounds:** The move lengths (`len - last_idx` and `len - first_idx`) correctly include the null terminator and stay within the buffer, since `len < max_len` is enforced beforehand.

No functional changes were required — the rewrite simply strips the annotations, as every flagged pattern already has a corresponding mitigation in place.
