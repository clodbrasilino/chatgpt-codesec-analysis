#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <limits.h>

int starts_with_number(const char *str, int number) {
    char *num_str = malloc(20);
    if (num_str == NULL) return -1;
    snprintf(num_str, 20, "%d", number);
    size_t num_len = strnlen(num_str, 20);
    int result = strncmp(str, num_str, num_len) == 0;
    free(num_str);
    return result;
}

int main() {
    const char *test_str = "1234_start";
    int test_num = 1234;
    if (starts_with_number(test_str, test_num)) {
        printf("The string starts with the number %d.\n", test_num);
    } else {
        printf("The string does not start with the number %d.\n", test_num);
    }
    return 0;
}
