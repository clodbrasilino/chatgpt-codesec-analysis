#include <stdio.h>
#include <stdlib.h>

#define ROWS 4
#define COLS 2

int index_wise_addition(const int tup1[ROWS][COLS],
                        const int tup2[ROWS][COLS],
                        int result[ROWS][COLS])
{
    size_t i;
    size_t j;

    if ((tup1 == NULL) || (tup2 == NULL) || (result == NULL))
    {
        return -1;
    }

    for (i = 0U; i < (size_t)ROWS; i++)
    {
        for (j = 0U; j < (size_t)COLS; j++)
        {
            long long sum = (long long)tup1[i][j] + (long long)tup2[i][j];
            if ((sum > (long long)2147483647) || (sum < (long long)(-2147483647 - 1)))
            {
                return -2;
            }
            result[i][j] = (int)sum;
        }
    }

    return 0;
}

int main(void)
{
    int tup1[ROWS][COLS] = { {1, 3}, {4, 5}, {2, 9}, {1, 10} };
    int tup2[ROWS][COLS] = { {6, 7}, {3, 9}, {1, 1}, {7, 3} };
    int result[ROWS][COLS] = { {0, 0}, {0, 0}, {0, 0}, {0, 0} };
    size_t i;
    int status;

    status = index_wise_addition(tup1, tup2, result);
    if (status != 0)
    {
        (void)fprintf(stderr, "Error: addition failed with code %d\n", status);
        return EXIT_FAILURE;
    }

    (void)printf("(");
    for (i = 0U; i < (size_t)ROWS; i++)
    {
        (void)printf("(%d, %d)", result[i][0], result[i][1]);
        if (i < ((size_t)ROWS - 1U))
        {
            (void)printf(", ");
        }
    }
    (void)printf(")\n");

    return EXIT_SUCCESS;
}
