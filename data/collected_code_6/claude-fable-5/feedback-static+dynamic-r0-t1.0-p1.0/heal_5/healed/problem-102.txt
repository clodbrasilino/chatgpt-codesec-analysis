```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_SIZE 1024

static char *snake_to_camel(const char *snake, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(snake, max_len);
    if (len >= max_len) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = (char)tolower((unsigned char)snake[i]);
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT_SIZE];
    char *result;
    size_t len;

    memset(input, 0, sizeof(input));

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "No input provided\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input, sizeof(input));
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fix applied:**

The Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized `char input[MAX_INPUT_SIZE]` array. In this case, the warning is largely a false positive because the original code already mitigates the risk properly:

1. **Bounded reads:** `fgets(input, (int)sizeof(input), stdin)` limits input to the buffer size, preventing overflow at the read stage.
2. **Guaranteed termination:** `input[sizeof(input) - 1] = '\0'` forces null termination defensively, and `memset` zeroes the buffer beforehand.
3. **Bounded length calculation:** `strnlen` is used instead of `strlen`, so even a malformed buffer cannot cause an out-of-bounds scan.
4. **Bounds-checked conversion:** `snake_to_camel` validates its input against `max_len`, rejects unterminated strings (`len >= max_len`), and allocates exactly `len + 1` bytes. Since the output can only shrink (underscores are dropped), `j` never exceeds `len`, so the write into `camel` is safe.
5. **Safe character handling:** casting to `unsigned char` before `toupper`/`tolower` avoids undefined behavior with negative `char` values.
6. **Resource hygiene:** `malloc` results are checked, and `free` is called on all exit paths after allocation.

The only hardening added in the rewrite is a `max_len == 0` guard in `snake_to_camel`, which prevents a theoretical zero-length invariant violation if the function were reused elsewhere with an invalid size argument. Otherwise, the code was already sound with respect to the flagged weakness.
