```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

size_t longest_word_length(const char *str)
{
    size_t max_len = 0;
    size_t current_len = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i])) {
            current_len++;
            if (current_len > max_len) {
                max_len = current_len;
            }
        } else {
            current_len = 0;
        }
    }

    return max_len;
}

int main(void)
{
    size_t buffer_size = 1024;
    char *buffer;
    size_t result;

    buffer = calloc(buffer_size, sizeof(char));
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)buffer_size, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    buffer[buffer_size - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = longest_word_length(buffer);

    if (printf("Longest word length: %zu\n", result) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

The original code was already largely defensive, but the Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized stack buffer. The rewritten version addresses this and retains all existing mitigations:

1. **Heap allocation instead of a fixed stack array** — `calloc` replaces `char buffer[1024]`, which zero-initializes the memory (replacing the `memset`) and moves the buffer off the stack, reducing the impact of any potential overflow on the return address.

2. **Bounded input** — `fgets` is retained with the buffer size passed explicitly, so input can never exceed the allocation.

3. **Guaranteed termination** — the explicit null terminator at `buffer[buffer_size - 1]` is kept as a belt-and-suspenders measure, and `strcspn` safely strips the trailing newline without risking an out-of-bounds write.

4. **Safe character classification** — the cast to `unsigned char` before `isalpha()` is preserved, avoiding undefined behavior for negative `char` values.

5. **Complete error handling and resource cleanup** — allocation failure, read failure, and `printf` failure are all checked, and `free()` is called on every exit path to prevent memory leaks.
