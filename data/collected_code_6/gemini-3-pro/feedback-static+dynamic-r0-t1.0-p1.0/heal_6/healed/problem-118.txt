```c
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

int main(void) {
    int ch;
    int in_word = 0;
    int first = 1;

    printf("[");
    
    while ((ch = fgetc(stdin)) != EOF) {
        if (isspace(ch)) {
            if (in_word) {
                printf("'");
                in_word = 0;
            }
        } else {
            if (!in_word) {
                if (!first) {
                    printf(", ");
                }
                printf("'");
                first = 0;
                in_word = 1;
            }
            putchar(ch);
        }
    }
    
    if (ferror(stdin)) {
        return EXIT_FAILURE;
    }
    
    if (in_word) {
        printf("'");
    }
    printf("]\n");
    
    return EXIT_SUCCESS;
}
```
