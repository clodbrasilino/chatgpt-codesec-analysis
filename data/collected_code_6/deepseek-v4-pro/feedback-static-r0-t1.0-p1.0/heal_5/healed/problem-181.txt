#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

char *longest_common_prefix(char **strs, int strs_len) {
    if (strs_len <= 0 || strs == NULL) {
        char *empty = (char *)malloc(1);
        if (empty == NULL) return NULL;
        empty[0] = '\0';
        return empty;
    }

    if (strs[0] == NULL) {
        char *empty = (char *)malloc(1);
        if (empty == NULL) return NULL;
        empty[0] = '\0';
        return empty;
    }

    size_t min_len = strnlen(strs[0], SIZE_MAX);
    for (int i = 1; i < strs_len; i++) {
        if (strs[i] == NULL) {
            min_len = 0;
            break;
        }
        size_t current_len = strnlen(strs[i], SIZE_MAX);
        if (current_len < min_len) {
            min_len = current_len;
        }
    }

    if (min_len == 0) {
        char *empty = (char *)malloc(1);
        if (empty == NULL) return NULL;
        empty[0] = '\0';
        return empty;
    }

    char *prefix = (char *)malloc(min_len + 1);
    if (prefix == NULL) return NULL;

    size_t prefix_len = 0;
    for (size_t i = 0; i < min_len; i++) {
        char current_char = strs[0][i];
        int mismatch = 0;
        for (int j = 1; j < strs_len; j++) {
            if (i >= strnlen(strs[j], min_len + 1) || strs[j][i] != current_char) {
                mismatch = 1;
                break;
            }
        }
        if (mismatch) {
            break;
        }
        prefix[prefix_len++] = current_char;
    }

    prefix[prefix_len] = '\0';
    return prefix;
}
