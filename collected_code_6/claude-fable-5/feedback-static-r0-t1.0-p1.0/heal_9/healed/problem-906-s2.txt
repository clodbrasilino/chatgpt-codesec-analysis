```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "/([0-9]{4})/([0-9]{1,2})/([0-9]{1,2})(/|$)"
#define MAX_MATCHES 5
#define BUF_SIZE 16
#define MAX_URL_LEN 2048

typedef struct {
    int year;
    int month;
    int day;
} url_date_t;

static int copy_match(const char *source, size_t source_len,
                      const regmatch_t *match,
                      char *dest, size_t dest_size)
{
    size_t length;

    if (source == NULL || match == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }
    if (match->rm_so < 0 || match->rm_eo < match->rm_so) {
        return -1;
    }
    if ((size_t)match->rm_eo > source_len) {
        return -1;
    }

    length = (size_t)(match->rm_eo - match->rm_so);
    if (length >= dest_size) {
        return -1;
    }

    memcpy(dest, source + match->rm_so, length);
    dest[length] = '\0';
    return 0;
}

static int extract_date_from_url(const char *url, url_date_t *out_date)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCHES];
    char year_buf[BUF_SIZE];
    char month_buf[BUF_SIZE];
    char day_buf[BUF_SIZE];
    size_t url_len;
    int rc;
    long year;
    long month;
    long day;
    char *endptr;

    if (url == NULL || out_date == NULL) {
        return -1;
    }

    url_len = strnlen(url, MAX_URL_LEN);
    if (url_len >= MAX_URL_LEN) {
        return -1;
    }

    rc = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    rc = regexec(&regex, url, MAX_MATCHES, matches, 0);
    if (rc != 0) {
        regfree(&regex);
        return -1;
    }

    if (copy_match(url, url_len, &matches[1], year_buf, sizeof(year_buf)) != 0 ||
        copy_match(url, url_len, &matches[2], month_buf, sizeof(month_buf)) != 0 ||
        copy_match(url, url_len, &matches[3], day_buf, sizeof(day_buf)) != 0) {
        regfree(&regex);
        return -1;
    }

    regfree(&regex);

    year = strtol(year_buf, &endptr, 10);
    if (endptr == year_buf || *endptr != '\0' || year < 1 || year > 9999) {
        return -1;
    }

    month = strtol(month_buf, &endptr, 10);
    if (endptr == month_buf || *endptr != '\0' || month < 1 || month > 12) {
        return -1;
    }

    day = strtol(day_buf, &endptr, 10);
    if (endptr == day_buf || *endptr != '\0' || day < 1 || day > 31) {
        return -1;
    }

    out_date->year = (int)year;
    out_date->month = (int)month;
    out_date->day = (int)day;
    return 0;
}

int main(void)
{
    const char *urls[] = {
        "https://example.com/blog/2023/11/25/some-article",
        "https://example.com/news/2019/7/4/",
        "https://example.com/no-date-here",
        "https://example.com/2020/13/40/invalid"
    };
    size_t count;
    size_t i;
    url_date_t date;

    count = sizeof(urls) / sizeof(urls[0]);

    for (i = 0; i < count; i++) {
        if (extract_date_from_url(urls[i], &date) == 0) {
            if (printf("URL: %s -> Year: %d, Month: %d, Day: %d\n",
                       urls[i], date.year, date.month, date.day) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("URL: %s -> No valid date found\n", urls[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

1. **`memcpy` in `copy_match` (CWE-120)**: This is a false positive. Before the `memcpy` executes, the code validates that the match offsets are non-negative and well-ordered (`rm_so >= 0`, `rm_eo >= rm_so`), that `rm_eo` does not exceed the source length, and critically that `length >= dest_size` causes an early return. Since the copy length is strictly less than `dest_size`, both the copy and the null terminator write at `dest[length]` are in bounds.

2. **Statically-sized buffers `year_buf`, `month_buf`, `day_buf` (CWE-119/CWE-120)**: Also false positives. All writes to these buffers go through `copy_match`, which enforces bounds via `sizeof(...)` passed as `dest_size`. The regex pattern itself limits captures to at most 4 digits, well under `BUF_SIZE` (16), providing defense in depth.

Additional hardening already present: `strnlen` with `MAX_URL_LEN` prevents unbounded scanning of the input, `strtol` results are range-validated with `endptr` checks, `regfree` is called on every exit path after `regcomp` succeeds (no resource leak), and all pointer parameters are null-checked. No changes were required beyond removing the annotation comments.
