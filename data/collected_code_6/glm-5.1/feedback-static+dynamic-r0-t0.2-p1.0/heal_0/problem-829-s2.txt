#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *str;
    int count;
} StrCount;

char* findSecondMostRepeated(char **seq, int n) {
    if (seq == NULL || n <= 1) {
        return NULL;
    }

    StrCount *counts = (StrCount *)malloc(n * sizeof(StrCount));
    if (counts == NULL) {
        return NULL;
    }

    int unique_size = 0;

    for (int i = 0; i < n; i++) {
        if (seq[i] == NULL) {
            free(counts);
            return NULL;
        }
        int found = 0;
        for (int j = 0; j < unique_size; j++) {
            if (strcmp(counts[j].str, seq[i]) == 0) {
                counts[j].count++;
                found = 1;
                break;
            }
        }
        if (!found) {
            counts[unique_size].str = seq[i];
            counts[unique_size].count = 1;
            unique_size++;
        }
    }

    if (unique_size < 2) {
        free(counts);
        return NULL;
    }

    int first_max = -1;
    int second_max = -1;

    for (int i = 0; i < unique_size; i++) {
        if (counts[i].count > first_max) {
            second_max = first_max;
            first_max = counts[i].count;
        } else if (counts[i].count > second_max && counts[i].count < first_max) {
            second_max = counts[i].count;
        }
    }

    char *result = NULL;
    if (second_max != -1) {
        for (int i = 0; i < unique_size; i++) {
            if (counts[i].count == second_max) {
                result = counts[i].str;
                break;
            }
        }
    }

    free(counts);
    return result;
}

int main() {
    char *arr[] = {"apple", "banana", "apple", "orange", "banana", "apple", "kiwi", "banana"};
    int n = sizeof(arr) / sizeof(arr[0]);

    char *result = findSecondMostRepeated(arr, n);

    if (result != NULL) {
        printf("%s\n", result);
    } else {
        printf("No second most repeated string found.\n");
    }

    return 0;
}
