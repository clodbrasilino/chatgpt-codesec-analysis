```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void remove_spaces(char *str) {
    if (str == NULL) {
        return;
    }

    int count = 0;
    for (int i = 0; str[i]; i++) {
        if (str[i] != ' ') {
            str[count++] = str[i];
        }
    }
    str[count] = '\0';
}

int main(void) {
    char str[] = "Hello World! This is a test string.";
    
    remove_spaces(str);
    
    printf("%s\n", str);
    
    return 0;
}
```
