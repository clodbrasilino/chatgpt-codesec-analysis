#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *create_node(int data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        exit(EXIT_FAILURE);
    }
    node->data = data;
    node->next = NULL;
    return node;
}

Node *add_lists(Node *list1, Node *list2) {
    Node dummy;
    dummy.next = NULL;
    Node *tail = &dummy;
    int carry = 0;

    while (list1 != NULL || list2 != NULL || carry != 0) {
        int sum = carry;
        if (list1 != NULL) {
            sum += list1->data;
            list1 = list1->next;
        }
        if (list2 != NULL) {
            sum += list2->data;
            list2 = list2->next;
        }
        carry = sum / 10;
        Node *new_node = create_node(sum % 10);
        tail->next = new_node;
        tail = tail->next;
    }

    return dummy.next;
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
}

int main(void) {
    Node *list1 = create_node(9);
    list1->next = create_node(9);
    list1->next->next = create_node(9);

    Node *list2 = create_node(1);
    list2->next = create_node(5);

    Node *result = add_lists(list1, list2);

    Node *curr = result;
    while (curr != NULL) {
        printf("%d ", curr->data);
        curr = curr->next;
    }
    printf("\n");

    free_list(list1);
    free_list(list2);
    free_list(result);

    return 0;
}
