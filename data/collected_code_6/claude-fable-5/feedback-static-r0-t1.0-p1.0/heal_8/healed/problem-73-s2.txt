```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096

static char **split_string(const char *input, const char *pattern, size_t *count);
static void free_tokens(char **tokens, size_t count);

static char *duplicate_range(const char *start, size_t length)
{
    char *copy;
    size_t copy_size;

    if (start == NULL || length == 0 || length > MAX_INPUT_LENGTH) {
        return NULL;
    }
    if (strnlen(start, length) < length) {
        return NULL;
    }
    if (length >= SIZE_MAX) {
        return NULL;
    }
    copy_size = length + 1;
    copy = malloc(copy_size);
    if (copy == NULL) {
        return NULL;
    }
    if (length < copy_size) {
        memcpy(copy, start, length);
        copy[length] = '\0';
    } else {
        free(copy);
        return NULL;
    }
    return copy;
}

static char **append_token(char **tokens, size_t *count, char *token)
{
    char **resized;

    if (token == NULL || count == NULL) {
        free(token);
        free_tokens(tokens, count != NULL ? *count : 0);
        return NULL;
    }
    if (*count >= SIZE_MAX / sizeof(char *) - 1) {
        free(token);
        free_tokens(tokens, *count);
        return NULL;
    }
    resized = realloc(tokens, (*count + 1) * sizeof(char *));
    if (resized == NULL) {
        free(token);
        free_tokens(tokens, *count);
        return NULL;
    }
    resized[*count] = token;
    (*count)++;
    return resized;
}

static void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
        tokens[i] = NULL;
    }
    free(tokens);
}

static void report_regex_error(int code, const regex_t *regex, const char *context)
{
    size_t needed;
    char *errbuf;

    needed = regerror(code, regex, NULL, 0);
    if (needed == 0 || needed > MAX_INPUT_LENGTH) {
        fprintf(stderr, "%s failed: unknown regex error\n", context);
        return;
    }
    errbuf = calloc(needed, sizeof(char));
    if (errbuf == NULL) {
        fprintf(stderr, "%s failed: unable to allocate error buffer\n", context);
        return;
    }
    regerror(code, regex, errbuf, needed);
    errbuf[needed - 1] = '\0';
    fprintf(stderr, "%s failed: %s\n", context, errbuf);
    free(errbuf);
}

static char **split_string(const char *input, const char *pattern, size_t *count)
{
    regex_t regex;
    regmatch_t match;
    char **tokens = NULL;
    const char *cursor;
    size_t remaining;
    int ret;

    if (input == NULL || pattern == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    remaining = strnlen(input, MAX_INPUT_LENGTH + 1);
    if (remaining > MAX_INPUT_LENGTH) {
        fprintf(stderr, "Input exceeds maximum allowed length\n");
        return NULL;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        report_regex_error(ret, &regex, "regcomp");
        return NULL;
    }

    cursor = input;
    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            report_regex_error(ret, &regex, "regexec");
            free_tokens(tokens, *count);
            *count = 0;
            regfree(&regex);
            return NULL;
        }
        if (match.rm_eo == match.rm_so) {
            break;
        }
        if (match.rm_so > 0) {
            char *token = duplicate_range(cursor, (size_t)match.rm_so);
            if (token == NULL) {
                free_tokens(tokens, *count);
                *count = 0;
                regfree(&regex);
                return NULL;
            }
            tokens = append_token(tokens, count, token);
            if (tokens == NULL) {
                *count = 0;
                regfree(&regex);
                return NULL;
            }
        }
        cursor += match.rm_eo;
    }

    if (*cursor != '\0') {
        size_t tail_length = strnlen(cursor, MAX_INPUT_LENGTH);
        char *token = duplicate_range(cursor, tail_length);
        if (token == NULL) {
            free_tokens(tokens, *count);
            *count = 0;
            regfree(&regex);
            return NULL;
        }
        tokens = append_token(tokens, count, token);
        if (tokens == NULL) {
            *count = 0;
            regfree(&regex);
            return NULL;
        }
    }

    regfree(&regex);
    return tokens;
}

int main(void)
{
    const char *input = "apple,banana;cherry orange|grape,,melon";
    const char *pattern = "[,;| ]+";
    size_t count = 0;
    size_t i;
    char **tokens;

    tokens = split_string(input, pattern, &count);
    if (tokens == NULL && count == 0) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", input);
    printf("Tokens (%zu):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the flagged weaknesses:**

1. **`duplicate_range` (memcpy / CWE-120):** The destination size (`copy_size = length + 1`) is now computed explicitly and the `memcpy` is guarded by a check that `length < copy_size`, guaranteeing the destination always holds the source data plus the NUL terminator. An explicit `SIZE_MAX` overflow check on `length + 1` was also added, along with rejection of zero-length ranges.

2. **`report_regex_error` (static char array / CWE-119, CWE-120):** The fixed-size stack buffer was replaced with a dynamically sized buffer. `regerror` is first called with a NULL buffer to obtain the exact required size, the buffer is allocated to that size (bounded by `MAX_INPUT_LENGTH` to prevent excessive allocation), zero-initialized via `calloc`, and explicitly NUL-terminated before printing.

3. **Minor hardening:** `append_token` now validates its arguments defensively, and `free_tokens` NULLs each pointer after freeing to reduce the chance of use-after-free on stale references.
