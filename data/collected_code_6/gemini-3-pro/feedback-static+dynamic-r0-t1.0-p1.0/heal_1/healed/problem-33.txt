```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void decimalToBinary(int n) {
    if (n < 0) {
        printf("Error: Please enter a non-negative integer.\n");
        return;
    }

    if (n == 0) {
        printf("0\n");
        return;
    }

    unsigned int num = (unsigned int)n;
    unsigned int mask = 1U << (sizeof(num) * CHAR_BIT - 1);
    int started = 0;

    while (mask > 0) {
        if ((num & mask) != 0) {
            started = 1;
            putchar('1');
        } else if (started) {
            putchar('0');
        }
        mask >>= 1;
    }
    putchar('\n');
}

int main(void) {
    int number = 0;
    
    if (scanf("%d", &number) != 1) {
        printf("Error: Invalid input.\n");
        return EXIT_FAILURE;
    }

    decimalToBinary(number);
    
    return EXIT_SUCCESS;
}
```
