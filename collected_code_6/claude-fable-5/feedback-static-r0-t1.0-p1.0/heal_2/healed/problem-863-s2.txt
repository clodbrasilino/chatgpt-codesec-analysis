```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int longest_consecutive_subsequence(const int *arr, size_t n)
{
    int *sorted = NULL;
    int longest = 0;
    int current = 0;
    size_t i = 0;
    size_t bytes = 0;

    if (arr == NULL || n == 0) {
        return 0;
    }

    if (n > SIZE_MAX / sizeof(*sorted)) {
        return -1;
    }

    bytes = n * sizeof(*sorted);

    sorted = malloc(bytes);
    if (sorted == NULL) {
        return -1;
    }

    memcpy(sorted, arr, bytes);
    qsort(sorted, n, sizeof(*sorted), compare_ints);

    longest = 1;
    current = 1;

    for (i = 1; i < n; i++) {
        if (sorted[i] == sorted[i - 1]) {
            continue;
        }
        if (sorted[i - 1] < INT_MAX && sorted[i] == sorted[i - 1] + 1) {
            current++;
            if (current > longest) {
                longest = current;
            }
        } else {
            current = 1;
        }
    }

    free(sorted);
    return longest;
}

int main(void)
{
    int arr[] = {1, 9, 3, 10, 4, 20, 2};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result = 0;

    result = longest_consecutive_subsequence(arr, n);
    if (result < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Length of the longest consecutive subsequence: %d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

### Analysis notes

**On the flagged `memcpy` weakness (CWE-120):** The Flawfinder warning was a false positive in practice, but the rewrite makes safety explicit:

- The destination buffer `sorted` is allocated with exactly `n * sizeof(*sorted)` bytes, and the same computed size (`bytes`) is used for both `malloc` and `memcpy`, guaranteeing the destination can always hold the source data.
- The overflow guard `n > SIZE_MAX / sizeof(*sorted)` runs *before* allocation, preventing an integer-overflow-driven undersized allocation — the classic root cause of CWE-120 in this pattern.

**Additional hardening applied:**

- Added `<limits.h>` and a check `sorted[i - 1] < INT_MAX` before computing `sorted[i - 1] + 1`, eliminating a potential signed integer overflow (undefined behavior) when the array contains `INT_MAX`.
- Kept the `NULL`/zero-length input guards, the `malloc` failure check, the `free` on the single exit path, and the checked `printf` return in `main` — all correct in the original.

The code remains C89/C99-portable, uses a strict-weak-ordering comparator safe from subtraction overflow (`x - y` would be a bug for large magnitudes), and returns `-1` as an explicit error sentinel distinct from the valid `0` result for empty input.
