#define MAX_INPUT_LEN 99

int min_swaps(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }
    
    size_t len1 = strnlen(str1, MAX_INPUT_LEN + 1);
    size_t len2 = strnlen(str2, MAX_INPUT_LEN + 1);

            count0_str1++;
        } else if (str1[i] == '1') {
            count1_str1++;
        } else {
            return -1;

            return -1;
        }

        return -1;

            } else {
                free(str1_copy);

            }
        }
    }

    return swaps;
}
