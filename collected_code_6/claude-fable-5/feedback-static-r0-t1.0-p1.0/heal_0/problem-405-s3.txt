```c
#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

typedef struct {
    int *elements;
    size_t size;
} Tuple;

bool tuple_contains(const Tuple *tuple, int value)
{
    if (tuple == NULL || tuple->elements == NULL) {
        return false;
    }

    for (size_t i = 0; i < tuple->size; i++) {
        if (tuple->elements[i] == value) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    int data[] = {10, 20, 30, 40, 50};
    Tuple tuple = { data, sizeof(data) / sizeof(data[0]) };

    int search_values[] = {30, 99};
    size_t num_searches = sizeof(search_values) / sizeof(search_values[0]);

    for (size_t i = 0; i < num_searches; i++) {
        if (tuple_contains(&tuple, search_values[i])) {
            if (printf("%d exists in the tuple.\n", search_values[i]) < 0) {
                return 1;
            }
        } else {
            if (printf("%d does not exist in the tuple.\n", search_values[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```
