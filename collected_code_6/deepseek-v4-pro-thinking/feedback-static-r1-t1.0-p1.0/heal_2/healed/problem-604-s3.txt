#define _GNU_SOURCE
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

static void reverse_range(char *begin, char *end) {
    while (begin < end) {
        char tmp = *begin;
        *begin = *end;
        *end = tmp;
        ++begin;
        --end;
    }
}

char *reverse_words(char *str, size_t size) {
    if (str == NULL || size == 0) {
        return NULL;
    }

    size_t len = strnlen(str, size);
    if (len == size) {
        str[size - 1] = '\0';
        len = size - 1;
    }

    if (len == 0) {
        return str;
    }

    reverse_range(str, str + len - 1);

    char *p = str;
    while (*p != '\0') {
        if (isspace((unsigned char)*p)) {
            ++p;
        } else {
            char *start = p;
            while (*p != '\0' && !isspace((unsigned char)*p)) {
                ++p;
            }
            reverse_range(start, p - 1);
        }
    }

    return str;
}

int main(void) {
    char *line = NULL;
    size_t bufsize = 0;
    ssize_t len = getline(&line, &bufsize, stdin);

    if (len != -1) {
        if (len > 0 && line[len - 1] == '\n') {
            line[len - 1] = '\0';
        }
        reverse_words(line, bufsize);
        puts(line);
    }

    free(line);
    return 0;
}
