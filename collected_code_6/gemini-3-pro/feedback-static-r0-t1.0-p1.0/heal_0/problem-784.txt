#include <stdio.h>
#include <stdlib.h>

long long findMultiplicationOfFirstEvenAndOdd(const int *list, size_t size) {
    if (list == NULL || size == 0) {
        return 0;
    }

    int firstEven = 0;
    int firstOdd = 0;
    int foundEven = 0;
    int foundOdd = 0;

    for (size_t i = 0; i < size; ++i) {
        if (!foundEven && list[i] % 2 == 0) {
            firstEven = list[i];
            foundEven = 1;
        }
        if (!foundOdd && list[i] % 2 != 0) {
            firstOdd = list[i];
            foundOdd = 1;
        }
        if (foundEven && foundOdd) {
            break;
        }
    }

    if (foundEven && foundOdd) {
        return (long long)firstEven * firstOdd;
    }

    return 0;
}

int main(void) {
    int list1[] = {1, 3, 5, 4, 7, 8};
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    long long result1 = findMultiplicationOfFirstEvenAndOdd(list1, size1);
    printf("Result 1: %lld\n", result1);

    int list2[] = {2, 4, 6};
    size_t size2 = sizeof(list2) / sizeof(list2[0]);
    long long result2 = findMultiplicationOfFirstEvenAndOdd(list2, size2);
    printf("Result 2: %lld\n", result2);

    int list3[] = {1, 3, 5};
    size_t size3 = sizeof(list3) / sizeof(list3[0]);
    long long result3 = findMultiplicationOfFirstEvenAndOdd(list3, size3);
    printf("Result 3: %lld\n", result3);

    return EXIT_SUCCESS;
}
