```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * CHAR_BIT)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = fgetc(stdin);
        guard++;
        if (guard >= (size_t)INT_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);

    return ch;
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        (void)discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`fgetc` in a loop (CWE-120, CWE-20):** This is a false positive in practical terms — `fgetc` reads a single character into an `int` and never writes to a user-controlled buffer. Nevertheless, the code defends against unbounded looping with the `guard` counter capped at `INT_MAX`, and correctly terminates on both `'\n'` and `EOF`. No fix required beyond what is present.

2. **Statically-sized `char` array (CWE-119/CWE-120):** All accesses to `input` are bounds-safe:
   - `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size including the null terminator.
   - Explicit re-termination via `input[sizeof(input) - 1U] = '\0'` guarantees a null-terminated string even in pathological cases.
   - `strnlen` is used instead of `strlen`, bounding the scan to the buffer size.
   - Overlong input (no newline found and buffer full) is detected, the remainder of stdin is drained via `discard_remaining_input()`, and the program rejects the input rather than silently truncating it.

3. **Integer overflow in conversion:** `binary_to_decimal` rejects strings longer than the bit width of `unsigned long` (now expressed portably with `CHAR_BIT` rather than a hard-coded `8U`) and additionally checks `value > (ULONG_MAX >> 1)` before each shift, making overflow impossible.

4. **Input validation (CWE-20):** Every character is validated to be `'0'` or `'1'`; NULL pointers and empty/oversized inputs are rejected with a distinct error path.

The code as given was already defensively written; the only substantive change in the rewrite is replacing the magic number `8U` with `CHAR_BIT` for portability to platforms where a byte is not 8 bits.
