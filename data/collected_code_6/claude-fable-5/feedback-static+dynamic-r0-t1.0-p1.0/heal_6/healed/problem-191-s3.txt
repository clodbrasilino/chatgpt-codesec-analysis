#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32U

static int has_thirty_days(const char *month);
static void to_lower_string(char *dest, const char *src, size_t dest_size);
static size_t bounded_strlen(const char *str, size_t max_len);
static void flush_stdin(void);

static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

static int has_thirty_days(const char *month)
{
    static const char *const thirty_day_months[] = {
        "april",
        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN];
    size_t i;
    size_t count;

    if (month == NULL) {
        return -1;
    }

    if (bounded_strlen(month, MAX_INPUT_LEN) >= MAX_INPUT_LEN) {
        return -1;
    }

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    for (i = 0U; i < count; i++) {
        if (strcmp(lowered, thirty_day_months[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = bounded_strlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        flush_stdin();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "Empty input provided.\n");
        return EXIT_FAILURE;
    }

    result = has_thirty_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name provided.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

1. **`getchar` in a loop (CWE-120, CWE-20)** — `flush_stdin()` reads and discards characters without storing them in any buffer, so no overflow is possible. The loop terminates correctly on both `'\n'` and `EOF`, preventing an infinite loop on closed streams.

2. **Statically-sized `char lowered[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — `to_lower_string()` receives `sizeof(lowered)` and writes at most `dest_size - 1` characters plus a null terminator. Additionally, `bounded_strlen()` rejects unterminated or oversized input before the copy occurs.

3. **Statically-sized `char input[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — Input is read with `fgets(input, sizeof(input), stdin)`, which enforces the buffer bound. Explicit null termination at `input[sizeof(input) - 1U]` is a defensive redundancy, and leftover characters beyond the buffer are drained via `flush_stdin()` so they cannot pollute subsequent reads.

Other good practices already present: `tolower()` is called with an `unsigned char` cast (avoids UB on negative `char` values), all pointer parameters are NULL-checked, `strcmp` operates only on guaranteed null-terminated strings, and array count is computed with `sizeof` division rather than a hardcoded constant. The Flawfinder hits are informational (risk 1–2) and all are properly mitigated; no functional changes were required.
