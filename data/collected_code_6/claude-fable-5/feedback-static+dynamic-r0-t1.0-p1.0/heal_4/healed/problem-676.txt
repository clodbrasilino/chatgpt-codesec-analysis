```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_non_alphanumeric(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_pos;
    size_t result_capacity;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result_capacity = input_len + 1;
    result = malloc(result_capacity);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_pos = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t match_len;
        size_t remaining_space;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret != 0) {
            break;
        }
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            break;
        }

        match_len = (size_t)(match.rm_eo - match.rm_so);

        if (result_pos >= result_capacity) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        remaining_space = result_capacity - 1 - result_pos;

        if (match_len > remaining_space) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        memcpy(result + result_pos, cursor + match.rm_so, match_len);
        result_pos += match_len;
        cursor += match.rm_eo;
    }

    result[result_pos] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a@b#c$d%e^1&2*3",
        "!!!???...",
        "AlreadyClean123"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alphanumeric(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n",
                    test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: %s\nCleaned:  %s\n\n",
               test_strings[i], cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive, but the flagged `memcpy` (CWE-120) warranted a cleaner, provably correct bounds check. Key changes in the rewrite:

1. **Explicit capacity tracking** — `result_capacity` is stored once at allocation time, so all subsequent bounds checks reference the actual allocated size rather than re-deriving it from `input_len`. This makes the invariant between allocation and copy self-evident.

2. **Simplified, non-overflowing bounds check** — the original triple condition (`match_len > input_len || result_pos > input_len - match_len || ...`) contained a redundant third clause that could theoretically wrap on unsigned arithmetic in other contexts. The rewrite computes `remaining_space = result_capacity - 1 - result_pos` only after verifying `result_pos < result_capacity`, guaranteeing no unsigned underflow, then checks `match_len > remaining_space`. This ensures `memcpy` can never write past `result` and always leaves room for the NUL terminator.

3. **Preserved correctness properties** — input length is capped via `strnlen` with `MAX_INPUT_LEN` (preventing unbounded reads on non-terminated input), all error paths free both the buffer and the compiled regex (no leaks), and the regex handle is released on every exit path after successful `regcomp`.

The destination buffer can now demonstrably always hold the copied data, resolving the Flawfinder finding.
