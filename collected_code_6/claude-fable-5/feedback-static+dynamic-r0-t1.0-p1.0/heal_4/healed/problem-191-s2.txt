#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stddef.h>

#define MAX_INPUT_LEN 32

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len; i++) {
        if (str[i] == '\0') {
            return i;
        }
    }

    return max_len;
}

static bool normalize_month(const char *month, char *normalized, size_t normalized_size)
{
    size_t len;
    size_t i;

    if (month == NULL || normalized == NULL || normalized_size == 0) {
        return false;
    }

    len = bounded_length(month, normalized_size);
    if (len == 0 || len >= normalized_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN] = {0};
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strncmp(normalized, thirty_day_months[i], sizeof(normalized)) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN] = {0};
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strncmp(normalized, all_months[i], sizeof(normalized)) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < 4096);
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;

    printf("Enter a month name: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = bounded_length(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %.31s\n", input);
        return 1;
    }

    if (has_thirty_days(input)) {
        printf("%.31s has 30 days.\n", input);
    } else {
        printf("%.31s does not have 30 days.\n", input);
    }

    return 0;
}

The rewritten code addresses each flagged weakness as follows:

- **CWE-119/CWE-120 (statically-sized `char` arrays):** All fixed-size buffers (`normalized`, `input`) are now zero-initialized at declaration, and every write into them remains strictly bounds-checked via `bounded_length()` and `normalize_month()`, which rejects input at or beyond buffer capacity before any copy occurs. `strcmp` was replaced with `strncmp` bounded by the buffer size, guaranteeing comparisons never read past the array even if a terminator were somehow missing.

- **CWE-120/CWE-20 (`getchar` in a loop):** `discard_remaining_input()` now includes a guard counter capping iterations, preventing an unbounded loop on pathological input streams while still draining a normal line. It never writes the characters anywhere, so no buffer can overflow.

- **Input handling in `main`:** `fgets` is already length-bounded by `sizeof(input)`; an explicit NUL terminator is forced at the final byte as defense-in-depth, and format specifiers were changed to `%.31s` so `printf`/`fprintf` output is precision-limited and cannot over-read even in the absence of a terminator.
