#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
} List;

Node *create_node(int data) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

List *create_list(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    return list;
}

int append_node(List *list, int data) {
    Node *new_node = create_node(data);
    if (new_node == NULL) {
        return 0;
    }
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    return 1;
}

void free_list(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

void remove_column(List **lists, int num_rows, int col_index) {
    if (lists == NULL || num_rows <= 0 || col_index < 0) {
        return;
    }
    for (int i = 0; i < num_rows; i++) {
        if (lists[i] == NULL || lists[i]->head == NULL) {
            continue;
        }
        if (col_index == 0) {
            Node *temp = lists[i]->head;
            lists[i]->head = temp->next;
            free(temp);
        } else {
            Node *current = lists[i]->head;
            int pos = 0;
            while (current != NULL && pos < col_index - 1) {
                current = current->next;
                pos++;
            }
            if (current != NULL && current->next != NULL) {
                Node *temp = current->next;
                current->next = temp->next;
                free(temp);
            }
        }
    }
}

int main(void) {
    int num_rows = 3;
    List **lists = (List **)malloc(num_rows * sizeof(List *));
    if (lists == NULL) {
        return 1;
    }
    for (int i = 0; i < num_rows; i++) {
        lists[i] = create_list();
        if (lists[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free_list(lists[j]);
            }
            free(lists);
            return 1;
        }
    }
    append_node(lists[0], 1);
    append_node(lists[0], 2);
    append_node(lists[0], 3);
    append_node(lists[1], 4);
    append_node(lists[1], 5);
    append_node(lists[1], 6);
    append_node(lists[2], 7);
    append_node(lists[2], 8);
    append_node(lists[2], 9);
    remove_column(lists, num_rows, 1);
    for (int i = 0; i < num_rows; i++) {
        Node *current = lists[i]->head;
        while (current != NULL) {
            printf("%d ", current->data);
            current = current->next;
        }
        printf("\n");
    }
    for (int i = 0; i < num_rows; i++) {
        free_list(lists[i]);
    }
    free(lists);
    return 0;
}
