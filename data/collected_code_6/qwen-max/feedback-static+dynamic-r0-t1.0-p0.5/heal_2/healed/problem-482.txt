#include <stdio.h>
#include <regex.h>
#include <string.h>
#include <stdlib.h>

void find_sequences(const char *input) {
    regex_t regex;
    if (regcomp(&regex, "[A-Z][a-z]+", REG_EXTENDED | REG_NEWLINE) != 0) {
        fprintf(stderr, "Could not compile regex\n");
        return;
    }

    regmatch_t pmatch[1];
    const size_t nmatch = 1;

    for (const char *ptr = input; *ptr; ptr += pmatch[0].rm_eo ? pmatch[0].rm_eo : 1) {
        int reti = regexec(&regex, ptr, nmatch, pmatch, 0);
        if (reti == 0) {
            printf("Found: %.*s\n", (int)(pmatch[0].rm_eo - pmatch[0].rm_so), &ptr[pmatch[0].rm_so]);
        } else if (reti == REG_NOMATCH) {
            break;
        } else {
            size_t msglen = regerror(reti, &regex, NULL, 0);
            char *msgbuf = (char *)malloc(msglen + 1);
            if (msgbuf == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                regfree(&regex);
                return;
            }
            regerror(reti, &regex, msgbuf, msglen + 1);
            fprintf(stderr, "Regex match failed: %s\n", msgbuf);
            free(msgbuf);
            break;
        }
    }

    regfree(&regex);
}

int main() {
    const char *test_string = "This is a Test String with SomeWords and OtherWords";
    find_sequences(test_string);
    return 0;
}
