#include <string.h>

int count_char_frequency(const char *str, size_t str_max, size_t freq_size, unsigned long freq[])
{
    size_t i;

    if (str == NULL || freq == NULL || freq_size < FREQ_SIZE || str_max == 0) {
        return -1;
    }

    for (i = 0; i < freq_size; i++) {
        freq[i] = 0;
    }

    for (i = 0; i < str_max && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (freq[c] < ULONG_MAX) {
            freq[c]++;
        }
    }

    }

    return 0;
}

        return EXIT_FAILURE;

        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

        fprintf(stderr, "Error counting character frequency\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i <= UCHAR_MAX; i++) {
        if (freq[i] > 0) {
            if (i >= 32 && i < 127) {
                printf("'%c' : %lu\n", (char)i, freq[i]);
            } else {
                printf("0x%02X : %lu\n", (unsigned int)i, freq[i]);

        }
    }

    return EXIT_SUCCESS;

1. **Bounded input**: `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to `BUFFER_SIZE - 1` characters plus the null terminator, so no overflow is possible on input.

2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure ensuring the buffer is null-terminated even if the input source misbehaves.

3. **Bounded length calculation**: `strnlen(buffer, sizeof(buffer))` never scans past the buffer, unlike plain `strlen`.

4. **Bounded processing**: `count_char_frequency` receives `str_max` and stops at either the null terminator or the maximum size, returning `-1` if no terminator was found within bounds. It also validates all pointers, sizes, and guards the frequency counters against `unsigned long` overflow.

5. **Safe indexing**: Characters are cast to `unsigned char` before indexing `freq[]`, preventing negative-index undefined behavior on platforms where `char` is signed, and `FREQ_SIZE` (`UCHAR_MAX + 1`) covers every possible byte value.
