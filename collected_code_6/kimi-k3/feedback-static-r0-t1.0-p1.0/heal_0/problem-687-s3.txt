```c
#include <stdio.h>

int gcd(int a, int b);

int gcd(int a, int b) {
    if (b == 0) {
        return a;
    }
    return gcd(b, a % b);
}

int main() {
    int num1, num2;
    printf("Enter two integers: ");
    if (scanf("%d %d", &num1, &num2) != 2) {
        printf("Invalid input\n");
        return 1;
    }
    printf("GCD: %d\n", gcd(num1, num2));
    return 0;
}
```
