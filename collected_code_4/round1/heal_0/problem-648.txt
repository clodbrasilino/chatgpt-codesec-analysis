```C
#include <stdio.h>

void swapElements(int* arr, int n) {
    int temp;
  
    for(int i = 0; i < n - 1; i = i + 2){
        temp = arr[i];
        arr[i] = arr[i + 1];
        arr[i + 1] = temp;
    }
} 

int main(){
    int arr[] = {1, 2, 3, 4, 5, 6};
    int n = sizeof(arr) / sizeof(*arr);
    
    swapElements(arr, n);

    for(int i=0; i<n; i++){
        printf("%d ", arr[i]);
    }
    
    return 0;
}
```
