```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    const char *input = "hello_world_example_string";
    char *result;

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\n", input) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (printf("Output: %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Key change:** The unbounded `strlen()` call was replaced with `strnlen()` bounded by `MAX_INPUT_LEN`. This addresses the CWE-126 (buffer over-read) concern flagged by Flawfinder: if the function ever receives a non-NUL-terminated buffer, the scan is now capped at a defined maximum rather than reading indefinitely past the buffer. Inputs at or beyond the limit are rejected by returning `NULL`, which the caller already handles. The rest of the logic was already safe: the destination buffer is sized from the measured length, `j` can never exceed `len`, `toupper()` is correctly given an `unsigned char` cast to avoid undefined behavior with negative `char` values, and all allocation and I/O error paths free the buffer before exiting.
