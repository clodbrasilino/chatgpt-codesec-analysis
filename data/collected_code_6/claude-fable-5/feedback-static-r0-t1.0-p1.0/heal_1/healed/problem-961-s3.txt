```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_ROMAN_LEN 64

static int roman_char_value(char c)
{
    switch (toupper((unsigned char)c)) {
    case 'I': return 1;
    case 'V': return 5;
    case 'X': return 10;
    case 'L': return 50;
    case 'C': return 100;
    case 'D': return 500;
    case 'M': return 1000;
    default:  return -1;
    }
}

static int roman_to_int(const char *s, size_t max_len, int *result)
{
    size_t len;
    size_t i;
    int total = 0;

    if (s == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(s, max_len);
    if (len == 0 || len >= max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        int current = roman_char_value(s[i]);
        if (current < 0) {
            return -1;
        }

        if (i + 1 < len) {
            int next = roman_char_value(s[i + 1]);
            if (next < 0) {
                return -1;
            }
            if (current < next) {
                total += next - current;
                i++;
                continue;
            }
        }
        total += current;
    }

    *result = total;
    return 0;
}

int main(void)
{
    char buffer[MAX_ROMAN_LEN];
    int value = 0;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a Roman numeral: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\r\n")] = '\0';

    if (roman_to_int(buffer, sizeof(buffer), &value) != 0) {
        fprintf(stderr, "Invalid Roman numeral: %s\n", buffer);
        return EXIT_FAILURE;
    }

    printf("%s = %d\n", buffer, value);
    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

1. **CWE-126 (over-read via `strlen`)**: Replaced `strlen` with `strnlen` bounded by an explicit `max_len` parameter passed by the caller. This guarantees the scan never reads past the buffer even if the string is not null-terminated. If `strnlen` returns `max_len` (meaning no terminator was found within bounds), the input is rejected.

2. **CWE-119/CWE-120 (statically-sized buffer)**: The buffer size is now defined by a single named constant (`MAX_ROMAN_LEN`) used consistently for allocation and bounds passing. The buffer is zero-initialized with `memset`, and a defensive `buffer[sizeof(buffer) - 1] = '\0'` guarantees null-termination regardless of what `fgets` did. `fgets` already limits input to `sizeof(buffer)`, so no overflow can occur on write.

3. The `strcspn` trim remains safe because the buffer is now guaranteed null-terminated before it runs.

The original code was already fairly defensive (fgets with sizeof, null checks); these changes eliminate the remaining theoretical over-read paths flagged by Flawfinder.
