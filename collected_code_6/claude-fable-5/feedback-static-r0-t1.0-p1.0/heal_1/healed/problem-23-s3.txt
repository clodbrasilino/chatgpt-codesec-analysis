#include <stdio.h>
#include <stddef.h>

int max_sum_of_lists(const int *const *lists, const size_t *lengths,
                     size_t num_lists, long *result)
{
    size_t i;
    long max;
    int found;

    if (lists == NULL || lengths == NULL || result == NULL || num_lists == 0U) {
        return -1;
    }

    max = 0L;
    found = 0;

    for (i = 0U; i < num_lists; i++) {
        size_t j;
        long sum;

        if (lists[i] == NULL) {
            return -1;
        }
        sum = 0L;
        for (j = 0U; j < lengths[i]; j++) {
            sum += (long)lists[i][j];
        }
        if (found == 0 || sum > max) {
            max = sum;
            found = 1;
        }
    }

    *result = max;
    return 0;
}

int main(void)
{
    int list1[] = { 1, 2, 3 };
    int list2[] = { 4, 5, 6 };
    int list3[] = { 10, 11, 12 };
    const int *lists[3];
    size_t lengths[3];
    long result;
    int status;

    lists[0] = list1;
    lists[1] = list2;
    lists[2] = list3;

    lengths[0] = sizeof(list1) / sizeof(list1[0]);
    lengths[1] = sizeof(list2) / sizeof(list2[0]);
    lengths[2] = sizeof(list3) / sizeof(list3[0]);

    status = max_sum_of_lists(lists, lengths, 3U, &result);
    if (status != 0) {
        (void)fprintf(stderr, "Error: invalid input\n");
        return 1;
    }

    if (printf("Maximum sum: %ld\n", result) < 0) {
        return 1;
    }

    return 0;
}
