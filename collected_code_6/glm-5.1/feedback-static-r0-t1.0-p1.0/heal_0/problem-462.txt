#include <stdio.h>
#include <stdlib.h>

void find_combinations(int *arr, int n, int start, int *current, int current_size) {
    if (current_size > 0) {
        printf("{");
        for (int i = 0; i < current_size; i++) {
            printf("%d", current[i]);
            if (i < current_size - 1) {
                printf(", ");
            }
        }
        printf("}\n");
    }

    for (int i = start; i < n; i++) {
        current[current_size] = arr[i];
        find_combinations(arr, n, i + 1, current, current_size + 1);
    }
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <integer1> <integer2> ...\n", argv[0]);
        return EXIT_FAILURE;
    }

    int n = argc - 1;
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (int i = 0; i < n; i++) {
        char *endptr;
        long val = strtol(argv[i + 1], &endptr, 10);
        if (*endptr != '\0' || val < 0 || val > 2147483647) {
            fprintf(stderr, "Invalid integer argument: %s\n", argv[i + 1]);
            free(arr);
            return EXIT_FAILURE;
        }
        arr[i] = (int)val;
    }

    int *current = (int *)malloc(n * sizeof(int));
    if (current == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(arr);
        return EXIT_FAILURE;
    }

    find_combinations(arr, n, 0, current, 0);

    free(arr);
    free(current);
    return EXIT_SUCCESS;
}
