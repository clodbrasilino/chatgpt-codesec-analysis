```C
#include <stdio.h>

// Function to find gcd
int gcd(int a, int b){
    if (b != 0)
        return gcd(b, a % b);
    else 
        return a;
}

int main(){
    int num1, num2, result;
    
    printf("Enter two positive integers: ");
    scanf("%d %d", &num1, &num2);
    
    result = gcd(num1, num2);
    
    printf("GCD of %d and %d is %d.", num1, num2, result);
    
    return 0;
}
```
