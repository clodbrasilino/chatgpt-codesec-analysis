```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

static bool is_undulating(long long num)
{
    unsigned long long value;

    if (num < 0) {
        value = (unsigned long long)(-(num + 1)) + 1ULL;
    } else {
        value = (unsigned long long)num;
    }

    if (value < 100ULL) {
        return false;
    }

    int first = (int)(value % 10ULL);
    value /= 10ULL;
    int second = (int)(value % 10ULL);
    value /= 10ULL;

    if (first == second) {
        return false;
    }

    bool expect_first = true;
    while (value > 0ULL) {
        int digit = (int)(value % 10ULL);
        if (expect_first) {
            if (digit != first) {
                return false;
            }
        } else {
            if (digit != second) {
                return false;
            }
        }
        expect_first = !expect_first;
        value /= 10ULL;
    }

    return true;
}

int main(void)
{
    long long tests[] = {121212LL, 1212121LL, 1991LL, 101LL, 12LL, 3434343LL, 1234LL};
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (size_t i = 0; i < count; i++) {
        if (printf("%lld is %sundulating\n", tests[i],
                   is_undulating(tests[i]) ? "" : "not ") < 0) {
            return EXIT_FAILURE;
        }
    }

    long long input = 0;
    if (printf("Enter a number: ") < 0) {
        return EXIT_FAILURE;
    }
    if (scanf("%lld", &input) != 1) {
        if (fprintf(stderr, "Invalid input\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("%lld is %sundulating\n", input,
               is_undulating(input) ? "" : "not ") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The key fix addresses the UBSan-reported undefined behavior: negating `LLONG_MIN` (`-9223372036854775808`) overflows because its positive counterpart is not representable in `long long`. The rewrite converts the value to `unsigned long long` using the expression `(unsigned long long)(-(num + 1)) + 1ULL`, which safely computes the absolute value for all inputs including `LLONG_MIN`, eliminating the AFL-discovered crash while preserving the original program's behavior for all other inputs.
