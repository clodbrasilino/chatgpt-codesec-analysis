```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *tests[] = {
        "python(java)",
        "string(remove)",
        "alpha(beta(gamma)delta)",
        "Hello (remove me) World",
        "abc(def(ghi)jkl)mno",
        "no parentheses here"
    };
    const char *expected[] = {
        "python",
        "string",
        "alpha",
        "Hello  World",
        "abcmno",
        "no parentheses here"
    };
    size_t num_tests;
    size_t i;
    int status;

    status = EXIT_SUCCESS;
    num_tests = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < num_tests; i++) {
        char *output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (strcmp(output, expected[i]) != 0) {
            fprintf(stderr, "test case %zu failed: expected '%s', got '%s'\n",
                    i, expected[i], output);
            status = EXIT_FAILURE;
        }
        if (printf("%s\n", output) < 0) {
            free(output);
            return EXIT_FAILURE;
        }
        free(output);
        output = NULL;
    }

    return status;
}
```

**Analysis of the weaknesses and how this version avoids them:**

The garbage output in the failing test cases (`got '��'`, `got '���'`) is the classic signature of reading **freed or uninitialized heap memory**. The likely causes in a flawed version of this code would be:

1. **Use-after-free** — printing or comparing `output` after calling `free(output)`. In this version, `strcmp` and `printf` both occur strictly *before* `free(output)`, and the pointer is set to `NULL` afterward to prevent accidental reuse.

2. **Missing null terminator / uninitialized buffer** — if the buffer were allocated with `malloc` and the terminator never written, `strcmp` would read past valid data into garbage. Here, `calloc` zero-initializes the entire buffer *and* `result[j] = '\0'` explicitly terminates the string, providing defense in depth.

3. **Returning a pointer to a stack buffer** — a common variant where `result` is a local array (`char result[len+1]`) returned from the function, leaving the caller with a dangling pointer. This version correctly heap-allocates the result, transferring ownership to the caller, which frees it exactly once.

4. **Buffer overflow** — `j` can never exceed `i < len`, so writes stay within the `len + 1` allocation, including the terminator.

Additionally, all allocation and I/O failures (`calloc`, `printf`) are checked, `NULL` input is handled defensively, and the nested-parentheses logic uses depth tracking with a guard against underflow from unbalanced `)` characters.
