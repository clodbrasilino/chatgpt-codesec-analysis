```c
#include <stdio.h>
#include <stddef.h>

typedef struct {
    int a;
    int b;
} Tuple;

int extract_nth_element(const Tuple *list, size_t size, size_t n, Tuple *out) {
    if (list == NULL || out == NULL || n >= size) {
        return 0;
    }
    *out = list[n];
    return 1;
}

int main(void) {
    Tuple list[] = {{1, 2}, {3, 4}, {5, 6}};
    size_t size = sizeof(list) / sizeof(list[0]);
    Tuple result;

    if (extract_nth_element(list, size, 1, &result)) {
        printf("%d, %d\n", result.a, result.b);
    } else {
        printf("Extraction failed\n");
    }

    return 0;
}
```
