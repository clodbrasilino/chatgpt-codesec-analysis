#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    char data;
    struct Node *next;
} Node;

Node *string_to_list(const char *str);
void print_list(const Node *head);
void free_list(Node *head);

int main(int argc, char *argv[])
{
    const char *input = (argc > 1) ? argv[1] : "Hello, World!";
    Node *list = string_to_list(input);

    if (list == NULL && input[0] != '\0') {
        fprintf(stderr, "Error: failed to convert string to list\n");
        return EXIT_FAILURE;
    }

    print_list(list);
    free_list(list);

    return EXIT_SUCCESS;
}

Node *string_to_list(const char *str)
{
    Node *head = NULL;
    Node *tail = NULL;

    if (str == NULL) {
        return NULL;
    }

    while (*str != '\0') {
        Node *new_node = malloc(sizeof(Node));
        if (new_node == NULL) {
            free_list(head);
            return NULL;
        }
        new_node->data = *str;
        new_node->next = NULL;

        if (head == NULL) {
            head = new_node;
        } else {
            tail->next = new_node;
        }
        tail = new_node;
        str++;
    }

    return head;
}

void print_list(const Node *head)
{
    const Node *current = head;

    while (current != NULL) {
        printf("'%c'", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

void free_list(Node *head)
{
    while (head != NULL) {
        Node *temp = head;
        head = head->next;
        free(temp);
    }
}
