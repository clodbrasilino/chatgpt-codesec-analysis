#include <stdio.h>
#include <stdint.h>

int get_rightmost_set_bit_position(uint32_t num) {
    int position = 0;
    while (((num & 1) == 0) && (position < 32)) {
        num >>= 1;
        position++;
    }
    if (position == 32) {
        return -1; // no set bit found
    }
    return position;
}

int main() {
    uint32_t num = 18; // example number, binary: 10010
    int position = get_rightmost_set_bit_position(num);
    if (position != -1) {
        printf("The position of the rightmost set bit is: %d\n", position);
    } else {
        printf("No set bits found.\n");
    }
    return 0;
}
