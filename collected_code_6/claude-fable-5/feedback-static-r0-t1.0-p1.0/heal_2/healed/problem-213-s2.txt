```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

typedef struct {
    char *first;
    char *second;
} StringTuple;

static char *concat_strings(const char *a, const char *b)
{
    size_t len_a;
    size_t len_b;
    size_t total_len;
    char *result;

    if (a == NULL || b == NULL) {
        return NULL;
    }

    len_a = strnlen(a, MAX_STRING_LEN);
    len_b = strnlen(b, MAX_STRING_LEN);

    if (len_a >= MAX_STRING_LEN || len_b >= MAX_STRING_LEN) {
        return NULL;
    }

    if (len_a > SIZE_MAX - len_b - 1U) {
        return NULL;
    }

    total_len = len_a + len_b + 1U;

    result = malloc(total_len);
    if (result == NULL) {
        return NULL;
    }

    if (len_a > 0U) {
        memcpy(result, a, len_a);
    }
    if (len_b > 0U) {
        memcpy(result + len_a, b, len_b);
    }
    result[len_a + len_b] = '\0';

    return result;
}

static int concat_tuples(const StringTuple *t1, const StringTuple *t2, StringTuple *out)
{
    if (t1 == NULL || t2 == NULL || out == NULL) {
        return -1;
    }

    out->first = concat_strings(t1->first, t2->first);
    if (out->first == NULL) {
        return -1;
    }

    out->second = concat_strings(t1->second, t2->second);
    if (out->second == NULL) {
        free(out->first);
        out->first = NULL;
        return -1;
    }

    return 0;
}

static void free_tuple(StringTuple *t)
{
    if (t != NULL) {
        free(t->first);
        free(t->second);
        t->first = NULL;
        t->second = NULL;
    }
}

int main(void)
{
    StringTuple t1;
    StringTuple t2;
    StringTuple result;

    t1.first = "Hello, ";
    t1.second = "Good ";
    t2.first = "World";
    t2.second = "Morning";

    result.first = NULL;
    result.second = NULL;

    if (concat_tuples(&t1, &t2, &result) != 0) {
        fprintf(stderr, "Tuple concatenation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("(%s, %s)\n", result.first, result.second) < 0) {
        free_tuple(&result);
        return EXIT_FAILURE;
    }

    free_tuple(&result);

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses:**

The Flawfinder warnings on both `memcpy` calls (CWE-120) are effectively **false positives** in the original code, but here is why the code is safe and how the rewrite hardens it further:

1. **Destination size is guaranteed sufficient.** The buffer is allocated with `malloc(len_a + len_b + 1U)`, and the two `memcpy` calls copy exactly `len_a` and `len_b` bytes into non-overlapping regions, with the final byte reserved for the NUL terminator. The destination can always hold the source data.

2. **Bounded length measurement.** `strnlen` with `MAX_STRING_LEN` prevents unbounded reads on non-terminated input, and the `>= MAX_STRING_LEN` rejection guarantees both lengths are strictly bounded.

3. **Integer overflow protection.** The `len_a > SIZE_MAX - len_b - 1U` check prevents `size_t` wraparound in the allocation size — this is the classic CWE-190 → CWE-120 chain, and it was already correctly guarded. It is technically redundant given `MAX_STRING_LEN` bounds (max total is well below `SIZE_MAX`), but defense-in-depth is good practice.

4. **Hardening added in the rewrite:** guarding `memcpy` with `len > 0U` checks. While `memcpy(dst, src, 0)` is defined behavior when pointers are valid, skipping zero-length copies satisfies stricter static analyzers and avoids pedantic UB concerns with pointer arithmetic edge cases.

5. **Memory management is correct:** partial-failure cleanup in `concat_tuples`, idempotent `free_tuple` via NULL assignment, and cleanup on the `printf` failure path — no leaks or double-frees.
