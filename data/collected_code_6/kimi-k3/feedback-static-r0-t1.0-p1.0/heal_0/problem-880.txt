#include <stdio.h>
#include <math.h>

int find_quadratic_solutions(double a, double b, double c)
{
    if (fabs(a) < 1e-12)
    {
        return -1;
    }

    double discriminant = b * b - 4.0 * a * c;

    if (discriminant > 0.0)
    {
        return 2;
    }
    else if (fabs(discriminant) < 1e-12)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

int main(void)
{
    double a = 0.0;
    double b = 0.0;
    double c = 0.0;
    int result = 0;

    if (printf("Enter coefficients a, b, c: ") < 0)
    {
        return 1;
    }

    if (scanf("%lf %lf %lf", &a, &b, &c) != 3)
    {
        (void)fprintf(stderr, "Invalid input\n");
        return 1;
    }

    result = find_quadratic_solutions(a, b, c);

    if (result == -1)
    {
        if (printf("Not a quadratic equation\n") < 0)
        {
            return 1;
        }
    }
    else
    {
        if (printf("Number of solutions: %d\n", result) < 0)
        {
            return 1;
        }
    }

    return 0;
}
