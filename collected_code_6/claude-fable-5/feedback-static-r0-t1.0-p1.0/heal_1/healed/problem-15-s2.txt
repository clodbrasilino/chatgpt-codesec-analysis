```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char **split_at_lowercase(const char *input, size_t *count)
{
    char **result = NULL;
    char **temp = NULL;
    size_t capacity = 8;
    size_t num_tokens = 0;
    size_t i = 0;
    size_t start = 0;
    size_t len;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && input[MAX_INPUT_LEN - 1] != '\0') {
        return NULL;
    }

    result = malloc(capacity * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    while (i <= len) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t token_len = i - start;
                char *token = malloc(token_len + 1);
                if (token == NULL) {
                    for (size_t j = 0; j < num_tokens; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }
                memcpy(token, &input[start], token_len);
                token[token_len] = '\0';

                if (num_tokens == capacity) {
                    if (capacity > SIZE_MAX / (2 * sizeof(char *))) {
                        free(token);
                        for (size_t j = 0; j < num_tokens; j++) {
                            free(result[j]);
                        }
                        free(result);
                        return NULL;
                    }
                    capacity *= 2;
                    temp = realloc(result, capacity * sizeof(char *));
                    if (temp == NULL) {
                        free(token);
                        for (size_t j = 0; j < num_tokens; j++) {
                            free(result[j]);
                        }
                        free(result);
                        return NULL;
                    }
                    result = temp;
                }
                result[num_tokens] = token;
                num_tokens++;
            }
            start = i + 1;
        }
        i++;
    }

    *count = num_tokens;
    return result;
}

void free_tokens(char **tokens, size_t count)
{
    if (tokens == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "HELLOworldFOObarBAZ123";
    size_t count = 0;
    char **tokens = split_at_lowercase(test, &count);

    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens: %zu\n", count);
    for (size_t i = 0; i < count; i++) {
        printf("  [%zu]: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

```c
#include <stdint.h>
```

**Summary of the fixes applied:**

1. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, and rejected inputs that are not null-terminated within that bound. This prevents an unbounded read on a non-terminated string.

2. **CWE-120 (memcpy overflow):** The destination buffer is allocated with `malloc(token_len + 1)` immediately before the `memcpy` of exactly `token_len` bytes, so the destination is guaranteed to hold the source data; the null terminator is written within bounds.

3. **Additional hardening (CWE-190):** Added an overflow check before doubling `capacity` in the `realloc` growth path, preventing a potential integer overflow in `capacity * sizeof(char *)` that could lead to an undersized allocation.
