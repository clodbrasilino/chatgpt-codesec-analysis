#include <stdio.h>
#include <stdlib.h>

int remove_negatives(int *arr, int size) {
    if (arr == NULL || size <= 0) {
        return 0;
    }
    
    int new_size = 0;
    for (int i = 0; i < size; i++) {
        if (arr[i] >= 0) {
            arr[new_size] = arr[i];
            new_size++;
        }
    }
    return new_size;
}

int main(void) {
    int size = 8;
    size_t alloc_size = (size_t)size * sizeof(int);
    
    if (size <= 0 || alloc_size / sizeof(int) != (size_t)size) {
        return EXIT_FAILURE;
    }
    
    int *arr = malloc(alloc_size);
    if (arr == NULL) {
        return EXIT_FAILURE;
    }
    
    arr[0] = 1;
    arr[1] = -1;
    arr[2] = 2;
    arr[3] = -2;
    arr[4] = 3;
    arr[5] = -3;
    arr[6] = 0;
    arr[7] = -4;
    
    int new_size = remove_negatives(arr, size);
    
    for (int i = 0; i < new_size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    free(arr);
    return EXIT_SUCCESS;
}
