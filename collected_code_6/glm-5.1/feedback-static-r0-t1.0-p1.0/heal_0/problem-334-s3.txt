#include <stdio.h>
#include <stdbool.h>

bool is_valid_triangle(int a, int b, int c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return false;
    }
    if ((long)a + (long)b <= c) {
        return false;
    }
    if ((long)a + (long)c <= b) {
        return false;
    }
    if ((long)b + (long)c <= a) {
        return false;
    }
    return true;
}

int main() {
    int a, b, c;
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        return 1;
    }
    if (is_valid_triangle(a, b, c)) {
        printf("Valid\n");
    } else {
        printf("Invalid\n");
    }
    return 0;
}
