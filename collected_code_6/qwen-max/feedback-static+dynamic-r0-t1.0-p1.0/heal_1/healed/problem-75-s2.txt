```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    int size;
} Tuple;

int isDivisibleByK(Tuple tuple, int k) {
    for (int i = 0; i < tuple.size; ++i) {
        if (tuple.elements[i] % k != 0) {
            return 0;
        }
    }
    return 1;
}

int main() {
    int elements1[] = {2, 4, 6, 8};
    int elements2[] = {9, 15, 21};
    int elements3[] = {3, 7, 10};
    int elements4[] = {12, 18, 24, 30};

    Tuple tuples[] = {
        {elements1, 4},
        {elements2, 3},
        {elements3, 3},
        {elements4, 4}
    };

    int numTuples = sizeof(tuples) / sizeof(tuples[0]);
    int k = 3;

    for (int i = 0; i < numTuples; ++i) {
        if (isDivisibleByK(tuples[i], k)) {
            for (int j = 0; j < tuples[i].size; ++j) {
                printf("%d ", tuples[i].elements[j]);
            }
            printf("\n");
        }
    }

    return 0;
}
```
