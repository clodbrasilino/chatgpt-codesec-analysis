#include <stdio.h>
#include <stdlib.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static long long sum_of_absolute_differences(const int *arr, size_t n)
{
    long long total;
    long long prefix;
    int *sorted;
    size_t i;

    if (arr == NULL || n < 2) {
        return 0;
    }

    sorted = malloc(n * sizeof(*sorted));
    if (sorted == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (i = 0; i < n; i++) {
        sorted[i] = arr[i];
    }

    qsort(sorted, n, sizeof(*sorted), compare_ints);

    total = 0;
    prefix = 0;
    for (i = 0; i < n; i++) {
        total += (long long)sorted[i] * (long long)i - prefix;
        prefix += (long long)sorted[i];
    }

    free(sorted);
    return total;
}

int main(void)
{
    int arr[] = {1, 8, 9, 15, 16};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result;

    result = sum_of_absolute_differences(arr, n);

    if (printf("Sum of absolute differences: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
