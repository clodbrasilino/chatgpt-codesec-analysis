#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 100
#define MAX_WORD_LEN 100

int is_duplicate(char words[][MAX_WORD_LEN], int count, const char *word) {
    for (int i = 0; i < count; i++) {
        if (strcmp(words[i], word) == 0) {
            return 1;
        }
    }
    return 0;
}

int remove_duplicates(char *input[], int input_count, char output[][MAX_WORD_LEN]) {
    int output_count = 0;
    
    for (int i = 0; i < input_count; i++) {
        if (input[i] == NULL) {
            continue;
        }
        
        if (!is_duplicate(output, output_count, input[i])) {
            strncpy(output[output_count], input[i], MAX_WORD_LEN - 1);
            output[output_count][MAX_WORD_LEN - 1] = '\0';
            output_count++;
        }
    }
    
    return output_count;
}

int main(void) {
    char *words[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    int input_count = sizeof(words) / sizeof(words[0]);
    char unique_words[MAX_WORDS][MAX_WORD_LEN];
    
    int unique_count = remove_duplicates(words, input_count, unique_words);
    
    printf("Unique words:\n");
    for (int i = 0; i < unique_count; i++) {
        printf("%s\n", unique_words[i]);
    }
    
    return 0;
}
