#include <stdio.h>
#include <stdbool.h>

bool sum_of_two_squares(int n) {
    if (n < 0) {
        return false;
    }
    
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            int count = 0;
            while (n % i == 0) {
                count++;
                n /= i;
            }
            if (i % 4 == 3 && count % 2 != 0) {
                return false;
            }
        }
    }
    return n % 4 != 3;
}

int main(void) {
    int num;
    
    if (scanf("%d", &num) == 1) {
        if (sum_of_two_squares(num)) {
            printf("Yes\n");
        } else {
            printf("No\n");
        }
    } else {
        printf("Invalid input\n");
    }
    
    return 0;
}
