```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

static void report_regex_error(int code, const regex_t *regex, const char *context);
char *remove_lowercase(const char *input);

static void report_regex_error(int code, const regex_t *regex, const char *context)
{
    size_t needed;
    char *errbuf;

    needed = regerror(code, regex, NULL, 0);
    if (needed == 0) {
        fprintf(stderr, "%s: unknown regex error\n", context);
        return;
    }

    errbuf = calloc(needed, sizeof(char));
    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate error buffer\n", context);
        return;
    }

    regerror(code, regex, errbuf, needed);
    errbuf[needed - 1] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
    free(errbuf);
}

char *remove_lowercase(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len;
    size_t input_len;
    size_t capacity;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not null-terminated\n");
        return NULL;
    }

    ret = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (ret != 0) {
        report_regex_error(ret, &regex, "Regex compilation failed");
        return NULL;
    }

    capacity = input_len + 1;
    result = calloc(capacity, sizeof(char));
    if (result == NULL) {
        regfree(&regex);
        fprintf(stderr, "Memory allocation failed\n");
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, capacity - 1 - result_len);
            if (remaining > capacity - 1 - result_len) {
                fprintf(stderr, "Buffer bounds exceeded\n");
                free(result);
                regfree(&regex);
                return NULL;
            }
            memmove(result + result_len, cursor, remaining);
            result_len += remaining;
            break;
        } else if (ret != 0) {
            report_regex_error(ret, &regex, "Regex execution failed");
            free(result);
            regfree(&regex);
            return NULL;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so ||
            (size_t)match.rm_so > capacity - 1 - result_len) {
            fprintf(stderr, "Buffer bounds exceeded\n");
            free(result);
            regfree(&regex);
            return NULL;
        }

        memmove(result + result_len, cursor, (size_t)match.rm_so);
        result_len += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    if (result_len >= capacity) {
        fprintf(stderr, "Buffer bounds exceeded\n");
        free(result);
        regfree(&regex);
        return NULL;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "HelloWorldFOO123bar",
        "KDeoALOklOOHserfLoAJSIskdsf",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix3d C4s3 Str1ng"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_lowercase(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: %s\n", test_strings[i]);
        printf("Cleaned:  %s\n\n", cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the flagged weakness (CWE-119/CWE-120) in `report_regex_error`: instead of a statically-sized `char errbuf[256]`, the buffer is now dynamically allocated by first calling `regerror` with a `NULL` buffer to query the exact required size. This guarantees the buffer can never be too small for the error message, eliminating any risk of truncation-related logic issues or improper bounds assumptions, while still explicitly null-terminating and freeing the allocation. The rest of the code already performed rigorous bounds checking (`strnlen` with a hard cap, validated `regmatch_t` offsets, capacity checks before every write), so it is preserved as-is.
