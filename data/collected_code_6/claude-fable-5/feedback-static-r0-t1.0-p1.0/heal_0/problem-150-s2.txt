#include <stdio.h>
#include <stdlib.h>

int is_present(long long a, long long b, long long c)
{
    long long diff;

    if (c == 0) {
        return (a == b) ? 1 : 0;
    }

    diff = b - a;

    if (c > 0) {
        if (diff < 0) {
            return 0;
        }
    } else {
        if (diff > 0) {
            return 0;
        }
    }

    return (diff % c == 0) ? 1 : 0;
}

int main(void)
{
    long long a;
    long long b;
    long long c;

    printf("Enter first term (a): ");
    if (scanf("%lld", &a) != 1) {
        fprintf(stderr, "Invalid input for a\n");
        return EXIT_FAILURE;
    }

    printf("Enter number to search (b): ");
    if (scanf("%lld", &b) != 1) {
        fprintf(stderr, "Invalid input for b\n");
        return EXIT_FAILURE;
    }

    printf("Enter common difference (c): ");
    if (scanf("%lld", &c) != 1) {
        fprintf(stderr, "Invalid input for c\n");
        return EXIT_FAILURE;
    }

    if (is_present(a, b, c)) {
        printf("%lld is present in the sequence\n", b);
    } else {
        printf("%lld is not present in the sequence\n", b);
    }

    return EXIT_SUCCESS;
}

This solves the classic problem of checking whether a number `b` appears in the infinite arithmetic sequence starting at `a` with common difference `c` (i.e., `a, a+c, a+2c, ...`).
