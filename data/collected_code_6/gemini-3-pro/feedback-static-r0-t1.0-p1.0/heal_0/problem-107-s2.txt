#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int is_hexadecimal(const char *str) {
    if (str == NULL || *str == '\0') {
        return 0;
    }

    size_t i = 0;
    if (str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        i = 2;
        if (str[i] == '\0') {
            return 0;
        }
    }

    for (; str[i] != '\0'; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

int count_hexadecimal_numbers(const char **strings, size_t count) {
    if (strings == NULL) {
        return 0;
    }

    int hex_count = 0;
    for (size_t i = 0; i < count; i++) {
        if (is_hexadecimal(strings[i])) {
            hex_count++;
        }
    }

    return hex_count;
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        return EXIT_FAILURE;
    }

    int hex_count = count_hexadecimal_numbers((const char **)(argv + 1), argc - 1);
    printf("%d\n", hex_count);

    return EXIT_SUCCESS;
}
