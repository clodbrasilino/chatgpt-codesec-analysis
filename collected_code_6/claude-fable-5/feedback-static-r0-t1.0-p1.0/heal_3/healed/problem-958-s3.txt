```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    buffer[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = strnlen(symbols[i], sizeof(symbols[i]));

            if (used >= buffer_size || sym_len >= buffer_size - used) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            number -= values[i];
        }
    }

    if (used >= buffer_size) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[used] = '\0';
    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            if (printf("%d -> %s\n", test_values[i], roman) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, roman, sizeof(roman)) != 0) {
        if (printf("0 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, roman, sizeof(roman)) != 0) {
        if (printf("4000 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

- **`memcpy` (CWE-120):** The original code was already safe — the guard `used >= buffer_size || sym_len >= buffer_size - used` guarantees the destination can hold `sym_len` bytes plus the eventual terminator, and the subtraction cannot underflow because `used < buffer_size` is checked first. In the rewrite, the hard-coded `symbol_lengths` table was replaced with a bounded `strnlen` on the symbol itself, eliminating the risk of the length table drifting out of sync with the symbols array (which would have been the only realistic way to trigger an overflow).

- **Statically-sized array `roman` (CWE-119/CWE-120):** This is a low-risk informational warning. The buffer is zero-initialized with `memset`, its size is always passed via `sizeof(roman)` rather than a repeated constant, and `int_to_roman` performs strict bounds checking and guarantees NUL termination on every path (including error paths). `ROMAN_BUFFER_SIZE` of 32 comfortably exceeds the longest possible output, `MMMDCCCLXXXVIII` (3888, 15 characters plus terminator).

- **Other correctness points verified:** all `printf`/`fprintf` return values are checked, out-of-range inputs (0 and 4000) are rejected before any writes beyond clearing `buffer[0]`, and NULL/zero-size buffers are handled defensively.
