#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_digits(const char *str) {
    if (str == NULL) {
        return NULL;
    }
    
    size_t len = strlen(str);
    char *result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }
    
    size_t j = 0;
    for (size_t i = 0; i < len; i++) {
        if (!isdigit((unsigned char)str[i])) {
            result[j++] = str[i];
        }
    }
    result[j] = '\0';
    
    char *shrunk = realloc(result, j + 1);
    if (shrunk != NULL) {
        result = shrunk;
    }
    
    return result;
}

int remove_all_digits(char **list, size_t size) {
    if (list == NULL) {
        return -1;
    }
    
    for (size_t i = 0; i < size; i++) {
        if (list[i] == NULL) {
            continue;
        }
        
        char *filtered = remove_digits(list[i]);
        if (filtered == NULL) {
            return -1;
        }
        
        free(list[i]);
        list[i] = filtered;
    }
    
    return 0;
}

int main(void) {
    size_t count = 4;
    char **list = malloc(count * sizeof(char *));
    if (list == NULL) {
        return EXIT_FAILURE;
    }
    
    const size_t buf_size = 32;
    int allocation_failed = 0;
    
    for (size_t i = 0; i < count; i++) {
        list[i] = malloc(buf_size);
        if (list[i] == NULL) {
            allocation_failed = 1;
            for (size_t j = 0; j < i; j++) {
                free(list[j]);
            }
            break;
        }
    }
    
    if (allocation_failed) {
        free(list);
        return EXIT_FAILURE;
    }
    
    snprintf(list[0], buf_size, "%s", "h3ll0");
    snprintf(list[1], buf_size, "%s", "w0rld123");
    snprintf(list[2], buf_size, "%s", "c0d1ng");
    snprintf(list[3], buf_size, "%s", "t3st456");
    
    if (remove_all_digits(list, count) != 0) {
        for (size_t i = 0; i < count; i++) {
            free(list[i]);
        }
        free(list);
        return EXIT_FAILURE;
    }
    
    for (size_t i = 0; i < count; i++) {
        printf("%s\n", list[i]);
    }
    
    for (size_t i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
    
    return EXIT_SUCCESS;
}
