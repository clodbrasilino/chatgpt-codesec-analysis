```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t capacity = 8;
    size_t n = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && input[MAX_INPUT_LEN - 1] != '\0') {
        return NULL;
    }

    if (capacity > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;
                size_t alloc_len;
                char *segment;

                if (start > len || seg_len > len - start || seg_len > SIZE_MAX - 1) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                alloc_len = seg_len + 1;
                segment = malloc(alloc_len);
                if (segment == NULL) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                memcpy(segment, input + start, seg_len);
                segment[seg_len] = '\0';

                if (n == capacity) {
                    if (capacity > SIZE_MAX / (2 * sizeof(char *))) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    capacity *= 2;
                    temp = realloc(tokens, capacity * sizeof(char *));
                    if (temp == NULL) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    tokens = temp;
                }
                tokens[n] = segment;
                n++;
            }
            start = i + 1;
        }
    }

    *count = n;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *text = "HELLOworldFOObarBAZ123";
    char **parts = NULL;
    size_t count = 0;
    size_t i;

    parts = split_at_lowercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);
    printf("Found %zu segment(s):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, parts[i]);
    }

    free_tokens(parts, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **Removed the redundant `seg_len >= SIZE_MAX` check** — it was folded into a single combined validation (`seg_len > SIZE_MAX - 1`) that guards the `seg_len + 1` addition against overflow *before* computing `alloc_len`, which is the correct place for that check.

2. **Removed the always-false `seg_len >= alloc_len` check** — since `alloc_len = seg_len + 1` is computed only after the overflow guard passes, the condition could never be true (as cppcheck's `knownConditionTrueFalse` correctly reported). Dead defensive code like this adds noise and can mask real issues.

3. **The `memcpy` (Flawfinder CWE-120 finding)** is retained but is now provably safe: the destination buffer is allocated with exactly `seg_len + 1` bytes, the copy length is `seg_len`, and the overflow guard guarantees `seg_len + 1` did not wrap. The source read is bounded by the `start > len || seg_len > len - start` validation against the `strnlen`-bounded length. This is a false positive once the surrounding invariants are established.

4. **Everything else preserved**: NULL-parameter checks, `strnlen` bounding against unterminated input, multiplication-overflow guards before `malloc`/`realloc`, the `realloc` temp-pointer pattern avoiding leaks on failure, and consistent cleanup of both `segment` and `tokens` on every error path.
