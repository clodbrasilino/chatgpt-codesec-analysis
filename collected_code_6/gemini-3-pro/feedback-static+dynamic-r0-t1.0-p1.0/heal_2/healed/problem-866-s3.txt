#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

int is_match(const char *input, const char *target) {
    if (input == NULL || target == NULL) {
        return 0;
    }
    while (*input != '\0' && *target != '\0') {
        if (tolower((unsigned char)*input) != *target) {
            return 0;
        }
        input++;
        target++;
    }
    return (*input == '\0' && *target == '\0');
}

int has_31_days(const char *month) {
    if (month == NULL) {
        return 0; 
    }

    if (is_match(month, "january") || is_match(month, "jan") ||
        is_match(month, "march") || is_match(month, "mar") ||
        is_match(month, "may") ||
        is_match(month, "july") || is_match(month, "jul") ||
        is_match(month, "august") || is_match(month, "aug") ||
        is_match(month, "october") || is_match(month, "oct") ||
        is_match(month, "december") || is_match(month, "dec")) {
        return 1;
    }

    return 0;
}

int main(void) {
    size_t buf_size = 256;
    char *input = (char *)malloc(buf_size);
    
    if (input == NULL) {
        return 1;
    }
    
    if (fgets(input, (int)buf_size, stdin) != NULL) {
        input[strcspn(input, "\n")] = '\0';

        if (has_31_days(input)) {
            printf("1\n");
        } else {
            printf("0\n");
        }
    }

    free(input);
    return 0;
}
