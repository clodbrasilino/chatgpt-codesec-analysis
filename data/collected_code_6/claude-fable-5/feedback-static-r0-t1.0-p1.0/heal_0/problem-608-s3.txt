#include <stdio.h>
#include <stdlib.h>

unsigned long long bell_number(int n, int *error)
{
    unsigned long long **triangle;
    unsigned long long result;
    int i;
    int j;

    if (error == NULL) {
        return 0ULL;
    }

    *error = 0;

    if (n < 0 || n > 25) {
        *error = 1;
        return 0ULL;
    }

    triangle = malloc((size_t)(n + 1) * sizeof(unsigned long long *));
    if (triangle == NULL) {
        *error = 2;
        return 0ULL;
    }

    for (i = 0; i <= n; i++) {
        triangle[i] = malloc((size_t)(i + 1) * sizeof(unsigned long long));
        if (triangle[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(triangle[j]);
            }
            free(triangle);
            *error = 2;
            return 0ULL;
        }
    }

    triangle[0][0] = 1ULL;

    for (i = 1; i <= n; i++) {
        triangle[i][0] = triangle[i - 1][i - 1];
        for (j = 1; j <= i; j++) {
            triangle[i][j] = triangle[i - 1][j - 1] + triangle[i][j - 1];
        }
    }

    result = triangle[n][0];

    for (i = 0; i <= n; i++) {
        free(triangle[i]);
    }
    free(triangle);

    return result;
}

int main(void)
{
    int n;
    int error;
    unsigned long long result;

    printf("Enter n (0 to 25): ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = bell_number(n, &error);

    if (error == 1) {
        fprintf(stderr, "n out of supported range\n");
        return EXIT_FAILURE;
    }

    if (error == 2) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Bell(%d) = %llu\n", n, result);

    return EXIT_SUCCESS;
}
