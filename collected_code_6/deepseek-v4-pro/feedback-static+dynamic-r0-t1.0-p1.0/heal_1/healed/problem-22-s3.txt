```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_ARRAY_SIZE 1000

int find_first_duplicate(const int arr[], int size, int *duplicate) {
    int *seen = NULL;
    int i;
    
    if (arr == NULL || duplicate == NULL || size <= 0) {
        return 0;
    }
    
    seen = (int *)calloc(MAX_ARRAY_SIZE, sizeof(int));
    if (seen == NULL) {
        return 0;
    }
    
    for (i = 0; i < size; i++) {
        if (arr[i] < 0 || arr[i] >= MAX_ARRAY_SIZE) {
            free(seen);
            return 0;
        }
        
        if (seen[arr[i]] == 1) {
            *duplicate = arr[i];
            free(seen);
            return 1;
        }
        
        seen[arr[i]] = 1;
    }
    
    free(seen);
    return 0;
}

int main(void) {
    int test_array[] = {3, 1, 4, 1, 5, 9, 2, 6, 5};
    int duplicate;
    int result;
    
    result = find_first_duplicate(test_array, sizeof(test_array) / sizeof(test_array[0]), &duplicate);
    
    if (result) {
        printf("First duplicate: %d\n", duplicate);
    } else {
        printf("No duplicate found.\n");
    }
    
    return 0;
}
```
