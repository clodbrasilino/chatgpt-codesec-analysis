```c
#include <stdio.h>
#include <math.h>
#include <stdbool.h>

typedef struct {
    double h;
    double k;
    double p;
} ParabolaParams;

typedef enum {
    PARABOLA_OK = 0,
    PARABOLA_ERROR_INVALID_PARAM,
    PARABOLA_ERROR_ZERO_P
} ParabolaError;

ParabolaError calculate_directrix_vertical(const ParabolaParams *params, double *directrix)
{
    if (params == NULL || directrix == NULL) {
        return PARABOLA_ERROR_INVALID_PARAM;
    }
    
    if (params->p == 0.0) {
        return PARABOLA_ERROR_ZERO_P;
    }
    
    *directrix = params->k - params->p;
    return PARABOLA_OK;
}

ParabolaError calculate_directrix_horizontal(const ParabolaParams *params, double *directrix)
{
    if (params == NULL || directrix == NULL) {
        return PARABOLA_ERROR_INVALID_PARAM;
    }
    
    if (params->p == 0.0) {
        return PARABOLA_ERROR_ZERO_P;
    }
    
    *directrix = params->h - params->p;
    return PARABOLA_OK;
}

int main(void)
{
    ParabolaParams vertical_parabola = {2.0, 3.0, 1.5};
    ParabolaParams horizontal_parabola = {1.0, 2.0, 2.0};
    double directrix = 0.0;
    ParabolaError status;
    
    status = calculate_directrix_vertical(&vertical_parabola, &directrix);
    if (status == PARABOLA_OK) {
        printf("Vertical parabola directrix: y = %.2f\n", directrix);
    } else {
        fprintf(stderr, "Error calculating vertical directrix: %d\n", status);
        return 1;
    }
    
    status = calculate_directrix_horizontal(&horizontal_parabola, &directrix);
    if (status == PARABOLA_OK) {
        printf("Horizontal parabola directrix: x = %.2f\n", directrix);
    } else {
        fprintf(stderr, "Error calculating horizontal directrix: %d\n", status);
        return 1;
    }
    
    return 0;
}
```
