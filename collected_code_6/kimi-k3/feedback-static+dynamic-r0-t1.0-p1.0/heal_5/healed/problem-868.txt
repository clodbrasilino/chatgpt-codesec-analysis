#include <stdio.h>
#include <string.h>

int lengthOfLastWord(const char *s) {
    if (s == NULL) {
        return 0;
    }
    
    size_t len = strnlen(s, 1024);
    if (len == 0) {
        return 0;
    }
    
    size_t i = len;
    int length = 0;

    while (i > 0 && s[i - 1] == ' ') {
        i--;
    }

    while (i > 0 && s[i - 1] != ' ') {
        length++;
        i--;
    }

    return length;
}

int main(void) {
    char buffer[1025];

    if (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        size_t len = strnlen(buffer, sizeof(buffer));
        if (len > 0 && buffer[len - 1] == '\n') {
            buffer[len - 1] = '\0';
        }
        printf("%d\n", lengthOfLastWord(buffer));
    }

    return 0;
}
