```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        int ch;
        do {
            ch = getchar();
        } while (ch != '\n' && ch != EOF);
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the original code and the flagged weaknesses:**

1. **Flawfinder `char input[INPUT_SIZE]` (CWE-119/CWE-120):** This is a low-risk, largely informational warning. The original code already mitigates it correctly — `fgets(input, sizeof(input), stdin)` bounds the read to the buffer size, and `strnlen` is used instead of `strlen` to guarantee bounded scanning. In the rewrite I added a defensive explicit NUL termination (`input[sizeof(input) - 1] = '\0'`) as belt-and-suspenders hardening, though `fgets` already guarantees termination on success.

2. **Flawfinder `getchar` in a loop (CWE-120/CWE-20):** Also a false positive in practice — the loop only discards excess characters and never writes them to a buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed stdin.

3. **Other observations:**
   - `octal_to_decimal` correctly guards against signed overflow with the `(LONG_MAX - digit) / 8` pre-check, avoiding undefined behavior.
   - The `len == max_len` check in the conversion function catches unterminated input within the given bound.
   - The overlong-input path correctly drains stdin before erroring out, preventing leftover data from affecting subsequent reads.
   - I added a `max_len == 0` guard and checked the return of `fflush` for completeness.

The original code was already well-hardened; both flagged findings were tool-level informational hits rather than genuine vulnerabilities.
