#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int a;
    int b;
} Pair;

int cmp_int(const void *a, const void *b) {
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    return (ia > ib) - (ia < ib);
}

Pair* find_k_pairs(int *arr1, int len1, int *arr2, int len2, int k, int *out_len) {
    if (arr1 == NULL || arr2 == NULL || out_len == NULL) {
        return NULL;
    }

    long long total_pairs = (long long)len1 * len2;
    if (len1 <= 0 || len2 <= 0 || k <= 0) {
        *out_len = 0;
        return NULL;
    }

    if ((long long)k > total_pairs) {
        k = (int)total_pairs;
    }

    Pair *result = (Pair *)malloc((size_t)k * sizeof(Pair));
    if (result == NULL) {
        return NULL;
    }

    qsort(arr1, (size_t)len1, sizeof(int), cmp_int);
    qsort(arr2, (size_t)len2, sizeof(int), cmp_int);

    int *indices = (int *)calloc((size_t)len1, sizeof(int));
    if (indices == NULL) {
        free(result);
        return NULL;
    }

    int count = 0;
    while (count < k) {
        int min_val = -1;
        int min_idx = -1;

        for (int i = 0; i < len1; i++) {
            if (indices[i] < len2) {
                int current_sum = arr1[i] + arr2[indices[i]];
                if (min_idx == -1 || current_sum < min_val) {
                    min_val = current_sum;
                    min_idx = i;
                }
            }
        }

        if (min_idx == -1) {
            break;
        }

        result[count].a = arr1[min_idx];
        result[count].b = arr2[indices[min_idx]];
        indices[min_idx]++;
        count++;
    }

    free(indices);
    *out_len = count;
    return result;
}

int main() {
    int arr1[] = {1, 7, 11};
    int arr2[] = {2, 4, 6};
    int k = 3;
    int out_len = 0;

    int len1 = sizeof(arr1) / sizeof(arr1[0]);
    int len2 = sizeof(arr2) / sizeof(arr2[0]);

    Pair *pairs = find_k_pairs(arr1, len1, arr2, len2, k, &out_len);
    if (pairs != NULL) {
        for (int i = 0; i < out_len; i++) {
            printf("(%d, %d)\n", pairs[i].a, pairs[i].b);
        }
        free(pairs);
    } else {
        printf("No pairs found or memory allocation failed.\n");
    }

    return 0;
}
