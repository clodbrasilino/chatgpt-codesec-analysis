#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    int size;
} Tuple;

int *find_similar_elements(Tuple *list1, int size1, Tuple *list2, int size2, int *result_size) {
    if (list1 == NULL || list2 == NULL || result_size == NULL) {
        return NULL;
    }
    
    int max_result_size = 0;
    for (int i = 0; i < size1; i++) {
        max_result_size += list1[i].size;
    }
    
    int *result = (int *)malloc(max_result_size * sizeof(int));
    if (result == NULL) {
        return NULL;
    }
    
    int count = 0;
    
    for (int i = 0; i < size1; i++) {
        for (int j = 0; j < list1[i].size; j++) {
            int current = list1[i].elements[j];
            int found = 0;
            
            for (int k = 0; k < size2; k++) {
                for (int l = 0; l < list2[k].size; l++) {
                    if (list2[k].elements[l] == current) {
                        found = 1;
                        break;
                    }
                }
                if (found) {
                    break;
                }
            }
            
            if (found) {
                int already_exists = 0;
                for (int m = 0; m < count; m++) {
                    if (result[m] == current) {
                        already_exists = 1;
                        break;
                    }
                }
                
                if (!already_exists) {
                    result[count++] = current;
                }
            }
        }
    }
    
    *result_size = count;
    
    if (count == 0) {
        free(result);
        return NULL;
    }
    
    int *final_result = (int *)malloc(count * sizeof(int));
    if (final_result == NULL) {
        free(result);
        return NULL;
    }
    
    for (int i = 0; i < count; i++) {
        final_result[i] = result[i];
    }
    
    free(result);
    
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - i - 1; j++) {
            if (final_result[j] > final_result[j + 1]) {
                int temp = final_result[j];
                final_result[j] = final_result[j + 1];
                final_result[j + 1] = temp;
            }
        }
    }
    
    return final_result;
}

void free_tuple_list(Tuple *list, int size) {
    if (list == NULL) {
        return;
    }
    
    for (int i = 0; i < size; i++) {
        free(list[i].elements);
    }
    free(list);
}
