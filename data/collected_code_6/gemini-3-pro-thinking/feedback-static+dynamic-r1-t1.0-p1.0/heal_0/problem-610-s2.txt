#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* create_node(int data);
Node* remove_kth_element(Node* head, size_t k);
void free_list(Node* head);
void print_list(Node* head);

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        exit(EXIT_FAILURE);
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

Node* remove_kth_element(Node* head, size_t k) {
    if (head == NULL) {
        return NULL;
    }

    if (k == 0) {
        Node* temp = head->next;
        free(head);
        return temp;
    }

    Node* current = head;
    size_t i = 0;

    while (current != NULL && current->next != NULL && i < k - 1) {
        current = current->next;
        i++;
    }

    if (current == NULL || current->next == NULL) {
        return head;
    }

    Node* temp = current->next;
    current->next = temp->next;
    free(temp);

    return head;
}

void free_list(Node* head) {
    Node* current = head;
    while (current != NULL) {
        Node* next = current->next;
        free(current);
        current = next;
    }
}

void print_list(Node* head) {
    Node* current = head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    Node* head = create_node(10);
    head->next = create_node(20);
    head->next->next = create_node(30);
    head->next->next->next = create_node(40);
    head->next->next->next->next = create_node(50);

    print_list(head);

    head = remove_kth_element(head, 2);
    print_list(head);

    head = remove_kth_element(head, 0);
    print_list(head);

    head = remove_kth_element(head, 100);
    print_list(head);

    free_list(head);

    return EXIT_SUCCESS;
}
