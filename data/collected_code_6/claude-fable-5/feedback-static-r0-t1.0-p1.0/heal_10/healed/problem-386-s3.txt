#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_SIZE 1024U

long min_swaps_for_balancing(const char *s, size_t max_len);
static int read_line(char *buffer, size_t size);
static int discard_remaining_input(void);

long min_swaps_for_balancing(const char *s, size_t max_len)
{
    long open_count = 0;
    long close_count = 0;
    long swaps = 0;
    long imbalance = 0;
    size_t i;
    size_t len;

    if (s == NULL) {
        return -1L;
    }

    if (max_len == 0U) {
        return -1L;
    }

    len = strnlen(s, max_len);
    if (len >= max_len) {
        return -1L;
    }

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            if (open_count == LONG_MAX) {
                return -1L;
            }
            open_count++;
        } else if (s[i] == ']') {
            if (close_count == LONG_MAX) {
                return -1L;
            }
            close_count++;
        } else {
            return -1L;
        }
    }

    if (open_count != close_count) {
        return -1L;
    }

    open_count = 0;
    close_count = 0;

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            open_count++;
            if (imbalance > 0) {
                if (swaps > (LONG_MAX - imbalance)) {
                    return -1L;
                }
                swaps += imbalance;
                imbalance--;
            }
        } else {
            close_count++;
            imbalance = close_count - open_count;
        }
    }

    return swaps;
}

static int discard_remaining_input(void)
{
    int ch;
    int truncated = 0;
    unsigned long guard = 0UL;

    do {
        ch = getchar();
        if ((ch != EOF) && (ch != '\n')) {
            truncated = 1;
        }
        guard++;
        if (guard >= ULONG_MAX) {
            return -1;
        }
    } while ((ch != EOF) && (ch != '\n'));

    return truncated;
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if ((buffer == NULL) || (size < 2U) || (size > (size_t)INT_MAX)) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1U] = '\0';

    len = strnlen(buffer, size);
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
        return 0;
    }

    if (len == (size - 1U)) {
        int truncated = discard_remaining_input();

        if (truncated < 0) {
            return -1;
        }
        if (truncated != 0) {
            return -2;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    long result;
    int status;

    (void)memset(buffer, 0, sizeof(buffer));

    printf("Enter a bracket string (only '[' and ']'): ");
    (void)fflush(stdout);

    status = read_line(buffer, sizeof(buffer));
    if (status == -2) {
        fprintf(stderr, "Error: input too long.\n");
        return EXIT_FAILURE;
    }
    if (status != 0) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    result = min_swaps_for_balancing(buffer, sizeof(buffer));

    if (result < 0) {
        fprintf(stderr, "Error: invalid or unbalanceable input string.\n");
        return EXIT_FAILURE;
    }

    printf("Minimum number of swaps required: %ld\n", result);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The two weaknesses flagged by Flawfinder were already mitigated in the original code, so the rewrite preserves those defenses:

1. **`getchar` in a loop (CWE-120, CWE-20):** The `discard_remaining_input` function reads into an `int` (correctly handling `EOF`), never writes to a buffer, and includes an unsigned `guard` counter to prevent an unbounded loop. No buffer boundary can be violated since no buffer is involved.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The `buffer[MAX_INPUT_SIZE]` array is protected because:
   - `read_line` receives the size via `sizeof(buffer)` and validates it (`size >= 2` and `size <= INT_MAX` before the cast to `int` for `fgets`).
   - `fgets` is a length-limited read function, and the buffer is explicitly NUL-terminated afterward (`buffer[size - 1U] = '\0'`).
   - `strnlen` is used instead of `strlen`, bounding all length computations.
   - Oversized input is detected and reported (`-2` status) rather than silently truncated, and leftover stdin is drained.

Additional hardening already present and retained: `NULL`/size parameter validation, `LONG_MAX` overflow guards on all counters and the swap accumulator, strict character-set validation (only `[` and `]` accepted), and `memset` initialization of the buffer before use.
