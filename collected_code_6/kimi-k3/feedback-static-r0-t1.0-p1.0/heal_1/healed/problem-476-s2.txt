#include <stdio.h>
#include <limits.h>

int sum_min_max(const int arr[], int size) {
    if (size <= 0 || arr == NULL) {
        return 0;
    }

    int min = arr[0];
    int max = arr[0];

    for (int i = 1; i < size; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
        if (arr[i] > max) {
            max = arr[i];
        }
    }

    return min + max;
}

int main(void) {
    int arr[] = {5, 3, 9, 1, 7};
    int size = sizeof(arr) / sizeof(arr[0]);

    int result = sum_min_max(arr, size);
    printf("Sum of largest and smallest: %d\n", result);

    return 0;
}
