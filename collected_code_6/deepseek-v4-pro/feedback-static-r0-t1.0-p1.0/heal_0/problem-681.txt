#include <stdio.h>
#include <stdlib.h>

int smallest_prime_divisor(int n) {
    if (n <= 1) return 0;
    if (n % 2 == 0) return 2;
    if (n % 3 == 0) return 3;

    for (int i = 5; i * i <= n; i += 6) {
        if (n % i == 0) return i;
        if (n % (i + 2) == 0) return i + 2;
    }

    return n;
}

int main(void) {
    int numbers[] = {15, 49, 97, 100, 121, 2, 1, 0, -7};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);

    for (size_t i = 0; i < count; i++) {
        int result = smallest_prime_divisor(numbers[i]);
        if (result == 0) {
            printf("Number %d has no prime divisor\n", numbers[i]);
        } else {
            printf("Smallest prime divisor of %d is %d\n", numbers[i], result);
        }
    }

    return 0;
}
