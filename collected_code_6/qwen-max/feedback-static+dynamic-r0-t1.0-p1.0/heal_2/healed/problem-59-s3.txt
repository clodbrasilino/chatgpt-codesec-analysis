```c
#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>

int64_t nth_octagonal_number(int64_t n) {
    return 3 * n * n - 2 * n;
}

int main() {
    int64_t n;
    printf("Enter the value of n: ");
    if (scanf("%" SCNd64, &n) != 1) {
        return 1;
    }
    int64_t result = nth_octagonal_number(n);
    printf("The %" PRId64 "th octagonal number is: %" PRId64 "\n", n, result);
    return 0;
}
```
