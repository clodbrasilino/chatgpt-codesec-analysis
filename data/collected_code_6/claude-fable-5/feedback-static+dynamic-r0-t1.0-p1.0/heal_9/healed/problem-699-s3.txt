#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static void strip_newline(char *s)
{
    size_t len;

    if (s == NULL) {
        return;
    }

    len = strnlen(s, MAX_INPUT_LEN + 2);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r')) {
        s[len - 1] = '\0';
        len--;
    }
}

static int read_token(char *buf, size_t size)
{
    size_t i = 0;
    int c;

    if (buf == NULL || size == 0) {
        return -1;
    }

    do {
        c = getchar();
    } while (c == ' ' || c == '\t' || c == '\n' || c == '\r');

    if (c == EOF) {
        return -1;
    }

    while (c != EOF && c != ' ' && c != '\t' && c != '\n' && c != '\r') {
        if (i < size - 1) {
            buf[i] = (char)c;
            i++;
        } else {
            return -1;
        }
        c = getchar();
    }

    buf[i] = '\0';
    return 0;
}

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count01 = 0;
    int count10 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN + 1);
    len2 = strnlen(s2, MAX_INPUT_LEN + 1);

    if (len1 > MAX_INPUT_LEN || len2 > MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2 || len1 == 0) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] != s2[i]) {
            if (s1[i] == '0') {
                count01++;
            } else {
                count10++;
            }
        }
    }

    if (count01 != count10) {
        return -1;
    }

    return count01;
}

int main(void)
{
    char s1[MAX_INPUT_LEN + 2];
    char s2[MAX_INPUT_LEN + 2];
    int result;

    memset(s1, 0, sizeof(s1));
    memset(s2, 0, sizeof(s2));

    if (read_token(s1, sizeof(s1)) != 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
        return 0;
    }

    if (read_token(s2, sizeof(s2)) != 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
        return 0;
    }

    strip_newline(s1);
    strip_newline(s2);

    result = min_swaps(s1, s2);

    if (result < 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
    } else {
        if (printf("%d\n", result) < 0) {
            return 1;
        }
    }

    return 0;
}

**Key changes made:**

1. **Fixed the `min_swaps` logic (the failing test cases):** The original formula `(count01 / 2) + (count10 / 2) + 2 * (count01 % 2)` was solving a different problem (cross-string swaps, LeetCode-1247 style). The test cases indicate the intended problem is *minimum swaps within one string* to transform `s1` into `s2`. Each swap pairs one `0→1` mismatch with one `1→0` mismatch, fixing both at once, so:
   - If `count01 != count10`, the strings have different character counts and conversion is impossible → return `-1` (printed as `Not Possible`).
   - Otherwise the answer is simply `count01`. This yields `1` instead of `2` (test 0), `-1`/`Not Possible` when mismatch counts are unbalanced (test 1), and `3` instead of `4` (test 2).

2. **Flawfinder `getchar` warnings (CWE-120/CWE-20):** Already mitigated — the read loop strictly bounds writes with `i < size - 1` and rejects oversized tokens rather than truncating silently. Retained as-is.

3. **Flawfinder statically-sized array warnings (CWE-119/CWE-120):** Already mitigated — buffers are zero-initialized, all writes are bounds-checked via `sizeof`, and `strnlen` is used everywhere instead of `strlen`, so no unbounded string operations exist. Retained as-is.
