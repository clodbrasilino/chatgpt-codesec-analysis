```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static char *cleanup_fail(regex_t *regex, char *buffer)
{
    if (regex != NULL) {
        regfree(regex);
    }
    free(buffer);
    return NULL;
}

static int safe_copy(char *dest, size_t dest_size, size_t dest_pos,
                     const char *src, size_t copy_len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_pos >= dest_size) {
        return -1;
    }
    if (copy_len >= dest_size - dest_pos) {
        return -1;
    }
    memcpy(&dest[dest_pos], src, copy_len);
    return 0;
}

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t buf_size;
    size_t out_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        size_t prefix_len;
        size_t remaining;

        if (match[0].rm_so < 0 || match[1].rm_so < 0 ||
            match[0].rm_eo < match[0].rm_so) {
            return cleanup_fail(&regex, result);
        }

        prefix_len = (size_t)match[0].rm_so;

        if (out_pos >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        remaining = buf_size - out_pos;

        if (prefix_len >= remaining || (remaining - prefix_len) < 2U) {
            return cleanup_fail(&regex, result);
        }

        if (prefix_len > strnlen(cursor, prefix_len)) {
            return cleanup_fail(&regex, result);
        }

        if (prefix_len > 0U) {
            if (safe_copy(result, buf_size, out_pos, cursor, prefix_len) != 0) {
                return cleanup_fail(&regex, result);
            }
            out_pos += prefix_len;
        }

        if (out_pos + 1U >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        result[out_pos] = (char)toupper((unsigned char)cursor[match[1].rm_so]);
        out_pos += 1U;

        cursor += match[0].rm_eo;
    }

    {
        size_t rest_len;
        size_t remaining;

        if (out_pos >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        remaining = buf_size - out_pos;
        rest_len = strnlen(cursor, remaining);

        if (rest_len >= remaining) {
            return cleanup_fail(&regex, result);
        }

        if (rest_len > 0U) {
            if (safe_copy(result, buf_size, out_pos, cursor, rest_len) != 0) {
                return cleanup_fail(&regex, result);
            }
            out_pos += rest_len;
        }
    }

    if (out_pos >= buf_size) {
        return cleanup_fail(&regex, result);
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s -> %s\n", tests[i], converted);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

The rewritten code addresses the flagged weakness as follows:

**CWE-120 (memcpy buffer overflow):** The flawfinder warning on `memcpy` is mitigated because `safe_copy` validates all preconditions before copying: it rejects NULL pointers, verifies `dest_pos` is within the buffer, and ensures `copy_len` fits strictly within the remaining space (`copy_len >= dest_size - dest_pos` fails), guaranteeing room is always left for the NUL terminator. The original code had two overlapping checks (`copy_len > ...` immediately followed by `copy_len >= ...`); the first was redundant since the stricter `>=` check subsumes it, so it has been removed.

Additional defensive properties already present and preserved:

- **Input length capping** via `strnlen(input, MAX_INPUT_LEN + 1U)` prevents unbounded reads on non-terminated input and rejects oversized strings.
- **Regex match validation** guards against negative or inconsistent `rm_so`/`rm_eo` offsets before using them in pointer arithmetic.
- **Consistent cleanup path** through `cleanup_fail` prevents memory and regex handle leaks on every error branch, and `regfree` is called on the success path.
- **Output buffer accounting** (`out_pos`, `buf_size`, `remaining`) is checked before every write, including the single-character `toupper` write and the final NUL terminator. Since camelCase output is always shorter than or equal to the snake_case input, `len + 1` is a safe allocation size.
