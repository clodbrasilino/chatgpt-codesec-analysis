```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
            best = max_int(best, dp[i - 1][j] - prices[j]);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

static int run_test(int k, const int *prices, int n, int expected, int index)
{
    int profit = maxProfit(k, prices, n);

    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return -1;
    }

    if (profit != expected) {
        fprintf(stderr, "test case %d failed: expected %d, got %d\n",
                index, expected, profit);
        return 1;
    }

    printf("Maximum profit with at most %d transactions: %d\n", k, profit);
    return 0;
}

int main(void)
{
    int prices0[] = {3, 2, 6, 5, 0, 3};
    int prices1[] = {2, 4, 1};
    int prices2[] = {1, 2, 4, 2, 5, 7, 2, 4, 9, 0};
    int status;
    int failures = 0;

    status = run_test(2, prices0,
                      (int)(sizeof(prices0) / sizeof(prices0[0])), 7, 0);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    status = run_test(2, prices1,
                      (int)(sizeof(prices1) / sizeof(prices1[0])), 2, 1);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    status = run_test(2, prices2,
                      (int)(sizeof(prices2) / sizeof(prices2[0])), 13, 2);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    if (failures > 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The core DP algorithm in the original code was actually sound — it implements the classic "Best Time to Buy and Sell Stock IV" solution with the `best = max(best, dp[i-1][j] - prices[j])` optimization, reducing complexity from O(k·n²) to O(k·n). The correct expected values for `k = 2` are:

- `{3, 2, 6, 5, 0, 3}` → **7** (buy 2 sell 6, buy 0 sell 3)
- `{2, 4, 1}` → **2** (buy 2 sell 4)
- `{1, 2, 4, 2, 5, 7, 2, 4, 9, 0}` → **13** (buy 2 sell 7, buy 2 sell 9; note `k >= n/2` fast path does not trigger here since 2 < 5)

The reported failures ("expected 10, got 5", "got 0") indicate the test harness was comparing against wrong expectations or the function was previously returning early/mis-indexed. The rewrite adds explicit verification against the mathematically correct expected results, distinguishes allocation failure (`-1`) from test failure, preserves the safe cleanup path on partial `calloc` failure, and keeps the guard clauses (`NULL` prices, `n <= 1`, `k <= 0`) and the greedy `k >= n/2` shortcut intact.
