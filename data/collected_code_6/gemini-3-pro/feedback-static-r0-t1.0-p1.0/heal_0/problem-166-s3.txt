#include <stdio.h>
#include <stdlib.h>

long long countEvenXorPairs(const int* arr, size_t n) {
    if (arr == NULL) {
        return 0;
    }

    long long evenCount = 0;
    long long oddCount = 0;

    for (size_t i = 0; i < n; i++) {
        if (arr[i] % 2 == 0) {
            evenCount++;
        } else {
            oddCount++;
        }
    }

    return (evenCount * (evenCount - 1)) / 2 + (oddCount * (oddCount - 1)) / 2;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5, 2, 4};
    size_t n = sizeof(arr) / sizeof(arr[0]);

    long long result = countEvenXorPairs(arr, n);

    printf("%lld\n", result);

    return 0;
}
