```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *camel_to_snake(const char *input)
{
    if (input == NULL) {
        return NULL;
    }

    size_t len = strlen(input);
    size_t max_len = (len * 2) + 1;

    char *output = malloc(max_len);
    if (output == NULL) {
        return NULL;
    }

    size_t out_idx = 0;
    for (size_t i = 0; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (isupper(c)) {
            if (i > 0) {
                output[out_idx++] = '_';
            }
            output[out_idx++] = (char)tolower(c);
        } else {
            output[out_idx++] = (char)c;
        }
    }
    output[out_idx] = '\0';

    return output;
}

int main(void)
{
    const char *tests[] = {
        "helloWorld",
        "camelCaseString",
        "myVeryLongVariableName",
        "simple",
        "A",
        "already_snake"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = camel_to_snake(tests[i]);
        if (result == NULL) {
            fprintf(stderr, "Error: conversion failed for input \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("Input:  %s\n", tests[i]);
        printf("Output: %s\n\n", result);
        free(result);
        result = NULL;
    }

    return EXIT_SUCCESS;
}
```

**Key safety measures addressing the CWEs:**

- **CWE-664 (Resource Lifetime):** Every `malloc` is paired with a `free` in `main`, pointers are set to `NULL` after freeing to prevent use-after-free, and the caller's ownership of the returned buffer is clear.
- **CWE-703 (Exceptional Conditions):** `NULL` input is rejected, `malloc` failure is checked and propagated, and the failure path in `main` returns `EXIT_FAILURE` instead of dereferencing a bad pointer.
- **CWE-710 (Coding Standards):** Uses `size_t` for indices/lengths, casts to `unsigned char` before `ctype.h` calls (avoiding UB), uses `const` correctness, and `int main(void)` with standard exit codes.

The buffer is sized at `2 * len + 1`, which is the worst case (every character uppercase becomes `_x`).
