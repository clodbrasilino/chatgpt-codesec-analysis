#include <stdio.h>
#include <stdlib.h>

int* remove_even(const int* arr, size_t size, size_t* new_size) {
    if (arr == NULL || new_size == NULL) {
        return NULL;
    }
    
    int* result = malloc(size * sizeof(int));
    if (result == NULL && size > 0) {
        return NULL;
    }
    
    size_t count = 0;
    for (size_t i = 0; i < size; i++) {
        if (arr[i] % 2 != 0) {
            result[count++] = arr[i];
        }
    }
    
    *new_size = count;
    return result;
}

int main(void) {
    int test1[] = {1, 2, 3, 4, 5};
    int test2[] = {5, 6, 7};
    int test3[] = {1, 2, 3, 4};
    
    size_t new_size;
    int* result;
    
    result = remove_even(test1, 5, &new_size);
    if (result != NULL) {
        printf("[");
        for (size_t i = 0; i < new_size; i++) {
            printf("%d", result[i]);
            if (i < new_size - 1) printf(", ");
        }
        printf("]\n");
        free(result);
    }
    
    result = remove_even(test2, 3, &new_size);
    if (result != NULL) {
        printf("[");
        for (size_t i = 0; i < new_size; i++) {
            printf("%d", result[i]);
            if (i < new_size - 1) printf(", ");
        }
        printf("]\n");
        free(result);
    }
    
    result = remove_even(test3, 4, &new_size);
    if (result != NULL) {
        printf("[");
        for (size_t i = 0; i < new_size; i++) {
            printf("%d", result[i]);
            if (i < new_size - 1) printf(", ");
        }
        printf("]\n");
        free(result);
    }
    
    return EXIT_SUCCESS;
}
