#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

char **extract_strings(const char *const *list, size_t list_size, size_t count)
{
    char **result;
    size_t i;
    size_t n;

    if (list == NULL) {
        return NULL;
    }

    n = count;
    if (n > list_size) {
        n = list_size;
    }

    result = calloc(n + 1, sizeof(*result));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        size_t len;
        size_t k;

        if (list[i] == NULL) {
            continue;
        }

        len = strnlen(list[i], MAX_STRING_LENGTH);

        result[i] = malloc(len + 1);
        if (result[i] == NULL) {
            size_t j;
            for (j = 0; j < i; j++) {
                free(result[j]);
            }
            free(result);
            return NULL;
        }

        for (k = 0; k < len; k++) {
            result[i][k] = list[i][k];
        }
        result[i][len] = '\0';
    }

    result[n] = NULL;

    return result;
}

void free_strings(char **strings, size_t count)
{
    size_t i;

    if (strings == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(strings[i]);
    }
    free(strings);
}

int main(void)
{
    const char *list[] = {"apple", "banana", "cherry", "date", "elderberry"};
    const size_t list_size = sizeof(list) / sizeof(list[0]);
    const size_t count = 3;
    char **extracted;
    size_t i;

    extracted = extract_strings(list, list_size, count);
    if (extracted == NULL) {
        fprintf(stderr, "Error: failed to extract strings\n");
        return EXIT_FAILURE;
    }

    for (i = 0; extracted[i] != NULL; i++) {
        printf("%s\n", extracted[i]);
    }

    free_strings(extracted, count);

    return EXIT_SUCCESS;
}
