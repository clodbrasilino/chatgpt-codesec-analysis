```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 255U

char *remove_even_chars(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len / 2 + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (i % 2 != 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN + 1];
    char *output;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    output = remove_even_chars(buffer);
    if (output == NULL) {
        fprintf(stderr, "Error processing string\n");
        return EXIT_FAILURE;
    }

    printf("Result: %s\n", output);
    free(output);

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-126 (`strlen` over-read)** — Both occurrences of `strlen` were replaced with `strnlen`, which takes an explicit upper bound. In `remove_even_chars`, the bound is `MAX_INPUT_LEN + 1`, and if the string turns out to be unterminated within that window (i.e., `len > MAX_INPUT_LEN`), the function fails safely by returning `NULL`. In `main`, the bound is `sizeof(buffer)`, guaranteeing the scan never leaves the buffer.

2. **CWE-119/CWE-120 (statically-sized array)** — The buffer size is now driven by a single named constant (`MAX_INPUT_LEN`), and `fgets` is still bounded by `sizeof(buffer)`, which prevents overflow on input. As an additional defensive measure, `buffer[sizeof(buffer) - 1] = '\0';` guarantees NUL-termination regardless of what `fgets` did, ensuring all downstream string operations operate on a properly terminated string.

3. **Existing good practices preserved** — `NULL` checks on the input pointer, `malloc` failure handling, `fgets` failure handling, newline stripping, and `free` of the allocated result all remain intact, so there are no memory leaks or unchecked error paths.

The original code was already fairly safe in practice (bounded `fgets`, checked `malloc`); the rewrite mainly hardens it against the theoretical unterminated-string scenarios flagged by Flawfinder.
