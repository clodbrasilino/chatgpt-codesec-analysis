```c
#include <stdbool.h>
#include <stdio.h>

bool is_even(int num) {
    return (num % 2) == 0;
}

int main() {
    int number;

    printf("Enter a number: ");
    scanf("%d", &number);

    if (is_even(number)) {
        printf("%d is even.\n", number);
    } else {
        printf("%d is not even.\n", number);
    }

    return 0;
}
```
