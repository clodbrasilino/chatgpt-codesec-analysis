```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix);
int replace_with_colon(char *str);

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    size_t errlen;
    char *errbuf;

    errlen = regerror(errcode, regex, NULL, 0);
    errbuf = malloc(errlen);
    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate error buffer\n", prefix);
        return;
    }
    regerror(errcode, regex, errbuf, errlen);
    fprintf(stderr, "%s: %s\n", prefix, errbuf);
    free(errbuf);
}

int replace_with_colon(char *str)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t offset = 0;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strlen(str);

    ret = regcomp(&regex, "[ ,.]", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return -1;
    }

    while (offset < len && str[offset] != '\0') {
        ret = regexec(&regex, str + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            print_regex_error(ret, &regex, "Regex execution failed");
            regfree(&regex);
            return -1;
        }
        if (offset + (size_t)match.rm_so >= len ||
            offset + (size_t)match.rm_eo > len) {
            regfree(&regex);
            return -1;
        }
        str[offset + (size_t)match.rm_so] = ':';
        offset += (size_t)match.rm_eo;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    char text[] = "Hello world, this is a.test string. Enjoy,coding today";

    printf("Original: %s\n", text);

    if (replace_with_colon(text) != 0) {
        fprintf(stderr, "Replacement failed\n");
        return EXIT_FAILURE;
    }

    printf("Modified: %s\n", text);

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **Eliminated the statically-sized `errbuf[256]` arrays (CWE-119/CWE-120):** The original code used fixed 256-byte stack buffers for `regerror()` messages. While `regerror()` itself respects the size limit, the flagged pattern is replaced with a dynamic approach — calling `regerror(errcode, regex, NULL, 0)` first to obtain the exact required buffer size, then allocating precisely that amount with `malloc()`. This guarantees the buffer is always large enough and cannot be improperly restricted.

2. **Centralized error reporting:** Both duplicated error-handling blocks were consolidated into a single `print_regex_error()` helper, reducing the surface area for buffer-handling mistakes and handling the allocation-failure case gracefully.

3. **Added defensive bounds checking on string writes:** The string length is captured up front, and both `match.rm_so` and `match.rm_eo` offsets are validated against it before writing to `str`, ensuring the colon replacement can never write out of bounds even if the regex library returned unexpected offsets.

The program's observable behavior is unchanged: it still replaces each space, comma, and period with a colon, producing `Hello:world::this:is:a:test:string::Enjoy:coding:today`.
