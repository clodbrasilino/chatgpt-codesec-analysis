#include <stdio.h>
#include <stdlib.h>

typedef struct {
    char *key;
    char *value;
} DictionaryEntry;

typedef struct {
    DictionaryEntry *entries;
    int count;
} Dictionary;

Dictionary* create_dictionary() {
    return (Dictionary*)calloc(1, sizeof(Dictionary));
}

void add_entry(Dictionary *dict, const char *key, const char *value) {
    dict->count++;
    dict->entries = (DictionaryEntry*)realloc(dict->entries, dict->count * sizeof(DictionaryEntry));
    dict->entries[dict->count - 1].key = strdup(key);
    dict->entries[dict->count - 1].value = strdup(value);
}

char** get_keys(Dictionary *dict, int *out_count) {
    *out_count = dict->count;
    char **keys = (char**)malloc(dict->count * sizeof(char*));
    for (int i = 0; i < dict->count; i++) {
        keys[i] = dict->entries[i].key;
    }
    return keys;
}

void free_dictionary(Dictionary *dict) {
    for (int i = 0; i < dict->count; i++) {
        free(dict->entries[i].key);
        free(dict->entries[i].value);
    }
    free(dict->entries);
    free(dict);
}

int main() {
    Dictionary *dict = create_dictionary();
    add_entry(dict, "key1", "value1");
    add_entry(dict, "key2", "value2");

    int key_count;
    char **keys = get_keys(dict, &key_count);

    for (int i = 0; i < key_count; i++) {
        printf("%s\n", keys[i]);
    }

    free(keys);
    free_dictionary(dict);

    return 0;
}
