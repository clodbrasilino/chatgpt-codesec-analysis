```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <ctype.h>

#define INPUT_BUFFER_SIZE 64

static int digit_sum(unsigned long long n)
{
    int sum = 0;
    while (n > 0ULL) {
        sum += (int)(n % 10ULL);
        n /= 10ULL;
    }
    return sum;
}

static int parse_non_negative(const char *str, unsigned long long *out)
{
    char *endptr = NULL;
    unsigned long long value = 0ULL;

    if (str == NULL || out == NULL) {
        return -1;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '-' || *str == '\0') {
        return -1;
    }

    errno = 0;
    value = strtoull(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }

    if (endptr == str) {
        return -1;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return -1;
    }

    *out = value;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch = 0;
    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

static int read_line(char *buf, size_t size)
{
    size_t len = 0;

    if (buf == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buf, (int)size, stdin) == NULL) {
        return -1;
    }

    len = strcspn(buf, "\n");

    if (buf[len] == '\0' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    unsigned long long number = 0ULL;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a non-negative integer: ");
    fflush(stdout);

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    if (parse_non_negative(buffer, &number) != 0) {
        fprintf(stderr, "Error: invalid non-negative integer input.\n");
        return EXIT_FAILURE;
    }

    result = digit_sum(number);

    if (printf("Digit sum of %llu is %d\n", number, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar` in a loop (CWE-120, CWE-20)** — This is a false positive. `getchar()` returns an `int` and writes into no buffer; the loop in `discard_remaining_input()` correctly stores the result in an `int` (allowing proper `EOF` comparison) and terminates on newline or end-of-file. There is no buffer to overflow.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — Also mitigated. The `buffer` is only ever written to by `fgets()` via `read_line()`, which receives `sizeof(buffer)` as the bound. `fgets` never writes more than `size - 1` characters plus a null terminator. Additionally, `read_line()`:
   - Validates the pointer and size range before use.
   - Detects truncated (over-long) input by checking whether the newline is missing at `size - 1`, then drains stdin and reports failure rather than silently processing partial input.
   - Guarantees null termination.

3. **Input validation** — `parse_non_negative()` rejects negative values, empty input, trailing garbage, and out-of-range values (`ERANGE` from `strtoull`), covering CWE-20 concerns.

The code as written already addresses both Flawfinder findings; the rewrite above simply removes the comments as requested. No functional changes were required.
