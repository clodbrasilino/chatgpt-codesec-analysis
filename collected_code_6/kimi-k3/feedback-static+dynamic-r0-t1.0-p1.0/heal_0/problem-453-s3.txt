#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long sumEvenFactors(long long n) {
    if (n < 0) {
        if (n == LLONG_MIN) {
            return 0;
        }
        n = -n;
    }
    
    if (n == 0) {
        return 0;
    }
    
    long long sum = 0;
    long long i;
    
    for (i = 2; i <= n / 2; i += 2) {
        if (n % i == 0) {
            if (sum > LLONG_MAX - i) {
                return 0;
            }
            sum += i;
        }
    }
    
    if (n % 2 == 0) {
        if (sum > LLONG_MAX - n) {
            return 0;
        }
        sum += n;
    }
    
    return sum;
}

int main(void) {
    long long number;
    long long result;
    
    printf("Enter a number: ");
    
    if (scanf("%lld", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    result = sumEvenFactors(number);
    
    if (result == 0 && number != 0) {
        printf("No even factors found or overflow occurred\n");
    } else {
        printf("Sum of even factors: %lld\n", result);
    }
    
    return EXIT_SUCCESS;
}
