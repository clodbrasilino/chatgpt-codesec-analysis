```c
#include <stdio.h>

long nth_centered_hexagonal(long n) {
    return 3 * n * (n - 1) + 1;
}

int main() {
    long n;
    printf("Enter the value of n: ");
    if(scanf("%ld", &n) != 1) {
        fprintf(stderr, "Input error\n");
        return 1;
    }
    printf("The %ldth centered hexagonal number is: %ld\n", n, nth_centered_hexagonal(n));
    return 0;
}
```
