```c
#include <stdio.h>
#include <stdlib.h>

int findTriplet(const int *arr, int n, int target) {
    for (int i = 0; i < n - 2; i++) {
        for (int j = i + 1; j < n - 1; j++) {
            for (int k = j + 1; k < n; k++) {
                if (arr[i] + arr[j] + arr[k] == target) {
                    return 1;
                }
            }
        }
    }
    return 0;
}

int main() {
    int arr[] = {1, 4, 9, 16, 25, 36, 49, 64, 81};
    int n = sizeof(arr) / sizeof(arr[0]);
    int target = 14;
    if (findTriplet(arr, n, target)) {
        printf("Triplet with sum %d found.\n", target);
    } else {
        printf("No triplet with sum %d found.\n", target);
    }
    return 0;
}
```
