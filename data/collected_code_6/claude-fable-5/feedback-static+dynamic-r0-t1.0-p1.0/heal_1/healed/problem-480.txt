#include <stdio.h>
#include <limits.h>
#include <stddef.h>
#include <string.h>

#define INPUT_SIZE 256

int find_max_occurring_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    unsigned char max_char = 0;
    int found = 0;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
            max_char = c;
            found = 1;
        }
    }

    if (found == 0) {
        return -1;
    }

    *result = (char)max_char;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    char max_char;
    size_t len;

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strlen(input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_max_occurring_char(input, &max_char) != 0) {
        fprintf(stderr, "Empty string or invalid input\n");
        return 1;
    }

    if (printf("Maximum occurring character: %c\n", max_char) < 0) {
        return 1;
    }

    return 0;
}

**Analysis of the original code and the applied hardening:**

The Flawfinder warning (CWE-119/CWE-120) flags the statically-sized buffer `char input[256]`. The original code was already largely safe because it used `fgets(input, sizeof(input), stdin)`, which performs proper bounds limiting tied to the actual buffer size via `sizeof`. However, the rewrite adds defense-in-depth:

1. **Explicit null termination** — `input[sizeof(input) - 1] = '\0'` guarantees the buffer is a valid C string even under unexpected `fgets` behavior, ensuring all subsequent string operations are bounded.
2. **Named constant** — `INPUT_SIZE` documents the buffer capacity and keeps size usage consistent.
3. **Safer newline stripping** — using `strlen` with a length check avoids iterating an unterminated buffer and correctly handles the empty-string edge case.
4. **Retained correctness properties** — the `unsigned char` cast before indexing `counts[]` prevents negative-index undefined behavior on platforms where `char` is signed; NULL-pointer checks and return-value checks on `fgets` and `printf` are preserved.

The frequency-counting logic itself is bounds-safe: `counts[UCHAR_MAX + 1]` covers every possible `unsigned char` value, so no index can exceed the array.
