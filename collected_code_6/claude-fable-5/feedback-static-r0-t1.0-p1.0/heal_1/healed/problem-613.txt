```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <stdbool.h>

#define NAME_MAX_LEN 32U

typedef struct {
    char name[NAME_MAX_LEN];
    int *records;
    size_t record_count;
} Tuple;

static bool set_tuple_name(Tuple *tuple, const char *name)
{
    size_t len;

    if (tuple == NULL || name == NULL) {
        return false;
    }

    len = strlen(name);
    if (len >= sizeof(tuple->name)) {
        return false;
    }

    memcpy(tuple->name, name, len);
    tuple->name[len] = '\0';
    return true;
}

bool find_max_record(const Tuple *tuples, size_t tuple_count, int *max_out)
{
    bool found = false;
    int max_val = INT_MIN;
    size_t i;
    size_t j;

    if (tuples == NULL || max_out == NULL || tuple_count == 0U) {
        return false;
    }

    for (i = 0U; i < tuple_count; i++) {
        if (tuples[i].records == NULL) {
            continue;
        }
        for (j = 0U; j < tuples[i].record_count; j++) {
            if (!found || tuples[i].records[j] > max_val) {
                max_val = tuples[i].records[j];
                found = true;
            }
        }
    }

    if (!found) {
        return false;
    }

    *max_out = max_val;
    return true;
}

int main(void)
{
    int records1[] = {3, 7, 2};
    int records2[] = {15, 4};
    int records3[] = {9, 11, 6, 8};
    Tuple tuples[3];
    int max_value = 0;

    memset(tuples, 0, sizeof(tuples));

    tuples[0].records = records1;
    tuples[0].record_count = sizeof(records1) / sizeof(records1[0]);
    if (!set_tuple_name(&tuples[0], "alpha")) {
        return EXIT_FAILURE;
    }

    tuples[1].records = records2;
    tuples[1].record_count = sizeof(records2) / sizeof(records2[0]);
    if (!set_tuple_name(&tuples[1], "beta")) {
        return EXIT_FAILURE;
    }

    tuples[2].records = records3;
    tuples[2].record_count = sizeof(records3) / sizeof(records3[0]);
    if (!set_tuple_name(&tuples[2], "gamma")) {
        return EXIT_FAILURE;
    }

    if (find_max_record(tuples, 3U, &max_value)) {
        if (printf("Maximum record value: %d\n", max_value) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (fprintf(stderr, "No records found\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes made:**

1. **CWE-119/CWE-120 (buffer handling on `char name[32]`):** The original code was already safe since it used `snprintf` with `sizeof`, but the rewrite makes bounds enforcement explicit and auditable. The `set_tuple_name` helper validates input pointers, checks the source length against the destination capacity **before** copying (rejecting oversized names rather than silently truncating), and guarantees null termination via `memcpy` plus an explicit terminator. This removes the format-string machinery entirely and makes the length check a hard failure path.

2. **Magic number removal:** The array size is now expressed via the `NAME_MAX_LEN` constant, keeping the declaration and any future validation logic in sync.

3. **Defensive initialization:** `memset(tuples, 0, sizeof(tuples))` ensures no uninitialized struct members (pointers, counts) are ever read, which protects `find_max_record`'s `records == NULL` guard.

4. **Error propagation:** Every fallible operation (`set_tuple_name`, `printf`, `fprintf`) has its return value checked, and failures exit with `EXIT_FAILURE`, consistent with the original code's strict style.

The core algorithm in `find_max_record` was already robust — it guards against `NULL` inputs, zero counts, and `NULL` record arrays, and correctly handles values equal to `INT_MIN` via the `found` flag — so it was preserved unchanged.
