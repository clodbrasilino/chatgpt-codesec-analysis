int *get_lucid_numbers(int n, size_t *count)
{
    if (count == NULL || n < 1) {
        return NULL;
    }
    if ((size_t)n > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    size_t capacity = (size_t)n;
    size_t len = capacity;

    int *arr = malloc(capacity * sizeof *arr);
    if (arr == NULL) {
        return NULL;
    }

    int *result = malloc(capacity * sizeof *result);
    if (result == NULL) {
        free(arr);
        return NULL;
    }

    for (size_t i = 0; i < capacity; i++) {
        arr[i] = (int)(i + 1);
    }

    size_t rcount = 0;
    result[rcount++] = arr[0];

    size_t start = 1;
    while (start < len && start < capacity) {
        int step = arr[start];
        if (step <= 0 || rcount >= capacity) {
            free(arr);
            free(result);
            return NULL;
        }
        result[rcount++] = step;

        size_t write = start;
        size_t idx = 0;

        for (size_t read = start; read < len && read < capacity; read++, idx++) {
            if (idx % (size_t)step != 0) {
                if (write >= capacity || write > read || read >= capacity) {
                    free(arr);
                    free(result);
                    return NULL;
                }
                arr[write] = arr[read];
                write++;
            }
        }

        if (write > capacity) {
            free(arr);
            free(result);
            return NULL;
        }
        len = write;
    }

    free(arr);

    if (rcount > 0 && rcount <= capacity) {
        int *shrunk = realloc(result, rcount * sizeof *shrunk);
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    *count = rcount;
    return result;
}
