#include <stdio.h>
#include <stdint.h>
#include <stdlib.h>

uint32_t highest_power_of_2(uint32_t n) {
    if (n == 0) {
        return 0;
    }
    
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    
    return n - (n >> 1);
}

int main(void) {
    uint32_t n;
    
    if (scanf("%u", &n) != 1) {
        return EXIT_FAILURE;
    }
    
    uint32_t result = highest_power_of_2(n);
    printf("%u\n", result);
    
    return EXIT_SUCCESS;
}
