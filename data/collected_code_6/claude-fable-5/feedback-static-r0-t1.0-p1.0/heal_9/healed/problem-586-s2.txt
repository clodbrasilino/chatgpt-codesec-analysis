#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k);
void print_array(const int *arr, size_t n);

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t temp_size;
    size_t move_size;
    size_t total_size;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*arr)) {
        return -1;
    }

    total_size = n * sizeof(*arr);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > SIZE_MAX / sizeof(*temp)) {
        return -1;
    }

    temp_size = k * sizeof(*temp);

    if (n - k > SIZE_MAX / sizeof(*arr)) {
        return -1;
    }

    move_size = (n - k) * sizeof(*arr);

    if (temp_size > total_size || move_size > total_size) {
        return -1;
    }

    if (temp_size > SIZE_MAX - move_size) {
        return -1;
    }

    if (temp_size + move_size != total_size) {
        return -1;
    }

    temp = malloc(temp_size);
    if (temp == NULL) {
        return -1;
    }

    if (k > n || temp_size > k * sizeof(*temp) || temp_size > total_size) {
        free(temp);
        return -1;
    }

    memcpy(temp, arr, temp_size);

    memmove(arr, arr + k, move_size);

    if ((n - k) + k > n || temp_size > (n - (n - k)) * sizeof(*arr)) {
        free(temp);
        return -1;
    }

    if ((n - k) * sizeof(*arr) > total_size - temp_size) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, temp_size);

    free(temp);
    return 0;
}

void print_array(const int *arr, size_t n)
{
    size_t i;

    if (arr == NULL || n == 0) {
        return;
    }

    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;

    printf("Original array: ");
    print_array(arr, n);

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    printf("Array after splitting at %zu and adding first part to end: ", k);
    print_array(arr, n);

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses:**

The two Flawfinder warnings (CWE-120, risk 2) on the `memcpy` calls are false positives in this code, and the existing guards already mitigate them. Here's why:

1. **First `memcpy(temp, arr, temp_size)`** — The destination `temp` is allocated with exactly `temp_size` bytes via `malloc(temp_size)`, and the allocation is checked for `NULL`. The source `arr` is guaranteed to hold at least `temp_size` bytes because `k < n` (enforced by `k = k % n` and the early return when `k == 0`), so `temp_size = k * sizeof(*arr) < total_size`. Overflow in the multiplication is ruled out by the `k > SIZE_MAX / sizeof(*temp)` check.

2. **Second `memcpy(arr + (n - k), temp, temp_size)`** — The destination region starts at offset `n - k` and receives exactly `k` elements, which fits precisely within the array bounds since `(n - k) + k == n`. The check `temp_size > total_size - temp_size + ...` logic and the `temp_size + move_size == total_size` invariant guarantee no write past `arr[n-1]`.

Additional defensive measures already present that I retained:

- `NULL`/zero-length input rejection.
- Integer overflow checks before every size multiplication (`SIZE_MAX / sizeof(...)`).
- Overflow check before the addition `temp_size + move_size`.
- Consistency invariant `temp_size + move_size == total_size`.
- Correct use of `memmove` (not `memcpy`) for the overlapping in-place shift.
- `free(temp)` on every exit path after allocation, preventing memory leaks (CWE-401).

The code is therefore memory-safe as written; the rewrite preserves all guards while satisfying the "no comments" requirement.
