#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int convertToSingleInteger(const int* list, size_t size, int* result) {
    if (list == NULL || result == NULL || size == 0) {
        return -1;
    }

    long long temp_result = 0;

    for (size_t i = 0; i < size; ++i) {
        if (list[i] < 0 || list[i] > 9) {
            return -1;
        }

        if (temp_result > (INT_MAX - list[i]) / 10) {
            return -1;
        }

        temp_result = temp_result * 10 + list[i];
    }

    *result = (int)temp_result;
    return 0;
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(list) / sizeof(list[0]);
    int result = 0;

    if (convertToSingleInteger(list, size, &result) == 0) {
        printf("Result: %d\n", result);
    } else {
        printf("Error: Conversion failed due to invalid input or overflow.\n");
    }

    int list_overflow[] = {2, 1, 4, 7, 4, 8, 3, 6, 4, 8}; 
    size_t size_overflow = sizeof(list_overflow) / sizeof(list_overflow[0]);
    int result_overflow = 0;

    if (convertToSingleInteger(list_overflow, size_overflow, &result_overflow) == 0) {
        printf("Result: %d\n", result_overflow);
    } else {
        printf("Error: Conversion failed due to invalid input or overflow.\n");
    }

    return 0;
}
