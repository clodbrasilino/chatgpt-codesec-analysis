#include <stdio.h>
#include <string.h>

void convertDateFormat(char *date) {
    char temp[9] = {0};
    if (strlen(date) >= 8) {
        strncpy(temp, date, 8);
    }
    if (strlen(date) >= 11) {
        memmove(date, date + 8, 3);
        memmove(date + 3, date + 5, 3);
        if (strlen(temp) >= 2) {
            memcpy(date + 6, temp, 2);
        }
    }
    date[10] = '\0';
}

int main() {
    char date[] = "2023-04-01";
    convertDateFormat(date);
    printf("%s\n", date);
    return 0;
}
