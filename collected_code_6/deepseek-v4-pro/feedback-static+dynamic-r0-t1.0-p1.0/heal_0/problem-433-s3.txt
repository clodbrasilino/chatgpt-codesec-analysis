```c
#include <stdio.h>
#include <stdlib.h>

int is_greater_than_all(int number, int *array, int size) {
    if (array == NULL || size <= 0) {
        return -1;
    }
    
    for (int i = 0; i < size; i++) {
        if (number <= array[i]) {
            return 0;
        }
    }
    
    return 1;
}

int main(void) {
    int array[] = {10, 20, 30, 40, 50};
    int size = sizeof(array) / sizeof(array[0]);
    int number;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    int result = is_greater_than_all(number, array, size);
    
    if (result == -1) {
        fprintf(stderr, "Error: Invalid array parameters\n");
        return EXIT_FAILURE;
    } else if (result == 1) {
        printf("%d is greater than all elements in the array\n", number);
    } else {
        printf("%d is not greater than all elements in the array\n", number);
    }
    
    return EXIT_SUCCESS;
}
```
