```c
#include <stdio.h>
#include <math.h>

#define PI 3.14159265358979323846

double cylinder_surface_area(double radius, double height) {
    if (radius <= 0 || height <= 0) {
        return -1.0;
    }
    return 2.0 * PI * radius * (radius + height);
}

int main(void) {
    double radius, height, area;

    printf("Enter radius: ");
    if (scanf("%lf", &radius) != 1) {
        printf("Invalid input for radius.\n");
        return 1;
    }

    printf("Enter height: ");
    if (scanf("%lf", &height) != 1) {
        printf("Invalid input for height.\n");
        return 1;
    }

    area = cylinder_surface_area(radius, height);
    if (area < 0) {
        printf("Radius and height must be positive values.\n");
        return 1;
    }

    printf("Surface area of cylinder: %.2f\n", area);

    return 0;
}
```
