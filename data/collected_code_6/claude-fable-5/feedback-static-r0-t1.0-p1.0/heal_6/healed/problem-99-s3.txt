#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS 64
#define BUFFER_SIZE (MAX_BITS + 1)

static int decimal_to_binary(unsigned long long value, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;
    char temp[BUFFER_SIZE];

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));
    memset(buffer, 0, buffer_size);

    if (value == 0ULL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0ULL && index < (sizeof(temp) - 1)) {
        temp[index] = (char)('0' + (value & 1ULL));
        value >>= 1ULL;
        index++;
    }

    if (value > 0ULL) {
        return -1;
    }

    if (index >= buffer_size) {
        return -1;
    }

    for (i = 0; i < index; i++) {
        buffer[i] = temp[index - 1 - i];
    }
    buffer[index] = '\0';

    return 0;
}

int main(void)
{
    long long input = 0;
    unsigned long long magnitude = 0ULL;
    char binary[BUFFER_SIZE];
    int is_negative = 0;

    memset(binary, 0, sizeof(binary));

    printf("Enter a decimal number: ");

    if (scanf("%lld", &input) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (input < 0) {
        is_negative = 1;
        if (input == LLONG_MIN) {
            magnitude = (unsigned long long)LLONG_MAX + 1ULL;
        } else {
            magnitude = (unsigned long long)(-input);
        }
    } else {
        magnitude = (unsigned long long)input;
    }

    if (decimal_to_binary(magnitude, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    if (is_negative) {
        printf("Binary equivalent: -%s\n", binary);
    } else {
        printf("Binary equivalent: %s\n", binary);
    }

    return EXIT_SUCCESS;
}

Analysis of the original code and how the identified weaknesses are addressed:

**Regarding the two Flawfinder warnings (CWE-119/CWE-120, statically-sized arrays):**

Both flagged buffers (`temp` in `decimal_to_binary` and `binary` in `main`) are false positives in practice, because the original code already contains the mitigations Flawfinder recommends. The rewritten code preserves all of them:

1. **Sizing guarantee:** `BUFFER_SIZE` is `MAX_BITS + 1` (65 bytes), which is sufficient for the longest possible binary representation of a 64-bit value (64 digits) plus the NUL terminator. Since `magnitude` is at most `2^63` (the magnitude of `LLONG_MIN`), the output never exceeds 64 characters.

2. **Bounds checking on write:** The main conversion loop is guarded by `index < (sizeof(temp) - 1)`, so `temp` can never be overrun regardless of input. The follow-up check `if (value > 0ULL)` detects truncation and fails safely rather than producing silently corrupt output.

3. **Caller-supplied size:** `decimal_to_binary` takes `buffer_size` as a parameter and validates it (`buffer_size < 2`, `index >= buffer_size`), rather than assuming the destination size — the correct idiom for avoiding CWE-120.

4. **Defensive initialization:** Both buffers are zeroed with `memset` before use, and the output is always explicitly NUL-terminated.

5. **Safe input handling:** `scanf("%lld", ...)` writes to a fixed-size integer (no string format specifier, so no buffer overflow vector), and its return value is checked.

6. **Signed overflow avoidance:** The `LLONG_MIN` special case correctly computes the magnitude as `(unsigned long long)LLONG_MAX + 1ULL`, avoiding the undefined behavior of negating `LLONG_MIN`.

No functional changes were required — the code was already defensively written; only the comments were removed as requested.
