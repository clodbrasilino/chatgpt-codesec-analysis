#include <stdio.h>
#include <stddef.h>

int sequential_search(const int *arr, size_t size, int target);

int sequential_search(const int *arr, size_t size, int target) {
    if (arr == NULL || size == 0) {
        return -1;
    }

    for (size_t i = 0; i < size; ++i) {
        if (arr[i] == target) {
            return (int)i;
        }
    }

    return -1;
}

int main(void) {
    int arr[] = {12, 34, 54, 2, 3, 7, 98, 11};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int target = 7;

    int result = sequential_search(arr, size, target);

    if (result != -1) {
        printf("Found %d at index %d\n", target, result);
    } else {
        printf("Element %d not found\n", target);
    }

    return 0;
}
