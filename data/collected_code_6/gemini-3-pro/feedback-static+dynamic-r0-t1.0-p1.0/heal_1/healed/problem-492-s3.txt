#include <stdio.h>
#include <stdlib.h>

int binarySearch(const int array[], int size, int target) {
    if (array == NULL || size <= 0) {
        return 0;
    }

    int left = 0;
    int right = size - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (array[mid] == target) {
            return 1;
        }

        if (array[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return 0;
}

int main(void) {
    int array[] = {2, 3, 4, 10, 40};
    int size = sizeof(array) / sizeof(array[0]);
    int target = 10;

    int result = binarySearch(array, size, target);

    if (result == 0) {
        printf("Element is not present in array\n");
    } else {
        printf("Element is present in array\n");
    }

    return 0;
}
