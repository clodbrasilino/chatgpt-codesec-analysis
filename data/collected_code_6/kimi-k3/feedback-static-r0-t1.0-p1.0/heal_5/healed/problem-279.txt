```c
#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <stdlib.h>
#include <errno.h>

uint64_t decagonal_number(uint64_t n) {
    return n * (4 * n - 3);
}

int main(void) {
    char buffer[32];
    char *endptr;
    uint64_t n;
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return 1;
    }
    
    errno = 0;
    n = strtoull(buffer, &endptr, 10);
    
    if (errno == ERANGE || endptr == buffer) {
        return 1;
    }
    
    printf("%" PRIu64 "\n", decagonal_number(n));
    
    return 0;
}
```
