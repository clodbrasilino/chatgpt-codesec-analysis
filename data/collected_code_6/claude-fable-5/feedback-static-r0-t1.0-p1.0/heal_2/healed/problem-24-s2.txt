#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

int binary_to_decimal(const char *binary, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, 128U);
    if (len == 0U || len >= 128U) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[128];
    unsigned long long decimal = 0ULL;
    size_t len;

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int ch;
        int truncated = 0;

        while ((ch = getchar()) != '\n' && ch != EOF) {
            truncated = 1;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code already mitigates both Flawfinder findings, so the rewritten version preserves those defenses:

1. **CWE-119/CWE-120 (statically-sized buffer `input[128]`)** — Mitigated by using `fgets` with `sizeof(input)` as the bound, forcing null-termination at `input[sizeof(input) - 1]`, and using `strnlen` instead of `strlen` to guarantee bounded scanning. The `binary_to_decimal` function independently validates length (`len == 0 || len >= 128`), null pointers, and character content, so it doesn't trust the caller.

2. **CWE-120/CWE-20 (`getchar` in a loop)** — This is a false positive in context: the loop only drains excess characters from `stdin` without storing them, so no buffer can overflow. It correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed streams, and it turns silent truncation into an explicit error, which is the correct input-validation behavior.

3. **Additional hardening already present** — The overflow guard `value > (ULLONG_MAX >> 1)` prevents unsigned wraparound during the shift-accumulate conversion, ensuring binary strings exceeding 64 significant bits are rejected rather than silently producing a wrong result.
