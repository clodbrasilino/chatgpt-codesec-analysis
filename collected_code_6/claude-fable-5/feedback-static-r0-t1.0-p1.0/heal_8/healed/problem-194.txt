```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses:**

1. **`getchar` in a loop (CWE-120, CWE-20, risk 1):** This is a false positive. The `getchar()` call in `discard_remaining_input()` never writes into a buffer — each character is read into a single `int` and immediately discarded. The loop terminates correctly on both `'\n'` and `EOF`, so there is no unbounded read into memory and no infinite-loop risk on stream errors. No change required.

2. **Statically-sized `char` array (CWE-119/CWE-120, risk 2):** Also a false positive, because the code already applies every recommended mitigation:
   - Input is read with `fgets(input, sizeof(input), stdin)`, which strictly bounds the write to the buffer size.
   - The buffer is explicitly NUL-terminated afterward as defense in depth.
   - Length checks use `strnlen` with `sizeof(input)` rather than `strlen`, so even a hypothetically unterminated buffer cannot cause an over-read.
   - Overlong input is detected (no trailing newline while the buffer is full), the remainder of the line is drained to keep `stdin` consistent, and the program rejects the input rather than silently truncating.

3. **Additional robustness already present:** `octal_to_decimal` validates all pointers, rejects empty/unterminated input, validates each character is in `'0'..'7'`, and performs a pre-multiplication overflow check against `LONG_MAX`, preventing signed integer overflow (CWE-190).

The code is safe as written; the Flawfinder hits are pattern-based warnings, and the mitigations they ask for are already implemented, so the rewrite is functionally identical with the annotation comments removed.
