```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32
#define DISCARD_GUARD_LIMIT 4096

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len && i < (sizeof(normalized) - 1); i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < DISCARD_GUARD_LIMIT);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    } else {
        (void)discard_remaining_input();
        if (len == sizeof(input) - 1) {
            fprintf(stderr, "Input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (len == 0) {
        fprintf(stderr, "Empty input.\n");
        return EXIT_FAILURE;
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why this code already mitigates them:

**1. `char normalized[MAX_INPUT]` in `has_28_days` (CWE-119/CWE-120):** The buffer is protected on multiple levels. `strnlen(month, MAX_INPUT)` bounds the length measurement, inputs of length `>= MAX_INPUT` are rejected, and the copy loop has a dual guard (`i < len && i < sizeof(normalized) - 1`) guaranteeing space for the null terminator. No overflow is possible.

**2. `getchar()` in a loop in `discard_remaining_input` (CWE-120/CWE-20):** The characters read are never stored in a buffer, so no overflow can occur. The `DISCARD_GUARD_LIMIT` bound additionally prevents unbounded looping on pathological streams, and EOF is handled explicitly.

**3. `char input[MAX_INPUT]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, sizeof(input), stdin)`, which limits writes to the buffer size. A defensive null terminator is set, `strnlen` bounds length checks, oversized input is detected and rejected after flushing the leftover stream data, and empty input is rejected before further processing.

All three Flawfinder hits are false positives in this implementation — the code is defensively written with bounds checking, length-limited functions (`fgets`, `strnlen`), guaranteed null termination, and character validation (`isalpha` with proper `unsigned char` casts to avoid UB with negative `char` values).
