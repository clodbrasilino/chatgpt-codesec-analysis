```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char *toggle_case(const char *input)
{
    size_t len;
    size_t i;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1U);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0U; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (isupper(c)) {
            result[i] = (char)tolower(c);
        } else if (islower(c)) {
            result[i] = (char)toupper(c);
        } else {
            result[i] = (char)c;
        }
    }
    result[len] = '\0';

    return result;
}

int main(void)
{
    const char *original = "Hello, World! 123 abcXYZ";
    char *toggled;

    toggled = toggle_case(original);
    if (toggled == NULL) {
        if (fprintf(stderr, "Error: failed to toggle case\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", original) < 0) {
        free(toggled);
        return EXIT_FAILURE;
    }

    if (printf("Toggled:  %s\n", toggled) < 0) {
        free(toggled);
        return EXIT_FAILURE;
    }

    free(toggled);
    return EXIT_SUCCESS;
}
```

**Key change addressing the identified weakness:**

- **CWE-126 (buffer over-read via `strlen`):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. This guarantees the length scan never reads beyond a fixed upper bound, so a non-NUL-terminated buffer cannot cause an unbounded over-read. If the input reaches the bound (meaning no terminator was found within the limit), the function safely rejects it by returning `NULL` rather than processing potentially malformed data.

The rest of the code was already defensively written (NULL checks, `malloc` result validation, `unsigned char` casts before `ctype.h` calls to avoid undefined behavior on negative `char` values, checked return values of `printf`/`fprintf`, and proper `free` on all exit paths), so it is preserved as-is.
