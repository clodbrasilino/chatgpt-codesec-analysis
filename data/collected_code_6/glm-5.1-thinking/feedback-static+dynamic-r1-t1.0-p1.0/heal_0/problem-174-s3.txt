#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct ValueNode {
    int value;
    struct ValueNode *next;
} ValueNode;

typedef struct DictEntry {
    char *key;
    ValueNode *values;
    struct DictEntry *next;
} DictEntry;

typedef struct Dictionary {
    DictEntry **buckets;
    size_t size;
} Dictionary;

Dictionary *create_dictionary(size_t size) {
    Dictionary *dict = malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    dict->buckets = calloc(size, sizeof(DictEntry *));
    if (!dict->buckets) {
        free(dict);
        return NULL;
    }
    dict->size = size;
    return dict;
}

unsigned long hash_key(const char *str, size_t size) {
    unsigned long hash = 5381;
    int c;
    while ((c = *str++))
        hash = ((hash << 5) + hash) + c;
    return hash % size;
}

int add_to_dictionary(Dictionary *dict, const char *key, int value) {
    if (!dict || !key) return 0;
    unsigned long idx = hash_key(key, dict->size);
    DictEntry *entry = dict->buckets[idx];
    while (entry) {
        if (strcmp(entry->key, key) == 0) break;
        entry = entry->next;
    }
    if (!entry) {
        entry = malloc(sizeof(DictEntry));
        if (!entry) return 0;
        entry->key = strdup(key);
        if (!entry->key) {
            free(entry);
            return 0;
        }
        entry->values = NULL;
        entry->next = dict->buckets[idx];
        dict->buckets[idx] = entry;
    }
    ValueNode *vnode = malloc(sizeof(ValueNode));
    if (!vnode) return 0;
    vnode->value = value;
    vnode->next = entry->values;
    entry->values = vnode;
    return 1;
}

void free_dictionary(Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        DictEntry *entry = dict->buckets[i];
        while (entry) {
            DictEntry *next_entry = entry->next;
            free(entry->key);
            ValueNode *vnode = entry->values;
            while (vnode) {
                ValueNode *next_vnode = vnode->next;
                free(vnode);
                vnode = next_vnode;
            }
            free(entry);
            entry = next_entry;
        }
    }
    free(dict->buckets);
    free(dict);
}

void print_dictionary(const Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        DictEntry *entry = dict->buckets[i];
        while (entry) {
            printf("%s: [", entry->key);
            ValueNode *vnode = entry->values;
            while (vnode) {
                printf("%d", vnode->value);
                if (vnode->next) printf(", ");
                vnode = vnode->next;
            }
            printf("]\n");
            entry = entry->next;
        }
    }
}

int main(void) {
    Dictionary *dict = create_dictionary(16);
    if (!dict) return EXIT_FAILURE;
    add_to_dictionary(dict, "apple", 1);
    add_to_dictionary(dict, "banana", 2);
    add_to_dictionary(dict, "apple", 3);
    add_to_dictionary(dict, "cherry", 4);
    add_to_dictionary(dict, "banana", 5);
    print_dictionary(dict);
    free_dictionary(dict);
    return EXIT_SUCCESS;
}
