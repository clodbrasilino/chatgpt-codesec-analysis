#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define BUFFER_SIZE 256

bool first_last_equal(const char *str, size_t max_len)
{
    size_t len;

    if (str == NULL || max_len == 0) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    return str[0] == str[len - 1];
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    printf("Enter a string: ");

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    }

    if (len == 0 || buffer[0] == '\0') {
        printf("Empty string provided\n");
        return 1;
    }

    if (first_last_equal(buffer, sizeof(buffer))) {
        printf("The first and last characters are equal\n");
    } else {
        printf("The first and last characters are not equal\n");
    }

    return 0;
}

1. **Bounded input:** `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to `BUFFER_SIZE - 1` characters plus the null terminator — no overflow is possible.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0';` is a defensive belt-and-suspenders measure ensuring null termination even though `fgets` already guarantees it on success.
3. **Bounded length calculation:** `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so even a hypothetically unterminated buffer would not cause an out-of-bounds read.
4. **Safe helper function:** `first_last_equal` validates for `NULL`, zero `max_len`, empty strings, and the `len == max_len` case (unterminated buffer), and all indexing (`str[0]`, `str[len - 1]`) is provably in bounds after those checks.
5. **Newline stripping** is done with an explicit length check (`len > 0`) before indexing `buffer[len - 1]`.
