#include <stdio.h>
#include <stdlib.h>

int find_smallest_missing(int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return 1;
    }

    size_t low = 0;
    size_t high = size - 1;

    if (arr[0] != 1) {
        return 1;
    }

    if (arr[high] == high + 1) {
        return (int)size + 1;
    }

    while (low <= high) {
        size_t mid = low + (high - low) / 2;

        if (arr[mid] == mid + 1) {
            low = mid + 1;
        } else {
            if (mid == 0 || arr[mid - 1] == mid) {
                return (int)mid + 1;
            }
            high = mid - 1;
        }
    }

    return -1;
}

int main(void) {
    int arr1[] = {1, 2, 3, 5, 6};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    int result1 = find_smallest_missing(arr1, size1);
    printf("%d\n", result1);

    int arr2[] = {1, 2, 3, 4, 5};
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);
    int result2 = find_smallest_missing(arr2, size2);
    printf("%d\n", result2);

    int arr3[] = {2, 3, 4, 5, 6};
    size_t size3 = sizeof(arr3) / sizeof(arr3[0]);
    int result3 = find_smallest_missing(arr3, size3);
    printf("%d\n", result3);

    int result4 = find_smallest_missing(NULL, 0);
    printf("%d\n", result4);

    return 0;
}
