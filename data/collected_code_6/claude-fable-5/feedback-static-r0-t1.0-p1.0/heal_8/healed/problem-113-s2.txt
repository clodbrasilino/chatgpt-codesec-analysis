#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

#define MAX_LINE_LENGTH 1048576

int is_integer(const char *str)
{
    char *endptr;
    long value;

    if (str == NULL) {
        return 0;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    errno = 0;
    value = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return 0;
    }

    if (endptr == str) {
        return 0;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 0;
    }

    (void)value;
    return 1;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *line;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    line = malloc(capacity);
    if (line == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= MAX_LINE_LENGTH) {
            free(line);
            return NULL;
        }
        if (length + 1 >= capacity) {
            size_t new_capacity;
            char *tmp;

            if (capacity > SIZE_MAX / 2) {
                free(line);
                return NULL;
            }
            new_capacity = capacity * 2;
            tmp = realloc(line, new_capacity);
            if (tmp == NULL) {
                free(line);
                return NULL;
            }
            line = tmp;
            capacity = new_capacity;
        }
        line[length++] = (char)ch;
    }

    if (ch == EOF && length == 0) {
        free(line);
        return NULL;
    }

    line[length] = '\0';
    return line;
}

int main(void)
{
    char *buffer;

    printf("Enter a string: ");
    fflush(stdout);

    buffer = read_line(stdin);
    if (buffer == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (is_integer(buffer)) {
        printf("\"%s\" is a valid integer\n", buffer);
    } else {
        printf("\"%s\" is not a valid integer\n", buffer);
    }

    free(buffer);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness (CWE-120, CWE-20 — `fgetc` in a loop):**

The Flawfinder warning is a low-risk (level 1) heuristic hit. Reviewing the loop, the code already mitigates the concern correctly, so no functional changes were required:

1. **Buffer overflow (CWE-120) is prevented** — the `length + 1 >= capacity` check guarantees space for both the incoming character and the terminating `'\0'` before every write. After the loop, `line[length] = '\0'` is always within bounds.

2. **Integer overflow in growth logic is prevented** — the `capacity > SIZE_MAX / 2` guard stops `capacity * 2` from wrapping around.

3. **Unbounded input (CWE-20 / resource exhaustion) is prevented** — the `MAX_LINE_LENGTH` cap (1 MiB) rejects pathologically long input rather than allocating without limit.

4. **Correct EOF handling** — `ch` is an `int`, so `EOF` is distinguishable from valid bytes (a classic pitfall when `char` is used), and the `ch == EOF && length == 0` case cleanly frees memory instead of returning an empty allocation.

5. **No memory leaks on error paths** — every failure branch (`realloc` failure, overflow guard, length cap) frees `line` before returning `NULL`, and `main` frees the buffer on success.

The `is_integer` validation is also robust: it uses `strtol` with `errno` checked for `ERANGE`, verifies `endptr != str` (no digits consumed), and tolerates only leading/trailing whitespace. The code as written already addresses the scanner's concerns; the warning can be treated as a verified false positive.
