#include <stdio.h>
#include <stdlib.h>

unsigned long long countSubsequences(const int arr[], int n, int k) {
    if (!arr || n <= 0 || k <= 0) {
        return 0;
    }

    int zeroCount = 0;
    int nonZeroCount = 0;
    int *nonZeroArr = (int *)malloc((size_t)n * sizeof(int));
    
    if (!nonZeroArr) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        if (arr[i] == 0) {
            zeroCount++;
        } else {
            nonZeroArr[nonZeroCount++] = arr[i];
        }
    }

    unsigned long long *dp = (unsigned long long *)calloc((size_t)k, sizeof(unsigned long long));
    if (!dp) {
        free(nonZeroArr);
        return 0;
    }

    for (int i = 0; i < nonZeroCount; i++) {
        int x = nonZeroArr[i];
        if (x >= k) {
            continue;
        }
        for (int j = k - 1; j >= x; j--) {
            dp[j] += dp[j / x] + 1ULL;
        }
    }

    unsigned long long V = dp[k - 1];
    
    free(dp);
    free(nonZeroArr);

    unsigned long long powZ = 1ULL;
    for (int i = 0; i < zeroCount; i++) {
        powZ *= 2ULL;
    }

    unsigned long long powNZ = 1ULL;
    for (int i = 0; i < nonZeroCount; i++) {
        powNZ *= 2ULL;
    }

    return V + (powZ - 1ULL) * powNZ;
}

int main(void) {
    int arr1[] = {1, 2, 3, 4};
    int k1 = 10;
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    
    unsigned long long result1 = countSubsequences(arr1, n1, k1);
    printf("%llu\n", result1);

    int arr2[] = {0, 2, 3};
    int k2 = 5;
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    
    unsigned long long result2 = countSubsequences(arr2, n2, k2);
    printf("%llu\n", result2);

    return 0;
}
