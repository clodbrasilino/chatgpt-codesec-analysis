```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(const char *input, char *output, size_t output_size);

int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    size_t remaining;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len;
        size_t space_left;

        if ((match.rm_so < 0) || (match.rm_eo <= match.rm_so)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        prefix_len = (size_t)match.rm_so;

        if (out_pos >= (output_size - 1U)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        space_left = output_size - 1U - out_pos;

        if (prefix_len >= space_left) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        memcpy(&output[out_pos], cursor, prefix_len);
        out_pos += prefix_len;
        output[out_pos] = ' ';
        out_pos++;

        cursor += match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    remaining = strlen(cursor);

    if (remaining > (output_size - 1U - out_pos)) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    memcpy(&output[out_pos], cursor, remaining);
    out_pos += remaining;
    output[out_pos] = '\0';

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256] = {0};

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two `memcpy` warnings (CWE-120) flagged by Flawfinder are false positives in the original code, since both copies are preceded by explicit bounds checks (`prefix_len >= space_left` and `remaining > output_size - 1U - out_pos`). They are retained as-is because the guards guarantee the destination can hold the source data plus the terminating byte.

One genuine subtle issue was fixed: the original used `strnlen(cursor, output_size)`, which silently truncates the measured length of the remaining input to `output_size`. If the tail of the input were longer than `output_size`, the length check could pass while the copy would drop data, or the check logic could behave inconsistently at the boundary. Replacing it with `strlen(cursor)` measures the true remaining length, so the subsequent bounds check correctly rejects inputs that don't fit rather than silently truncating.

The statically-sized `result[256]` buffer flagged in `main` (CWE-119/CWE-120) is safe because `sizeof(result)` is passed to the function, which enforces all writes stay within bounds and always NUL-terminates; the buffer is also zero-initialized. All error paths release the compiled regex with `regfree` and clear the output buffer, avoiding both resource leaks and the return of partially written data.
