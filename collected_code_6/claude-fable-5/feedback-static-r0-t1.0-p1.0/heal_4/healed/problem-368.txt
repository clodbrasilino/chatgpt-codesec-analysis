```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int *elements;
    size_t size;
} Tuple;

Tuple *tuple_create(const int *values, size_t size);
void tuple_destroy(Tuple *t);
Tuple *tuple_repeat(const Tuple *t, size_t n);
void tuple_print(const Tuple *t);

Tuple *tuple_create(const int *values, size_t size)
{
    Tuple *t = NULL;
    size_t bytes = 0U;

    if (values == NULL || size == 0U) {
        return NULL;
    }

    if (size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    bytes = size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->elements = malloc(bytes);
    if (t->elements == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->elements, values, bytes);
    t->size = size;

    return t;
}

void tuple_destroy(Tuple *t)
{
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->size = 0U;
        free(t);
    }
}

Tuple *tuple_repeat(const Tuple *t, size_t n)
{
    Tuple *result = NULL;
    size_t total = 0U;
    size_t chunk_bytes = 0U;
    size_t total_bytes = 0U;
    size_t i = 0U;

    if (t == NULL || t->elements == NULL || t->size == 0U || n == 0U) {
        return NULL;
    }

    if (t->size > SIZE_MAX / n) {
        return NULL;
    }

    total = t->size * n;

    if (total > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    chunk_bytes = t->size * sizeof(int);
    total_bytes = total * sizeof(int);

    if (chunk_bytes > total_bytes) {
        return NULL;
    }

    result = malloc(sizeof(Tuple));
    if (result == NULL) {
        return NULL;
    }

    result->elements = malloc(total_bytes);
    if (result->elements == NULL) {
        free(result);
        return NULL;
    }

    for (i = 0U; i < n; i++) {
        size_t offset = i * t->size;

        if (offset > total - t->size) {
            free(result->elements);
            free(result);
            return NULL;
        }
        memcpy(&result->elements[offset], t->elements, chunk_bytes);
    }

    result->size = total;

    return result;
}

void tuple_print(const Tuple *t)
{
    size_t i = 0U;

    if (t == NULL || t->elements == NULL) {
        printf("(empty)\n");
        return;
    }

    printf("(");
    for (i = 0U; i < t->size; i++) {
        printf("%d", t->elements[i]);
        if (i < t->size - 1U) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void)
{
    int values[] = {1, 2, 3};
    Tuple *original = NULL;
    Tuple *repeated = NULL;
    size_t n = 3U;

    original = tuple_create(values, sizeof(values) / sizeof(values[0]));
    if (original == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    repeated = tuple_repeat(original, n);
    if (repeated == NULL) {
        fprintf(stderr, "Failed to repeat tuple\n");
        tuple_destroy(original);
        return EXIT_FAILURE;
    }

    printf("Original tuple: ");
    tuple_print(original);
    printf("Repeated %zu times: ", n);
    tuple_print(repeated);

    tuple_destroy(original);
    tuple_destroy(repeated);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **`offset` variable scope reduced (cppcheck `variableScope` / `unreadVariable`)**: The `offset` variable was previously declared at function scope in `tuple_repeat` and initialized to `0U`, a value that was never read before being overwritten inside the loop. It is now declared inside the `for` loop body, reducing its scope to where it is actually used and eliminating the dead initialization.

2. **`memcpy` calls (Flawfinder CWE-120)**: These are flagged by static analysis as a matter of policy, but the code already mitigates the risk correctly, so the calls were retained:
   - In `tuple_create`, the destination buffer is allocated with exactly `bytes` (`size * sizeof(int)`), guarded against overflow via the `SIZE_MAX / sizeof(int)` check, so the copy of `bytes` cannot overflow the destination.
   - In `tuple_repeat`, overflow checks on `t->size * n` and `total * sizeof(int)` guarantee `total_bytes` is valid, and the per-iteration `offset > total - t->size` guard ensures each `chunk_bytes` copy stays within the allocated buffer.

No functional behavior changed; the fixes are purely hygiene improvements confirming the analyzer findings were either resolved (variable scope) or already defended against (bounded copies).
