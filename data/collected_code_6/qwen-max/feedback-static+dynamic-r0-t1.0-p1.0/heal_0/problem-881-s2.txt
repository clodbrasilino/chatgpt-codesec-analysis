#include <stdio.h>
#include <stdbool.h>

int sum_first_even_odd(int *arr, size_t len, int *sum_even, int *sum_odd) {
    bool found_even = false;
    bool found_odd = false;

    for(size_t i = 0; i < len; ++i) {
        if (!found_even && arr[i] % 2 == 0) {
            *sum_even = arr[i];
            found_even = true;
        }
        if (!found_odd && arr[i] % 2 != 0) {
            *sum_odd = arr[i];
            found_odd = true;
        }
        if (found_even && found_odd) {
            break;
        }
    }

    return found_even && found_odd ? 0 : -1;
}

int main() {
    int arr[] = {3, 6, 9, 2, 11};
    size_t len = sizeof(arr) / sizeof(arr[0]);
    int sum_even, sum_odd;

    int result = sum_first_even_odd(arr, len, &sum_even, &sum_odd);

    if (result == 0) {
        printf("Sum of first even and odd: %d\n", sum_even + sum_odd);
    } else {
        printf("Failed to find both first even and odd numbers.\n");
    }

    return 0;
}
