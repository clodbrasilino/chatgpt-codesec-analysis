#include <stdio.h>

void findCombSums(int arr[], int n, int sum) {
    int dp[sum+1];
    dp[0]=1; 
    for (int i=1; i<=sum; i++)
        dp[i] = 0;
    for (int i=0; i<n; i++)
        for (int j=sum; j>=arr[i]; j--)
            dp[j] |= dp[j-arr[i]];
    printf("%d\n", dp[sum]);
}

int main(){
    int arr[] = {1, 2, 3};
    int sum = 4;
    int n = sizeof(arr)/sizeof(arr[0]);
    findCombSums(arr, n, sum);
    return 0;
}
