#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

static bool is_word_length_even(const char *word, size_t max_len)
{
    size_t len;

    if (word == NULL) {
        return false;
    }

    len = strnlen(word, max_len);
    if (len == max_len) {
        return false;
    }

    return (len % 2U) == 0U;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;
    bool newline_found;

    printf("Enter a word: ");

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    newline_found = (len > 0U && buffer[len - 1U] == '\n');

    if (newline_found) {
        buffer[len - 1U] = '\0';
        len--;
    } else if (len == sizeof(buffer) - 1U) {
        discard_remaining_input();
    }

    if (len == 0U) {
        fprintf(stderr, "No word entered\n");
        return EXIT_FAILURE;
    }

    if (is_word_length_even(buffer, sizeof(buffer))) {
        printf("The word \"%s\" has an even length (%zu)\n", buffer, len);
    } else {
        printf("The word \"%s\" has an odd length (%zu)\n", buffer, len);
    }

    return EXIT_SUCCESS;
}

1. **`char buffer[BUFFER_SIZE]` (CWE-119/CWE-120, risk 2):** The fixed-size buffer is safe because `fgets()` is called with `sizeof(buffer)`, which guarantees no write beyond the array bounds. The explicit `buffer[sizeof(buffer) - 1U] = '\0'` termination is defensive redundancy (fgets already null-terminates), and all subsequent length operations use the bounded `strnlen()` rather than `strlen()`.

2. **`getchar()` in a loop (CWE-120/CWE-20, risk 1):** This is a false positive in context — `discard_remaining_input()` reads and discards characters without storing them into any buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed/redirected stdin.
