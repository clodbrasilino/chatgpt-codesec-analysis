```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static int append_chunk(char *dest, size_t capacity, size_t *dest_pos,
                        const char *src, size_t chunk_len)
{
    if (dest == NULL || dest_pos == NULL || src == NULL) {
        return -1;
    }
    if (*dest_pos >= capacity) {
        return -1;
    }
    if (chunk_len >= capacity - *dest_pos) {
        return -1;
    }
    memcpy(dest + *dest_pos, src, chunk_len);
    *dest_pos += chunk_len;
    return 0;
}

char *remove_non_alnum(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t capacity;
    size_t result_pos;
    size_t offset;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    capacity = input_len + 1;
    result = malloc(capacity);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[^a-zA-Z0-9]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_pos = 0;
    offset = 0;

    while (offset < input_len) {
        ret = regexec(&regex, input + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = input_len - offset;
            if (append_chunk(result, capacity, &result_pos,
                             input + offset, remaining) != 0) {
                regfree(&regex);
                free(result);
                return NULL;
            }
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        if (match.rm_so > 0) {
            size_t chunk = (size_t)match.rm_so;
            if (chunk > input_len - offset) {
                regfree(&regex);
                free(result);
                return NULL;
            }
            if (append_chunk(result, capacity, &result_pos,
                             input + offset, chunk) != 0) {
                regfree(&regex);
                free(result);
                return NULL;
            }
        }
        if (match.rm_eo <= 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        if ((size_t)match.rm_eo > input_len - offset) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        offset += (size_t)match.rm_eo;
    }

    if (result_pos >= capacity) {
        regfree(&regex);
        free(result);
        return NULL;
    }

    result[result_pos] = '\0';

    regfree(&regex);

    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a@b#c$d%e^1&2*3",
        "!!!???...",
        "AlreadyClean123"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alnum(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: %s\nCleaned:  %s\n\n", test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

Analysis of the original code and the fix applied:

The flawfinder warning on `memcpy` (CWE-120) was already substantially mitigated in the original code, since `append_chunk` validated that `chunk_len < capacity - *dest_pos` before copying. However, that check has a subtle weakness: if `*dest_pos` ever exceeded `capacity`, the subtraction `capacity - *dest_pos` would underflow (both are unsigned `size_t`), producing a huge value and allowing the bounds check to pass incorrectly. The rewrite adds an explicit `*dest_pos >= capacity` guard before the subtraction, making the bounds check safe against unsigned wraparound and fully addressing the CWE-120 concern.

All other defensive measures from the original are preserved: NULL-pointer checks, `strnlen` with a hard cap to avoid unbounded reads, validation of `rm_so`/`rm_eo` regex offsets before use, cleanup of the regex and heap allocation on every error path, and a final bounds check before writing the NUL terminator.
