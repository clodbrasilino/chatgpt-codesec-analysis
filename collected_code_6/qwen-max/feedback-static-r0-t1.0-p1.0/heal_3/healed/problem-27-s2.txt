```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void remove_digits(char *str, size_t len) {
    char *write = str;
    for (char *read = str; read - str < len; read++) {
        if (!isdigit((unsigned char)*read)) {
            if (write - str < len - 1) {
                *write++ = *read;
            }
        }
    }
    *write = '\0';
}

int main() {
    char strings[][20] = {"hello123", "world4567", "test890"};
    int num_strings = sizeof(strings) / sizeof(strings[0]);

    for (int i = 0; i < num_strings; i++) {
        size_t len = strnlen(strings[i], sizeof(strings[i]));
        remove_digits(strings[i], len);
        printf("%s\n", strings[i]);
    }

    return 0;
}
```
