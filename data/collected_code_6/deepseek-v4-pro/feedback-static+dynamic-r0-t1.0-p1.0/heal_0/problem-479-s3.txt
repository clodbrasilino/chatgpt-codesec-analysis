```c
#include <stdio.h>
#include <stdlib.h>

int first_digit(int number) {
    number = abs(number);
    while (number >= 10) {
        number /= 10;
    }
    return number;
}

int main(void) {
    int values[] = {0, 5, -7, 42, -123, 9876, -10000};
    size_t count = sizeof(values) / sizeof(values[0]);
    
    for (size_t i = 0; i < count; i++) {
        printf("First digit of %d is %d\n", values[i], first_digit(values[i]));
    }
    
    return 0;
}
```
