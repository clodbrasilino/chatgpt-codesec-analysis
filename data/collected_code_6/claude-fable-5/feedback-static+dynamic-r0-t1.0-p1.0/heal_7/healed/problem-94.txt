```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 32

typedef struct {
    char name[MAX_NAME_LEN];
    int value;
} Record;

int index_minimum(const Record *records, size_t count, char *result, size_t result_size)
{
    size_t i;
    size_t min_index;
    size_t name_len;

    if (records == NULL || result == NULL || count == 0 || result_size == 0) {
        return -1;
    }

    min_index = 0;
    for (i = 1; i < count; i++) {
        if (records[i].value < records[min_index].value) {
            min_index = i;
        }
    }

    name_len = strnlen(records[min_index].name, MAX_NAME_LEN);
    if (name_len >= MAX_NAME_LEN || name_len >= result_size) {
        return -1;
    }

    memcpy(result, records[min_index].name, name_len);
    result[name_len] = '\0';

    return 0;
}

int main(void)
{
    Record records[] = {
        {"Rash", 143},
        {"Manjeet", 200},
        {"Varsha", 100}
    };
    size_t count = sizeof(records) / sizeof(records[0]);
    char result[MAX_NAME_LEN];

    if (index_minimum(records, count, result, sizeof(result)) != 0) {
        fprintf(stderr, "Error: could not extract minimum record\n");
        return EXIT_FAILURE;
    }

    printf("Record with minimum value: %s\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char name[MAX_NAME_LEN]` and `char result[MAX_NAME_LEN]` (CWE-119/CWE-120):** Statically-sized arrays are only dangerous when writes to them are unbounded. Here, all writes are bounds-checked. The struct arrays in `main` are initialized with string literals shorter than `MAX_NAME_LEN`, and `result` is only ever written to via the validated `memcpy` path. No change needed.

2. **`memcpy` (CWE-120):** The copy is already guarded correctly:
   - `strnlen(..., MAX_NAME_LEN)` bounds the source length even if `name` were not NUL-terminated.
   - The check `name_len >= MAX_NAME_LEN` rejects unterminated source strings.
   - The check `name_len >= result_size` guarantees `name_len + 1` bytes fit in the destination, so both the `memcpy` and the subsequent NUL-terminator write at `result[name_len]` are in bounds.

3. **General robustness:** NULL-pointer checks, zero-count, and zero-size guards are present, and the caller passes `sizeof(result)` rather than a hardcoded size, keeping the buffer size and its usage in sync.
