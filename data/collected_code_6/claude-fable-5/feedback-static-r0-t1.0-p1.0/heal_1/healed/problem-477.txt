```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *string_to_lower(const char *input)
{
    size_t len;
    size_t i;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < len; i++) {
        result[i] = (char)tolower((unsigned char)input[i]);
    }
    result[len] = '\0';

    return result;
}

int main(void)
{
    const char *original = "Hello, World! THIS Is A Test String.";
    char *lowered;

    lowered = string_to_lower(original);
    if (lowered == NULL) {
        fprintf(stderr, "Error: conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", original) < 0) {
        free(lowered);
        return EXIT_FAILURE;
    }

    if (printf("Lowered:  %s\n", lowered) < 0) {
        free(lowered);
        return EXIT_FAILURE;
    }

    free(lowered);
    lowered = NULL;

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder CWE-126 warning: `strlen` was replaced with `strnlen` bounded by `MAX_INPUT_LEN`, so a non-NUL-terminated string can no longer cause an unbounded over-read. If the bound is reached (meaning no terminator was found within the limit), the function safely returns `NULL` instead of processing potentially invalid data. All other defensive measures from the original — the `NULL` input check, `malloc` failure check, the `(unsigned char)` cast before `tolower` to avoid undefined behavior with negative `char` values, explicit NUL termination, checked `printf` return values, and freeing plus nulling the pointer — are preserved.
