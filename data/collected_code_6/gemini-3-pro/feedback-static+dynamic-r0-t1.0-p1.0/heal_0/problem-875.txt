```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int first;
    int second;
} Tuple;

int find_min_difference(Tuple* tuples, size_t size) {
    if (tuples == NULL || size == 0) {
        return -1;
    }

    int min_diff = INT_MAX;

    for (size_t i = 0; i < size; i++) {
        int diff = abs(tuples[i].first - tuples[i].second);
        if (diff < min_diff) {
            min_diff = diff;
        }
    }

    return min_diff;
}

int main(void) {
    size_t size = 4;
    Tuple* tuples = (Tuple*)malloc(size * sizeof(Tuple));
    
    if (tuples == NULL) {
        return EXIT_FAILURE;
    }

    tuples[0].first = 10;
    tuples[0].second = 15;
    
    tuples[1].first = 20;
    tuples[1].second = 22;
    
    tuples[2].first = 30;
    tuples[2].second = 29;
    
    tuples[3].first = 40;
    tuples[3].second = 48;

    int min_diff = find_min_difference(tuples, size);
    
    if (min_diff != -1) {
        printf("%d\n", min_diff);
    }

    free(tuples);
    tuples = NULL;

    return EXIT_SUCCESS;
}
```
