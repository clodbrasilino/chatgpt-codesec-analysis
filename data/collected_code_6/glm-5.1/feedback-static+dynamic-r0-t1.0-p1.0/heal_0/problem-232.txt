#include <stdio.h>
#include <stdlib.h>

int cmp_desc(const void *a, const void *b) {
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    return (ia < ib) - (ia > ib);
}

int *get_n_largest(const int *dataset, size_t dataset_len, size_t n, size_t *out_len) {
    if (dataset == NULL || out_len == NULL) {
        return NULL;
    }

    *out_len = 0;

    if (dataset_len == 0 || n == 0) {
        return NULL;
    }

    size_t k = n < dataset_len ? n : dataset_len;
    int *temp = (int *)malloc(dataset_len * sizeof(int));
    if (temp == NULL) {
        return NULL;
    }

    for (size_t i = 0; i < dataset_len; i++) {
        temp[i] = dataset[i];
    }

    qsort(temp, dataset_len, sizeof(int), cmp_desc);

    int *result = (int *)malloc(k * sizeof(int));
    if (result == NULL) {
        free(temp);
        return NULL;
    }

    for (size_t i = 0; i < k; i++) {
        result[i] = temp[i];
    }

    free(temp);
    *out_len = k;

    return result;
}

int main(void) {
    int data[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5};
    size_t data_len = sizeof(data) / sizeof(data[0]);
    size_t n = 4;
    size_t out_len = 0;

    int *largest = get_n_largest(data, data_len, n, &out_len);

    if (largest != NULL) {
        for (size_t i = 0; i < out_len; i++) {
            printf("%d ", largest[i]);
        }
        printf("\n");
        free(largest);
    }

    return 0;
}
