#include <stdio.h>
#include <limits.h>
#include <stddef.h>

int minJumps(const int arr[], size_t n)
{
    if (arr == NULL || n == 0)
    {
        return -1;
    }

    if (n == 1)
    {
        return 0;
    }

    if (arr[0] <= 0)
    {
        return -1;
    }

    int jumps = 1;
    size_t maxReach = (size_t)arr[0];
    size_t steps = (size_t)arr[0];

    for (size_t i = 1; i < n; i++)
    {
        if (i == n - 1)
        {
            return jumps;
        }

        if (arr[i] > 0 && i + (size_t)arr[i] > maxReach)
        {
            maxReach = i + (size_t)arr[i];
        }

        steps--;

        if (steps == 0)
        {
            jumps++;

            if (i >= maxReach)
            {
                return -1;
            }

            steps = maxReach - i;
        }
    }

    return -1;
}

int main(void)
{
    int arr1[] = {1, 3, 5, 8, 9, 2, 6, 7, 6, 8, 9};
    size_t n1 = sizeof(arr1) / sizeof(arr1[0]);

    int arr2[] = {1, 1, 0, 2};
    size_t n2 = sizeof(arr2) / sizeof(arr2[0]);

    int arr3[] = {2, 3, 1, 1, 4};
    size_t n3 = sizeof(arr3) / sizeof(arr3[0]);

    int result1 = minJumps(arr1, n1);
    int result2 = minJumps(arr2, n2);
    int result3 = minJumps(arr3, n3);

    if (result1 >= 0)
    {
        if (printf("Minimum jumps for array 1: %d\n", result1) < 0)
        {
            return 1;
        }
    }
    else
    {
        if (printf("End of array 1 is not reachable\n") < 0)
        {
            return 1;
        }
    }

    if (result2 >= 0)
    {
        if (printf("Minimum jumps for array 2: %d\n", result2) < 0)
        {
            return 1;
        }
    }
    else
    {
        if (printf("End of array 2 is not reachable\n") < 0)
        {
            return 1;
        }
    }

    if (result3 >= 0)
    {
        if (printf("Minimum jumps for array 3: %d\n", result3) < 0)
        {
            return 1;
        }
    }
    else
    {
        if (printf("End of array 3 is not reachable\n") < 0)
        {
            return 1;
        }
    }

    return 0;
}
