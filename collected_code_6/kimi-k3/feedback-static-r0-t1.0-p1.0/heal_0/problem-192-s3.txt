```c
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int has_letter_and_number(const char *str) {
    int has_letter = 0;
    int has_number = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i])) {
            has_letter = 1;
        }
        if (isdigit((unsigned char)str[i])) {
            has_number = 1;
        }
        if (has_letter && has_number) {
            return 1;
        }
    }

    return 0;
}

int main(void) {
    const char *test_strings[] = {
        "hello123",
        "hello",
        "12345",
        "abc1",
        "",
        NULL
    };
    size_t i;

    for (i = 0; test_strings[i] != NULL; i++) {
        if (has_letter_and_number(test_strings[i])) {
            printf("\"%s\": has at least one letter and one number\n", test_strings[i]);
        } else {
            printf("\"%s\": does NOT have both\n", test_strings[i]);
        }
    }

    return 0;
}
```
