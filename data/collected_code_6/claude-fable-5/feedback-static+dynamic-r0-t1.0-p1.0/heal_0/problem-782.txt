```c
#include <stdio.h>
#include <stdlib.h>

long long sum_odd_length_subarrays(const int *arr, size_t n)
{
    long long total = 0;

    if (arr == NULL || n == 0) {
        return 0;
    }

    for (size_t i = 0; i < n; i++) {
        long long left = (long long)i + 1;
        long long right = (long long)(n - i);
        long long total_subarrays = left * right;
        long long odd_count = (total_subarrays + 1) / 2;
        total += odd_count * (long long)arr[i];
    }

    return total;
}

int main(void)
{
    int arr[] = {1, 4, 2, 5, 3};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result;

    result = sum_odd_length_subarrays(arr, n);

    if (printf("Sum of all odd length subarrays: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
