#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256

void remove_char(char *str, char c)
{
    size_t read_idx = 0;
    size_t write_idx = 0;

    if (str == NULL) {
        return;
    }

    while (str[read_idx] != '\0') {
        if (str[read_idx] != c) {
            str[write_idx] = str[read_idx];
            write_idx++;
        }
        read_idx++;
    }
    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    const char *source = "hello world, hello everyone";
    char target = 'l';
    size_t source_len;
    int written;

    source_len = strnlen(source, sizeof(buffer));
    if (source_len >= sizeof(buffer)) {
        fprintf(stderr, "Source string too long\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, sizeof(buffer), "%s", source);
    if (written < 0 || (size_t)written >= sizeof(buffer)) {
        fprintf(stderr, "Failed to copy source string safely\n");
        return EXIT_FAILURE;
    }

    printf("Original string: %s\n", buffer);
    remove_char(buffer, target);
    printf("After removing '%c': %s\n", target, buffer);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code already addresses the Flawfinder warning (CWE-119/CWE-120) regarding the statically-sized `buffer[BUFFER_SIZE]`. The mitigations in place are:

1. **Length validation before copy** — `strnlen(source, sizeof(buffer))` bounds the length check itself, and the result is compared against the buffer capacity, rejecting oversized input before any write occurs.

2. **Bounded copy with `snprintf`** — instead of unsafe `strcpy`, the copy uses `snprintf(buffer, sizeof(buffer), "%s", source)`, which guarantees null-termination and never writes past the buffer. The return value is checked for both encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof(buffer)`).

3. **NULL-guard in `remove_char`** — the helper checks `str == NULL` before dereferencing, and its in-place compaction algorithm only ever writes at or before the read index, so it cannot expand the string or overflow the buffer. The final null terminator is written at `write_idx`, which is always within the original string bounds.
