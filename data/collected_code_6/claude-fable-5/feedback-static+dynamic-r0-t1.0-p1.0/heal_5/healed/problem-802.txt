static int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        size_t mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else if (arr[mid] < arr[high]) {
            high = mid;
        } else {
            high--;
        }
    }

    return (int)low;
}

int main(void)
{
