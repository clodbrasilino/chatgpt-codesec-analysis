#include <stdio.h>
#include <stdlib.h>

void split_and_add(int *arr, int n, int k) {
    int *temp = (int *)malloc(n * sizeof(int));
    if (temp == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }
    
    k = k % n;
    
    for (int i = 0; i < n; i++) {
        temp[i] = arr[(i + k) % n];
    }
    
    for (int i = 0; i < n; i++) {
        arr[i] = temp[i];
    }
    
    free(temp);
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 2;
    
    split_and_add(arr, n, k);
    
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
