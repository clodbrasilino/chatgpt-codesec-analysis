#include<stdio.h>
#include<math.h>

int isDifferenceofSquares(int num) {
    int start = 0, end = (int)sqrt(num) + 1;
    while (start <= end) {
        int curr = start * start - end * end;
        if (curr == num)
            return 1;
        else if (curr < num)
            start++;
        else
            end--;
    }
    return 0;
}

int main() {
    int num;
    printf("\nEnter a number: ");
    scanf("%d", &num);
    if(isDifferenceofSquares(num)) {
        printf("\nThe given number %d can be expressed as difference of two squares\n", num);
    } else {
        printf("\nThe given number %d cannot be expressed as difference of two squares\n", num);
    }
    return 0;
}
