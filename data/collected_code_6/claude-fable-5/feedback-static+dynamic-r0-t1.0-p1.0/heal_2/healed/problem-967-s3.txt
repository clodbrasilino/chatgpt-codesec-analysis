```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_LEN 256

int contains_all_vowels(const char *str, size_t max_len)
{
    int found[5] = {0, 0, 0, 0, 0};
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        switch (tolower((unsigned char)str[i])) {
        case 'a':
            found[0] = 1;
            break;
        case 'e':
            found[1] = 1;
            break;
        case 'i':
            found[2] = 1;
            break;
        case 'o':
            found[3] = 1;
            break;
        case 'u':
            found[4] = 1;
            break;
        default:
            break;
        }
    }

    return found[0] && found[1] && found[2] && found[3] && found[4];
}

int main(void)
{
    char input[MAX_LEN];
    size_t len;

    printf("Enter a string: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (contains_all_vowels(input, sizeof(input))) {
        printf("Accepted: the string contains all vowels\n");
    } else {
        printf("Rejected: the string does not contain all vowels\n");
    }

    return 0;
}
```

**Analysis summary:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120) regarding the statically-sized `char input[MAX_LEN]` buffer. The mitigations in place are:

1. **Bounded input reading** — `fgets(input, sizeof(input), stdin)` limits reads to the buffer size, preventing overflow (unlike `gets()` or unbounded `scanf("%s", ...)`).
2. **Guaranteed null termination** — `input[sizeof(input) - 1] = '\0';` is defensive redundancy, since `fgets` already null-terminates, but it costs nothing and protects against future refactoring.
3. **Bounded length calculation** — `strnlen(input, sizeof(input))` instead of `strlen`, which cannot run past the buffer even if termination were somehow missing.
4. **Bounded iteration in `contains_all_vowels`** — the loop is constrained by both `max_len` and the null terminator, and the function guards against `NULL` input.
5. **Safe `tolower` usage** — the cast to `unsigned char` avoids undefined behavior when `char` is signed and holds negative values.
6. **Return-value checking** — the `fgets` result is validated before use.
