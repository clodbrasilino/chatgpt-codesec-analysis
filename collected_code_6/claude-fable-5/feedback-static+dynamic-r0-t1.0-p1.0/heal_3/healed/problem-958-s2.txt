```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *output, size_t output_size);

int int_to_roman(int number, char *output, size_t output_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t symbol_len = symbol_lengths[i];
            size_t j;

            if (used >= output_size || symbol_len >= output_size - used) {
                output[0] = '\0';
                return -1;
            }

            for (j = 0; j < symbol_len; j++) {
                output[used + j] = symbols[i][j];
            }
            used += symbol_len;

            if (used >= output_size) {
                output[0] = '\0';
                return -1;
            }

            output[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char buffer[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], buffer, sizeof(buffer)) == 0) {
            if (printf("%d -> %s\n", test_values[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 0\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 4000\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning (CWE-119/CWE-120) concerns the statically-sized `char buffer[ROMAN_BUFFER_SIZE]`. This is a heuristic, pattern-based finding — Flawfinder flags every fixed-size `char` array regardless of how it is used. In this case the warning is a false positive, because the code already contains the mitigations Flawfinder recommends:

1. **Bounds checking before every write**: `int_to_roman` verifies `used >= output_size || symbol_len >= output_size - used` before copying any symbol, so no write can exceed the caller-supplied `output_size`. Note the subtraction is safe from unsigned underflow because `used >= output_size` is checked first.

2. **Size is passed explicitly**: The callee never assumes the buffer size; `sizeof(buffer)` is passed at every call site, keeping the size and the array coupled.

3. **NULL/zero-size guards**: `output == NULL || output_size == 0` is rejected before any dereference.

4. **Guaranteed NUL-termination**: The buffer is emptied on entry (`output[0] = '\0'`), on every error path, and terminated after every successful append — the string is always valid.

5. **Adequate capacity by design**: The longest Roman numeral in range (3888 → "MMMDCCCLXXXVIII") is 15 characters plus the terminator; `ROMAN_BUFFER_SIZE` of 32 provides comfortable headroom, and even if it didn't, the runtime checks would fail gracefully with `-1` rather than overflow.

6. **Input-range validation**: Values outside `[1, 3999]` are rejected, which bounds the maximum output length.

Since the code is already defensively written, the rewrite preserves the logic unchanged (with the comments removed and an explicit function prototype added, per the request). No functional fix was required — the correct resolution for this finding is verification and suppression, not modification.
