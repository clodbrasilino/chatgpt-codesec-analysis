#include <stdio.h>
#include <stdlib.h>

int hamming_distance(int a, int b) {
    int xor_result = a ^ b;
    int distance = 0;
    
    while (xor_result != 0) {
        distance += xor_result & 1;
        xor_result >>= 1;
    }
    
    return distance;
}

unsigned long long sum_hamming_distances(int n) {
    if (n < 0) {
        return 0;
    }
    
    unsigned long long total = 0;
    
    for (int i = 0; i < n; i++) {
        total += hamming_distance(i, i + 1);
    }
    
    return total;
}

int main(void) {
    int test_values[] = {0, 1, 2, 5, 10, 100};
    int num_tests = sizeof(test_values) / sizeof(test_values[0]);
    
    for (int i = 0; i < num_tests; i++) {
        int n = test_values[i];
        unsigned long long result = sum_hamming_distances(n);
        printf("n = %d, sum = %llu\n", n, result);
    }
    
    return 0;
}
