```c
#include <stdio.h>
#include <stdlib.h>

int find_nth_non_multiple(int n, int m) {
    int count = 0;
    for (int i = 1; ; i++) {
        if (i % m != 0) {
            count++;
            if (count == n) {
                return i;
            }
        }
    }
}

int main() {
    int n = 10;
    int m = 3;
    int result = find_nth_non_multiple(n, m);
    printf("%d\n", result);
    return 0;
}
```
