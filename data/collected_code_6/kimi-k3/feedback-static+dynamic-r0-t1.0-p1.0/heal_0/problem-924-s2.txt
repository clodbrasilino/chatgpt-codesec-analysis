#include <stdio.h>
#include <limits.h>

int find_maximum(int first_number, int second_number) {
    if (first_number > second_number) {
        return first_number;
    }
    return second_number;
}

int main(void) {
    int first_number = 0;
    int second_number = 0;
    int result = 0;

    printf("Enter two integers: ");
    
    if (scanf("%d %d", &first_number, &second_number) != 2) {
        fprintf(stderr, "Error: Invalid input\n");
        return 1;
    }

    result = find_maximum(first_number, second_number);
    printf("Maximum: %d\n", result);

    return 0;
}
