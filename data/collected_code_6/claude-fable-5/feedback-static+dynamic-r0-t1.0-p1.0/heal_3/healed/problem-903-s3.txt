```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>

static unsigned int countUnsetBits(unsigned int num)
{
    unsigned int count = 0U;
    while (num > 0U)
    {
        if ((num & 1U) == 0U)
        {
            count++;
        }
        num >>= 1U;
    }
    return count;
}

static unsigned long long totalUnsetBits(unsigned int n)
{
    unsigned long long total = 0ULL;
    unsigned int i = 1U;
    for (;;)
    {
        total += (unsigned long long)countUnsetBits(i);
        if (i == n)
        {
            break;
        }
        i++;
    }
    return total;
}

int main(int argc, char *argv[])
{
    unsigned int n = 0U;

    if (argc > 1)
    {
        char *end = NULL;
        unsigned long parsed;
        errno = 0;
        parsed = strtoul(argv[1], &end, 10);
        if (errno != 0 || end == argv[1] || *end != '\0' || parsed < 1UL || parsed > (unsigned long)UINT_MAX)
        {
            fprintf(stderr, "Error: invalid input\n");
            return EXIT_FAILURE;
        }
        n = (unsigned int)parsed;
    }
    else
    {
        if (scanf("%u", &n) != 1)
        {
            fprintf(stderr, "Error: invalid input\n");
            return EXIT_FAILURE;
        }
        if (n < 1U)
        {
            fprintf(stderr, "Error: input must be at least 1\n");
            return EXIT_FAILURE;
        }
    }

    printf("%llu\n", totalUnsetBits(n));

    return EXIT_SUCCESS;
}
```

Key points of the analysis and fixes:

1. **Test case failures (expected 4/17, got 1):** Those failures describe an earlier, buggy revision that printed `countUnsetBits(n)` for the single value `n` (e.g., `countUnsetBits(5)` = 1, `countUnsetBits(14)` = 1) instead of the cumulative total. The correct behavior is summing unset bits across `1..n` (`n=5` → 4, `n=14` → 17), which `totalUnsetBits` now provides — verified: 1→0, 2→1, 3→0, 4→2, 5→1, summing to 4.

2. **`variableScope` / `unreadVariable`:** `parsed` was declared and initialized at function scope but only used inside the `argc > 1` branch, and its `0UL` initializer was dead. It is now declared uninitialized inside that block, eliminating both warnings.

3. **`constParameter` on `argv`:** Left as `char *argv[]` because `main`'s signature is mandated by the C standard (C11 §5.1.2.2.1); changing it to a const-qualified form is non-conforming and can break hosted-environment guarantees. This cppcheck hint is a false positive for `main` specifically.

4. **Robustness retained:** `strtoul` parsing with `errno`, `end` pointer, trailing-character, and range checks (`1..UINT_MAX`) is preserved, as is the `scanf` fallback path, and the `for(;;)`/`i == n` loop structure avoids infinite looping when `n == UINT_MAX`.
