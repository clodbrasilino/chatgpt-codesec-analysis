#include <stdio.h>
#include <stdlib.h>

unsigned long long jacobsthal_lucas(int n) {
    if (n < 0) return 0;
    if (n == 0) return 2;
    if (n == 1) return 1;
    
    unsigned long long prev2 = 2;
    unsigned long long prev1 = 1;
    unsigned long long current = 0;
    
    for (int i = 2; i <= n; i++) {
        current = prev1 + 2 * prev2;
        prev2 = prev1;
        prev1 = current;
    }
    
    return current;
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <n>\n", argv[0]);
        return EXIT_FAILURE;
    }
    
    char *endptr;
    long n = strtol(argv[1], &endptr, 10);
    
    if (*endptr != '\0' || n < 0 || n > 93) {
        fprintf(stderr, "Invalid input. Please provide a non-negative integer up to 93.\n");
        return EXIT_FAILURE;
    }
    
    printf("%llu\n", jacobsthal_lucas((int)n));
    
    return EXIT_SUCCESS;
}
