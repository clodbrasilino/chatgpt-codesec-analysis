#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool is_divisible_by_digits(int num) {
    if (num == 0) {
        return false;
    }
    
    int temp = num;
    if (temp < 0) {
        temp = -temp;
    }
    
    while (temp > 0) {
        int digit = temp % 10;
        if (digit == 0 || num % digit != 0) {
            return false;
        }
        temp /= 10;
    }
    return true;
}

void find_numbers_in_range(int start, int end, int **result, int *count) {
    if (count == NULL || result == NULL) {
        return;
    }

    if (start > end) {
        int temp = start;
        start = end;
        end = temp;
    }

    *count = 0;
    int capacity = 10;
    *result = (int *)malloc(capacity * sizeof(int));
    if (*result == NULL) {
        return;
    }

    for (int i = start; i <= end; ++i) {
        if (is_divisible_by_digits(i)) {
            if (*count >= capacity) {
                capacity *= 2;
                int *new_result = (int *)realloc(*result, capacity * sizeof(int));
                if (new_result == NULL) {
                    free(*result);
                    *result = NULL;
                    *count = 0;
                    return;
                }
                *result = new_result;
            }
            (*result)[*count] = i;
            (*count)++;
        }
    }
}

int main(void) {
    int start = 1;
    int end = 100;
    int *numbers = NULL;
    int count = 0;

    find_numbers_in_range(start, end, &numbers, &count);

    if (numbers != NULL) {
        for (int i = 0; i < count; ++i) {
            printf("%d ", numbers[i]);
        }
        printf("\n");
        free(numbers);
        numbers = NULL;
    } else {
        printf("Memory allocation failed or invalid input.\n");
    }

    return 0;
}
