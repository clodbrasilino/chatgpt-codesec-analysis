#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct KeyValuePair {
    char *key;
    char *value;
    struct KeyValuePair *next;
} KeyValuePair;

typedef struct {
    KeyValuePair **buckets;
    size_t num_buckets;
} Dictionary;

static size_t hash_string(const char *str, size_t num_buckets) {
    size_t hash = 5381;
    int c;
    while ((c = *str++)) {
        hash = ((hash << 5) + hash) + c;
    }
    return hash % num_buckets;
}

static char *duplicate_string(const char *src) {
    if (!src) {
        return NULL;
    }
    size_t len = strlen(src) + 1;
    char *dest = malloc(len);
    if (!dest) {
        return NULL;
    }
    memcpy(dest, src, len);
    return dest;
}

Dictionary *create_dictionary(size_t num_buckets) {
    if (num_buckets == 0) {
        return NULL;
    }
    Dictionary *dict = malloc(sizeof(Dictionary));
    if (!dict) {
        return NULL;
    }
    dict->buckets = calloc(num_buckets, sizeof(KeyValuePair *));
    if (!dict->buckets) {
        free(dict);
        return NULL;
    }
    dict->num_buckets = num_buckets;
    return dict;
}

void free_dictionary(Dictionary *dict) {
    if (!dict) {
        return;
    }
    for (size_t i = 0; i < dict->num_buckets; ++i) {
        KeyValuePair *current = dict->buckets[i];
        while (current) {
            KeyValuePair *next = current->next;
            free(current->key);
            free(current->value);
            free(current);
            current = next;
        }
    }
    free(dict->buckets);
    free(dict);
}

int insert_dictionary(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) {
        return -1;
    }
    size_t index = hash_string(key, dict->num_buckets);
    KeyValuePair *current = dict->buckets[index];

    while (current) {
        if (strcmp(current->key, key) == 0) {
            char *new_value = duplicate_string(value);
            if (!new_value) {
                return -1;
            }
            free(current->value);
            current->value = new_value;
            return 0;
        }
        current = current->next;
    }

    KeyValuePair *new_pair = malloc(sizeof(KeyValuePair));
    if (!new_pair) {
        return -1;
    }
    new_pair->key = duplicate_string(key);
    if (!new_pair->key) {
        free(new_pair);
        return -1;
    }
    new_pair->value = duplicate_string(value);
    if (!new_pair->value) {
        free(new_pair->key);
        free(new_pair);
        return -1;
    }
    new_pair->next = dict->buckets[index];
    dict->buckets[index] = new_pair;
    return 0;
}

Dictionary *merge_dictionaries(const Dictionary *dict1, const Dictionary *dict2) {
    if (!dict1 || !dict2) {
        return NULL;
    }
    size_t new_size = dict1->num_buckets + dict2->num_buckets;
    Dictionary *merged = create_dictionary(new_size);
    if (!merged) {
        return NULL;
    }

    for (size_t i = 0; i < dict1->num_buckets; ++i) {
        KeyValuePair *current = dict1->buckets[i];
        while (current) {
            if (insert_dictionary(merged, current->key, current->value) != 0) {
                free_dictionary(merged);
                return NULL;
            }
            current = current->next;
        }
    }

    for (size_t i = 0; i < dict2->num_buckets; ++i) {
        KeyValuePair *current = dict2->buckets[i];
        while (current) {
            if (insert_dictionary(merged, current->key, current->value) != 0) {
                free_dictionary(merged);
                return NULL;
            }
            current = current->next;
        }
    }

    return merged;
}

void print_dictionary(const Dictionary *dict) {
    if (!dict) {
        return;
    }
    for (size_t i = 0; i < dict->num_buckets; ++i) {
        KeyValuePair *current = dict->buckets[i];
        while (current) {
            printf("%s: %s\n", current->key, current->value);
            current = current->next;
        }
    }
}

int main(void) {
    Dictionary *dict1 = create_dictionary(16);
    if (!dict1) {
        return EXIT_FAILURE;
    }

    Dictionary *dict2 = create_dictionary(16);
    if (!dict2) {
        free_dictionary(dict1);
        return EXIT_FAILURE;
    }

    if (insert_dictionary(dict1, "host", "127.0.0.1") != 0 ||
        insert_dictionary(dict1, "port", "8080") != 0 ||
        insert_dictionary(dict2, "user", "admin") != 0 ||
        insert_dictionary(dict2, "port", "443") != 0) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return EXIT_FAILURE;
    }

    Dictionary *merged = merge_dictionaries(dict1, dict2);
    if (!merged) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return EXIT_FAILURE;
    }

    print_dictionary(merged);

    free_dictionary(dict1);
    free_dictionary(dict2);
    free_dictionary(merged);

    return EXIT_SUCCESS;
}
