```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_WORD_LEN 64
#define MAX_SENTENCE_LEN 4096

typedef struct {
    char word[MAX_WORD_LEN];
    size_t position;
} AdverbInfo;

size_t find_adverbs(const char *sentence, AdverbInfo *results, size_t max_results)
{
    size_t count = 0;
    size_t i = 0;
    size_t len;

    if (sentence == NULL || results == NULL || max_results == 0) {
        return 0;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);

    while (i < len && count < max_results) {
        while (i < len && !isalpha((unsigned char)sentence[i])) {
            i++;
        }

        if (i < len) {
            size_t start = i;
            size_t word_len;

            while (i < len && isalpha((unsigned char)sentence[i])) {
                i++;
            }

            word_len = i - start;

            if (word_len >= 3 && word_len < sizeof(results[count].word)) {
                if (tolower((unsigned char)sentence[i - 2]) == 'l' &&
                    tolower((unsigned char)sentence[i - 1]) == 'y') {
                    size_t copy_len = word_len;

                    if (copy_len >= sizeof(results[count].word)) {
                        copy_len = sizeof(results[count].word) - 1;
                    }

                    memcpy(results[count].word, &sentence[start], copy_len);
                    results[count].word[copy_len] = '\0';
                    results[count].position = start;
                    count++;
                }
            }
        }
    }

    return count;
}

int main(void)
{
    const char *sentence = "She quickly and quietly walked home, then suddenly stopped.";
    AdverbInfo *results;
    size_t count;
    const size_t max_results = 32;

    results = calloc(max_results, sizeof(AdverbInfo));
    if (results == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    count = find_adverbs(sentence, results, max_results);

    if (count == 0) {
        printf("No adverbs found.\n");
    } else {
        printf("Found %zu adverb(s):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("  \"%s\" at position %zu\n", results[i].word, results[i].position);
        }
    }

    free(results);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **CWE-119/CWE-120 (statically-sized array `word`):** The fixed-size buffer is safe here because every write path is bounds-checked. The condition `word_len < sizeof(results[count].word)` guarantees the word plus its NUL terminator fits within `MAX_WORD_LEN`, and the secondary clamp on `copy_len` provides defense-in-depth.

2. **CWE-120 (`memcpy` without overflow check):** This is a false positive in practice — the destination size is verified before the copy. `copy_len` can never exceed `sizeof(results[count].word) - 1`, so `memcpy` plus the explicit NUL termination at `word[copy_len]` always stays within bounds.

Additional defensive measures already present and preserved:

- `strnlen` with `MAX_SENTENCE_LEN` bounds the scan even for non-terminated input.
- All `ctype.h` calls cast to `unsigned char`, avoiding UB on negative `char` values.
- NULL-pointer and zero-capacity guards at function entry.
- `calloc` result checked before use; memory freed before exit.
