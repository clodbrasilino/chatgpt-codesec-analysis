#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

int count_lists(Node **lists, int num_lists) {
    if (lists == NULL || num_lists <= 0) {
        return 0;
    }

    int count = 0;
    for (int i = 0; i < num_lists; i++) {
        if (lists[i] != NULL) {
            count++;
        }
    }
    return count;
}

Node *create_node(int data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int main(void) {
    int num_lists = 5;
    Node **lists = (Node **)malloc(num_lists * sizeof(Node *));
    if (lists == NULL) {
        return 1;
    }

    lists[0] = create_node(1);
    lists[1] = NULL;
    lists[2] = create_node(3);
    lists[3] = NULL;
    lists[4] = create_node(5);

    if (lists[0] != NULL) {
        Node *second = create_node(2);
        if (second != NULL) {
            lists[0]->next = second;
        }
    }

    int result = count_lists(lists, num_lists);
    printf("%d\n", result);

    for (int i = 0; i < num_lists; i++) {
        free_list(lists[i]);
    }
    free(lists);

    return 0;
}
