#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    int value;
} KeyValuePair;

typedef struct {
    KeyValuePair *pairs;
    size_t size;
    size_t capacity;
} Dictionary;

Dictionary *dictionary_create(size_t capacity) {
    Dictionary *dict = malloc(sizeof(Dictionary));
    if (dict == NULL) {
        return NULL;
    }
    dict->pairs = malloc(sizeof(KeyValuePair) * capacity);
    if (dict->pairs == NULL) {
        free(dict);
        return NULL;
    }
    dict->size = 0;
    dict->capacity = capacity;
    return dict;
}

int dictionary_insert(Dictionary *dict, const char *key, int value) {
    if (dict == NULL || key == NULL) {
        return -1;
    }
    if (dict->size >= dict->capacity) {
        size_t new_capacity = dict->capacity * 2;
        KeyValuePair *new_pairs = realloc(dict->pairs, sizeof(KeyValuePair) * new_capacity);
        if (new_pairs == NULL) {
            return -1;
        }
        dict->pairs = new_pairs;
        dict->capacity = new_capacity;
    }
    char *key_copy = strdup(key);
    if (key_copy == NULL) {
        return -1;
    }
    dict->pairs[dict->size].key = key_copy;
    dict->pairs[dict->size].value = value;
    dict->size++;
    return 0;
}

int dictionary_get_by_index(Dictionary *dict, size_t index, char **key, int *value) {
    if (dict == NULL || key == NULL || value == NULL) {
        return -1;
    }
    if (index >= dict->size) {
        return -1;
    }
    *key = dict->pairs[index].key;
    *value = dict->pairs[index].value;
    return 0;
}

void dictionary_destroy(Dictionary *dict) {
    if (dict == NULL) {
        return;
    }
    for (size_t i = 0; i < dict->size; i++) {
        free(dict->pairs[i].key);
    }
    free(dict->pairs);
    free(dict);
}

int main(void) {
    Dictionary *dict = dictionary_create(4);
    if (dict == NULL) {
        return EXIT_FAILURE;
    }
    if (dictionary_insert(dict, "apple", 1) != 0) {
        dictionary_destroy(dict);
        return EXIT_FAILURE;
    }
    if (dictionary_insert(dict, "banana", 2) != 0) {
        dictionary_destroy(dict);
        return EXIT_FAILURE;
    }
    if (dictionary_insert(dict, "cherry", 3) != 0) {
        dictionary_destroy(dict);
        return EXIT_FAILURE;
    }
    if (dictionary_insert(dict, "date", 4) != 0) {
        dictionary_destroy(dict);
        return EXIT_FAILURE;
    }
    if (dictionary_insert(dict, "elderberry", 5) != 0) {
        dictionary_destroy(dict);
        return EXIT_FAILURE;
    }
    char *key = NULL;
    int value = 0;
    for (size_t i = 0; i < dict->size; i++) {
        if (dictionary_get_by_index(dict, i, &key, &value) == 0) {
            printf("[%zu] key: %s, value: %d\n", i, key, value);
        }
    }
    if (dictionary_get_by_index(dict, 10, &key, &value) != 0) {
        printf("Index 10 is out of bounds\n");
    }
    dictionary_destroy(dict);
    return EXIT_SUCCESS;
}
