```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

int parse_tuple(const char *input, long **result, size_t *count)
{
    const char *p;
    long *values;
    long *tmp;
    size_t capacity;
    size_t n;
    char *endptr;

    if (input == NULL || result == NULL || count == NULL) {
        return -1;
    }

    *result = NULL;
    *count = 0;

    p = input;
    while (isspace((unsigned char)*p)) {
        p++;
    }

    if (*p != '(') {
        return -1;
    }
    p++;

    capacity = 4;
    values = malloc(capacity * sizeof(*values));
    if (values == NULL) {
        return -1;
    }

    n = 0;

    while (isspace((unsigned char)*p)) {
        p++;
    }

    if (*p == ')') {
        p++;
        while (isspace((unsigned char)*p)) {
            p++;
        }
        if (*p != '\0') {
            free(values);
            return -1;
        }
        *result = values;
        *count = 0;
        return 0;
    }

    for (;;) {
        long val;

        while (isspace((unsigned char)*p)) {
            p++;
        }

        errno = 0;
        val = strtol(p, &endptr, 10);
        if (endptr == p || errno == ERANGE) {
            free(values);
            return -1;
        }
        p = endptr;

        if (n == capacity) {
            if (capacity > SIZE_MAX / (2 * sizeof(*values))) {
                free(values);
                return -1;
            }
            capacity *= 2;
            tmp = realloc(values, capacity * sizeof(*values));
            if (tmp == NULL) {
                free(values);
                return -1;
            }
            values = tmp;
        }
        values[n] = val;
        n++;

        while (isspace((unsigned char)*p)) {
            p++;
        }

        if (*p == ',') {
            p++;
            continue;
        }

        if (*p == ')') {
            p++;
            break;
        }

        free(values);
        return -1;
    }

    while (isspace((unsigned char)*p)) {
        p++;
    }

    if (*p != '\0') {
        free(values);
        return -1;
    }

    *result = values;
    *count = n;
    return 0;
}

int main(void)
{
    const char *tests[] = {
        "(1, 2, 3)",
        "( 10 , -20 , 30 )",
        "(42)",
        "()",
        "(1, 2,",
        "not a tuple"
    };
    size_t num_tests;
    size_t i;
    size_t j;
    long *values;
    size_t count;

    num_tests = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < num_tests; i++) {
        values = NULL;
        count = 0;
        if (parse_tuple(tests[i], &values, &count) == 0) {
            printf("Input: %s -> Parsed %zu values: ", tests[i], count);
            for (j = 0; j < count; j++) {
                printf("%ld", values[j]);
                if (j + 1 < count) {
                    printf(", ");
                }
            }
            printf("\n");
            free(values);
        } else {
            printf("Input: %s -> Parse error\n", tests[i]);
        }
    }

    return 0;
}
```

The key changes made to address the identified weakness:

1. **Reduced scope of `val`** — The variable `val` was declared at function scope but only used inside the parsing loop. It is now declared inside the `for (;;)` loop body, minimizing its scope and making the code easier to reason about.

2. **Added `<stdint.h>`** — The original code used `SIZE_MAX`, which is defined in `<stdint.h>`, not `<limits.h>`. Relying on it being available transitively is non-portable, so the proper header is now explicitly included.

The rest of the code was already sound: all allocation failures are handled, the `realloc` pattern correctly uses a temporary pointer to avoid leaks, the overflow check before doubling `capacity` prevents integer overflow in the size calculation, `strtol` errors are detected via both `endptr` and `errno == ERANGE`, and every error path frees the allocated buffer.
