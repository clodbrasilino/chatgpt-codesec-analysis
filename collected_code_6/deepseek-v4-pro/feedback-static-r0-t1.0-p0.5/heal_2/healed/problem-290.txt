#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
    int length;
} List;

int list_length(const List* list) {
    if (list == NULL) {
        return 0;
    }
    return list->length;
}

List** find_max_length_lists(List** lists, int num_lists, int* result_count) {
    if (lists == NULL || num_lists <= 0 || result_count == NULL) {
        return NULL;
    }

    int max_length = 0;
    int count = 0;
    int i;

    for (i = 0; i < num_lists; i++) {
        int current_length = list_length(lists[i]);
        if (current_length > max_length) {
            max_length = current_length;
            count = 1;
        } else if (current_length == max_length) {
            count++;
        }
    }

    if (count == 0) {
        *result_count = 0;
        return NULL;
    }

    List** result = (List**)malloc(count * sizeof(List*));
    if (result == NULL) {
        *result_count = 0;
        return NULL;
    }

    int index = 0;
    for (i = 0; i < num_lists; i++) {
        if (list_length(lists[i]) == max_length) {
            result[index++] = lists[i];
        }
    }

    *result_count = count;
    return result;
}

Node* create_node(int data) {
    Node* node = (Node*)malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

List* create_list(const int* arr, int size) {
    List* list = (List*)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->length = 0;

    if (size == 0 || arr == NULL) {
        return list;
    }

    Node* current = NULL;
    for (int i = 0; i < size; i++) {
        Node* node = create_node(arr[i]);
        if (node == NULL) {
            Node* temp = list->head;
            while (temp != NULL) {
                Node* next = temp->next;
                free(temp);
                temp = next;
            }
            free(list);
            return NULL;
        }
        if (list->head == NULL) {
            list->head = node;
            current = node;
        } else {
            current->next = node;
            current = node;
        }
        list->length++;
    }
    return list;
}

void free_list(List* list) {
    if (list == NULL) {
        return;
    }
    Node* current = list->head;
    while (current != NULL) {
        Node* next = current->next;
        free(current);
        current = next;
    }
    free(list);
}

int main() {
    const int arr1[] = {1, 2, 3};
    const int arr2[] = {4, 5, 6, 7};
    const int arr3[] = {8, 9};
    const int arr4[] = {10, 11, 12, 13};

    List* list1 = create_list(arr1, 3);
    List* list2 = create_list(arr2, 4);
    List* list3 = create_list(arr3, 2);
    List* list4 = create_list(arr4, 4);

    if (list1 == NULL || list2 == NULL || list3 == NULL || list4 == NULL) {
        free_list(list1);
        free_list(list2);
        free_list(list3);
        free_list(list4);
        return 1;
    }

    List* lists[] = {list1, list2, list3, list4};
    int result_count = 0;

    List** max_lists = find_max_length_lists(lists, 4, &result_count);

    if (max_lists != NULL && result_count > 0) {
        printf("Maximum length: %d\n", list_length(max_lists[0]));
        printf("Number of lists with maximum length: %d\n", result_count);
        for (int i = 0; i < result_count; i++) {
            printf("List %d: ", i + 1);
            Node* current = max_lists[i]->head;
            while (current != NULL) {
                printf("%d ", current->data);
                current = current->next;
            }
            printf("\n");
        }
        free(max_lists);
    }

    free_list(list1);
    free_list(list2);
    free_list(list3);
    free_list(list4);

    return 0;
}
