int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    size_t i;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = strnlen(word, WORD_MAX);
    if (word_len == 0U || word_len >= WORD_MAX) {
        return 0;
    }

    i = 0U;

        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               !isalnum((unsigned char)sentence[i])) {
            i++;
        }

            break;

            i++;

            return 1;
        }
    }

    return 0;
}

2. **Guaranteed termination:** `memset` zero-initializes both buffers, and an explicit `buf[sizeof(buf) - 1U] = '\0'` provides defense-in-depth even though `fgets` already NUL-terminates.
3. **Safe newline stripping:** `strcspn(buf, "\n")` is guaranteed to return an index within the buffer since the string is NUL-terminated, so the write is in-bounds.
