#include <stdio.h>
#include <stdint.h>

int last_digit_factorial(int n) {
    if (n == 0 || n == 1) {
        return 1;
    }
    int64_t count = 0, i;
    for (i = 5; n / i >= 1; i *= 5) {
        count += n / i;
    }
    if (count > 1) {
        return 0;
    } else if (count == 1) {
        return 5;
    } else {
        n %= 10;
        switch (n) {
            case 2: return 2;
            case 3: return 6;
            case 4: return 4;
            case 6: return 6;
            case 7: return 2;
            case 8: return 4;
            case 9: return 8;
            default: return 1;
        }
    }
}

int main() {
    int n;
    scanf("%d", &n);
    printf("%d\n", last_digit_factorial(n));
    return 0;
}
