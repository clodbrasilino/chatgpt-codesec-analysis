#include<stdio.h>

int findOddOccurrence(int arr[], int size) {
    int res = 0;
    for(int i = 0; i < size; i++){
        res ^= arr[i];
    }
    return res;
}

int main(){
    int arr[] = {2, 3, 5, 4, 5, 2, 4, 3, 5, 2, 4, 4, 2};
    int size = sizeof(arr)/sizeof(arr[0]);
    printf("%d", findOddOccurrence(arr, size));
    return 0;
}
