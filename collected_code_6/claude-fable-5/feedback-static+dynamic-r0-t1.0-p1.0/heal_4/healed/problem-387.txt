static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static int is_valid_hex(const char *str, size_t max_len)
{
    size_t i = 0;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = bounded_strlen(str, max_len);

    if (len == 0 || len >= max_len) {
        return 0;
    }

    if (len > 2 && str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        i = 2;
    }

    if (i >= len) {
        return 0;
    }

    for (; i < len; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

static int is_hex_even(const char *hex, size_t max_len, int *result)
{
    size_t len;
    char last;

    if (hex == NULL || result == NULL) {
        return -1;
    }

    if (!is_valid_hex(hex, max_len)) {
        return -1;
    }

    len = bounded_strlen(hex, max_len);

    if (len == 0) {
        return -1;
    }

    last = hex[len - 1];

    if (last >= '0' && last <= '9') {
        *result = ((last - '0') % 2 == 0) ? 1 : 0;
    } else {
        last = (char)tolower((unsigned char)last);
        if (last < 'a' || last > 'f') {
            return -1;
        }
        *result = ((last - 'a' + 10) % 2 == 0) ? 1 : 0;
    }

    return 0;
}

static void flush_stdin(void)
{
    int c;
    size_t guard = 0;
    const size_t guard_limit = 1000000;

    do {
        c = getchar();
        guard++;
    } while (c != '\n' && c != EOF && guard < guard_limit);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = bounded_strlen(buffer, size);

    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        return 0;
    }

    if (len == size - 1) {
        flush_stdin();
        return -2;
    }

    return 0;
}

{
    char buffer[BUFFER_SIZE];
    int even = 0;
    int status;

    memset(buffer, 0, sizeof(buffer));

        return EXIT_FAILURE;

        return EXIT_FAILURE;

    } else {

**Analysis of the flagged weaknesses and how they are mitigated:**

1. **`getchar` in a loop (CWE-120, CWE-20)** — The `flush_stdin` function reads into no buffer at all; each character is discarded, so no overflow is possible. The `guard_limit` counter additionally prevents an unbounded loop on pathological input streams. This finding is a false positive, and the guard makes the loop provably terminating.

2. **Statically-sized array `buffer` (CWE-119/CWE-120)** — All reads into `buffer` go through `read_line`, which uses `fgets` with the exact buffer size, forcibly NUL-terminates at `size - 1`, detects and rejects over-long input (returning `-2` and draining stdin), and validates `size` bounds including a cast-safety check against `INT_MAX` before converting to `int` for `fgets`. All subsequent string operations use `bounded_strlen` rather than `strlen`, so they can never scan past the buffer even if termination were somehow lost. This finding is also mitigated; the static array is safe as used.

The remaining changes (`fflush(stdout)` after the prompt, the `INT_MAX` check) are hardening improvements rather than fixes for exploitable defects — the original code was already sound with respect to both Flawfinder hits.
