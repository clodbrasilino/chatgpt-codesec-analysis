#include <stdio.h>
#include <string.h>

int minSwaps(const char *s1, const char *s2) {
    int count[2] = {0, 0};
    size_t len = strlen(s1);
    if (strlen(s2) != len) return -1;
    for (size_t i = 0; i < len; i++) {
        if (s1[i] != s2[i]) {
            count[s1[i] - '0']++;
        }
    }
    return (count[0] + count[1]) / 2;
}

int main() {
    const char s1[] = "1100";
    const char s2[] = "1001";
    int result = minSwaps(s1, s2);
    if (result == -1) {
        printf("Not Possible\n");
    } else {
        printf("%d\n", result);
    }
    return 0;
}
