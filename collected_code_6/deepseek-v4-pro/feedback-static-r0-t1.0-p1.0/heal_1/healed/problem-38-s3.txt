#include <stdio.h>

float find_division(const int arr[], int size) {
    int first_even = 0, first_odd = 0;
    int found_even = 0, found_odd = 0;
    
    for (int i = 0; i < size; i++) {
        if (arr[i] % 2 == 0 && !found_even) {
            first_even = arr[i];
            found_even = 1;
        } else if (arr[i] % 2 != 0 && !found_odd) {
            first_odd = arr[i];
            found_odd = 1;
        }
        
        if (found_even && found_odd) {
            break;
        }
    }
    
    if (!found_even || !found_odd) {
        return -1.0f;
    }
    
    if (first_odd == 0) {
        return -1.0f;
    }
    
    return (float)first_even / first_odd;
}

int main() {
    int arr[] = {3, 5, 7, 2, 4, 6};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    float result = find_division(arr, size);
    
    if (result == -1.0f) {
        printf("Could not find both even and odd numbers\n");
    } else {
        printf("Result: %.2f\n", result);
    }
    
    int arr2[] = {1, 3, 5, 7};
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    
    result = find_division(arr2, size2);
    
    if (result == -1.0f) {
        printf("Could not find both even and odd numbers\n");
    } else {
        printf("Result: %.2f\n", result);
    }
    
    int arr3[] = {2, 4, 6, 8};
    int size3 = sizeof(arr3) / sizeof(arr3[0]);
    
    result = find_division(arr3, size3);
    
    if (result == -1.0f) {
        printf("Could not find both even and odd numbers\n");
    } else {
        printf("Result: %.2f\n", result);
    }
    
    return 0;
}
