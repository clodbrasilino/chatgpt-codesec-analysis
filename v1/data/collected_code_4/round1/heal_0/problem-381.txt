```c
#include <stdio.h>
#include <stdlib.h>

#define ROW 5
#define COL 3

int colInd; 

int compare(const void* a, const void* b) 
{ 
    int* row1 = (int*)a; 
    int* row2 = (int*)b; 
    return row1[colInd] - row2[colInd]; 
} 
  
void sort2DArray(int arr[ROW][COL], int col) 
{ 
    colInd = col; 
    qsort((void*)arr, ROW, COL*sizeof(int), compare); 
} 

int main() 
{ 
    int arr[ROW][COL] = { {3, 2, 1}, {5, 6, 9}, {8, 9, 7}, {7, 16, 14}, {2, 0, 1}}; 
    int col = 2; 

    sort2DArray(arr, col); 

    for (int i=0; i<ROW; i++) 
    { 
        for (int j=0; j<COL; j++) 
            printf("%d ", arr[i][j]); 
        printf("\n"); 
    } 
    return 0;
}
```
