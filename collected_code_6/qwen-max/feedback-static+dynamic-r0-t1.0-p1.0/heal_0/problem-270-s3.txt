```c
#include <stdio.h>
#include <stdlib.h>

int sum_even_at_even_positions(int *array, int size) {
    int sum = 0;
    for (int i = 0; i < size; i += 2) {
        if (array[i] % 2 == 0) {
            sum += array[i];
        }
    }
    return sum;
}

int main() {
    int arr[] = {2, 3, 4, 5, 6, 7, 8};
    int size = sizeof(arr) / sizeof(arr[0]);
    int result = sum_even_at_even_positions(arr, size);
    printf("Sum of even numbers at even positions: %d\n", result);
    return 0;
}
```
