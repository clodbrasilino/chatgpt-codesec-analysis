#include <stdio.h>

int findMissingPositive(int arr[], int length) {
    if(length <= 0) {
        return 1;
    }
    
    for(int i = 0; i < length; ++i) {
        while(arr[i] > 0 && arr[i] <= length && arr[arr[i] - 1] != arr[i]) {
            int temp = arr[i];
            arr[i] = arr[temp - 1];
            arr[temp - 1] = temp;
         }
    }
    
    for(int i = 0; i < length; ++i) {
        if(arr[i] != i + 1) {
            return i + 1;
        }
    }
    
    return length + 1;
}

int main() {
    int arr[] = {3, 4, -1, 1};
    int arr_length = sizeof(arr) / sizeof(arr[0]);
    int result = findMissingPositive(arr, arr_length);
    
    printf("The first missing positive number is %d\n", result);
    
    return 0;
}
