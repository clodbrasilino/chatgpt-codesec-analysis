#include <stdio.h>
#include <stdlib.h>

int newman_conway(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n == 1 || n == 2) {
        return 1;
    }

    int *sequence = (int *)malloc((n + 1) * sizeof(int));
    if (sequence == NULL) {
        return -1;
    }

    sequence[1] = 1;
    sequence[2] = 1;

    for (int i = 3; i <= n; i++) {
        sequence[i] = sequence[sequence[i - 1]] + sequence[i - sequence[i - 1]];
    }

    int result = sequence[n];
    free(sequence);

    return result;
}

int main() {
    int n;
    
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }

    int result = newman_conway(n);
    if (result == -1) {
        return EXIT_FAILURE;
    }

    printf("%d\n", result);

    return EXIT_SUCCESS;
}
