#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void left_rotate(char *str, size_t len, int d) {
    d = (d % (int)len + len) % len;
    if (d == 0) return;

    char *temp = (char *)malloc(d + 1);
    if (temp == NULL) return;

    memcpy(temp, str, d);
    temp[d] = '\0';

    memmove(str, str + d, len - d + 1);
    strncat(str, temp, d);

    free(temp);
}

int main() {
    char str[] = "abcdef";
    int d = 2;
    left_rotate(str, sizeof(str) / sizeof(str[0]) - 1, d);
    printf("%s\n", str);
    return 0;
}
