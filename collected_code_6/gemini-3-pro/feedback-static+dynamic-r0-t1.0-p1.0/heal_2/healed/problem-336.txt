#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_equal_case_insensitive(const char *str, const char *target) {
    while (*str && *target) {
        if (tolower((unsigned char)*str) != tolower((unsigned char)*target)) {
            return 0;
        }
        str++;
        target++;
    }
    return *str == '\0' && *target == '\0';
}

int has_28_days(const char *month) {
    if (month == NULL) {
        return 0;
    }

    if (is_equal_case_insensitive(month, "february") || 
        is_equal_case_insensitive(month, "feb")) {
        return 1;
    }

    return 0;
}

int main(void) {
    const char *test_months[] = {"February", "March", "Feb", "january", NULL};
    
    for (int i = 0; i < 5; i++) {
        if (test_months[i] != NULL) {
            if (has_28_days(test_months[i])) {
                printf("%s has 28 days.\n", test_months[i]);
            } else {
                printf("%s does not have 28 days.\n", test_months[i]);
            }
        }
    }

    return 0;
}
