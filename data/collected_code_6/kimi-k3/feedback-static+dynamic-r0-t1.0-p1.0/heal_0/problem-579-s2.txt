#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *data;
    size_t size;
} Tuple;

int tuple_init(Tuple *t, size_t size) {
    if (t == NULL) {
        return -1;
    }
    if (size == 0) {
        t->data = NULL;
        t->size = 0;
        return 0;
    }
    t->data = malloc(size * sizeof(int));
    if (t->data == NULL) {
        return -1;
    }
    t->size = size;
    return 0;
}

void tuple_free(Tuple *t) {
    if (t != NULL) {
        free(t->data);
        t->data = NULL;
        t->size = 0;
    }
}

int tuple_contains(const Tuple *t, int value) {
    size_t i;
    if (t == NULL || t->data == NULL) {
        return 0;
    }
    for (i = 0; i < t->size; i++) {
        if (t->data[i] == value) {
            return 1;
        }
    }
    return 0;
}

int tuple_add_unique(Tuple *t, int value) {
    int *new_data;
    if (t == NULL) {
        return -1;
    }
    if (tuple_contains(t, value)) {
        return 0;
    }
    new_data = realloc(t->data, (t->size + 1) * sizeof(int));
    if (new_data == NULL) {
        return -1;
    }
    t->data = new_data;
    t->data[t->size] = value;
    t->size++;
    return 0;
}

int find_dissimilar(const Tuple *t1, const Tuple *t2, Tuple *result) {
    size_t i;
    if (t1 == NULL || t2 == NULL || result == NULL) {
        return -1;
    }
    result->data = NULL;
    result->size = 0;
    
    for (i = 0; i < t1->size; i++) {
        if (!tuple_contains(t2, t1->data[i])) {
            if (tuple_add_unique(result, t1->data[i]) != 0) {
                tuple_free(result);
                return -1;
            }
        }
    }
    for (i = 0; i < t2->size; i++) {
        if (!tuple_contains(t1, t2->data[i])) {
            if (tuple_add_unique(result, t2->data[i]) != 0) {
                tuple_free(result);
                return -1;
            }
        }
    }
    return 0;
}

int main(void) {
    Tuple t1, t2, result;
    size_t i;
    int status;
    
    if (tuple_init(&t1, 4) != 0) {
        fprintf(stderr, "Failed to initialize tuple 1\n");
        return EXIT_FAILURE;
    }
    if (tuple_init(&t2, 4) != 0) {
        fprintf(stderr, "Failed to initialize tuple 2\n");
        tuple_free(&t1);
        return EXIT_FAILURE;
    }
    
    t1.data[0] = 3;
    t1.data[1] = 4;
    t1.data[2] = 5;
    t1.data[3] = 6;
    
    t2.data[0] = 5;
    t2.data[1] = 6;
    t2.data[2] = 7;
    t2.data[3] = 8;
    
    status = find_dissimilar(&t1, &t2, &result);
    if (status != 0) {
        fprintf(stderr, "Failed to find dissimilar elements\n");
        tuple_free(&t1);
        tuple_free(&t2);
        return EXIT_FAILURE;
    }
    
    printf("Dissimilar elements: ");
    for (i = 0; i < result.size; i++) {
        printf("%d ", result.data[i]);
    }
    printf("\n");
    
    tuple_free(&t1);
    tuple_free(&t2);
    tuple_free(&result);
    
    return EXIT_SUCCESS;
}
