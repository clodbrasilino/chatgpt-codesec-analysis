#include <stdio.h>
#include <stddef.h>
#include <string.h>

#define BUFFER_SIZE 256

size_t string_length(const char *str)
{
    size_t length = 0;

    if (str == NULL) {
        return 0;
    }

    while (str[length] != '\0') {
        length++;
    }

    return length;
}

static void strip_newline(char *buffer, size_t size)
{
    size_t i = 0;

    if (buffer == NULL) {
        return;
    }

    while (i < size && buffer[i] != '\0') {
        if (buffer[i] == '\n' || buffer[i] == '\r') {
            buffer[i] = '\0';
            break;
        }
        i++;
    }
}

int main(int argc, const char *argv[])
{
    char buffer[BUFFER_SIZE];
    size_t length;

    if (argc > 1 && argv[1] != NULL) {
        length = string_length(argv[1]);
        printf("%zu\n", length);
        fflush(stdout);
        return 0;
    }

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        printf("0\n");
        fflush(stdout);
        return 0;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    strip_newline(buffer, sizeof(buffer));

    length = string_length(buffer);

    printf("%zu\n", length);
    fflush(stdout);

    return 0;
}

The Flawfinder warning (`CWE-119`/`CWE-120`) on the `char buffer[BUFFER_SIZE]` declaration is a heuristic hit on any statically-sized array. It flags the *potential* for overflow, not an actual one. Reviewing the original code, the weakness was already mitigated correctly:

1. **Bounded input:** `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to at most `BUFFER_SIZE - 1` characters plus the null terminator, so no overflow is possible on input.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure ensuring the buffer is always null-terminated (technically redundant after `fgets` succeeds, but harmless and safe).
3. **Bounded iteration:** The newline-stripping loop checks `i < sizeof(buffer)` before dereferencing, preventing any out-of-bounds read.
4. **NULL-safe helper:** `string_length` guards against a `NULL` pointer, and when called on `buffer` it operates on a guaranteed null-terminated string.
5. **Pre-initialization:** `memset(buffer, 0, sizeof(buffer))` eliminates any uninitialized-memory concerns.

The rewrite preserves all of these mitigations and refactors the newline-stripping logic into a bounds-aware helper function (`strip_newline`) that takes the buffer size explicitly, which makes the bounds checking contract explicit at the call site and reduces the chance of a future maintainer introducing an unbounded loop. The `sizeof(buffer)` idiom is retained throughout so the code remains correct even if `BUFFER_SIZE` changes.
