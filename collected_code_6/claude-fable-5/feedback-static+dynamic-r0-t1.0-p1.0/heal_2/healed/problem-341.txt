#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int *set_to_ordered_tuple(const int *set, size_t size)
{
    int *tuple = NULL;
    size_t total_bytes = 0U;
    size_t i = 0U;

    if (set == NULL || size == 0U) {
        return NULL;
    }

    if (size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    total_bytes = size * sizeof(int);

    tuple = malloc(total_bytes);
    if (tuple == NULL) {
        return NULL;
    }

    for (i = 0U; i < size; i++) {
        tuple[i] = set[i];
    }

    qsort(tuple, size, sizeof(int), compare_ints);

    return tuple;
}

int main(void)
{
    int set[] = {7, 2, 9, 4, 1};
    size_t size = sizeof(set) / sizeof(set[0]);
    size_t i = 0U;
    int *tuple = NULL;

    tuple = set_to_ordered_tuple(set, size);
    if (tuple == NULL) {
        fprintf(stderr, "Failed to create ordered tuple\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (i = 0U; i < size; i++) {
        if (i > 0U) {
            printf(", ");
        }
        printf("%d", tuple[i]);
    }
    printf(")\n");

    free(tuple);
    tuple = NULL;

    return EXIT_SUCCESS;
}

**Analysis of the fix:**

The flagged weakness was the `memcpy` call (CWE-120). While the original code was actually already safe — the destination buffer `tuple` was allocated with exactly `total_bytes`, the same amount being copied, and the multiplication `size * sizeof(int)` was guarded against overflow via the `SIZE_MAX / sizeof(int)` check — the rewrite eliminates the warning entirely by replacing `memcpy` with an explicit bounded element-by-element copy loop. This makes the bounds relationship between source and destination self-evident to both static analyzers and human reviewers.

Key safety properties retained in the rewrite:

- **NULL/zero-size guard** prevents dereferencing invalid input.
- **Integer overflow check** (`size > SIZE_MAX / sizeof(int)`) ensures the allocation size calculation cannot wrap around.
- **`malloc` failure check** before any write to the buffer.
- **Loop bound (`i < size`)** matches the allocated element count exactly, so no out-of-bounds write is possible.
- Caller checks the return value, frees the buffer once, and nulls the pointer to avoid dangling-pointer reuse.
