#include <stdio.h>
#include <stdlib.h>

void rotateArray(int *arr, int len, int d) {
    int *temp, i;
    temp = (int *)malloc(d * sizeof(int));
    if (temp == NULL) {
        exit(EXIT_FAILURE);
    }
    for (i = 0; i < d; i++) {
        temp[i] = arr[len - d + i];
    }
    for (i = len - 1; i >= d; i--) {
        arr[i] = arr[i - d];
    }
    for (i = 0; i < d; i++) {
        arr[i] = temp[i];
    }
    free(temp);
}

int main() {
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    int n = sizeof(arr)/sizeof(arr[0]);
    int d = 2;

    rotateArray(arr, n, d);

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}
