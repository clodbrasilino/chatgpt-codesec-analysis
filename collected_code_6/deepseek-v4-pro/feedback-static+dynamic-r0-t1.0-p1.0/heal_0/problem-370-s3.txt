```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    float value;
} Tuple;

int compare_tuples(const void *a, const void *b) {
    const Tuple *tuple_a = (const Tuple *)a;
    const Tuple *tuple_b = (const Tuple *)b;
    
    if (tuple_a->value < tuple_b->value) return -1;
    if (tuple_a->value > tuple_b->value) return 1;
    return 0;
}

void sort_tuples_by_float(Tuple *tuples, size_t count) {
    if (tuples == NULL || count == 0) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

int main(void) {
    Tuple tuples[] = {
        {1, 3.14f},
        {2, 1.41f},
        {3, 2.71f},
        {4, 0.58f},
        {5, 1.73f}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    size_t i;
    
    printf("Before sorting:\n");
    for (i = 0; i < count; i++) {
        printf("(%d, %.2f) ", tuples[i].id, tuples[i].value);
    }
    printf("\n");
    
    sort_tuples_by_float(tuples, count);
    
    printf("After sorting:\n");
    for (i = 0; i < count; i++) {
        printf("(%d, %.2f) ", tuples[i].id, tuples[i].value);
    }
    printf("\n");
    
    return 0;
}
```
