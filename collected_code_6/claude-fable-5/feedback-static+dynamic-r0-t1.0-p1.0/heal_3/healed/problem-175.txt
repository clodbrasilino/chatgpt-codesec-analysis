```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 4096

bool is_valid_parentheses(const char *input)
{
    size_t length;
    char *stack;
    size_t top = 0;
    size_t i;
    bool result = true;

    if (input == NULL) {
        return false;
    }

    length = strnlen(input, MAX_INPUT_LENGTH);
    if (length == MAX_INPUT_LENGTH && input[MAX_INPUT_LENGTH - 1] != '\0') {
        return false;
    }

    if (length == 0) {
        return true;
    }

    stack = malloc(length);
    if (stack == NULL) {
        return false;
    }

    for (i = 0; i < length; i++) {
        char c = input[i];

        if (c == '(' || c == '[' || c == '{') {
            if (top >= length) {
                result = false;
                break;
            }
            stack[top] = c;
            top++;
        } else if (c == ')' || c == ']' || c == '}') {
            if (top == 0) {
                result = false;
                break;
            }
            top--;
            if ((c == ')' && stack[top] != '(') ||
                (c == ']' && stack[top] != '[') ||
                (c == '}' && stack[top] != '{')) {
                result = false;
                break;
            }
        } else {
            result = false;
            break;
        }
    }

    if (result && top != 0) {
        result = false;
    }

    free(stack);
    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LENGTH];

    while (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        size_t len = strlen(buffer);

        while (len > 0 && (buffer[len - 1] == '\n' || buffer[len - 1] == '\r')) {
            buffer[len - 1] = '\0';
            len--;
        }

        if (is_valid_parentheses(buffer)) {
            printf("True\n");
        } else {
            printf("False\n");
        }
    }

    return EXIT_SUCCESS;
}
```

The key weakness identified by the failing test cases was that the program produced no output for the test harness: the original `main` only ran a hardcoded internal test array and never read the actual test inputs. The rewritten `main` reads each line from standard input, strips trailing newline/carriage-return characters, validates it, and prints `True` or `False` accordingly, which matches the expected outputs (`True` for cases 0 and 2, `False` for case 1, etc.). The validation function itself was already robust — it handles `NULL` input, unterminated oversized strings via `strnlen`, empty strings, allocation failure, stack underflow/overflow, mismatched bracket types, and unbalanced leftovers — so it is preserved unchanged.
