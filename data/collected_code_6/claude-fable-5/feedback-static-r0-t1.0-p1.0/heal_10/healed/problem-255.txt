```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NUM_COLOURS 3

static const char *const colours[NUM_COLOURS] = { "Red", "Green", "Blue" };

static void print_combination(const int *indices, size_t count)
{
    size_t i;

    if (indices == NULL || count == 0) {
        return;
    }

    for (i = 0; i < count; i++) {
        if (indices[i] < 0 || indices[i] >= NUM_COLOURS) {
            continue;
        }
        fputs(colours[indices[i]], stdout);
        if (i + 1 < count) {
            fputc(' ', stdout);
        }
    }
    fputc('\n', stdout);
}

static void generate_combinations(int *indices, size_t depth, size_t count, int start)
{
    int i;

    if (indices == NULL || count == 0) {
        return;
    }

    if (depth >= count) {
        print_combination(indices, count);
        return;
    }

    if (start < 0) {
        start = 0;
    }
    if (start >= NUM_COLOURS) {
        return;
    }

    for (i = start; i < NUM_COLOURS; i++) {
        indices[depth] = i;
        generate_combinations(indices, depth + 1, count, i);
    }
}

static int choose_colours(size_t count)
{
    int *indices;

    if (count == 0 || count > SIZE_MAX / sizeof(*indices)) {
        return -1;
    }

    indices = calloc(count, sizeof(*indices));
    if (indices == NULL) {
        return -1;
    }

    generate_combinations(indices, 0, count, 0);
    free(indices);
    return 0;
}

int main(void)
{
    size_t count = 2;

    printf("Combinations with repetition choosing %zu colours from %d:\n",
           count, NUM_COLOURS);

    if (choose_colours(count) != 0) {
        fprintf(stderr, "Failed to generate combinations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning targets the statically-sized array `colours[NUM_COLOURS]`. Reviewing the code, the original source already mitigates this risk correctly, so the rewrite preserves those defenses rather than restructuring:

1. **Bounds checking on every read** — In `print_combination`, each index is validated with `indices[i] < 0 || indices[i] >= NUM_COLOURS` before being used to access `colours[]`, preventing any out-of-bounds read even if the indices array were corrupted.

2. **Write bounds guaranteed by construction** — In `generate_combinations`, writes to `indices[depth]` only occur when `depth < count` (enforced by the `depth >= count` base case), and the buffer is allocated with exactly `count` elements. The loop variable `i` is clamped to `[start, NUM_COLOURS)` with `start` sanitized to be non-negative.

3. **Overflow-safe allocation** — `choose_colours` checks `count > SIZE_MAX / sizeof(*indices)` before calling `calloc`, preventing integer-overflow-induced under-allocation, and `calloc` itself performs the same multiplication check as a second layer of defense.

4. **NULL checks** — Both helper functions reject NULL pointers, and the `calloc` result is verified before use.

5. **No unbounded string operations** — Output uses `fputs`/`fputc` on compile-time constant strings only; there are no `strcpy`/`gets`-style calls that could overflow the fixed-size structures.

The Flawfinder hit is a false positive in this instance: the array is `static const`, never written at runtime, and every access is range-checked. The code is retained as-is with comments stripped, since altering the safe pattern would add no security value.
