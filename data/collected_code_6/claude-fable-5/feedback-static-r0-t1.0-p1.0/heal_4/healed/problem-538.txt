```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define TUPLE_MAX_ITEM_LEN 4096U

typedef struct {
    char **items;
    size_t count;
} Tuple;

static Tuple *tuple_create(const char *const *list, size_t count);
static void tuple_destroy(Tuple *tuple);
static void tuple_print(const Tuple *tuple);

static Tuple *tuple_create(const char *const *list, size_t count)
{
    Tuple *tuple = NULL;
    size_t i = 0U;

    if (list == NULL || count == 0U) {
        return NULL;
    }

    if (count > (SIZE_MAX / sizeof(*tuple->items))) {
        return NULL;
    }

    tuple = malloc(sizeof(*tuple));
    if (tuple == NULL) {
        return NULL;
    }

    tuple->items = calloc(count, sizeof(*tuple->items));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }
    tuple->count = count;

    for (i = 0U; i < count; i++) {
        size_t len = 0U;
        size_t alloc_size = 0U;

        if (list[i] == NULL) {
            tuple_destroy(tuple);
            return NULL;
        }

        len = strnlen(list[i], TUPLE_MAX_ITEM_LEN);
        if (len >= TUPLE_MAX_ITEM_LEN) {
            tuple_destroy(tuple);
            return NULL;
        }

        alloc_size = len + 1U;
        tuple->items[i] = malloc(alloc_size);
        if (tuple->items[i] == NULL) {
            tuple_destroy(tuple);
            return NULL;
        }

        if (len > 0U) {
            if (len >= alloc_size) {
                tuple_destroy(tuple);
                return NULL;
            }
            memcpy(tuple->items[i], list[i], len);
        }
        tuple->items[i][len] = '\0';
    }

    return tuple;
}

static void tuple_destroy(Tuple *tuple)
{
    if (tuple == NULL) {
        return;
    }

    if (tuple->items != NULL) {
        size_t i = 0U;

        for (i = 0U; i < tuple->count; i++) {
            free(tuple->items[i]);
        }
        free(tuple->items);
    }
    free(tuple);
}

static void tuple_print(const Tuple *tuple)
{
    size_t i = 0U;

    if (tuple == NULL || tuple->items == NULL) {
        return;
    }

    if (printf("(") < 0) {
        return;
    }
    for (i = 0U; i < tuple->count; i++) {
        if (tuple->items[i] == NULL) {
            return;
        }
        if (printf("'%s'", tuple->items[i]) < 0) {
            return;
        }
        if (i + 1U < tuple->count) {
            if (printf(", ") < 0) {
                return;
            }
        }
    }
    if (printf(")\n") < 0) {
        return;
    }
}

int main(void)
{
    const char *const list[] = { "apple", "banana", "cherry", "date" };
    const size_t count = sizeof(list) / sizeof(list[0]);
    Tuple *tuple = NULL;

    tuple = tuple_create(list, count);
    if (tuple == NULL) {
        (void)fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    tuple_print(tuple);
    tuple_destroy(tuple);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) was a false positive in the original code, but the rewrite hardens it defensively:

1. **Destination sizing is provably correct:** `alloc_size = len + 1U` guarantees the destination buffer can hold `len` bytes plus the null terminator. Since `len < TUPLE_MAX_ITEM_LEN`, the addition `len + 1U` cannot overflow `size_t`.

2. **Explicit guard added:** The rewrite adds an explicit `len >= alloc_size` check immediately before the `memcpy`, making the invariant machine-checkable and satisfying static analyzers rather than relying on the reader to trace the allocation logic.

3. **Bounded length measurement:** `strnlen` with `TUPLE_MAX_ITEM_LEN` prevents unbounded reads on non-terminated input, and the `len >= TUPLE_MAX_ITEM_LEN` rejection ensures no truncation ever occurs silently.

4. **Manual null termination:** `tuple->items[i][len] = '\0'` guarantees termination regardless of copy behavior, avoiding the classic `strncpy` non-termination pitfall.

Other correctness properties preserved: overflow check before `calloc` sizing, full cleanup on every error path via `tuple_destroy` (safe because `calloc` zero-initializes the pointer array, so `free(NULL)` is well-defined), and checked `printf` return values.
