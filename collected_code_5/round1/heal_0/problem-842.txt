```C
#include<stdio.h>

int findOdd(int arr[], int n) {
   int i, res = 0;
   for(i = 0; i < n; i++) {
      res = res ^ arr[i];
   }
   return res;
}

int main() {
   int arr[] = {2, 3, 5, 4, 5, 2, 4, 3, 5, 2, 4, 4, 2};
   int n = sizeof(arr)/sizeof(arr[0]);
   
   printf ("The number which occurs odd number of times is: %d ", findOdd(arr, n));
   return 0;
}
```
