#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *camel_to_snake(const char *input);

char *camel_to_snake(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *output = NULL;
    size_t input_len;
    size_t out_capacity;
    size_t out_len = 0;
    size_t offset = 0;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    out_capacity = (input_len * 2U) + 1U;

    output = malloc(out_capacity);
    if (output == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[A-Z]", REG_EXTENDED);
    if (ret != 0) {
        free(output);
        return NULL;
    }

    while (offset < input_len) {
        ret = regexec(&regex, input + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = input_len - offset;
            if (remaining > (out_capacity - 1U) ||
                out_len > ((out_capacity - 1U) - remaining) ||
                (out_len + remaining) >= out_capacity) {
                regfree(&regex);
                free(output);
                return NULL;
            }
            memcpy(output + out_len, input + offset, remaining);
            out_len += remaining;
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            free(output);
            return NULL;
        }

        if (match.rm_so < 0 || match.rm_eo <= 0) {
            regfree(&regex);
            free(output);
            return NULL;
        }

        if ((size_t)match.rm_so > (input_len - offset)) {
            regfree(&regex);
            free(output);
            return NULL;
        }

        if ((size_t)match.rm_so > (out_capacity - 1U) ||
            out_len > ((out_capacity - 1U) - (size_t)match.rm_so) ||
            (out_len + (size_t)match.rm_so + 2U) >= out_capacity) {
            regfree(&regex);
            free(output);
            return NULL;
        }

        memcpy(output + out_len, input + offset, (size_t)match.rm_so);
        out_len += (size_t)match.rm_so;

        if ((offset + (size_t)match.rm_so) > 0U) {
            if (out_len >= (out_capacity - 1U)) {
                regfree(&regex);
                free(output);
                return NULL;
            }
            output[out_len] = '_';
            out_len++;
        }

        if (out_len >= (out_capacity - 1U)) {
            regfree(&regex);
            free(output);
            return NULL;
        }

        output[out_len] = (char)tolower((unsigned char)input[offset + (size_t)match.rm_so]);
        out_len++;

        offset += (size_t)match.rm_eo;
    }

    if (out_len >= out_capacity) {
        regfree(&regex);
        free(output);
        return NULL;
    }

    output[out_len] = '\0';
    regfree(&regex);
    return output;
}

int main(void)
{
    const char *tests[] = {
        "camelCaseString",
        "ConvertThisToSnake",
        "alreadylower",
        "XMLParser",
        ""
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *result = camel_to_snake(tests[i]);
        if (result == NULL) {
            fprintf(stderr, "Conversion failed for input: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("%s -> %s\n", tests[i], result) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}

**Analysis summary of the fixes applied:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) were already guarded in the original code, but the rewrite strengthens the guarantees:

1. **First `memcpy` (tail copy on `REG_NOMATCH`):** An additional explicit check `(out_len + remaining) >= out_capacity` was added so the destination bound is verified directly against the total write size before copying, not just via the subtraction-based overflow-safe checks.

2. **Second `memcpy` (prefix copy before a match):** The existing three-part guard already ensures `out_len + match.rm_so + 2` fits within `out_capacity`, covering the copy plus the subsequent underscore and lowered character writes. This was retained.

3. **Single-byte writes hardened:** Explicit capacity checks were added before writing the `'_'` separator and the lowercased character, so each write into `output` is independently bounds-verified rather than relying solely on the earlier combined check.

The allocation of `(input_len * 2U) + 1U` cannot overflow because `input_len < 4096`, and the final NUL-terminator write is verified against capacity before execution. All error paths correctly release both the regex and the heap buffer, so there are no leaks or double-frees.
