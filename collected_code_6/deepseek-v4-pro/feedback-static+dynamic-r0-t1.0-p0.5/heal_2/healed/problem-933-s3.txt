#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *camel_to_snake(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t input_len = 0;
    while (input[input_len] != '\0') {
        if (input_len >= SIZE_MAX / 2) {
            return NULL;
        }
        input_len++;
    }

    size_t max_result_len = input_len * 2 + 1;
    char *result = malloc(max_result_len);
    if (result == NULL) {
        return NULL;
    }

    size_t dst_pos = 0;
    size_t i = 0;

    while (i < input_len) {
        if (i > 0 && isupper((unsigned char)input[i]) && 
            islower((unsigned char)input[i - 1]) && 
            isalpha((unsigned char)input[i - 1])) {
            if (dst_pos + 2 >= max_result_len) {
                free(result);
                return NULL;
            }
            result[dst_pos++] = '_';
            result[dst_pos++] = (char)tolower((unsigned char)input[i]);
            i++;
        } else if (i > 0 && isupper((unsigned char)input[i]) && 
                   i + 1 < input_len && islower((unsigned char)input[i + 1]) &&
                   isupper((unsigned char)input[i - 1])) {
            if (dst_pos + 2 >= max_result_len) {
                free(result);
                return NULL;
            }
            result[dst_pos++] = '_';
            result[dst_pos++] = (char)tolower((unsigned char)input[i]);
            i++;
        } else {
            if (dst_pos + 1 >= max_result_len) {
                free(result);
                return NULL;
            }
            result[dst_pos++] = (char)tolower((unsigned char)input[i]);
            i++;
        }
    }

    result[dst_pos] = '\0';

    char *final_result = realloc(result, dst_pos + 1);
    if (final_result == NULL) {
        free(result);
        return NULL;
    }

    return final_result;
}

int main(void) {
    const char *test_cases[] = {
        "chromeCast",
        "googleAssistant",
        "quadCore",
        "camelCase",
        "thisIsATest",
        "already_snake",
        "with123Numbers",
        "UPPERCASE",
        "",
        "a",
        "A",
        NULL
    };

    for (int i = 0; test_cases[i] != NULL; i++) {
        char *converted = camel_to_snake(test_cases[i]);
        if (converted != NULL) {
            printf("%s -> %s\n", test_cases[i], converted);
            free(converted);
        } else {
            printf("%s -> (conversion failed)\n", test_cases[i]);
        }
    }

    return 0;
}
