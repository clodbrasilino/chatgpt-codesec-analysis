```c
#include <stdio.h>

int countUnequalPairs(const int arr[], int n) {
    int count = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] != arr[j]) {
                count++;
            }
        }
    }
    return count;
}

int main() {
    const int arr[] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = countUnequalPairs(arr, n);
    printf("Number of unequal pairs: %d\n", result);
    return 0;
}
```
