```c
#include <stdio.h>
#include <stdlib.h>

double calculateAverage(const int *numbers, int size) {
    if (size <= 0) return 0.0;
    double sum = 0.0;
    for (int i = 0; i < size; ++i) {
        sum += numbers[i];
    }
    return sum / size;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    double avg = calculateAverage(arr, n);
    printf("Average: %.2f\n", avg);
    return 0;
}
```
