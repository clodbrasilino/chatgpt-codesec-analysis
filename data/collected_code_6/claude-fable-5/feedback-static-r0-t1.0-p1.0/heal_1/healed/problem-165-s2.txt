```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

size_t count_position_matches(const char *str, size_t max_len)
{
    size_t count = 0;
    size_t len;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);

    for (i = 0; i < len && i < 26; i++) {
        if (str[i] == (char)('a' + i) || str[i] == (char)('A' + i)) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t result;
    size_t len;

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = count_position_matches(buffer, sizeof(buffer));

    printf("Characters at same position as in alphabet: %zu\n", result);

    return EXIT_SUCCESS;
}
```

The key fixes applied:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen`, which takes a maximum length bound and will never read past the buffer even if the string is not null-terminated.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is now defined via a named constant, `fgets` is already bounded by `sizeof(buffer)`, and an explicit null terminator is written at the last byte as a defensive guarantee before any string operations occur.

3. **Hardened API contract**: `count_position_matches` now accepts an explicit `max_len` parameter so it cannot over-read regardless of the caller's buffer state, making the function safe even if passed a non-terminated string.
