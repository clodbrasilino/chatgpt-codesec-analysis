#include <stdio.h>
#include <stdlib.h>

char *tuple_join(const char **tuple, size_t count, const char *delimiter) {
    if (tuple == NULL || delimiter == NULL || count == 0) {
        return NULL;
    }

    int delim_len_int = snprintf(NULL, 0, "%s", delimiter);
    if (delim_len_int < 0) {
        return NULL;
    }
    size_t delim_len = (size_t)delim_len_int;
    size_t total_len = 0;

    for (size_t i = 0; i < count; i++) {
        if (tuple[i] == NULL) {
            return NULL;
        }
        int len_int = snprintf(NULL, 0, "%s", tuple[i]);
        if (len_int < 0) {
            return NULL;
        }
        total_len += (size_t)len_int;
        if (i < count - 1) {
            total_len += delim_len;
        }
    }

    char *result = (char *)malloc(total_len + 1);
    if (result == NULL) {
        return NULL;
    }

    char *ptr = result;
    size_t remaining = total_len + 1;

    for (size_t i = 0; i < count; i++) {
        int written = snprintf(ptr, remaining, "%s", tuple[i]);
        if (written < 0 || (size_t)written >= remaining) {
            break;
        }
        ptr += written;
        remaining -= written;

        if (i < count - 1) {
            written = snprintf(ptr, remaining, "%s", delimiter);
            if (written < 0 || (size_t)written >= remaining) {
                break;
            }
            ptr += written;
            remaining -= written;
        }
    }

    return result;
}

int main(void) {
    const char *tuple[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    const char *delimiter = ", ";

    char *result = tuple_join(tuple, count, delimiter);
    if (result == NULL) {
        fprintf(stderr, "Failed to join tuple\n");
        return 1;
    }

    printf("%s\n", result);

    free(result);

    return 0;
}
