#include <stdio.h>
#include <regex.h>
#include <string.h>

void find_sequences(const char *input, size_t input_len) {
    regex_t re;
    int reti;
    char msgbuf[100];

    if (regcomp(&re, "[A-Z][a-z]+", 0) != 0) {
        fprintf(stderr, "Could not compile regex\n");
        return;
    }

    if ((reti = regexec(&re, input, 0, NULL, 0)) == REG_NOMATCH) {
        printf("No matches found\n");
    } else if (reti != 0) {
        regerror(reti, &re, msgbuf, sizeof(msgbuf));
        fprintf(stderr, "Regex match failed: %s\n", msgbuf);
    } else {
        const char *current_input = input;

        while (1) {
            regmatch_t pmatch;
            if (regexec(&re, current_input, 1, &pmatch, 0) != 0) {
                break;
            }
            if (pmatch.rm_so == (size_t)-1 || pmatch.rm_eo > input_len) {
                break;
            }
            const char *start = current_input + pmatch.rm_so;
            printf("%.*s\n", (int)(pmatch.rm_eo - pmatch.rm_so), start);
            current_input += pmatch.rm_eo;
        }
    }
    regfree(&re);
}

int main() {
    const char *test_str = "HelloWorld isAnExample OfPattern Matching";
    size_t test_str_len = strlen(test_str);
    find_sequences(test_str, test_str_len);
    return 0;
}
