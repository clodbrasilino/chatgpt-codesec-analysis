```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(t->first));
    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        if (printf("(%.*s, %d)\n", MAX_NAME_LEN - 1, tuples[i].first, tuples[i].second) < 0) {
            break;
        }
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        if (fprintf(stderr, "Failed to initialize tuples\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        if (fprintf(stderr, "Failed to sort tuples\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **Statically-sized array `char first[MAX_NAME_LEN]` (CWE-119/CWE-120):** The flagged risk is mitigated because every write into `first` is bounds-checked. `set_tuple` is the only function that writes to it, and it validates the input length before copying.

2. **`memcpy` without overflow check (CWE-120):** This is a false positive in context. The call is preceded by `strnlen(name, sizeof(t->first))`, which caps the measured length at the buffer size, and the guard `if (len >= sizeof(t->first)) return -1;` guarantees `len <= MAX_NAME_LEN - 1`. Therefore `memcpy(t->first, name, len)` plus the null terminator at `t->first[len]` can never exceed the 64-byte buffer.

**Additional defensive qualities worth noting:**

- `strnlen` instead of `strlen` prevents reading past unterminated input.
- `strncmp` with an explicit bound in the comparator avoids over-reads even if a string were somehow unterminated.
- `printf("%.*s", ...)` limits output length, preventing over-reads during printing.
- NULL-pointer checks on all public entry points (`set_tuple`, `sort_tuples`, `print_tuples`).
- Return values of `printf`/`fprintf` are checked, and all error paths propagate `EXIT_FAILURE`.

The code as written already implements the fixes the static analyzer suggests; no functional changes were required, only removal of the annotation comments.
