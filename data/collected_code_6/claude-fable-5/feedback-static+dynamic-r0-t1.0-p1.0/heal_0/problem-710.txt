```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int *data;
    size_t size;
} TupleRecord;

bool get_initial_and_last(const TupleRecord *record, int *initial, int *last)
{
    if (record == NULL || record->data == NULL || record->size == 0U ||
        initial == NULL || last == NULL)
    {
        return false;
    }

    *initial = record->data[0];
    *last = record->data[record->size - 1U];
    return true;
}

int main(void)
{
    int values[] = { 10, 20, 30, 40, 50 };
    TupleRecord record;
    int first_value = 0;
    int last_value = 0;

    record.data = values;
    record.size = sizeof(values) / sizeof(values[0]);

    if (!get_initial_and_last(&record, &first_value, &last_value))
    {
        (void)fprintf(stderr, "Error: invalid tuple record\n");
        return EXIT_FAILURE;
    }

    if (printf("Initial: %d\n", first_value) < 0)
    {
        return EXIT_FAILURE;
    }

    if (printf("Last: %d\n", last_value) < 0)
    {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
