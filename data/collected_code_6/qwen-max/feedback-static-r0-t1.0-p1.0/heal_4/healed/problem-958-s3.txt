#include <stdio.h>
#include <string.h>

char *intToRoman(int num) {
    const char *romanNumerals[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    static char roman[20];
    int index = 0;

    for (int i = 0; i < 13; i++) {
        while (num >= values[i]) {
            size_t len = strlen(romanNumerals[i]);
            if (index + len + 1 < sizeof(roman)) {
                strncpy(roman + index, romanNumerals[i], len);
                index += len;
            }
            num -= values[i];
        }
    }

    roman[index] = '\0';
    return roman;
}

int main() {
    int number = 1994;
    printf("%s\n", intToRoman(number));
    return 0;
}
