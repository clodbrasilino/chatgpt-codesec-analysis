```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *left_rotate(const char *str, size_t d)
{
    size_t len;
    char *result;

    if (str == NULL) {
        return NULL;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    if (len == 0) {
        result[0] = '\0';
        return result;
    }

    d = d % len;

    if (len - d > len || d > len) {
        free(result);
        return NULL;
    }

    memcpy(result, str + d, len - d);
    memcpy(result + (len - d), str, d);
    result[len] = '\0';

    return result;
}

int main(void)
{
    const char *original = "HelloWorld";
    size_t shift = 3;
    char *rotated;

    rotated = left_rotate(original, shift);
    if (rotated == NULL) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", original) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    if (printf("Rotated:  %s\n", rotated) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    free(rotated);
    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weaknesses:

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. If the string is not `\0`-terminated within that limit, the function safely rejects the input instead of reading past the buffer.

2. **CWE-120 (`memcpy` buffer overflow):** The destination buffer is allocated as `len + 1`, and the two `memcpy` calls together write exactly `len` bytes (`len - d` plus `d`), so the destination always holds the source data. An explicit sanity check on `d` and `len - d` guards against any arithmetic anomaly before copying, making the bounds guarantee defensive rather than implicit.

3. Allocation failure, `NULL` input, and empty-string cases remain handled, and all heap memory is freed on every exit path in `main`.
