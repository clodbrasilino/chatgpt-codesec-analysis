#include <stdio.h>
#include <string.h>

int countSubstringsWithSameStartEnd(const char *str, size_t n) {
    int count = 0;
    
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (str[i] == str[j]) {
                count++;
            }
        }
    }
    
    return count / 2;
}

int main() {
    const char input[] = "abcab";
    size_t input_len = sizeof(input) / sizeof(input[0]) - 1;
    int result = countSubstringsWithSameStartEnd(input, input_len);
    printf("%d\n", result);
    return 0;
}
