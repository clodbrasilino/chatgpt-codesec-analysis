```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "/([0-9]{4})/([0-9]{1,2})/([0-9]{1,2})(/|$)"
#define MAX_FIELD_LEN 8
#define MAX_URL_LEN 2048
#define ERRBUF_LEN 128
#define MATCH_COUNT 5

typedef struct {
    char year[MAX_FIELD_LEN];
    char month[MAX_FIELD_LEN];
    char day[MAX_FIELD_LEN];
} UrlDate;

static int copy_match(const char *source, size_t source_len,
                      const regmatch_t *match,
                      char *dest, size_t dest_size)
{
    size_t length;

    if (source == NULL || match == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }
    if (match->rm_so < 0 || match->rm_eo < match->rm_so) {
        return -1;
    }
    if ((size_t)match->rm_eo > source_len) {
        return -1;
    }

    length = (size_t)(match->rm_eo - match->rm_so);
    if (length >= dest_size) {
        return -1;
    }

    memcpy(dest, source + match->rm_so, length);
    dest[length] = '\0';
    return 0;
}

int extract_date_from_url(const char *url, UrlDate *result)
{
    regex_t regex;
    regmatch_t matches[MATCH_COUNT];
    size_t url_len;
    int ret;

    if (url == NULL || result == NULL) {
        return -1;
    }

    memset(result, 0, sizeof(*result));

    url_len = strnlen(url, MAX_URL_LEN);
    if (url_len >= MAX_URL_LEN) {
        return -1;
    }

    ret = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, url, MATCH_COUNT, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (copy_match(url, url_len, &matches[1], result->year, sizeof(result->year)) != 0 ||
        copy_match(url, url_len, &matches[2], result->month, sizeof(result->month)) != 0 ||
        copy_match(url, url_len, &matches[3], result->day, sizeof(result->day)) != 0) {
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *urls[] = {
        "https://example.com/blog/2023/07/15/some-article",
        "https://news.site.org/2019/1/9",
        "https://example.com/no-date-here"
    };
    size_t count = sizeof(urls) / sizeof(urls[0]);
    size_t i;
    UrlDate date;

    for (i = 0; i < count; i++) {
        if (extract_date_from_url(urls[i], &date) == 0) {
            if (printf("URL: %s\n  Year: %s, Month: %s, Day: %s\n",
                       urls[i], date.year, date.month, date.day) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("URL: %s\n  No date found\n", urls[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:** The flagged weaknesses in the original code were already properly mitigated, so the rewrite preserves those defenses:

1. **Statically-sized struct fields (`year`, `month`, `day`) — CWE-119/CWE-120:** All writes go through `copy_match()`, which validates the match offsets against the source length and rejects any match whose length would equal or exceed `dest_size`, guaranteeing room for the null terminator.

2. **`memcpy` — CWE-120:** The copy is guarded by the `length >= dest_size` check and the `rm_eo <= source_len` bounds validation before the `memcpy` executes, so the destination can always hold the data.

3. **`errbuf` stack array — CWE-119/CWE-120:** `regerror()` is passed `sizeof(errbuf)` so it cannot overflow, and an explicit null terminator is enforced defensively at `errbuf[sizeof(errbuf) - 1]`.

Additional hardening already present and retained: `strnlen()` with `MAX_URL_LEN` caps input scanning, all pointer parameters are null-checked, `result` is zeroed before use, and `regfree()` is called on every exit path after successful compilation to avoid resource leaks.
