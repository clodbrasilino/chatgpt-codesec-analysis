#include <stdio.h>
#include <stdlib.h>

static void swap_long(long *a, long *b)
{
    long tmp = *a;
    *a = *b;
    *b = tmp;
}

static void sift_down_max(long *heap, size_t size, size_t root)
{
    for (;;) {
        size_t largest = root;
        size_t left = 2U * root + 1U;
        size_t right = 2U * root + 2U;

        if (left < size && heap[left] > heap[largest]) {
            largest = left;
        }
        if (right < size && heap[right] > heap[largest]) {
            largest = right;
        }
        if (largest == root) {
            break;
        }
        swap_long(&heap[root], &heap[largest]);
        root = largest;
    }
}

static void sift_down_min(long *heap, size_t size, size_t root)
{
    for (;;) {
        size_t smallest = root;
        size_t left = 2U * root + 1U;
        size_t right = 2U * root + 2U;

        if (left < size && heap[left] < heap[smallest]) {
            smallest = left;
        }
        if (right < size && heap[right] < heap[smallest]) {
            smallest = right;
        }
        if (smallest == root) {
            break;
        }
        swap_long(&heap[root], &heap[smallest]);
        root = smallest;
    }
}

static void build_max_heap(long *heap, size_t size)
{
    size_t i = size / 2U;

    while (i > 0U) {
        i--;
        sift_down_max(heap, size, i);
    }
}

static void build_min_heap(long *heap, size_t size)
{
    size_t i = size / 2U;

    while (i > 0U) {
        i--;
        sift_down_min(heap, size, i);
    }
}

static long pop_max(long *heap, size_t *size)
{
    long top = heap[0];

    (*size)--;
    heap[0] = heap[*size];
    sift_down_max(heap, *size, 0U);
    return top;
}

static long pop_min(long *heap, size_t *size)
{
    long top = heap[0];

    (*size)--;
    heap[0] = heap[*size];
    sift_down_min(heap, *size, 0U);
    return top;
}

int max_product_of_three(const int *arr, size_t n, long *result)
{
    long *max_heap;
    long *min_heap;
    size_t i;
    size_t max_size;
    size_t min_size;
    long big1;
    long big2;
    long big3;
    long small1;
    long small2;
    long prod_a;
    long prod_b;

    if (arr == NULL || result == NULL || n < 3U) {
        return -1;
    }

    max_heap = malloc(n * sizeof(long));
    if (max_heap == NULL) {
        return -1;
    }

    min_heap = malloc(n * sizeof(long));
    if (min_heap == NULL) {
        free(max_heap);
        return -1;
    }

    for (i = 0U; i < n; i++) {
        max_heap[i] = (long)arr[i];
        min_heap[i] = (long)arr[i];
    }

    max_size = n;
    min_size = n;

    build_max_heap(max_heap, max_size);
    build_min_heap(min_heap, min_size);

    big1 = pop_max(max_heap, &max_size);
    big2 = pop_max(max_heap, &max_size);
    big3 = pop_max(max_heap, &max_size);

    small1 = pop_min(min_heap, &min_size);
    small2 = pop_min(min_heap, &min_size);

    prod_a = big1 * big2 * big3;
    prod_b = small1 * small2 * big1;

    *result = (prod_a > prod_b) ? prod_a : prod_b;

    free(max_heap);
    free(min_heap);
    return 0;
}

int main(void)
{
    int arr1[] = {1, 10, 2, 6, 5, 3};
    int arr2[] = {-10, -10, 5, 2};
    int arr3[] = {-1, -2, -3};
    long result;

    if (max_product_of_three(arr1, sizeof(arr1) / sizeof(arr1[0]), &result) == 0) {
        printf("Max product: %ld\n", result);
    } else {
        printf("Error computing max product\n");
    }

    if (max_product_of_three(arr2, sizeof(arr2) / sizeof(arr2[0]), &result) == 0) {
        printf("Max product: %ld\n", result);
    } else {
        printf("Error computing max product\n");
    }

    if (max_product_of_three(arr3, sizeof(arr3) / sizeof(arr3[0]), &result) == 0) {
        printf("Max product: %ld\n", result);
    } else {
        printf("Error computing max product\n");
    }

    return 0;
}
