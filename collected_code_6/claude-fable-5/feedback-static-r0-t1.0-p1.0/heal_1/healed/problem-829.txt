```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

struct WordCount {
    char *word;
    size_t count;
};

static char *duplicateBounded(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_WORD_LEN);
    if (len >= MAX_WORD_LEN) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static void freeTable(struct WordCount *table, size_t tableSize)
{
    size_t j;

    for (j = 0; j < tableSize; j++) {
        free(table[j].word);
    }
    free(table);
}

static char *findSecondMostFrequent(const char **words, size_t n)
{
    struct WordCount *table = NULL;
    size_t tableSize = 0;
    size_t i;
    size_t j;
    char *result = NULL;

    if (words == NULL || n < 2) {
        return NULL;
    }

    table = calloc(n, sizeof(struct WordCount));
    if (table == NULL) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        int found = 0;

        if (words[i] == NULL) {
            continue;
        }

        if (strnlen(words[i], MAX_WORD_LEN) >= MAX_WORD_LEN) {
            freeTable(table, tableSize);
            return NULL;
        }

        for (j = 0; j < tableSize; j++) {
            if (strcmp(table[j].word, words[i]) == 0) {
                table[j].count++;
                found = 1;
                break;
            }
        }

        if (found == 0) {
            char *copy = duplicateBounded(words[i]);

            if (copy == NULL) {
                freeTable(table, tableSize);
                return NULL;
            }

            table[tableSize].word = copy;
            table[tableSize].count = 1;
            tableSize++;
        }
    }

    if (tableSize >= 2) {
        size_t firstIdx = 0;
        size_t secondIdx = tableSize;

        for (j = 1; j < tableSize; j++) {
            if (table[j].count > table[firstIdx].count) {
                secondIdx = firstIdx;
                firstIdx = j;
            } else if (secondIdx == tableSize ||
                       table[j].count > table[secondIdx].count) {
                secondIdx = j;
            }
        }

        if (secondIdx < tableSize) {
            result = duplicateBounded(table[secondIdx].word);
        }
    }

    freeTable(table, tableSize);

    return result;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "orange",
        "banana", "apple", "orange", "banana", "banana"
    };
    size_t count = sizeof(sequence) / sizeof(sequence[0]);
    char *second = NULL;

    second = findSecondMostFrequent(sequence, count);
    if (second == NULL) {
        fprintf(stderr, "Could not determine the second most frequent string\n");
        return EXIT_FAILURE;
    }

    if (printf("Second most frequent string: %s\n", second) < 0) {
        free(second);
        return EXIT_FAILURE;
    }

    free(second);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (over-read via `strlen`)**: Both `strlen` calls were replaced with `strnlen` bounded by `MAX_WORD_LEN`, so a string missing its NUL terminator can no longer cause an unbounded read. Inputs at or beyond the bound are rejected rather than silently truncated.

2. **CWE-120 (unchecked `memcpy`)**: All duplication logic is centralized in `duplicateBounded`, where the destination buffer is allocated at exactly `len + 1` bytes based on the bounded length measurement, and the NUL terminator is written explicitly. The `memcpy` size can therefore never exceed the destination capacity.

3. **Refactoring for safety**: Cleanup logic was consolidated into `freeTable` to eliminate the duplicated free-loops on error paths, reducing the chance of a future leak or double-free being introduced during maintenance. The original memory-management behavior (caller frees the returned string) and tie-breaking semantics are preserved.
