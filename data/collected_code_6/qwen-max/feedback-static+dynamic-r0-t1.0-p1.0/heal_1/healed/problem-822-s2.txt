#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool is_valid_password(const char *password, size_t min_length) {
    if (password == NULL || min_length == 0) {
        return false;
    }

    bool has_upper = 0, has_lower = 0, has_digit = 0;
    size_t length = 0;

    while (password[length] != '\0' && password[length]) {
        if (password[length] >= 'A' && password[length] <= 'Z') {
            has_upper = 1;
        } else if (password[length] >= 'a' && password[length] <= 'z') {
            has_lower = 1;
        } else if (password[length] >= '0' && password[length] <= '9') {
            has_digit = 1;
        }
        length++;
        if (length < min_length) {
            continue;
        } else {
            break;
        }
    }

    return has_upper && has_lower && has_digit && length >= min_length;
}

int main() {
    const char *password = "Test123";
    size_t min_length = 8;

    if (is_valid_password(password, min_length)) {
        printf("Password is valid.\n");
    } else {
        printf("Password is not valid.\n");
    }

    return 0;
}
