```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

#define BUFFER_SIZE 100

size_t string_length(const char *str) {
    size_t length = 0;
    if (str == NULL) {
        return 0;
    }
    while (length < BUFFER_SIZE && str[length] != '\0') {
        length++;
    }
    return length;
}

int main(void) {
    char str[BUFFER_SIZE];
    
    printf("Enter a string: ");
    if (fgets(str, BUFFER_SIZE, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }
    
    size_t len = strcspn(str, "\n");
    if (len < BUFFER_SIZE - 1) {
        str[len] = '\0';
    } else {
        str[BUFFER_SIZE - 1] = '\0';
        int c;
        bool overflow = false;
        while ((c = getchar()) != '\n' && c != EOF) {
            overflow = true;
        }
        if (overflow) {
            fprintf(stderr, "Warning: Input was truncated to %d characters\n", BUFFER_SIZE - 1);
        }
    }
    
    printf("Length of the string: %zu\n", string_length(str));
    
    return 0;
}
```
