```c
#include <stdio.h>
#include <string.h>

int count_std_occurrences(const char *str) {
    if (str == NULL) {
        return 0;
    }

    int count = 0;
    const char *target = "std";
    size_t target_len = sizeof("std") - 1;
    const char *ptr = str;

    while ((ptr = strstr(ptr, target)) != NULL) {
        count++;
        ptr += target_len;
    }

    return count;
}

int main(void) {
    const char *test_string = "1. list item 1. list item std 1. list item 2. list item 2. list item std 2. list item";
    int occurrences = count_std_occurrences(test_string);

    printf("The substring 'std' occurs %d times.\n", occurrences);

    return 0;
}
```
