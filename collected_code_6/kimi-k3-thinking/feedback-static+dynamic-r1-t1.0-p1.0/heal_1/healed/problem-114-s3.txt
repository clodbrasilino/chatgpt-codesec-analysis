```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int first;
    int second;
} Tuple;

typedef struct {
    Tuple tuple;
    size_t frequency;
} TupleFrequency;

static bool tuples_equal(const Tuple *a, const Tuple *b)
{
    return (a->first == b->first) && (a->second == b->second);
}

size_t assign_tuple_frequencies(const Tuple *tuples, size_t count, TupleFrequency *result)
{
    size_t unique_count = 0;
    size_t i;
    size_t j;

    if (tuples == NULL || result == NULL) {
        return 0;
    }

    for (i = 0; i < count; i++) {
        bool found = false;
        for (j = 0; j < unique_count; j++) {
            if (tuples_equal(&tuples[i], &result[j].tuple)) {
                result[j].frequency++;
                found = true;
                break;
            }
        }
        if (!found) {
            result[unique_count].tuple = tuples[i];
            result[unique_count].frequency = 1;
            unique_count++;
        }
    }

    return unique_count;
}

int main(void)
{
    Tuple tuples[] = {
        {1, 2}, {3, 4}, {1, 2}, {5, 6}, {3, 4}, {1, 2}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    TupleFrequency *frequencies = NULL;
    size_t unique_count;
    size_t i;

    frequencies = malloc(count * sizeof(*frequencies));
    if (frequencies == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    unique_count = assign_tuple_frequencies(tuples, count, frequencies);

    for (i = 0; i < unique_count; i++) {
        printf("Tuple (%d, %d) : Frequency %zu\n",
               frequencies[i].tuple.first,
               frequencies[i].tuple.second,
               frequencies[i].frequency);
    }

    free(frequencies);
    frequencies = NULL;

    return EXIT_SUCCESS;
}
```
