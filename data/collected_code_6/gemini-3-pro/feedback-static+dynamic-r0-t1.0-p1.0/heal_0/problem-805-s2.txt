```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int* data;
    size_t size;
} List;

typedef struct {
    List* lists;
    size_t count;
} ListOfLists;

List* findMaxSumList(const ListOfLists* lol) {
    if (!lol || !lol->lists || lol->count == 0) {
        return NULL;
    }

    List* maxList = NULL;
    long long maxSum = LLONG_MIN;

    for (size_t i = 0; i < lol->count; ++i) {
        if (!lol->lists[i].data) {
            continue;
        }

        long long currentSum = 0;
        for (size_t j = 0; j < lol->lists[i].size; ++j) {
            currentSum += lol->lists[i].data[j];
        }

        if (currentSum > maxSum) {
            maxSum = currentSum;
            maxList = &lol->lists[i];
        }
    }

    if (maxSum == LLONG_MIN) {
        return NULL;
    }

    return maxList;
}

int main() {
    int arr1[] = {1, 2, 3};
    int arr2[] = {4, 5, -1};
    int arr3[] = {10, -5, 2, 1};

    List lists[3];
    lists[0].data = arr1;
    lists[0].size = sizeof(arr1) / sizeof(arr1[0]);

    lists[1].data = arr2;
    lists[1].size = sizeof(arr2) / sizeof(arr2[0]);

    lists[2].data = arr3;
    lists[2].size = sizeof(arr3) / sizeof(arr3[0]);

    ListOfLists lol;
    lol.lists = lists;
    lol.count = 3;

    List* result = findMaxSumList(&lol);

    if (result) {
        printf("List with max sum: ");
        for (size_t i = 0; i < result->size; ++i) {
            printf("%d ", result->data[i]);
        }
        printf("\n");
    } else {
        printf("No valid list found.\n");
    }

    return 0;
}
```
