#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* concatenate(char *a, char *b) {
    size_t len_a = strnlen(a, 256);
    size_t len_b = strnlen(b, 256);
    char *result = (char *)malloc(len_a + len_b + 1);
    if (result == NULL) {
        return NULL;
    }
    snprintf(result, len_a + len_b + 1, "%s%s", a, b);
    return result;
}

void processTuples(char *tuples[], int length, char **output) {
    for (int i = 0; i < length - 1; i++) {
        output[i] = concatenate(tuples[i], tuples[i + 1]);
    }
    output[length - 1] = NULL;
}

int main() {
    char *tuples[] = {"hello", "world", "foo", "bar"};
    int length = sizeof(tuples) / sizeof(tuples[0]);
    char *output[length];
    
    processTuples(tuples, length, output);

    for (int i = 0; i < length - 1; i++) {
        printf("%s\n", output[i]);
        free(output[i]);
    }

    return 0;
}
