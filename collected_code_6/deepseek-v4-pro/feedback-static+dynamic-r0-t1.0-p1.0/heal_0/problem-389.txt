#include <stdio.h>
#include <stdlib.h>

unsigned long long lucas_number(int n) {
    if (n < 0) {
        fprintf(stderr, "Error: Input must be non-negative\n");
        exit(EXIT_FAILURE);
    }
    
    if (n == 0) return 2;
    if (n == 1) return 1;
    
    unsigned long long prev2 = 2;
    unsigned long long prev1 = 1;
    unsigned long long current = 0;
    
    for (int i = 2; i <= n; i++) {
        current = prev1 + prev2;
        prev2 = prev1;
        prev1 = current;
    }
    
    return current;
}

int main(void) {
    int n;
    
    printf("Enter n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    unsigned long long result = lucas_number(n);
    printf("Lucas number L%d = %llu\n", n, result);
    
    return EXIT_SUCCESS;
}
