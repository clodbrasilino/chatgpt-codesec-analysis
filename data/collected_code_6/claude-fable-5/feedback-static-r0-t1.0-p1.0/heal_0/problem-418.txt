#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
} SubList;

const SubList *find_max_sublist(const SubList *lists, size_t count)
{
    const SubList *max_list;
    size_t i;

    if (lists == NULL || count == 0) {
        return NULL;
    }

    max_list = &lists[0];
    for (i = 1; i < count; i++) {
        if (lists[i].length > max_list->length) {
            max_list = &lists[i];
        }
    }

    return max_list;
}

static int init_sublist(SubList *list, const int *values, size_t length)
{
    size_t i;

    if (list == NULL || values == NULL || length == 0) {
        return -1;
    }

    list->data = malloc(length * sizeof(int));
    if (list->data == NULL) {
        return -1;
    }

    for (i = 0; i < length; i++) {
        list->data[i] = values[i];
    }
    list->length = length;

    return 0;
}

static void free_sublists(SubList *lists, size_t count)
{
    size_t i;

    if (lists == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(lists[i].data);
        lists[i].data = NULL;
        lists[i].length = 0;
    }
}

int main(void)
{
    SubList lists[3] = { {NULL, 0}, {NULL, 0}, {NULL, 0} };
    const int a[] = {1, 2};
    const int b[] = {3, 4, 5, 6};
    const int c[] = {7, 8, 9};
    const SubList *result;
    size_t i;

    if (init_sublist(&lists[0], a, sizeof(a) / sizeof(a[0])) != 0) {
        free_sublists(lists, 3);
        return EXIT_FAILURE;
    }
    if (init_sublist(&lists[1], b, sizeof(b) / sizeof(b[0])) != 0) {
        free_sublists(lists, 3);
        return EXIT_FAILURE;
    }
    if (init_sublist(&lists[2], c, sizeof(c) / sizeof(c[0])) != 0) {
        free_sublists(lists, 3);
        return EXIT_FAILURE;
    }

    result = find_max_sublist(lists, 3);
    if (result == NULL) {
        free_sublists(lists, 3);
        return EXIT_FAILURE;
    }

    printf("Maximum length sublist (length %zu): ", result->length);
    for (i = 0; i < result->length; i++) {
        printf("%d ", result->data[i]);
    }
    printf("\n");

    free_sublists(lists, 3);

    return EXIT_SUCCESS;
}
