```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int *items;
    size_t size;
} List;

typedef struct {
    List *element;
} SingletonTuple;

List *copy_list_from_tuple(const SingletonTuple *tuple)
{
    List *copy;
    size_t bytes;

    if (tuple == NULL || tuple->element == NULL) {
        return NULL;
    }

    copy = malloc(sizeof(List));
    if (copy == NULL) {
        return NULL;
    }

    copy->size = tuple->element->size;

    if (copy->size == 0) {
        copy->items = NULL;
        return copy;
    }

    if (tuple->element->items == NULL) {
        free(copy);
        return NULL;
    }

    if (copy->size > SIZE_MAX / sizeof(int)) {
        free(copy);
        return NULL;
    }

    bytes = copy->size * sizeof(int);

    copy->items = malloc(bytes);
    if (copy->items == NULL) {
        free(copy);
        return NULL;
    }

    memcpy(copy->items, tuple->element->items, bytes);

    return copy;
}

void free_list(List *list)
{
    if (list != NULL) {
        free(list->items);
        free(list);
    }
}

int main(void)
{
    int data[] = {10, 20, 30, 40, 50};
    size_t count = sizeof(data) / sizeof(data[0]);
    size_t bytes;
    size_t i;

    if (count > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Size overflow\n");
        return EXIT_FAILURE;
    }

    bytes = count * sizeof(int);

    List original;
    original.items = malloc(bytes);
    if (original.items == NULL) {
        fprintf(stderr, "Allocation failed\n");
        return EXIT_FAILURE;
    }

    memcpy(original.items, data, bytes);
    original.size = count;

    SingletonTuple tuple;
    tuple.element = &original;

    List *copied = copy_list_from_tuple(&tuple);
    if (copied == NULL) {
        fprintf(stderr, "Copy failed\n");
        free(original.items);
        return EXIT_FAILURE;
    }

    printf("Original list: ");
    for (i = 0; i < original.size; i++) {
        printf("%d ", original.items[i]);
    }
    printf("\n");

    printf("Copied list:   ");
    for (i = 0; i < copied->size; i++) {
        printf("%d ", copied->items[i]);
    }
    printf("\n");

    free(original.items);
    free_list(copied);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**CWE-120 (memcpy in `copy_list_from_tuple`):** This is a false positive. The destination buffer `copy->items` is allocated with exactly `bytes` (`copy->size * sizeof(int)`), and the same `bytes` value is passed to `memcpy`. The source list's size field is used to compute both the allocation and the copy length, so the destination can always hold the source data. The overflow guard `copy->size > SIZE_MAX / sizeof(int)` prevents the multiplication from wrapping, which would otherwise be the real vulnerability (a small allocation followed by a huge copy). NULL checks on `tuple`, `tuple->element`, and `tuple->element->items` prevent invalid reads, and the zero-size case is handled explicitly before any allocation or copy occurs.

**CWE-120 (memcpy in `main`):** Also a false positive. `count` and `bytes` are derived directly from `sizeof(data)`, so the copy length exactly matches both the source array size and the destination allocation. The `SIZE_MAX / sizeof(int)` check guards the multiplication, though with a compile-time-known array it can never trigger.

Additional correctness points already in place: every `malloc` return is checked, all error paths free previously acquired memory (no leaks on failure), `free_list` tolerates NULL and correctly frees a list whose `items` is NULL (empty-list case, since `free(NULL)` is defined behavior), and `original.items` plus the copied list are both released before normal exit.
