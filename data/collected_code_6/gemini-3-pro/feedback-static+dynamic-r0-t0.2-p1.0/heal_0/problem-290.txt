#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
} List;

typedef struct {
    List *lists;
    size_t count;
} ListOfLists;

ListOfLists find_max_length_lists(const ListOfLists *input) {
    ListOfLists result = {NULL, 0};
    if (!input || !input->lists || input->count == 0) {
        return result;
    }

    size_t max_length = 0;
    for (size_t i = 0; i < input->count; ++i) {
        if (input->lists[i].length > max_length) {
            max_length = input->lists[i].length;
        }
    }

    size_t max_count = 0;
    for (size_t i = 0; i < input->count; ++i) {
        if (input->lists[i].length == max_length) {
            max_count++;
        }
    }

    result.lists = (List *)malloc(max_count * sizeof(List));
    if (!result.lists) {
        return result;
    }

    result.count = max_count;
    size_t index = 0;
    for (size_t i = 0; i < input->count; ++i) {
        if (input->lists[i].length == max_length) {
            result.lists[index].length = input->lists[i].length;
            result.lists[index].data = (int *)malloc(max_length * sizeof(int));
            if (!result.lists[index].data) {
                for (size_t j = 0; j < index; ++j) {
                    free(result.lists[j].data);
                }
                free(result.lists);
                result.lists = NULL;
                result.count = 0;
                return result;
            }
            for (size_t j = 0; j < max_length; ++j) {
                result.lists[index].data[j] = input->lists[i].data[j];
            }
            index++;
        }
    }

    return result;
}

void free_list_of_lists(ListOfLists *lol) {
    if (lol && lol->lists) {
        for (size_t i = 0; i < lol->count; ++i) {
            free(lol->lists[i].data);
        }
        free(lol->lists);
        lol->lists = NULL;
        lol->count = 0;
    }
}

int main(void) {
    int data1[] = {1, 2, 3};
    int data2[] = {4, 5};
    int data3[] = {6, 7, 8};
    int data4[] = {9};

    List lists[] = {
        {data1, 3},
        {data2, 2},
        {data3, 3},
        {data4, 1}
    };

    ListOfLists input = {lists, 4};

    ListOfLists max_lists = find_max_length_lists(&input);

    if (max_lists.lists) {
        for (size_t i = 0; i < max_lists.count; ++i) {
            for (size_t j = 0; j < max_lists.lists[i].length; ++j) {
                printf("%d ", max_lists.lists[i].data[j]);
            }
            printf("\n");
        }
        free_list_of_lists(&max_lists);
    }

    return 0;
}
