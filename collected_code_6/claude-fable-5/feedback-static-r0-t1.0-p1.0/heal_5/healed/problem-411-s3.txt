```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t buf_size;
    size_t out_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        size_t prefix_len;
        size_t remaining;

        if (match[0].rm_so < 0 || match[1].rm_so < 0 ||
            match[0].rm_eo < match[0].rm_so) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        prefix_len = (size_t)match[0].rm_so;

        if (out_pos >= buf_size) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        remaining = buf_size - out_pos;

        if (prefix_len >= remaining || (remaining - prefix_len) < 2U) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if (prefix_len > strnlen(cursor, prefix_len)) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, prefix_len);
        out_pos += prefix_len;

        result[out_pos] = (char)toupper((unsigned char)cursor[match[1].rm_so]);
        out_pos += 1U;

        cursor += match[0].rm_eo;
    }

    {
        size_t rest_len;
        size_t remaining;

        if (out_pos >= buf_size) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        remaining = buf_size - out_pos;
        rest_len = strnlen(cursor, remaining);

        if (rest_len >= remaining) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, rest_len);
        out_pos += rest_len;
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s -> %s\n", tests[i], converted);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) were already effectively mitigated in the original code, and the rewrite preserves and slightly hardens those mitigations:

1. **First `memcpy` (prefix copy):** Before copying, the code verifies `out_pos < buf_size`, computes the exact `remaining` space, and rejects the operation unless `prefix_len < remaining` **and** at least 2 bytes remain (one for the uppercased character, one for the eventual NUL terminator). I added a defensive validation that `match[0].rm_eo >= match[0].rm_so` and a `strnlen` sanity check on the source region so the copy length can never exceed readable source data.

2. **Second `memcpy` (tail copy):** The length is derived via `strnlen(cursor, remaining)`, which is inherently bounded by the destination's remaining capacity, and the `rest_len >= remaining` check guarantees room for the terminating `'\0'`.

Additional safety properties retained: input length is capped at `MAX_INPUT_LEN` using `strnlen` (avoiding unbounded `strlen`), camelCase output is provably never longer than the input (each `_X` pair collapses to one character), so `buf_size = len + 1` is always sufficient, all error paths release both the regex and the heap buffer (no leaks), and `toupper` receives an `unsigned char` cast to avoid UB on negative `char` values.
