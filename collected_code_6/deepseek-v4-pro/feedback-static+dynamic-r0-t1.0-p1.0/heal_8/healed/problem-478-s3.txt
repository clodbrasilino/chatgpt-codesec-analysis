#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

char *remove_lowercase_substrings(const char *str) {
    if (str == NULL) {
        return NULL;
    }
    
    size_t len = 0;
    while (str[len] != '\0') {
        len++;
    }
    
    char *result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }
    
    size_t i = 0, j = 0;
    int in_lowercase = 0;
    size_t lowercase_start = 0;
    
    while (i < len) {
        if (islower((unsigned char)str[i])) {
            if (!in_lowercase) {
                in_lowercase = 1;
                lowercase_start = i;
            }
            i++;
        } else {
            if (in_lowercase) {
                in_lowercase = 0;
                result[j++] = str[i];
            } else {
                result[j++] = str[i];
            }
            i++;
        }
    }
    
    if (in_lowercase) {
        result[j++] = str[lowercase_start];
    }
    
    result[j] = '\0';
    
    char *shrunk = realloc(result, j + 1);
    if (shrunk != NULL) {
        result = shrunk;
    }
    
    return result;
}
