```c
#include <stdio.h>

int maxSumDivideThree(int n);

int main() {
    int n;
    scanf("%d", &n);
    printf("%d\n", maxSumDivideThree(n));
    return 0;
}

int maxSumDivideThree(int n) {
    if (n == 0 || n == 1) return 0;
    if (n == 2) return 1;
    return n / 3 * 2 + maxSumDivideThree(n / 3);
}
```
