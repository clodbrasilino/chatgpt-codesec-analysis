```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int first;
    int second;
} Tuple;

char *flatten_tuple_list(const Tuple *tuples, size_t count)
{
    size_t buffer_size;
    size_t offset;
    size_t i;
    char *result;
    int written;

    if (tuples == NULL || count == 0) {
        result = malloc(1);
        if (result == NULL) {
            return NULL;
        }
        result[0] = '\0';
        return result;
    }

    buffer_size = count * 32 + 1;
    result = malloc(buffer_size);
    if (result == NULL) {
        return NULL;
    }

    offset = 0;
    for (i = 0; i < count; i++) {
        if (i < count - 1) {
            written = snprintf(result + offset, buffer_size - offset,
                               "%d %d ", tuples[i].first, tuples[i].second);
        } else {
            written = snprintf(result + offset, buffer_size - offset,
                               "%d %d", tuples[i].first, tuples[i].second);
        }
        if (written < 0 || (size_t)written >= buffer_size - offset) {
            free(result);
            return NULL;
        }
        offset += (size_t)written;
    }

    return result;
}

int main(void)
{
    Tuple tuples[] = {
        {1, 4},
        {6, 7},
        {5, 4}
    };
    size_t count;
    char *flattened;

    count = sizeof(tuples) / sizeof(tuples[0]);
    flattened = flatten_tuple_list(tuples, count);
    if (flattened == NULL) {
        fprintf(stderr, "Failed to flatten tuple list\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", flattened) < 0) {
        free(flattened);
        return EXIT_FAILURE;
    }

    free(flattened);
    return EXIT_SUCCESS;
}
```
