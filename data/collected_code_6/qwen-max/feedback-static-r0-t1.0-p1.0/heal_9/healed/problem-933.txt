#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

int camel_to_snake(const char *input, char *output, size_t output_size) {
    regex_t re;
    regmatch_t pmatch[1];
    int reti, offset = 0;

    if (regcomp(&re, "([a-z0-9])([A-Z])", REG_EXTENDED | REG_NEWLINE) != 0) {
        return -1;
    }

    while ((reti = regexec(&re, input + offset, 1, pmatch, 0)) == 0) {
        size_t match_len = pmatch[0].rm_eo - pmatch[0].rm_so;
        if (offset + match_len + 2 >= output_size) {
            regfree(&re);
            return -1;
        }
        for (int i = 0; i < match_len; i++) {
            output[offset + i] = input[offset + i];
        }
        output[offset + pmatch[0].rm_so] = '_';
        output[offset + pmatch[0].rm_so + 1] = tolower((unsigned char)input[offset + pmatch[0].rm_so + 1]);
        offset += pmatch[0].rm_so + 2;
    }

    if (reti != REG_NOMATCH) {
        regfree(&re);
        return -1;
    }

    size_t remaining_len = output_size - offset - 1;
    if (remaining_len > 0) {
        strncat(output + offset, input + offset, remaining_len - 1);
        output[output_size - 1] = '\0';
    } else {
        regfree(&re);
        return -1;
    }

    for (size_t i = 0; i < output_size && output[i] != '\0'; i++) {
        if (output[i] >= 'A' && output[i] <= 'Z') {
            output[i] = tolower((unsigned char)output[i]);
        }
    }

    regfree(&re);
    return 0;
}

int main() {
    const char input[] = "ThisIsACamelCaseString";
    char output[256];
    size_t input_len = strlen(input);

    if (input_len < sizeof(output)) {
        if (camel_to_snake(input, output, sizeof(output) - 1) == 0) {
            printf("%s\n", output);
        } else {
            printf("Conversion failed.\n");
        }
    } else {
        printf("Input string too long.\n");
    }

    return 0;
}
