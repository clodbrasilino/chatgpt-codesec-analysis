```C
#include<stdio.h>

int findOdd(int arr[], int arr_size)
{
    int res = 0;
    for (int i = 0; i < arr_size; i++){
        res = res ^ arr[i];
    }
    return res;
}

int main(void)
{
    int arr[] = {1, 2, 3, 2, 3, 1, 3};
    int n = sizeof(arr)/sizeof(arr[0]);
    printf("%d", findOdd(arr, n));
    return 0;
}
```
