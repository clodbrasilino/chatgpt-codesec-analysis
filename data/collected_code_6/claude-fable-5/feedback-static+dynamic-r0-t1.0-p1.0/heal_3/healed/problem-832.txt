```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <regex.h>

#define NUM_BUFFER_SIZE 64

static int copy_match(char *dest, size_t dest_size, const char *src, size_t len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_size == 0 || len >= dest_size) {
        return -1;
    }
    memcpy(dest, src, len);
    dest[len] = '\0';
    return 0;
}

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[NUM_BUFFER_SIZE];
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;
        long value;
        char *endptr;

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            regfree(&regex);
            return -1;
        }

        len = (size_t)(match.rm_eo - match.rm_so);

        if (len >= sizeof(buffer)) {
            regfree(&regex);
            return -1;
        }

        if (copy_match(buffer, sizeof(buffer), cursor + match.rm_so, len) != 0) {
            regfree(&regex);
            return -1;
        }

        errno = 0;
        value = strtol(buffer, &endptr, 10);
        if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test1 = "abc 42 def -17 ghi 305 jkl 7";
    const char *test2 = "no numbers here";
    long max;

    if (extract_max_number(test1, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test1, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test1);
    }

    if (extract_max_number(test2, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test2, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test2);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes made:**

1. **CWE-120 (`memcpy` in `copy_match`)**: The original code already guarded the copy with `len >= dest_size`, ensuring room for the NUL terminator. I retained that check and added a defensive `dest_size == 0` guard to eliminate any edge case where `len >= dest_size` could theoretically pass with an unusable buffer size.

2. **CWE-119/CWE-120 (statically-sized `buffer[NUM_BUFFER_SIZE]`)**: The buffer itself is safe because every write path goes through `copy_match`, which enforces bounds. I added an explicit `len >= sizeof(buffer)` check at the call site before invoking `copy_match`, making the bounds enforcement visible and redundant at both layers — a defense-in-depth measure that satisfies static analyzers and reviewers alike.

3. **Everything else was already sound**: `strtol` is used with full `errno`/`endptr` validation (guarding against overflow and partial parses), `regfree` is called on every exit path preventing resource leaks, and NULL-pointer checks exist on all public entry points. No further remediation was required.
