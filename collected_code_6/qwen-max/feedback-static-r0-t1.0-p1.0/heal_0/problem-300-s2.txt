#include <stdio.h>
#include <stdlib.h>

long long int findCount(long long int n);

int main() {
    long long int n;
    printf("Enter n: ");
    scanf("%lld", &n);
    if (n < 1) {
        return 1;
    }
    long long int count = findCount(n);
    printf("Count: %lld\n", count);
    return 0;
}

long long int findCount(long long int n) {
    long long int dp[n+1][n+1];
    for (long long int i = 0; i <= n; i++)
        for (long long int j = 0; j <= n; j++)
            dp[i][j] = 0;
    dp[0][0] = 1;
    for (long long int len = 1; len <= n; len++) {
        dp[len][0] = dp[len-1][0];
        for (long long int diff = 1; diff <= len; diff++)
            dp[len][diff] = dp[len-1][diff-1] + dp[len-1][diff+1];
    }
    return dp[n][0];
}
