```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 1024
#define MAX_WORD_LEN 256
#define MAX_INPUT_LEN 65536

static char *duplicate_string(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(src, max_len - 1);
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static int extract_word(const char *str, size_t str_len, size_t *pos,
                        char *word, size_t word_size)
{
    size_t i;
    size_t w = 0;

    if (str == NULL || pos == NULL || word == NULL || word_size == 0) {
        return 0;
    }

    i = *pos;

    while (i < str_len && str[i] != '\0' && !isalnum((unsigned char)str[i])) {
        i++;
    }

    if (i >= str_len || str[i] == '\0') {
        *pos = i;
        word[0] = '\0';
        return 0;
    }

    while (i < str_len && str[i] != '\0' && isalnum((unsigned char)str[i])) {
        if (w + 1 < word_size) {
            word[w] = str[i];
            w++;
        }
        i++;
    }

    word[w] = '\0';
    *pos = i;
    return 1;
}

static int store_word(char (*words)[MAX_WORD_LEN], size_t index,
                      const char *src, size_t dest_size)
{
    size_t len;

    if (words == NULL || src == NULL || dest_size == 0 ||
        dest_size > MAX_WORD_LEN || index >= MAX_WORDS) {
        return -1;
    }

    len = strnlen(src, dest_size - 1);
    if (len > 0) {
        memcpy(words[index], src, len);
    }
    words[index][len] = '\0';
    return 0;
}

char *first_repeated_word(const char *str)
{
    char (*words)[MAX_WORD_LEN];
    char current[MAX_WORD_LEN];
    size_t pos = 0;
    size_t count = 0;
    size_t i;
    size_t str_len;
    char *result;

    if (str == NULL) {
        return duplicate_string("None", MAX_WORD_LEN);
    }

    str_len = strnlen(str, MAX_INPUT_LEN);

    words = calloc((size_t)MAX_WORDS, sizeof(*words));
    if (words == NULL) {
        return NULL;
    }

    while (extract_word(str, str_len, &pos, current, sizeof(current)) == 1) {
        for (i = 0; i < count; i++) {
            if (strncmp(words[i], current, MAX_WORD_LEN) == 0) {
                result = duplicate_string(current, MAX_WORD_LEN);
                free(words);
                return result;
            }
        }
        if (count < MAX_WORDS) {
            if (store_word(words, count, current, sizeof(words[count])) != 0) {
                free(words);
                return NULL;
            }
            count++;
        } else {
            break;
        }
    }

    free(words);
    return duplicate_string("None", MAX_WORD_LEN);
}

static int run_test(const char *input)
{
    char *repeated;

    repeated = first_repeated_word(input);
    if (repeated == NULL) {
        if (printf("Memory allocation failure\n") < 0) {
            return -1;
        }
        return -1;
    }

    if (strncmp(repeated, "None", MAX_WORD_LEN) == 0) {
        if (printf("No repeated word found\n") < 0) {
            free(repeated);
            return -1;
        }
    } else {
        if (printf("First repeated word: %s\n", repeated) < 0) {
            free(repeated);
            return -1;
        }
    }

    free(repeated);
    return 0;
}

int main(void)
{
    const char *test1 = "the quick brown fox jumps over the lazy dog";
    const char *test2 = "hello world foo bar";
    const char *test3 = "one two two three";

    if (run_test(test1) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test2) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test3) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

The Flawfinder hits in the original code were all low-risk (risk 2) advisories, and inspection shows they were already properly mitigated:

1. **`memcpy` in `duplicate_string` (CWE-120):** Safe. The destination is allocated with `malloc(len + 1)` where `len` is bounded by `strnlen(src, max_len - 1)`, so the buffer always holds `len` bytes plus the NUL terminator.

2. **`memcpy` in `store_word` (CWE-120):** Safe, and now hardened further. The copy length is derived from `strnlen(src, dest_size - 1)`, guaranteeing it fits in the destination row. I added two defensive checks: `dest_size > MAX_WORD_LEN` guards against a caller passing an inconsistent size relative to the actual row width, and `index >= MAX_WORDS` prevents out-of-bounds row access even though the only call site already enforces `count < MAX_WORDS`.

3. **Statically-sized arrays (CWE-119/CWE-120):** The `char current[MAX_WORD_LEN]` buffer and the `words` array-of-arrays are always accessed with explicit bounds: `extract_word` truncates via `w + 1 < word_size`, comparisons use `strncmp(..., MAX_WORD_LEN)`, and the word count is capped at `MAX_WORDS`.

Other correctness properties preserved: all allocations are checked, `words` is freed on every exit path (no leaks), `isalnum` receives an `unsigned char` cast avoiding UB on negative `char` values, and input scanning is bounded by `MAX_INPUT_LEN` via `strnlen`.
