```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

bool is_binary_string(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (max_len == 0) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        if (str[i] != '0' && str[i] != '1') {
            return false;
        }
    }

    if (i == max_len) {
        return false;
    }

    return true;
}

static void flush_stdin(void)
{
    int ch;
    size_t count = 0;
    const size_t max_flush = 1048576;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < max_flush);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    if (is_binary_string(buffer, sizeof(buffer))) {
        printf("\"%s\" is a binary string\n", buffer);
    } else {
        printf("\"%s\" is not a binary string\n", buffer);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The `flush_stdin()` function does not store the characters read into any buffer, so no overflow is possible. Additionally, the loop is bounded by `max_flush` (1 MiB), preventing an unbounded loop on a malicious or pathological input stream, and it correctly terminates on both `'\n'` and `EOF`.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The `buffer` is protected by multiple layers:
   - `fgets(buffer, sizeof(buffer), stdin)` limits the read to `BUFFER_SIZE - 1` characters and always null-terminates within bounds.
   - A defensive `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination even if `fgets` behavior were somehow violated.
   - `strnlen` is used instead of `strlen`, bounding the length scan to the buffer size.
   - When input exceeds the buffer (no newline found and the buffer is full), the remaining stdin data is drained via `flush_stdin()` so it cannot poison subsequent reads.
   - `is_binary_string()` takes an explicit `max_len` parameter, validates `NULL`, empty strings, zero length, and rejects strings that are not properly terminated within `max_len`.

Both Flawfinder findings are low-risk informational hits already mitigated by the existing bounds checking, so the rewritten code retains the same hardened logic with the comments removed.
