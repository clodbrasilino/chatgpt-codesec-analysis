```c
#include <stdio.h>
#include <math.h>

double lateral_surface_area(double radius, double height) {
    return 2.0 * M_PI * radius * height;
}

int main(void) {
    double radius = 5.0;
    double height = 10.0;
    
    double area = lateral_surface_area(radius, height);
    printf("Lateral surface area: %.2f\n", area);
    
    return 0;
}
```
