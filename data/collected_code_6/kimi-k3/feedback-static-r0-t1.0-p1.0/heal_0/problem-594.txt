#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

int find_difference(const int *arr, size_t size, int *result)
{
    if (arr == NULL || result == NULL || size == 0)
    {
        return -1;
    }

    bool found_even = false;
    bool found_odd = false;
    int first_even = 0;
    int first_odd = 0;

    for (size_t i = 0; i < size; i++)
    {
        if (!found_even && arr[i] % 2 == 0)
        {
            first_even = arr[i];
            found_even = true;
        }
        else if (!found_odd && arr[i] % 2 != 0)
        {
            first_odd = arr[i];
            found_odd = true;
        }

        if (found_even && found_odd)
        {
            break;
        }
    }

    if (!found_even || !found_odd)
    {
        return -1;
    }

    *result = first_even - first_odd;
    return 0;
}

int main(void)
{
    int arr[] = {1, 3, 5, 8, 9, 10};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int difference = 0;

    if (find_difference(arr, size, &difference) == 0)
    {
        printf("Difference: %d\n", difference);
    }
    else
    {
        printf("Could not find both even and odd numbers.\n");
    }

    return 0;
}
