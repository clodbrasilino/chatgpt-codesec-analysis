#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>

bool is_valid_parentheses(const char *str, size_t len) {
    if (str == NULL) {
        return false;
    }

    if (len == 0) {
        return true;
    }

    if (len == SIZE_MAX) {
        return false;
    }

    char *stack = malloc(len);
    if (stack == NULL) {
        return false;
    }

    size_t top = 0;

    for (size_t i = 0; i < len; i++) {
        char c = str[i];
        if (c == '(' || c == '{' || c == '[') {
            stack[top++] = c;
        } else if (c == ')' || c == '}' || c == ']') {
            if (top == 0) {
                free(stack);
                return false;
            }
            char popped = stack[--top];
            if ((c == ')' && popped != '(') ||
                (c == '}' && popped != '{') ||
                (c == ']' && popped != '[')) {
                free(stack);
                return false;
            }
        }
    }

    bool result = (top == 0);
    free(stack);
    return result;
}

int main(void) {
    printf("%d\n", is_valid_parentheses("()[]{}", 6));
    printf("%d\n", is_valid_parentheses("([)]", 4));
    printf("%d\n", is_valid_parentheses(NULL, 0));
    printf("%d\n", is_valid_parentheses("(", 1));
    printf("%d\n", is_valid_parentheses(")", 1));
    return 0;
}
