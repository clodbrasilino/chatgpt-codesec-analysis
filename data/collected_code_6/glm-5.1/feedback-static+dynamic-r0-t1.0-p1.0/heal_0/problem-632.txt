#include <stdio.h>
#include <stdlib.h>

void move_zeroes(int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return;
    }

    size_t write_idx = 0;

    for (size_t i = 0; i < size; i++) {
        if (arr[i] != 0) {
            arr[write_idx] = arr[i];
            write_idx++;
        }
    }

    while (write_idx < size) {
        arr[write_idx] = 0;
        write_idx++;
    }
}

int main(void) {
    int arr[] = {1, 0, 2, 0, 3, 4, 0, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    move_zeroes(arr, size);

    for (size_t i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
