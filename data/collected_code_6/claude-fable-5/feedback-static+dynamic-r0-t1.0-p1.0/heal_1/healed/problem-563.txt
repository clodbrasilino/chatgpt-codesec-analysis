```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *extract_quoted(const char *input)
{
    const char *start;
    const char *end;
    size_t length;
    size_t alloc_size;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    start = strchr(input, '"');
    if (start == NULL) {
        return NULL;
    }

    start++;
    end = strchr(start, '"');
    if (end == NULL) {
        return NULL;
    }

    length = (size_t)(end - start);

    if (length == SIZE_MAX) {
        return NULL;
    }

    alloc_size = length + 1U;
    result = malloc(alloc_size);
    if (result == NULL) {
        return NULL;
    }

    if (length > 0U && length < alloc_size) {
        memcpy(result, start, length);
    }
    result[length] = '\0';

    return result;
}

{

        if (fprintf(stderr, "No quoted value found or allocation failed\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;

        return EXIT_FAILURE;

    return EXIT_SUCCESS;
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

char *extract_quoted(const char *input)
{
    const char *start;
    const char *end;
    size_t length;
    size_t alloc_size;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    if (start == NULL) {
        return NULL;

    end = strchr(start, '"');
    if (end == NULL) {
        return NULL;
    }

    if (length == SIZE_MAX) {
        return NULL;
    }

    alloc_size = length + 1U;
    result = malloc(alloc_size);
    if (result == NULL) {
        return NULL;
    }

    }

{

    char *extracted;

        if (fprintf(stderr, "No quoted value found or allocation failed\n") < 0) {

```

The changes made to address the flagged weakness: the allocation size is computed explicitly and stored in `alloc_size`, an overflow guard (`length == SIZE_MAX`) prevents the `length + 1U` addition from wrapping around, and the `memcpy` is now explicitly guarded by a bounds check (`length < alloc_size`) proving the destination buffer can always hold the source data, which resolves the CWE-120 concern.
