#include <stdio.h>

typedef struct {
    int first;
    int second;
    int valid;
} Pair;

Pair find_highest_product_pair(const int arr[], size_t size)
{
    Pair result = {0, 0, 0};
    int max1, max2, min1, min2;
    size_t i;
    long long product_max, product_min;

    if (arr == NULL || size < 2U) {
        return result;
    }

    if (arr[0] >= arr[1]) {
        max1 = arr[0];
        max2 = arr[1];
        min1 = arr[1];
        min2 = arr[0];
    } else {
        max1 = arr[1];
        max2 = arr[0];
        min1 = arr[0];
        min2 = arr[1];
    }

    for (i = 2U; i < size; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    product_max = (long long)max1 * (long long)max2;
    product_min = (long long)min1 * (long long)min2;

    if (product_max >= product_min) {
        result.first = max1;
        result.second = max2;
    } else {
        result.first = min1;
        result.second = min2;
    }
    result.valid = 1;

    return result;
}

int main(void)
{
    int arr[] = {-10, -3, 5, 6, -2};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    Pair result = find_highest_product_pair(arr, size);

    if (result.valid != 0) {
        printf("Pair with highest product: %d and %d\n", result.first, result.second);
        printf("Product: %lld\n", (long long)result.first * (long long)result.second);
    } else {
        fprintf(stderr, "Error: array is NULL or has fewer than two elements.\n");
        return 1;
    }

    return 0;
}
