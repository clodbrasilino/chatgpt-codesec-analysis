```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    char *data;
} Tuple;

void rearrange_tuples(Tuple* tuples, int num_tuples, const int* order, int num_order, Tuple** result, int* num_result) {
    if (!tuples || !order || !result || !num_result || num_tuples <= 0 || num_order <= 0) {
        if (result) *result = NULL;
        if (num_result) *num_result = 0;
        return;
    }

    *result = (Tuple*)malloc(num_order * sizeof(Tuple));
    if (!*result) {
        *num_result = 0;
        return;
    }

    int count = 0;
    for (int i = 0; i < num_order; i++) {
        for (int j = 0; j < num_tuples; j++) {
            if (tuples[j].id == order[i]) {
                (*result)[count] = tuples[j];
                count++;
                break;
            }
        }
    }

    *num_result = count;
}

int main(void) {
    Tuple tuples[] = {
        {3, "Tuple 3"},
        {1, "Tuple 1"},
        {4, "Tuple 4"},
        {2, "Tuple 2"}
    };
    int num_tuples = sizeof(tuples) / sizeof(tuples[0]);

    int order[] = {2, 4, 1, 3};
    int num_order = sizeof(order) / sizeof(order[0]);

    Tuple* rearranged = NULL;
    int num_rearranged = 0;

    rearrange_tuples(tuples, num_tuples, order, num_order, &rearranged, &num_rearranged);

    if (rearranged) {
        for (int i = 0; i < num_rearranged; i++) {
            printf("ID: %d, Data: %s\n", rearranged[i].id, rearranged[i].data);
        }
        free(rearranged);
    }

    return 0;
}
```
