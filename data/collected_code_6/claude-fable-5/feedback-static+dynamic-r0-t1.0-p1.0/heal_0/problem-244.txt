#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <limits.h>

static int next_perfect_square(unsigned long long n, unsigned long long *result)
{
    unsigned long long root;
    unsigned long long next_root;

    if (result == NULL) {
        return -1;
    }

    root = (unsigned long long)sqrt((double)n);

    while (root * root > n) {
        root--;
    }
    while ((root + 1ULL) * (root + 1ULL) <= n) {
        root++;
    }

    next_root = root + 1ULL;

    if (next_root > 0xFFFFFFFFULL) {
        return -1;
    }

    *result = next_root * next_root;
    return 0;
}

int main(void)
{
    unsigned long long input;
    unsigned long long square;

    printf("Enter a non-negative integer: ");

    if (scanf("%llu", &input) != 1) {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    if (next_perfect_square(input, &square) != 0) {
        fprintf(stderr, "Result would overflow.\n");
        return EXIT_FAILURE;
    }

    printf("Next perfect square greater than %llu is %llu\n", input, square);

    return EXIT_SUCCESS;
}
