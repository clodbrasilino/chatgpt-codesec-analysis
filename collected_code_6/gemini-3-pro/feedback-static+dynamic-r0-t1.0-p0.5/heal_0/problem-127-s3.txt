#include <stdio.h>

int multiply(int a, int b);

int multiply(int a, int b) {
    unsigned int uresult = 0;
    unsigned int ua = (a < 0) ? -(unsigned int)a : (unsigned int)a;
    unsigned int ub = (b < 0) ? -(unsigned int)b : (unsigned int)b;
    int is_negative = (a < 0) ^ (b < 0);

    while (ub > 0) {
        if (ub & 1) {
            uresult += ua;
        }
        ua <<= 1;
        ub >>= 1;
    }

    return is_negative ? -(int)uresult : (int)uresult;
}

int main(void) {
    int a = 15;
    int b = -4;
    int result = multiply(a, b);
    
    printf("%d\n", result);
    
    return 0;
}
