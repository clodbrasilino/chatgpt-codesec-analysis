```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            if (match_len >= sizeof(matches[found].text)) {
                match_len = sizeof(matches[found].text) - 1;
            }

            if (match_len > 0 && start + match_len <= len) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended against the weaknesses flagged by Flawfinder — those were informational warnings triggered by pattern matching on `char[]` and `memcpy`, not actual vulnerabilities. Here's why the code is safe as written:

1. **Statically-sized buffers (CWE-119/CWE-120):** Both `input` and `Match.text` are fixed at `MAX_INPUT` bytes, but every read/write is bounds-checked. `fgets(input, sizeof(input), stdin)` limits input length, and a forced NUL terminator is added defensively at `input[sizeof(input) - 1]`.

2. **`memcpy` (CWE-120):** The copy is safe because `match_len` is explicitly clamped to `sizeof(matches[found].text) - 1` before the copy, guaranteeing room for the NUL terminator. The additional `start + match_len <= len` check prevents reading past the source.

3. **Other good practices already present:** `strnlen` instead of `strlen` (bounded scanning), `isupper`/`islower` called with `unsigned char` casts (avoiding UB on negative `char` values), NULL-parameter validation in `find_sequences`, and checked return values for `fgets` and `find_sequences`.

The only functional addition in the rewrite is `fflush(stdout)` after the prompt, ensuring the prompt appears before blocking on `stdin` when output is line- or fully-buffered (e.g., when piped).
