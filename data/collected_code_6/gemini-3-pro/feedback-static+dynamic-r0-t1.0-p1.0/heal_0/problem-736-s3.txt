#include <stdio.h>
#include <stdlib.h>

int find_left_insertion_point(int *arr, int size, int value) {
    if (arr == NULL || size < 0) {
        return -1;
    }

    int left = 0;
    int right = size;

    while (left < right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] < value) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }

    return left;
}

int main(void) {
    int arr[] = {1, 3, 3, 3, 5, 7, 9};
    int size = sizeof(arr) / sizeof(arr[0]);
    int value = 3;

    int insertion_point = find_left_insertion_point(arr, size, value);

    if (insertion_point != -1) {
        printf("The left insertion point for %d is at index %d\n", value, insertion_point);
    } else {
        printf("Error: Invalid array or size.\n");
    }

    return 0;
}
