#include <stdio.h>
#include <string.h>
#include <ctype.h>

static int is_vowel(char c) {
    c = tolower((unsigned char)c);
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

size_t count_chars_with_vowel_neighbors(const char *str) {
    size_t len = strlen(str);
    size_t count = 0;
    size_t i;

    if (len < 2) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        int vowel_neighbor = 0;

        if (i > 0) {
            if (is_vowel(str[i - 1])) {
                vowel_neighbor = 1;
            }
        }

        if (i < len - 1) {
            if (is_vowel(str[i + 1])) {
                vowel_neighbor = 1;
            }
        }

        if (vowel_neighbor) {
            count++;
        }
    }

    return count;
}

int main(void) {
    const char *input = "abcdefghi";
    size_t result = count_chars_with_vowel_neighbors(input);
    printf("%zu\n", result);
    return 0;
}
