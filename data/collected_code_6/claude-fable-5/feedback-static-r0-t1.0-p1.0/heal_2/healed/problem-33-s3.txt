#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (decimal > 0 && index < buffer_size - 1) {
        buffer[index] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
        index++;
    }

    if (decimal > 0) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[index] = '\0';

    for (i = 0; i < index / 2; i++) {
        char temp = buffer[i];
        buffer[i] = buffer[index - 1 - i];
        buffer[index - 1 - i] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[MAX_BITS + 1] = {0};
    int scan_result;

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &number);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal: %u\n", number);
    printf("Binary: %s\n", binary);

    return EXIT_SUCCESS;
}

**Analysis notes:**

The original code was already well-defended against the Flawfinder finding (CWE-119/CWE-120). The warning is a heuristic false positive triggered simply by the presence of a fixed-size `char` array. Here's why the code is safe, and what I adjusted:

1. **Buffer sizing is provably correct**: `MAX_BITS + 1` guarantees room for every bit of an `unsigned int` plus the NUL terminator, so overflow is mathematically impossible for any input.

2. **Bounds checking is enforced at the point of write**: the loop condition `index < buffer_size - 1` prevents any out-of-bounds write, and the caller passes `sizeof(binary)` rather than a hard-coded constant, so the check stays correct if the buffer size ever changes.

3. **Defensive validation**: `NULL` and minimum-size checks reject unusable buffers before any write occurs, and `scanf`'s return value is validated so an uninitialized read of `number` cannot happen.

4. **Hardening additions in the rewrite**: I zero-initialized `binary` so it is always a valid NUL-terminated string even on error paths, and I NUL-terminate the buffer (`buffer[0] = '\0'`) before returning the truncation error, so a caller that ignores the return code can never print or process uninitialized memory.
