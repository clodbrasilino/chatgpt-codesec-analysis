```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define IP_PATTERN "^((25[0-5]|2[0-4][0-9]|1[0-9]{2}|[1-9]?[0-9])\\.){3}(25[0-5]|2[0-4][0-9]|1[0-9]{2}|[1-9]?[0-9])$"

int is_valid_ip(const char *ip)
{
    regex_t regex;
    int result;

    if (ip == NULL) {
        return 0;
    }

    result = regcomp(&regex, IP_PATTERN, REG_EXTENDED | REG_NOSUB);
    if (result != 0) {
        size_t errbuf_size = regerror(result, &regex, NULL, 0);
        char *errbuf = malloc(errbuf_size);
        if (errbuf != NULL) {
            regerror(result, &regex, errbuf, errbuf_size);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Regex compilation failed\n");
        }
        return -1;
    }

    result = regexec(&regex, ip, 0, NULL, 0);
    regfree(&regex);

    if (result == 0) {
        return 1;
    } else if (result == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex execution failed\n");
        return -1;
    }
}

int main(void)
{
    const char *test_ips[] = {
        "192.168.1.1",
        "255.255.255.255",
        "0.0.0.0",
        "256.1.1.1",
        "192.168.1",
        "abc.def.ghi.jkl",
        "10.0.0.1",
        "999.999.999.999",
        "1.2.3.4.5",
        ""
    };
    size_t count = sizeof(test_ips) / sizeof(test_ips[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int status = is_valid_ip(test_ips[i]);
        if (status < 0) {
            fprintf(stderr, "Error validating IP: %s\n", test_ips[i]);
            return EXIT_FAILURE;
        }
        printf("%-20s : %s\n", test_ips[i], status == 1 ? "Valid" : "Invalid");
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **Flawfinder warning (CWE-119/CWE-120) — statically-sized `errbuf[256]`:** Replaced the fixed-size stack buffer with a dynamically allocated buffer sized exactly by a first call to `regerror(result, &regex, NULL, 0)`, which returns the required buffer length including the null terminator. This eliminates any risk of truncation or improper sizing, and the `malloc` failure path is handled gracefully with a fallback error message.

2. **cppcheck `variableScope` warning on `status`:** The declaration of `status` was moved from function scope in `main` into the `for` loop body, reducing its scope to the smallest block where it is used. This improves readability and prevents accidental reuse of stale values across iterations.

**Notes on remaining code quality:** The original code was already fairly robust — `regfree` is called on every path after a successful `regcomp`, `NULL` input is handled, and `REG_NOSUB` avoids unnecessary submatch allocation. The regex itself correctly validates dotted-quad IPv4 addresses (0–255 per octet, no leading-zero ambiguity beyond single digits), and the test suite in `main` exercises valid, out-of-range, malformed, short, long, and empty inputs.
