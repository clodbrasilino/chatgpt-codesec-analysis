```C
#include <stdio.h>
#include <stdlib.h>

int compare(const void* a, const void* b) {
    return (*(int*)b - *(int*)a);
}

void find_largest_products(int* arr1, int* arr2, int n, int k) {
    int* results = malloc(k * sizeof(int));
    if(results == NULL) {
        printf("Memory allocation failed.\n");
        exit(0);
    }

    qsort(arr1, n, sizeof(int), compare);
    qsort(arr2, n, sizeof(int), compare);

    for(int i=0; i<k; i++) {
        results[i] = arr1[i] * arr2[i];
    }

    qsort(results, k, sizeof(int), compare);
    printf("%d largest products are: ", k);
    for(int i=0; i<k; i++) {
        printf("%d ", results[i]);
    }
    printf("\n");

    free(results);
}

int main() {
    int n = 5, k = 3;
    int arr1[] = {1, 5, 2, 6, 3};
    int arr2[] = {3, 5, 2, 8, 1};

    find_largest_products(arr1, arr2, n, k);

    return 0;
}
```
