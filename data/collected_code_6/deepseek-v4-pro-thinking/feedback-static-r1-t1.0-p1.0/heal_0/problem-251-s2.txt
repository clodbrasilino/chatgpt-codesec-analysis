#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

static Node *create_node(int data)
{
    Node *node = malloc(sizeof(*node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

static void print_list(const Node *head)
{
    const Node *curr = head;
    while (curr != NULL) {
        printf("%d ", curr->data);
        curr = curr->next;
    }
    printf("\n");
}

static void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

int insert_before_each(Node **head, int value)
{
    if (head == NULL) {
        return -1;
    }

    Node *curr = *head;
    Node *prev = NULL;

    while (curr != NULL) {
        Node *next = curr->next;
        Node *new_node = malloc(sizeof(*new_node));

        if (new_node == NULL) {
            return -1;
        }

        new_node->data = value;
        new_node->next = curr;

        if (prev == NULL) {
            *head = new_node;
        } else {
            prev->next = new_node;
        }

        prev = curr;
        curr = next;
    }

    return 0;
}

int main(void)
{
    Node *head = NULL;
    Node *n1 = create_node(1);
    if (n1 == NULL) {
        return EXIT_FAILURE;
    }

    Node *n2 = create_node(2);
    if (n2 == NULL) {
        free(n1);
        return EXIT_FAILURE;
    }

    Node *n3 = create_node(3);
    if (n3 == NULL) {
        free(n1);
        free(n2);
        return EXIT_FAILURE;
    }

    n1->next = n2;
    n2->next = n3;
    head = n1;

    if (insert_before_each(&head, 0) != 0) {
        free_list(head);
        return EXIT_FAILURE;
    }

    print_list(head);
    free_list(head);

    return EXIT_SUCCESS;
}
