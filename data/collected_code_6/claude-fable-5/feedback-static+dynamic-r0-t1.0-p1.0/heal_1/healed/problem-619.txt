```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *move_numbers_to_end(const char *input)
{
    size_t len;
    char *result;
    size_t i;
    size_t pos = 0;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < len; i++) {
        if (!isdigit((unsigned char)input[i])) {
            result[pos] = input[i];
            pos++;
        }
    }

    for (i = 0; i < len; i++) {
        if (isdigit((unsigned char)input[i])) {
            result[pos] = input[i];
            pos++;
        }
    }

    result[pos] = '\0';
    return result;
}

int main(void)
{
    const char *test = "a1b2c3d4e5";
    char *output;

    output = move_numbers_to_end(test);
    if (output == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\n", test) < 0) {
        free(output);
        return EXIT_FAILURE;
    }

    if (printf("Output: %s\n", output) < 0) {
        free(output);
        return EXIT_FAILURE;
    }

    free(output);
    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **`strlen` → `strnlen` with a bound (`MAX_INPUT_LEN`):** This addresses the Flawfinder CWE-126 warning. `strnlen` will never read beyond `MAX_INPUT_LEN` bytes, so a non-null-terminated buffer cannot cause an unbounded over-read.

2. **Rejection of over-long input:** If `strnlen` returns `MAX_INPUT_LEN`, we cannot confirm the string was properly terminated within the allowed bound, so the function returns `NULL` rather than processing potentially truncated or malformed data.

3. **Everything else preserved:** The original code was already solid in other respects — `malloc` result checked, `isdigit` correctly given an `unsigned char` cast (avoiding UB for negative `char` values), `NULL` input handled, buffer sized `len + 1` for the terminator, `printf` return values checked, and `output` freed on all paths. No changes were needed there.
