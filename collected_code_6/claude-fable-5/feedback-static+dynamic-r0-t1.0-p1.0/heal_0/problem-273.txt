```c
#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

int subtract_tuples(const int *tuple1, const int *tuple2, int *result, size_t length)
{
    size_t i;

    if (tuple1 == NULL || tuple2 == NULL || result == NULL || length == 0) {
        return -1;
    }

    for (i = 0; i < length; i++) {
        result[i] = tuple1[i] - tuple2[i];
    }

    return 0;
}

int main(void)
{
    const int tuple1[] = {10, 4, 5, 12, 9};
    const int tuple2[] = {2, 5, 18, 24, 11};
    size_t length = sizeof(tuple1) / sizeof(tuple1[0]);
    int *result;
    size_t i;
    int status;

    result = malloc(length * sizeof(*result));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    status = subtract_tuples(tuple1, tuple2, result, length);
    if (status != 0) {
        fprintf(stderr, "Subtraction failed due to invalid input\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("Result: (");
    for (i = 0; i < length; i++) {
        printf("%d", result[i]);
        if (i < length - 1) {
            printf(", ");
        }
    }
    printf(")\n");

    free(result);
    result = NULL;

    return EXIT_SUCCESS;
}
```
