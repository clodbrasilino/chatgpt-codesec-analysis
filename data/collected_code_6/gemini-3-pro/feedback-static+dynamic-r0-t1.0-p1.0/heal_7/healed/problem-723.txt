#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int a;
    int b;
} Pair;

int compare_pairs(const void* p1, const void* p2) {
    Pair* pair1 = (Pair*)p1;
    Pair* pair2 = (Pair*)p2;
    if (pair1->a != pair2->a) {
        return (pair1->a < pair2->a) ? -1 : 1;
    }
    if (pair1->b != pair2->b) {
        return (pair1->b < pair2->b) ? -1 : 1;
    }
    return 0;
}

int main(void) {
    int size_in;
    if (scanf("%d", &size_in) != 1 || size_in <= 0) {
        printf("0\n");
        return 0;
    }

    size_t size = (size_t)size_in;
    int* list1 = (int*)malloc(size * sizeof(int));
    int* list2 = (int*)malloc(size * sizeof(int));

    if (!list1 || !list2) {
        free(list1);
        free(list2);
        printf("0\n");
        return 0;
    }

    for (size_t i = 0; i < size; i++) {
        if (scanf("%d", &list1[i]) != 1) {
            list1[i] = 0;
        }
    }

    for (size_t i = 0; i < size; i++) {
        if (scanf("%d", &list2[i]) != 1) {
            list2[i] = 0;
        }
    }

    Pair* p1 = (Pair*)malloc(size * sizeof(Pair));
    Pair* p2 = (Pair*)malloc(size * sizeof(Pair));

    if (!p1 || !p2) {
        free(list1); free(list2);
        free(p1); free(p2);
        printf("0\n");
        return 0;
    }

    for (size_t i = 0; i < size; i++) {
        p1[i].a = list1[i];
        p1[i].b = list2[i];
        p2[i].a = list2[i];
        p2[i].b = list1[i];
    }

    qsort(p1, size, sizeof(Pair), compare_pairs);
    qsort(p2, size, sizeof(Pair), compare_pairs);

    int count = 0;
    size_t i = 0, j = 0;

    while (i < size && j < size) {
        int cmp = compare_pairs(&p1[i], &p2[j]);
        if (cmp == 0) {
            count++;
            i++;
            j++;
        } else if (cmp < 0) {
            i++;
        } else {
            j++;
        }
    }

    printf("%d\n", count);

    free(list1);
    free(list2);
    free(p1);
    free(p2);

    return 0;
}
