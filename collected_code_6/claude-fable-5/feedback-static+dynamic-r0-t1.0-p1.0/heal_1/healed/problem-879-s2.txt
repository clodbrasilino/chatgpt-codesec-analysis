```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

int match_pattern(const char *text)
{
    regex_t regex;
    int ret;

    ret = regcomp(&regex, "^a.*b$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        size_t errsize = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errsize);
        if (errbuf != NULL) {
            regerror(ret, &regex, errbuf, errsize);
            fprintf(stderr, "Failed to compile regex: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Failed to compile regex and could not allocate error buffer\n");
        }
        return -1;
    }

    ret = regexec(&regex, text, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex match failed\n");
        return -1;
    }
}

int main(void)
{
    const char *tests[] = {
        "ab",
        "acb",
        "a123b",
        "abc",
        "b",
        "a"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = match_pattern(tests[i]);
        if (result < 0) {
            return EXIT_FAILURE;
        }
        printf("\"%s\" -> %s\n", tests[i], result ? "match" : "no match");
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the change:**

The Flawfinder warning flagged the statically-sized `char errbuf[256]` as a potential CWE-119/CWE-120 buffer issue. While the original code was actually safe (it correctly passed `sizeof(errbuf)` to `regerror`, which guarantees truncation rather than overflow), the rewritten version eliminates the fixed-size buffer entirely:

1. **Two-pass `regerror` call**: Calling `regerror(ret, &regex, NULL, 0)` first returns the exact buffer size required for the full error message, including the null terminator.
2. **Dynamic allocation**: The buffer is then allocated with `malloc` to precisely that size, so the message can never be truncated and no static bound can be exceeded.
3. **Allocation failure handling**: If `malloc` fails, a fallback error message is printed and the function still returns `-1` cleanly, avoiding a NULL pointer dereference.
4. **No leaks**: `free(errbuf)` is called on the success path of the error branch, and `regfree(&regex)` continues to release the compiled pattern after `regexec`.

The rest of the program logic (match semantics, return codes, and the test harness in `main`) is unchanged and was already correct.
