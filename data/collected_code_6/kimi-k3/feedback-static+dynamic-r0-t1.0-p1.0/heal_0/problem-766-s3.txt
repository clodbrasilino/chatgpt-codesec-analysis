```c
#include <stdio.h>
#include <stdlib.h>

void iterate_consecutive_pairs(const int *arr, size_t size) {
    if (arr == NULL || size < 2) {
        return;
    }
    for (size_t i = 0; i < size - 1; ++i) {
        printf("(%d, %d) ", arr[i], arr[i + 1]);
    }
    printf("\n");
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(list) / sizeof(list[0]);
    iterate_consecutive_pairs(list, size);
    return EXIT_SUCCESS;
}
```
