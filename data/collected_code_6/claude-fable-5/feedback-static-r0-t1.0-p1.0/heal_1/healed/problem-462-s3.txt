```c
#include <stdio.h>
#include <stdlib.h>

static void print_combination(const int *arr, const int *indices, size_t k)
{
    size_t i;

    printf("{ ");
    for (i = 0; i < k; i++) {
        printf("%d ", arr[indices[i]]);
    }
    printf("}\n");
}

static int combinations_of_size(const int *arr, size_t n, size_t k)
{
    int *indices;
    size_t i;

    if (k == 0) {
        printf("{ }\n");
        return 0;
    }

    indices = malloc(k * sizeof(*indices));
    if (indices == NULL) {
        return -1;
    }

    for (i = 0; i < k; i++) {
        indices[i] = (int)i;
    }

    for (;;) {
        size_t pos;

        print_combination(arr, indices, k);

        pos = k;
        while (pos > 0) {
            pos--;
            if ((size_t)indices[pos] < n - (k - pos)) {
                indices[pos]++;
                for (i = pos + 1; i < k; i++) {
                    indices[i] = indices[i - 1] + 1;
                }
                break;
            }
            if (pos == 0) {
                free(indices);
                return 0;
            }
        }
    }
}

static int find_all_combinations(const int *arr, size_t n)
{
    size_t k;

    if (arr == NULL || n == 0) {
        return -1;
    }

    for (k = 0; k <= n; k++) {
        if (combinations_of_size(arr, n, k) != 0) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    int list[] = { 1, 2, 3, 4 };
    size_t n = sizeof(list) / sizeof(list[0]);

    if (find_all_combinations(list, n) != 0) {
        fprintf(stderr, "Failed to generate combinations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The fix applied addresses the `[variableScope]` weakness: the variable `pos` was declared at function scope in `combinations_of_size`, but it is only used inside the infinite `for (;;)` loop. Its declaration has been moved into that loop body, reducing its scope to the smallest region where it is actually needed. This improves readability and helps the compiler and static analyzers reason about the variable's lifetime.
