#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int add_k_to_tuple(const int *input, int *output, size_t size, int k)
{
    size_t i;

    if (input == NULL || output == NULL || size == 0U) {
        return -1;
    }

    for (i = 0U; i < size; i++) {
        if ((k > 0 && input[i] > INT_MAX - k) ||
            (k < 0 && input[i] < INT_MIN - k)) {
            return -1;
        }
        output[i] = input[i] + k;
    }

    return 0;
}

int main(void)
{
    int tuple[] = {1, 3, 4, 5, 6};
    size_t size = sizeof(tuple) / sizeof(tuple[0]);
    int k = 4;
    int *result;
    size_t i;

    result = (int *)malloc(size * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (add_k_to_tuple(tuple, result, size, k) != 0) {
        fprintf(stderr, "Failed to add k to tuple elements\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("Original tuple: (");
    for (i = 0U; i < size; i++) {
        printf("%d%s", tuple[i], (i < size - 1U) ? ", " : "");
    }
    printf(")\n");

    printf("Result tuple: (");
    for (i = 0U; i < size; i++) {
        printf("%d%s", result[i], (i < size - 1U) ? ", " : "");
    }
    printf(")\n");

    free(result);
    return EXIT_SUCCESS;
}
