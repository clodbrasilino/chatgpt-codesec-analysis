```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

size_t count_lists(Node *const *lists, size_t num_lists)
{
    size_t count = 0;

    if (lists == NULL) {
        return 0;
    }

    for (size_t i = 0; i < num_lists; i++) {
        if (lists[i] != NULL) {
            count++;
        }
    }

    return count;
}

Node *create_node(int data)
{
    Node *node = malloc(sizeof(*node));

    if (node == NULL) {
        return NULL;
    }

    node->data = data;
    node->next = NULL;
    return node;
}

void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

int main(void)
{
    const size_t num_lists = 4;
    Node **lists = calloc(num_lists, sizeof(*lists));

    if (lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < num_lists; i++) {
        if (i % 2 == 0) {
            lists[i] = create_node((int)(i + 1));
            if (lists[i] == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                for (size_t j = 0; j < num_lists; j++) {
                    free_list(lists[j]);
                }
                free(lists);
                return EXIT_FAILURE;
            }
        }
    }

    printf("Number of lists: %zu\n", count_lists(lists, num_lists));

    for (size_t i = 0; i < num_lists; i++) {
        free_list(lists[i]);
    }
    free(lists);

    return EXIT_SUCCESS;
}
```

**Fix explanation:** The parameter type was changed from `const Node *const *` to `Node *const *`. In C, a `Node **` implicitly converts to `Node *const *` (adding `const` to the pointer itself is safe), but **not** to `const Node *const *` — because allowing `T **` → `const T **` could permit writing a `const` object through a non-const alias. The function only reads `lists[i]` and never modifies the array elements, so `Node *const *` preserves the needed safety guarantee while eliminating the `-Wincompatible-pointer-types` warning without resorting to a cast.
