```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int *extract_consecutive(const int *list, size_t size, size_t start, size_t count, size_t *out_count)
{
    int *result;
    size_t i;

    if (out_count == NULL) {
        return NULL;
    }
    *out_count = 0;

    if (list == NULL || count == 0 || start >= size || count > size - start) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    result = (int *)malloc(count * sizeof(int));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count && (start + i) < size; i++) {
        result[i] = list[start + i];
    }
    *out_count = i;

    return result;
}

int main(void)
{
    int list[] = {5, 12, 7, 9, 21, 3, 18, 42};
    size_t size = sizeof(list) / sizeof(list[0]);
    size_t start = 2;
    size_t count = 4;
    size_t extracted_count = 0;
    size_t invalid_count = 0;
    size_t i;
    int *extracted;
    int *invalid;

    printf("Original list: ");
    for (i = 0; i < size; i++) {
        printf("%d ", list[i]);
    }
    printf("\n");

    extracted = extract_consecutive(list, size, start, count, &extracted_count);
    if (extracted == NULL) {
        fprintf(stderr, "Error: extraction failed (invalid arguments or allocation failure).\n");
        return EXIT_FAILURE;
    }

    printf("Extracted %zu elements starting at index %zu: ", extracted_count, start);
    for (i = 0; i < extracted_count; i++) {
        printf("%d ", extracted[i]);
    }
    printf("\n");

    free(extracted);
    extracted = NULL;

    invalid = extract_consecutive(list, size, 6, 5, &invalid_count);
    if (invalid == NULL) {
        printf("Out-of-range extraction request was correctly rejected.\n");
    } else {
        free(invalid);
        invalid = NULL;
    }

    return EXIT_SUCCESS;
}
```
