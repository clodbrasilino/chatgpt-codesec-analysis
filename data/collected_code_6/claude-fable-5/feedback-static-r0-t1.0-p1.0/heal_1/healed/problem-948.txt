```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TUPLE_MAX_ITEM_SIZE 4096
#define TEXT_MAX_LEN 256

typedef struct {
    void **items;
    size_t *sizes;
    size_t count;
} Tuple;

Tuple *tuple_create(size_t count)
{
    Tuple *t = NULL;

    if (count == 0) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->items = calloc(count, sizeof(void *));
    if (t->items == NULL) {
        free(t);
        return NULL;
    }

    t->sizes = calloc(count, sizeof(size_t));
    if (t->sizes == NULL) {
        free(t->items);
        free(t);
        return NULL;
    }

    t->count = count;
    return t;
}

int tuple_set_item(Tuple *t, size_t index, const void *data, size_t size)
{
    void *copy = NULL;

    if (t == NULL || data == NULL || size == 0 || size > TUPLE_MAX_ITEM_SIZE) {
        return -1;
    }

    if (index >= t->count) {
        return -1;
    }

    copy = malloc(size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, data, size);

    free(t->items[index]);
    t->items[index] = copy;
    t->sizes[index] = size;

    return 0;
}

const void *tuple_get_item(const Tuple *t, size_t index, size_t *size_out)
{
    if (t == NULL) {
        return NULL;
    }

    if (index >= t->count) {
        return NULL;
    }

    if (size_out != NULL) {
        *size_out = t->sizes[index];
    }

    return t->items[index];
}

void tuple_destroy(Tuple *t)
{
    size_t i;

    if (t == NULL) {
        return;
    }

    for (i = 0; i < t->count; i++) {
        free(t->items[i]);
    }

    free(t->items);
    free(t->sizes);
    free(t);
}

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0;

    if (s == NULL) {
        return 0;
    }

    while (len < max_len && s[len] != '\0') {
        len++;
    }

    return len;
}

int main(void)
{
    Tuple *t = NULL;
    int number = 42;
    double value = 3.14;
    const char *text = "hello";
    const void *item = NULL;
    size_t size = 0;
    size_t text_len = 0;

    t = tuple_create(3);
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 0, &number, sizeof(number)) != 0) {
        fprintf(stderr, "Failed to set item 0\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 1, &value, sizeof(value)) != 0) {
        fprintf(stderr, "Failed to set item 1\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    text_len = bounded_strlen(text, TEXT_MAX_LEN);
    if (text_len == 0 || text_len >= TEXT_MAX_LEN) {
        fprintf(stderr, "Invalid text length\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 2, text, text_len + 1) != 0) {
        fprintf(stderr, "Failed to set item 2\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    item = tuple_get_item(t, 0, &size);
    if (item != NULL && size == sizeof(int)) {
        printf("Item 0: %d\n", *(const int *)item);
    }

    item = tuple_get_item(t, 1, &size);
    if (item != NULL && size == sizeof(double)) {
        printf("Item 1: %f\n", *(const double *)item);
    }

    item = tuple_get_item(t, 2, &size);
    if (item != NULL && size > 0 && ((const char *)item)[size - 1] == '\0') {
        printf("Item 2: %s\n", (const char *)item);
    }

    item = tuple_get_item(t, 5, &size);
    if (item == NULL) {
        printf("Index 5 is out of bounds\n");
    }

    tuple_destroy(t);
    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **`memcpy` (CWE-120):** The original code was already safe here since `copy` is allocated with exactly `size` bytes immediately before the copy. To further harden it, I added an upper bound check (`TUPLE_MAX_ITEM_SIZE`) so an attacker-controlled or corrupted `size` value cannot trigger an excessive allocation or integer-related issues downstream.

2. **`strlen` (CWE-126):** Replaced the unbounded `strlen` with a `bounded_strlen` helper (equivalent to POSIX `strnlen`, implemented portably for strict ISO C). It caps traversal at `TEXT_MAX_LEN`, so a non-NUL-terminated buffer can no longer cause an over-read. The result is validated before use, rejecting both empty and truncated (unterminated) inputs.

3. **Defensive read of item 2:** Before printing the stored string with `%s`, the code now verifies the last stored byte is a NUL terminator, preventing an over-read if the tuple were ever populated with non-string data at that index.

The rest of the code (ownership semantics, cleanup paths, out-of-bounds handling) was already correct and leak-free, so it is preserved unchanged.
