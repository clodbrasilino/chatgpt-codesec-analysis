#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    int length;
} Tuple;

typedef struct {
    Tuple *tuples;
    int count;
} TupleList;

TupleList remove_tuples_of_length_k(TupleList list, int k) {
    TupleList result;
    result.count = 0;
    
    if (list.count == 0 || list.tuples == NULL) {
        result.tuples = NULL;
        return result;
    }

    result.tuples = (Tuple *)malloc(list.count * sizeof(Tuple));
    if (result.tuples == NULL) {
        return result;
    }

    for (int i = 0; i < list.count; i++) {
        if (list.tuples[i].length != k) {
            result.tuples[result.count].length = list.tuples[i].length;
            if (list.tuples[i].length > 0) {
                result.tuples[result.count].elements = (int *)malloc(list.tuples[i].length * sizeof(int));
                if (result.tuples[result.count].elements == NULL) {
                    for (int j = 0; j < result.count; j++) {
                        free(result.tuples[j].elements);
                    }
                    free(result.tuples);
                    result.tuples = NULL;
                    result.count = 0;
                    return result;
                }
                for (int j = 0; j < list.tuples[i].length; j++) {
                    result.tuples[result.count].elements[j] = list.tuples[i].elements[j];
                }
            } else {
                result.tuples[result.count].elements = NULL;
            }
            result.count++;
        }
    }

    if (result.count == 0) {
        free(result.tuples);
        result.tuples = NULL;
    } else {
        Tuple *temp = (Tuple *)realloc(result.tuples, result.count * sizeof(Tuple));
        if (temp != NULL) {
            result.tuples = temp;
        }
    }

    return result;
}

void free_tuple_list(TupleList *list) {
    if (list != NULL && list->tuples != NULL) {
        for (int i = 0; i < list->count; i++) {
            free(list->tuples[i].elements);
        }
        free(list->tuples);
        list->tuples = NULL;
        list->count = 0;
    }
}

int main(void) {
    TupleList list;
    list.count = 3;
    list.tuples = (Tuple *)malloc(list.count * sizeof(Tuple));
    if (list.tuples == NULL) {
        return 1;
    }

    list.tuples[0].length = 2;
    list.tuples[0].elements = (int *)malloc(2 * sizeof(int));
    if (list.tuples[0].elements != NULL) {
        list.tuples[0].elements[0] = 1;
        list.tuples[0].elements[1] = 2;
    }

    list.tuples[1].length = 3;
    list.tuples[1].elements = (int *)malloc(3 * sizeof(int));
    if (list.tuples[1].elements != NULL) {
        list.tuples[1].elements[0] = 1;
        list.tuples[1].elements[1] = 2;
        list.tuples[1].elements[2] = 3;
    }

    list.tuples[2].length = 2;
    list.tuples[2].elements = (int *)malloc(2 * sizeof(int));
    if (list.tuples[2].elements != NULL) {
        list.tuples[2].elements[0] = 4;
        list.tuples[2].elements[1] = 5;
    }

    TupleList filtered_list = remove_tuples_of_length_k(list, 2);

    for (int i = 0; i < filtered_list.count; i++) {
        for (int j = 0; j < filtered_list.tuples[i].length; j++) {
            printf("%d ", filtered_list.tuples[i].elements[j]);
        }
        printf("\n");
    }

    free_tuple_list(&list);
    free_tuple_list(&filtered_list);

    return 0;
}
