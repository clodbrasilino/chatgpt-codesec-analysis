```c
#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

long long sumSumAbsoluteDifferences(int* arr, size_t n) {
    if (arr == NULL || n == 0) {
        return 0;
    }

    qsort(arr, n, sizeof(int), compare);

    long long sum = 0;
    for (size_t i = 0; i < n; i++) {
        sum += (long long)arr[i] * i - (long long)arr[i] * (n - 1 - i);
    }

    return sum;
}

int main(void) {
    int arr[] = {1, 2, 3, 4};
    size_t n = sizeof(arr) / sizeof(arr[0]);

    long long result = sumSumAbsoluteDifferences(arr, n);
    printf("%lld\n", result);

    return 0;
}
```
