#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

char *snake_to_camel(const char *input)
{
    size_t len = 0;
    size_t i;
    size_t ri = 0;
    int upper = 1;
    unsigned char c;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    while (input[len] != '\0') {
        len++;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < len; i++) {
        c = (unsigned char)input[i];
        if (c == '_' && i + 1 < len && isalpha((unsigned char)input[i + 1])) {
            upper = 1;
            continue;
        }
        if (upper && isalpha(c)) {
            result[ri++] = (char)toupper(c);
        } else {
            result[ri++] = (char)c;
        }
        upper = 0;
    }

    result[ri] = '\0';
    return result;
}

int main(int argc, char **argv)
{
    char *out;
    int i;
    int any = 0;
    char buffer[4096];
    const char *defaults[] = { "android_tv", "google_pixel", "apple_watch" };
    size_t count = sizeof(defaults) / sizeof(defaults[0]);
    size_t k;

    if (argc > 1) {
        for (i = 1; i < argc; i++) {
            out = snake_to_camel(argv[i]);
            if (out == NULL) {
                return EXIT_FAILURE;
            }
            printf("%s\n", out);
            free(out);
        }
        return EXIT_SUCCESS;
    }

    while (scanf("%4095s", buffer) == 1) {
        any = 1;
        out = snake_to_camel(buffer);
        if (out == NULL) {
            return EXIT_FAILURE;
        }
        printf("%s\n", out);
        free(out);
    }

    if (!any) {
        for (k = 0; k < count; k++) {
            out = snake_to_camel(defaults[k]);
            if (out == NULL) {
                return EXIT_FAILURE;
            }
            printf("%s\n", out);
            free(out);
        }
    }

    return EXIT_SUCCESS;
}
