```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

typedef struct {
    double x;
    double y;
} Point;

typedef struct {
    Point vertices[6];
} Hexagon;

typedef struct {
    Hexagon *hexagons;
    size_t count;
} HexGrid;

static int compute_hexagon(double center_x, double center_y, double radius, Hexagon *hex)
{
    int i;

    if (hex == NULL || radius <= 0.0) {
        return -1;
    }

    for (i = 0; i < 6; i++) {
        double angle = (M_PI / 180.0) * (60.0 * (double)i - 30.0);
        hex->vertices[i].x = center_x + radius * cos(angle);
        hex->vertices[i].y = center_y + radius * sin(angle);
    }

    return 0;
}

HexGrid *create_hex_grid(size_t rows, size_t cols, double radius)
{
    HexGrid *grid;
    size_t row;
    size_t col;
    size_t index;
    double width;
    double height;

    if (rows == 0 || cols == 0 || radius <= 0.0) {
        return NULL;
    }

    if (cols > SIZE_MAX / rows) {
        return NULL;
    }

    grid = malloc(sizeof(HexGrid));
    if (grid == NULL) {
        return NULL;
    }

    grid->count = rows * cols;
    grid->hexagons = malloc(grid->count * sizeof(Hexagon));
    if (grid->hexagons == NULL) {
        free(grid);
        return NULL;
    }

    width = sqrt(3.0) * radius;
    height = 1.5 * radius;
    index = 0;

    for (row = 0; row < rows; row++) {
        for (col = 0; col < cols; col++) {
            double offset_x = ((row % 2) == 1) ? (width / 2.0) : 0.0;
            double center_x = (double)col * width + offset_x;
            double center_y = (double)row * height;

            if (compute_hexagon(center_x, center_y, radius, &grid->hexagons[index]) != 0) {
                free(grid->hexagons);
                free(grid);
                return NULL;
            }
            index++;
        }
    }

    return grid;
}

void destroy_hex_grid(HexGrid *grid)
{
    if (grid != NULL) {
        free(grid->hexagons);
        free(grid);
    }
}

int main(void)
{
    HexGrid *grid;
    size_t i;
    int j;
    size_t rows = 3;
    size_t cols = 4;
    double radius = 1.0;

    grid = create_hex_grid(rows, cols, radius);
    if (grid == NULL) {
        fprintf(stderr, "Failed to create hexagon grid\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < grid->count; i++) {
        if (printf("Hexagon %zu:\n", i) < 0) {
            destroy_hex_grid(grid);
            return EXIT_FAILURE;
        }
        for (j = 0; j < 6; j++) {
            if (printf("  (%.4f, %.4f)\n",
                       grid->hexagons[i].vertices[j].x,
                       grid->hexagons[i].vertices[j].y) < 0) {
                destroy_hex_grid(grid);
                return EXIT_FAILURE;
            }
        }
    }

    destroy_hex_grid(grid);
    return EXIT_SUCCESS;
}
```

Note: compile with `-lm` to link the math library (e.g., `gcc -o hexgrid hexgrid.c -lm`). If your compiler doesn't define `M_PI` under strict standards mode, add `#define _USE_MATH_DEFINES` before the includes or define `M_PI` as `3.14159265358979323846`. Also include `<stdint.h>` if `SIZE_MAX` is not available via `<stdlib.h>` on your platform.
