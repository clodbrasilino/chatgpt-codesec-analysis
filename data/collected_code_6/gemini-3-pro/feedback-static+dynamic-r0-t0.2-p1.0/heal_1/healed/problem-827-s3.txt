#include <stdio.h>
#include <stdlib.h>

int sum_column(int **list_of_lists, int num_rows, const int *row_lengths, int col_index) {
    if (list_of_lists == NULL || row_lengths == NULL || num_rows <= 0 || col_index < 0) {
        return 0;
    }

    int sum = 0;
    for (int i = 0; i < num_rows; i++) {
        if (list_of_lists[i] != NULL && col_index < row_lengths[i]) {
            sum += list_of_lists[i][col_index];
        }
    }

    return sum;
}

int main(void) {
    int num_rows = 3;
    int row_lengths[] = {3, 2, 4};
    
    int **list_of_lists = malloc(num_rows * sizeof(int *));
    if (list_of_lists == NULL) {
        return 1;
    }

    for (int i = 0; i < num_rows; i++) {
        list_of_lists[i] = malloc(row_lengths[i] * sizeof(int));
        if (list_of_lists[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(list_of_lists[j]);
            }
            free(list_of_lists);
            return 1;
        }
    }

    list_of_lists[0][0] = 1; 
    list_of_lists[0][1] = 2; 
    list_of_lists[0][2] = 3;
    list_of_lists[1][0] = 4; 
    list_of_lists[1][1] = 5;
    list_of_lists[2][0] = 6; 
    list_of_lists[2][1] = 7; 
    list_of_lists[2][2] = 8; 
    list_of_lists[2][3] = 9;

    int col_to_sum = 1;
    int sum = sum_column(list_of_lists, num_rows, row_lengths, col_to_sum);

    printf("%d\n", sum);

    for (int i = 0; i < num_rows; i++) {
        free(list_of_lists[i]);
    }
    free(list_of_lists);

    return 0;
}
