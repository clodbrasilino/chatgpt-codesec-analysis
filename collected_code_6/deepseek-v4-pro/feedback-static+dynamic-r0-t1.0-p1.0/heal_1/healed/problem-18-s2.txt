#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 1024

char *remove_chars(char *str1, const char *str2) {
    int table[256] = {0};
    size_t i, j = 0;
    
    if (str1 == NULL || str2 == NULL) {
        return str1;
    }
    
    for (i = 0; str2[i] != '\0'; i++) {
        table[(unsigned char)str2[i]] = 1;
    }
    
    for (i = 0; str1[i] != '\0'; i++) {
        if (!table[(unsigned char)str1[i]]) {
            str1[j++] = str1[i];
        }
    }
    
    str1[j] = '\0';
    return str1;
}

int main(void) {
    char *str1 = NULL;
    char *str2 = NULL;
    size_t len1 = 0;
    size_t len2 = 0;
    
    printf("Enter first string: ");
    if (getline(&str1, &len1, stdin) == -1) {
        free(str1);
        return 1;
    }
    str1[strcspn(str1, "\n")] = '\0';
    
    printf("Enter second string: ");
    if (getline(&str2, &len2, stdin) == -1) {
        free(str1);
        free(str2);
        return 1;
    }
    str2[strcspn(str2, "\n")] = '\0';
    
    remove_chars(str1, str2);
    printf("Result: %s\n", str1);
    
    free(str1);
    free(str2);
    
    return 0;
}
