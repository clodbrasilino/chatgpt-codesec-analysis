#include <stdio.h>

int most_significant_set_bit(unsigned int num) {
    if (num == 0) {
        return -1;
    }
    
    int msb = 0;
    while (num >>= 1) {
        msb++;
    }
    
    return msb;
}

int main(void) {
    unsigned int test_values[] = {0, 1, 2, 3, 4, 8, 16, 255, 256, 1024, 0x80000000, 0xFFFFFFFF};
    int num_tests = sizeof(test_values) / sizeof(test_values[0]);
    
    for (int i = 0; i < num_tests; i++) {
        int result = most_significant_set_bit(test_values[i]);
        
        if (result == -1) {
            printf("Number: %u, MSB position: No set bits\n", test_values[i]);
        } else {
            printf("Number: %u, MSB position: %d\n", test_values[i], result);
        }
    }
    
    return 0;
}
