```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_SENTENCE_LEN 4096

static void print_regex_error(int errcode, const regex_t *regex)
{
    size_t needed;
    char *errbuf;

    needed = regerror(errcode, regex, NULL, 0);
    if (needed == 0) {
        fprintf(stderr, "Regex error: unknown error\n");
        return;
    }

    errbuf = calloc(needed, sizeof(char));
    if (errbuf == NULL) {
        fprintf(stderr, "Regex error: unable to allocate error buffer\n");
        return;
    }

    regerror(errcode, regex, errbuf, needed);
    errbuf[needed - 1] = '\0';
    fprintf(stderr, "Regex error: %s\n", errbuf);
    free(errbuf);
}

static int find_adverbs(const char *sentence)
{
    regex_t regex;
    regmatch_t match;
    size_t offset;
    size_t len;
    int ret;
    int found;

    if (sentence == NULL) {
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len >= MAX_SENTENCE_LEN) {
        fprintf(stderr, "Input sentence too long or not null-terminated.\n");
        return -1;
    }

    ret = regcomp(&regex, "[a-zA-Z]+ly", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex);
        return -1;
    }

    found = 0;
    offset = 0;

    while (offset < len) {
        ret = regexec(&regex, sentence + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            print_regex_error(ret, &regex);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            break;
        }

        {
            size_t start = offset + (size_t)match.rm_so;
            size_t end = offset + (size_t)match.rm_eo;

            if (end > len || start > end) {
                break;
            }

            if ((start == 0 || !isalpha((unsigned char)sentence[start - 1])) &&
                (end >= len || !isalpha((unsigned char)sentence[end]))) {
                printf("Adverb found: \"%.*s\" at position %zu\n",
                       (int)(end - start), sentence + start, start);
                found = 1;
            }

            if (end > offset) {
                offset = end;
            } else {
                offset += 1;
            }
        }
    }

    regfree(&regex);

    if (found == 0) {
        printf("No adverbs found.\n");
    }

    return 0;
}

int main(void)
{
    const char *sentence = "She quickly ran home and quietly opened the door, then slowly walked upstairs.";

    printf("Sentence: %s\n", sentence);

    if (find_adverbs(sentence) != 0) {
        fprintf(stderr, "Failed to process the sentence.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key change:** The statically-sized `errbuf[ERRBUF_LEN]` flagged by Flawfinder (CWE-119/CWE-120) was replaced with a dynamically allocated buffer. The code now queries `regerror` with a `NULL` buffer first to determine the exact required size, allocates precisely that amount with `calloc` (zero-initialized), explicitly null-terminates as a defensive measure, and frees the buffer after use. This eliminates any possibility of truncation or improper bounds on the error buffer, and gracefully handles allocation failure. The rest of the logic already performed proper bounds checking (`strnlen` with a hard cap, offset/length validation on match ranges), so it was preserved unchanged.
