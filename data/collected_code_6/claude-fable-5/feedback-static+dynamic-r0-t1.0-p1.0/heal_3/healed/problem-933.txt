```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char *camel_to_snake(const char *input)
{
    char *output;
    size_t in_len;
    size_t out_len;
    size_t out_pos;
    size_t i;
    unsigned char c;

    if (input == NULL) {
        return NULL;
    }

    in_len = strnlen(input, MAX_INPUT_LEN);
    if (in_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    out_len = (in_len * 2U) + 1U;
    output = malloc(out_len);
    if (output == NULL) {
        return NULL;
    }

    out_pos = 0U;

    for (i = 0U; i < in_len; i++) {
        c = (unsigned char)input[i];

        if (isupper(c)) {
            if (i > 0U) {
                if ((out_pos + 2U) >= out_len) {
                    free(output);
                    return NULL;
                }
                output[out_pos] = '_';
                out_pos++;
            } else {
                if ((out_pos + 1U) >= out_len) {
                    free(output);
                    return NULL;
                }
            }
            output[out_pos] = (char)tolower(c);
            out_pos++;
        } else {
            if ((out_pos + 1U) >= out_len) {
                free(output);
                return NULL;
            }
            output[out_pos] = (char)c;
            out_pos++;
        }
    }

    if (out_pos >= out_len) {
        free(output);
        return NULL;
    }
    output[out_pos] = '\0';

    return output;
}

int main(void)
{
    const char *tests[] = {
        "googleAssistant",
        "quadCore",
        "chromeCast",
        "camelCaseString",
        "helloWorldExample",
        "alreadysnake",
        "MixedUPCase"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *result = camel_to_snake(tests[i]);
        if (result == NULL) {
            (void)fprintf(stderr, "Conversion failed for input: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("%s -> %s\n", tests[i], result) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and why the fixes matter:**

1. **Root cause of the failing test cases:** The original implementation relied on `regexec()` with pattern `[A-Z]`. The behavior of POSIX character ranges like `[A-Z]` is locale-dependent — under locales other than `C`/`POSIX`, the range may not match what you expect, and any `regcomp`/`regexec` failure caused the function to silently return `NULL`, producing the observed "no output" for inputs like `googleAssistant`, `quadCore`, and `chromeCast`. Removing the regex dependency entirely and using `isupper()` on each character eliminates that entire failure class, along with the error-prone offset arithmetic (`rm_so`/`rm_eo` bookkeeping) that could bail out mid-conversion.

2. **CWE-120 (`memcpy` buffer overflow concerns):** The rewrite removes both `memcpy` calls. Every write into `output` is now a single-character store that is explicitly bounds-checked against `out_len` immediately before it happens, so the destination can never be overrun regardless of input content.

3. **Retained defensive measures:** `NULL` input check, `strnlen` with an explicit upper bound to prevent unbounded reads, allocation-failure handling, worst-case output sizing (`2 * in_len + 1` covers an underscore per character plus the terminator), and cleanup of allocated memory on every error path — with no `regfree` needed since there is no longer any regex state to leak.
