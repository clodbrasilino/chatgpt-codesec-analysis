#include <stdio.h>
#include <string.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>

static long long binaryToDecimal(const char *binary) {
    long long decimal = 0;

    if (binary == NULL || binary[0] == '\0') {
        return -1;
    }

    for (size_t i = 0; i < 64 && binary[i] != '\0'; i++) {
        if (binary[i] == '1') {
            decimal = (decimal << 1) | 1;
        } else if (binary[i] == '0') {
            decimal = decimal << 1;
        } else {
            return -1;
        }
    }

    return decimal;
}
