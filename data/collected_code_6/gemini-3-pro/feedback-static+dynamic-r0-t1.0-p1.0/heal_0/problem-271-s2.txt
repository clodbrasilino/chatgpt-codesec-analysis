#include <stdio.h>
#include <stdlib.h>

unsigned long long sum_of_fifth_power_of_even_numbers(int n) {
    if (n < 0) {
        return 0;
    }
    
    unsigned long long sum = 0;
    
    for (int i = 1; i <= n; i++) {
        unsigned long long even_num = 2ULL * i;
        unsigned long long power = even_num * even_num * even_num * even_num * even_num;
        sum += power;
    }
    
    return sum;
}

int main(void) {
    int n;
    
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (n < 0) {
        fprintf(stderr, "Number must be non-negative\n");
        return EXIT_FAILURE;
    }
    
    unsigned long long result = sum_of_fifth_power_of_even_numbers(n);
    printf("%llu\n", result);
    
    return EXIT_SUCCESS;
}
