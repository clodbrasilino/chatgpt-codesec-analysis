#include <stdio.h>
#include <stdlib.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

int maxSumNoThreeConsecutive(int arr[], int n) {
    if (n == 0) {
        return 0;
    }
    if (n == 1) {
        return arr[0];
    }
    if (n == 2) {
        return arr[0] + arr[1];
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        exit(EXIT_FAILURE);
    }

    dp[0] = arr[0];
    dp[1] = arr[0] + arr[1];
    dp[2] = max(max(dp[1], arr[1] + arr[2]), arr[0] + arr[2]);

    for (int i = 3; i < n; i++) {
        dp[i] = max(max(dp[i - 1], dp[i - 2] + arr[i]), arr[i] + arr[i - 1] + dp[i - 3]);
    }

    int result = dp[n - 1];
    free(dp);
    return result;
}

int main(void) {
    int n;
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }

    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        return EXIT_FAILURE;
    }

    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            free(arr);
            return EXIT_FAILURE;
        }
    }

    int result = maxSumNoThreeConsecutive(arr, n);
    printf("%d\n", result);

    free(arr);
    return EXIT_SUCCESS;
}
