```c
#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

size_t count_true_booleans(const bool *arr, size_t len) {
    size_t count = 0;
    for (size_t i = 0; i < len; ++i) {
        if (arr[i]) {
            ++count;
        }
    }
    return count;
}

int main() {
    bool arr[] = { true, false, true, true, false, true, false };
    size_t len = sizeof(arr) / sizeof(arr[0]);
    size_t true_count = count_true_booleans(arr, len);
    printf("Number of true booleans: %zu\n", true_count);
    return 0;
}
```
