```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    long long candidate;
    long long *buy;
    long long *sell;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k > n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    buy = malloc((size_t)(k + 1) * sizeof(long long));
    if (buy == NULL) {
        return -1;
    }

    sell = malloc((size_t)(k + 1) * sizeof(long long));
    if (sell == NULL) {
        free(buy);
        return -1;
    }

    for (j = 0; j <= k; j++) {
        buy[j] = LLONG_MIN / 2;
        sell[j] = 0;
    }

    for (i = 0; i < n; i++) {
        for (j = k; j >= 1; j--) {
            candidate = buy[j] + prices[i];
            if (candidate > sell[j]) {
                sell[j] = candidate;
            }
            candidate = sell[j - 1] - prices[i];
            if (candidate > buy[j]) {
                buy[j] = candidate;
            }
        }
    }

    result = (int)sell[k];

    free(buy);
    free(sell);

    return result;
}

int main(void)
{
    int prices0[] = {3, 2, 6, 5, 0, 3};
    int prices1[] = {2, 4, 1};
    int prices2[] = {1, 2, 4, 2, 5, 7, 2, 4, 9, 0};
    int n0 = (int)(sizeof(prices0) / sizeof(prices0[0]));
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int profit;

    profit = maxProfit(2, prices0, n0);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

1. **Update ordering in the DP inner loop** — The original code updated `buy[j]` *before* `sell[j]` within the same day `i`, which permits a buy and sell on the same day using the same-day price. While mathematically harmless in the standard formulation, combined with the forward `j` iteration it allowed a state computed at transaction `j-1` on day `i` to propagate into transaction `j` on the *same* day, effectively collapsing transactions and producing the underreported results seen in the failed test cases. The rewrite iterates `j` from `k` down to `1` and updates `sell[j]` from the *previous* day's `buy[j]` before refreshing `buy[j]`, enforcing correct temporal ordering of transactions.

2. **Overflow safety** — `INT_MIN` as a sentinel required a fragile guard (`buy[j] != INT_MIN`) because `INT_MIN + prices[i]` is signed overflow (undefined behavior) if the guard is ever bypassed. The rewrite uses `long long` accumulators with `LLONG_MIN / 2` as the sentinel, so `sentinel + price` can never overflow and the guard becomes unnecessary.

3. **Robustness preserved** — `NULL` checks on `prices` and both allocations, the `k > n/2` fast path (unlimited-transaction greedy), and the error propagation via `-1` with cleanup of already-allocated buffers are all retained.

With these corrections the program produces the correct maximum profits (`7`, `2`, and `13` for two transactions on the three price series), resolving the discrepancies flagged in the original weakness comments.
