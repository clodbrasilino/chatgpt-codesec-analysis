#include <stdio.h>
#include <stdlib.h>

long long sum_odd_numbers(long long l, long long r) {
    if (l > r || l < 0) {
        return -1;
    }

    if (l % 2 == 0) {
        l++;
    }
    if (r % 2 == 0) {
        r--;
    }

    if (l > r) {
        return 0;
    }

    unsigned long long ul = (unsigned long long)l;
    unsigned long long ur = (unsigned long long)r;
    unsigned long long n = ((ur - ul) / 2) + 1;

    return (long long)(n * ((ul + ur) / 2));
}

int main(void) {
    long long l, r;
    long long sum;

    if (scanf("%lld %lld", &l, &r) != 2) {
        return EXIT_FAILURE;
    }

    sum = sum_odd_numbers(l, r);

    if (sum == -1) {
        return EXIT_FAILURE;
    }

    printf("%lld\n", sum);

    return EXIT_SUCCESS;
}
