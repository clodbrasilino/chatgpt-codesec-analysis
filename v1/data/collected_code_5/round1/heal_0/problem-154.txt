```C
#include <stdio.h>
#include <stdlib.h>

#define ROWS 4
#define COLS 4

int** extract_elements(int** arr, int size, int index) {
    int** extracted_elements = (int**) malloc(size * sizeof(int*));

    for(int i=0; i<size; i++) {
        if (arr[i] != NULL) {
            extracted_elements[i] = (int*) malloc(sizeof(int));
            if (index >= 0 && index < COLS) {
                *extracted_elements[i] = arr[i][index];
            } else {
                free(extracted_elements[i]);
                extracted_elements[i] = NULL;
            }
        } else {
            extracted_elements[i] = NULL;
        }
    }
    return extracted_elements;
}

int main() {
    int** array = (int**) malloc(ROWS * sizeof(int*));
    for(int i=0; i<ROWS; i++) {
        array[i] = (int*) malloc(COLS * sizeof(int));
        for(int j=0; j<COLS; j++) {
            array[i][j] = i*j;
        }
    }

    int** extracted_elements = extract_elements(array, ROWS, 2);
    for(int i=0; i<ROWS; i++) {
        if (extracted_elements[i] != NULL) {
            printf("%d\n", *extracted_elements[i]);
            free(extracted_elements[i]);
        }
    }

    for(int i=0; i<ROWS; i++) {
        if (array[i] != NULL) 
            free(array[i]);
    }

    free(extracted_elements);
    free(array);

    return 0;
}
```
This program creates a 2D array dynamically and fills it with values. Then, it extracts a particular column of the 2D array into another dynamically allocated 1D array and prints it, handling all the exceptions carefully such as null pointers and out of bounds array indexing. While applying proper check for all dynamic allocations, freeing all resources after usage to avoid memory leaks. It doesn't ignore any returned values and doesn't apply any bad coding practices.
