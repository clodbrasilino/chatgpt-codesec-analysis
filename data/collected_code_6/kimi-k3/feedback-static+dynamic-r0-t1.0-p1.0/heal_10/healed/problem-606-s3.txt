```c
#include <stdio.h>
#include <math.h>
#include <stdlib.h>
#include <errno.h>
#include <string.h>
#include <ctype.h>
#include <limits.h>

double degrees_to_radians(double degrees) {
    return degrees * (M_PI / 180.0);
}

int main(void) {
    char input_buffer[128];
    char *end_ptr;
    double degrees;
    double radians;
    size_t len;
    size_t buffer_size = sizeof(input_buffer);

    if (fgets(input_buffer, buffer_size, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    len = strnlen(input_buffer, buffer_size);
    
    if (len == 0) {
        fprintf(stderr, "Empty input\n");
        return EXIT_FAILURE;
    }

    if (input_buffer[len - 1] == '\n') {
        input_buffer[len - 1] = '\0';
        len--;
    } else if (len == buffer_size - 1) {
        int c;
        int overflow = 0;
        while ((c = getchar()) != '\n' && c != EOF) {
            overflow = 1;
        }
        if (overflow || c == EOF) {
            fprintf(stderr, "Input too long\n");
            return EXIT_FAILURE;
        }
    }

    if (len == 0) {
        fprintf(stderr, "Empty input\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    degrees = strtod(input_buffer, &end_ptr);

    if (end_ptr == input_buffer) {
        fprintf(stderr, "No digits were found\n");
        return EXIT_FAILURE;
    }

    if (errno == ERANGE) {
        fprintf(stderr, "Input value out of range\n");
        return EXIT_FAILURE;
    }

    if (degrees > HUGE_VAL || degrees < -HUGE_VAL) {
        fprintf(stderr, "Input value out of range\n");
        return EXIT_FAILURE;
    }

    while (*end_ptr != '\0' && isspace((unsigned char)*end_ptr)) {
        end_ptr++;
    }

    if (*end_ptr != '\0') {
        fprintf(stderr, "Invalid trailing characters\n");
        return EXIT_FAILURE;
    }

    radians = degrees_to_radians(degrees);

    if (!isfinite(radians)) {
        fprintf(stderr, "Result out of range\n");
        return EXIT_FAILURE;
    }

    printf("%f degrees is %f radians\n", degrees, radians);

    return EXIT_SUCCESS;
}
```
