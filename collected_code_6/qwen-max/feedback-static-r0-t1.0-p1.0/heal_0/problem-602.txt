#include <stdio.h>
#include <string.h>
#include <stdbool.h>

char find_first_repeated_char(const char *str) {
    int hash[256] = {0};
    for (int i = 0; str[i] != '\0'; ++i) {
        if (hash[(unsigned char)str[i]] == 1) {
            return str[i];
        }
        hash[(unsigned char)str[i]] = 1;
    }
    return '\0';
}

int main() {
    const char *test_str = "programming";
    char result = find_first_repeated_char(test_str);
    if (result != '\0') {
        printf("First repeated character: %c\n", result);
    } else {
        printf("No repeated characters found.\n");
    }
    return 0;
}
