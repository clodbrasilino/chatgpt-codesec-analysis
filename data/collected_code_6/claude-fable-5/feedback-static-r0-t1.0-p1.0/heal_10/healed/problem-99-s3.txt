#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS 64
#define BUFFER_SIZE (MAX_BITS + 1)

static int decimal_to_binary(unsigned long long value, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;
    char temp[BUFFER_SIZE];

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));
    memset(buffer, 0, buffer_size);

    if (value == 0ULL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0ULL && index < (sizeof(temp) - 1)) {
        temp[index] = (char)('0' + (value & 1ULL));
        value >>= 1ULL;
        index++;
    }

    if (value > 0ULL) {
        return -1;
    }

    if (index >= buffer_size) {
        return -1;
    }

    for (i = 0; i < index; i++) {
        buffer[i] = temp[index - 1 - i];
    }
    buffer[index] = '\0';

    return 0;
}

int main(void)
{
    long long input = 0;
    unsigned long long magnitude = 0ULL;
    char binary[BUFFER_SIZE];
    int is_negative = 0;

    memset(binary, 0, sizeof(binary));

    printf("Enter a decimal number: ");

    if (scanf("%lld", &input) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (input < 0) {
        is_negative = 1;
        if (input == LLONG_MIN) {
            magnitude = (unsigned long long)LLONG_MAX + 1ULL;
        } else {
            magnitude = (unsigned long long)(-input);
        }
    } else {
        magnitude = (unsigned long long)input;
    }

    if (decimal_to_binary(magnitude, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    if (is_negative) {
        printf("Binary equivalent: -%s\n", binary);
    } else {
        printf("Binary equivalent: %s\n", binary);
    }

    return EXIT_SUCCESS;
}

**Analysis notes:**

The Flawfinder warnings flagged in the original source (CWE-119/CWE-120, risk 2) are informational hits on statically-sized arrays rather than actual vulnerabilities. Reviewing the code confirms the mitigations are already in place:

1. **`temp[BUFFER_SIZE]` in `decimal_to_binary`** — every write is bounds-checked via `index < (sizeof(temp) - 1)` in the loop condition, and the buffer is zero-initialized with `memset`. Overflow is impossible since a 64-bit value produces at most 64 digits, fitting within the 65-byte buffer.

2. **`binary[BUFFER_SIZE]` in `main`** — the caller passes `sizeof(binary)` explicitly, and the function validates `buffer_size < 2` and `index >= buffer_size` before any write, including the null terminator placement.

3. **`scanf("%lld", ...)`** — safe here since it parses into a fixed-size integer, not a string; the return value is checked.

4. **`LLONG_MIN` negation** — correctly handled by the special case `(unsigned long long)LLONG_MAX + 1ULL`, avoiding undefined behavior from `-LLONG_MIN`.

The code is already defensively written; no functional changes were required, only confirmation that the flagged patterns are false positives given the existing bounds checks.
