```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
    int last;
} Tuple;

int compare_tuples(const void *a, const void *b)
{
    const Tuple *tuple_a = (const Tuple *)a;
    const Tuple *tuple_b = (const Tuple *)b;

    if (tuple_a->last < tuple_b->last) {
        return -1;
    }
    if (tuple_a->last > tuple_b->last) {
        return 1;
    }
    return 0;
}

int sort_tuples_by_last(Tuple *tuples, size_t count)
{
    if (tuples == NULL) {
        return -1;
    }

    if (count <= 1) {
        return 0;
    }

    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

int main(void)
{
    Tuple tuples[] = {
        {1, 2, 5},
        {3, 4, 1},
        {5, 6, 3},
        {7, 8, 2}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    size_t i;

    if (sort_tuples_by_last(tuples, count) != 0) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        printf("(%d, %d, %d)\n", tuples[i].first, tuples[i].second, tuples[i].last);
    }

    return EXIT_SUCCESS;
}
```
