#include <stdio.h>
#include <string.h>

void remove_chars(char *str1, const char *str2) {
    int i, j, k;
    for (i = 0; str1[i] != '\0'; i++) {
        for (j = 0; str2[j] != '\0'; j++) {
            if (str1[i] == str2[j]) {
                for (k = i; str1[k] != '\0'; k++) {
                    str1[k] = str1[k + 1];
                }
                str1[k - 1] = '\0';
                i--;
                break;
            }
        }
    }
}

int main() {
    char str1[100] = "hello world";
    const char *str2 = "lo";
    remove_chars(str1, str2);
    printf("%s\n", str1);
    return 0;
}
