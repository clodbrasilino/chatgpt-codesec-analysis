#include <stdio.h>
#include <stdlib.h>

double power(double a, int b) {
    double result = 1.0;
    int abs_b = b < 0 ? -b : b;

    for (int i = 0; i < abs_b; i++) {
        result *= a;
    }

    if (b < 0) {
        if (result == 0.0) {
            fprintf(stderr, "Error: Division by zero.\n");
            exit(EXIT_FAILURE);
        }
        result = 1.0 / result;
    }

    return result;
}

int main(void) {
    double a;
    int b;
    double result;

    if (scanf("%lf %d", &a, &b) != 2) {
        fprintf(stderr, "Error: Invalid input.\n");
        return EXIT_FAILURE;
    }

    if (a == 0.0 && b < 0) {
        fprintf(stderr, "Error: Division by zero.\n");
        return EXIT_FAILURE;
    }

    result = power(a, b);

    printf("%.6f\n", result);

    return EXIT_SUCCESS;
}
