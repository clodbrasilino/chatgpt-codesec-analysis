#include <stdio.h>
#include <stdlib.h>

long long max_sum(long long n, long long *memo, long long memo_size) {
    if (n == 0) {
        return 0;
    }
    if (n < memo_size && memo[n] != -1) {
        return memo[n];
    }

    long long sum = max_sum(n / 2, memo, memo_size) +
                    max_sum(n / 3, memo, memo_size) +
                    max_sum(n / 4, memo, memo_size) +
                    max_sum(n / 5, memo, memo_size);

    long long result = (sum > n) ? sum : n;

    if (n < memo_size) {
        memo[n] = result;
    }

    return result;
}

int main(void) {
    long long n;
    if (scanf("%lld", &n) != 1) {
        return 1;
    }

    if (n < 0) {
        return 1;
    }

    long long memo_size = (n < 1000000) ? (n + 1) : 1000000;
    long long *memo = (long long *)malloc(memo_size * sizeof(long long));
    if (memo == NULL) {
        return 1;
    }

    for (long long i = 0; i < memo_size; i++) {
        memo[i] = -1;
    }

    long long result = max_sum(n, memo, memo_size);
    printf("%lld\n", result);

    free(memo);
    return 0;
}
