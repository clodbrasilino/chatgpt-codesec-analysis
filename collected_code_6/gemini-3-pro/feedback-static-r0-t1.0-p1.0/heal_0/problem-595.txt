#include <stdio.h>
#include <string.h>

int min_swaps(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }

    size_t len1 = strlen(str1);
    size_t len2 = strlen(str2);

    if (len1 != len2) {
        return -1;
    }

    int count_0_to_1 = 0;
    int count_1_to_0 = 0;

    for (size_t i = 0; i < len1; ++i) {
        if ((str1[i] != '0' && str1[i] != '1') || 
            (str2[i] != '0' && str2[i] != '1')) {
            return -1;
        }

        if (str1[i] == '0' && str2[i] == '1') {
            count_0_to_1++;
        } else if (str1[i] == '1' && str2[i] == '0') {
            count_1_to_0++;
        }
    }

    if (count_0_to_1 != count_1_to_0) {
        return -1;
    }

    return count_0_to_1;
}

int main(void) {
    const char *str1 = "1100";
    const char *str2 = "1001";

    int result = min_swaps(str1, str2);

    if (result != -1) {
        printf("Minimum swaps required: %d\n", result);
    } else {
        printf("Invalid input strings or conversion not possible.\n");
    }

    return 0;
}
