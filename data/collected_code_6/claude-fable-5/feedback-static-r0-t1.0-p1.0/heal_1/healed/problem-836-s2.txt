```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static int add_would_overflow(long long a, long long b)
{
    if (b > 0 && a > LLONG_MAX - b) {
        return 1;
    }
    if (b < 0 && a < LLONG_MIN - b) {
        return 1;
    }
    return 0;
}

size_t max_sum_subarray_length(const long long *arr, size_t n)
{
    long long max_sum;
    long long current_sum;
    size_t max_len;
    size_t current_len;
    size_t i;

    if (arr == NULL || n == 0U) {
        return 0U;
    }

    max_sum = arr[0];
    current_sum = arr[0];
    max_len = 1U;
    current_len = 1U;

    for (i = 1U; i < n; i++) {
        if (current_sum < 0LL || add_would_overflow(current_sum, arr[i])) {
            current_sum = arr[i];
            current_len = 1U;
        } else {
            current_sum += arr[i];
            current_len++;
        }

        if (current_sum > max_sum) {
            max_sum = current_sum;
            max_len = current_len;
        } else if (current_sum == max_sum && current_len > max_len) {
            max_len = current_len;
        }
    }

    return max_len;
}

int main(void)
{
    long long arr[] = { -2LL, 1LL, -3LL, 4LL, -1LL, 2LL, 1LL, -5LL, 4LL };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t result;

    result = max_sum_subarray_length(arr, n);

    if (result == 0U) {
        if (printf("Invalid input or empty array\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Length of subarray with maximum sum: %zu\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Removed the invalid overflow test** — the original `current_sum + arr[i] < arr[i]` relies on signed integer overflow wrapping, which is undefined behavior in C. Optimizing compilers (GCC, Clang with `-O2`) are entitled to assume overflow never happens and may reduce that expression to `current_sum < 0`, silently removing the intended overflow "handling."

2. **Added `add_would_overflow()`** — a portable, UB-free pre-condition check using `LLONG_MAX`/`LLONG_MIN` bounds *before* performing the addition. This is the canonical safe pattern (equivalent to CERT INT32-C guidance applied to `long long`).

3. **Made the Kadane reset condition explicit** — `current_sum < 0LL` is the mathematically intended condition (a negative running sum never helps a future subarray). The overflow guard is a separate, explicit safety check rather than being conflated with the algorithm's logic, so behavior is now well-defined for all inputs including extreme values near `LLONG_MAX`/`LLONG_MIN`.

The rest of the code (NULL/empty guards, `size_t` usage, `printf` return-value checks, `EXIT_SUCCESS`/`EXIT_FAILURE`) was already sound and is preserved. Expected output: `Length of subarray with maximum sum: 4` (the subarray `{4, -1, 2, 1}` summing to 6).
