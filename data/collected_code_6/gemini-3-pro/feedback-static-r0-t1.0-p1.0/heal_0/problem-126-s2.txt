#include <stdio.h>
#include <stdlib.h>

int gcd(int a, int b) {
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int sum_of_common_divisors(int num1, int num2) {
    if (num1 == 0 && num2 == 0) {
        return 0;
    }

    int common_gcd = gcd(num1, num2);
    int sum = 0;

    for (int i = 1; i <= common_gcd; i++) {
        if (common_gcd % i == 0) {
            sum += i;
        }
    }

    return sum;
}

int main(void) {
    int num1, num2;
    
    if (scanf("%d %d", &num1, &num2) != 2) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    int result = sum_of_common_divisors(num1, num2);
    printf("%d\n", result);

    return EXIT_SUCCESS;
}
