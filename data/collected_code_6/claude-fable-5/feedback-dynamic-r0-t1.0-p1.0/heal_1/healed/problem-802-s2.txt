```c
#include <stdio.h>
#include <stdlib.h>

int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;
    size_t mid;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        if (arr[low] <= arr[high]) {
            return (int)low;
        }

        mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else {
            high = mid;
        }
    }

    return (int)low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    int arr3[] = {1, 2, 3, 4, 5};
    int result;

    result = count_rotations(arr1, sizeof(arr1) / sizeof(arr1[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr1\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr1: %d\n", result);

    result = count_rotations(arr2, sizeof(arr2) / sizeof(arr2[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr2\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr2: %d\n", result);

    result = count_rotations(arr3, sizeof(arr3) / sizeof(arr3[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr3\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr3: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the original weakness:**

The failing test case (`expected 2, got 3`) points to fragility in the original algorithm's minimum-detection logic. The original code relied on checking `arr[mid] <= arr[next] && arr[mid] <= arr[prev]` with modular wraparound indexing, combined with `high = mid - 1U` when discarding the right half. That combination is brittle: excluding `mid` from the search range (`high = mid - 1U`) can skip over the actual pivot when the neighbor check doesn't fire on the right iteration, and the modular `prev`/`next` computation wraps around the array boundary, which can incorrectly validate a candidate against elements from the opposite end of the rotation.

**Fixes applied in the rewrite:**

1. **Simplified invariant-based binary search:** compare `arr[mid]` against `arr[high]`. If `arr[mid] > arr[high]`, the pivot must lie strictly to the right (`low = mid + 1U`); otherwise `mid` itself may be the pivot, so it stays in range (`high = mid`), never discarding a potential answer.
2. **Removed modular neighbor checks** (`next`, `prev`), eliminating the wraparound comparison hazard entirely.
3. **Loop condition changed to `low < high`**, guaranteeing termination with `low == high` pointing at the minimum, so the function always returns a valid index for non-empty input — the `-1` fall-through for valid arrays is gone.
4. **Retained the early-exit** for already-sorted subranges (`arr[low] <= arr[high]`) and the `NULL`/zero-length guards, and kept `size_t` arithmetic underflow-safe (no `mid - 1U` on an unsigned zero).
