#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <ctype.h>

char* snake_to_camel(const char* snake_str) {
    if (snake_str == NULL) {
        return NULL;
    }

    regex_t regex;
    int reti;
    const char* pattern = "_([a-z])";
    size_t nmatch = 2;
    regmatch_t pmatch[2];

    reti = regcomp(&regex, pattern, REG_EXTENDED);
    if (reti) {
        return NULL;
    }

    size_t len = strnlen(snake_str, 1048576);
    char* camel_str = (char*)malloc(len + 1);
    if (camel_str == NULL) {
        regfree(&regex);
        return NULL;
    }

    const char* cursor = snake_str;
    char* dest = camel_str;
    size_t remaining_src = len;
    size_t remaining_dest = len;

    while (regexec(&regex, cursor, nmatch, pmatch, 0) == 0) {
        size_t prefix_len = pmatch[0].rm_so;
        
        if (prefix_len > 0) {
            size_t copy_len = (prefix_len < remaining_dest) ? prefix_len : remaining_dest;
            memcpy(dest, cursor, copy_len);
            dest += copy_len;
            remaining_dest -= copy_len;
        }

        if (remaining_dest > 0) {
            *dest = toupper((unsigned char)cursor[pmatch[1].rm_so]);
            dest++;
            remaining_dest--;
        }

        size_t step = pmatch[0].rm_eo;
        cursor += step;
        
        if (step > remaining_src) {
            remaining_src = 0;
        } else {
            remaining_src -= step;
        }
    }

    if (remaining_src > 0) {
        size_t copy_len = (remaining_src < remaining_dest) ? remaining_src : remaining_dest;
        memcpy(dest, cursor, copy_len);
        dest += copy_len;
    }
    *dest = '\0';

    regfree(&regex);
    return camel_str;
}

int main(void) {
    const char* snake = "this_is_a_snake_case_string";
    char* camel = snake_to_camel(snake);

    if (camel != NULL) {
        printf("Snake: %s\n", snake);
        printf("Camel: %s\n", camel);
        free(camel);
    } else {
        printf("Conversion failed or memory allocation error.\n");
    }

    return 0;
}
