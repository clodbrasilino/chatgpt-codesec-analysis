```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int extract_min_max_k(const int *tuple, size_t n, size_t k,
                             int *min_out, int *max_out)
{
    int *sorted;
    size_t i;
    size_t bytes;

    if (tuple == NULL || min_out == NULL || max_out == NULL) {
        return -1;
    }
    if (n == 0 || k == 0 || k > n) {
        return -1;
    }
    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    bytes = n * sizeof(int);

    sorted = malloc(bytes);
    if (sorted == NULL) {
        return -1;
    }

    memcpy(sorted, tuple, bytes);
    qsort(sorted, n, sizeof(int), compare_ints);

    for (i = 0; i < k; i++) {
        min_out[i] = sorted[i];
        max_out[i] = sorted[n - 1 - i];
    }

    free(sorted);
    return 0;
}

int main(void)
{
    int tuple[] = { 5, 20, 3, 7, 6, 8, 100, -2, 15 };
    size_t n = sizeof(tuple) / sizeof(tuple[0]);
    size_t k = 3;
    int *min_k;
    int *max_k;
    size_t i;

    if (k == 0 || k > n || k > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Invalid value of k\n");
        return EXIT_FAILURE;
    }

    min_k = malloc(k * sizeof(int));
    if (min_k == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    max_k = malloc(k * sizeof(int));
    if (max_k == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(min_k);
        return EXIT_FAILURE;
    }

    if (extract_min_max_k(tuple, n, k, min_k, max_k) != 0) {
        fprintf(stderr, "Failed to extract elements\n");
        free(min_k);
        free(max_k);
        return EXIT_FAILURE;
    }

    printf("Minimum %zu elements: ", k);
    for (i = 0; i < k; i++) {
        printf("%d ", min_k[i]);
    }
    printf("\n");

    printf("Maximum %zu elements: ", k);
    for (i = 0; i < k; i++) {
        printf("%d ", max_k[i]);
    }
    printf("\n");

    free(min_k);
    free(max_k);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

1. **Flawfinder `memcpy` warning (CWE-120):** This was already a false positive in practice — the destination buffer `sorted` is allocated with exactly `n * sizeof(int)` bytes, matching the copy size. The rewrite makes this explicit by computing `bytes` once and using the same variable for both the `malloc` and the `memcpy`, so the destination size and copy size can never diverge.

2. **Real bug fixed — missing `<stdint.h>`:** The original code used `SIZE_MAX` without including `<stdint.h>`, which would fail to compile (or rely on transitive includes) on strict C implementations. This has been added.

3. **Overflow protection retained:** The `n > SIZE_MAX / sizeof(int)` check guards against integer overflow in the allocation size, and an equivalent check for `k` was added in `main` before allocating `min_k`/`max_k`.

4. **Existing good practices preserved:** NULL-pointer parameter validation, `k > n` bounds checking, allocation failure handling with proper cleanup on every error path, and no memory leaks (`sorted`, `min_k`, `max_k` all freed on all paths).
