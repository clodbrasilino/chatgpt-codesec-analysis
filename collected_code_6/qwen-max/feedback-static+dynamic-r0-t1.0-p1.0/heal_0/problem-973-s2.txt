#include <stdio.h>
#include <string.h>

void left_rotate_string(char *str, int n) {
    int len = strlen(str);
    if (n > 0 && len > 1) {
        n = n % len;
        char temp[n + 1];
        strncpy(temp, str, n);
        temp[n] = '\0';
        memmove(str, str + n, len + 1 - n);
        strcat(str, temp);
    }
}

int main() {
    char str[] = "HelloWorld";
    int n = 3;
    left_rotate_string(str, n);
    printf("%s\n", str);
    return 0;
}
