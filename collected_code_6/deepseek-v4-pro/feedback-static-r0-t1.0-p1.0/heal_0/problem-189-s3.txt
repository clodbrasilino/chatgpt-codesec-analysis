#include <stdio.h>
#include <stdlib.h>

int firstMissingPositive(int *nums, int numsSize) {
    for (int i = 0; i < numsSize; i++) {
        while (nums[i] > 0 && nums[i] <= numsSize && nums[nums[i] - 1] != nums[i]) {
            int temp = nums[i];
            nums[i] = nums[temp - 1];
            nums[temp - 1] = temp;
        }
    }
    
    for (int i = 0; i < numsSize; i++) {
        if (nums[i] != i + 1) {
            return i + 1;
        }
    }
    
    return numsSize + 1;
}

int main(void) {
    int nums1[] = {1, 2, 0};
    int size1 = sizeof(nums1) / sizeof(nums1[0]);
    printf("%d\n", firstMissingPositive(nums1, size1));
    
    int nums2[] = {3, 4, -1, 1};
    int size2 = sizeof(nums2) / sizeof(nums2[0]);
    printf("%d\n", firstMissingPositive(nums2, size2));
    
    int nums3[] = {7, 8, 9, 11, 12};
    int size3 = sizeof(nums3) / sizeof(nums3[0]);
    printf("%d\n", firstMissingPositive(nums3, size3));
    
    return 0;
}
