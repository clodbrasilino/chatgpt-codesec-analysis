```c
#include <stdio.h>
#include <stddef.h>

int average(const int *list, size_t size, double *result)
{
    double sum = 0.0;
    size_t i;

    if (list == NULL || result == NULL || size == 0) {
        return -1;
    }

    for (i = 0; i < size; i++) {
        sum += (double)list[i];
    }

    *result = sum / (double)size;
    return 0;
}

int main(void)
{
    int numbers[] = {10, 20, 30, 40, 50};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    double avg = 0.0;

    if (average(numbers, count, &avg) != 0) {
        fprintf(stderr, "Error: could not compute average\n");
        return 1;
    }

    if (printf("Average: %.2f\n", avg) < 0) {
        return 1;
    }

    return 0;
}
```
