```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256
#define WORD_SIZE 256
#define MIN_WORD_LEN 3
#define MAX_WORD_LEN 5

int find_words(const char *text);

int find_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;
    char errbuf[ERRBUF_SIZE];
    char word[WORD_SIZE];
    int found;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[[:alpha:]]+", REG_EXTENDED);
    if (ret != 0) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    cursor = text;
    found = 0;

    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            break;
        }

        size_t length = (size_t)(match.rm_eo - match.rm_so);

        if (length >= MIN_WORD_LEN && length <= MAX_WORD_LEN &&
            length < sizeof(word)) {
            memcpy(word, cursor + match.rm_so, length);
            word[length] = '\0';
            printf("%s\n", word);
            found++;
        }

        if (match.rm_eo == 0) {
            break;
        }

        cursor += match.rm_eo;
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regexec failed: %s\n", errbuf);
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return found;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog nearby";
    int count;

    count = find_words(text);
    if (count < 0) {
        fprintf(stderr, "Error while searching for words\n");
        return EXIT_FAILURE;
    }

    printf("Total words found: %d\n", count);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already largely safe — the Flawfinder hits were advisory rather than actual vulnerabilities. Here is why each one is mitigated:

1. **`char errbuf[ERRBUF_SIZE]` (CWE-119/CWE-120):** Safe because `regerror()` is always called with `sizeof(errbuf)` as the size limit, and the buffer is explicitly NUL-terminated afterward as a defensive measure. No overflow is possible.

2. **`char word[WORD_SIZE]` (CWE-119/CWE-120):** Safe because every write into `word` is preceded by an explicit bounds check. I tightened the check from `length + 1 <= sizeof(word)` to `length < sizeof(word)` — functionally equivalent, but it eliminates any theoretical concern about integer overflow in `length + 1` (impossible in practice here since `length <= MAX_WORD_LEN`, but the simpler form is strictly safer and clearer).

3. **`memcpy` (CWE-120):** Safe because the copy of `length` bytes plus the NUL terminator is guaranteed to fit — `length` is validated against both `MAX_WORD_LEN` (5) and the buffer size (256) before the copy occurs.

Additional robustness already present and preserved:
- `NULL` input check.
- Validation of `rm_so`/`rm_eo` before computing the match length.
- Guard against infinite loops on zero-length advancement (`match.rm_eo == 0`).
- `regfree()` called on all exit paths after successful `regcomp`, so no resource leak.
- Distinction between `REG_NOMATCH` (normal termination) and genuine `regexec` errors.

The code compiles cleanly with `-Wall -Wextra -pedantic` and has no exploitable buffer issues.
