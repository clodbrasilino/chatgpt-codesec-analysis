#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_max_occurrences(int *list, size_t size) {
    if (list == NULL || size == 0) {
        return INT_MIN;
    }

    int max_val = list[0];
    int min_val = list[0];
    for (size_t i = 1; i < size; i++) {
        if (list[i] > max_val) {
            max_val = list[i];
        }
        if (list[i] < min_val) {
            min_val = list[i];
        }
    }

    size_t range = 0;
    if (max_val >= min_val) {
        range = (size_t)max_val - (size_t)min_val + 1;
    } else {
        return list[0];
    }

    size_t *counts = (size_t *)calloc(range, sizeof(size_t));
    if (counts == NULL) {
        return INT_MIN;
    }

    for (size_t i = 0; i < size; i++) {
        counts[list[i] - min_val]++;
    }

    size_t max_count = 0;
    int result = list[0];
    for (size_t i = 0; i < size; i++) {
        size_t count_index = list[i] - min_val;
        if (counts[count_index] > max_count) {
            max_count = counts[count_index];
            result = list[i];
        }
    }

    free(counts);
    return result;
}

int main(void) {
    int list[] = {1, 3, 2, 3, 4, 3, 2, 2, 2};
    size_t size = sizeof(list) / sizeof(list[0]);

    int max_occurrence_item = find_max_occurrences(list, size);

    printf("%d\n", max_occurrence_item);

    return 0;
}
