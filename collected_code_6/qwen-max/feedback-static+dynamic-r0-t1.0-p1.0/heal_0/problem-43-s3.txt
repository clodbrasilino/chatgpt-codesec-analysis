#include <stdio.h>
#include <regex.h>

int find_sequences(const char *str) {
    regex_t regex;
    int ret;
    
    ret = regcomp(&regex, "^[a-z]+(_[a-z]+)*$", 0);
    if (ret)
        return -1;

    ret = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);

    if (!ret)
        return 1;

    return 0;
}

int main(void) {
    const char *test_strings[] = {"hello_world", "Hello_World", "helloWorld", "hello_world_test", "hello_World"};
    size_t i;

    for (i = 0; i < sizeof(test_strings) / sizeof(test_strings[0]); ++i) {
        printf("%s: %d\n", test_strings[i], find_sequences(test_strings[i]));
    }

    return 0;
}
