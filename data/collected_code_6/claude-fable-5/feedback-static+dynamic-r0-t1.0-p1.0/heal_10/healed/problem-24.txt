```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_BUFFER_SIZE 128U
#define MAX_BINARY_DIGITS (sizeof(unsigned long) * 8U)
#define FLUSH_GUARD_LIMIT 4096U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int c;
    size_t guard = 0U;

    do {
        c = getchar();
        guard++;
    } while (c != '\n' && c != EOF && guard < FLUSH_GUARD_LIMIT);
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == sizeof(input) - 1U) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The `flush_stdin()` function never writes the characters it reads into any buffer — it simply discards them. There is no buffer to overflow. Additionally, the `guard` counter bounds the loop at `FLUSH_GUARD_LIMIT` iterations, and the loop terminates on `\n` or `EOF`, preventing an unbounded loop on a stream that never ends. No fix needed.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The `input` buffer is used safely throughout:
   - `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size and always null-terminates within bounds.
   - The buffer is zero-initialized with `memset` before use.
   - A defensive `input[sizeof(input) - 1U] = '\0'` guarantees termination.
   - `strnlen(input, sizeof(input))` is used instead of `strlen`, so length computation cannot run past the buffer.
   - The truncation case (`len == sizeof(input) - 1U` with no newline) is detected, the remaining stdin is drained, and the program exits with an error rather than processing partial input.
   - `binary_to_decimal()` re-validates the pointer, length bounds, character set, and checks for `unsigned long` overflow (`value > ULONG_MAX >> 1`) before every shift.

Both Flawfinder hits are false positives here; the code already implements the recommended mitigations (bounded reads, bounded length functions, explicit termination, overflow checks), so the rewrite preserves the logic unchanged with the comments removed.
