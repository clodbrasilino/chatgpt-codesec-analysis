```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int starts_with_number(const char *str) {
    if (str == NULL) {
        return 0;
    }
    if (*str == '\0') {
        return 0;
    }
    return isdigit((unsigned char)*str) ? 1 : 0;
}

int main(void) {
    const char *test_strings[] = {
        "123abc",
        "abc123",
        "",
        " 123abc",
        "1",
        "9hello",
        NULL
    };
    
    for (int i = 0; test_strings[i] != NULL; i++) {
        printf("\"%s\" -> %s\n", 
               test_strings[i], 
               starts_with_number(test_strings[i]) ? "starts with number" : "does not start with number");
    }
    
    return 0;
}
```
