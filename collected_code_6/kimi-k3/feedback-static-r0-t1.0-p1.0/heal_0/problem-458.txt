```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

double calculate_rectangle_area(double length, double width);

int main(void)
{
    double length;
    double width;
    double area;

    printf("Enter length: ");
    if (scanf("%lf", &length) != 1)
    {
        fprintf(stderr, "Error: Invalid input for length\n");
        return EXIT_FAILURE;
    }

    printf("Enter width: ");
    if (scanf("%lf", &width) != 1)
    {
        fprintf(stderr, "Error: Invalid input for width\n");
        return EXIT_FAILURE;
    }

    if (length < 0.0 || width < 0.0)
    {
        fprintf(stderr, "Error: Dimensions must be non-negative\n");
        return EXIT_FAILURE;
    }

    area = calculate_rectangle_area(length, width);
    printf("Area: %.2f\n", area);

    return EXIT_SUCCESS;
}

double calculate_rectangle_area(double length, double width)
{
    return length * width;
}
```
