```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    size_t needed = regerror(errcode, regex, NULL, 0);
    char *errbuf = malloc(needed);
    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate error buffer\n", prefix);
        return;
    }
    regerror(errcode, regex, errbuf, needed);
    fprintf(stderr, "%s: %s\n", prefix, errbuf);
    free(errbuf);
}

int ends_with_alphanumeric(const char *str)
{
    regex_t regex;
    int ret;
    int result = 0;

    if (str == NULL) {
        return 0;
    }

    ret = regcomp(&regex, "[a-zA-Z0-9]$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return -1;
    }

    ret = regexec(&regex, str, 0, NULL, 0);
    if (ret == 0) {
        result = 1;
    } else if (ret == REG_NOMATCH) {
        result = 0;
    } else {
        print_regex_error(ret, &regex, "Regex execution failed");
        result = -1;
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "world!",
        "test_",
        "abcXYZ",
        "ends with space ",
        "9"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = ends_with_alphanumeric(tests[i]);
        if (result < 0) {
            fprintf(stderr, "Error checking string: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (result == 1) {
            printf("\"%s\" ends with an alphanumeric character.\n", tests[i]);
        } else {
            printf("\"%s\" does not end with an alphanumeric character.\n", tests[i]);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already largely safe — the Flawfinder warnings on `char errbuf[256]` are heuristic flags on fixed-size buffers (CWE-119/CWE-120), not confirmed vulnerabilities, since `regerror()` was correctly called with `sizeof(errbuf)` and guarantees null-termination within the given size.

To fully address the flagged concern, the rewrite eliminates the statically-sized buffers entirely:

1. **Dynamic error buffer sizing** — `regerror()` is first called with a `NULL` buffer and size `0`, which returns the exact number of bytes required for the full error message. The buffer is then allocated with `malloc()` to that precise size, so no message can ever be truncated and no fixed bound can be misjudged.
2. **Allocation failure handling** — if `malloc()` fails, a fallback message is printed rather than dereferencing a null pointer.
3. **Centralized error reporting** — the duplicated error-handling blocks were factored into `print_regex_error()`, reducing the surface area for future mistakes and ensuring `free()` is always paired with the allocation.
4. **Resource management preserved** — `regfree()` is still called on every path after a successful `regcomp()`, avoiding memory leaks in the regex internals.
