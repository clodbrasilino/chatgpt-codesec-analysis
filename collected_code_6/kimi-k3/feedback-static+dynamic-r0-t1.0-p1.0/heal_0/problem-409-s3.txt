```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

struct Tuple {
    int first;
    int second;
};

int find_min_product(const struct Tuple *list, size_t size) {
    if (list == NULL || size == 0) {
        return INT_MAX;
    }
    
    int min_product = INT_MAX;
    
    for (size_t i = 0; i < size; i++) {
        long long product = (long long)list[i].first * (long long)list[i].second;
        if (product < (long long)min_product) {
            if (product > (long long)INT_MAX) {
                min_product = INT_MAX;
            } else if (product < (long long)INT_MIN) {
                min_product = INT_MIN;
            } else {
                min_product = (int)product;
            }
        }
    }
    
    return min_product;
}

int main(void) {
    struct Tuple list[] = {{2, 3}, {1, 5}, {4, 2}, {0, 10}};
    size_t size = sizeof(list) / sizeof(list[0]);
    
    int result = find_min_product(list, size);
    
    if (result == INT_MAX && size == 0) {
        fprintf(stderr, "Error: Empty list\n");
        return EXIT_FAILURE;
    }
    
    printf("Minimum product: %d\n", result);
    
    return EXIT_SUCCESS;
}
```
