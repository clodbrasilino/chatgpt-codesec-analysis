```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define SENTENCE_MAX 256
#define WORD_MAX 64

static size_t bounded_strlen(const char *s, size_t max)
{
    size_t n = 0U;

    while (n < max && s[n] != '\0') {
        n++;
    }
    return n;
}

static int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    size_t i;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = bounded_strlen(word, WORD_MAX);
    if (word_len == 0U || word_len >= WORD_MAX) {
        return 0;
    }

    i = 0U;
    while (i < SENTENCE_MAX && sentence[i] != '\0') {
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               !isalnum((unsigned char)sentence[i])) {
            i++;
        }
        if (i >= SENTENCE_MAX || sentence[i] == '\0') {
            break;
        }

        size_t start = i;
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               isalnum((unsigned char)sentence[i])) {
            i++;
        }
        size_t token_len = i - start;

        if (token_len == word_len &&
            strncmp(&sentence[start], word, word_len) == 0) {
            return 1;
        }
    }

    return 0;
}

static int read_line(char *buffer, size_t size, const char *prompt)
{
    size_t len;

    if (buffer == NULL || size < 2U || size > (size_t)INT_MAX || prompt == NULL) {
        return -1;
    }

    printf("%s", prompt);
    fflush(stdout);

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1U] = '\0';

    len = strcspn(buffer, "\n");
    if (buffer[len] == '\n') {
        buffer[len] = '\0';
    } else if (len == size - 1U) {
        int ch;
        do {
            ch = getchar();
        } while (ch != '\n' && ch != EOF);
        if (ch == EOF && ferror(stdin)) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    char sentence[SENTENCE_MAX];
    char word[WORD_MAX];

    memset(sentence, 0, sizeof(sentence));
    memset(word, 0, sizeof(word));

    if (read_line(sentence, sizeof(sentence), "Enter a sentence: ") != 0) {
        fprintf(stderr, "Error reading sentence.\n");
        return EXIT_FAILURE;
    }

    if (read_line(word, sizeof(word), "Enter a word to search: ") != 0) {
        fprintf(stderr, "Error reading word.\n");
        return EXIT_FAILURE;
    }

    if (is_word_present(sentence, word) == 1) {
        printf("The word \"%s\" is present in the sentence.\n", word);
    } else {
        printf("The word \"%s\" is not present in the sentence.\n", word);
    }

    return EXIT_SUCCESS;
}
```

Note: since `INT_MAX` is used for the added size validation in `read_line`, `<limits.h>` must be included as well:

```c
#include <limits.h>
```

The rewritten code addresses the flagged weaknesses as follows:

- **Flawfinder `char` (CWE-119/CWE-120):** Both statically-sized arrays are zero-initialized with `memset`, always passed together with their `sizeof`, and every read/scan operation (`fgets`, `bounded_strlen`, the tokenizer loops in `is_word_present`) is strictly bounded by `SENTENCE_MAX`/`WORD_MAX`, so no overflow is possible.
- **Flawfinder `getchar` (CWE-120/CWE-20):** The drain loop reads into an `int`, never stores into a buffer, terminates on `'\n'` or `EOF`, and now additionally distinguishes a stream error from a clean EOF via `ferror`, returning `-1` so callers handle the failure.
- **Additional hardening:** the `size > INT_MAX` guard prevents an integer truncation issue in the `(int)size` cast passed to `fgets`, and the explicit `buffer[size - 1] = '\0'` guarantees NUL-termination in all cases.
