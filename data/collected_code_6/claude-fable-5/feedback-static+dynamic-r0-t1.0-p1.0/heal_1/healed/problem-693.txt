```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(const char *input, char *output, size_t output_size);

int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len;

        if (match.rm_so < 0) {
            regfree(&regex);
            return -1;
        }

        prefix_len = (size_t)match.rm_so;

        if (prefix_len >= (output_size - out_pos)) {
            regfree(&regex);
            return -1;
        }

        if ((out_pos + prefix_len + 1U) >= output_size) {
            regfree(&regex);
            return -1;
        }

        memcpy(&output[out_pos], cursor, prefix_len);
        out_pos += prefix_len;
        output[out_pos] = ' ';
        out_pos++;

        cursor += match.rm_eo;
    }

    {
        size_t remaining = strnlen(cursor, output_size);

        if (remaining >= (output_size - out_pos)) {
            regfree(&regex);
            return -1;
        }

        if ((out_pos + remaining) >= output_size) {
            regfree(&regex);
            return -1;
        }

        memcpy(&output[out_pos], cursor, remaining);
        out_pos += remaining;
        output[out_pos] = '\0';
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256];

    memset(result, 0, sizeof(result));

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Summary of changes addressing the flagged weaknesses:**

1. **CWE-120 (memcpy buffer overflow):** Every `memcpy` is now guarded by an explicit bounds check expressed as `len >= (output_size - out_pos)`, which avoids potential integer overflow in the addition-based check (`out_pos + len + 1 >= output_size` can wrap around if `len` is attacker-controlled and near `SIZE_MAX`). Both check forms are kept for defense-in-depth. A negative `rm_so` from the regex engine is also rejected before casting to `size_t`.

2. **CWE-126 (strlen over-read):** Replaced `strlen(cursor)` with `strnlen(cursor, output_size)`, bounding the read. Since anything longer than `output_size` would fail the subsequent capacity check anyway, this loses no functionality while guaranteeing the scan terminates.

3. **CWE-119/CWE-120 (statically-sized array):** The buffer size is always passed via `sizeof(result)` and never assumed inside the function; the output buffer is zero-initialized with `memset`, and the function writes an empty string to `output` immediately after validating parameters, so the caller never reads uninitialized memory even on early error paths.

4. **General hardening:** `regfree` is called on every exit path after successful `regcomp`, preventing resource leaks, and all parameters are NULL/zero-checked up front.
