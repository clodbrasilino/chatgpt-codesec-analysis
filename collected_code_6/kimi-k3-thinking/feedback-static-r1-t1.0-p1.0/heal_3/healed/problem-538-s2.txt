```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

typedef struct {
    char **items;
    size_t size;
} StringTuple;

static size_t bounded_strlen(const char *s, size_t max)
{
    size_t len = 0;
    while (len < max && s[len] != '\0') {
        len++;
    }
    return len;
}

StringTuple *string_list_to_tuple(const char *list[], size_t count)
{
    if (list == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    StringTuple *tuple = malloc(sizeof(StringTuple));
    if (tuple == NULL) {
        return NULL;
    }

    tuple->items = calloc(count, sizeof(char *));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }

    tuple->size = 0;

    for (size_t i = 0; i < count; i++) {
        if (list[i] == NULL) {
            goto fail;
        }

        size_t len = bounded_strlen(list[i], MAX_STRING_LENGTH + 1);
        if (len > MAX_STRING_LENGTH) {
            goto fail;
        }

        size_t dest_capacity = len + 1;
        if (dest_capacity <= len) {
            goto fail;
        }

        tuple->items[i] = malloc(dest_capacity);
        if (tuple->items[i] == NULL) {
            goto fail;
        }

        int written = snprintf(tuple->items[i], dest_capacity, "%s", list[i]);
        if (written < 0 || (size_t)written != len || (size_t)written >= dest_capacity) {
            free(tuple->items[i]);
            tuple->items[i] = NULL;
            goto fail;
        }

        tuple->size++;
    }

    return tuple;

fail:
    for (size_t j = 0; j < count; j++) {
        free(tuple->items[j]);
    }
    free(tuple->items);
    free(tuple);
    return NULL;
}

void free_string_tuple(StringTuple *tuple)
{
    if (tuple == NULL) {
        return;
    }

    if (tuple->items != NULL) {
        for (size_t i = 0; i < tuple->size; i++) {
            free(tuple->items[i]);
        }
        free(tuple->items);
    }

    free(tuple);
}

void print_string_tuple(const StringTuple *tuple)
{
    if (tuple == NULL) {
        printf("()\n");
        return;
    }

    printf("(");
    for (size_t i = 0; i < tuple->size; i++) {
        printf("\"%s\"", tuple->items[i]);
        if (i + 1 < tuple->size) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void)
{
    const char *list[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(list) / sizeof(list[0]);
    StringTuple *tuple = string_list_to_tuple(list, count);

    if (tuple == NULL) {
        fprintf(stderr, "Error: failed to convert string list to tuple\n");
        return EXIT_FAILURE;
    }

    print_string_tuple(tuple);

    free_string_tuple(tuple);
    tuple = NULL;

    return EXIT_SUCCESS;
}
```

Key fixes applied to address the flagged CWE-120 `memcpy` weakness:

1. **Replaced `memcpy` with `snprintf`** — The copy is now bounded by the destination capacity, and the return value is validated (`written < 0`, `written != len`, or `written >= dest_capacity` all trigger failure). This guarantees no buffer overflow and proper null termination, eliminating the manual `items[i][len] = '\0'` step.

2. **Added overflow guard for `dest_capacity`** — `if (dest_capacity <= len)` catches any wraparound from `len + 1` before allocation.

3. **Used `calloc` for the items array** — All slots start as `NULL`, so the `fail` cleanup path can safely free all `count` slots without risking freeing uninitialized pointers.

4. **Added `bounded_strlen` helper** — Provides a portable, standards-compliant bounded string length calculation (replacing the POSIX-only `strnlen`), ensuring strings longer than `MAX_STRING_LENGTH` are rejected without reading unbounded memory.
