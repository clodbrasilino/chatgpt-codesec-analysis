```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LEN 100

struct entry {
    char str[MAX_LEN];
    int count;
};

static const char *second_most_frequent(const char *arr[], size_t n,
                                        struct entry *table, size_t *table_size)
{
    size_t i;
    size_t j;
    size_t unique = 0;
    size_t first_idx;
    size_t second_idx;
    int found_second = 0;

    if (arr == NULL || table == NULL || table_size == NULL || n == 0) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        int found = 0;
        size_t len;

        if (arr[i] == NULL) {
            return NULL;
        }

        len = strnlen(arr[i], MAX_LEN);
        if (len >= MAX_LEN) {
            return NULL;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(table[j].str, arr[i], MAX_LEN) == 0) {
                table[j].count++;
                found = 1;
                break;
            }
        }

        if (found == 0) {
            memcpy(table[unique].str, arr[i], len);
            table[unique].str[len] = '\0';
            table[unique].count = 1;
            unique++;
        }
    }

    *table_size = unique;

    if (unique < 2) {
        return NULL;
    }

    first_idx = 0;
    second_idx = 0;

    for (i = 1; i < unique; i++) {
        if (table[i].count > table[first_idx].count) {
            first_idx = i;
        }
    }

    for (i = 0; i < unique; i++) {
        if (i == first_idx) {
            continue;
        }
        if (found_second == 0 || table[i].count > table[second_idx].count) {
            second_idx = i;
            found_second = 1;
        }
    }

    if (found_second == 0) {
        return NULL;
    }

    return table[second_idx].str;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "orange",
        "banana", "apple", "orange", "banana", "banana"
    };
    size_t n = sizeof(sequence) / sizeof(sequence[0]);
    struct entry *table;
    size_t table_size = 0;
    const char *result;

    table = calloc(n, sizeof(struct entry));
    if (table == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    result = second_most_frequent(sequence, n, table, &table_size);

    if (result == NULL) {
        fprintf(stderr, "No second most frequent string found\n");
        free(table);
        return EXIT_FAILURE;
    }

    if (printf("Second most frequent string: %s\n", result) < 0) {
        free(table);
        return EXIT_FAILURE;
    }

    free(table);
    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen()` with `strnlen(arr[i], MAX_LEN)`, which bounds the read to at most `MAX_LEN` bytes. Even if a non-terminated string is passed, the scan cannot read past the limit, and the `len >= MAX_LEN` check rejects anything too long or unterminated within the bound.

2. **CWE-120 (`strncpy` misuse):** Replaced `strncpy()` with `memcpy()` using the already-validated length, followed by explicit null termination at `str[len]`. This avoids `strncpy`'s two classic pitfalls — silent truncation without termination, and needless zero-padding — since the length is verified before the copy.

3. **CWE-119/CWE-120 (statically-sized buffer):** The fixed-size `str[MAX_LEN]` array is retained, but every write path is now provably bounded: the length check guarantees `len <= MAX_LEN - 1`, so `memcpy` plus the terminator can never exceed the buffer. Comparison also uses `strncmp(..., MAX_LEN)` as defense-in-depth against unterminated table entries.

4. **Existing safeguards preserved:** NULL-pointer checks on all parameters and array elements, `calloc` result validation, `printf` return-value check, and `free()` on every exit path remain intact, so there are no leaks or unchecked failures.
