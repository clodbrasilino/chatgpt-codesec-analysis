```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 16
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used > buffer_size - sym_len - 1) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **Flawfinder `memcpy` (CWE-120):** This is a false positive in practice. The code already guards the copy with `if (sym_len >= buffer_size || used > buffer_size - sym_len - 1)`, which guarantees that `used + sym_len + 1 <= buffer_size` before every `memcpy` and null-terminator write. Note the check is carefully written to avoid unsigned integer underflow/overflow: it compares `used > buffer_size - sym_len - 1` only after confirming `sym_len < buffer_size`, so `buffer_size - sym_len - 1` cannot wrap around. No change is required — the mitigation is already in place.

2. **Flawfinder `char` static array (CWE-119/CWE-120):** Also mitigated. The buffer size is passed to `int_to_roman` via `sizeof(roman)` rather than a hardcoded constant, the callee validates `buffer_size >= 2`, and the buffer is zero-initialized with `memset`. `ROMAN_BUFFER_SIZE` of 16 comfortably holds the longest possible Roman numeral in range (`MMMDCCCLXXXVIII` = 15 chars + NUL for 3888), and even if it didn't, the bounds check would fail safely with `-1` rather than overflow.

3. **`scanf("%d", ...)`:** Safe as used — `%d` with an `int*` cannot overflow a buffer, and the return value is checked. Out-of-range values are rejected by `int_to_roman`.

The code as written is already defensively correct; the rewrite above preserves all existing bounds checks with the audit comments removed as requested.
