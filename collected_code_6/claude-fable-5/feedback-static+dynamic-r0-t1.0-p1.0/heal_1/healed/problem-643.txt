#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int contains_interior_z(const char *word, size_t len)
{
    size_t i;

    if (word == NULL || len < 3) {
        return 0;
    }

    for (i = 1; i < len - 1; i++) {
        if (word[i] == 'z' || word[i] == 'Z') {
            return 1;
        }
    }

    return 0;
}

int find_word_with_interior_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    while (text[i] != '\0') {
        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }

        const size_t start = i;

        while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
            i++;
        }

        const size_t len = i - start;

        if (len > 0 && contains_interior_z(text + start, len)) {
            if (len >= result_size) {
                return -1;
            }
            memcpy(result, text + start, len);
            result[len] = '\0';
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    const char *tests[] = {
        "the lazy dog jumped",
        "zebra runs fast",
        "jazz music is nice",
        "no matching words here",
        "amazing puzzle pieces"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    char buffer[256];
    size_t i;

    for (i = 0; i < num_tests; i++) {
        const int status = find_word_with_interior_z(tests[i], buffer, sizeof(buffer));
        if (status == 0) {
            printf("Input: \"%s\" -> Match: \"%s\"\n", tests[i], buffer);
        } else if (status == 1) {
            printf("Input: \"%s\" -> No match found\n", tests[i]);
        } else {
            fprintf(stderr, "Error processing input: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

The changes made to address the identified weaknesses:

1. **`start` variable scope** — moved its declaration inside the outer `while` loop in `find_word_with_interior_z`, and declared it `const` since it is assigned exactly once per iteration.

2. **`len` variable scope** — similarly moved inside the loop, declared at the point of its calculation as `const`.

3. **`status` variable scope** — moved its declaration inside the `for` loop in `main`, also declared `const` since it is only assigned once per iteration.

These changes improve the code in several ways: the variables can no longer accidentally carry stale values across loop iterations, the `const` qualifiers document intent and let the compiler enforce single assignment, and the reduced lifetimes make the code easier to reason about during review and maintenance.
