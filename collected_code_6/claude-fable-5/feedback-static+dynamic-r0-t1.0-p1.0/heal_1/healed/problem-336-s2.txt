```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32

int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int ch;
        do {
            ch = getchar();
        } while (ch != '\n' && ch != EOF);
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and fixes applied:**

The original code was already quite defensive, and the Flawfinder hits were largely low-risk static-analysis noise. Still, here is how each flagged item was hardened:

1. **`char normalized[MAX_INPUT]` / `char input[MAX_INPUT]` (CWE-119/CWE-120):** These statically-sized buffers were already protected — `fgets` bounds the read with `sizeof(input)`, and `has_28_days` rejects any string with `len >= MAX_INPUT` before writing into `normalized`, guaranteeing room for the null terminator. I kept those checks and added an explicit `input[sizeof(input) - 1] = '\0'` as belt-and-suspenders termination.

2. **`strlen` (CWE-126):** The risk is an over-read on non-terminated strings. Replaced both calls with `strnlen(..., MAX_INPUT)`, which bounds the scan even if termination were ever missing. Combined with the `len >= MAX_INPUT` rejection, an unterminated buffer can never be processed further.

3. **`getchar` in a loop (CWE-120/CWE-20):** The drain loop never writes to a buffer, so no overflow is possible; it correctly discards excess input and terminates on both `'\n'` and `EOF`. Restructured as a `do/while` for clarity but the semantics are identical and safe.

4. **Additional input validation (CWE-20):** Added an `isalpha` check in `has_28_days` so non-alphabetic input is rejected as invalid rather than silently normalized, tightening the validation surface. Also added `fflush(stdout)` so the prompt reliably appears before blocking on `stdin`.
