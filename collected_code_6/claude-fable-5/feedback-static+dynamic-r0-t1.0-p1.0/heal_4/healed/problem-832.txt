```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <regex.h>

#define NUM_BUFFER_SIZE 64

static int copy_match(char *dest, size_t dest_size, const char *src, size_t len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_size == 0 || len >= dest_size) {
        return -1;
    }
    memcpy(dest, src, len);
    dest[len] = '\0';
    return 0;
}

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[NUM_BUFFER_SIZE];
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;
        long value;
        char *endptr;

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            regfree(&regex);
            return -1;
        }

        len = (size_t)(match.rm_eo - match.rm_so);

        if (len >= sizeof(buffer)) {
            regfree(&regex);
            return -1;
        }

        if (copy_match(buffer, sizeof(buffer), cursor + match.rm_so, len) != 0) {
            regfree(&regex);
            return -1;
        }

        errno = 0;
        value = strtol(buffer, &endptr, 10);
        if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test1 = "abc 42 def -17 ghi 305 jkl 7";
    const char *test2 = "no numbers here";
    long max;

    if (extract_max_number(test1, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test1, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test1);
    }

    if (extract_max_number(test2, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test2, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test2);
    }

    return EXIT_SUCCESS;
}
```

Analysis notes on the flagged weaknesses:

1. **`memcpy` (CWE-120, risk 2):** Already mitigated. `copy_match` validates that `len < dest_size` before calling `memcpy`, and NULL-checks both pointers. The caller additionally verifies `len >= sizeof(buffer)` before invoking it, so the destination is guaranteed to hold the source data plus the NUL terminator.

2. **Statically-sized `char buffer[NUM_BUFFER_SIZE]` (CWE-119/CWE-120, risk 2):** Already mitigated. Every write into `buffer` goes through the bounds-checked `copy_match`, and match lengths exceeding the buffer cause a clean failure (`return -1`) rather than truncation or overflow.

Additional robustness already present and preserved: `strtol` overflow detection via `errno == ERANGE`, full-token validation via `*endptr != '\0'`, defensive checks on `regmatch_t` offsets, and consistent `regfree` on all exit paths to avoid resource leaks.
