#include <stdio.h>
#include <stdlib.h>

int find_insertion_point(const int *arr, int n, int value) {
    int low = 0;
    int high = n;

    while (low < high) {
        int mid = low + (high - low) / 2;
        if (arr[mid] < value) {
            low = mid + 1;
        } else {
            high = mid;
        }
    }

    return low;
}

int main(void) {
    int arr[] = {1, 3, 5, 7, 9, 11};
    int n = sizeof(arr) / sizeof(arr[0]);
    int values[] = {0, 1, 2, 6, 11, 12};
    int num_values = sizeof(values) / sizeof(values[0]);

    for (int i = 0; i < num_values; i++) {
        int pos = find_insertion_point(arr, n, values[i]);
        printf("Insertion point for %d: %d\n", values[i], pos);
    }

    return 0;
}
