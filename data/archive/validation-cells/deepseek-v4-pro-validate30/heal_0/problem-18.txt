#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void remove_chars(char *str1, const char *str2) {
    int table[256] = {0};
    int i = 0, j = 0;
    size_t len = strlen(str2);
    for (i = 0; i < len; i++) {
        table[(unsigned char)str2[i]] = 1;
    }
    i = 0;
    while (str1[i] != '\0') {
        if (!table[(unsigned char)str1[i]]) {
            str1[j++] = str1[i];
        }
        i++;
    }
    str1[j] = '\0';
}

int main(void) {
    char str1[256];
    char str2[256];

    if (fgets(str1, sizeof(str1), stdin) == NULL) {
        return 1;
    }
    str1[strcspn(str1, "\n")] = '\0';

    if (fgets(str2, sizeof(str2), stdin) == NULL) {
        return 1;
    }
    str2[strcspn(str2, "\n")] = '\0';

    remove_chars(str1, str2);
    printf("%s\n", str1);

    return 0;
}
