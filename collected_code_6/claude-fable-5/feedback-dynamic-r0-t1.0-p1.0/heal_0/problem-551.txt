#include <stdio.h>
#include <stdlib.h>

int extract_column(int **matrix, size_t rows, size_t cols, size_t col_index, int *result)
{
    size_t i;

    if (matrix == NULL || result == NULL) {
        return -1;
    }
    if (col_index >= cols) {
        return -1;
    }
    for (i = 0; i < rows; i++) {
        if (matrix[i] == NULL) {
            return -1;
        }
        result[i] = matrix[i][col_index];
    }
    return 0;
}

int **create_matrix(size_t rows, size_t cols)
{
    int **matrix;
    size_t i;
    size_t j;

    matrix = calloc(rows, sizeof(int *));
    if (matrix == NULL) {
        return NULL;
    }
    for (i = 0; i < rows; i++) {
        matrix[i] = calloc(cols, sizeof(int));
        if (matrix[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(matrix[j]);
            }
            free(matrix);
            return NULL;
        }
    }
    return matrix;
}

void free_matrix(int **matrix, size_t rows)
{
    size_t i;

    if (matrix == NULL) {
        return;
    }
    for (i = 0; i < rows; i++) {
        free(matrix[i]);
    }
    free(matrix);
}

int main(void)
{
    size_t rows = 3;
    size_t cols = 4;
    size_t col_index = 2;
    size_t i;
    size_t j;
    int **matrix;
    int *column;
    int value = 1;

    matrix = create_matrix(rows, cols);
    if (matrix == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < rows; i++) {
        for (j = 0; j < cols; j++) {
            matrix[i][j] = value;
            value++;
        }
    }

    column = calloc(rows, sizeof(int));
    if (column == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free_matrix(matrix, rows);
        return EXIT_FAILURE;
    }

    if (extract_column(matrix, rows, cols, col_index, column) != 0) {
        fprintf(stderr, "Column extraction failed\n");
        free(column);
        free_matrix(matrix, rows);
        return EXIT_FAILURE;
    }

    printf("Column %zu:\n", col_index);
    for (i = 0; i < rows; i++) {
        printf("%d\n", column[i]);
    }

    free(column);
    free_matrix(matrix, rows);
    return EXIT_SUCCESS;
}
