#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define INPUT_CAPACITY 256

static bool is_vowel(char c)
{
    return strchr("aeiouAEIOU", c) != NULL;
}

static size_t bounded_strlen(const char *str, size_t max_length)
{
    size_t length = 0U;

    while (length < max_length && str[length] != '\0') {
        length++;
    }

    return length;
}

void reverse_vowels(char *str, size_t max_length)
{
    if (str == NULL || max_length == 0U) {
        return;
    }

    size_t length = bounded_strlen(str, max_length);
    if (length < 2U) {
        return;
    }

    size_t left = 0U;
    size_t right = length - 1U;

    while (left < right) {
        while (left < right && !is_vowel(str[left])) {
            left++;
        }
        while (left < right && !is_vowel(str[right])) {
            right--;
        }
        if (left < right) {
            char temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            right--;
        }
    }
}

int main(void)
{
    char input[INPUT_CAPACITY];

    printf("Enter a string: ");
    if (fgets(input, sizeof input, stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    if (strchr(input, '\n') != NULL) {
        input[strcspn(input, "\n")] = '\0';
    } else if (!feof(stdin)) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    printf("Original: %s\n", input);
    reverse_vowels(input, sizeof input);
    printf("Vowels reversed: %s\n", input);

    return EXIT_SUCCESS;
}
