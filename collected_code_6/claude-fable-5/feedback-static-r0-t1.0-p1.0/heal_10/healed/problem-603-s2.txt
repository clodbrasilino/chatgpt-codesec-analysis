#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>
#include <string.h>

int *get_lucid_numbers(int n, size_t *count);

int *get_lucid_numbers(int n, size_t *count)
{
    if (count == NULL || n < 1) {
        return NULL;
    }
    if ((size_t)n > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    size_t capacity = (size_t)n;
    size_t len = capacity;

    int *arr = malloc(capacity * sizeof *arr);
    if (arr == NULL) {
        return NULL;
    }

    int *result = malloc(capacity * sizeof *result);
    if (result == NULL) {
        free(arr);
        return NULL;
    }

    for (size_t i = 0; i < capacity; i++) {
        arr[i] = (int)(i + 1);
    }

    size_t rcount = 0;
    result[rcount++] = arr[0];

    size_t start = 1;
    while (start < len && start < capacity) {
        int step = arr[start];
        if (step <= 0 || rcount >= capacity) {
            free(arr);
            free(result);
            return NULL;
        }
        result[rcount++] = step;

        size_t write = start;
        size_t idx = 0;
        size_t limit = (len < capacity) ? len : capacity;

        for (size_t read = start; read < limit; read++, idx++) {
            if (idx % (size_t)step != 0) {
                if (write >= capacity || write > read || read >= capacity) {
                    free(arr);
                    free(result);
                    return NULL;
                }
                arr[write] = arr[read];
                write++;
            }
        }

        if (write > capacity) {
            free(arr);
            free(result);
            return NULL;
        }
        len = write;
    }

    free(arr);

    if (rcount > 0 && rcount <= capacity) {
        int *shrunk = realloc(result, rcount * sizeof *shrunk);
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    *count = rcount;
    return result;
}

int main(int argc, const char *const argv[])
{
    int n = 25;

    if (argc > 1 && argv[1] != NULL) {
        char *end = NULL;
        errno = 0;
        long value = strtol(argv[1], &end, 10);
        if (errno != 0 || end == argv[1] || end == NULL || *end != '\0' ||
            value < 1 || value > INT_MAX) {
            fprintf(stderr, "Invalid input: expected a positive integer\n");
            return EXIT_FAILURE;
        }
        n = (int)value;
    }

    size_t count = 0;
    int *lucid = get_lucid_numbers(n, &count);
    if (lucid == NULL) {
        fprintf(stderr, "Failed to compute lucid numbers\n");
        return EXIT_FAILURE;
    }

    printf("Lucid numbers <= %d:\n", n);
    for (size_t i = 0; i < count; i++) {
        printf("%d ", lucid[i]);
    }
    printf("\n");

    free(lucid);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses (CWE-120 / CWE-20, buffer reads in loops):**

The Flawfinder warnings pointed at the sieve loop where `arr[read]` and `arr[write]` are accessed. On review, the original code already mitigates all of them, so the rewrite preserves those guards rather than removing them:

1. **Read bounds (`arr[read]`)** — the loop bound is `limit = min(len, capacity)`, and `len` can never exceed `capacity` (it's initialized to `capacity` and only ever reassigned from `write`, which is checked against `capacity`). This guarantees `read < capacity` at all times; the explicit `read >= capacity` check inside the loop is defensive redundancy.

2. **Write bounds (`arr[write]`)** — before every store, `write >= capacity` and `write > read` are checked, so a write can never go past the allocation or ahead of the read cursor (which would corrupt unread data).

3. **Result buffer (`result[rcount++]`)** — bounded because there can be at most `n` lucid numbers ≤ `n`, and `rcount >= capacity` is verified before each insertion inside the loop.

4. **Allocation overflow (CWE-190)** — `(size_t)n > SIZE_MAX / sizeof(int)` prevents multiplication overflow in `malloc`.

5. **Input validation (CWE-20)** — `strtol` is used with full error checking (`errno`, `end` pointer, empty-string, trailing garbage, range `1..INT_MAX`), so untrusted `argv[1]` cannot produce an invalid `n`.

6. **Division by zero / negative modulus** — `step <= 0` is rejected before `idx % (size_t)step`.

7. **Memory hygiene** — every error path frees both buffers; the `realloc` shrink correctly falls back to the original pointer on failure, avoiding a leak or use of NULL.

The code is safe as written; the warnings are risk-level-1 informational hits inherent to any indexed loop, and all invariants they ask you to verify hold.
