```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

static char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = 0;
    while (len < MAX_INPUT_LEN && snake[len] != '\0') {
        len++;
    }
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    char *result;
    size_t len;

    memset(input, 0, sizeof(input));

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "No input provided\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strlen(input);
    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **`strnlen` portability issue** — The most likely cause of the three failed test cases (`<no output>`) is a build failure: `strnlen` is a POSIX function, not part of strict C90/C99, so compilation can fail without `_POSIX_C_SOURCE` defined. I replaced it with a portable bounded length loop that provides identical over-read protection.

2. **CWE-119/CWE-120 (statically-sized buffer)** — The buffer is now zero-initialized with `memset`, `fgets` is still bounded by `sizeof(input)`, and an explicit sentinel `input[sizeof(input) - 1] = '\0'` guarantees termination regardless of input.

3. **CWE-126 (`strlen` over-read)** — With the buffer zeroed and explicitly NUL-terminated before `strlen` is called, the string is guaranteed terminated, eliminating the over-read risk.

4. **Defensive hardening retained** — NULL checks on the input pointer and `malloc` result, bounded allocation (`len + 1` cannot overflow since `len < MAX_INPUT_LEN`), and the result buffer is always terminated before use.
