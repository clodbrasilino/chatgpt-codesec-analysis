```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static void report_regex_error(int errcode, const regex_t *regex, const char *context)
{
    size_t needed;
    char *error_buffer;

    needed = regerror(errcode, regex, NULL, 0);
    if (needed == 0) {
        fprintf(stderr, "%s: unknown regex error\n", context);
        return;
    }

    error_buffer = malloc(needed);
    if (error_buffer == NULL) {
        fprintf(stderr, "%s: unable to allocate error buffer\n", context);
        return;
    }

    memset(error_buffer, 0, needed);
    regerror(errcode, regex, error_buffer, needed);
    error_buffer[needed - 1] = '\0';

    fprintf(stderr, "%s: %s\n", context, error_buffer);

    free(error_buffer);
}

static int match_pattern(const char *text)
{
    regex_t regex;
    int ret;

    if (text == NULL) {
        fprintf(stderr, "match_pattern: NULL input\n");
        return -1;
    }

    ret = regcomp(&regex, "a.*b$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        report_regex_error(ret, &regex, "Regex compilation failed");
        regfree(&regex);
        return -1;
    }

    ret = regexec(&regex, text, 0, NULL, 0);

    if (ret == 0) {
        regfree(&regex);
        return 1;
    }

    if (ret == REG_NOMATCH) {
        regfree(&regex);
        return 0;
    }

    report_regex_error(ret, &regex, "Regex execution failed");
    regfree(&regex);
    return -1;
}

int main(void)
{
    const char *tests[] = {
        "acb",
        "ab",
        "axyzb",
        "a123b",
        "bca",
        "abc",
        "b",
        "a"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = match_pattern(tests[i]);
        if (result < 0) {
            return EXIT_FAILURE;
        }
        if (result == 1) {
            printf("\"%s\" matches\n", tests[i]);
        } else {
            printf("\"%s\" does not match\n", tests[i]);
        }
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder finding (CWE-119/CWE-120): the statically-sized `error_buffer[256]` was replaced with a dynamically allocated buffer sized by a first call to `regerror(errcode, regex, NULL, 0)`, which returns the exact required length including the null terminator. This eliminates any possibility of truncation or improper bounds assumptions, since the buffer is always exactly as large as the message requires, is zero-initialized, explicitly null-terminated as a defensive measure, and freed after use. Allocation failure is handled gracefully without dereferencing a NULL pointer. The rest of the logic — NULL input validation, proper `regfree` on every exit path, and distinct handling of match, no-match, and error cases — was already sound and is preserved.
