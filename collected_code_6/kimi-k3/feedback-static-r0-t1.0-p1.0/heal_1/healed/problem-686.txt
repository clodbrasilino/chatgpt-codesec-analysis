#include <stdio.h>
#include <stdlib.h>

void findFrequency(const int arr[], int n)
{
    int *freq = (int *)calloc(n, sizeof(int));
    if (freq == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return;
    }

    for (int i = 0; i < n; i++)
    {
        freq[i] = 1;
        for (int j = i + 1; j < n; j++)
        {
            if (arr[i] == arr[j])
            {
                freq[i]++;
                freq[j] = 0;
            }
        }
    }

    for (int i = 0; i < n; i++)
    {
        if (freq[i] != 0)
        {
            printf("%d occurs %d times\n", arr[i], freq[i]);
        }
    }

    free(freq);
    freq = NULL;
}

int main(void)
{
    int n;

    printf("Enter the number of elements: ");
    if (scanf("%d", &n) != 1 || n <= 0)
    {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    int *arr = (int *)malloc((size_t)n * sizeof(int));
    if (arr == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter %d elements: ", n);
    for (int i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
        {
            fprintf(stderr, "Invalid input\n");
            free(arr);
            arr = NULL;
            return EXIT_FAILURE;
        }
    }

    findFrequency(arr, n);

    free(arr);
    arr = NULL;

    return EXIT_SUCCESS;
}
