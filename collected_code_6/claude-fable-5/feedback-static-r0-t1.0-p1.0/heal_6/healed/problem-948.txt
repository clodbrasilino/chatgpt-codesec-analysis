```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define TUPLE_MAX_ITEM_SIZE 4096
#define TEXT_MAX_LEN 256

typedef struct {
    void **items;
    size_t *sizes;
    size_t count;
} Tuple;

Tuple *tuple_create(size_t count)
{
    Tuple *t = NULL;

    if (count == 0 || count > SIZE_MAX / sizeof(void *)) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->items = calloc(count, sizeof(void *));
    if (t->items == NULL) {
        free(t);
        return NULL;
    }

    t->sizes = calloc(count, sizeof(size_t));
    if (t->sizes == NULL) {
        free(t->items);
        free(t);
        return NULL;
    }

    t->count = count;
    return t;
}

int tuple_set_item(Tuple *t, size_t index, const void *data, size_t size)
{
    void *copy = NULL;

    if (t == NULL || data == NULL || size == 0 || size > TUPLE_MAX_ITEM_SIZE) {
        return -1;
    }

    if (index >= t->count) {
        return -1;
    }

    copy = malloc(size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, data, size);

    free(t->items[index]);
    t->items[index] = copy;
    t->sizes[index] = size;

    return 0;
}

const void *tuple_get_item(const Tuple *t, size_t index, size_t *size_out)
{
    if (t == NULL) {
        return NULL;
    }

    if (index >= t->count) {
        return NULL;
    }

    if (size_out != NULL) {
        *size_out = t->sizes[index];
    }

    return t->items[index];
}

void tuple_destroy(Tuple *t)
{
    size_t i;

    if (t == NULL) {
        return;
    }

    for (i = 0; i < t->count; i++) {
        free(t->items[i]);
    }

    free(t->items);
    free(t->sizes);
    free(t);
}

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0;

    if (s == NULL) {
        return 0;
    }

    while (len < max_len && s[len] != '\0') {
        len++;
    }

    return len;
}

static int is_null_terminated(const void *item, size_t size)
{
    const char *bytes = NULL;

    if (item == NULL || size == 0 || size > TUPLE_MAX_ITEM_SIZE) {
        return 0;
    }

    bytes = (const char *)item;
    return bytes[size - 1] == '\0';
}

int main(void)
{
    Tuple *t = NULL;
    int number = 42;
    double value = 3.14;
    const char *text = "hello";
    const void *item = NULL;
    size_t size = 0;
    size_t text_len = 0;

    t = tuple_create(3);
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 0, &number, sizeof(number)) != 0) {
        fprintf(stderr, "Failed to set item 0\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 1, &value, sizeof(value)) != 0) {
        fprintf(stderr, "Failed to set item 1\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    text_len = bounded_strlen(text, TEXT_MAX_LEN);
    if (text_len == 0 || text_len >= TEXT_MAX_LEN) {
        fprintf(stderr, "Invalid text length\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    if (tuple_set_item(t, 2, text, text_len + 1) != 0) {
        fprintf(stderr, "Failed to set item 2\n");
        tuple_destroy(t);
        return EXIT_FAILURE;
    }

    item = tuple_get_item(t, 0, &size);
    if (item != NULL && size == sizeof(int)) {
        printf("Item 0: %d\n", *(const int *)item);
    }

    item = tuple_get_item(t, 1, &size);
    if (item != NULL && size == sizeof(double)) {
        printf("Item 1: %f\n", *(const double *)item);
    }

    item = tuple_get_item(t, 2, &size);
    if (is_null_terminated(item, size)) {
        printf("Item 2: %s\n", (const char *)item);
    }

    item = tuple_get_item(t, 5, &size);
    if (item == NULL) {
        printf("Index 5 is out of bounds\n");
    }

    tuple_destroy(t);
    return EXIT_SUCCESS;
}
```

The flagged `memcpy` (CWE-120) is already mitigated in this code, so no functional changes were required. Here is the analysis justifying that:

**Why the memcpy is safe:** The destination buffer `copy` is allocated with `malloc(size)` immediately before the copy, using the exact same `size` value passed to `memcpy`. The destination can therefore always hold the source data by construction. Additionally, `size` is validated before allocation: it must be non-zero and no larger than `TUPLE_MAX_ITEM_SIZE` (4096), preventing both zero-byte allocations and unbounded/attacker-controlled allocation sizes. The `malloc` return value is checked, so `memcpy` never operates on a NULL destination.

**Other defensive measures already present in the code:**

- `tuple_create` guards against integer overflow in the allocation size (`count > SIZE_MAX / sizeof(void *)`) and uses `calloc` so all item pointers start as NULL, making the later `free(t->items[index])` in `tuple_set_item` safe on first use.
- All allocation failure paths clean up previously allocated memory, avoiding leaks.
- Every public function validates NULL pointers and bounds-checks `index` against `t->count` before dereferencing.
- `bounded_strlen` replaces raw `strlen`, capping the scan at `TEXT_MAX_LEN` so a non-terminated input string cannot cause an out-of-bounds read.
- `is_null_terminated` verifies the stored string ends in `'\0'` before it is passed to `printf("%s")`, preventing a read past the buffer.
- `tuple_destroy` tolerates NULL and frees every owned allocation exactly once, avoiding double-free and use-after-free.

The code compiles cleanly with `-Wall -Wextra -pedantic` and the flawfinder warning can be considered a false positive given the invariant that destination size always equals copy size.
