#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define IPV4_PATTERN "^((25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9][0-9]|[0-9])\\.){3}(25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9][0-9]|[0-9])$"
#define IPV6_PATTERN "^([0-9a-fA-F]{1,4}:){7}[0-9a-fA-F]{1,4}$"

int is_valid_ip(const char *ip)
{
    regex_t regex;
    int ret;

    if (ip == NULL) {
        return 0;
    }

    ret = regcomp(&regex, IPV4_PATTERN, REG_EXTENDED);
    if (ret == 0) {
        ret = regexec(&regex, ip, 0, NULL, 0);
        regfree(&regex);
        if (ret == 0) {
            return 1;
        }
    }

    ret = regcomp(&regex, IPV6_PATTERN, REG_EXTENDED);
    if (ret == 0) {
        ret = regexec(&regex, ip, 0, NULL, 0);
        regfree(&regex);
        if (ret == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    const char *test_ips[] = {
        "192.168.1.1",
        "256.1.1.1",
        "2001:0db8:85a3:0000:0000:8a2e:0370:7334",
        "2001:db8::1",
        "10.0.0.256",
        "172.16.0.1",
        "invalid_ip"
    };
    size_t num_tests = sizeof(test_ips) / sizeof(test_ips[0]);
    size_t i;

    for (i = 0; i < num_tests; i++) {
        printf("%s: %s\n", test_ips[i], is_valid_ip(test_ips[i]) ? "Valid" : "Invalid");
    }

    return EXIT_SUCCESS;
}
