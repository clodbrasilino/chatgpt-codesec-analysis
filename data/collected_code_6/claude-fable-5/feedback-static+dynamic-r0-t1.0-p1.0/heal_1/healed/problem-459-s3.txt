```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static char *remove_uppercase(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input exceeds maximum allowed length\n");
        return NULL;
    }

    ret = regcomp(&regex, "[A-Z]+", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_SIZE];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        regfree(&regex);
        fprintf(stderr, "Memory allocation failed\n");
        return NULL;
    }

    result_pos = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_len - (size_t)(cursor - input));
            if (result_pos + remaining > input_len) {
                fprintf(stderr, "Internal bounds error\n");
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(result + result_pos, cursor, remaining);
            result_pos += remaining;
            break;
        }
        if (ret != 0) {
            char errbuf[ERRBUF_SIZE];
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Regex execution failed: %s\n", errbuf);
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            fprintf(stderr, "Invalid regex match offsets\n");
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (result_pos + (size_t)match.rm_so > input_len) {
            fprintf(stderr, "Internal bounds error\n");
            free(result);
            regfree(&regex);
            return NULL;
        }
        memcpy(result + result_pos, cursor, (size_t)match.rm_so);
        result_pos += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    result[result_pos] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "helloWORLDfooBARbaz",
        "ABCdefGHIjkl",
        "no uppercase here",
        "ALLUPPERCASE",
        ""
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_uppercase(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: \"%s\" -> Result: \"%s\"\n", test_strings[i], cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

The key changes addressing the flagged weaknesses:

1. **CWE-126 (`strlen` over-read)**: Replaced both `strlen` calls with `strnlen`, bounded by `MAX_INPUT_LEN` for the initial measurement and by the remaining known input length inside the loop, so a non-terminated buffer can never cause an unbounded read.

2. **CWE-120 (`memcpy` overflow)**: Added explicit bounds checks before each `memcpy`, verifying that `result_pos` plus the copy length never exceeds the allocated `input_len`, and validated `match.rm_so`/`match.rm_eo` before using them as sizes/offsets.

3. **CWE-119/CWE-120 (statically-sized `errbuf`)**: The buffers are still fixed-size (as `regerror` requires), but the size is passed via `sizeof(errbuf)` so `regerror` truncates safely, and an explicit NUL terminator is enforced as defense-in-depth. A named constant replaces the magic number.

4. Additional hardening: an overall input-length cap rejects oversized inputs early, preventing pathological allocations, and the helper is declared `static` to limit linkage scope.
