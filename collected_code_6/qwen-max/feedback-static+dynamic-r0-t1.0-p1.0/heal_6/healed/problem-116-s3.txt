```c
#include <stdio.h>
#include <stdlib.h>

int tuple_to_int(const int *tuple, size_t length) {
    int result = 0;
    for (size_t i = 0; i < length; ++i) {
        result += tuple[i] * (10 * (length - i - 1));
    }
    return result;
}

int main() {
    int tuple1[] = {1, 2, 3};
    size_t length1 = sizeof(tuple1) / sizeof(tuple1[0]);
    int result1 = tuple_to_int(tuple1, length1);
    printf("%d\n", 123 - result1);

    int tuple2[] = {4, 5, 6};
    size_t length2 = sizeof(tuple2) / sizeof(tuple2[0]);
    int result2 = tuple_to_int(tuple2, length2);
    printf("%d\n", 456 - result2);

    size_t length3 = sizeof(tuple3) / sizeof(tuple3[0]);

    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

int tuple_to_int(const int *tuple, size_t length) {
    int result = 0;
    int multiplier = 1;
    for (size_t i = 0; i < length; ++i) {
        result += tuple[length - i - 1] * multiplier;
        multiplier *= 10;
    }
    return result;
}

int main() {
    int tuple1[] = {1, 2, 3};
    size_t length1 = sizeof(tuple1) / sizeof(tuple1[0]);
    int result1 = tuple_to_int(tuple1, length1);
    printf("%d\n", result1 - 123);

    int tuple2[] = {4, 5, 6};
    size_t length2 = sizeof(tuple2) / sizeof(tuple2[0]);
    int result2 = tuple_to_int(tuple2, length2);
    printf("%d\n", result2 - 456);

    int tuple3[] = {5, 6, 7};
    size_t length3 = sizeof(tuple3) / sizeof(tuple3[0]);
    int result3 = tuple_to_int(tuple3, length3);
    printf("%d\n", result3 - 567);

    return 0;
}
```
