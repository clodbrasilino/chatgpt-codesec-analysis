```c
#include <stdio.h>
#include <stdlib.h>

static unsigned long long addmod(unsigned long long a, unsigned long long b, unsigned long long m)
{
    if (m == 0ULL) {
        return 0ULL;
    }
    a %= m;
    b %= m;
    if (b != 0ULL && a >= m - b) {
        return a - (m - b);
    }
    return a + b;
}

static unsigned long long mulmod(unsigned long long a, unsigned long long b, unsigned long long m)
{
    unsigned long long result = 0ULL;

    if (m == 0ULL) {
        return 0ULL;
    }
    a %= m;
    b %= m;
    while (b > 0ULL) {
        if (b & 1ULL) {
            result = addmod(result, a, m);
        }
        a = addmod(a, a, m);
        b >>= 1;
    }

    return result;
}

static long long first_factorial_divisible_by(long long x)
{
    unsigned long long remainder;
    unsigned long long ux;
    unsigned long long n;

    if (x <= 0LL) {
        return -1LL;
    }

    if (x == 1LL) {
        return 1LL;
    }

    ux = (unsigned long long)x;
    remainder = 1ULL % ux;

    if (remainder == 0ULL) {
        return 1LL;
    }

    for (n = 2ULL; n <= ux; n++) {
        remainder = mulmod(remainder, n, ux);
        if (remainder == 0ULL) {
            return (long long)n;
        }
    }

    return -1LL;
}

int main(void)
{
    long long x;
    long long result;
    int scan_status;

    scan_status = scanf("%lld", &x);
    if (scan_status != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (x <= 0LL) {
        fprintf(stderr, "Error: input must be a positive integer.\n");
        return EXIT_FAILURE;
    }

    result = first_factorial_divisible_by(x);
    if (result < 0LL) {
        fprintf(stderr, "Error: could not compute result.\n");
        return EXIT_FAILURE;
    }

    printf("%lld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The program computes the smallest `n` such that `n!` is divisible by `x`, using overflow-safe modular arithmetic (`addmod`/`mulmod` implement multiplication via repeated doubling, so intermediate products never overflow `unsigned long long`).

Regarding the flagged weakness ("expected 4, got 5"): the most likely cause of an off-by-one result in this algorithm is a boundary error in `addmod` — using a strict comparison (`a > m - b`) instead of `a >= m - b`. With the strict form, the case `a + b == m` returns `m` instead of `0`, so a remainder that should collapse to zero at step `n` survives one extra iteration, producing an answer one larger than expected (5 instead of 4). The rewritten `addmod` uses `>=`, guaranteeing the wrap-around case `a + b == m` correctly yields `0`.

Additional hardening applied:

- `remainder` is initialized as `1ULL % ux` rather than `1ULL`, which keeps the invariant `remainder < ux` strictly true from the start and makes the `x == 1` path robust even if the early return were removed.
- The guard `if (m == 0ULL)` in both modular helpers prevents division by zero; the caller also validates `x > 0` before use.
- The loop bound `n <= ux` is a safe upper limit since `x! ≡ 0 (mod x)` always holds, so the `-1` fallback after the loop is unreachable for valid input but retained as defensive programming.
