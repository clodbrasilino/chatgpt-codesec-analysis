```c
#include <stdio.h>
#include <stdlib.h>

int maxProfit(int k, const int *prices, int n)
{
    int *prev;
    int *curr;
    int *tmp;
    int t;
    int d;
    int result;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (d = 1; d < n; d++) {
            if (prices[d] > prices[d - 1]) {
                result += prices[d] - prices[d - 1];
            }
        }
        return result;
    }

    prev = calloc((size_t)n, sizeof(int));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc((size_t)n, sizeof(int));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    result = 0;

    for (t = 1; t <= k; t++) {
        int maxDiff = prev[0] - prices[0];
        curr[0] = 0;
        for (d = 1; d < n; d++) {
            if (prev[d] - prices[d] > maxDiff) {
                maxDiff = prev[d] - prices[d];
            }
            curr[d] = curr[d - 1];
            if (prices[d] + maxDiff > curr[d]) {
                curr[d] = prices[d] + maxDiff;
            }
        }
        if (curr[n - 1] > result) {
            result = curr[n - 1];
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    free(prev);
    free(curr);

    return result;
}

int main(void)
{
    int prices0[] = {10, 22, 5, 75, 65, 80};
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int prices2[] = {2, 4, 1};

    struct {
        int k;
        const int *prices;
        int n;
        int expected;
    } tests[] = {
        {2, prices0, (int)(sizeof(prices0) / sizeof(prices0[0])), 87},
        {2, prices1, (int)(sizeof(prices1) / sizeof(prices1[0])), 7},
        {2, prices2, (int)(sizeof(prices2) / sizeof(prices2[0])), 2}
    };

    int numTests = (int)(sizeof(tests) / sizeof(tests[0]));
    int i;
    int failures = 0;

    for (i = 0; i < numTests; i++) {
        int actual = maxProfit(tests[i].k, tests[i].prices, tests[i].n);
        if (actual == tests[i].expected) {
            printf("Test %d passed: expected %d, got %d\n",
                   i, tests[i].expected, actual);
        } else {
            printf("Test %d FAILED: expected %d, got %d\n",
                   i, tests[i].expected, actual);
            failures++;
        }
    }

    if (failures > 0) {
        printf("%d of %d tests failed\n", failures, numTests);
        return EXIT_FAILURE;
    }

    printf("All %d tests passed\n", numTests);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses listed in the original comments referred to an earlier, broken revision. I verified the current implementation against all three failure cases and they are resolved:

1. **Test 2 (`{2, 4, 1}`, k=2, expected 2):** The `k >= n / 2` guard correctly routes this to the greedy unlimited-transaction path (since `2 >= 3/2`), summing all positive deltas: `4 - 2 = 2`. An earlier version likely lacked this shortcut or mishandled the `n <= 1` / small-`n` boundary, returning 0.

2. **Test 0 (`{10, 22, 5, 75, 65, 80}`, k=2, expected 87):** The DP correctly maintains `maxDiff = max(prev[d] - prices[d])` *inside* the day loop before computing `curr[d]`, yielding the O(k·n) formulation. Tracing: transaction 1 gives 75, transaction 2 gives 87 (buy 10→sell 22, buy 5→sell... actually buy 10→22 = 12 plus buy 5→80 = 75, total 87). The "got 5" symptom suggests the old code only captured a single price delta — likely `maxDiff` was updated after `curr[d]` or reset per iteration.

3. **Test 1 (`{3, 2, 6, 5, 0, 3}`, k=2, expected 7):** Correctly computes 4 (buy 2→sell 6) + 3 (buy 0→sell 3) = 7 via the buffer-swap between transaction rounds. The "got 0" failure hints the old version failed to swap/carry `prev` state.

The rewrite also preserves the defensive qualities worth keeping: NULL/degenerate-input guards, `calloc` for zero-initialized DP buffers, allocation-failure handling with cleanup of the first buffer before exit, and no leaks (both buffers freed regardless of swap parity).
