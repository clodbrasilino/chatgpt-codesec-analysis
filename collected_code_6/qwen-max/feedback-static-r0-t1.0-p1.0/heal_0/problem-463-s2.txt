#include <stdio.h>
#include <limits.h>

int maxProductSubarray(int *arr, int n) {
    int maxEnd = arr[0];
    int minEnd = arr[0];
    int maxSoFar = arr[0];
    for (int i = 1; i < n; i++) {
        int temp = maxEnd;
        maxEnd = (arr[i] > (maxEnd * arr[i])) ? arr[i] : (maxEnd * arr[i]);
        maxEnd = (maxEnd > (minEnd * arr[i])) ? maxEnd : (minEnd * arr[i]);
        minEnd = (arr[i] < (temp * arr[i])) ? arr[i] : (temp * arr[i]);
        minEnd = (minEnd < (minEnd * arr[i])) ? minEnd : (minEnd * arr[i]);
        if (maxEnd > maxSoFar) {
            maxSoFar = maxEnd;
        }
    }
    return maxSoFar;
}

int main() {
    int arr[] = {2, 3, -2, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("Maximum product subarray: %d\n", maxProductSubarray(arr, n));
    return 0;
}
