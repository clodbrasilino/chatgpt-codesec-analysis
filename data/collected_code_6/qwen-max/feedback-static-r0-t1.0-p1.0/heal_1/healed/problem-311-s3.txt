#include <stdio.h>
#include <limits.h>

unsigned int setLeftMostUnsetBit(unsigned int n) {
    unsigned int bit = 1;
    while (bit) {
        if (!(n & bit)) {
            return n | bit;
        }
        bit = bit << 1;
        if (bit == 0) break; // Prevent infinite loop when bit overflows
    }
    return n;
}

int main() {
    unsigned int num = 0b101010; // Example number in binary: 42 in decimal
    printf("Original number: %u (in binary: %08b)\n", num, num);
    num = setLeftMostUnsetBit(num);
    printf("After setting left most unset bit: %u (in binary: %08b)\n", num, num);
    return 0;
}
