```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;
    char tmp;
    char *endptr;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }
    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }
    if (j <= i) {
        return -1;
    }

    tmp = digits[i];
    digits[i] = digits[j];
    digits[j] = tmp;

    for (int a = i + 1, b = len - 1; a < b; a++, b--) {
        tmp = digits[a];
        digits[a] = digits[b];
        digits[b] = tmp;
    }

    errno = 0;
    result = strtoll(digits, &endptr, 10);
    if (errno == ERANGE || endptr == digits || *endptr != '\0') {
        return -1;
    }

    if (result <= n) {
        return -1;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            if (printf("next_bigger(%lld) = no bigger number exists\n", inputs[k]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("next_bigger(%lld) = %lld\n", inputs[k], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged weakness (CWE-119/CWE-120, statically-sized buffer `digits[DIGITS_BUF_SIZE]`) is already properly mitigated in this code, so the rewrite preserves the existing defenses rather than restructuring them:

1. **Buffer sizing is provably safe.** A `long long` is at most 64 bits; its decimal representation (`LLONG_MAX` = 19 digits) plus a NUL terminator fits comfortably in 32 bytes. Negative values are rejected before formatting, so no sign character is ever needed.

2. **Bounded write with verification.** `snprintf(digits, sizeof(digits), ...)` never writes past the buffer, and the return value is checked both for encoding errors (`len < 0`) and truncation (`(size_t)len >= sizeof(digits)`), so subsequent indexing via `len` is always within bounds.

3. **All index arithmetic is bounded by `len`.** The scan loops (`i`, `j`, and the reversal loop) operate strictly within `[0, len - 1]`, with explicit guards (`i < 0`, `j <= i`) preventing out-of-range access.

4. **Robust reconversion.** `strtoll` is validated via `errno == ERANGE`, `endptr == digits`, and `*endptr != '\0'`, catching overflow and malformed intermediate states.

5. **Overflow-safe result check.** `result <= n` guards against the swapped permutation not actually producing a larger value.

The `memset` before `snprintf` is technically redundant (snprintf NUL-terminates on success), but it is harmless defense-in-depth and I retained it. No functional or security changes were required — the original code correctly addressed the Flawfinder finding.
