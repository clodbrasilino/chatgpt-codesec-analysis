```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

size_t longest_word_length(const char *str)
{
    size_t max_len = 0;
    size_t current_len = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i])) {
            current_len++;
            if (current_len > max_len) {
                max_len = current_len;
            }
        } else {
            current_len = 0;
        }
    }

    return max_len;
}

int main(void)
{
    char buffer[1024];
    size_t result;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = longest_word_length(buffer);

    if (printf("Longest word length: %zu\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning about the statically-sized `buffer` array (CWE-119/CWE-120) is already properly mitigated in this code:

1. **Bounded input reading**: `fgets(buffer, sizeof(buffer), stdin)` limits input to at most 1023 characters plus the null terminator, preventing buffer overflow. Using `sizeof(buffer)` rather than a hardcoded literal ensures the bound stays correct even if the array size changes.

2. **NULL check on input**: The return value of `fgets` is checked, handling EOF or read errors gracefully.

3. **Safe newline stripping**: `strcspn(buffer, "\n")` returns an index guaranteed to be within the string bounds (it points to the newline or the existing null terminator), so the write is always in-bounds — safer than the common `buffer[strlen(buffer)-1]` idiom which can underflow on an empty string.

4. **Correct `isalpha` usage**: The cast to `(unsigned char)` before passing to `isalpha` avoids undefined behavior for characters with negative values on platforms where `char` is signed.

5. **NULL guard in `longest_word_length`**: Defends against a NULL pointer dereference.
