// Importing necessary libraries
#include<stdio.h>
#include<stdlib.h>

// Function Declaration
void findPairs(int* arr1, int* arr2, int size1, int size2, int k);

int main(void) {

    int arr1[] = {1, 3, 5};
    int arr2[] = {2, 4, 6};
    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    int k = 2;

    findPairs(arr1, arr2, size1, size2, k);

    return EXIT_SUCCESS;
}

void findPairs(int* arr1, int* arr2, int size1, int size2, int k) {
    if(arr1 == NULL || arr2 == NULL || size1 <= 0 || size2 <= 0 || k <= 0){
        printf("Invalid inputs\n");
        return;
    }

    int count = 0;

    for(int i = 0; i < size1; i++) {
        for(int j = 0; j < size2; j++) {
            printf("(%d,%d)\n", arr1[i], arr2[j]);
            count++;

            if(count == k)
                return;

        }
    }

    if(count < k){
        printf("Not enough pairs\n");
    }
}
