```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    double numerator;
    double denominator;
} Tuple;

typedef struct {
    double quotient;
    bool success;
} DivResult;

DivResult divide_tuple(const Tuple input_tuple) {
    DivResult result;
    
    if (input_tuple.denominator == 0.0) {
        result.success = false;
        result.quotient = 0.0;
    } else {
        result.success = true;
        result.quotient = input_tuple.numerator / input_tuple.denominator;
    }
    
    return result;
}

void process_tuples(const Tuple* tuples, size_t count) {
    if (tuples == NULL && count > 0) {
        return;
    }

    for (size_t i = 0; i < count; i++) {
        DivResult result = divide_tuple(tuples[i]);
        if (result.success) {
            printf("%.6f\n", result.quotient);
        } else {
            printf("Error: Division by zero\n");
        }
    }
}

int main(void) {
    Tuple tuples[] = {
        {10.0, 2.0},
        {5.0, 0.0},
        {-9.0, 3.0},
        {4.2, 2.1}
    };
    
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    
    process_tuples(tuples, count);
    
    return EXIT_SUCCESS;
}
```
