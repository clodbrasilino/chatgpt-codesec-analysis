#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int compare_chars(const void *a, const void *b) {
    return (*(const char *)a - *(const char *)b);
}

int sort_string(char *str, size_t max_len) {
    size_t len;
    
    if (str == NULL || max_len == 0) {
        return -1;
    }
    
    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }
    
    qsort(str, len, sizeof(char), compare_chars);
    return 0;
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read_len;
    char *newline;
    size_t len;
    
    read_len = getline(&buffer, &buffer_size, stdin);
    if (read_len == -1) {
        free(buffer);
        return EXIT_FAILURE;
    }
    
    newline = strchr(buffer, '\n');
    if (newline != NULL) {
        *newline = '\0';
    }
    
    len = strnlen(buffer, buffer_size);
    if (len == 0 || len >= buffer_size) {
        free(buffer);
        return EXIT_FAILURE;
    }
    
    if (sort_string(buffer, buffer_size) != 0) {
        free(buffer);
        return EXIT_FAILURE;
    }
    
    printf("%s\n", buffer);
    
    free(buffer);
    return EXIT_SUCCESS;
}
