```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>

int sum_array(const int *arr, size_t size, long *result)
{
    size_t i;
    long sum;

    if (arr == NULL || result == NULL || size == 0) {
        return -1;
    }

    sum = 0;
    for (i = 0; i < size; i++) {
        sum += arr[i];
    }

    *result = sum;
    return 0;
}

int main(void)
{
    int arr[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    long result;
    int status;

    status = sum_array(arr, size, &result);
    if (status != 0) {
        fprintf(stderr, "Error: failed to compute sum\n");
        return EXIT_FAILURE;
    }

    printf("Sum: %ld\n", result);
    return EXIT_SUCCESS;
}
```
