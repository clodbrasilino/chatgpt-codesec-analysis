```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int safe_copy(void *dest, size_t dest_size, const void *src, size_t copy_size)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (copy_size == 0) {
        return 0;
    }
    if (copy_size > dest_size) {
        return -1;
    }
    memcpy(dest, src, copy_size);
    return 0;
}

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t copy_size;
    size_t move_size;
    size_t total_size;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    total_size = n * sizeof(int);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    copy_size = k * sizeof(int);
    move_size = (n - k) * sizeof(int);

    if (copy_size > total_size || move_size > total_size) {
        return -1;
    }

    if (copy_size + move_size != total_size) {
        return -1;
    }

    temp = malloc(copy_size);
    if (temp == NULL) {
        return -1;
    }

    if (safe_copy(temp, copy_size, arr, copy_size) != 0) {
        free(temp);
        return -1;
    }

    memmove(arr, arr + k, move_size);

    if (safe_copy(arr + (n - k), total_size - move_size, temp, copy_size) != 0) {
        free(temp);
        return -1;
    }

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weakness (CWE-120, `memcpy` in `safe_copy`):**

The flawfinder warning on `memcpy` is a **false positive** in this code, and no functional change is required. Here's why the code is already safe:

1. **Guarded copy size**: `safe_copy` explicitly validates `copy_size > dest_size` before calling `memcpy`, which is exactly the mitigation CWE-120 requires. The destination capacity is passed as an explicit parameter, so the check is enforced at the call boundary rather than relying on caller discipline alone.

2. **NULL checks**: Both `dest` and `src` are validated, preventing undefined behavior from `memcpy(NULL, ...)`.

3. **Integer overflow protection**: `split_and_rotate` guards `n > SIZE_MAX / sizeof(int)` before computing `total_size = n * sizeof(int)`, so `copy_size` and `move_size` cannot wrap around.

4. **Consistency checks**: The invariant `copy_size + move_size == total_size` is verified, and each partial size is checked against `total_size`, making the subsequent `memmove` and second `safe_copy` bounds-correct by construction.

5. **Correct destination sizing**: In the second `safe_copy`, the remaining capacity at `arr + (n - k)` is correctly expressed as `total_size - move_size`, which equals `copy_size` — the exact space available.

Memory hygiene is also sound: `temp` is freed on every exit path after allocation, so there are no leaks. The expected output for the sample input is `52 36 12 10 5 6` (left rotation by 2 places the first `k` elements at the end).
