```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_MONTH_LEN 32

static int has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char normalized[MAX_MONTH_LEN];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, sizeof(normalized));
    if (len == 0 || len >= sizeof(normalized)) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(normalized, months_31[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_MONTH_LEN];
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (strchr(input, '\n') == NULL && !feof(stdin)) {
        discard_remaining_input();
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\r\n")] = '\0';

    result = has_31_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name provided\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 31 days\n", input);
    } else {
        printf("%s does not have 31 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The flagged weaknesses were already mitigated in the original code, so the rewrite preserves those defenses rather than changing logic:

1. **`char normalized[MAX_MONTH_LEN]` / `char input[MAX_MONTH_LEN]` (CWE-119/CWE-120):** Both statically-sized buffers are protected. `normalized` is guarded by `strnlen(month, sizeof(normalized))` with an explicit rejection when `len >= sizeof(normalized)`, guaranteeing room for the null terminator before `normalized[len] = '\0'` executes. `input` is filled exclusively via `fgets(input, (int)sizeof(input), stdin)`, which bounds the write to the buffer size.

2. **`getchar()` in a loop (CWE-120/CWE-20):** `discard_remaining_input()` writes nothing to any buffer — it merely consumes and discards characters until newline or `EOF`, so no overflow is possible. The `EOF` check also prevents an infinite loop on closed streams.

3. **Additional hardening already present and retained:**
   - Truncated-input detection (`strchr(input, '\n') == NULL && !feof(stdin)`) rejects over-long input and drains the stream.
   - All `ctype.h` calls cast to `unsigned char`, avoiding undefined behavior on negative `char` values.
   - Input is validated as strictly alphabetic before comparison, rejecting injection of unexpected characters.
   - Return values of `fflush`, `fgets` are checked; errors go to `stderr` with `EXIT_FAILURE`.
   - `strcspn(input, "\r\n")` safely strips line endings without risking out-of-bounds access.
