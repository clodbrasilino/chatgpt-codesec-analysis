#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* snake_to_camel(const char* snake_str) {
    if (!snake_str) return NULL;

    size_t len = strlen(snake_str);
    char* camel_str = (char*)malloc(len + 1);
    if (!camel_str) {
        return NULL;
    }

    int capitalize_next = 1;
    char* out = camel_str;

    for (size_t i = 0; i < len; i++) {
        if (snake_str[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next) {
                *out++ = toupper((unsigned char)snake_str[i]);
                capitalize_next = 0;
            } else {
                *out++ = snake_str[i];
            }
        }
    }
    *out = '\0';

    return camel_str;
}

int main(void) {
    const char* snake = "this_is_a_snake_case_string";
    char* camel = snake_to_camel(snake);

    if (camel) {
        printf("%s\n", camel);
        free(camel);
    } else {
        fprintf(stderr, "Conversion failed.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
