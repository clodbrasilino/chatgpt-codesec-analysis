#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

int find_substrings(const char *str, size_t str_len, const char *substr, size_t substr_len, size_t **positions) {
    if (str == NULL || substr == NULL || positions == NULL) {
        return -1;
    }

    if (substr_len == 0 || str_len == 0 || substr_len > str_len) {
        *positions = NULL;
        return 0;
    }

    size_t count = 0;
    size_t capacity = 10;
    size_t *pos_array = (size_t *)malloc(capacity * sizeof(size_t));
    
    if (pos_array == NULL) {
        return -1;
    }

    size_t i = 0;
    while (i <= str_len - substr_len) {
        int match = 1;
        for (size_t j = 0; j < substr_len; j++) {
            if (str[i + j] != substr[j]) {
                match = 0;
                break;
            }
        }

                    free(pos_array);
                    return -1;
                }
                size_t *new_array = (size_t *)realloc(pos_array, new_capacity * sizeof(size_t));
                
                if (new_array == NULL) {
                    free(pos_array);
                    return -1;
                }
                
                pos_array = new_array;
                capacity = new_capacity;
            }

            pos_array[count] = i;
            count++;
            i++;
        } else {
            i++;
        }
    }

    if (count == 0) {
        free(pos_array);
        *positions = NULL;
        return 0;
    }

    if (count > SIZE_MAX / sizeof(size_t)) {
        free(pos_array);
        return -1;
    }

    size_t *final_array = (size_t *)realloc(pos_array, count * sizeof(size_t));
    if (final_array != NULL) {
        pos_array = final_array;
    }

    *positions = pos_array;
    return (int)count;
}
