```c
#include <stdio.h>
#include <stdlib.h>

void square_elements(const int *input, int *output, size_t length) {
    for (size_t i = 0; i < length; i++) {
        output[i] = input[i] * input[i];
    }
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5};
    size_t length = sizeof(list) / sizeof(list[0]);
    int *squares = malloc(length * sizeof(int));
    if (squares == NULL) {
        return EXIT_FAILURE;
    }
    square_elements(list, squares, length);
    for (size_t i = 0; i < length; i++) {
        printf("%d ", squares[i]);
    }
    printf("\n");
    free(squares);
    return EXIT_SUCCESS;
}
```
