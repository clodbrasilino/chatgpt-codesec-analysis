#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define BUFFER_SIZE 258
#define MIN_LENGTH 8

bool validate_string(const char *str, size_t min_length) {
    if (str == NULL) {
        return false;
    }
    
    size_t len = strnlen(str, BUFFER_SIZE - 1);
    if (len < min_length) {
        return false;
    }
    
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    
    for (size_t i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        }
    }
    
    return has_upper && has_lower && has_digit;
}

int main(void) {
    char buffer[BUFFER_SIZE];
    size_t min_len = MIN_LENGTH;
    
    printf("Enter string to validate: ");
    if (fgets(buffer, BUFFER_SIZE, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }
    
    size_t len = strnlen(buffer, BUFFER_SIZE);
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == BUFFER_SIZE - 1) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
    }
    
    if (validate_string(buffer, min_len)) {
        printf("String is valid\n");
    } else {
        printf("String is invalid\n");
    }
    
    return 0;
}
