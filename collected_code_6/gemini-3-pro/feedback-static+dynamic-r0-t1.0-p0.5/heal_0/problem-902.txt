#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    int value;
} KeyValuePair;

typedef struct {
    KeyValuePair *pairs;
    size_t size;
    size_t capacity;
} Dictionary;

Dictionary* create_dictionary(size_t capacity) {
    Dictionary *dict = (Dictionary*)malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    
    dict->pairs = (KeyValuePair*)malloc(capacity * sizeof(KeyValuePair));
    if (!dict->pairs) {
        free(dict);
        return NULL;
    }
    
    dict->size = 0;
    dict->capacity = capacity;
    return dict;
}

void free_dictionary(Dictionary *dict) {
    if (dict) {
        for (size_t i = 0; i < dict->size; i++) {
            free(dict->pairs[i].key);
        }
        free(dict->pairs);
        free(dict);
    }
}

int add_to_dictionary(Dictionary *dict, const char *key, int value) {
    if (!dict || !key) return -1;

    for (size_t i = 0; i < dict->size; i++) {
        if (strcmp(dict->pairs[i].key, key) == 0) {
            dict->pairs[i].value += value;
            return 0;
        }
    }

    if (dict->size >= dict->capacity) {
        size_t new_capacity = dict->capacity == 0 ? 4 : dict->capacity * 2;
        KeyValuePair *new_pairs = (KeyValuePair*)realloc(dict->pairs, new_capacity * sizeof(KeyValuePair));
        if (!new_pairs) return -1;
        dict->pairs = new_pairs;
        dict->capacity = new_capacity;
    }

    dict->pairs[dict->size].key = strdup(key);
    if (!dict->pairs[dict->size].key) return -1;
    
    dict->pairs[dict->size].value = value;
    dict->size++;
    return 0;
}

Dictionary* combine_dictionaries(const Dictionary *dict1, const Dictionary *dict2) {
    if (!dict1 || !dict2) return NULL;

    Dictionary *result = create_dictionary(dict1->size + dict2->size);
    if (!result) return NULL;

    for (size_t i = 0; i < dict1->size; i++) {
        if (add_to_dictionary(result, dict1->pairs[i].key, dict1->pairs[i].value) != 0) {
            free_dictionary(result);
            return NULL;
        }
    }

    for (size_t i = 0; i < dict2->size; i++) {
        if (add_to_dictionary(result, dict2->pairs[i].key, dict2->pairs[i].value) != 0) {
            free_dictionary(result);
            return NULL;
        }
    }

    return result;
}

int main(void) {
    Dictionary *dict1 = create_dictionary(4);
    Dictionary *dict2 = create_dictionary(4);

    if (!dict1 || !dict2) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return 1;
    }

    add_to_dictionary(dict1, "apple", 5);
    add_to_dictionary(dict1, "banana", 3);
    
    add_to_dictionary(dict2, "banana", 4);
    add_to_dictionary(dict2, "orange", 2);

    Dictionary *combined = combine_dictionaries(dict1, dict2);

    if (combined) {
        for (size_t i = 0; i < combined->size; i++) {
            printf("%s: %d\n", combined->pairs[i].key, combined->pairs[i].value);
        }
    }

    free_dictionary(dict1);
    free_dictionary(dict2);
    free_dictionary(combined);

    return 0;
}
