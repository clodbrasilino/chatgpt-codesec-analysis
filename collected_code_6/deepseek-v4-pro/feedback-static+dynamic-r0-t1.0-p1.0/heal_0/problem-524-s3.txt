#include <stdio.h>
#include <stdlib.h>

int maxIncreasingSubsequenceSum(int arr[], int n) {
    if (n <= 0) {
        return 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        dp[i] = arr[i];
    }

    int maxSum = dp[0];

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[j] < arr[i] && dp[j] + arr[i] > dp[i]) {
                dp[i] = dp[j] + arr[i];
            }
        }
        if (dp[i] > maxSum) {
            maxSum = dp[i];
        }
    }

    free(dp);
    return maxSum;
}

int main(void) {
    int arr1[] = {1, 101, 2, 3, 100, 4, 5};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int result1 = maxIncreasingSubsequenceSum(arr1, n1);
    printf("%d\n", result1);

    int arr2[] = {3, 4, 5, 10};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int result2 = maxIncreasingSubsequenceSum(arr2, n2);
    printf("%d\n", result2);

    int arr3[] = {10, 5, 4, 3};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    int result3 = maxIncreasingSubsequenceSum(arr3, n3);
    printf("%d\n", result3);

    int arr4[] = {1};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);

    printf("%d\n", result4);

    printf("%d\n", result6);

    return 0;
}
