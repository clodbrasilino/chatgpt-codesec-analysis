```c
#include <stdio.h>
#include <stddef.h>

static size_t count_odd_rotations(const char *str)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] == '1') {
            count++;
        }
    }

    return count;
}

int main(void)
{
    const char *tests[] = { "011001", "11011", "1010", "0000", "1111" };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < num_tests; i++) {
        size_t result = count_odd_rotations(tests[i]);
        if (printf("String: %s -> Odd rotations: %zu\n", tests[i], result) < 0) {
            return 1;
        }
    }

    return 0;
}
```

The key insight: a rotation of a binary string has an odd decimal value if and only if it ends in `'1'`. Since every character becomes the last character in exactly one rotation, the answer equals the number of `'1'` characters in the string. This gives an O(n) solution with a NULL check, bounded iteration via the null terminator, checked `printf` return values, and no dynamic resources to mismanage.
