#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int compare_desc(const void *a, const void *b)
{
    long long val_a = *(const long long *)a;
    long long val_b = *(const long long *)b;
    
    if (val_a < val_b)
        return 1;
    if (val_a > val_b)
        return -1;
    return 0;
}

int find_largest_products(const int *list1, size_t size1, const int *list2, size_t size2, size_t count, long long *result)
{
    long long *products;
    size_t total_products;
    size_t i;
    size_t j;
    size_t index;

    if (list1 == NULL || list2 == NULL || result == NULL)
        return -1;
    
    if (size1 == 0 || size2 == 0 || count == 0)
        return -1;
    
    if (size1 > SIZE_MAX / size2)
        return -1;
    
    total_products = size1 * size2;
    
    if (count > total_products)
        return -1;

    products = malloc(total_products * sizeof(long long));
    if (products == NULL)
        return -1;

    index = 0;
    for (i = 0; i < size1; i++)
    {
        for (j = 0; j < size2; j++)
        {
            products[index] = (long long)list1[i] * (long long)list2[j];
            index++;
        }
    }

    qsort(products, total_products, sizeof(long long), compare_desc);

    for (i = 0; i < count; i++)
    {
        result[i] = products[i];
    }

    free(products);
    return 0;
}

int main(void)
{
    int list1[] = {5, 3, 8, 1};
    int list2[] = {2, 7, 4};
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    size_t size2 = sizeof(list2) / sizeof(list2[0]);
    size_t count = 5;
    long long *result;
    size_t i;
    int status;

    result = malloc(count * sizeof(long long));
    if (result == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    status = find_largest_products(list1, size1, list2, size2, count, result);
    if (status != 0)
    {
        fprintf(stderr, "Error finding largest products\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("The %zu largest products are:\n", count);
    for (i = 0; i < count; i++)
    {
        printf("%lld\n", result[i]);
    }

    free(result);
    return EXIT_SUCCESS;
}
