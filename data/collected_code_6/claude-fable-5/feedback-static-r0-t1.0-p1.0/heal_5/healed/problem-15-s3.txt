#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t capacity = 8;
    size_t n = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && input[MAX_INPUT_LEN - 1] != '\0') {
        return NULL;
    }

    if (capacity > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;
                char *segment;

                if (seg_len >= SIZE_MAX) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                if (start > len || seg_len > len - start) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                segment = malloc(seg_len + 1);
                if (segment == NULL) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                memcpy(segment, input + start, seg_len);
                segment[seg_len] = '\0';

                if (n == capacity) {
                    if (capacity > SIZE_MAX / (2 * sizeof(char *))) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    capacity *= 2;
                    temp = realloc(tokens, capacity * sizeof(char *));
                    if (temp == NULL) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    tokens = temp;
                }
                tokens[n] = segment;
                n++;
            }
            start = i + 1;
        }
    }

    *count = n;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *text = "HELLOworldFOObarBAZ123";
    char **parts = NULL;
    size_t count = 0;
    size_t i;

    parts = split_at_lowercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);
    printf("Found %zu segment(s):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, parts[i]);
    }

    free_tokens(parts, count);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) is a **false positive** in this code, and no functional change is required. Here's why the copy is provably safe:

1. **Destination size is guaranteed sufficient**: `segment` is allocated with exactly `seg_len + 1` bytes immediately before the copy, and `malloc`'s return value is checked for `NULL`. The `memcpy` writes exactly `seg_len` bytes, leaving room for the null terminator written afterward.

2. **Source bounds are validated**: `seg_len = i - start` where the loop guarantees `i <= len` and `i > start`, so `input + start` through `input + start + seg_len - 1` always lies within the validated input string. The explicit guards (`start > len || seg_len > len - start`) are defensive redundancy on top of the loop invariants.

3. **Input length is bounded**: `strnlen(input, MAX_INPUT_LEN)` caps `len` at 4096, and the check for a missing terminator rejects unterminated buffers, preventing over-reads.

4. **Integer overflow is prevented**: `seg_len >= SIZE_MAX` guards `seg_len + 1` against wrapping, and the `capacity > SIZE_MAX / (2 * sizeof(char *))` check prevents overflow in the `realloc` size computation.

The code as written is already hardened — the static analyzer flags `memcpy` heuristically because it cannot verify the surrounding invariants.
