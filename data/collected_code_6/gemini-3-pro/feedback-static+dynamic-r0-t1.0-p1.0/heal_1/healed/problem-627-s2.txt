#include <stdio.h>
#include <stdlib.h>

int findSmallestMissing(const int* arr, int size) {
    if (arr == NULL || size <= 0) {
        return 1;
    }

    int* present = (int*)calloc(size + 1, sizeof(int));
    if (present == NULL) {
        return -1; 
    }

    for (int i = 0; i < size; i++) {
        if (arr[i] > 0 && arr[i] <= size) {
            present[arr[i]] = 1;
        }
    }

    int missing = size + 1;
    for (int i = 1; i <= size; i++) {
        if (present[i] == 0) {
            missing = i;
            break;
        }
    }

    free(present);
    return missing;
}

int main(void) {
    int arr[] = {3, 4, -1, 1};
    int size = sizeof(arr) / sizeof(arr[0]);

    int missing = findSmallestMissing(arr, size);
    
    if (missing != -1) {
        printf("Smallest missing positive number: %d\n", missing);
    } else {
        printf("Memory allocation failed.\n");
    }

    return 0;
}
