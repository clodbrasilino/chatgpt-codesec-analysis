#include <stdio.h>
#include <stdlib.h>

#define MAX_STR_LEN 100000

char* longestCommonSubsequence(const char* str1, const char* str2) {
    if (!str1 || !str2) {
        return NULL;
    }

    size_t len1 = 0;
    while (len1 < MAX_STR_LEN && str1[len1] != '\0') {
        len1++;
    }

    size_t len2 = 0;
    while (len2 < MAX_STR_LEN && str2[len2] != '\0') {
        len2++;
    }

    size_t** dp = (size_t**)malloc((len1 + 1) * sizeof(size_t*));
    if (!dp) {
        return NULL;
    }

    for (size_t i = 0; i <= len1; i++) {
        dp[i] = (size_t*)malloc((len2 + 1) * sizeof(size_t));
        if (!dp[i]) {
            for (size_t k = 0; k < i; k++) {
                free(dp[k]);
            }
            free(dp);
            return NULL;
        }
        for (size_t j = 0; j <= len2; j++) {
            dp[i][j] = 0;
        }
    }

    for (size_t i = 1; i <= len1; i++) {
        for (size_t j = 1; j <= len2; j++) {
            if (str1[i - 1] == str2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = (dp[i - 1][j] > dp[i][j - 1]) ? dp[i - 1][j] : dp[i][j - 1];
            }
        }
    }

    size_t idx = dp[len1][len2];
    char* lcs = (char*)malloc((idx + 1) * sizeof(char));
    if (!lcs) {
        for (size_t i = 0; i <= len1; i++) {
            free(dp[i]);
        }
        free(dp);
        return NULL;
    }
    
    lcs[idx] = '\0';
    size_t i = len1, j = len2;

    while (i > 0 && j > 0) {
        if (str1[i - 1] == str2[j - 1]) {
            lcs[idx - 1] = str1[i - 1];
            i--;
            j--;
            idx--;
        } else if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }

    for (size_t k = 0; k <= len1; k++) {
        free(dp[k]);
    }
    free(dp);

    return lcs;
}

int main(void) {
    const char* str1 = "AGGTAB";
    const char* str2 = "GXTXAYB";

    char* result = longestCommonSubsequence(str1, str2);

    if (result) {
        printf("Longest Common Subsequence: %s\n", result);
        free(result);
    } else {
        printf("Memory allocation failed or invalid input.\n");
    }

    return 0;
}
