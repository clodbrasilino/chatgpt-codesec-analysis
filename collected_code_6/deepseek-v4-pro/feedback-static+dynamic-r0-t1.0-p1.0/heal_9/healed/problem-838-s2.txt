#include <stdio.h>
#include <string.h>
#include <stddef.h>
#include <stdlib.h>

#define MAX_STRING_LENGTH 100

static int minSwapsToEqual(const char *s1, const char *s2, size_t len) {
    int count1_s1 = 0, count1_s2 = 0;
    int type1 = 0, type2 = 0;

    for (size_t i = 0; i < len; i++) {
        if (s1[i] == '1') count1_s1++;
        if (s2[i] == '1') count1_s2++;
        if (s1[i] != s2[i]) {
            if (s1[i] == '0') type1++;
            else type2++;
        }
    }

    if (count1_s1 != count1_s2) return -1;
    if (type1 % 2 != 0 || type2 % 2 != 0) return -1;

    return type1 / 2 + type2 / 2 + (type1 % 2) * 2;
}

static void clear_input_buffer(void) {
    int c;
    while ((c = getchar()) != '\n' && c != EOF);
}

static int validate_and_trim(char *str, size_t size, const char *input_name) {
    size_t len = strcspn(str, "\n");
    if (len < size && str[len] == '\n') {
        str[len] = '\0';
    } else {
        clear_input_buffer();
        if (size > 0) {
            str[size - 1] = '\0';
        }
        printf("Warning: %s was truncated to %zu characters\n", input_name, size - 1);
    }
    return 0;
}

static int is_valid_binary(const char *str, size_t len) {
    for (size_t i = 0; i < len; i++) {
        if (str[i] != '0' && str[i] != '1') {
            return 0;
        }
    }
    return 1;
}

static char *safe_fgets_dynamic(const char *prompt, size_t *out_len) {
    size_t capacity = 64;
    size_t length = 0;
    char *buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    printf("%s", prompt);

    int c;
    while ((c = getchar()) != '\n' && c != EOF) {
        if (length + 1 >= capacity) {
            size_t new_capacity = capacity * 2;
            if (new_capacity > MAX_STRING_LENGTH + 2) {
                new_capacity = MAX_STRING_LENGTH + 2;
            }
            if (new_capacity <= length + 1) {
                free(buffer);
                clear_input_buffer();
                return NULL;
            }
            char *new_buffer = realloc(buffer, new_capacity);
            if (new_buffer == NULL) {
                free(buffer);
                clear_input_buffer();
                return NULL;
            }
            buffer = new_buffer;
            capacity = new_capacity;
        }
        buffer[length++] = (char)c;
    }

    if (c == EOF && length == 0) {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';
    *out_len = length;

    if (length > MAX_STRING_LENGTH) {
        free(buffer);
        return NULL;
    }

    return buffer;
}

int main(void) {
    size_t len1 = 0, len2 = 0;

    char *s1 = safe_fgets_dynamic("Enter first binary string: ", &len1);
    if (s1 == NULL) {
        printf("Error reading first string or string too long\n");
        return 1;
    }

    char *s2 = safe_fgets_dynamic("Enter second binary string: ", &len2);
    if (s2 == NULL) {
        printf("Error reading second string or string too long\n");
        free(s1);
        return 1;
    }

    if (len1 != len2) {
        printf("Strings must be of equal length\n");
        free(s1);
        free(s2);
        return 1;
    }

    if (len1 == 0) {
        printf("Minimum swaps required: 0\n");
        free(s1);
        free(s2);
        return 0;
    }

    if (!is_valid_binary(s1, len1)) {
        printf("Invalid character in first string\n");
        free(s1);
        free(s2);
        return 1;
    }
    if (!is_valid_binary(s2, len2)) {
        printf("Invalid character in second string\n");
        free(s1);
        free(s2);
        return 1;
    }

    int result = minSwapsToEqual(s1, s2, len1);
    free(s1);
    free(s2);

    if (result == -1) {
        printf("Cannot make strings equal\n");
        return 1;
    }
    printf("Minimum swaps required: %d\n", result);

    return 0;
}
