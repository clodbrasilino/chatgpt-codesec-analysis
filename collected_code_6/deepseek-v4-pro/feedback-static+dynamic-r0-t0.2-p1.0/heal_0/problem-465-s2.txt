#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct DictItem {
    char *key;
    char *value;
    struct DictItem *next;
} DictItem;

typedef struct {
    DictItem *head;
    size_t size;
} Dictionary;

Dictionary *dict_create(void) {
    Dictionary *dict = (Dictionary *)malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    dict->head = NULL;
    dict->size = 0;
    return dict;
}

int dict_add(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) return -1;
    
    DictItem *item = (DictItem *)malloc(sizeof(DictItem));
    if (!item) return -1;
    
    item->key = strdup(key);
    item->value = strdup(value);
    if (!item->key || !item->value) {
        free(item->key);
        free(item->value);
        free(item);
        return -1;
    }
    
    item->next = dict->head;
    dict->head = item;
    dict->size++;
    return 0;
}

void dict_drop_empty(Dictionary *dict) {
    if (!dict) return;
    
    DictItem *current = dict->head;
    DictItem *prev = NULL;
    
    while (current) {
        if (current->value == NULL || strlen(current->value) == 0) {
            DictItem *to_delete = current;
            
            if (prev) {
                prev->next = current->next;
            } else {
                dict->head = current->next;
            }
            
            current = current->next;
            
            free(to_delete->key);
            free(to_delete->value);
            free(to_delete);
            dict->size--;
        } else {
            prev = current;
            current = current->next;
        }
    }
}

void dict_print(const Dictionary *dict) {
    if (!dict) return;
    
    DictItem *current = dict->head;
    while (current) {
        printf("%s: %s\n", current->key, current->value);
        current = current->next;
    }
}

void dict_destroy(Dictionary *dict) {
    if (!dict) return;
    
    DictItem *current = dict->head;
    while (current) {
        DictItem *next = current->next;
        free(current->key);
        free(current->value);
        free(current);
        current = next;
    }
    
    free(dict);
}

int main(void) {
    Dictionary *dict = dict_create();
    if (!dict) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }
    
    dict_add(dict, "name", "John");
    dict_add(dict, "email", "");
    dict_add(dict, "phone", "555-1234");
    dict_add(dict, "address", "");
    dict_add(dict, "city", "New York");
    
    printf("Before dropping empty items:\n");
    dict_print(dict);
    printf("Size: %zu\n\n", dict->size);
    
    dict_drop_empty(dict);
    
    printf("After dropping empty items:\n");
    dict_print(dict);
    printf("Size: %zu\n", dict->size);
    
    dict_destroy(dict);
    return EXIT_SUCCESS;
}
