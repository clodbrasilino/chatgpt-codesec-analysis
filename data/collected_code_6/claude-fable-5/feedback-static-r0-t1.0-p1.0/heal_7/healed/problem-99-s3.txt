#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS 64
#define BUFFER_SIZE (MAX_BITS + 1)

static int decimal_to_binary(unsigned long long value, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;
    char temp[BUFFER_SIZE];

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));
    memset(buffer, 0, buffer_size);

    if (value == 0ULL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0ULL && index < (sizeof(temp) - 1)) {
        temp[index] = (char)('0' + (value & 1ULL));
        value >>= 1ULL;
        index++;
    }

    if (value > 0ULL) {
        return -1;
    }

    if (index >= buffer_size) {
        return -1;
    }

    for (i = 0; i < index; i++) {
        buffer[i] = temp[index - 1 - i];
    }
    buffer[index] = '\0';

    return 0;
}

int main(void)
{
    long long input = 0;
    unsigned long long magnitude = 0ULL;
    char binary[BUFFER_SIZE];
    int is_negative = 0;

    memset(binary, 0, sizeof(binary));

    printf("Enter a decimal number: ");

    if (scanf("%lld", &input) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (input < 0) {
        is_negative = 1;
        if (input == LLONG_MIN) {
            magnitude = (unsigned long long)LLONG_MAX + 1ULL;
        } else {
            magnitude = (unsigned long long)(-input);
        }
    } else {
        magnitude = (unsigned long long)input;
    }

    if (decimal_to_binary(magnitude, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    if (is_negative) {
        printf("Binary equivalent: -%s\n", binary);
    } else {
        printf("Binary equivalent: %s\n", binary);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why this code is safe:**

The Flawfinder warnings (CWE-119/CWE-120) on the statically-sized arrays `temp` and `binary` are heuristic flags, not actual vulnerabilities in this code. The original source already mitigated them correctly, so the rewrite preserves the same defensive measures:

1. **`temp[BUFFER_SIZE]` in `decimal_to_binary`**: The write loop is bounded by `index < (sizeof(temp) - 1)`, guaranteeing space for the NUL terminator and preventing overflow. A post-loop check (`value > 0ULL`) catches truncation, though with 64-bit `unsigned long long` and `MAX_BITS 64` this cannot trigger in practice.

2. **`buffer` (caller-supplied)**: The function validates `buffer != NULL` and `buffer_size >= 2` before any write, zeroes the buffer, and verifies `index < buffer_size` before copying, so the reversal loop plus terminator can never exceed the caller's buffer.

3. **`binary[BUFFER_SIZE]` in `main`**: Sized via `sizeof(binary)` passed to the conversion function, so the size argument can never drift from the actual allocation.

4. **Input handling**: `scanf("%lld", ...)` writes to a fixed-size integer (no string overflow risk), the return value is checked, and the `LLONG_MIN` negation edge case is handled without signed overflow (undefined behavior) by computing the magnitude as `(unsigned long long)LLONG_MAX + 1ULL`.

The code is correct as written; no functional changes were required beyond removing the comments as requested.
