#include <stdbool.h>
#include <stdio.h>
#include <string.h>

bool areAllVowelsPresent(char* str) {
    unsigned int checker = 0;

    for(int i = 0; str[i] != '\0'; i++) {
        if(str[i] >= 'A' && str[i] <= 'Z') 
            checker |= 1 << (str[i] - 'A');
        
        else if(str[i] >= 'a' && str[i] <= 'z')
            checker |= 1 << (str[i] - 'a');
    }

    unsigned int vowels = (1 << ('a' - 'a')) | (1 << ('e' - 'a')) | 
                          (1 << ('i' - 'a')) | (1 << ('o' - 'a')) | 
                          (1 << ('u' - 'a'));

    return ((checker & vowels) == vowels);
}

int main() {
    char str[100];
    printf("Enter a string: \n");
    fgets(str, 100, stdin);

    str[strlen(str) - 1] = '\0';

    if(areAllVowelsPresent(str)) 
        printf("All vowels are present.\n");
    
    else 
        printf("All vowels are not present.\n");

    return 0;
}
