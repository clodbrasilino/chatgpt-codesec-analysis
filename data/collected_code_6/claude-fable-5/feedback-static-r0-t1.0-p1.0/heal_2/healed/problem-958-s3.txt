```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    buffer[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (used >= buffer_size || sym_len >= buffer_size - used) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            number -= values[i];
        }
    }

    if (used >= buffer_size) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[used] = '\0';
    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            if (printf("%d -> %s\n", test_values[i], roman) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, roman, sizeof(roman)) != 0) {
        if (printf("0 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, roman, sizeof(roman)) != 0) {
        if (printf("4000 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **Flawfinder `memcpy` warning (CWE-120):** This is a false positive in the original code. The copy is already guarded by `used >= buffer_size || sym_len >= buffer_size - used`, which is written carefully to avoid unsigned integer underflow (checking `used >= buffer_size` first before computing `buffer_size - used`). The check guarantees `sym_len < buffer_size - used`, leaving room for both the copied symbol and the eventual NUL terminator. The `memcpy` is therefore safe as-is and is retained.

2. **Flawfinder `char` array warning (CWE-119/CWE-120):** Also a false positive. The statically-sized `roman[ROMAN_BUFFER_SIZE]` buffer is always passed together with `sizeof(roman)`, and every write path in `int_to_roman` is bounds-checked against that size. The buffer is also zero-initialized with `memset` before use. The worst-case Roman numeral (3888 → "MMMDCCCLXXXVIII", 15 chars + NUL) fits comfortably in 32 bytes.

The rewritten code preserves the correct bounds checking, replaces the magic range values with named constants (`ROMAN_MIN`/`ROMAN_MAX`) for maintainability, and keeps all return-value checks on `printf`/`fprintf`. No functional changes were required, since the flagged lines were already properly mitigated — the warnings are pattern-based heuristics from Flawfinder rather than actual exploitable defects.
