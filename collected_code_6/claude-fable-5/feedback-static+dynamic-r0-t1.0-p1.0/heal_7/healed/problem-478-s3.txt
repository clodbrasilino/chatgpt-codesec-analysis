```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (isupper((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char *buffer;
    size_t bufsize;

    bufsize = MAX_INPUT_LEN;
    buffer = malloc(bufsize);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    memset(buffer, 0, bufsize);

    while (fgets(buffer, (int)bufsize, stdin) != NULL) {
        buffer[bufsize - 1] = '\0';

        size_t len = strnlen(buffer, bufsize);

        while (len > 0 && (buffer[len - 1] == '\n' || buffer[len - 1] == '\r')) {
            buffer[len - 1] = '\0';
            len--;
        }

        char *output = remove_lowercase(buffer);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free(buffer);
            return EXIT_FAILURE;
        }

        printf("%s\n", output);
        fflush(stdout);
        free(output);

        memset(buffer, 0, bufsize);
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```
