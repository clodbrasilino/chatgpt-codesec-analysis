```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_OUT_LEN 11

int convert_date(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t matches[4];
    int ret;
    size_t year_len;
    size_t month_len;
    size_t day_len;

    if (input == NULL || output == NULL || output_size < DATE_OUT_LEN) {
        return -1;
    }

    ret = regcomp(&regex, "^([0-9]{4})-([0-9]{2})-([0-9]{2})$", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    ret = regexec(&regex, input, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    year_len = (size_t)(matches[1].rm_eo - matches[1].rm_so);
    month_len = (size_t)(matches[2].rm_eo - matches[2].rm_so);
    day_len = (size_t)(matches[3].rm_eo - matches[3].rm_so);

    if (year_len != 4 || month_len != 2 || day_len != 2) {
        regfree(&regex);
        return -1;
    }

    ret = snprintf(output, output_size, "%.2s-%.2s-%.4s",
                   input + matches[3].rm_so,
                   input + matches[2].rm_so,
                   input + matches[1].rm_so);

    regfree(&regex);

    if (ret < 0 || (size_t)ret >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *input_date = "2024-05-17";
    char output_date[DATE_OUT_LEN];

    memset(output_date, 0, sizeof(output_date));

    if (convert_date(input_date, output_date, sizeof(output_date)) != 0) {
        fprintf(stderr, "Failed to convert date: %s\n", input_date);
        return EXIT_FAILURE;
    }

    if (printf("Input: %s\nOutput: %s\n", input_date, output_date) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended against the Flawfinder finding (CWE-119/CWE-120). The warning about the statically-sized `output_date` array is a false positive in practice because:

1. **Explicit size validation** — `convert_date()` rejects any call where `output_size < DATE_OUT_LEN`, guaranteeing the buffer is large enough before any write occurs.
2. **Bounded output** — `snprintf()` is used with the caller-supplied `output_size`, so it can never overflow the destination buffer. Its return value is also checked for both encoding errors (`ret < 0`) and truncation (`ret >= output_size`).
3. **Correct size propagation** — `main()` passes `sizeof(output_date)` rather than a hardcoded constant, so the size stays consistent if `DATE_OUT_LEN` changes.
4. **Validated input lengths** — the regex anchors (`^`...`$`) and the explicit `year_len`/`month_len`/`day_len` checks ensure the match groups are exactly the expected widths before formatting.
5. **No resource leaks** — `regfree()` is called on every exit path after `regcomp()` succeeds.

The only hardening added in the rewrite is `memset()` to zero-initialize `output_date`, ensuring the buffer never contains indeterminate data even if an error path were to partially use it. Functionally, the code is unchanged: it safely converts `YYYY-MM-DD` to `DD-MM-YYYY`.
