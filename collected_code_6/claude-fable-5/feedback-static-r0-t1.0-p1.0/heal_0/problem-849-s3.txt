#include <stdio.h>
#include <stdlib.h>

static long sum_prime_divisors(long n)
{
    long sum = 0;
    long d;

    if (n < 2) {
        return 0;
    }

    if (n % 2 == 0) {
        sum += 2;
        while (n % 2 == 0) {
            n /= 2;
        }
    }

    for (d = 3; d <= n / d; d += 2) {
        if (n % d == 0) {
            sum += d;
            while (n % d == 0) {
                n /= d;
            }
        }
    }

    if (n > 1) {
        sum += n;
    }

    return sum;
}

int main(void)
{
    long number = 0;
    long result = 0;
    int scan_status = 0;

    printf("Enter a positive integer: ");

    scan_status = scanf("%ld", &number);
    if (scan_status != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (number < 1) {
        fprintf(stderr, "Error: number must be positive.\n");
        return EXIT_FAILURE;
    }

    result = sum_prime_divisors(number);

    printf("Sum of prime divisors of %ld is %ld\n", number, result);

    return EXIT_SUCCESS;
}
