```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int value;
    size_t count;
} Frequency;

static int find_index(const Frequency *freq, size_t freq_len, int value)
{
    size_t i;

    for (i = 0; i < freq_len; i++) {
        if (freq[i].value == value) {
            return (int)i;
        }
    }
    return -1;
}

int element_frequencies(const int *const *lists, const size_t *list_sizes,
                        size_t num_lists, Frequency **result, size_t *result_len)
{
    size_t total = 0;
    size_t i;
    size_t j;
    size_t freq_len = 0;
    Frequency *freq = NULL;

    if (lists == NULL || list_sizes == NULL || result == NULL ||
        result_len == NULL) {
        return -1;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] == NULL && list_sizes[i] > 0) {
            return -1;
        }
        if (list_sizes[i] > SIZE_MAX - total) {
            return -1;
        }
        total += list_sizes[i];
    }

    if (total == 0) {
        *result = NULL;
        *result_len = 0;
        return 0;
    }

    freq = malloc(total * sizeof(*freq));
    if (freq == NULL) {
        return -1;
    }

    for (i = 0; i < num_lists; i++) {
        for (j = 0; j < list_sizes[i]; j++) {
            int idx = find_index(freq, freq_len, lists[i][j]);
            if (idx >= 0) {
                freq[idx].count++;
            } else {
                freq[freq_len].value = lists[i][j];
                freq[freq_len].count = 1;
                freq_len++;
            }
        }
    }

    *result = freq;
    *result_len = freq_len;
    return 0;
}

int main(void)
{
    int list1[] = {1, 2, 3, 2};
    int list2[] = {4, 5, 6, 2, 7};
    int list3[] = {1, 3, 5, 6, 1};
    const int *lists[] = {list1, list2, list3};
    size_t sizes[] = {sizeof(list1) / sizeof(list1[0]),
                      sizeof(list2) / sizeof(list2[0]),
                      sizeof(list3) / sizeof(list3[0])};
    size_t num_lists = sizeof(lists) / sizeof(lists[0]);
    Frequency *result = NULL;
    size_t result_len = 0;
    size_t i;

    if (element_frequencies(lists, sizes, num_lists, &result, &result_len) != 0) {
        fprintf(stderr, "Failed to compute frequencies\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_len; i++) {
        if (printf("%d: %zu\n", result[i].value, result[i].count) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
    }

    free(result);
    return EXIT_SUCCESS;
}
```
