#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char **split_string_regex(const char *str, const char *pattern, size_t *count) {
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t capacity = 10;
    size_t len;
    char **result = NULL;
    char *dup = NULL;
    char *current;
    const char *original;

    if (str == NULL || pattern == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result = (char **)malloc(capacity * sizeof(char *));
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    len = strlen(str);
    dup = (char *)malloc(len + 1);
    if (dup == NULL) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    memcpy(dup, str, len + 1);
    current = dup;
    original = str;

    while (1) {
        ret = regexec(&regex, current, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            if (*current != '\0') {
                if (*count >= capacity) {
                    capacity *= 2;
                    char **temp = (char **)realloc(result, capacity * sizeof(char *));
                    if (temp == NULL) {
                        for (size_t i = 0; i < *count; i++) free(result[i]);
                        free(result);
                        free(dup);
                        regfree(&regex);
                        return NULL;
                    }
                    result = temp;
                }
                size_t token_len = strlen(current);
                result[*count] = (char *)malloc(token_len + 1);
                if (result[*count] == NULL) {
                    for (size_t i = 0; i < *count; i++) free(result[i]);
                    free(result);
                    free(dup);
                    regfree(&regex);
                    return NULL;
                }
                memcpy(result[*count], current, token_len + 1);
                (*count)++;
            }
            break;
        } else if (ret != 0) {
            for (size_t i = 0; i < *count; i++) free(result[i]);
            free(result);
            free(dup);
            regfree(&regex);
            return NULL;
        }

        if (match.rm_so > 0) {
            if (*count >= capacity) {
                capacity *= 2;
                char **temp = (char **)realloc(result, capacity * sizeof(char *));
                if (temp == NULL) {
                    for (size_t i = 0; i < *count; i++) free(result[i]);
                    free(result);
                    free(dup);
                    regfree(&regex);
                    return NULL;
                }
                result = temp;
            }
            size_t token_len = match.rm_so;
            result[*count] = (char *)malloc(token_len + 1);
            if (result[*count] == NULL) {
                for (size_t i = 0; i < *count; i++) free(result[i]);
                free(result);
                free(dup);
                regfree(&regex);
                return NULL;
            }
            memcpy(result[*count], current, token_len);
            result[*count][token_len] = '\0';
            (*count)++;
        }

        current += match.rm_eo;
    }

    free(dup);
    regfree(&regex);

    char **final_result = (char **)realloc(result, (*count + 1) * sizeof(char *));
    if (final_result != NULL) {
        final_result[*count] = NULL;
        return final_result;
    }
    result[*count] = NULL;
    return result;
}

void free_split_result(char **result) {
    if (result == NULL) return;
    for (size_t i = 0; result[i] != NULL; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void) {
    const char *str = "Hello,world;this.is:a/test";
    const char *pattern = "[,;:. /]+";
    size_t count = 0;

    char **tokens = split_string_regex(str, pattern, &count);
    if (tokens == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < count; i++) {
        printf("%s\n", tokens[i]);
    }

    free_split_result(tokens);

    return EXIT_SUCCESS;
}
