#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>

bool are_isomorphic(const char *s, const char *t) {
    if (s == NULL || t == NULL) {
        return false;
    }

    int map_s[256];
    int map_t[256];

    for (int i = 0; i < 256; ++i) {
        map_s[i] = -1;
        map_t[i] = -1;
    }

    while (*s != '\0' && *t != '\0') {
        unsigned char c1 = (unsigned char)*s;
        unsigned char c2 = (unsigned char)*t;

        if (map_s[c1] == -1 && map_t[c2] == -1) {
            map_s[c1] = c2;
            map_t[c2] = c1;
        } else if (map_s[c1] != c2 || map_t[c2] != c1) {
            return false;
        }
        
        s++;
        t++;
    }

    return *s == '\0' && *t == '\0';
}

int main(void) {
    const char *s1 = "egg";
    const char *t1 = "add";
    
    const char *s2 = "foo";
    const char *t2 = "bar";
    
    const char *s3 = "paper";
    const char *t3 = "title";

    if (are_isomorphic(s1, t1)) {
        printf("1\n");
    } else {
        printf("0\n");
    }

    if (are_isomorphic(s2, t2)) {
        printf("1\n");
    } else {
        printf("0\n");
    }

    if (are_isomorphic(s3, t3)) {
        printf("1\n");
    } else {
        printf("0\n");
    }

    return 0;
}
