#include <stdlib.h>
#include <string.h>

int split_and_rotate(int *arr, size_t n, size_t k);

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t temp_size;
    size_t move_size;
    size_t total_size;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*arr)) {
        return -1;
    }

    total_size = n * sizeof(*arr);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > SIZE_MAX / sizeof(*temp)) {
        return -1;
    }

    temp_size = k * sizeof(*temp);
    move_size = (n - k) * sizeof(*arr);

    if (temp_size > total_size || move_size > total_size) {
        return -1;
    }

    if (temp_size + move_size != total_size) {
        return -1;
    }

    temp = malloc(temp_size);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, temp_size);
    memmove(arr, arr + k, move_size);
    memcpy(arr + (n - k), temp, temp_size);

    free(temp);
    return 0;
}

void print_array(const int *arr, size_t n)
{
    size_t i;

    if (arr == NULL) {
        return;
    }

    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;

    printf("Original array: ");
    print_array(arr, n);

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    printf("Array after splitting at %zu and adding first part to end: ", k);
    print_array(arr, n);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The two Flawfinder `memcpy` warnings (CWE-120) are false positives in this code, and the existing guards already constitute the fix. Here's why the rewritten code is safe as-is:

1. **First `memcpy(temp, arr, temp_size)`:** The destination `temp` is allocated with exactly `temp_size` bytes, and the `malloc` return value is checked for `NULL`. The source `arr` holds `total_size` bytes, and `temp_size <= total_size` is explicitly verified, so the read never exceeds the source buffer.

2. **`memmove(arr, arr + k, move_size)`:** Since `k = k % n` guarantees `k < n`, the source region `arr + k` spans exactly `(n - k)` elements, which is `move_size` bytes — entirely within bounds. `memmove` is correctly used here because source and destination overlap.

3. **Second `memcpy(arr + (n - k), temp, temp_size)`:** The destination region starts at element `n - k` and receives `k` elements, ending exactly at element `n`. The arithmetic identity is defensively verified by the `temp_size + move_size != total_size` check.

4. **Integer overflow protection:** Both `n * sizeof(*arr)` and `k * sizeof(*temp)` are guarded against `size_t` overflow via the `SIZE_MAX` division checks before any multiplication occurs.

5. **Resource handling:** `temp` is freed on the success path, and no allocation occurs before the early-return validation paths, so there are no leaks.
