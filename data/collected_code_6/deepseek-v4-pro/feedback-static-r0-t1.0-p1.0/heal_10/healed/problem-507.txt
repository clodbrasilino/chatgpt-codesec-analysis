#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct node {
    char *word;
    struct node *next;
} node_t;

static node_t *create_node(const char *word) {
    node_t *new_node;
    size_t len;

    if (word == NULL) {
        return NULL;
    }

    len = strnlen(word, SIZE_MAX);
    
    new_node = (node_t *)malloc(sizeof(node_t));
    if (new_node == NULL) {
        return NULL;
    }

    new_node->word = (char *)malloc(len + 1);
    if (new_node->word == NULL) {
        free(new_node);
        return NULL;
    }

    if (len > 0) {
        memcpy(new_node->word, word, len);
    }
    new_node->word[len] = '\0';
    new_node->next = NULL;
    return new_node;
}

static void free_list(node_t *head) {
    node_t *current = head;
    node_t *next;
    while (current != NULL) {
        next = current->next;
        free(current->word);
        free(current);
        current = next;
    }
}

static node_t *remove_words(node_t *head, const char *target) {
    node_t *dummy;
    node_t *prev;
    node_t *current;

    if (head == NULL || target == NULL) {
        return head;
    }

    dummy = create_node("");
    if (dummy == NULL) {
        return head;
    }

    dummy->next = head;
    prev = dummy;
    current = head;

    while (current != NULL) {
        if (current->word != NULL && strcmp(current->word, target) == 0) {
            prev->next = current->next;
            free(current->word);
            free(current);
            current = prev->next;
        } else {
            prev = current;
            current = current->next;
        }
    }

    head = dummy->next;
    free(dummy->word);
    free(dummy);
    return head;
}

static void print_list(node_t *head) {
    node_t *current = head;
    while (current != NULL) {
        if (current->word != NULL) {
            printf("%s ", current->word);
        }
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    node_t *head = NULL;
    node_t *tail = NULL;
    const char *words[] = {"apple", "banana", "apple", "cherry", "apple"};
    int count = sizeof(words) / sizeof(words[0]);
    int i;

    for (i = 0; i < count; i++) {
        node_t *new_node = create_node(words[i]);
        if (new_node == NULL) {
            free_list(head);
            return 1;
        }
        if (head == NULL) {
            head = new_node;
            tail = new_node;
        } else {
            tail->next = new_node;
            tail = new_node;
        }
    }

    printf("Original list: ");
    print_list(head);

    head = remove_words(head, "apple");

    printf("After removing 'apple': ");
    print_list(head);

    free_list(head);
    return 0;
}
