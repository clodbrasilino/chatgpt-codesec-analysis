#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>

#define CHAR_SET_SIZE 256

static char find_most_frequent_char(const char *str, size_t len)
{
    size_t freq[CHAR_SET_SIZE] = {0};
    size_t max_count = 0;
    size_t i;
    char most_frequent;

    if (len == 0) {
        return '\0';
    }

    most_frequent = str[0];

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        freq[c]++;
        if (freq[c] > max_count) {
            max_count = freq[c];
            most_frequent = (char)c;
        }
    }

    return most_frequent;
}

char *minimize_string(const char *str, size_t max_len)
{
    char *result;
    char target;
    size_t len;
    size_t i;
    size_t j = 0;

    if (str == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(str, max_len);

    result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    if (len == 0) {
        result[0] = '\0';
        return result;
    }

    target = find_most_frequent_char(str, len);

    for (i = 0; i < len; i++) {
        if (str[i] != target) {
            result[j] = str[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_capacity = 0;
    ssize_t bytes_read;
    char *minimized;
    size_t input_len;

    printf("Enter a string: ");
    fflush(stdout);

    bytes_read = getline(&buffer, &buffer_capacity, stdin);
    if (bytes_read < 0) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (bytes_read > 0 && buffer[bytes_read - 1] == '\n') {
        buffer[bytes_read - 1] = '\0';
        bytes_read--;
    }

    input_len = (size_t)bytes_read;

    if (input_len == 0) {
        fprintf(stderr, "Empty string provided\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    minimized = minimize_string(buffer, input_len);
    if (minimized == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Minimized string: %s\n", minimized);
    printf("Original length: %zu\n", input_len);
    printf("Minimized length: %zu\n", strnlen(minimized, input_len + 1));

    free(minimized);
    minimized = NULL;

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
