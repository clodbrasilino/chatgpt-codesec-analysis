#define MAX_LEN 256

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t i = 0;

    if (str == NULL) {
        return 0;
    }

    while (i < max_len && str[i] != '\0') {
        i++;
    }

    return i;
}

int contains_all_vowels(const char *str, size_t max_len)
{

    size_t len;

        return 0;

    for (i = 0; i < len; i++) {

            found[0] = 1;
            break;

            found[1] = 1;
            break;
        case 'i':
            found[2] = 1;
            break;
        case 'o':
            found[3] = 1;
            break;
        case 'u':
            found[4] = 1;
            break;
        default:

        }
    }

}

        printf("Accepted: the string contains all vowels\n");
