```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

typedef long long (*binary_op)(long long, long long);

static long long power(long long base, long long exp)
{
    long long result = 1;

    if (exp < 0) {
        return 0;
    }

    while (exp > 0) {
        if (exp % 2 == 1) {
            result *= base;
        }
        exp /= 2;
        if (exp > 0) {
            base *= base;
        }
    }
    return result;
}

static long long *map_binary(binary_op op, const long long *a, const long long *b, size_t n)
{
    long long *result;
    size_t i;

    if (op == NULL || a == NULL || b == NULL || n == 0) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(*result)) {
        return NULL;
    }

    result = malloc(n * sizeof(*result));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        result[i] = op(a[i], b[i]);
    }
    return result;
}

int main(void)
{
    const long long bases[] = {10, 20, 30};
    const long long indices[] = {1, 2, 3};
    size_t n = sizeof(bases) / sizeof(bases[0]);
    long long *powers;
    size_t i;

    powers = map_binary(power, bases, indices, n);
    if (powers == NULL) {
        fprintf(stderr, "Failed to compute powers\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (i = 0; i < n; i++) {
        printf("%lld", powers[i]);
        if (i < n - 1) {
            printf(", ");
        }
    }
    printf("]\n");

    free(powers);
    powers = NULL;

    return EXIT_SUCCESS;
}
```

Changes made:

1. **Fixed the `constVariable` weakness**: Declared `indices` as `const long long[]`. I also applied the same treatment to `bases` since it is never modified either, and both are passed to `map_binary` which accepts `const long long *` parameters — so this is safe and improves const-correctness.

2. **Added missing include**: The original code uses `SIZE_MAX` in `map_binary` but never includes `<stdint.h>`, which is where `SIZE_MAX` is defined. Without it, the code would fail to compile on a conforming implementation (or silently pick it up via transitive includes, which is non-portable). This was a latent defect beyond the flagged style issue.

The rest of the code is already sound: the overflow guard before `malloc`, the NULL checks, the `free` followed by nulling the pointer, and the exponentiation-by-squaring implementation (with the `exp > 0` guard preventing an unnecessary final `base *= base` overflow) are all correct as written.
