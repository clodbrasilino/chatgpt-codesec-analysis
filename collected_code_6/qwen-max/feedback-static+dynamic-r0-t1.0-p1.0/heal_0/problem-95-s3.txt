#include <stdio.h>
#include <limits.h>

int findMinSublistLength(int arr[], int n) {
    if (n <= 0) return 0;

    int min_length = INT_MAX;
    int start = 0, sum = 0, min_sum = 0;

    for (int i = 0; i < n; ++i) {
        sum += arr[i];
        if (sum - min_sum > 0) {
            min_length = (i - start + 1 < min_length) ? i - start + 1 : min_length;
        } else {
            min_sum = sum;
            start = i + 1;
        }
    }

    return min_length == INT_MAX ? 0 : min_length;
}

int main() {
    int arr[] = {1, -1, 2, -2, 3, -3};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = findMinSublistLength(arr, n);
    printf("Minimum length of sublist: %d\n", result);
    return 0;
}
