```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    char *first;
    char *last;
    size_t tail_len;

    if (str == NULL) {
        return;
    }

    len = strnlen(str, max_len);
    if (len == 0U || len >= max_len) {
        if (len >= max_len) {
            str[max_len - 1U] = '\0';
            len = max_len - 1U;
        }
        if (len == 0U) {
            return;
        }
    }

    first = memchr(str, (int)ch, len);
    if (first == NULL) {
        return;
    }

    last = NULL;
    {
        size_t i;
        for (i = len; i > 0U; i--) {
            if (str[i - 1U] == ch) {
                last = &str[i - 1U];
                break;
            }
        }
    }

    if (last != NULL && last != first) {
        tail_len = len - (size_t)(last - str) - 1U;
        memmove(last, last + 1, tail_len + 1U);
        len -= 1U;
    }

    tail_len = len - (size_t)(first - str) - 1U;
    memmove(first, first + 1, tail_len + 1U);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

The key hardening changes made to address the flagged weaknesses:

1. **`strlen` over-read (CWE-126)**: Replaced all `strlen` calls with `strnlen` bounded by an explicit `max_len` parameter passed to `remove_first_last`, and force-terminate the string if no `\0` is found within bounds. Subsequent lengths for `memmove` are computed arithmetically from the known bounded length rather than re-scanning with `strlen`.

2. **Unbounded `strchr`/`strrchr`**: Replaced `strchr` with `memchr` (explicitly length-bounded) and `strrchr` with a bounded reverse loop, so no scan can run past the validated length.

3. **Statically-sized array (CWE-119/CWE-120)**: The buffer size is now a single named constant, `fgets` remains bounded by `sizeof(buffer)`, and the same bound is propagated into `remove_first_last` so the function can never operate beyond the caller's buffer.

4. **`getchar` (CWE-120/CWE-20)**: It is used once (not in a loop) with an explicit `EOF` check, which is already the correct pattern; that remains unchanged.
