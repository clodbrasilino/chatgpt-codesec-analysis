#include <stdio.h>
#include <math.h>
#include <stdint.h>
#include <limits.h>

int64_t sumEvenFactors(int64_t n) {
    if (n <= 0) {
        return 0;
    }

    int64_t sum = 0;
    int64_t limit = (int64_t)sqrt((double)n);

    for (int64_t i = 1; i <= limit; i++) {
        if (n % i == 0) {
            if (i % 2 == 0) {
                if (sum > INT64_MAX - i) {
                    return -1;
                }
                sum += i;
            }
            int64_t counterpart = n / i;
            if (counterpart != i && counterpart % 2 == 0) {
                if (sum > INT64_MAX - counterpart) {
                    return -1;
                }
                sum += counterpart;
            }
        }
    }

    return sum;
}

int main(void) {
    int64_t number;

    printf("Enter a number: ");
    if (scanf("%ld", &number) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    int64_t result = sumEvenFactors(number);
    if (result == -1) {
        printf("Overflow occurred during calculation.\n");
        return 1;
    }
    printf("Sum of even factors: %ld\n", result);

    return 0;
}
