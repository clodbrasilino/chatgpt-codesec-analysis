```c
#include <stdio.h>
#include <stddef.h>
#include <string.h>
#include <stdbool.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096U
#define MAX_WORDS 7U

static bool is_palindrome(const char *str)
{
    size_t left;
    size_t right;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_STRING_LENGTH);
    if ((len == 0U) || (len >= MAX_STRING_LENGTH)) {
        return false;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        if (tolower((unsigned char)str[left]) != tolower((unsigned char)str[right])) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

static size_t find_palindromes(const char *const *strings, size_t count,
                               const char **results, size_t results_capacity)
{
    size_t found;

    if ((strings == NULL) || (results == NULL) || (results_capacity == 0U)) {
        return 0U;
    }

    if (count > results_capacity) {
        count = results_capacity;
    }

    found = 0U;
    for (size_t i = 0U; i < count; i++) {
        if (is_palindrome(strings[i])) {
            if (found < results_capacity) {
                results[found] = strings[i];
                found++;
            } else {
                break;
            }
        }
    }

    return found;
}

int main(void)
{
    const char *const words[MAX_WORDS] = {
        "racecar",
        "hello",
        "Level",
        "world",
        "noon",
        "example",
        "madam"
    };
    const size_t word_count = sizeof(words) / sizeof(words[0]);
    const char *palindromes[MAX_WORDS] = { NULL };
    const size_t results_capacity = sizeof(palindromes) / sizeof(palindromes[0]);
    size_t found;

    found = find_palindromes(words, word_count, palindromes, results_capacity);

    if (found > results_capacity) {
        return 1;
    }

    if (found == 0U) {
        if (puts("No palindromes found.") == EOF) {
            return 1;
        }
    } else {
        if (printf("Found %zu palindrome(s):\n", found) < 0) {
            return 1;
        }
        for (size_t i = 0U; i < found; i++) {
            if (palindromes[i] == NULL) {
                return 1;
            }
            if (printf("%s\n", palindromes[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

1. **`words[MAX_WORDS]`** — Fully initialized with string literals at compile time, never written to at runtime (`const char *const`), and its element count is derived via `sizeof(words) / sizeof(words[0])` rather than a hardcoded value, so no out-of-bounds access is possible.

2. **`palindromes[MAX_WORDS]`** — All writes are bounds-checked: `find_palindromes()` clamps `count` to `results_capacity`, guards each write with `found < results_capacity`, and rejects `NULL` pointers and zero capacity up front. The caller also validates `found > results_capacity` defensively and checks each pointer for `NULL` before dereferencing.

3. **String handling** — `strnlen()` with `MAX_STRING_LENGTH` bounds all string scans, rejecting unterminated or oversized inputs, and `tolower()` receives an `unsigned char` cast to avoid undefined behavior with negative `char` values.

4. **I/O robustness** — All `printf`/`puts` return values are checked, so stream failures propagate as a nonzero exit status.
