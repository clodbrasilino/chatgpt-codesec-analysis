```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *remove_lowercase_substrings(const char *str, size_t len) {
    if (str == NULL) {
        return NULL;
    }

    if (len + 1 <= len) {
        return NULL;
    }

    char *safe_str = malloc(len + 1);
    if (safe_str == NULL) {
        return NULL;
    }
    
    snprintf(safe_str, len + 1, "%s", str);

    regex_t regex;
    int ret = regcomp(&regex, "[[:lower:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(safe_str);
        return NULL;
    }

    char *result = malloc(len + 1);
    if (result == NULL) {
        regfree(&regex);
        free(safe_str);
        return NULL;
    }

    const char *p = safe_str;
    size_t out_idx = 0;
    regmatch_t match;

    while (1) {
        ret = regexec(&regex, p, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            while (*p) {
                result[out_idx++] = *p++;
            }
            break;
        }

        if (ret != 0) {
            break;
        }

        for (int i = 0; i < match.rm_so; i++) {
            result[out_idx++] = p[i];
        }

        p += match.rm_eo;
    }

    result[out_idx] = '\0';
    regfree(&regex);
    free(safe_str);

    char *shrunk = realloc(result, out_idx + 1);
    return shrunk != NULL ? shrunk : result;
}

int main(void) {
    const char *input = "HelloWorldABCdefGHI";
    size_t input_len = sizeof("HelloWorldABCdefGHI") - 1;
    char *output = remove_lowercase_substrings(input, input_len);
    
    if (output != NULL) {
        printf("Original: %s\nResult: %s\n", input, output);
        free(output);
    } else {
        fprintf(stderr, "Memory allocation or regex compilation failed\n");
        return 1;
    }
    
    return 0;
}
```
