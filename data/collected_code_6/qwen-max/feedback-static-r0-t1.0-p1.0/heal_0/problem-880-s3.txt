#include <stdio.h>
#include <math.h>

int find_number_of_solutions(double a, double b, double c) {
    double discriminant = b * b - 4 * a * c;
    if (discriminant > 0) return 2;
    if (discriminant == 0) return 1;
    return 0;
}

int main() {
    double a, b, c;
    printf("Enter coefficients a, b, c: ");
    int ret = scanf("%lf %lf %lf", &a, &b, &c);
    if (ret != 3) {
        fprintf(stderr, "Input error.\n");
        return 1;
    }
    if (a == 0) {
        fprintf(stderr, "Coefficient 'a' must not be zero for a quadratic equation.\n");
        return 1;
    }
    int solutions = find_number_of_solutions(a, b, c);
    printf("The number of solutions is: %d\n", solutions);
    return 0;
}
