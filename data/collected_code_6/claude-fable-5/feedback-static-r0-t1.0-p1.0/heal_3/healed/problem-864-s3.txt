#include <stdio.h>
#include <stddef.h>
#include <stdbool.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096U

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t len = 0U;

    while ((len < max_len) && (str[len] != '\0')) {
        len++;
    }

    return len;
}

static bool is_palindrome(const char *str)
{
    size_t left;
    size_t right;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = bounded_length(str, MAX_STRING_LENGTH);
    if ((len == 0U) || (len >= MAX_STRING_LENGTH)) {
        return false;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        if (tolower((unsigned char)str[left]) != tolower((unsigned char)str[right])) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

static size_t find_palindromes(const char *const *strings, size_t count,
                               const char **results, size_t results_capacity)
{
    size_t found;

    if ((strings == NULL) || (results == NULL) || (results_capacity == 0U)) {
        return 0U;
    }

    found = 0U;
    for (size_t i = 0U; i < count; i++) {
        if (is_palindrome(strings[i])) {
            if (found < results_capacity) {
                results[found] = strings[i];
                found++;
            } else {
                break;
            }
        }
    }

    return found;
}

int main(void)
{
    static const char *const words[] = {
        "racecar",
        "hello",
        "Level",
        "world",
        "noon",
        "example",
        "madam"
    };
    const size_t word_count = sizeof(words) / sizeof(words[0]);
    const char *palindromes[sizeof(words) / sizeof(words[0])] = { NULL };
    const size_t palindromes_capacity = sizeof(palindromes) / sizeof(palindromes[0]);
    size_t found;

    found = find_palindromes(words, word_count, palindromes, palindromes_capacity);

    if (found > palindromes_capacity) {
        return 1;
    }

    if (found == 0U) {
        if (puts("No palindromes found.") == EOF) {
            return 1;
        }
    } else {
        if (printf("Found %zu palindrome(s):\n", found) < 0) {
            return 1;
        }
        for (size_t i = 0U; i < found; i++) {
            if (palindromes[i] == NULL) {
                return 1;
            }
            if (printf("%s\n", palindromes[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning on the statically-sized `palindromes` array is a false positive in this code, and no structural changes were required. Here is why the code is already safe:

1. **Capacity is derived, not hard-coded:** `palindromes` is sized with `sizeof(words) / sizeof(words[0])`, so it can never be smaller than the maximum number of possible matches. If `words` grows, the results buffer grows automatically.

2. **Bounds are enforced at the point of write:** `find_palindromes()` receives the capacity explicitly (`results_capacity`) and checks `found < results_capacity` before every write, breaking out of the loop when full. It also rejects `NULL` pointers and zero capacity up front.

3. **String reads are length-bounded:** `bounded_length()` never scans past `MAX_STRING_LENGTH`, protecting against unterminated strings, and `is_palindrome()` rejects strings that hit the cap rather than operating on a truncated view.

4. **Defense-in-depth checks in `main()`:** The `found > palindromes_capacity` guard and the per-element `NULL` check before printing catch any contract violation before it can cause an out-of-bounds read.

5. **`tolower()` is called with an `unsigned char` cast,** avoiding undefined behavior for negative `char` values — a common subtle bug that scanners often miss.

The code as written already performs the bounds checking that the Flawfinder hint recommends, so the rewrite preserves the original logic with the advisory comment removed.
