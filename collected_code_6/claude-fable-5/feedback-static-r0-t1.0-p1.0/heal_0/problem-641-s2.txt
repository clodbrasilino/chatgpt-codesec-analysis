```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long nonagonal_number(long n)
{
    if (n < 1) {
        return -1;
    }
    if (n > (LONG_MAX - 5) / 7) {
        return -1;
    }
    long term = 7 * n - 5;
    if (n > LONG_MAX / term) {
        return -1;
    }
    return (n * term) / 2;
}

int main(void)
{
    long n = 0;
    long result = 0;

    printf("Enter n: ");
    if (scanf("%ld", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = nonagonal_number(n);
    if (result < 0) {
        fprintf(stderr, "Error: n must be positive and within range\n");
        return EXIT_FAILURE;
    }

    printf("The %ld-th nonagonal number is %ld\n", n, result);
    return EXIT_SUCCESS;
}
```
