#include <stdio.h>
#include <stdlib.h>

int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;
    size_t mid;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        if (arr[low] <= arr[high]) {
            return (int)low;
        }

        mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else {
            high = mid;
        }
    }

    return (int)low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    int arr3[] = {1, 2, 3, 4, 5};
    int result;

    result = count_rotations(arr1, sizeof(arr1) / sizeof(arr1[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr1\n");
        return EXIT_FAILURE;
    }
    if (result != 2) {
        fprintf(stderr, "Test case 1 failed: expected 2, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr1: %d\n", result);

    result = count_rotations(arr2, sizeof(arr2) / sizeof(arr2[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr2\n");
        return EXIT_FAILURE;
    }
    if (result != 4) {
        fprintf(stderr, "Test case 2 failed: expected 4, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr2: %d\n", result);

    result = count_rotations(arr3, sizeof(arr3) / sizeof(arr3[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr3\n");
        return EXIT_FAILURE;
    }
    if (result != 0) {
        fprintf(stderr, "Test case 3 failed: expected 0, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr3: %d\n", result);

    return EXIT_SUCCESS;
}

Analysis notes: the reported weakness ("test case 1 failed: expected 2, got 3") typically arises when the binary search advances `low = mid` on equality or uses `arr[mid] >= arr[high]`, causing an off-by-one past the pivot. The corrected logic above guarantees convergence to the minimum element's index: it compares `arr[mid]` strictly against `arr[high]`, moving `low = mid + 1` only when the minimum is provably in the right half, and shrinking `high = mid` otherwise (keeping `mid` as a candidate). The early-exit check `arr[low] <= arr[high]` handles already-sorted subranges (test case 3 returns 0 immediately), the `NULL`/zero-length guard prevents undefined behavior, and `size_t` arithmetic with `low + (high - low) / 2` avoids both underflow and midpoint overflow. Verified traces: arr1 → 2, arr2 → 4, arr3 → 0, so all three test cases pass.
