#include <stdio.h>
#include <regex.h>
#include <string.h>

int starts_with_vowel(const char *str) {
    regex_t regex;
    int ret;
    
    if (str == NULL) {
        return 0;
    }
    
    ret = regcomp(&regex, "^[aeiouAEIOU]", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }
    
    ret = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);
    
    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    }
    
    return -1;
}

int main(void) {
    const char *test_strings[] = {
        "apple",
        "banana",
        "Elephant",
        "car",
        "orange",
        "Umbrella",
        "",
        NULL
    };
    
    for (int i = 0; test_strings[i] != NULL; i++) {
        int result = starts_with_vowel(test_strings[i]);
        printf("'%s': ", test_strings[i]);
        if (result == 1) {
            printf("starts with vowel\n");
        } else if (result == 0) {
            printf("does not start with vowel\n");
        } else {
            printf("error\n");
        }
    }
    
    return 0;
}
