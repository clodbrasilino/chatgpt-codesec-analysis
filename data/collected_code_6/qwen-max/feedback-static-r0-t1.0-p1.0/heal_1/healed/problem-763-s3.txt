#include <stdio.h>
#include <limits.h>

int getMinDifference(const int *arr, int n) {
    int min_diff = INT_MAX;
    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {
            int diff = arr[i] > arr[j] ? arr[i] - arr[j] : arr[j] - arr[i];
            if (diff < min_diff) {
                min_diff = diff;
            }
        }
    }
    return min_diff;
}

int main() {
    int arr[] = {5, 12, 3, 18, 7};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = getMinDifference(arr, n);
    printf("The minimum difference is %d\n", result);
    return 0;
}
