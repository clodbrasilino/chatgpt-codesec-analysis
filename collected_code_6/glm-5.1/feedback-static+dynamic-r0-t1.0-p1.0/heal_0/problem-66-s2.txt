```c
#include <stdio.h>
#include <stdlib.h>

int count_positive(const int *list, size_t size) {
    int count = 0;
    for (size_t i = 0; i < size; i++) {
        if (list[i] > 0) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int list[] = {1, -2, 3, 0, -5, 8};
    size_t size = sizeof(list) / sizeof(list[0]);
    
    int result = count_positive(list, size);
    
    printf("%d\n", result);
    
    return 0;
}
```
