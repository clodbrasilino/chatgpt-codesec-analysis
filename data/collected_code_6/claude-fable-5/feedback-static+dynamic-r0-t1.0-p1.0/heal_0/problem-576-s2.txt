#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

bool is_subarray(const int *arr, size_t arr_len, const int *sub, size_t sub_len)
{
    if (sub_len == 0U) {
        return true;
    }
    if (arr == NULL || sub == NULL || sub_len > arr_len) {
        return false;
    }
    for (size_t i = 0U; i <= arr_len - sub_len; i++) {
        size_t j = 0U;
        while (j < sub_len && arr[i + j] == sub[j]) {
            j++;
        }
        if (j == sub_len) {
            return true;
        }
    }
    return false;
}

int main(void)
{
    int arr[] = {2, 3, 0, 5, 1, 1, 2};
    int sub1[] = {0, 5, 1};
    int sub2[] = {1, 1, 3};
    size_t arr_len = sizeof(arr) / sizeof(arr[0]);
    size_t sub1_len = sizeof(sub1) / sizeof(sub1[0]);
    size_t sub2_len = sizeof(sub2) / sizeof(sub2[0]);

    if (is_subarray(arr, arr_len, sub1, sub1_len)) {
        if (printf("sub1 is a subarray of arr\n") < 0) {
            return 1;
        }
    } else {
        if (printf("sub1 is not a subarray of arr\n") < 0) {
            return 1;
        }
    }

    if (is_subarray(arr, arr_len, sub2, sub2_len)) {
        if (printf("sub2 is a subarray of arr\n") < 0) {
            return 1;
        }
    } else {
        if (printf("sub2 is not a subarray of arr\n") < 0) {
            return 1;
        }
    }

    return 0;
}
