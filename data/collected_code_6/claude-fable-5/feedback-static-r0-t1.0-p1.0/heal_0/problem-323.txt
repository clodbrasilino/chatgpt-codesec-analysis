#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

static int rearrange_alternate(int *arr, size_t n)
{
    int *pos;
    int *neg;
    size_t pcount = 0;
    size_t ncount = 0;
    size_t i;
    size_t pi = 0;
    size_t ni = 0;
    size_t k = 0;

    if (arr == NULL || n == 0) {
        return -1;
    }

    pos = malloc(n * sizeof(int));
    if (pos == NULL) {
        return -1;
    }

    neg = malloc(n * sizeof(int));
    if (neg == NULL) {
        free(pos);
        return -1;
    }

    for (i = 0; i < n; i++) {
        if (arr[i] >= 0) {
            pos[pcount] = arr[i];
            pcount++;
        } else {
            neg[ncount] = arr[i];
            ncount++;
        }
    }

    while (pi < pcount && ni < ncount) {
        arr[k] = pos[pi];
        k++;
        pi++;
        arr[k] = neg[ni];
        k++;
        ni++;
    }

    while (pi < pcount) {
        arr[k] = pos[pi];
        k++;
        pi++;
    }

    while (ni < ncount) {
        arr[k] = neg[ni];
        k++;
        ni++;
    }

    free(pos);
    free(neg);
    return 0;
}

static void print_array(const int *arr, size_t n)
{
    size_t i;

    if (arr == NULL) {
        return;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return;
        }
    }
    (void)printf("\n");
}

int main(void)
{
    int data[] = { -5, 3, 4, -1, -2, 8, -9, 7, 6, -3 };
    size_t n = sizeof(data) / sizeof(data[0]);

    (void)printf("Original array:\n");
    print_array(data, n);

    if (rearrange_alternate(data, n) != 0) {
        (void)fprintf(stderr, "Failed to rearrange array\n");
        return EXIT_FAILURE;
    }

    (void)printf("Rearranged array:\n");
    print_array(data, n);

    return EXIT_SUCCESS;
}
