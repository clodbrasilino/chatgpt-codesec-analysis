#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char name[50];
    double price;
} Item;

typedef struct {
    Item* items;
    int size;
    int capacity;
} MaxHeap;

MaxHeap* createMaxHeap(int capacity) {
    if (capacity <= 0) return NULL;
    MaxHeap* heap = (MaxHeap*)malloc(sizeof(MaxHeap));
    if (!heap) return NULL;
    heap->items = (Item*)malloc(capacity * sizeof(Item));
    if (!heap->items) {
        free(heap);
        return NULL;
    }
    heap->size = 0;
    heap->capacity = capacity;
    return heap;
}

void swap(Item* a, Item* b) {
    Item temp = *a;
    *a = *b;
    *b = temp;
}

void heapifyDown(MaxHeap* heap, int idx) {
    int largest = idx;
    int left = 2 * idx + 1;
    int right = 2 * idx + 2;

    if (left < heap->size && heap->items[left].price > heap->items[largest].price)
        largest = left;

    if (right < heap->size && heap->items[right].price > heap->items[largest].price)
        largest = right;

    if (largest != idx) {
        swap(&heap->items[idx], &heap->items[largest]);
        heapifyDown(heap, largest);
    }
}

void heapifyUp(MaxHeap* heap, int idx) {
    int parent = (idx - 1) / 2;
    if (idx > 0 && heap->items[idx].price > heap->items[parent].price) {
        swap(&heap->items[idx], &heap->items[parent]);
        heapifyUp(heap, parent);
    }
}

int insertMaxHeap(MaxHeap* heap, Item item) {
    if (!heap) return -1;
    if (heap->size < heap->capacity) {
        heap->items[heap->size] = item;
        heapifyUp(heap, heap->size);
        heap->size++;
    } else if (item.price < heap->items[0].price) {
        heap->items[0] = item;
        heapifyDown(heap, 0);
    }
    return 0;
}

void freeMaxHeap(MaxHeap* heap) {
    if (heap) {
        free(heap->items);
        free(heap);
    }
}

Item* findNCheapestItems(Item* dataset, int datasetSize, int n, int* resultSize) {
    if (!dataset || datasetSize <= 0 || n <= 0 || !resultSize) return NULL;

    int heapCapacity = (n < datasetSize) ? n : datasetSize;
    MaxHeap* heap = createMaxHeap(heapCapacity);
    if (!heap) return NULL;

    for (int i = 0; i < datasetSize; i++) {
        insertMaxHeap(heap, dataset[i]);
    }

    *resultSize = heap->size;
    Item* result = (Item*)malloc(heap->size * sizeof(Item));
    if (!result) {
        freeMaxHeap(heap);
        return NULL;
    }

    for (int i = 0; i < heap->size; i++) {
        result[i] = heap->items[i];
    }

    freeMaxHeap(heap);
    return result;
}

int main() {
    Item dataset[] = {
        {"Item A", 15.5},
        {"Item B", 8.2},
        {"Item C", 20.0},
        {"Item D", 5.0},
        {"Item E", 12.3},
        {"Item F", 3.5}
    };
    int datasetSize = sizeof(dataset) / sizeof(dataset[0]);
    int n = 3;
    int resultSize = 0;

    Item* cheapestItems = findNCheapestItems(dataset, datasetSize, n, &resultSize);

    if (cheapestItems) {
        for (int i = 0; i < resultSize; i++) {
            printf("%s: $%.2f\n", cheapestItems[i].name, cheapestItems[i].price);
        }
        free(cheapestItems);
    } else {
        printf("Failed to find cheapest items.\n");
    }

    return 0;
}
