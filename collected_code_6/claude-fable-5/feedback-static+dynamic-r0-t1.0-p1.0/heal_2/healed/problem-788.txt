```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(len + 1);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->str, str, len);
    t->str[len] = '\0';

    if (list_size > 0) {
        t->list = malloc(list_size * sizeof(int));
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_size * sizeof(int));
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *s = "WEB";
    int values[] = {1, 2};
    Tuple *t;

    t = create_tuple(s, values, sizeof(values) / sizeof(values[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**First `memcpy` (CWE-120 warning):** This is a false positive. The destination buffer `t->str` is allocated with exactly `len + 1` bytes, and the copy transfers exactly `len` bytes. Since `len` is computed via `strnlen(str, MAX_STR_LEN)` and validated with `len >= MAX_STR_LEN` before allocation, the destination is always guaranteed to hold the source data, and the null terminator is written explicitly within bounds.

**Second `memcpy` (CWE-120 warning):** Also a false positive. The destination `t->list` is allocated with `list_size * sizeof(int)` bytes, exactly matching the copy size. The multiplication cannot overflow because of the guard `list_size > SIZE_MAX / sizeof(int)`, which rejects any size that would wrap around.

Additional defensive measures already present in the code:
- NULL pointer validation on both `str` and `list` (with the `list_size > 0` condition allowing an empty tuple)
- `strnlen` instead of `strlen`, bounding the scan of potentially unterminated input
- Every `malloc` return value is checked, with correct partial cleanup on failure paths to prevent memory leaks
- `free_tuple` tolerates NULL and frees all owned allocations, avoiding double-free or leak issues
- `print_tuple` guards against NULL input

The code compiles cleanly and follows correct ownership semantics, so no functional changes were required — the Flawfinder findings are risk-level-2 heuristic matches on `memcpy` usage rather than actual vulnerabilities.
