#include <stdio.h>
#include <stdbool.h>

bool isEquilateral(double a, double b, double c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return false;
    }
    if ((a + b <= c) || (a + c <= b) || (b + c <= a)) {
        return false;
    }
    return ((a == b) && (b == c));
}

int main(void) {
    double a = 0.0;
    double b = 0.0;
    double c = 0.0;

    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return 1;
    }

    if (isEquilateral(a, b, c)) {
        printf("Equilateral\n");
    } else {
        printf("Not Equilateral\n");
    }

    return 0;
}
