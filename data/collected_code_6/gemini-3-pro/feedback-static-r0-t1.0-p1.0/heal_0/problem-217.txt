#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char find_first_repeated_char(const char *str) {
    if (str == NULL) {
        return '\0';
    }

    int seen[256] = {0};

    for (int i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] > 0) {
            return str[i];
        }
        seen[c]++;
    }

    return '\0';
}

int main(void) {
    const char *test_str = "programming";
    char result = find_first_repeated_char(test_str);

    if (result != '\0') {
        printf("%c\n", result);
    } else {
        printf("None\n");
    }

    return 0;
}
