#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

static char *duplicate_word(const char *word, size_t len)
{
    char *copy;
    size_t alloc_size;

    if (word == NULL || len >= MAX_WORD_LEN) {
        return NULL;
    }

    if (len > SIZE_MAX - 1) {
        return NULL;
    }

    alloc_size = len + 1;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        if (len > alloc_size - 1) {
            free(copy);
            return NULL;
        }
        memcpy(copy, word, len);
    }
    copy[len] = '\0';

    return copy;
}

char **remove_duplicates(const char **words, size_t count, size_t *out_count)
{
    char **result = NULL;
    size_t unique = 0;
    size_t i;
    size_t j;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            result[unique] = duplicate_word(words[i], len);
            if (result[unique] == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Unique words:\n");
    for (i = 0; i < result_count; i++) {
        printf("%s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

**The `memcpy` finding (CWE-120, risk 2):** Flawfinder flags every `memcpy` call because it cannot statically verify destination buffer sizes. In this code, the finding is a false positive — the destination is already guaranteed to hold the source data:

1. `alloc_size = len + 1` is computed after confirming `len <= SIZE_MAX - 1`, so no integer overflow can occur in the size calculation.
2. The buffer is allocated with exactly `alloc_size` bytes, and the copy length is `len`, which is strictly less than `alloc_size`.
3. The defensive check `len > alloc_size - 1` before the `memcpy` re-verifies the invariant at the point of use.
4. Callers only pass `len` values obtained from `strnlen(words[i], MAX_WORD_LEN)` and reject anything `>= MAX_WORD_LEN`, so the source is guaranteed to contain at least `len` readable bytes.

Additional defensive measures already present that address related CWE classes: `strnlen`/`strncmp` bounded string operations (avoiding CWE-126 over-reads on unterminated input), the `count > SIZE_MAX / sizeof(char *)` guard against multiplication overflow in the allocation (CWE-190), `calloc` zero-initialization so cleanup on failure never frees uninitialized pointers, and full rollback of partial allocations on error to prevent leaks (CWE-401). The code is correct as-is; no functional changes were required.
