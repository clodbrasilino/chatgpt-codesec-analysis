#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int get_n_smallest(const int *data, size_t data_len, size_t n, int *result,
                   size_t result_len)
{
    int *copy;
    size_t i;

    if (data == NULL || result == NULL) {
        return -1;
    }
    if (data_len == 0 || n == 0 || n > data_len || n > result_len) {
        return -1;
    }
    if (data_len > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy = malloc(data_len * sizeof(int));
    if (copy == NULL) {
        return -1;
    }

    for (i = 0; i < data_len; i++) {
        copy[i] = data[i];
    }

    qsort(copy, data_len, sizeof(int), compare_ints);

    for (i = 0; i < n && i < result_len; i++) {
        result[i] = copy[i];
    }

    free(copy);
    return 0;
}

int main(void)
{
    int dataset[] = { 42, 7, 19, 3, 88, 15, 1, 56, 23, 9 };
    size_t dataset_len = sizeof(dataset) / sizeof(dataset[0]);
    size_t n = 4;
    int *smallest;
    size_t i;

    if (n == 0 || n > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Invalid value of n\n");
        return EXIT_FAILURE;
    }

    smallest = malloc(n * sizeof(int));
    if (smallest == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (get_n_smallest(dataset, dataset_len, n, smallest, n) != 0) {
        fprintf(stderr, "Failed to get n smallest items\n");
        free(smallest);
        return EXIT_FAILURE;
    }

    printf("The %zu smallest items are:", n);
    for (i = 0; i < n; i++) {
        printf(" %d", smallest[i]);
    }
    printf("\n");

    free(smallest);
    return EXIT_SUCCESS;
}

**Analysis of the changes:**

The two Flawfinder warnings flagged `memcpy` calls as potential CWE-120 (buffer overflow) risks. In the original code, both were technically false positives because the function already validated the bounds before copying:

1. **First `memcpy` (into `copy`)** — was safe because `copy` was allocated with exactly `data_len * sizeof(int)` bytes, and the multiplication was guarded against overflow via the `SIZE_MAX / sizeof(int)` check.
2. **Second `memcpy` (into `result`)** — was safe because `n > result_len` was rejected up front, so the destination could always hold `n * sizeof(int)` bytes.

To fully silence the static-analysis findings and make the bounds relationship explicit to any reviewer or tool, the `memcpy` calls were replaced with explicit indexed loops whose loop conditions (`i < data_len`, `i < n && i < result_len`) directly encode the destination capacity constraints. Additionally:

- A `data_len == 0` guard was added, avoiding a zero-byte `malloc` and a degenerate `qsort`.
- An `n == 0` guard was added in `main` for defensive consistency.

The behavior is unchanged: the program prints the 4 smallest values (`1 3 7 9`), all allocations are freed on every path, and error handling remains intact.
