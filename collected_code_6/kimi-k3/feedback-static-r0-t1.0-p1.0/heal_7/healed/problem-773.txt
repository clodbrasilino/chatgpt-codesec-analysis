#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>
#include <errno.h>

int find_substrings(const char *str, const char *substr, size_t **positions) {
    if (str == NULL || substr == NULL || positions == NULL) {
        return -1;
    }

    size_t str_len = strnlen(str, 1025);
    if (str_len > 1024) {
        return -1;
    }

    size_t substr_len = strnlen(substr, 257);
    if (substr_len > 256) {
        return -1;
    }

    if (substr_len == 0 || str_len == 0 || substr_len > str_len) {
        *positions = NULL;
        return 0;
    }

    size_t count = 0;
    size_t capacity = 10;
    size_t *pos_array = (size_t *)malloc(capacity * sizeof(size_t));
    
    if (pos_array == NULL) {
        return -1;
    }

    const char *current = str;
    const char *found;

    while ((found = strstr(current, substr)) != NULL) {
        if (count >= capacity) {
            size_t new_capacity = capacity * 2;
            if (new_capacity > SIZE_MAX / sizeof(size_t)) {
                free(pos_array);
                return -1;
            }
            size_t *new_array = (size_t *)realloc(pos_array, new_capacity * sizeof(size_t));
            
            if (new_array == NULL) {
                free(pos_array);
                return -1;
            }
            
            pos_array = new_array;
            capacity = new_capacity;
        }

        pos_array[count] = (size_t)(found - str);
        count++;
        current = found + 1;
    }

    if (count == 0) {
        free(pos_array);
        *positions = NULL;
        return 0;
    }

    size_t *final_array = (size_t *)realloc(pos_array, count * sizeof(size_t));
    if (final_array != NULL) {
        pos_array = final_array;
    }

    *positions = pos_array;
    return (int)count;
}

int main(void) {
    char *str = NULL;
    char *substr = NULL;
    size_t *positions = NULL;
    int result;
    size_t str_size = 1025;
    size_t substr_size = 257;
    ssize_t str_read;
    ssize_t substr_read;

    str = (char *)malloc(str_size);
    if (str == NULL) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        return 1;
    }

    substr = (char *)malloc(substr_size);
    if (substr == NULL) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        free(str);
        return 1;
    }

    printf("Enter the main string: ");
    str_read = getline(&str, &str_size, stdin);
    if (str_read == -1) {
        fprintf(stderr, "Error reading input\n");
        free(str);
        free(substr);
        return 1;
    }
    if (str_read > 0 && str[str_read - 1] == '\n') {
        str[str_read - 1] = '\0';
    }

    printf("Enter the substring to find: ");
    substr_read = getline(&substr, &substr_size, stdin);
    if (substr_read == -1) {
        fprintf(stderr, "Error reading input\n");
        free(str);
        free(substr);
        return 1;
    }
    if (substr_read > 0 && substr[substr_read - 1] == '\n') {
        substr[substr_read - 1] = '\0';
    }

    result = find_substrings(str, substr, &positions);

    if (result < 0) {
        fprintf(stderr, "Error: Memory allocation failed or invalid parameters\n");
        free(str);
        free(substr);
        return 1;
    } else if (result == 0) {
        printf("Substring not found\n");
    } else {
        printf("Found %d occurrence(s) at position(s): ", result);
        for (int i = 0; i < result; i++) {
            printf("%zu", positions[i]);
            if (i < result - 1) {
                printf(", ");
            }
        }
        printf("\n");
    }

    free(positions);
    free(str);
    free(substr);
    return 0;
}
