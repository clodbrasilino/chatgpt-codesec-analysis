#include <stdlib.h>

typedef struct {
    int *data;
    size_t size;
} List;

int compare(const void *a, const void *b) {
    List *listA = (List *)a;
    List *listB = (List *)b;
    return listA->data[index] - listB->data[index];
}

void free_lists(List *lists, size_t count) {
    for (size_t i = 0; i < count; i++) {
        free(lists[i].data);
    }
    free(lists);
}
