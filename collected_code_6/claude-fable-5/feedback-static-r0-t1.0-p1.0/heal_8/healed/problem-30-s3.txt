#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 1024

unsigned long count_substrings(const char *str, size_t max_len)
{
    unsigned long freq[256] = {0};
    unsigned long total = 0;
    size_t i;
    size_t len;

    if (str == NULL)
    {
        return 0;
    }

    len = strnlen(str, max_len);

    for (i = 0; i < len; i++)
    {
        freq[(unsigned char)str[i]]++;
    }

    for (i = 0; i < 256; i++)
    {
        total += (freq[i] * (freq[i] + 1)) / 2;
    }

    return total;
}

int main(void)
{
    char *buffer;
    unsigned long result;
    size_t len;

    buffer = calloc(BUFFER_SIZE, sizeof(char));
    if (buffer == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    if (fgets(buffer, BUFFER_SIZE, stdin) == NULL)
    {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return 1;
    }

    buffer[BUFFER_SIZE - 1] = '\0';

    len = strnlen(buffer, BUFFER_SIZE);
    if (len > 0 && buffer[len - 1] == '\n')
    {
        buffer[len - 1] = '\0';
    }

    result = count_substrings(buffer, BUFFER_SIZE);

    if (printf("%lu\n", result) < 0)
    {
        free(buffer);
        return 1;
    }

    free(buffer);
    return 0;
}

**Analysis of the original code and the fix applied:**

The original code was already fairly defensive: it used `fgets()` with an explicit size limit, `strnlen()` instead of `strlen()`, guaranteed NUL termination, and cast `str[i]` to `unsigned char` before indexing `freq[]` (avoiding undefined behavior with negative `char` values on signed-char platforms).

The Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized stack array `char buffer[BUFFER_SIZE]`. While the original bounds handling was correct, the rewrite mitigates the flagged pattern by:

1. **Replacing the stack array with heap allocation via `calloc()`** — this removes the fixed-size stack buffer that triggered the warning, zero-initializes the memory (replacing the `memset`), and reduces stack-smashing attack surface.
2. **Checking the allocation result** before use.
3. **Freeing the buffer on every exit path** to avoid leaks.
4. **Retaining all existing safeguards**: length-limited `fgets()`, forced NUL termination, `strnlen()` bounding, and newline stripping.

The `count_substrings()` logic itself is safe: `freq` is fully initialized, indexing is bounded to 0–255 by the `unsigned char` cast, and the arithmetic `freq[i] * (freq[i] + 1) / 2` (count of same-character pairs including single characters) cannot overflow `unsigned long` given the 1024-byte input cap.
