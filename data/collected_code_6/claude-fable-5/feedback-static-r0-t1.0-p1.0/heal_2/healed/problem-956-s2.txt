```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count)
{
    regex_t regex;
    regmatch_t match;
    char **result = NULL;
    char **temp = NULL;
    size_t capacity = 8;
    size_t n = 0;
    const char *cursor = input;
    size_t remaining;
    int rc;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    remaining = strnlen(input, MAX_INPUT_LEN);
    if (remaining >= MAX_INPUT_LEN && input[MAX_INPUT_LEN] != '\0') {
        return NULL;
    }

    rc = regcomp(&regex, "[A-Z][^A-Z]*|^[^A-Z]+", REG_EXTENDED);
    if (rc != 0) {
        return NULL;
    }

    result = malloc(capacity * sizeof(char *));
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    while (remaining > 0 && *cursor != '\0') {
        rc = regexec(&regex, cursor, 1, &match, 0);
        if (rc != 0) {
            break;
        }

        if (match.rm_so < 0 || match.rm_eo < 0 || match.rm_so == match.rm_eo) {
            break;
        }

        size_t start = (size_t)match.rm_so;
        size_t end = (size_t)match.rm_eo;

        if (end > remaining || start > end) {
            break;
        }

        size_t len = end - start;

        if (len >= SIZE_MAX) {
            break;
        }

        char *token = malloc(len + 1);
        if (token == NULL) {
            for (size_t i = 0; i < n; i++) {
                free(result[i]);
            }
            free(result);
            regfree(&regex);
            return NULL;
        }

        memcpy(token, cursor + start, len);
        token[len] = '\0';

        if (n >= capacity) {
            if (capacity > SIZE_MAX / (2 * sizeof(char *))) {
                free(token);
                for (size_t i = 0; i < n; i++) {
                    free(result[i]);
                }
                free(result);
                regfree(&regex);
                return NULL;
            }
            capacity *= 2;
            temp = realloc(result, capacity * sizeof(char *));
            if (temp == NULL) {
                free(token);
                for (size_t i = 0; i < n; i++) {
                    free(result[i]);
                }
                free(result);
                regfree(&regex);
                return NULL;
            }
            result = temp;
        }

        result[n] = token;
        n++;
        cursor += end;
        remaining -= end;
    }

    regfree(&regex);
    *count = n;
    return result;
}

void free_tokens(char **tokens, size_t count)
{
    if (tokens == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *text = "helloWorldThisIsATestString";
    size_t count = 0;
    char **tokens = split_at_uppercase(text, &count);

    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);
    for (size_t i = 0; i < count; i++) {
        printf("Token %zu: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Summary of the changes addressing the flagged weaknesses:

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, and the function rejects inputs that are not NUL-terminated within that bound. This guarantees the length computation can never read past a defined limit, and the subsequent loop is additionally guarded by `remaining > 0`.

2. **CWE-120 (`memcpy` buffer overflow):** The destination buffer is allocated as `len + 1` immediately before the copy, with an explicit `len >= SIZE_MAX` check preventing integer overflow in the allocation size, so the destination is always guaranteed to hold exactly `len` bytes plus the terminator. The `end > remaining` check also ensures the source read stays within the validated input bounds.

3. **Missing include fixed:** The original code used `SIZE_MAX` without including `<stdint.h>`, which would fail to compile on conforming implementations; that header is now included.

The overall structure, memory-management discipline (cleanup on every failure path, `regfree` before every return), and the public API remain unchanged.
