```c
#include <stdio.h>
#include <stdlib.h>

double geometric_sum(int n)
{
    double sum = 0.0;
    double term = 1.0;
    int i;

    if (n < 0) {
        return 0.0;
    }

    for (i = 0; i <= n; i++) {
        sum += term;
        term /= 2.0;
    }

    return sum;
}

int main(void)
{
    int n = 0;
    double result = 0.0;

    printf("Enter a non-negative integer n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0) {
        fprintf(stderr, "Error: n must be non-negative\n");
        return EXIT_FAILURE;
    }

    result = geometric_sum(n);
    printf("Geometric sum for n = %d is %f\n", n, result);

    return EXIT_SUCCESS;
}
```

This computes the geometric series sum `1 + 1/2 + 1/4 + ... + 1/2^n` iteratively rather than recursively, which avoids stack exhaustion for large `n` (addressing CWE-664). The return value of `scanf` is checked and negative inputs are rejected before use (addressing CWE-703). The code uses `int main(void)`, explicit `EXIT_SUCCESS`/`EXIT_FAILURE` return codes, initialized variables, and consistent brace style (addressing CWE-710).
