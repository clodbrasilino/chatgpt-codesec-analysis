#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

static int *get_lucid_numbers(int n, size_t *count);

static int *get_lucid_numbers(int n, size_t *count)
{
    if (count == NULL || n < 1) {
        return NULL;
    }
    if ((size_t)n > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    size_t capacity = (size_t)n;
    size_t len = capacity;

    int *arr = malloc(len * sizeof *arr);
    if (arr == NULL) {
        return NULL;
    }

    int *result = malloc(capacity * sizeof *result);
    if (result == NULL) {
        free(arr);
        return NULL;
    }

    for (size_t i = 0; i < len; i++) {
        arr[i] = (int)(i + 1);
    }

    size_t rcount = 0;
    result[rcount++] = arr[0];

    size_t start = 1;
    while (start < len && rcount < capacity) {
        int step = arr[start];
        if (step <= 0) {
            break;
        }
        result[rcount++] = step;

        size_t write = start;
        size_t idx = 0;
        for (size_t read = start; read < len; read++, idx++) {
            if (idx % (size_t)step != 0) {
                if (write >= len || write > read || read >= capacity) {
                    break;
                }
                arr[write] = arr[read];
                write++;
            }
        }
        len = write;
    }

    free(arr);

    if (rcount > 0) {
        int *shrunk = realloc(result, rcount * sizeof *shrunk);
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    *count = rcount;
    return result;
}

int main(int argc, const char *const argv[])
{
    int n = 25;

    if (argc > 1) {
        const char *input = argv[1];
        char *end = NULL;
        errno = 0;
        long value = strtol(input, &end, 10);
        if (errno != 0 || end == input || *end != '\0' ||
            value < 1 || value > INT_MAX) {
            fprintf(stderr, "Invalid input: expected a positive integer\n");
            return EXIT_FAILURE;
        }
        n = (int)value;
    }

    size_t count = 0;
    int *lucid = get_lucid_numbers(n, &count);
    if (lucid == NULL) {
        fprintf(stderr, "Failed to compute lucid numbers\n");
        return EXIT_FAILURE;
    }

    printf("Lucid numbers <= %d:\n", n);
    for (size_t i = 0; i < count; i++) {
        printf("%d ", lucid[i]);
    }
    printf("\n");

    free(lucid);
    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

1. **First flagged read (`read < len` loop condition)** — CWE-120/CWE-20: The loop iterates `read` from `start` to `len`, and `len` is always ≤ `capacity` (it only ever shrinks via `len = write`, and `write` starts at `start < len`). All reads of `arr[read]` are therefore within the allocated bounds.

2. **Second flagged check (`write >= len || write > read || read >= capacity`)** — This explicit guard clause is the fix itself: it defensively verifies that both the write index and read index remain within valid array bounds before any element copy occurs, breaking out of the loop otherwise.

3. **Third flagged access (`arr[write] = arr[read]`)** — This access is only reachable after the guard above passes, so `write < len ≤ capacity` and `read < capacity` are guaranteed at the point of dereference.

Additional hardening already present and preserved:

- **Integer overflow protection**: `(size_t)n > SIZE_MAX / sizeof(int)` prevents overflow in the `malloc` size computation.
- **NULL checks**: Both `malloc` results and the `count` output pointer are validated; `arr` is freed if the second allocation fails, avoiding a leak.
- **Safe `realloc` pattern**: The result of `realloc` is assigned to a temporary (`shrunk`) so the original pointer is not lost on failure.
- **Robust input parsing**: `strtol` with `errno` reset, full-consumption check (`*end != '\0'`), and range validation (`1..INT_MAX`) prevents CWE-20 issues in `main`.
- **Memory hygiene**: `arr` and `lucid` are both freed on all paths, with no double-free or use-after-free.

The code is functionally correct (it computes ludic/lucid numbers via successive sieving) and the flagged Flawfinder findings are risk-1 informational hits that are already mitigated by the existing bounds guards, so the rewrite retains that logic verbatim with the comments removed.
