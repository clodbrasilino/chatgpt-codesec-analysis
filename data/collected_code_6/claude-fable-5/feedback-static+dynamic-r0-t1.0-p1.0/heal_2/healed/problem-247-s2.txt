```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT 1024

static void free_table(size_t **table, size_t rows)
{
    size_t i;

    if (table == NULL) {
        return;
    }
    for (i = 0; i < rows; i++) {
        free(table[i]);
    }
    free(table);
}

static char *longest_palindromic_subsequence(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t j;
    size_t len;
    size_t left;
    size_t right;
    size_t **dp;
    char *result;

    if (s == NULL) {
        return NULL;
    }

    n = strnlen(s, max_len);
    if (n == 0) {
        result = malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    if (n > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    dp = calloc(n, sizeof *dp);
    if (dp == NULL) {
        return NULL;
    }
    for (i = 0; i < n; i++) {
        dp[i] = calloc(n, sizeof **dp);
        if (dp[i] == NULL) {
            free_table(dp, i);
            return NULL;
        }
    }

    for (i = 0; i < n; i++) {
        dp[i][i] = 1;
    }

    for (len = 2; len <= n; len++) {
        for (i = 0; i + len - 1 < n; i++) {
            j = i + len - 1;
            if (s[i] == s[j]) {
                dp[i][j] = dp[i + 1][j - 1] + 2;
            } else {
                dp[i][j] = (dp[i + 1][j] > dp[i][j - 1]) ? dp[i + 1][j] : dp[i][j - 1];
            }
        }
    }

    if (dp[0][n - 1] > SIZE_MAX - 1) {
        free_table(dp, n);
        return NULL;
    }

    result = malloc(dp[0][n - 1] + 1);
    if (result == NULL) {
        free_table(dp, n);
        return NULL;
    }

    i = 0;
    j = n - 1;
    left = 0;
    right = dp[0][n - 1] - 1;
    while (i <= j) {
        if (i == j) {
            result[left] = s[i];
            break;
        }
        if (s[i] == s[j]) {
            result[left] = s[i];
            result[right] = s[j];
            left++;
            right--;
            i++;
            j--;
        } else if (dp[i + 1][j] >= dp[i][j - 1]) {
            i++;
        } else {
            j--;
        }
    }
    result[dp[0][n - 1]] = '\0';

    free_table(dp, n);
    return result;
}

int main(void)
{
    char input[MAX_INPUT];
    char *lps;
    size_t len;

    printf("Enter a string: ");
    if (fgets(input, sizeof input, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }
    input[sizeof input - 1] = '\0';

    len = strnlen(input, sizeof input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    lps = longest_palindromic_subsequence(input, sizeof input);
    if (lps == NULL) {
        fprintf(stderr, "Failed to compute longest palindromic subsequence\n");
        return EXIT_FAILURE;
    }

    printf("Longest palindromic subsequence: %s\n", lps);
    printf("Length: %zu\n", strnlen(lps, MAX_INPUT));

    free(lps);
    return EXIT_SUCCESS;
}
```

Analysis of the original code and the fixes applied:

**Weaknesses addressed:**

1. **Missing `<stdint.h>` include** — The original code used `SIZE_MAX` without including `<stdint.h>`, which is a portability defect that could cause a compilation failure or, worse, an implicit declaration issue on older toolchains. This is now included.

2. **Flawfinder CWE-119/CWE-120 (statically-sized buffer `input[MAX_INPUT]`)** — This was already mitigated correctly in the original: `fgets(input, sizeof input, stdin)` bounds the read, a defensive NUL terminator is written at `input[sizeof input - 1]`, and `strnlen` is used instead of `strlen` throughout, so no unbounded string operations can overrun the buffer. These mitigations are preserved.

3. **Integer overflow hardening** — The original already guarded the per-row `calloc` size with `n > SIZE_MAX / sizeof(size_t)`. I added an explicit guard before `malloc(dp[0][n - 1] + 1)` so the `+ 1` cannot wrap to zero, which would otherwise yield a zero-byte allocation followed by out-of-bounds writes.

**Other properties verified as sound:**
- All allocation failures (`calloc`, `malloc`) are checked, and `free_table` correctly releases partially constructed tables, so there are no memory leaks on error paths.
- The DP reconstruction indices (`i + 1`, `j - 1`) stay within bounds because `calloc` zero-initializes the table and the loop invariants keep `i <= j < n`.
- The newline strip checks `len > 0` before indexing `input[len - 1]`, avoiding an underflow on empty input.
