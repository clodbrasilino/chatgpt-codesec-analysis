```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    j = 0;
    i = 0;
    while (i < len) {
        if (snake[i] == '_') {
            i++;
            if (i < len && snake[i] != '_') {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                j++;
                i++;
            }
        } else {
            camel[j] = snake[i];
            j++;
            i++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    const char *input = "hello_world_example_string";
    char *result;

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\n", input) < 0) {
        free(result);
        return EXIT_FAILURE;
    }
    if (printf("Output: %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Key change:** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. This addresses the Flawfinder CWE-126 warning: if a non-NUL-terminated buffer is ever passed in, the read is now capped at a fixed maximum rather than running off the end of the allocation. If the string reaches the cap (meaning no terminator was found within the limit), the function rejects the input by returning `NULL` instead of silently truncating.

The rest of the logic was already sound: the `malloc(len + 1)` allocation is always sufficient since camelCase output can never exceed the input length (underscores are dropped), `toupper` is correctly called with an `unsigned char` cast to avoid undefined behavior on negative `char` values, all allocation failures are checked, and `result` is freed on every exit path in `main`.
