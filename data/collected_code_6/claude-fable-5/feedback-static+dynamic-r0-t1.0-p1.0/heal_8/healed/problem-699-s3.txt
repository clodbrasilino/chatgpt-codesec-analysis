```c
#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static void strip_newline(char *s)
{
    size_t len;

    if (s == NULL) {
        return;
    }

    len = strnlen(s, MAX_INPUT_LEN + 2);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r')) {
        s[len - 1] = '\0';
        len--;
    }
}

static int read_token(char *buf, size_t size)
{
    size_t i = 0;
    int c;

    if (buf == NULL || size == 0) {
        return -1;
    }

    do {
        c = getchar();
    } while (c == ' ' || c == '\t' || c == '\n' || c == '\r');

    if (c == EOF) {
        return -1;
    }

    while (c != EOF && c != ' ' && c != '\t' && c != '\n' && c != '\r') {
        if (i < size - 1) {
            buf[i] = (char)c;
            i++;
        } else {
            return -1;
        }
        c = getchar();
    }

    buf[i] = '\0';
    return 0;
}

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count01 = 0;
    int count10 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN + 1);
    len2 = strnlen(s2, MAX_INPUT_LEN + 1);

    if (len1 > MAX_INPUT_LEN || len2 > MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2 || len1 == 0) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] != s2[i]) {
            if (s1[i] == '0') {
                count01++;
            } else {
                count10++;
            }
        }
    }

    if ((count01 + count10) % 2 != 0) {
        return -1;
    }

    return (count01 / 2) + (count10 / 2) + 2 * (count01 % 2);
}

int main(void)
{
    char s1[MAX_INPUT_LEN + 2];
    char s2[MAX_INPUT_LEN + 2];
    int result;

    memset(s1, 0, sizeof(s1));
    memset(s2, 0, sizeof(s2));

    if (read_token(s1, sizeof(s1)) != 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
        return 0;
    }

    if (read_token(s2, sizeof(s2)) != 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
        return 0;
    }

    strip_newline(s1);
    strip_newline(s2);

    result = min_swaps(s1, s2);

    if (result < 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
    } else {
        if (printf("%d\n", result) < 0) {
            return 1;
        }
    }

    return 0;
}
```

**Key change explained (briefly):** The test failures pointed to a logic bug in `min_swaps`, not a memory-safety issue. The original code required `count01 == count10` and returned `count01`, which is the wrong model for this problem (minimum swaps to make two binary strings equal, swapping between the strings). The correct approach:

- A pair of `0/1` mismatches (or a pair of `1/0` mismatches) is resolved in **one** swap → `count01 / 2 + count10 / 2`.
- One leftover `0/1` plus one leftover `1/0` mismatch requires **two** swaps → `+ 2 * (count01 % 2)`.
- If the total mismatch count is odd (`(count01 + count10) % 2 != 0`), it's impossible → `Not Possible`.

The Flawfinder warnings (`getchar` in loops, statically-sized `char` arrays) were already mitigated in the original: `read_token` enforces `i < size - 1` before every write and rejects oversized tokens, buffers are zero-initialized with `memset`, and all string operations use `strnlen` with explicit bounds — so that defensive structure was preserved as-is.
