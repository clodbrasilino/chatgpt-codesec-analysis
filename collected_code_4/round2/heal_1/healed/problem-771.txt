```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Stack
{
    char* arr;
    int top;
    int capacity;
} Stack;

Stack* createStack(unsigned int capacity)
{
    Stack* stack = (Stack*)malloc(sizeof(Stack));

    if(stack == NULL)
    {
        return NULL;
    }

    stack->capacity = capacity;
    stack->top = -1;
    stack->arr = (char*)malloc(stack->capacity * sizeof(char));

    if(stack->arr == NULL)
    {
        free(stack);
        return NULL;
    }

    return stack;
}

int isFull(Stack* stack)
{
    return stack->top == stack->capacity - 1;
}

int isEmpty(Stack* stack)
{
    return stack->top == -1;
}

void push(Stack* stack, char item)
{
    if (!isFull(stack))
    {
        stack->arr[++stack->top] = item;
    }  
}

char pop(Stack* stack)
{
    return !isEmpty(stack) ? stack->arr[stack->top--] : 0;
}

char peek(Stack* stack)
{
    return !isEmpty(stack) ? stack->arr[stack->top] : 0;
}

int isBalanced(char* expr)
{
    Stack* stack = createStack(strlen(expr));
    
    if (stack == NULL)
    {
        return 0;
    }

    for (int i = 0; i<strlen(expr); i++)
    {
        if (expr[i] == '{' || expr[i] == '(' || expr[i] == '[')
            push(stack, expr[i]);
        else if (expr[i] == '}' || expr[i] == ')' || expr[i] == ']')
        {
            if (isEmpty(stack) || (expr[i] == '}' && peek(stack) != '{') || 
                (expr[i] == ')' && peek(stack) != '(') || 
                (expr[i] == ']' && peek(stack) != '['))
            {
                free(stack->arr);
                free(stack);
                return 0;
            }

            pop(stack);
        }
    }
  
    int balance_flag = isEmpty(stack) ? 1 : 0;
    free(stack->arr);
    free(stack);

    return balance_flag;
}

int main()
{
    char expr[100];
    printf("Enter an expression: ");
    scanf("%s", expr);
    if (isBalanced(expr))
        printf("Balanced \n");
    else
        printf("Not Balanced \n");
    return 0;
}
```
