static int compare_chars(const void *a, const void *b)
{
    unsigned char ca = *(const unsigned char *)a;
    unsigned char cb = *(const unsigned char *)b;

    if (ca < cb) {
        return -1;
    }
    if (ca > cb) {
        return 1;
    }
    return 0;
}

int sort_string(char *str, size_t max_len)
{
    size_t len;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        return -1;
    }

    if (len > 1) {
        qsort(str, len, sizeof(char), compare_chars);
    }

    return 0;
}

        return EXIT_FAILURE;

    if (len > 0 && buffer[len - 1] == '\n') {

    if (sort_string(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error sorting string\n");

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` is defensive belt-and-suspenders; `fgets` already null-terminates, but this guards against any future refactoring that might replace the input function.

3. **Bounded length calculation** — `strnlen(buffer, sizeof(buffer))` is used instead of `strlen()`, so length computation can never read past the buffer even if termination were somehow lost.

4. **Validation in `sort_string`** — the function rejects `NULL` pointers, zero-length limits, and unterminated strings (`len >= max_len`), and `qsort` operates only on the measured length, never beyond it.
