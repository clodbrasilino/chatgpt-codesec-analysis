```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    buffer[0] = '\0';

    if (number < 1 || number > 3999) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];
            if (used >= buffer_size || sym_len >= buffer_size - used) {
                buffer[0] = '\0';
                return -1;
            }
            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            number -= values[i];
        }
    }

    buffer[used] = '\0';
    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            if (printf("%d -> %s\n", test_values[i], roman) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, roman, sizeof(roman)) != 0) {
        if (printf("0 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, roman, sizeof(roman)) != 0) {
        if (printf("4000 is out of range, as expected\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Summary of changes addressing the flagged weaknesses:**

1. **CWE-126 (`strlen` over-read):** Removed the runtime `strlen` call entirely. Symbol lengths are now stored in a static, compile-time `symbol_lengths` table, and the symbols themselves are held in a fixed-size two-dimensional array (`char[][3]`), guaranteeing null termination and eliminating any dependence on runtime string scanning.

2. **CWE-120 (`memcpy` without bounds check):** The bounds check before `memcpy` was rewritten to avoid any possibility of integer overflow in the arithmetic: `used >= buffer_size || sym_len >= buffer_size - used` guarantees `used + sym_len < buffer_size` without computing a potentially wrapping sum, so the copy plus the terminating null byte always fit.

3. **CWE-119/CWE-120 (statically-sized array `roman`):** The buffer is zero-initialized with `memset` before first use, its size is always passed via `sizeof(roman)` rather than a hard-coded constant, and every write path in `int_to_roman` is bounds-checked against that size, so the fixed-size array cannot be overflowed.
