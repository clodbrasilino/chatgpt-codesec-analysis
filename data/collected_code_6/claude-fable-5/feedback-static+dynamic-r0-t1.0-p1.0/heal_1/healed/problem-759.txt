```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 128

bool is_decimal_precision_two(const char *str, size_t max_len)
{
    size_t len;
    size_t i = 0;
    size_t digits_before = 0;
    size_t digits_after = 0;
    bool dot_found = false;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    if (str[i] == '+' || str[i] == '-') {
        i++;
        if (i == len) {
            return false;
        }
    }

    for (; i < len; i++) {
        if (isdigit((unsigned char)str[i])) {
            if (dot_found) {
                digits_after++;
            } else {
                digits_before++;
            }
        } else if (str[i] == '.') {
            if (dot_found) {
                return false;
            }
            dot_found = true;
        } else {
            return false;
        }
    }

    if (digits_before == 0) {
        return false;
    }

    if (dot_found && (digits_after == 0 || digits_after > 2)) {
        return false;
    }

    return true;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a number: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Input error\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (is_decimal_precision_two(buffer, sizeof(buffer))) {
        printf("Valid decimal with precision of 2: %s\n", buffer);
    } else {
        printf("Invalid decimal with precision of 2: %s\n", buffer);
    }

    return 0;
}
```

**Analysis of the original code and applied fixes:**

The original code was already reasonably safe — the Flawfinder hits were largely low-risk advisories rather than exploitable defects. Here is how each was addressed:

1. **`strlen` on unbounded strings (CWE-126):** Replaced both calls with `strnlen()`, bounded by the known buffer capacity. In `is_decimal_precision_two()`, the caller now passes the maximum length explicitly, so the function can never over-read even if handed a non-terminated buffer. A result equal to `max_len` is treated as invalid input (no terminator found within bounds).

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer size is now defined via a single `MAX_INPUT_LEN` constant, `fgets()` remains bounded by `sizeof(buffer)`, and an explicit `buffer[sizeof(buffer) - 1] = '\0'` guarantees NUL-termination defensively regardless of libc behavior.

3. **Additional hardening (not flagged, but good practice):**
   - `fflush(stdout)` after the prompt ensures it appears before blocking on input.
   - When input exceeds the buffer (no `\n` captured), the remainder of the line is drained from `stdin` so leftover bytes cannot corrupt subsequent reads.
   - The `isdigit()` cast to `unsigned char` was already correct in the original and is preserved — passing a raw `char` that might be negative is undefined behavior.
