```C
#include <stdio.h>

int maxProduct(int* arr, int n) {
    if(n < 2) {
        return -1;
    }
    int max_product = arr[0] * arr[1];
    for(int i = 0; i < n; i++) {
        for(int j = i + 1; j < n; j++) {
            if(arr[i] * arr[j] > max_product) {
                max_product = arr[i] * arr[j];
            }
        }
    }
    return max_product;
}

int main() {
    int arr[] = {1, 20, 30, 44, 5, 56, 57, 8, 9, 10};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("Max product is %d", maxProduct(arr, n));
    return 0;
}
```
