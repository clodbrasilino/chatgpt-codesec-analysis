#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char** extract_strings(char **strings, int string_count, int size) {
    int i;
    int count = 0;
    char **result = malloc((string_count + 1) * sizeof(char*));
    
    if(result == NULL) {
        return NULL;
    }
    
    for(i = 0; i < string_count; ++i) {
        if(strlen(strings[i]) == size) {
            result[count] = malloc((size + 1) * sizeof(char));
            if(result[count] == NULL) {
                while(count > 0) {
                    --count;
                    free(result[count]);
                }
                free(result);
                return NULL;
            }
            strcpy(result[count], strings[i]);
            ++count;
        }
    }
    
    result[count] = NULL;
    
    return result;
}

int main() {
    int i;
    char *my_strings[] = {
        "hello",
        "world",
        "my",
        "name",
        "is",
        "developer",
        NULL
    };
    
    char **result = extract_strings(my_strings, 6, 5);
    
    if(result == NULL) {
        printf("Memory allocation failed.\n");
        return 1;
    }
    
    for(i = 0; result[i] != NULL; ++i) {
        printf("%s\n", result[i]);
    }
    
    for(i = 0; result[i] != NULL; ++i) {
        free(result[i]);
    }
    
    free(result);

    return 0;
}
