#include <stdio.h>
#include <stdlib.h>

int numWaysToPaintFence(int n, int k) {
    if (n == 0) return 0;
    if (n == 1) return k;
    if (n == 2) return k * k;
    
    int *dp = (int *)malloc((n + 1) * sizeof(int));
    if (dp == NULL) {
        return -1;
    }
    
    dp[1] = k;
    dp[2] = k * k;
    
    for (int i = 3; i <= n; i++) {
        dp[i] = (k - 1) * (dp[i - 1] + dp[i - 2]);
    }
    
    int result = dp[n];
    free(dp);
    return result;
}

int main(void) {
    int n = 3;
    int k = 2;
    int result = numWaysToPaintFence(n, k);
    
    if (result == -1) {
        printf("Memory allocation failed\n");
        return 1;
    }
    
    printf("Number of ways to paint %d posts with %d colors: %d\n", n, k, result);
    return 0;
}
