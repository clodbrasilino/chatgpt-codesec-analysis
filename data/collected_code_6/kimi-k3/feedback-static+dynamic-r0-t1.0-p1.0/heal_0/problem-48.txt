```c
#include <stdio.h>
#include <stdint.h>

uint32_t set_odd_bits(uint32_t num) {
    return num | 0xAAAAAAAAu;
}

int main(void) {
    uint32_t num = 0x12345678u;
    uint32_t result = set_odd_bits(num);
    
    printf("Original: 0x%08X\n", num);
    printf("Modified: 0x%08X\n", result);
    
    return 0;
}
```
