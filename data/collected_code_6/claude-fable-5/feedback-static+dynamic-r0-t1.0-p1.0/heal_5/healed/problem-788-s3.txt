```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_NAME_LEN 4096

typedef struct {
    char *name;
    int *values;
    size_t count;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t count)
{
    Tuple *t;
    size_t len;
    size_t values_size;

    if (str == NULL || (list == NULL && count > 0)) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    values_size = count * sizeof(int);

    len = strnlen(str, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->name = malloc(len + 1);
    if (t->name == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->name, str, len);
    t->name[len] = '\0';

    if (count > 0) {
        t->values = malloc(values_size);
        if (t->values == NULL) {
            free(t->name);
            free(t);
            return NULL;
        }
        memcpy(t->values, list, values_size);
    } else {
        t->values = NULL;
    }

    t->count = count;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->name);
        free(t->values);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL) {
        return;
    }

    printf("(\"%s\", [", t->name);
    for (i = 0; i < t->count; i++) {
        printf("%d", t->values[i]);
        if (i + 1 < t->count) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    int numbers[] = {10, 20, 30, 40};
    Tuple *t;

    t = create_tuple("example", numbers, sizeof(numbers) / sizeof(numbers[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

The two Flawfinder warnings (CWE-120, buffer overflow via `memcpy`) are effectively **false positives** in the original code, but the rewrite makes the safety guarantees more explicit:

1. **`memcpy(t->name, str, len)`** — The destination is allocated with exactly `len + 1` bytes, and `len` is bounded by `strnlen(str, MAX_NAME_LEN)` with an explicit rejection of unterminated/oversized strings (`len >= MAX_NAME_LEN`). The destination always holds the source data plus the null terminator, which is written separately.

2. **`memcpy(t->values, list, count * sizeof(int))`** — The overflow guard `count > SIZE_MAX / sizeof(int)` prevents integer overflow in the size calculation. In the rewrite, the size is computed once into `values_size` and used consistently for both `malloc` and `memcpy`, eliminating any possibility of the allocation size and copy size diverging (a common source of real CWE-120 bugs during future maintenance).

Other correctness properties preserved: all error paths clean up prior allocations (no leaks), `free_tuple` is null-safe, `t->values` is explicitly set to `NULL` for empty tuples so `free()` remains valid, and `print_tuple` guards against `NULL` input.
