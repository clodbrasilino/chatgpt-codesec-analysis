#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
} List;

Node *create_node(int data) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

List *create_list(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    return list;
}

bool list_append(List *list, int data) {
    if (list == NULL) {
        return false;
    }
    Node *new_node = create_node(data);
    if (new_node == NULL) {
        return false;
    }
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    return true;
}

void list_free(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

bool contains_sublist(List *list, List *sublist) {
    if (list == NULL || sublist == NULL || sublist->head == NULL) {
        return false;
    }
    if (list->head == NULL) {
        return false;
    }
    Node *main_node = list->head;
    Node *sub_head = sublist->head;
    while (main_node != NULL) {
        Node *main_temp = main_node;
        Node *sub_temp = sub_head;
        while (main_temp != NULL && sub_temp != NULL && main_temp->data == sub_temp->data) {
            main_temp = main_temp->next;
            sub_temp = sub_temp->next;
        }
        if (sub_temp == NULL) {
            return true;
        }
        main_node = main_node->next;
    }
    return false;
}

int main(void) {
    List *list = create_list();
    List *sublist = create_list();
    if (list == NULL || sublist == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    list_append(list, 1);
    list_append(list, 2);
    list_append(list, 3);
    list_append(list, 4);
    list_append(list, 5);
    list_append(sublist, 2);
    list_append(sublist, 3);
    list_append(sublist, 4);
    if (contains_sublist(list, sublist)) {
        printf("Sublist found\n");
    } else {
        printf("Sublist not found\n");
    }
    list_free(list);
    list_free(sublist);
    return EXIT_SUCCESS;
}
