```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LENGTH 4096

size_t remove_words_containing(char **words, size_t count, const char *pattern)
{
    size_t write_index = 0;
    size_t read_index;

    if (words == NULL || pattern == NULL) {
        return count;
    }

    for (read_index = 0; read_index < count; read_index++) {
        if (words[read_index] == NULL) {
            continue;
        }
        if (strstr(words[read_index], pattern) != NULL) {
            free(words[read_index]);
            words[read_index] = NULL;
        } else {
            words[write_index] = words[read_index];
            if (write_index != read_index) {
                words[read_index] = NULL;
            }
            write_index++;
        }
    }

    return write_index;
}

char *duplicate_string(const char *source)
{
    size_t length;
    size_t buffer_size;
    char *copy;

    if (source == NULL) {
        return NULL;
    }

    length = strnlen(source, MAX_WORD_LENGTH);
    if (length >= MAX_WORD_LENGTH) {
        return NULL;
    }

    buffer_size = length + 1;
    copy = malloc(buffer_size);
    if (copy == NULL) {
        return NULL;
    }

    if (length > 0) {
        memcpy(copy, source, length);
    }
    copy[buffer_size - 1] = '\0';
    return copy;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
        words[i] = NULL;
    }
    free(words);
}

int main(void)
{
    const char *initial_words[] = {
        "apple", "banana", "cherry", "grape", "pineapple", "mango"
    };
    const char *pattern = "ap";
    size_t total = sizeof(initial_words) / sizeof(initial_words[0]);
    size_t i;
    size_t remaining;
    char **words;

    words = malloc(total * sizeof(char *));
    if (words == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < total; i++) {
        words[i] = duplicate_string(initial_words[i]);
        if (words[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_word_list(words, i);
            return EXIT_FAILURE;
        }
    }

    printf("Original list:\n");
    for (i = 0; i < total; i++) {
        printf("  %s\n", words[i]);
    }

    remaining = remove_words_containing(words, total, pattern);

    printf("After removing words containing \"%s\":\n", pattern);
    for (i = 0; i < remaining; i++) {
        printf("  %s\n", words[i]);
    }

    free_word_list(words, remaining);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged weakness (CWE-120, unchecked `memcpy`) was already mitigated in the original code, since the destination buffer is allocated as `length + 1` bytes immediately before copying exactly `length` bytes. In the rewrite I made the safety guarantee more explicit:

- Introduced `buffer_size = length + 1` so the relationship between the allocation size and the copy length is visible at a glance.
- Guarded the `memcpy` with `if (length > 0)` to avoid any copy on an empty string.
- Wrote the null terminator via `copy[buffer_size - 1]`, tying the terminator placement directly to the allocated size.

Other defensive properties preserved from the original:

- `strnlen` with `MAX_WORD_LENGTH` bounds the length computation, preventing unbounded reads on non-terminated input, and inputs at or beyond the cap are rejected before any allocation.
- `remove_words_containing` performs stable in-place compaction, frees removed entries, and nulls out moved slots so no pointer is ever double-freed.
- `free_word_list` nulls each entry after freeing and tolerates a `NULL` list.
- The partial-failure path in `main` frees only the `i` successfully allocated strings, avoiding leaks and invalid frees.

One subtle point worth noting: after `remove_words_containing`, `free_word_list(words, remaining)` is correct only because the function nulls out all slots beyond `write_index`; freeing with `remaining` rather than `total` still leaks nothing since the removed entries were already freed and the trailing slots are `NULL`.
