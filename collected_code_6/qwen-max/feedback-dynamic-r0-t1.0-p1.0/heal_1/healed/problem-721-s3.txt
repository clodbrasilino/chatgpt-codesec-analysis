#include <stdio.h>
#include <stdlib.h>
#include <float.h>

double max(double a, double b) {
    return (a > b) ? a : b;
}

double findPathWithMaxAverage(int n, int **matrix, int *path) {
    double maxAvg = -DBL_MAX;
    int i, j, k;

    for (i = 0; i < n; ++i) {
        double sum = 0.0;
        for (k = 0; k < n; ++k) {
            sum += matrix[k][i];
        }
        if (sum / n > maxAvg) {
            maxAvg = sum / n;
            for (j = 0; j < n; ++j) {
                path[j] = j * n + i;
            }
        }
    }

    for (i = 0; i < n; ++i) {
        double sum = 0.0;
        for (k = 0; k < n; ++k) {
            sum += matrix[i][k];
        }
        if (sum / n > maxAvg) {
            maxAvg = sum / n;
            for (j = 0; j < n; ++j) {
                path[j] = i * n + j;
            }
        }
    }

    return maxAvg;
}

int main() {
    int n;
    printf("Enter the size of the matrix: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    int **matrix = (int **)malloc(n * sizeof(int *));
    if (matrix == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    for (int i = 0; i < n; ++i) {
        matrix[i] = (int *)malloc(n * sizeof(int));

            for (int j = 0; j < i; ++j) {
                free(matrix[j]);

            return 1;
        }

        for (int j = 0; j < n; ++j) {

                fprintf(stderr, "Invalid input\n");
                for (int k = 0; k < n; ++k) {

                }
                free(matrix);
