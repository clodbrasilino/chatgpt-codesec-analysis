#include <stdio.h>
#include <math.h>

double sum_of_inverse_divisors(unsigned int n) {
    double sum = 0.0;
    unsigned int limit;
    
    if (n == 0) {
        return 0.0;
    }

    limit = (unsigned int)sqrt((double)n);

    for (unsigned int i = 1; i <= limit; i++) {
        if (n % i == 0) {
            sum += 1.0 / i;
            if (i != (n / i)) {
                sum += 1.0 / (n / i);
            }
        }
    }

    return sum;
}

int main(void) {
    unsigned int num = 6;
    double result;

    result = sum_of_inverse_divisors(num);
    printf("%.6f\n", result);

    return 0;
}
