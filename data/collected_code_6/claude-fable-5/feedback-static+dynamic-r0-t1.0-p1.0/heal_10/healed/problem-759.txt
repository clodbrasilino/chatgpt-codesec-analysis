```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 128
#define MAX_FLUSH_ITERATIONS 4096

bool is_decimal_precision_two(const char *str, size_t max_len)
{
    size_t len;
    size_t i = 0;
    size_t digits_before = 0;
    size_t digits_after = 0;
    bool dot_found = false;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    if (str[i] == '+' || str[i] == '-') {
        i++;
        if (i == len) {
            return false;
        }
    }

    for (; i < len; i++) {
        if (isdigit((unsigned char)str[i])) {
            if (dot_found) {
                digits_after++;
            } else {
                digits_before++;
            }
        } else if (str[i] == '.') {
            if (dot_found) {
                return false;
            }
            dot_found = true;
        } else {
            return false;
        }
    }

    if (digits_before == 0) {
        return false;
    }

    if (dot_found && (digits_after == 0 || digits_after > 2)) {
        return false;
    }

    return true;
}

static void flush_stdin(void)
{
    int ch;
    size_t iterations = 0;

    do {
        ch = getchar();
        iterations++;
    } while (ch != '\n' && ch != EOF && iterations < MAX_FLUSH_ITERATIONS);
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Output error\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Input error\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    if (is_decimal_precision_two(buffer, sizeof(buffer))) {
        printf("Valid decimal with precision of 2: %s\n", buffer);
    } else {
        printf("Invalid decimal with precision of 2: %s\n", buffer);
    }

    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** in `flush_stdin`: This is a false positive in practice. The loop never stores the character into a buffer — it only discards input. Additionally, the `MAX_FLUSH_ITERATIONS` bound prevents unbounded looping on a pathological stream, and both `'\n'` and `EOF` are checked correctly (note that `ch` is declared as `int`, which is required to distinguish `EOF` from valid characters).

2. **Flawfinder `char buffer[MAX_INPUT_LEN]` warning (CWE-119/CWE-120)**: The statically-sized array is properly protected:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` limits the read to the buffer size and guarantees null termination on success.
   - The explicit `buffer[sizeof(buffer) - 1] = '\0';` is defense-in-depth.
   - `strnlen` (rather than `strlen`) bounds the length scan.
   - Overly long input lines are detected (`len == sizeof(buffer) - 1` with no trailing newline) and the remainder of stdin is drained via `flush_stdin`, preventing leftover input from corrupting subsequent reads.

3. **Robust validation logic**: `is_decimal_precision_two` guards against `NULL`, empty strings, unterminated buffers (`len == max_len`), lone signs, multiple dots, and casts to `unsigned char` before calling `isdigit`, avoiding undefined behavior on negative `char` values.

No functional changes were required — the original code already implements the mitigations for both flagged findings, so the rewrite simply removes the annotation comments.
