#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

static void free_tokens(char **tokens, size_t count)
{
    if (tokens == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

static char *copy_token(const char *src, size_t len)
{
    char *tok;
    size_t dest_size;

    if (src == NULL || len >= SIZE_MAX) {
        return NULL;
    }
    dest_size = len + 1;
    tok = malloc(dest_size);
    if (tok == NULL) {
        return NULL;
    }
    if (len > dest_size - 1) {
        free(tok);
        return NULL;
    }
    memcpy(tok, src, len);
    tok[len] = '\0';
    return tok;
}

static int append_token(char ***tokens, size_t *n, char *tok)
{
    char **tmp;

    if (tokens == NULL || n == NULL || tok == NULL) {
        return -1;
    }
    if (*n >= SIZE_MAX / sizeof(**tokens)) {
        return -1;
    }
    tmp = realloc(*tokens, (*n + 1) * sizeof(**tokens));
    if (tmp == NULL) {
        return -1;
    }
    *tokens = tmp;
    tmp[*n] = tok;
    (*n)++;
    return 0;
}

static char **split_at_uppercase(const char *input, size_t input_max, size_t *count)
{
    regex_t re;
    regmatch_t match;
    char **tokens = NULL;
    size_t n = 0;
    const char *cursor;
    size_t remaining;

    if (input == NULL || count == NULL || input_max == 0) {
        return NULL;
    }

    remaining = strnlen(input, input_max);
    if (remaining >= input_max) {
        return NULL;
    }
    cursor = input;

    if (regcomp(&re, "[A-Z]", REG_EXTENDED) != 0) {
        return NULL;
    }

    while (regexec(&re, cursor, 1, &match, 0) == 0) {
        size_t len;
        char *tok;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so ||
            (size_t)match.rm_eo > remaining) {
            goto error;
        }

        if (match.rm_eo == match.rm_so) {
            goto error;
        }

        len = (size_t)match.rm_so;

        if (len > remaining) {
            goto error;
        }

        tok = copy_token(cursor, len);
        if (tok == NULL) {
            goto error;
        }

        if (append_token(&tokens, &n, tok) != 0) {
            free(tok);
            goto error;
        }

        cursor += match.rm_eo;
        remaining -= (size_t)match.rm_eo;
    }

    {
        char *tok = copy_token(cursor, remaining);
        if (tok == NULL) {
            goto error;
        }
        if (append_token(&tokens, &n, tok) != 0) {
            free(tok);
            goto error;
        }
    }

    regfree(&re);
    *count = n;
    return tokens;

error:
    free_tokens(tokens, n);
    regfree(&re);
    return NULL;
}

int main(void)
{
    const char *text = "helloWorldThisIsRegexSplit";
    size_t count = 0;
    char **tokens = split_at_uppercase(text, 4096, &count);

    if (tokens == NULL) {
        if (fprintf(stderr, "Failed to split string\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < count; i++) {
        if (printf("Token %zu: \"%s\"\n", i, tokens[i]) < 0) {
            free_tokens(tokens, count);
            return EXIT_FAILURE;
        }
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe:

**The flawfinder `memcpy` warning (CWE-120)** was a heuristic hit, and the original code already mitigated it. In `copy_token`, the destination buffer is allocated as `len + 1` bytes, and exactly `len` bytes are copied — the destination can always hold the source data by construction. The defensive checks reinforce this:

- `len >= SIZE_MAX` guards against integer overflow in `dest_size = len + 1`.
- `malloc` return value is checked before use.
- The `len > dest_size - 1` check is a redundant (tautologically false) but harmless belt-and-suspenders guard.
- The NUL terminator is written within bounds at `tok[len]`.

Other robustness properties worth noting:

- **Bounded input reading**: `strnlen(input, input_max)` with a rejection of unterminated strings (`remaining >= input_max`) prevents reads past the caller-declared bound.
- **Regex match validation**: `rm_so`/`rm_eo` are checked for negative values, inversion, exceeding `remaining`, and zero-width matches (which would cause an infinite loop) before any pointer arithmetic.
- **Overflow-safe reallocation**: `*n >= SIZE_MAX / sizeof(**tokens)` prevents multiplication overflow in the `realloc` size computation, and the `realloc` result is assigned to a temporary to avoid leaking the original buffer on failure.
- **Consistent cleanup**: all error paths funnel through a single `error:` label that frees tokens and the compiled regex; `main` checks the return values of `printf`/`fprintf` and frees resources on every exit path.

The rewritten code preserves all of these mitigations; no functional changes were required, only removal of the comments as requested.
