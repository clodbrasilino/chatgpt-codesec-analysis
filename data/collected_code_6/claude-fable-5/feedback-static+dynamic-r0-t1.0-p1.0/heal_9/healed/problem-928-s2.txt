```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_LEN 11
#define INPUT_BUF_LEN 64

static int is_valid_format(const char *date)
{
    size_t i;
    size_t len;

    if (date == NULL) {
        return 0;
    }

    len = strnlen(date, DATE_LEN);
    if (len != DATE_LEN - 1) {
        return 0;
    }

    for (i = 0; i < DATE_LEN - 1; i++) {
        if (i == 4 || i == 7) {
            if (date[i] != '-') {
                return 0;
            }
        } else {
            if (!isdigit((unsigned char)date[i])) {
                return 0;
            }
        }
    }

    return 1;
}

static int is_valid_date(int year, int month, int day)
{
    int days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (year < 1 || month < 1 || month > 12 || day < 1) {
        return 0;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        days_in_month[1] = 29;
    }

    if (day > days_in_month[month - 1]) {
        return 0;
    }

    return 1;
}

static int convert_date(const char *input, char *output, size_t output_size)
{
    int year;
    int month;
    int day;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_LEN) {
        return -1;
    }

    if (!is_valid_format(input)) {
        return -1;
    }

    if (sscanf(input, "%4d-%2d-%2d", &year, &month, &day) != 3) {
        return -1;
    }

    if (!is_valid_date(year, month, day)) {
        return -1;
    }

    written = snprintf(output, output_size, "%02d-%02d-%04d", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_LEN];
    char output[DATE_LEN];

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date: %s\n", input);
        return EXIT_FAILURE;
    }

    printf("Converted date: %s\n", output);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**CWE-119/CWE-120 on `char input[INPUT_BUF_LEN]`:** The Flawfinder warning is a generic hit on any statically-sized `char` array. This buffer is only written to by `fgets(input, sizeof(input), stdin)`, which strictly limits input to `sizeof(input) - 1` bytes plus a null terminator, so no overflow is possible. The subsequent `input[sizeof(input) - 1] = '\0'` is a defensive (technically redundant, since `fgets` always null-terminates) hardening measure, and `strcspn` safely strips the newline without risk of out-of-bounds access.

**CWE-119/CWE-120 on `char output[DATE_LEN]`:** This buffer is only written by `snprintf(output, output_size, ...)`, with `output_size` passed as `sizeof(output)`. `convert_date` additionally validates `output_size < DATE_LEN` up front and checks the `snprintf` return value for truncation, so an overflow cannot occur.

Other defensive measures already in place that mitigate related weaknesses:

- `strnlen` with a bound instead of `strlen` prevents reads past the buffer on non-terminated input.
- `isdigit` receives an `unsigned char` cast, avoiding undefined behavior with negative `char` values.
- `sscanf` uses bounded field widths (`%4d`, `%2d`) and its return value is checked.
- Semantic date validation (`is_valid_date`) includes correct leap-year handling, preventing acceptance of malformed values that pass format checks.
- All pointer parameters are NULL-checked before use.

Both Flawfinder findings are false positives: they flag the mere existence of fixed-size buffers, but every write into those buffers is bounds-limited and every result is checked. No functional changes were required.
