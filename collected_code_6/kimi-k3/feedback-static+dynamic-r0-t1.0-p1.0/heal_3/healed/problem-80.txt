int tetrahedral_number(uint32_t n, uint64_t *result)
{
    uint64_t term1;
    uint64_t term2;
    uint64_t term3;
    uint64_t temp;
    
    if (result == NULL)
    {
        return -1;
    }
    
    if (n > 1811332U)
    {
        return -1;
    }
    
    term1 = (uint64_t)n;
    term2 = (uint64_t)(n + 1U);
    term3 = (uint64_t)(n + 2U);
    
    if (term1 % 6U == 0U)
    {
        term1 = term1 / 6U;
    }
    else if (term2 % 6U == 0U)
    {
        term2 = term2 / 6U;
    }
    else
    {
        term3 = term3 / 6U;
    }
    
    temp = term1 * term2;
    *result = temp * term3;
    
    return 0;
}
