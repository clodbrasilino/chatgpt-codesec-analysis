```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int *elements;
    size_t count;
} Tuple;

typedef struct {
    Tuple *tuples;
    size_t count;
} TupleList;

static int tuple_init(Tuple *t, const int *values, size_t count)
{
    size_t bytes;

    if (t == NULL) {
        return -1;
    }
    t->elements = NULL;
    t->count = 0;
    if (count == 0 || values == NULL) {
        return 0;
    }
    if (count > SIZE_MAX / sizeof(int)) {
        return -1;
    }
    bytes = count * sizeof(int);
    t->elements = malloc(bytes);
    if (t->elements == NULL) {
        return -1;
    }
    memcpy(t->elements, values, bytes);
    t->count = count;
    return 0;
}

static void tuple_free(Tuple *t)
{
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->count = 0;
    }
}

static void tuple_list_free(TupleList *list)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < list->count; i++) {
        tuple_free(&list->tuples[i]);
    }
    free(list->tuples);
    list->tuples = NULL;
    list->count = 0;
}

static int remove_empty_tuples(TupleList *list)
{
    size_t i;
    size_t write_index = 0;
    Tuple *shrunk;

    if (list == NULL) {
        return -1;
    }
    for (i = 0; i < list->count; i++) {
        if (list->tuples[i].count > 0) {
            list->tuples[write_index] = list->tuples[i];
            write_index++;
        } else {
            tuple_free(&list->tuples[i]);
        }
    }
    list->count = write_index;
    if (write_index == 0) {
        free(list->tuples);
        list->tuples = NULL;
        return 0;
    }
    if (write_index > SIZE_MAX / sizeof(Tuple)) {
        return -1;
    }
    shrunk = realloc(list->tuples, write_index * sizeof(Tuple));
    if (shrunk != NULL) {
        list->tuples = shrunk;
    }
    return 0;
}

static void print_tuple_list(const TupleList *list)
{
    size_t i;
    size_t j;

    if (list == NULL) {
        return;
    }
    printf("[");
    for (i = 0; i < list->count; i++) {
        printf("(");
        for (j = 0; j < list->tuples[i].count; j++) {
            printf("%d", list->tuples[i].elements[j]);
            if (j + 1 < list->tuples[i].count) {
                printf(", ");
            }
        }
        printf(")");
        if (i + 1 < list->count) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main(void)
{
    TupleList list;
    const int a[] = {1, 2};
    const int b[] = {3, 4, 5};
    const int c[] = {6};
    size_t i;

    list.count = 5;
    if (list.count > SIZE_MAX / sizeof(Tuple)) {
        fprintf(stderr, "Allocation size overflow\n");
        return EXIT_FAILURE;
    }
    list.tuples = malloc(list.count * sizeof(Tuple));
    if (list.tuples == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    for (i = 0; i < list.count; i++) {
        list.tuples[i].elements = NULL;
        list.tuples[i].count = 0;
    }

    if (tuple_init(&list.tuples[0], a, sizeof(a) / sizeof(a[0])) != 0 ||
        tuple_init(&list.tuples[1], NULL, 0) != 0 ||
        tuple_init(&list.tuples[2], b, sizeof(b) / sizeof(b[0])) != 0 ||
        tuple_init(&list.tuples[3], NULL, 0) != 0 ||
        tuple_init(&list.tuples[4], c, sizeof(c) / sizeof(c[0])) != 0) {
        fprintf(stderr, "Tuple initialization failed\n");
        tuple_list_free(&list);
        return EXIT_FAILURE;
    }

    printf("Before: ");
    print_tuple_list(&list);

    if (remove_empty_tuples(&list) != 0) {
        fprintf(stderr, "Failed to remove empty tuples\n");
        tuple_list_free(&list);
        return EXIT_FAILURE;
    }

    printf("After:  ");
    print_tuple_list(&list);

    tuple_list_free(&list);
    return EXIT_SUCCESS;
}
```

Analysis of the original code and the changes made:

**The flagged weakness (CWE-120, memcpy):** The original code was actually already safe. The destination buffer is allocated with `malloc(count * sizeof(int))` immediately before the `memcpy` of exactly `count * sizeof(int)` bytes, and integer overflow in the size calculation is guarded by the `count > SIZE_MAX / sizeof(int)` check. This is a classic Flawfinder false positive — it flags `memcpy` by pattern, not by data-flow analysis. In the rewrite, I compute the size once into `bytes` and use it for both `malloc` and `memcpy`, making the invariant "destination size equals copy size" explicit and impossible to desynchronize during future maintenance.

**Additional hardening added:**

1. **Overflow guard before `realloc` in `remove_empty_tuples`** — technically unreachable since `write_index` can never exceed the original allocation count, but it makes the multiplication defensively safe if the function is ever reused in another context.

2. **Overflow guard before the `malloc` in `main`** — same rationale, applied to `list.count * sizeof(Tuple)`.

3. **`sizeof(x)/sizeof(x[0])` instead of hard-coded lengths** in the `tuple_init` calls — the original passed literal `2`, `3`, `1`, which could silently desynchronize from the array definitions and cause an out-of-bounds read in `memcpy`'s source. Deriving the count from the array itself eliminates that class of bug.

**Correctness properties preserved:** the compaction in `remove_empty_tuples` correctly transfers ownership of `elements` pointers (no double-free, since freed slots beyond `write_index` are never touched again after `count` is reduced), the `realloc` failure path safely keeps the original larger buffer, and all error paths in `main` release resources before exit.
