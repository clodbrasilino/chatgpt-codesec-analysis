```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *get_nth_colon_separated_value(const char *input, int n) {
    const char *start = input;
    for (int i = 0; i < n; ++i) {
        start = strchr(start, ':');
        if (start == NULL) return NULL;
        start++;
    }
    const char *end = strchr(start, ':');
    if (end == NULL) {
        size_t len = strlen(input);
        end = input + len;
    } else {
        size_t start_offset = start - input;
        size_t end_offset = end - input;
        if (end_offset <= start_offset) return NULL; // Ensure end is after start
    }
    size_t len = end - start;
    char *result = malloc(len + 1);
    if (result == NULL) return NULL;
    memcpy(result, start, len);
    result[len] = '\0';
    return result;
}

int main() {
    const char *tuple = "field1:field2:field3:field4";
    int n = 2;
    char *value = get_nth_colon_separated_value(tuple, n);
    if (value != NULL) {
        printf("%s\n", value);
        free(value);
    } else {
        printf("Value not found\n");
    }
    return 0;
}
```
