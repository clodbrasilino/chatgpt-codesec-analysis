#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and how the identified weaknesses are addressed:**

1. **CWE-119/CWE-120 (statically-sized buffers `normalized` and `input`):** Both flagged buffers were already handled defensively, and the rewrite preserves those mitigations:
   - `input` is filled via `fgets()` with an explicit `sizeof(input)` bound, then force-terminated with `input[sizeof(input) - 1] = '\0'` as a belt-and-braces measure.
   - `normalized` is only written inside a loop bounded by both `len` and `sizeof(normalized) - 1`, and is always null-terminated afterward. Length is measured with `strnlen()` rather than `strlen()`, so an unterminated string can never cause an over-read.

2. **CWE-120/CWE-20 (`getchar()` in a loop):** The drain loop never stores the characters read — it only discards excess input until newline or `EOF` — so no buffer can overflow. It is extracted into a `flush_stdin()` helper for clarity, with proper `EOF` handling to avoid an infinite loop when stdin closes.

3. **Input validation (CWE-20):** Non-alphabetic characters cause rejection (`-1`), empty input and overlong input (`len >= MAX_INPUT_LEN`) are rejected, and `NULL` is checked before any dereference. Characters passed to `isalpha()`/`tolower()` are cast to `unsigned char`, avoiding undefined behavior for negative `char` values.

4. **Robustness details retained:** `fflush(stdout)` return value is checked, `fgets()` failure is reported to `stderr`, and exit codes use `EXIT_SUCCESS`/`EXIT_FAILURE`.

The original code was already well-hardened; the flawfinder hits are informational (risk 1–2) and all are mitigated by the existing bounds checks, which the rewrite keeps intact.
