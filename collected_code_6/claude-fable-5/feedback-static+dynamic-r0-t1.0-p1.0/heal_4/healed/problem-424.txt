```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0U;

    while (len < max_len && s[len] != '\0') {
        len++;
    }

    return len;
}

static char *extract_rear(const char *const *strings, size_t count)
{
    char *result = NULL;
    size_t i;

    if (strings == NULL || count == 0U || count >= (SIZE_MAX - 1U)) {
        return NULL;
    }

    result = malloc(count + 1U);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0U; i < count; i++) {
        size_t len;

        if (strings[i] == NULL) {
            free(result);
            return NULL;
        }

        len = bounded_strlen(strings[i], MAX_STRING_LEN);
        if (len >= MAX_STRING_LEN) {
            free(result);
            return NULL;
        }

        if (len == 0U) {
            result[i] = ' ';
        } else {
            result[i] = strings[i][len - 1U];
        }
    }

    result[count] = '\0';
    return result;
}

static int print_rear(const char *rear, size_t count)
{
    size_t i;

    if (printf("[") < 0) {
        return -1;
    }

    for (i = 0U; i < count; i++) {
        if (printf("'%c'", rear[i]) < 0) {
            return -1;
        }
        if (i + 1U < count) {
            if (printf(", ") < 0) {
                return -1;
            }
        }
    }

    if (printf("]\n") < 0) {
        return -1;
    }

    if (fflush(stdout) != 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *const words[] = { "Mers", "for", "Vers" };
    size_t count = sizeof(words) / sizeof(words[0]);
    char *rear = extract_rear(words, count);

    if (rear == NULL) {
        (void)fprintf(stderr, "Extraction failed\n");
        return EXIT_FAILURE;
    }

    if (print_rear(rear, count) != 0) {
        free(rear);
        return EXIT_FAILURE;
    }

    free(rear);
    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the reported weaknesses:**

The core logic of `extract_rear` was actually sound — for `{"Mers", "for", "Vers"}` it correctly produces `['s', 'r', 's']`. The "no output" failures point to output delivery rather than extraction logic. The key issues addressed:

1. **Missing `fflush(stdout)`** — the most likely cause of `<no output>` in the test harness. If the process output is captured through a pipe, `stdout` is fully buffered, and if the harness reads before process exit (or the process is terminated abnormally), buffered output is lost. An explicit flush guarantees delivery.

2. **Off-by-one risk in the overflow guard** — `count >= SIZE_MAX` still permits `count == SIZE_MAX - 1`... wait, no: it permits `count` up to `SIZE_MAX - 1`, making `malloc(count + 1U)` request `SIZE_MAX`, which is technically valid but pathological. Tightening to `count >= SIZE_MAX - 1U` keeps `count + 1U` safely representable.

3. **Refactored printing into `print_rear`** — consolidates the repetitive error-handling/`free` pattern in `main`, reducing the chance of a leaked allocation on a missed error path.

The rest of the original design is retained deliberately: `bounded_strlen` protects against non-terminated strings, `NULL` element checks prevent dereference faults, and empty strings map to a space rather than reading `strings[i][-1]` — all of which were already correct defenses.
