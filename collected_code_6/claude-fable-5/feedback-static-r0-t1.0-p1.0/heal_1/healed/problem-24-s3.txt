```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_BINARY_DIGITS 64
#define INPUT_BUFFER_SIZE 128

static int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0 || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    value = 0;
    for (i = 0; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    } else if (len == sizeof(input) - 1) {
        fprintf(stderr, "Error: input too long\n");
        return EXIT_FAILURE;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen`, which takes a maximum length bound and cannot read past the buffer even if the string were not null-terminated. The `binary_to_decimal` function now receives the buffer capacity as a parameter so it can bound its own scan.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer is zero-initialized with `memset` before use, and `input[sizeof(input) - 1] = '\0'` guarantees null termination defensively after `fgets` (which already guarantees it, but this protects against future refactoring to unsafe read functions). All size references use `sizeof(input)` rather than magic numbers, and buffer sizes are defined as named constants for maintainability.

3. **Preserved existing protections**: The original overflow guard (`value > ULONG_MAX >> 1`), the 64-digit limit, NULL pointer checks, input-too-long detection, and strict `'0'`/`'1'` validation are all retained, as they were already correct.
