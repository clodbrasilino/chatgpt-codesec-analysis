```c
#include <stdio.h>
#include <stdbool.h>

bool is_majority_element(int arr[], int n, int x) {
    int count = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] == x) {
            count++;
        }
    }
    return count > n / 2;
}

int find_majority_element(int arr[], int n) {
    if (n == 0) {
        return -1;
    }

    int candidate = arr[n / 2];
    if (is_majority_element(arr, n, candidate)) {
        return candidate;
    }

    return -1;
}

int main() {
    int arr[] = {1, 2, 3, 3, 3, 3, 10};
    int n = sizeof(arr) / sizeof(arr[0]);

    int majority = find_majority_element(arr, n);

    if (majority != -1) {
        printf("Majority element is %d\n", majority);
    } else {
        printf("No majority element\n");
    }

    return 0;
}
```
