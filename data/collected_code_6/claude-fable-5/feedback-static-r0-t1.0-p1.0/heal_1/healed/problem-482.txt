```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_LEN 256
#define ERRBUF_LEN 128

static int find_sequences(const char *input);

static int find_sequences(const char *input)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor;
    char buffer[MAX_MATCH_LEN];
    int ret;
    int found;

    if (input == NULL) {
        fprintf(stderr, "Error: input string is NULL\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Error: failed to compile regex: %s\n", errbuf);
        return -1;
    }

    found = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            char errbuf[ERRBUF_LEN];
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Error: regex execution failed: %s\n", errbuf);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            regfree(&regex);
            fprintf(stderr, "Error: invalid match offsets\n");
            return -1;
        }

        {
            size_t length = (size_t)(match.rm_eo - match.rm_so);
            if (length >= sizeof(buffer)) {
                length = sizeof(buffer) - 1;
            }
            memcpy(buffer, cursor + match.rm_so, length);
            buffer[length] = '\0';
        }

        printf("Found match: %s\n", buffer);
        found++;

        if (match.rm_eo == 0) {
            cursor++;
        } else {
            cursor += match.rm_eo;
        }
    }

    regfree(&regex);

    if (found == 0) {
        printf("No matches found.\n");
    }

    return found;
}

int main(void)
{
    const char *test_string = "HelloWorld this Is a TestString with Mixed caseWords";
    int result;

    printf("Input: %s\n", test_string);

    result = find_sequences(test_string);
    if (result < 0) {
        fprintf(stderr, "Error: sequence search failed\n");
        return EXIT_FAILURE;
    }

    printf("Total matches: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Summary of changes made:**

1. **CWE-119/CWE-120 (buffer for matches):** The original code already clamped `length` to `sizeof(buffer) - 1` before `memcpy`, so the overflow risk was mitigated. I kept the bounds check and added validation of `rm_so`/`rm_eo` offsets before computing `length`, so a negative or inverted offset pair can never produce a huge unsigned value that would bypass the clamp logic.

2. **CWE-120 (memcpy):** Guaranteed safe by the clamped `length` combined with the new offset sanity check — the destination can always hold the source data plus the NUL terminator.

3. **`errbuf` arrays:** `regerror` is length-bounded via `sizeof(errbuf)`, but I added an explicit NUL termination as defense in depth, and factored the size into an `ERRBUF_LEN` constant.

4. **`variableScope` (cppcheck):** Moved `length` into the innermost block where it is actually used, reducing its scope as suggested.

5. **Robustness bonus:** Added protection against a zero-length match (`rm_eo == 0`), which would cause an infinite loop with a different pattern; the cursor now always advances. Also declared `find_sequences` as `static` with a forward declaration for proper linkage hygiene.
