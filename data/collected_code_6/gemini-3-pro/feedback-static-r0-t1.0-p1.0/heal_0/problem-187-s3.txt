#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* longestCommonSubsequence(const char* str1, const char* str2) {
    if (!str1 || !str2) {
        return NULL;
    }

    int len1 = strlen(str1);
    int len2 = strlen(str2);

    int** dp = (int**)malloc((len1 + 1) * sizeof(int*));
    if (!dp) {
        return NULL;
    }

    for (int i = 0; i <= len1; i++) {
        dp[i] = (int*)malloc((len2 + 1) * sizeof(int));
        if (!dp[i]) {
            for (int k = 0; k < i; k++) {
                free(dp[k]);
            }
            free(dp);
            return NULL;
        }
        for (int j = 0; j <= len2; j++) {
            dp[i][j] = 0;
        }
    }

    for (int i = 1; i <= len1; i++) {
        for (int j = 1; j <= len2; j++) {
            if (str1[i - 1] == str2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = (dp[i - 1][j] > dp[i][j - 1]) ? dp[i - 1][j] : dp[i][j - 1];
            }
        }
    }

    int idx = dp[len1][len2];
    char* lcs = (char*)malloc((idx + 1) * sizeof(char));
    if (!lcs) {
        for (int i = 0; i <= len1; i++) {
            free(dp[i]);
        }
        free(dp);
        return NULL;
    }
    
    lcs[idx] = '\0';
    int i = len1, j = len2;

    while (i > 0 && j > 0) {
        if (str1[i - 1] == str2[j - 1]) {
            lcs[idx - 1] = str1[i - 1];
            i--;
            j--;
            idx--;
        } else if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }

    for (int k = 0; k <= len1; k++) {
        free(dp[k]);
    }
    free(dp);

    return lcs;
}

int main(void) {
    const char* str1 = "AGGTAB";
    const char* str2 = "GXTXAYB";

    char* result = longestCommonSubsequence(str1, str2);

    if (result) {
        printf("Longest Common Subsequence: %s\n", result);
        free(result);
    } else {
        printf("Memory allocation failed or invalid input.\n");
    }

    return 0;
}
