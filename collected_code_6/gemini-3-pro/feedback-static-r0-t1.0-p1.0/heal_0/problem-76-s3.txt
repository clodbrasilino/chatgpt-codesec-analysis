#include <stdio.h>

unsigned long long count_squares(int m, int n) {
    if (m <= 0 || n <= 0) {
        return 0;
    }

    unsigned long long total_squares = 0;
    int min_dimension = (m < n) ? m : n;

    for (int k = 1; k <= min_dimension; ++k) {
        total_squares += (unsigned long long)(m - k + 1) * (n - k + 1);
    }

    return total_squares;
}

int main(void) {
    int m = 0;
    int n = 0;

    if (scanf("%d %d", &m, &n) != 2) {
        return 1;
    }

    unsigned long long result = count_squares(m, n);
    
    if (printf("%llu\n", result) < 0) {
        return 1;
    }

    return 0;
}
