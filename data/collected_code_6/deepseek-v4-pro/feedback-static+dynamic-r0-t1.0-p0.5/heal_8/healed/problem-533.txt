#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_DOUBLE,
    TYPE_CHAR,
    TYPE_STRING,
    TYPE_POINTER,
    TYPE_UNKNOWN
} DataType;

typedef struct {
    void *data;
    DataType type;
    size_t data_size;
} TupleElement;

typedef struct {
    TupleElement *elements;
    size_t size;
    size_t capacity;
} Tuple;

Tuple* create_tuple(size_t initial_capacity) {
    Tuple *tuple = (Tuple*)malloc(sizeof(Tuple));
    if (!tuple) return NULL;
    
    if (initial_capacity == 0) {
        initial_capacity = 1;
    }
    
    tuple->elements = (TupleElement*)malloc(initial_capacity * sizeof(TupleElement));
    if (!tuple->elements) {
        free(tuple);
        return NULL;
    }
    
    tuple->size = 0;
    tuple->capacity = initial_capacity;
    return tuple;
}

void destroy_tuple(Tuple *tuple) {
    if (!tuple) return;
    
    for (size_t i = 0; i < tuple->size; i++) {
        if (tuple->elements[i].type == TYPE_STRING && tuple->elements[i].data) {
            free(tuple->elements[i].data);
        }
    }
    
    free(tuple->elements);
    free(tuple);
}

int tuple_add(Tuple *tuple, void *data, DataType type, size_t data_size) {
    if (!tuple || !data) return -1;
    
    if (data_size == 0) return -1;
    
    if (tuple->size >= tuple->capacity) {
        size_t new_capacity;
        if (tuple->capacity > SIZE_MAX / 2) {
            return -1;
        }
        new_capacity = tuple->capacity * 2;
        
        TupleElement *new_elements = (TupleElement*)realloc(tuple->elements, new_capacity * sizeof(TupleElement));
        if (!new_elements) return -1;
        
        tuple->elements = new_elements;
        tuple->capacity = new_capacity;
    }
    
    if (type == TYPE_STRING) {
        const char *str = (const char*)data;
        size_t len = strnlen(str, data_size);
        if (len >= data_size) {
            return -1;
        }
        
        char *str_copy = (char*)malloc(len + 1);
        if (!str_copy) return -1;
        
        if (len > 0) {
            memcpy(str_copy, str, len);
        }
        str_copy[len] = '\0';
        tuple->elements[tuple->size].data = str_copy;
        tuple->elements[tuple->size].data_size = len + 1;
    } else {
        if (data_size > 0) {
            void *data_copy = malloc(data_size);
            if (!data_copy) return -1;
            memcpy(data_copy, data, data_size);
            tuple->elements[tuple->size].data = data_copy;
        } else {
            tuple->elements[tuple->size].data = NULL;
        }
        tuple->elements[tuple->size].data_size = data_size;
    }
    
    tuple->elements[tuple->size].type = type;
    tuple->size++;
    
    return 0;
}

static int copy_element_to_tuple(Tuple *tuple, const TupleElement *element) {
    if (!tuple || !element) return -1;
    
    if (tuple->size >= tuple->capacity) {
        size_t new_capacity;
        if (tuple->capacity > SIZE_MAX / 2) {
            return -1;
        }
        new_capacity = tuple->capacity * 2;
        
        TupleElement *new_elements = (TupleElement*)realloc(tuple->elements, new_capacity * sizeof(TupleElement));
        if (!new_elements) return -1;
        
        tuple->elements = new_elements;
        tuple->capacity = new_capacity;
    }
    
    if (element->type == TYPE_STRING) {
        if (!element->data) return -1;
        
        size_t len = strlen((const char*)element->data);
        char *str_copy = (char*)malloc(len + 1);
        if (!str_copy) return -1;
        
        memcpy(str_copy, element->data, len + 1);
        tuple->elements[tuple->size].data = str_copy;
        tuple->elements[tuple->size].data_size = len + 1;
    } else {
        if (element->data_size > 0 && element->data) {
            void *data_copy = malloc(element->data_size);
            if (!data_copy) return -1;
            memcpy(data_copy, element->data, element->data_size);
            tuple->elements[tuple->size].data = data_copy;
        } else {
            tuple->elements[tuple->size].data = NULL;
        }
        tuple->elements[tuple->size].data_size = element->data_size;
    }
    
    tuple->elements[tuple->size].type = element->type;
    tuple->size++;
    
    return 0;
}

Tuple* remove_elements_by_type(Tuple *tuple, DataType type_to_remove) {
    if (!tuple) return NULL;
    
    Tuple *new_tuple = create_tuple(tuple->capacity);
    if (!new_tuple) return NULL;
    
    for (size_t i = 0; i < tuple->size; i++) {
        if (tuple->elements[i].type != type_to_remove) {
            if (copy_element_to_tuple(new_tuple, &tuple->elements[i]) != 0) {
                destroy_tuple(new_tuple);
                return NULL;
            }
        }
    }
    
    return new_tuple;
}

void print_tuple(Tuple *tuple) {
    if (!tuple) return;
    
    printf("(");
    for (size_t i = 0; i < tuple->size; i++) {
        if (!tuple->elements[i].data) {
            printf("null");
        } else {
            switch (tuple->elements[i].type) {
                case TYPE_INT:
                    if (tuple->elements[i].data_size >= sizeof(int)) {
                        printf("%d", *(int*)tuple->elements[i].data);
                    } else {
                        printf("invalid");
                    }
                    break;
                case TYPE_FLOAT:
                    if (tuple->elements[i].data_size >= sizeof(float)) {
                        printf("%f", *(float*)tuple->elements[i].data);
                    } else {
                        printf("invalid");
                    }
                    break;
                case TYPE_DOUBLE:
                    if (tuple->elements[i].data_size >= sizeof(double)) {
                        printf("%lf", *(double*)tuple->elements[i].data);
                    } else {
                        printf("invalid");
                    }
                    break;
                case TYPE_CHAR:
                    if (tuple->elements[i].data_size >= sizeof(char)) {
                        printf("%c", *(char*)tuple->elements[i].data);
                    } else {
                        printf("invalid");
                    }
                    break;
                case TYPE_STRING:
                    printf("%s", (char*)tuple->elements[i].data);
                    break;
                case TYPE_POINTER:
                    printf("%p", tuple->elements[i].data);
                    break;
                default:
                    printf("unknown");
                    break;
            }
        }
        
        if (i < tuple->size - 1) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main() {
    Tuple *tuple = create_tuple(4);
    if (!tuple) {
        fprintf(stderr, "Failed to create tuple\n");
        return 1;
    }
    
    int int_val1 = 10;
    int int_val2 = 20;
    float float_val = 3.14f;
    double double_val = 2.71828;
    char char_val = 'A';
    char str_val[] = "hello";
    int *ptr_val = &int_val1;
    
    tuple_add(tuple, &int_val1, TYPE_INT, sizeof(int));
    tuple_add(tuple, &float_val, TYPE_FLOAT, sizeof(float));
    tuple_add(tuple, &double_val, TYPE_DOUBLE, sizeof(double));
    tuple_add(tuple, &char_val, TYPE_CHAR, sizeof(char));
    tuple_add(tuple, str_val, TYPE_STRING, sizeof(str_val));
    tuple_add(tuple, &int_val2, TYPE_INT, sizeof(int));
    tuple_add(tuple, ptr_val, TYPE_POINTER, sizeof(void*));
    
    printf("Original tuple: ");
    print_tuple(tuple);
    
    Tuple *filtered_tuple = remove_elements_by_type(tuple, TYPE_INT);
    if (filtered_tuple) {
        printf("Tuple after removing TYPE_INT elements: ");
        print_tuple(filtered_tuple);
        destroy_tuple(filtered_tuple);
    }
    
    filtered_tuple = remove_elements_by_type(tuple, TYPE_STRING);
    if (filtered_tuple) {
        printf("Tuple after removing TYPE_STRING elements: ");
        print_tuple(filtered_tuple);
        destroy_tuple(filtered_tuple);
    }
    
    filtered_tuple = remove_elements_by_type(tuple, TYPE_FLOAT);
    if (filtered_tuple) {
        printf("Tuple after removing TYPE_FLOAT elements: ");
        print_tuple(filtered_tuple);
        destroy_tuple(filtered_tuple);
    }
    
    destroy_tuple(tuple);
    
    return 0;
}
