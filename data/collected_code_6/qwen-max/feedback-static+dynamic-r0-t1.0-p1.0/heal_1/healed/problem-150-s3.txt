#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

bool is_in_sequence(int n) {
    int a = 0, b = 1;
    while (a <= n) {
        if (a == n) return true;
        int next = a + b;
        if (next < 0 || next > INT_MAX) break;
        a = b;
        b = next;
    }
    return false;
}

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);
    if (is_in_sequence(num)) {
        printf("Number is in the sequence\n");
    } else {
        printf("Number is not in the sequence\n");
    }
    return 0;
}
