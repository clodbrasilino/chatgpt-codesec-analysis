#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    Node *tail;
    int size;
} List;

List *create_list(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    return list;
}

void append_node(List *list, int data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        return;
    }
    node->data = data;
    node->next = NULL;
    if (list->tail == NULL) {
        list->head = node;
        list->tail = node;
    } else {
        list->tail->next = node;
        list->tail = node;
    }
    list->size++;
}

void free_list(List *list) {
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    free(list);
}

List **split_list(List *list, int n, int *num_sublists) {
    if (list == NULL || n <= 0 || num_sublists == NULL) {
        return NULL;
    }
    if (list->size == 0) {
        *num_sublists = 0;
        return NULL;
    }

    *num_sublists = (list->size + n - 1) / n;
    List **sublists = (List **)malloc(*num_sublists * sizeof(List *));
    if (sublists == NULL) {
        return NULL;
    }

    Node *current = list->head;
    for (int i = 0; i < *num_sublists; i++) {
        sublists[i] = create_list();
        if (sublists[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free_list(sublists[j]);
            }
            free(sublists);
            return NULL;
        }
        for (int j = 0; j < n && current != NULL; j++) {
            append_node(sublists[i], current->data);
            current = current->next;
        }
    }

    return sublists;
}

void print_list(List *list) {
    Node *current = list->head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    List *list = create_list();
    if (list == NULL) {
        return 1;
    }

    for (int i = 1; i <= 10; i++) {
        append_node(list, i);
    }

    printf("Original list: ");
    print_list(list);

    int num_sublists = 0;
    List **sublists = split_list(list, 3, &num_sublists);

    if (sublists != NULL) {
        printf("Split into %d sublists:\n", num_sublists);
        for (int i = 0; i < num_sublists; i++) {
            printf("Sublist %d: ", i + 1);
            print_list(sublists[i]);
            free_list(sublists[i]);
        }
        free(sublists);
    }

    free_list(list);
    return 0;
}
