#include <stdlib.h>

#define ROWS 3

static int compute_averages(const double data[ROWS][COLS], size_t rows, size_t cols, double *result, size_t result_len)
{
    size_t i;
    size_t j;
    double sum;

    if (data == NULL || result == NULL) {
        return -1;
    }
    if (rows == 0 || cols == 0 || result_len < cols) {
        return -1;
    }

    for (j = 0; j < cols; j++) {
        sum = 0.0;

            sum += data[i][j];
        }
        result[j] = sum / (double)rows;
    }

    return 0;
}

{
