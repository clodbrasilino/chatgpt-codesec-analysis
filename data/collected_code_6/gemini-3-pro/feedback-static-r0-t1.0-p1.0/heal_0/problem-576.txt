#include <stdio.h>
#include <stdbool.h>

bool isSubarray(const int* array1, size_t size1, const int* array2, size_t size2) {
    if (array1 == NULL || array2 == NULL || size1 < size2) {
        return false;
    }

    if (size2 == 0) {
        return true;
    }

    for (size_t i = 0; i <= size1 - size2; i++) {
        size_t j;
        for (j = 0; j < size2; j++) {
            if (array1[i + j] != array2[j]) {
                break;
            }
        }
        if (j == size2) {
            return true;
        }
    }

    return false;
}

int main(void) {
    int array1[] = {1, 2, 3, 4, 5, 6, 7};
    int array2[] = {3, 4, 5};
    int array3[] = {3, 5, 4};

    size_t size1 = sizeof(array1) / sizeof(array1[0]);
    size_t size2 = sizeof(array2) / sizeof(array2[0]);
    size_t size3 = sizeof(array3) / sizeof(array3[0]);

    if (isSubarray(array1, size1, array2, size2)) {
        printf("array2 is a subarray of array1\n");
    } else {
        printf("array2 is not a subarray of array1\n");
    }

    if (isSubarray(array1, size1, array3, size3)) {
        printf("array3 is a subarray of array1\n");
    } else {
        printf("array3 is not a subarray of array1\n");
    }

    return 0;
}
