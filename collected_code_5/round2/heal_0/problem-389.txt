```c
#include <stdio.h>

int lucasNumber(int n) {
    if (n < 0) {
        printf("Invalid input, please enter a non-negative integer.\n");
        return -1;
    }
    
    int a = 2, b = 1, c;
    if (n==0) 
        return a;
    else if (n==1)
        return b;
    
    for (int i=2; i<=n; i++) {
        c = a + b;
        a = b;
        b = c;
    }
    return b;
}

int main() {
    int n;
    printf("Enter a non-negative integer: ");
    scanf("%d", &n);
    
    int result = lucasNumber(n);
    if (result != -1) {
        printf("The %d'th Lucas number is: %d\n", n, result);
    }
    return 0;
}
```
