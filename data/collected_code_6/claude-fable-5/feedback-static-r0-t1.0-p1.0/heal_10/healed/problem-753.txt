#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_MAX_LEN 64U
#define RECORD_COUNT 5U

typedef struct {
    char name[NAME_MAX_LEN];
    int value;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static int set_tuple_name(Tuple *t, const char *name)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(t->name));
    if (len >= sizeof(t->name)) {
        return -1;
    }

    memcpy(t->name, name, len);
    t->name[len] = '\0';
    return 0;
}

static Tuple *find_min_k_records(const Tuple *records, size_t n, size_t k)
{
    Tuple *sorted = NULL;
    Tuple *result = NULL;
    size_t total_size;
    size_t result_size;

    if (records == NULL || n == 0U || k == 0U || k > n) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(Tuple)) {
        return NULL;
    }

    total_size = n * sizeof(Tuple);
    result_size = k * sizeof(Tuple);

    sorted = malloc(total_size);
    if (sorted == NULL) {
        return NULL;
    }

    memcpy(sorted, records, total_size);
    qsort(sorted, n, sizeof(Tuple), compare_tuples);

    result = malloc(result_size);
    if (result == NULL) {
        free(sorted);
        return NULL;
    }

    memcpy(result, sorted, result_size);
    free(sorted);

    return result;
}

int main(void)
{
    Tuple records[RECORD_COUNT];
    const char *names[RECORD_COUNT] = {"Manjeet", "Akshat", "Akash", "Nikhil", "Rohit"};
    const int values[RECORD_COUNT] = {10, 4, 2, 8, 6};
    size_t n = RECORD_COUNT;
    size_t k = 2U;
    Tuple *min_records = NULL;
    size_t i;

    memset(records, 0, sizeof(records));

    for (i = 0U; i < n; i++) {
        if (set_tuple_name(&records[i], names[i]) != 0) {
            if (fprintf(stderr, "Failed to initialize record\n") < 0) {
                return EXIT_FAILURE;
            }
            return EXIT_FAILURE;
        }
        records[i].value = values[i];
    }

    min_records = find_min_k_records(records, n, k);
    if (min_records == NULL) {
        if (fprintf(stderr, "Failed to find minimum records\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Minimum %zu records:\n", k) < 0) {
        free(min_records);
        return EXIT_FAILURE;
    }

    for (i = 0U; i < k; i++) {
        if (printf("(%s, %d)\n", min_records[i].name, min_records[i].value) < 0) {
            free(min_records);
            return EXIT_FAILURE;
        }
    }

    free(min_records);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why this code is safe:**

1. **`char name[NAME_MAX_LEN]` (CWE-119/CWE-120):** The statically-sized array is safe here because every write to it goes through `set_tuple_name()`, which validates the length with `strnlen()` bounded by `sizeof(t->name)` and rejects any string that would not fit including the null terminator. The array is never written to directly elsewhere.

2. **`memcpy` in `set_tuple_name` (CWE-120):** Safe because `len` is guaranteed to be strictly less than `sizeof(t->name)` before the copy, leaving room for the explicit null terminator written afterward.

3. **`memcpy(sorted, records, total_size)` (CWE-120):** Safe because `sorted` is allocated with exactly `total_size` bytes, and the multiplication `n * sizeof(Tuple)` is guarded against integer overflow with the `n > SIZE_MAX / sizeof(Tuple)` check.

4. **`memcpy(result, sorted, result_size)` (CWE-120):** Safe because `result` is allocated with exactly `result_size` bytes, `k <= n` is enforced, so `result_size <= total_size` and cannot overflow given the earlier overflow guard.

5. **`const char *names[5]` (CWE-119):** Safe because it is a fixed array of string literal pointers, indexed only by the loop bounded by `n`, which matches the array size (now tied together via the `RECORD_COUNT` macro to prevent divergence).

The Flawfinder hits are all false positives in this code — bounds checking, overflow guards, allocation-size matching, and null-pointer validation are already correctly in place. The rewrite preserves those mitigations and replaces the magic number `5` with a shared constant so the array sizes and loop bounds cannot silently drift apart.
