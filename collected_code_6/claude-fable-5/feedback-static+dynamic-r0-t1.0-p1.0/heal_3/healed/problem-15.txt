```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096U

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t num_tokens = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len = 0;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;
                size_t alloc_size;
                char *segment;

                if (seg_len >= MAX_INPUT_LEN || seg_len >= SIZE_MAX - 1U) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                alloc_size = seg_len + 1U;

                segment = malloc(alloc_size);
                if (segment == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                memcpy(segment, &input[start], seg_len);
                segment[seg_len] = '\0';

                if (num_tokens >= (SIZE_MAX / sizeof(char *)) - 1U) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                temp = realloc(tokens, (num_tokens + 1U) * sizeof(char *));
                if (temp == NULL) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }
                tokens = temp;
                tokens[num_tokens] = segment;
                num_tokens++;
            }
            start = i + 1U;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "HELLOworldFOObarBAZ123";
    char **tokens = NULL;
    size_t count = 0;
    size_t i;

    tokens = split_at_lowercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens: %zu\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu]: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Integer overflow in `seg_len + 1`**: The original comment flagged `alloc_size = seg_len + 1` as a possible overflow point. I added an explicit guard `seg_len >= SIZE_MAX - 1U` combined with the existing `MAX_INPUT_LEN` bound before computing `alloc_size`, guaranteeing the addition cannot wrap.

2. **Always-true condition (`knownConditionTrueFalse`)**: The redundant `if (seg_len < alloc_size)` check was removed. Since `alloc_size` is now provably `seg_len + 1` with no overflow possible, the dead `else` branch was eliminated.

3. **Flawfinder `memcpy` warning (CWE-120)**: The `memcpy` is retained but is now demonstrably safe — the destination buffer is allocated with exactly `seg_len + 1` bytes, the copy writes exactly `seg_len` bytes, and the null terminator lands at index `seg_len`, the last valid byte. The overflow-proof allocation size makes the destination guaranteed to hold the source.

4. **Declarations hoisted**: `alloc_size` and `segment` are declared before use in the block for stricter C90-style compatibility, and unsigned literals (`1U`) are used consistently to avoid implicit signed/unsigned conversions.

All other defensive measures (NULL checks, `strnlen` bounding, `realloc` overflow guard via `SIZE_MAX / sizeof(char *)`, and full cleanup on every error path) were preserved, so the function still leaks no memory on any failure path.
