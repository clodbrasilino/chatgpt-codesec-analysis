#include <stdio.h>
#include <stdlib.h>

void swap_adjacent(int *list, size_t length) {
    if (list == NULL || length < 2) {
        return;
    }

    for (size_t i = 0; i < length - 1; i += 2) {
        int temp = list[i];
        list[i] = list[i + 1];
        list[i + 1] = temp;
    }
}

int main(void) {
    size_t length = 6;
    int *list = (int *)malloc(length * sizeof(int));
    if (list == NULL) {
        return 1;
    }

    list[0] = 1;
    list[1] = 2;
    list[2] = 3;
    list[3] = 4;
    list[4] = 5;
    list[5] = 6;

    swap_adjacent(list, length);

    for (size_t i = 0; i < length; i++) {
        printf("%d ", list[i]);
    }
    printf("\n");

    free(list);
    return 0;
}
