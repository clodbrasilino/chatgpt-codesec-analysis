```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define KEY_SIZE 64

typedef struct {
    char key[KEY_SIZE];
    int value;
} CounterEntry;

static int set_entry(CounterEntry *entry, const char *key, int value)
{
    int written;

    if (entry == NULL || key == NULL) {
        return -1;
    }

    if (strlen(key) >= sizeof(entry->key)) {
        return -1;
    }

    written = snprintf(entry->key, sizeof(entry->key), "%s", key);
    if (written < 0 || (size_t)written >= sizeof(entry->key)) {
        return -1;
    }

    entry->value = value;
    return 0;
}

static int compare_by_value_desc(const void *a, const void *b)
{
    const CounterEntry *ea = (const CounterEntry *)a;
    const CounterEntry *eb = (const CounterEntry *)b;

    if (ea->value < eb->value) {
        return 1;
    }
    if (ea->value > eb->value) {
        return -1;
    }
    return strcmp(ea->key, eb->key);
}

static int sort_counter_by_value(CounterEntry *entries, size_t count)
{
    if (entries == NULL) {
        return -1;
    }
    if (count == 0) {
        return 0;
    }
    qsort(entries, count, sizeof(CounterEntry), compare_by_value_desc);
    return 0;
}

int main(void)
{
    size_t count = 5;
    size_t i;
    CounterEntry *entries;

    entries = calloc(count, sizeof(CounterEntry));
    if (entries == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (set_entry(&entries[0], "apple", 3) != 0 ||
        set_entry(&entries[1], "banana", 7) != 0 ||
        set_entry(&entries[2], "cherry", 1) != 0 ||
        set_entry(&entries[3], "date", 7) != 0 ||
        set_entry(&entries[4], "elderberry", 5) != 0) {
        fprintf(stderr, "Failed to initialize entries\n");
        free(entries);
        return EXIT_FAILURE;
    }

    if (sort_counter_by_value(entries, count) != 0) {
        fprintf(stderr, "Sorting failed\n");
        free(entries);
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s: %d\n", entries[i].key, entries[i].value) < 0) {
            free(entries);
            return EXIT_FAILURE;
        }
    }

    free(entries);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened against the Flawfinder-identified weakness (CWE-119/CWE-120, statically-sized buffer). The key mitigations, preserved and strengthened in the rewrite:

1. **Bounded copy via `snprintf`** — `set_entry` never uses `strcpy`/`strcat`; `snprintf` with `sizeof(entry->key)` guarantees no overflow and NUL-termination.
2. **Truncation detection** — the return value of `snprintf` is checked (`written < 0 || (size_t)written >= sizeof(...)`), so silently truncated keys are rejected rather than accepted, avoiding data-integrity issues (CWE-120 mitigation done correctly, not just overflow prevention).
3. **Added an explicit `strlen` pre-check** — a defense-in-depth guard that rejects oversized keys before formatting.
4. **NULL-pointer checks** on all externally-callable helpers (`set_entry`, `sort_counter_by_value`).
5. **`calloc` with element count and size** — avoids manual multiplication overflow (CWE-190) and zero-initializes the buffer, so `key` arrays are NUL-terminated even before assignment.
6. **Deterministic `qsort` comparator** — ties on `value` fall back to `strcmp` on the key, ensuring stable, reproducible output; comparisons avoid the classic `a - b` integer-overflow pitfall.
7. **Consistent cleanup** — `free(entries)` on every exit path prevents leaks, and I/O return values (`printf`) are checked.
