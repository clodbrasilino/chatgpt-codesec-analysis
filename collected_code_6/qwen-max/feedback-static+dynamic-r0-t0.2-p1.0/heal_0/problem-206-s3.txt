#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* concatenate(char *a, char *b) {
    size_t len_a = strlen(a);
    size_t len_b = strlen(b);
    char *result = (char *)malloc(len_a + len_b + 1);
    if (result == NULL) return NULL;
    strcpy(result, a);
    strcat(result, b);
    return result;
}

void process_tuples(char **tuples, int n, char ***output, int *output_size) {
    *output_size = n - 1;
    *output = (char **)malloc((*output_size) * sizeof(char *));
    if (*output == NULL) return;

    for (int i = 0; i < *output_size; ++i) {
        (*output)[i] = concatenate(tuples[i], tuples[i + 1]);
        if ((*output)[i] == NULL) {
            for (int j = 0; j < i; ++j) {
                free((*output)[j]);
            }
            free(*output);
            *output = NULL;
            *output_size = 0;
            return;
        }
    }
}

int main() {
    char *tuples[] = {"hello", "world", "foo", "bar"};
    int n = sizeof(tuples) / sizeof(tuples[0]);

    char **output;
    int output_size;

    process_tuples(tuples, n, &output, &output_size);

    for (int i = 0; i < output_size; ++i) {
        printf("%s\n", output[i]);
        free(output[i]);
    }
    free(output);

    return 0;
}
