```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LENGTH 4096

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t len = 0;

    while (len < max_len && str[len] != '\0') {
        len++;
    }

    return len;
}

int find_first_non_repeated(const char *str, char *result)
{
    size_t counts[256] = {0};
    size_t i;
    size_t len;

    if (str == NULL || result == NULL) {
        return -1;
    }

    len = bounded_length(str, MAX_INPUT_LENGTH);

    if (len == MAX_INPUT_LENGTH && str[len] != '\0') {
        return -1;
    }

    for (i = 0; i < len; i++) {
        counts[(unsigned char)str[i]]++;
    }

    for (i = 0; i < len; i++) {
        if (counts[(unsigned char)str[i]] == 1) {
            *result = str[i];
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    const char *test_strings[] = {
        "swiss",
        "aabbcc",
        "programming",
        ""
    };
    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;
    char result;

    for (i = 0; i < num_tests; i++) {
        int status = find_first_non_repeated(test_strings[i], &result);
        if (status == 0) {
            if (printf("First non-repeated character in \"%s\": %c\n",
                       test_strings[i], result) < 0) {
                return EXIT_FAILURE;
            }
        } else if (status == 1) {
            if (printf("No non-repeated character found in \"%s\"\n",
                       test_strings[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Invalid input provided\n") < 0) {
                return EXIT_FAILURE;
            }
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **CWE-126 (strlen over-read):** Replaced the unbounded `strlen` call with a `bounded_length` helper that scans at most `MAX_INPUT_LENGTH` bytes, guaranteeing the read is bounded even if a non-null-terminated buffer is passed. If the string reaches the limit without termination, the function rejects the input rather than risking an over-read. (On POSIX systems `strnlen` could be used directly, but the manual loop keeps it portable to strict ISO C.)

2. **Variable scope reduction:** `status` is now declared inside the `for` loop body in `main`, limiting its scope to where it is actually used, as flagged by the static analyzer.

3. All other defensive properties were preserved: NULL-pointer checks, the `(unsigned char)` cast before indexing `counts` (avoiding UB with negative `char` values), and checking the return values of `printf`/`fprintf`.
