```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int a;
    int b;
} Tuple;

typedef struct {
    Tuple t1;
    Tuple t2;
} NestedTuple;

int subtract_tuple_elements(NestedTuple *nt, int *res) {
    if (nt == NULL || res == NULL) {
        return -1;
    }
    res[0] = nt->t1.a - nt->t2.a;
    res[1] = nt->t1.b - nt->t2.b;
    return 0;
}

int main() {
    NestedTuple nt = {{20, 15}, {5, 7}};
    int res[2];
    if (subtract_tuple_elements(&nt, res) != 0) {
        return EXIT_FAILURE;
    }
    printf("%d %d\n", res[0], res[1]);
    return EXIT_SUCCESS;
}
```
