#include <stdio.h>
#include <stdlib.h>

long long merge_count(int *arr, int *temp, int left, int mid, int right) {
    int i = left;
    int j = mid;
    int k = left;
    long long count = 0;

    while (i <= mid - 1 && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            temp[k++] = arr[j++];
            count += (mid - i);
        }
    }

    while (i <= mid - 1) {
        temp[k++] = arr[i++];
    }

    while (j <= right) {
        temp[k++] = arr[j++];
    }

    for (i = left; i <= right; i++) {
        arr[i] = temp[i];
    }

    return count;
}

long long merge_sort_count(int *arr, int *temp, int left, int right) {
    long long count = 0;
    if (left < right) {
        int mid = left + (right - left) / 2;
        count += merge_sort_count(arr, temp, left, mid);
        count += merge_sort_count(arr, temp, mid + 1, right);
        count += merge_count(arr, temp, left, mid + 1, right);
    }
    return count;
}

long long count_inversions(int *arr, int n) {
    long long count = 0;
    if (arr == NULL || n <= 1) {
        return 0;
    }

    int *temp = (int *)malloc(n * sizeof(int));
    if (temp == NULL) {
        return -1;
    }

    count = merge_sort_count(arr, temp, 0, n - 1);

    free(temp);

    return count;
}

int main(void) {
    int arr1[] = {1, 20, 6, 4, 5};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    long long result1 = count_inversions(arr1, n1);
    printf("%lld\n", result1);

    int arr2[] = {1, 2, 3, 4, 5};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    long long result2 = count_inversions(arr2, n2);
    printf("%lld\n", result2);

    int arr3[] = {5, 4, 3, 2, 1};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    long long result3 = count_inversions(arr3, n3);
    printf("%lld\n", result3);

    long long result4 = count_inversions(NULL, 5);
    printf("%lld\n", result4);

    long long result5 = count_inversions(arr1, 0);
    printf("%lld\n", result5);

    return 0;
}
