#include <stdio.h>
#include <stdlib.h>

int canSortByCorners(int arr[], int n) {
    if (n <= 1) {
        return 1;
    }

    int *sorted = (int *)malloc(n * sizeof(int));
    if (sorted == NULL) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        sorted[i] = arr[i];
    }

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                int temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }

    int left = 0;
    int right = n - 1;
    int idx = 0;

    while (left <= right && idx < n) {
        if (arr[left] == sorted[idx]) {
            left++;
            idx++;
        } else if (arr[right] == sorted[idx]) {
            right--;
            idx++;
        } else {
            free(sorted);
            return 0;
        }
    }

    free(sorted);
    return 1;
}

int main(void) {
    int arr1[] = {2, 3, 5, 4, 1};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);

    if (canSortByCorners(arr1, n1)) {
        printf("Array 1 can be sorted by picking corner elements.\n");
    } else {
        printf("Array 1 cannot be sorted by picking corner elements.\n");
    }

    int arr2[] = {2, 4, 1, 3, 5};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);

    if (canSortByCorners(arr2, n2)) {
        printf("Array 2 can be sorted by picking corner elements.\n");
    } else {
        printf("Array 2 cannot be sorted by picking corner elements.\n");
    }

    int arr3[] = {1, 2, 3, 4, 5};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);

    if (canSortByCorners(arr3, n3)) {
        printf("Array 3 can be sorted by picking corner elements.\n");
    } else {
        printf("Array 3 cannot be sorted by picking corner elements.\n");
    }

    return 0;
}
