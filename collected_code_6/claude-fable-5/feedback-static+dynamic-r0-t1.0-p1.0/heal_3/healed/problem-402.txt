#include <stdio.h>
#include <stdlib.h>

static long long mulmod(long long a, long long b, long long mod)
{
    return (long long)((unsigned __int128)a * (unsigned __int128)b %
                       (unsigned __int128)mod);
}

long long power(long long base, long long exp, long long mod)
{
    long long result = 1 % mod;
    base %= mod;
    if (base < 0) {
        base += mod;
    }
    while (exp > 0) {
        if (exp & 1) {
            result = mulmod(result, base, mod);
        }
        exp >>= 1;
        base = mulmod(base, base, mod);
    }
    return result;
}

long long mod_inverse(long long a, long long p)
{
    return power(a, p - 2, p);
}

static long long ncr_small(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }
    if (r == 0 || r == n) {
        return 1 % p;
    }
    if (r > n - r) {
        r = n - r;
    }

    long long result = 1 % p;

    for (long long i = 0; i < r; i++) {
        result = mulmod(result, (n - i) % p, p);
        result = mulmod(result, mod_inverse((i + 1) % p, p), p);
    }

    return result;
}

long long ncr_mod_p(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }

    long long result = 1 % p;

    while (n > 0 || r > 0) {
        long long ni = n % p;
        long long ri = r % p;

        if (ri > ni) {
            return 0;
        }

        result = mulmod(result, ncr_small(ni, ri, p), p);

        n /= p;
        r /= p;
    }

    return result;
}

int main(void)
{
    long long n, r, p;

    if (scanf("%lld %lld %lld", &n, &r, &p) != 3) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0 || r < 0 || p <= 1) {
        fprintf(stderr, "Invalid values: require n >= 0, r >= 0, p > 1\n");
        return EXIT_FAILURE;
    }

    long long answer = ncr_mod_p(n, r, p);
    printf("%lld\n", answer);

    return EXIT_SUCCESS;
}

**Analysis of the fix:**

The core algorithm was mathematically sound — it correctly implements Lucas' theorem for computing C(n, r) mod p with a prime modulus, using `__int128` to avoid overflow in modular multiplication, Fermat's little theorem for modular inverses, and the `r > n - r` symmetry optimization.

The reported test failures (`expected 6, got <no output>`; `expected 25...`; `expected 10...`) point to an **I/O contract mismatch**, not a computational bug. The automated test harness expected the program to emit only the numeric result, but the original program:

1. Printed an interactive prompt (`"Enter n, r and prime p: "`) before reading input.
2. Wrapped the answer in a decorated format string (`"C(%lld, %lld) %% %lld = %lld\n"`), so the test's exact-match comparison against `6`, `25`, or `10` failed.

The fix removes the prompt and reduces output to `printf("%lld\n", answer)`, so a test case such as `n=4, r=2, p=7` now produces exactly `6`. Error handling for malformed input and invalid domain values (negative `n`/`r`, non-prime-eligible `p <= 1`) is preserved on `stderr`, which keeps `stdout` clean for the harness.
