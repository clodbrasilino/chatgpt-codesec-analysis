#include <stdio.h>
#include <stdlib.h>

int countRotations(const int arr[], int n) {
    if (n <= 0) {
        return 0;
    }

    int low = 0;
    int high = n - 1;

    while (low <= high) {
        if (arr[low] <= arr[high]) {
            return low;
        }

        int mid = low + (high - low) / 2;
        int next = (mid + 1) % n;
        int prev = (mid + n - 1) % n;

        if (arr[mid] <= arr[next] && arr[mid] <= arr[prev]) {
            return mid;
        }

        if (arr[mid] <= arr[high]) {
            high = mid - 1;
        } else if (arr[mid] >= arr[low]) {
            low = mid + 1;
        } else {
            low++;
        }
    }

    return 0;
}

int main(void) {
    int capacity = 16;
    int *arr = (int *)malloc(capacity * sizeof(int));
    int n = 0;
    
    if (arr == NULL) {
        return 1;
    }

    while (scanf("%d", &arr[n]) == 1) {
        n++;
        if (n >= capacity) {
            capacity *= 2;
            int *temp = (int *)realloc(arr, capacity * sizeof(int));
            if (temp == NULL) {
                free(arr);
                return 1;
            }
            arr = temp;
        }
    }
    
    if (n > 0) {
        int rotations = countRotations(arr, n);
        printf("%d\n", rotations);
    }
    
    free(arr);
    return 0;
}
