#include <stdio.h>
#include <stdlib.h>

int sum_of_divisors(int n) {
    if (n <= 1) {
        return 0;
    }
    int sum = 1;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i * i != n) {
                sum += n / i;
            }
        }
    }
    return sum;
}

long long sum_amicable_numbers(int limit) {
    if (limit <= 0) {
        return 0;
    }

    int *divisors_sum = (int *)calloc((size_t)(limit + 1), sizeof(int));
    if (divisors_sum == NULL) {
        exit(EXIT_FAILURE);
    }

    for (int i = 1; i <= limit; i++) {
        divisors_sum[i] = sum_of_divisors(i);
    }

    long long total_sum = 0;
    for (int i = 1; i <= limit; i++) {
        int a = i;
        int b = divisors_sum[a];
        if (b > 0 && b <= limit && b != a && divisors_sum[b] == a) {
            total_sum += a;
        }
    }

    free(divisors_sum);
    return total_sum;
}

int main(void) {
    int limit;
    
    if (scanf("%d", &limit) != 1) {
        return EXIT_FAILURE;
    }

    long long result = sum_amicable_numbers(limit);
    printf("%lld\n", result);

    return EXIT_SUCCESS;
}
