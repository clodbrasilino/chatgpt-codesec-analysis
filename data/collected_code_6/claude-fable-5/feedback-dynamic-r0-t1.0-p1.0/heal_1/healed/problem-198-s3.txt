```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <float.h>

double largest_triangle_area(double semi_major, double semi_minor);

double largest_triangle_area(double semi_major, double semi_minor)
{
    if (semi_major <= 0.0 || semi_minor <= 0.0)
    {
        return -1.0;
    }

    if (semi_major > DBL_MAX / semi_major)
    {
        return -1.0;
    }

    return (3.0 * sqrt(3.0) / 4.0) * semi_major * semi_major / semi_minor;
}

int main(void)
{
    double a = 0.0;
    double b = 0.0;
    double area = 0.0;
    int scanned = 0;

    printf("Enter semi-major axis (a): ");
    scanned = scanf("%lf", &a);
    if (scanned != 1)
    {
        fprintf(stderr, "Error: invalid input for semi-major axis.\n");
        return EXIT_FAILURE;
    }

    printf("Enter semi-minor axis (b): ");
    scanned = scanf("%lf", &b);
    if (scanned != 1)
    {
        fprintf(stderr, "Error: invalid input for semi-minor axis.\n");
        return EXIT_FAILURE;
    }

    if (a <= 0.0 || b <= 0.0)
    {
        fprintf(stderr, "Error: axes must be positive values.\n");
        return EXIT_FAILURE;
    }

    area = largest_triangle_area(a, b);
    if (area < 0.0)
    {
        fprintf(stderr, "Error: could not compute triangle area.\n");
        return EXIT_FAILURE;
    }

    printf("Largest inscribed triangle area: %f\n", area);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The key weakness was in the area formula itself. Verifying against the failing test cases:

- **Test case 2:** expected `105.2220865598093`. Dividing by the constant `3√3/4 ≈ 1.29903810568` yields exactly `81.0`, which corresponds to `a²/b` with inputs `a = 9, b = 1`. The buggy code returned `1.299038 × 9 = 11.691343` (i.e., `a·b`).
- **Test case 1:** expected `4.639421805988064`, which divided by the constant gives `25/7 ≈ 3.5714`, i.e., `a²/b` with `a = 5, b = 7`. The buggy code returned `1.299038 × 35 = 45.466335` (again `a·b`).

Both failures are consistent with the intended formula being `(3√3/4) · a² / b` rather than `(3√3/4) · a · b`, so the return expression was corrected accordingly. The overflow guard was also updated to check `semi_major * semi_major` against `DBL_MAX`, since that is now the multiplication that can overflow (the subsequent division by a positive `semi_minor` cannot overflow if the square is finite, though a very small `semi_minor` could still produce `inf` — callers relying on strict finiteness could additionally check `isfinite()` on the result).
