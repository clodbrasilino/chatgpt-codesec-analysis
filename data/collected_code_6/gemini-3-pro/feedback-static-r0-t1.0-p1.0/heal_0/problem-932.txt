#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void removeDuplicates(char **words, int *count) {
    if (words == NULL || count == NULL || *count <= 0) {
        return;
    }

    int n = *count;
    int newCount = 0;

    for (int i = 0; i < n; i++) {
        if (words[i] == NULL) {
            continue;
        }

        int isDuplicate = 0;
        for (int j = 0; j < newCount; j++) {
            if (strcmp(words[i], words[j]) == 0) {
                isDuplicate = 1;
                break;
            }
        }

        if (!isDuplicate) {
            words[newCount] = words[i];
            newCount++;
        } else {
            free(words[i]);
        }
    }

    *count = newCount;
}

int main(void) {
    int count = 5;
    char **words = malloc(count * sizeof(char *));
    
    if (words == NULL) {
        return 1;
    }

    words[0] = strdup("apple");
    words[1] = strdup("banana");
    words[2] = strdup("apple");
    words[3] = strdup("cherry");
    words[4] = strdup("banana");

    for (int i = 0; i < count; i++) {
        if (words[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(words[j]);
            }
            free(words);
            return 1;
        }
    }

    removeDuplicates(words, &count);

    for (int i = 0; i < count; i++) {
        printf("%s\n", words[i]);
    }

    for (int i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);

    return 0;
}
