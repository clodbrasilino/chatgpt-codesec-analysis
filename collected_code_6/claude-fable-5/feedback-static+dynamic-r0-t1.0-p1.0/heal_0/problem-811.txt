#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

typedef struct {
    int first;
    int second;
} Tuple;

bool tuples_equal(const Tuple *a, const Tuple *b)
{
    if (a == NULL || b == NULL) {
        return false;
    }
    return (a->first == b->first) && (a->second == b->second);
}

bool lists_identical(const Tuple *list1, size_t len1, const Tuple *list2, size_t len2)
{
    size_t i;

    if (len1 != len2) {
        return false;
    }
    if (len1 == 0U) {
        return true;
    }
    if (list1 == NULL || list2 == NULL) {
        return false;
    }
    for (i = 0U; i < len1; i++) {
        if (!tuples_equal(&list1[i], &list2[i])) {
            return false;
        }
    }
    return true;
}

int main(void)
{
    Tuple listA[] = { {1, 2}, {3, 4}, {5, 6} };
    Tuple listB[] = { {1, 2}, {3, 4}, {5, 6} };
    Tuple listC[] = { {1, 2}, {3, 4}, {7, 8} };
    size_t lenA = sizeof(listA) / sizeof(listA[0]);
    size_t lenB = sizeof(listB) / sizeof(listB[0]);
    size_t lenC = sizeof(listC) / sizeof(listC[0]);

    if (lists_identical(listA, lenA, listB, lenB)) {
        if (printf("List A and List B are identical\n") < 0) {
            return 1;
        }
    } else {
        if (printf("List A and List B are not identical\n") < 0) {
            return 1;
        }
    }

    if (lists_identical(listA, lenA, listC, lenC)) {
        if (printf("List A and List C are identical\n") < 0) {
            return 1;
        }
    } else {
        if (printf("List A and List C are not identical\n") < 0) {
            return 1;
        }
    }

    return 0;
}
