```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis notes:**

The two Flawfinder hits in the original are both false positives, and the code as written already mitigates them, so the rewrite preserves the same logic:

1. **`getchar` in a loop (CWE-120/CWE-20):** `discard_remaining_input()` never writes into a buffer — it only reads and discards characters until newline or `EOF`, with a proper `EOF` guard preventing an infinite loop on closed stdin. No overflow is possible.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer is only ever written by `fgets(input, sizeof(input), stdin)`, which enforces the bound. Defense-in-depth is already present: forced NUL termination at `sizeof(input) - 1`, length measurement via `strnlen` (never `strlen`), explicit detection of truncated/oversized input with stream draining, and rejection rather than silent truncation.

Additional strengths worth noting: `octal_to_decimal` validates all pointers, rejects empty and unterminated strings, restricts input to `'0'`–`'7'`, and performs a pre-multiplication overflow check against `LONG_MAX`, so no signed overflow (UB) can occur. The `fflush(stdout)` return value is checked, and all error paths report to `stderr` with a non-zero exit status. The code is effectively CERT-C compliant as-is.
