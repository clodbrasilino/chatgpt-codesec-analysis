```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

#define INPUT_BUFFER_SIZE 64
#define DISCARD_GUARD_LIMIT 4096

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < (size_t)DISCARD_GUARD_LIMIT);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = strnlen(buffer, size);
    if (len > 0 && buffer[len - 1] != '\n' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    return 0;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and fixes applied:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20):** The original `discard_remaining_input` was already bounded, but the guard limit of `INT_MAX` was excessively large, meaning a malicious or malformed input stream could force up to ~2 billion iterations. I replaced it with a sane `DISCARD_GUARD_LIMIT` of 4096 iterations, which is more than sufficient to drain a line of leftover input while preventing a denial-of-service via an unbounded stream. The loop still terminates correctly on `'\n'` and `EOF`, and `getchar` never writes into a buffer, so no overflow is possible.

2. **Flawfinder `char buffer[64]` warning (CWE-119/CWE-120):** The statically-sized array is safe because every access is bounds-checked: `read_line` receives `sizeof(buffer)` explicitly, `fgets` limits the read to `size` bytes, the buffer is forcibly NUL-terminated at `size - 1`, and `strnlen` is used instead of `strlen` to avoid reading past the buffer. I additionally tightened the precondition to `size >= 2`, since a size of 1 would make the truncation-detection logic (`len == size - 1`) misbehave and leave no room for actual input.

3. **Other defensive properties preserved:** `parse_int` correctly handles `errno`/`ERANGE`, empty parses, trailing garbage, and `long`-to-`int` range narrowing. `closest_smaller` guards against signed integer underflow at `INT_MIN` (CWE-191). All return values, including `printf`, are checked.
