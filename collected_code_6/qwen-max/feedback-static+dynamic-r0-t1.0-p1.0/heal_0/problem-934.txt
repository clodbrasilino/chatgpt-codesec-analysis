#include <stdio.h>
#include <stdlib.h>

unsigned long delannoy_number(int n, int m) {
    unsigned long dp[n+1][m+1];
    for (int i = 0; i <= n; ++i) {
        for (int j = 0; j <= m; ++j) {
            if (i == 0 || j == 0)
                dp[i][j] = 1;
            else
                dp[i][j] = dp[i-1][j] + dp[i][j-1] + dp[i-1][j-1];
        }
    }
    return dp[n][m];
}

int main() {
    int n, m;
    printf("Enter the values of n and m: ");
    if (scanf("%d %d", &n, &m) != 2) {
        fprintf(stderr, "Invalid input. Please enter two integers.\n");
        return EXIT_FAILURE;
    }
    if (n < 0 || m < 0) {
        fprintf(stderr, "Values must be non-negative integers.\n");
        return EXIT_FAILURE;
    }
    unsigned long result = delannoy_number(n, m);
    printf("The Delannoy number D(%d, %d) is: %lu\n", n, m, result);
    return EXIT_SUCCESS;
}
