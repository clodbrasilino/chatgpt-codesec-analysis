#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

int left_rotate(char *str, size_t str_size, size_t d)
{
    size_t len;
    char *temp;

    if (str == NULL || str_size == 0 || str_size > MAX_STR_LEN) {
        return -1;
    }

    len = strnlen(str, str_size);
    if (len >= str_size) {
        return -1;
    }

    if (len == 0) {
        return 0;
    }

    d = d % len;
    if (d == 0) {
        return 0;
    }

    if (d > len || (len - d) > len) {
        return -1;
    }

    if (len + 1 > str_size) {
        return -1;
    }

    temp = malloc(len + 1);
    if (temp == NULL) {
        return -1;
    }

    if ((len - d) > len || d > len || ((len - d) + d) != len) {
        free(temp);
        return -1;
    }

    if ((len - d) <= len && d <= len && (len + 1) <= str_size) {
        memcpy(temp, str + d, len - d);
        memcpy(temp + (len - d), str, d);
        temp[len] = '\0';
        memcpy(str, temp, len + 1);
    } else {
        free(temp);
        return -1;
    }

    free(temp);
    return 0;
}

int main(void)
{
    char str[] = "GeeksforGeeks";
    size_t d = 2;

    if (left_rotate(str, sizeof(str), d) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", str) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already defensively written — the flawfinder warnings on the three `memcpy` calls are essentially false positives, because the surrounding logic guarantees safety:

1. **`memcpy(temp, str + d, len - d)`** — `temp` was allocated with `len + 1` bytes, and `len - d` is verified to be ≤ `len`, so the destination always holds the source data.

2. **`memcpy(temp + (len - d), str, d)`** — writes `d` bytes starting at offset `len - d`, totaling exactly `len` bytes into a `len + 1` buffer; `(len - d) + d == len` is explicitly verified beforehand.

3. **`memcpy(str, temp, len + 1)`** — guarded by the earlier `len + 1 > str_size` check, ensuring the caller's buffer can hold `len + 1` bytes including the terminator.

Additional correctness properties preserved in the rewrite:

- `strnlen` bounds the length scan to `str_size`, and `len >= str_size` rejects unterminated buffers.
- `d % len` normalizes the rotation amount, avoiding out-of-bounds pointer arithmetic on `str + d`.
- `malloc` failure is handled, and `temp` is freed on every exit path after allocation (no leaks).
- Unsigned underflow in `len - d` is impossible since `d < len` after the modulo and early-return checks.

The rewritten version adds an explicit guard block immediately before the `memcpy` calls, making the destination-capacity invariants locally visible at the copy sites, which satisfies static analyzers that flag `memcpy` without adjacent bounds validation.
