#include <stdio.h>
#include <stdlib.h>

long long max_path_sum(int **triangle, int rows) {
    if (triangle == NULL || rows <= 0) {
        return 0;
    }
    for (int i = 0; i < rows; ++i) {
        if (triangle[i] == NULL) {
            return 0;
        }
    }
    long long *dp = malloc(rows * sizeof(long long));
    if (dp == NULL) {
        return 0;
    }
    for (int i = 0; i < rows; ++i) {
        dp[i] = triangle[rows - 1][i];
    }
    for (int i = rows - 2; i >= 0; --i) {
        for (int j = 0; j <= i; ++j) {
            if (dp[j] > dp[j + 1]) {
                dp[j] = triangle[i][j] + dp[j];
            } else {
                dp[j] = triangle[i][j] + dp[j + 1];
            }
        }
    }
    long long result = dp[0];
    free(dp);
    return result;
}

int main(void) {
    int rows = 3;
    int **triangle = malloc(rows * sizeof(int *));
    if (triangle == NULL) {
        return 1;
    }
    for (int i = 0; i < rows; ++i) {
        triangle[i] = NULL;
    }
    triangle[0] = malloc(1 * sizeof(int));
    triangle[1] = malloc(2 * sizeof(int));
    triangle[2] = malloc(3 * sizeof(int));
    if (triangle[0] == NULL || triangle[1] == NULL || triangle[2] == NULL) {
        for (int i = 0; i < rows; ++i) {
            free(triangle[i]);
        }
        free(triangle);
        return 1;
    }
    triangle[0][0] = 1;
    triangle[1][0] = 2;
    triangle[1][1] = 3;
    triangle[2][0] = 1;
    triangle[2][1] = 5;
    triangle[2][2] = 1;
    long long result = max_path_sum(triangle, rows);
    printf("%lld\n", result);
    for (int i = 0; i < rows; ++i) {
        free(triangle[i]);
    }
    free(triangle);
    return 0;
}
