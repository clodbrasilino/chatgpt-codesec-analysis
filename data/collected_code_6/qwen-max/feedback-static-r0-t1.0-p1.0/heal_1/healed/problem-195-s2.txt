#include <stdio.h>
#include <stdlib.h>

int find_first_position(const int *arr, int n, int x) {
    int low = 0, high = n - 1;
    int result = -1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        if (arr[mid] == x) {
            result = mid;
            high = mid - 1;
        } else if (arr[mid] < x) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    return result;
}

int main() {
    int arr[] = {1, 2, 4, 4, 5, 6, 6};
    int n = sizeof(arr) / sizeof(arr[0]);
    int x = 4;
    int index = find_first_position(arr, n, x);
    if (index != -1) {
        printf("First position of %d is at index %d\n", x, index);
    } else {
        printf("%d not found in the array\n", x);
    }
    return 0;
}
