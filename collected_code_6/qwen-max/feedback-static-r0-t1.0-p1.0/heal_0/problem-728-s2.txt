#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

void free_list(Node* head) {
    Node* temp;
    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

int sum_lists(Node* head1, Node* head2) {
    int sum = 0;
    while (head1 != NULL || head2 != NULL) {
        if (head1 != NULL) {
            sum += head1->data;
            head1 = head1->next;
        }
        if (head2 != NULL) {
            sum += head2->data;
            head2 = head2->next;
        }
    }
    return sum;
}

int main() {
    Node* list1 = create_node(3);
    Node* list2 = create_node(4);
    Node* temp1 = list1;
    Node* temp2 = list2;

    for (int i = 5; i < 8; i++) {
        temp1->next = create_node(i);
        temp1 = temp1->next;
    }

    for (int i = 2; i < 6; i++) {
        temp2->next = create_node(i);
        temp2 = temp2->next;
    }

    int result = sum_lists(list1, list2);
    printf("Sum of elements: %d\n", result);

    free_list(list1);
    free_list(list2);

    return 0;
}
