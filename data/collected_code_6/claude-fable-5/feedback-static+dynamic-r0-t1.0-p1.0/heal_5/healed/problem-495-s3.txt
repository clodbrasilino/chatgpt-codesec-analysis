```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096
#define RESULT_BUF_SIZE 256

static int append_segment(char *output, size_t output_size, size_t *out_len,
                          const char *src, size_t seg_len)
{
    if (output == NULL || out_len == NULL || src == NULL) {
        return -1;
    }
    if (output_size == 0 || *out_len >= output_size) {
        return -1;
    }
    if (seg_len >= output_size - *out_len) {
        return -1;
    }
    memmove(output + *out_len, src, seg_len);
    *out_len += seg_len;
    return 0;
}

int remove_lowercase(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t out_len = 0;
    size_t input_len;
    const char *cursor;
    const char *end;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return -1;
    }

    if (input_len >= output_size) {
        return -1;
    }

    ret = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    end = input + input_len;

    while (cursor < end && *cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = (size_t)(end - cursor);
            if (append_segment(output, output_size, &out_len,
                               cursor, remaining) != 0) {
                regfree(&regex);
                return -1;
            }
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            return -1;
        }
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so ||
            (size_t)match.rm_eo > (size_t)(end - cursor)) {
            regfree(&regex);
            return -1;
        }
        if (match.rm_so > 0) {
            if (append_segment(output, output_size, &out_len,
                               cursor, (size_t)match.rm_so) != 0) {
                regfree(&regex);
                return -1;
            }
        }
        cursor += match.rm_eo;
    }

    if (out_len >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[out_len] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test_strings[] = {
        "HelloWorldFooBAR",
        "abcDEFghiJKL",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix3d C4s3 Str1ng"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *result = calloc(RESULT_BUF_SIZE, sizeof(char));
        if (result == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (remove_lowercase(test_strings[i], result, RESULT_BUF_SIZE) == 0) {
            printf("Input:  %s\n", test_strings[i]);
            printf("Output: %s\n\n", result);
        } else {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}
```

1. **Heap allocation with `calloc`** — The `result` buffer in `main` is now dynamically allocated and zero-initialized, replacing the statically-sized stack array flagged by Flawfinder, and is explicitly freed on every exit path to avoid leaks.

2. **Early capacity check** — `remove_lowercase` now verifies up front that `input_len < output_size`, guaranteeing the output (which can never exceed the input length) always fits, including the NUL terminator.

3. **Preserved existing defenses** — All original bounds checks remain: `strnlen` capping input at `MAX_INPUT_LEN`, per-segment overflow checks in `append_segment` (`seg_len >= output_size - *out_len` avoids integer overflow), validation of `regmatch_t` offsets against the remaining buffer, and a final `out_len < output_size` check before writing the terminator.

4. **Resource safety** — `regfree` is called on every error path, and the compiled regex never leaks.
