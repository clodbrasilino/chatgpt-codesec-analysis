```c
#include <stdio.h>
#include <stdlib.h>

int last_digit_of_factorial(int n)
{
    int digits[5] = {1, 1, 2, 6, 4};

    if (n < 0)
    {
        return -1;
    }

    if (n < 5)
    {
        return digits[n];
    }

    return 0;
}

int main(void)
{
    int number = 0;
    int result = 0;

    printf("Enter a non-negative integer: ");

    if (scanf("%d", &number) != 1)
    {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    result = last_digit_of_factorial(number);

    if (result < 0)
    {
        fprintf(stderr, "Error: number must be non-negative.\n");
        return EXIT_FAILURE;
    }

    printf("The last digit of %d! is %d\n", number, result);

    return EXIT_SUCCESS;
}
```

The key insight here is that computing the full factorial is unnecessary and would overflow quickly. For any `n >= 5`, the factorial contains both 2 and 5 as factors, making it divisible by 10, so the last digit is always 0. Only the values for 0 through 4 need to be handled explicitly, which avoids overflow entirely and keeps the function O(1).
