#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

bool are_all_characters_unique(const char *str);

bool are_all_characters_unique(const char *str) {
    if (str == NULL) {
        return false;
    }

    bool seen[UCHAR_MAX + 1] = { false };
    const unsigned char *ptr = (const unsigned char *)str;

    while (*ptr != '\0') {
        if (seen[*ptr]) {
            return false;
        }
        seen[*ptr] = true;
        ptr++;
    }

    return true;
}

int main(void) {
    const char *test_cases[] = {
        "abcdefg",
        "hello",
        NULL,
        "",
        "1234567890",
        "AaBbCc"
    };
    
    size_t num_cases = sizeof(test_cases) / sizeof(test_cases[0]);

    for (size_t i = 0; i < num_cases; i++) {
        bool result = are_all_characters_unique(test_cases[i]);
        
        if (test_cases[i] != NULL) {
            printf("\"%s\": %s\n", test_cases[i], result ? "true" : "false");
        } else {
            printf("NULL: %s\n", result ? "true" : "false");
        }
    }

    return EXIT_SUCCESS;
}
