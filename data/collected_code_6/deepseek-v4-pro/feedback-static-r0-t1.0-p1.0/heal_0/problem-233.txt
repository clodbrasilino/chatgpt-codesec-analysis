```c
#include <stdio.h>
#include <math.h>

#define PI 3.14159265358979323846

double lateral_surface_area(double radius, double height) {
    if (radius < 0 || height < 0) {
        return -1.0;
    }
    return 2.0 * PI * radius * height;
}

int main(void) {
    double radius = 0.0;
    double height = 0.0;
    double area = 0.0;

    printf("Enter radius: ");
    if (scanf("%lf", &radius) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    printf("Enter height: ");
    if (scanf("%lf", &height) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    area = lateral_surface_area(radius, height);
    if (area < 0) {
        printf("Radius and height must be non-negative.\n");
        return 1;
    }

    printf("Lateral surface area: %.2f\n", area);
    return 0;
}
```
