```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_SIZE 1024

static int min_flips_to_alternate(const char *s, size_t max_len)
{
    size_t len;
    size_t i;
    int flips_start_zero;
    int flips_start_one;
    char expected_zero;
    char expected_one;

    if (s == NULL) {
        return -1;
    }

    len = strnlen(s, max_len);
    if (len == max_len) {
        return -1;
    }
    if (len == 0) {
        return 0;
    }

    flips_start_zero = 0;
    flips_start_one = 0;

    for (i = 0; i < len; i++) {
        if (s[i] != '0' && s[i] != '1') {
            return -1;
        }

        if ((i % 2) == 0) {
            expected_zero = '0';
            expected_one = '1';
        } else {
            expected_zero = '1';
            expected_one = '0';
        }

        if (s[i] != expected_zero) {
            flips_start_zero++;
        }
        if (s[i] != expected_one) {
            flips_start_one++;
        }
    }

    return (flips_start_zero < flips_start_one) ? flips_start_zero : flips_start_one;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    int result;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = min_flips_to_alternate(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid binary string\n");
        return EXIT_FAILURE;
    }

    printf("%d\n", result);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced both `strlen` calls with `strnlen`, bounding the scan to the buffer size so a non-terminated string can never cause an over-read. In `min_flips_to_alternate`, the caller now passes an explicit maximum length, and the function rejects input if no terminator is found within that bound.

2. **CWE-119/CWE-120 (statically-sized buffer):** The buffer is zero-initialized with `memset`, `fgets` remains bounded by `sizeof(buffer)`, and an explicit `buffer[sizeof(buffer) - 1] = '\0'` guarantees null-termination as defense-in-depth before any string operation runs.

3. **Defensive API design:** `min_flips_to_alternate` no longer trusts its input to be terminated — the size contract is now part of its signature, which eliminates the original implicit assumption that made the Flawfinder finding valid.
