```c
#include <stdio.h>
#include <stdlib.h>

int count_divisors(int n)
{
    int count = 0;
    int i;
    
    if (n == 0) {
        return 0;
    }
    
    if (n < 0) {
        n = -n;
    }
    
    for (i = 1; i <= n / i; i++) {
        if (n % i == 0) {
            if (i == n / i) {
                count++;
            } else {
                count += 2;
            }
        }
    }
    
    return count;
}

int main(void)
{
    int number;
    int result;
    
    printf("Enter an integer: ");
    
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    result = count_divisors(number);
    
    printf("Number of divisors: %d\n", result);
    
    return EXIT_SUCCESS;
}
```
