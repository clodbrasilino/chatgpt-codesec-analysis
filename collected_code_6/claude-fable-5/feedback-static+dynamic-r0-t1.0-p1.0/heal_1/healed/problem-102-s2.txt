```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    const char *input = "hello_world_example_string";
    char *result;

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\n", input) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (printf("Output: %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Test case failures (main functional bug):** The failing test cases (`pythonProgram` vs `PythonProgram`, etc.) show the expected output is **PascalCase**, not lowerCamelCase. The fix is initializing `capitalize_next = 1` instead of `0`, so the first character is also capitalized.

2. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, which prevents unbounded over-reads on non-null-terminated input. If the string reaches the limit, the function safely rejects it by returning `NULL` rather than processing potentially truncated/invalid data.

The rest of the code was already sound: `malloc` result is checked, `toupper` receives an `unsigned char` cast (avoiding UB with negative `char` values), the output buffer cannot overflow since `j <= i < len` and the buffer holds `len + 1` bytes, and `result` is freed on all exit paths in `main`.
