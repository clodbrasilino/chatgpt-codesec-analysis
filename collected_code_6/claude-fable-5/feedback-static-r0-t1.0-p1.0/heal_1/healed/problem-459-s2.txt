```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_uppercase(const char *input);

char *remove_uppercase(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[A-Z]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_len - (size_t)(cursor - input));
            if (result_len + remaining > input_len) {
                regfree(&regex);
                free(result);
                return NULL;
            }
            memcpy(result + result_len, cursor, remaining);
            result_len += remaining;
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        if (result_len + (size_t)match.rm_so > input_len) {
            regfree(&regex);
            free(result);
            return NULL;
        }
        memcpy(result + result_len, cursor, (size_t)match.rm_so);
        result_len += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "helloWORLDthisIS a TEST string",
        "ALLUPPERCASE",
        "nolowercasehere",
        "MixedCASEString"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_uppercase(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: \"%s\" -> Result: \"%s\"\n", test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Summary of applied fixes:**

1. **CWE-126 (`strlen` over-read):** Both `strlen` calls were replaced with `strnlen` bounded by `MAX_INPUT_LEN` (for the initial input) and by the remaining known input length (inside the loop), so a non-terminated string can no longer cause an unbounded read. Inputs at or beyond the cap are rejected.

2. **CWE-120 (`memcpy` overflow):** Every `memcpy` is now preceded by an explicit bounds check verifying `result_len + copy_length <= input_len`, guaranteeing the destination buffer (of size `input_len + 1`) can hold the data before copying.

3. **Defensive validation of regex offsets:** Added a sanity check that `match.rm_so >= 0` and `match.rm_eo >= match.rm_so` before casting to `size_t`, preventing a negative-to-huge-unsigned conversion that could corrupt the copy length or cursor arithmetic.

All error paths correctly release both the compiled regex and the allocated buffer, so no leaks are introduced by the added checks.
