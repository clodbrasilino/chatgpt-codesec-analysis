```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    char name[50];
} NestedRecord;

typedef struct {
    int id;
    char data[50];
    NestedRecord* nested;
} Tuple;

void remove_nested_record(Tuple* tuple) {
    if (tuple != NULL && tuple->nested != NULL) {
        free(tuple->nested);
        tuple->nested = NULL;
    }
}

int main(void) {
    Tuple tuple;
    tuple.id = 1;
    
    snprintf(tuple.data, sizeof(tuple.data), "%s", "Tuple Data");
    
    tuple.nested = (NestedRecord*)malloc(sizeof(NestedRecord));
    if (tuple.nested == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    
    tuple.nested->id = 2;
    snprintf(tuple.nested->name, sizeof(tuple.nested->name), "%s", "Nested Data");
    
    remove_nested_record(&tuple);
    
    return 0;
}
```
