#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define R 3
#define C 3

int min(int x, int y, int z) {
    if (x < y) {
        return (x < z) ? x : z;
    } else {
        return (y < z) ? y : z;
    }
}

int minCost(int cost[R][C], int m, int n) {
    int minCost[R][C];
    int i, j;

    minCost[0][0] = cost[0][0];

    for (i = 1; i <= m; i++) {
        minCost[i][0] = minCost[i-1][0] + cost[i][0];
    }

    for (j = 1; j <= n; j++) {
        minCost[0][j] = minCost[0][j-1] + cost[0][j];
    }

    for (i = 1; i <= m; i++) {
        for (j = 1; j <= n; j++) {
            minCost[i][j] = min(minCost[i-1][j-1], minCost[i-1][j], minCost[i][j-1]) + cost[i][j];
        }
    }

    return minCost[m][n];
}

int main() {
    int cost[R][C] = { {1, 2, 3}, {4, 8, 2}, {1, 5, 3} };
    printf("The minimum cost from (0, 0) to (2, 2) is: %d\n", minCost(cost, 2, 2));
    return 0;
}
