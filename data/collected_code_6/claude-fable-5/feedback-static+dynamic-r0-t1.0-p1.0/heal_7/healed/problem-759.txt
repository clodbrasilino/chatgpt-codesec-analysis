```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 128
#define MAX_FLUSH_ITERATIONS 4096

bool is_decimal_precision_two(const char *str, size_t max_len)
{
    size_t len;
    size_t i = 0;
    size_t digits_before = 0;
    size_t digits_after = 0;
    bool dot_found = false;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    if (str[i] == '+' || str[i] == '-') {
        i++;
        if (i == len) {
            return false;
        }
    }

    for (; i < len; i++) {
        if (isdigit((unsigned char)str[i])) {
            if (dot_found) {
                digits_after++;
            } else {
                digits_before++;
            }
        } else if (str[i] == '.') {
            if (dot_found) {
                return false;
            }
            dot_found = true;
        } else {
            return false;
        }
    }

    if (digits_before == 0) {
        return false;
    }

    if (dot_found && (digits_after == 0 || digits_after > 2)) {
        return false;
    }

    return true;
}

static void flush_stdin(void)
{
    int ch;
    size_t iterations = 0;

    do {
        ch = getchar();
        iterations++;
    } while (ch != '\n' && ch != EOF && iterations < MAX_FLUSH_ITERATIONS);
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Output error\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Input error\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    if (is_decimal_precision_two(buffer, sizeof(buffer))) {
        printf("Valid decimal with precision of 2: %s\n", buffer);
    } else {
        printf("Invalid decimal with precision of 2: %s\n", buffer);
    }

    return 0;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `getchar` warning (CWE-120, CWE-20) in `flush_stdin`:**
The concern is an unbounded read loop. The code already mitigates this properly — no data is stored in a buffer (so no overflow is possible), and the loop is bounded by `MAX_FLUSH_ITERATIONS`, terminating on `'\n'`, `EOF`, or after 4096 iterations. This prevents an infinite loop on a stream that never delivers a newline or EOF.

**2. Flawfinder `char buffer[MAX_INPUT_LEN]` warning (CWE-119/CWE-120):**
The statically-sized buffer is used safely:
- Input is read with `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces the bound and always NUL-terminates on success.
- A defensive `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination regardless.
- Length is computed with `strnlen(..., sizeof(buffer))` rather than `strlen`, so it can never scan past the buffer.
- When the input line exceeds the buffer (no newline found and `len == sizeof(buffer) - 1`), the remainder of stdin is drained via `flush_stdin()`, preventing stale data from leaking into subsequent reads.

**3. Validation logic (`is_decimal_precision_two`):**
Defensively written — NULL check, bounded `strnlen`, rejection of non-terminated input (`len == max_len`), correct `isdigit` usage with an `unsigned char` cast (avoiding UB for negative `char` values), and strict format checking (single optional sign, at least one integer digit, at most one dot, 1–2 fractional digits).
