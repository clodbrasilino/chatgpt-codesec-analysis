static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size)
{
    size_t index;
    size_t start;
    size_t end;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    index = 0;

    if (decimal < 0) {
        buffer[index] = '-';
        index++;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    start = index;

    if (value == 0UL) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index] = '0';
        index++;
    } else {
        while (value > 0UL) {
            if (index + 1 >= buffer_size) {
                return -1;
            }
            buffer[index] = (char)('0' + (char)(value % 2UL));
            index++;
            value /= 2UL;
        }
    }

    if (index >= buffer_size) {
        return -1;
    }

    buffer[index] = '\0';

    end = index - 1;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    long decimal = 0;
