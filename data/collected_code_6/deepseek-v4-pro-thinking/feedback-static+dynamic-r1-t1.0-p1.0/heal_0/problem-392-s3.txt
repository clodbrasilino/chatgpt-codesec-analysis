#include <stdio.h>
#include <stdlib.h>

struct MemoEntry {
    unsigned long long key;
    unsigned long long value;
    int used;
};

struct Memo {
    struct MemoEntry *entries;
    size_t capacity;
    size_t count;
};

static size_t hash_key(unsigned long long key) {
    key ^= key >> 33;
    key *= 0xff51afd7ed558ccdULL;
    key ^= key >> 33;
    key *= 0xc4ceb9fe1a85ec53ULL;
    key ^= key >> 33;
    return (size_t)key;
}

static int memo_init(struct Memo *memo) {
    memo->capacity = 64;
    memo->count = 0;
    memo->entries = calloc(memo->capacity, sizeof(*memo->entries));
    if (!memo->entries) {
        memo->capacity = 0;
        return 0;
    }
    return 1;
}

static void memo_free(struct Memo *memo) {
    free(memo->entries);
    memo->entries = NULL;
    memo->capacity = 0;
    memo->count = 0;
}

static int memo_resize(struct Memo *memo) {
    if (memo->capacity > ((size_t)-1) / 2) {
        return 0;
    }
    size_t new_capacity = memo->capacity * 2;
    if (new_capacity > ((size_t)-1) / sizeof(*memo->entries)) {
        return 0;
    }
    struct MemoEntry *new_entries = calloc(new_capacity, sizeof(*new_entries));
    if (!new_entries) {
        return 0;
    }
    size_t mask = new_capacity - 1;
    for (size_t i = 0; i < memo->capacity; ++i) {
        if (!memo->entries[i].used) {
            continue;
        }
        size_t idx = hash_key(memo->entries[i].key) & mask;
        while (new_entries[idx].used) {
            idx = (idx + 1) & mask;
        }
        new_entries[idx] = memo->entries[i];
    }
    free(memo->entries);
    memo->entries = new_entries;
    memo->capacity = new_capacity;
    return 1;
}

static int memo_get(const struct Memo *memo, unsigned long long key, unsigned long long *value) {
    if (memo->capacity == 0 || memo->count == 0) {
        return 0;
    }
    size_t mask = memo->capacity - 1;
    size_t idx = hash_key(key) & mask;
    while (memo->entries[idx].used) {
        if (memo->entries[idx].key == key) {
            *value = memo->entries[idx].value;
            return 1;
        }
        idx = (idx + 1) & mask;
    }
    return 0;
}

static int memo_put(struct Memo *memo, unsigned long long key, unsigned long long value) {
    if (memo->count >= (memo->capacity >> 1) + (memo->capacity >> 2)) {
        if (!memo_resize(memo)) {
            return 0;
        }
    }
    size_t mask = memo->capacity - 1;
    size_t idx = hash_key(key) & mask;
    while (memo->entries[idx].used) {
        if (memo->entries[idx].key == key) {
            memo->entries[idx].value = value;
            return 1;
        }
        idx = (idx + 1) & mask;
    }
    memo->entries[idx].key = key;
    memo->entries[idx].value = value;
    memo->entries[idx].used = 1;
    ++memo->count;
    return 1;
}

static unsigned long long f_rec(unsigned long long n, struct Memo *memo) {
    if (n <= 1) {
        return n;
    }
    unsigned long long cached;
    if (memo_get(memo, n, &cached)) {
        return cached;
    }
    unsigned long long split = f_rec(n / 2, memo) +
                               f_rec(n / 3, memo) +
                               f_rec(n / 4, memo) +
                               f_rec(n / 5, memo);
    unsigned long long result = split > n ? split : n;
    if (!memo_put(memo, n, result)) {
        return result;
    }
    return result;
}

unsigned long long f(unsigned long long n) {
    struct Memo memo;
    if (!memo_init(&memo)) {
        exit(EXIT_FAILURE);
    }
    unsigned long long result = f_rec(n, &memo);
    memo_free(&memo);
    return result;
}

int main(void) {
    unsigned long long n;
    while (scanf("%llu", &n) == 1) {
        printf("%llu\n", f(n));
    }
    return 0;
}
