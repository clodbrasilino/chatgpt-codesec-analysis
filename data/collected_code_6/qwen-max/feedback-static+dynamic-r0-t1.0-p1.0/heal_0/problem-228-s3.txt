```c
#include <stdio.h>
#include <stdint.h>

int check_bits_unset(uint32_t num, int l, int r) {
    uint32_t mask = ((1 << (r - l + 1)) - 1) << l;
    return (num & mask) == 0;
}

int main() {
    uint32_t num = 0b10101010101010101010101010101010;
    int l = 6, r = 15;
    if (check_bits_unset(num, l, r)) {
        printf("All bits in the given range are unset.\n");
    } else {
        printf("Not all bits in the given range are unset.\n");
    }
    return 0;
}
```
