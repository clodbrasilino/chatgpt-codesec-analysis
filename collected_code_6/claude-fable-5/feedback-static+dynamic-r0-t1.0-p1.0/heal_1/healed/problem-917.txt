```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            {
                size_t match_len = j - start;

                if (match_len >= sizeof(matches[found].text)) {
                    match_len = sizeof(matches[found].text) - 1;
                }

                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Summary of fixes applied:**

1. **`strlen` over-read (CWE-126):** Replaced both `strlen` calls with `strnlen`, bounding the scan to the buffer size so a non-terminated string cannot cause an over-read. Additionally, `input[sizeof(input) - 1] = '\0'` guarantees termination defensively after `fgets` (which already terminates, but this makes the invariant explicit).

2. **`memcpy` overflow (CWE-120):** The bounds check now uses `sizeof(matches[found].text)` directly rather than the `MAX_INPUT` macro, so the clamp remains correct even if the struct definition changes independently of the macro. The copy can never exceed the destination capacity, and space for the null terminator is always reserved.

3. **Statically-sized arrays (CWE-119/CWE-120):** All writes into `input` and `Match::text` are bounded — `fgets` receives `sizeof(input)`, and the `memcpy` length is clamped against the destination's actual size, satisfying the bounds-checking requirement.

4. **Variable scope (`variableScope` warning):** The loop counter `i` in `main` is now declared inside the `for` statement, reducing its scope to the minimum needed (C99+).

The code preserves the original behavior: it reads a line, strips the trailing newline, extracts sequences of an uppercase letter followed by one or more lowercase letters, and prints them.
