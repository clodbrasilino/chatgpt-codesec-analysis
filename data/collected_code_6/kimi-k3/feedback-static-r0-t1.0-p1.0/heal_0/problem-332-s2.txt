#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define CHAR_SET_SIZE 256

typedef struct {
    unsigned char character;
    size_t count;
} CharFrequency;

int compare_frequencies(const void *a, const void *b)
{
    const CharFrequency *fa = (const CharFrequency *)a;
    const CharFrequency *fb = (const CharFrequency *)b;
    
    if (fa->count < fb->count) {
        return 1;
    }
    if (fa->count > fb->count) {
        return -1;
    }
    return (int)fa->character - (int)fb->character;
}

CharFrequency *count_character_frequency(const char *str, size_t *unique_count)
{
    size_t counts[CHAR_SET_SIZE] = {0};
    CharFrequency *result = NULL;
    size_t i;
    size_t unique = 0;
    
    if (str == NULL || unique_count == NULL) {
        return NULL;
    }
    
    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (counts[c] == 0) {
            unique++;
        }
        if (counts[c] < SIZE_MAX) {
            counts[c]++;
        }
    }
    
    if (unique == 0) {
        *unique_count = 0;
        return NULL;
    }
    
    result = (CharFrequency *)malloc(unique * sizeof(CharFrequency));
    if (result == NULL) {
        return NULL;
    }
    
    size_t idx = 0;
    for (i = 0; i < CHAR_SET_SIZE; i++) {
        if (counts[i] > 0) {
            result[idx].character = (unsigned char)i;
            result[idx].count = counts[i];
            idx++;
        }
    }
    
    *unique_count = unique;
    return result;
}

void print_frequencies(const CharFrequency *freqs, size_t count)
{
    size_t i;
    
    if (freqs == NULL || count == 0) {
        return;
    }
    
    for (i = 0; i < count; i++) {
        if (freqs[i].character >= 32 && freqs[i].character <= 126) {
            printf("'%c': %zu\n", freqs[i].character, freqs[i].count);
        } else {
            printf("'\\x%02X': %zu\n", freqs[i].character, freqs[i].count);
        }
    }
}

int main(int argc, char *argv[])
{
    const char *input = "Hello, World!";
    CharFrequency *frequencies = NULL;
    size_t count = 0;
    
    if (argc > 1) {
        input = argv[1];
    }
    
    frequencies = count_character_frequency(input, &count);
    
    if (frequencies == NULL && count > 0) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    if (count > 0) {
        qsort(frequencies, count, sizeof(CharFrequency), compare_frequencies);
        printf("Character frequencies for: \"%s\"\n", input);
        print_frequencies(frequencies, count);
        free(frequencies);
    } else {
        printf("Empty string provided.\n");
    }
    
    return EXIT_SUCCESS;
}
