#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

char* find_first_repeated_word(const char* str) {
    const int MAX_WORDS = 100;
    const int MAX_WORD_LENGTH = 50;
    char words[MAX_WORDS][MAX_WORD_LENGTH] = {0};
    int word_count = 0;

    char* mutable_str = strdup(str);
    if (mutable_str == NULL) return NULL;
    char* token = strtok(mutable_str, " ");
    while (token && word_count < MAX_WORDS) {
        bool found = false;
        for (int i = 0; i < word_count; i++) {
            if (strcmp(words[i], token) == 0) {
                free(mutable_str);
                return strdup(token);
            }
        }
        strcpy(words[word_count++], token);
        token = strtok(NULL, " ");
    }

    free(mutable_str);
    return NULL;
}

int main() {
    const char* input0 = "a b a";
    const char* input1 = "c a a c";
    const char* input2 = "";
    const char* input3 = "This is a test string with test words";

    char* result0 = find_first_repeated_word(input0);
    if (result0 != NULL) {
        printf("First repeated word: %s\n", result0);
        free(result0);
    } else {
        printf("None\n");
    }

    char* result1 = find_first_repeated_word(input1);
