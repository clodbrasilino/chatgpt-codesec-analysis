```c
#include <stdio.h>
#include <stdlib.h>

long long calculate_product(const int *arr, size_t size, int *error) {
    if (arr == NULL || error == NULL) {
        if (error != NULL) {
            *error = -1;
        }
        return 0;
    }
    
    if (size == 0) {
        *error = -2;
        return 0;
    }
    
    long long product = 1;
    for (size_t i = 0; i < size; i++) {
        product *= arr[i];
    }
    
    *error = 0;
    return product;
}

int main(void) {
    int numbers[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);
    int error = 0;
    
    long long result = calculate_product(numbers, size, &error);
    
    if (error == 0) {
        printf("Product: %lld\n", result);
    } else {
        fprintf(stderr, "Error calculating product (code: %d)\n", error);
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
```
