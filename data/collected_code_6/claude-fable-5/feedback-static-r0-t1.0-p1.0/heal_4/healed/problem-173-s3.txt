```c
#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

void remove_non_alnum(char *str)
{
    size_t read_idx;
    size_t write_idx;

    if (str == NULL) {
        return;
    }

    read_idx = 0;
    write_idx = 0;

    while (str[read_idx] != '\0') {
        if (isalnum((unsigned char)str[read_idx])) {
            str[write_idx] = str[read_idx];
            write_idx++;
        }
        read_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    static const char input[] = "Hello, World! 123 #$% foo_bar";
    size_t input_len;
    char *buffer;
    int written;

    input_len = strnlen(input, sizeof(input));
    if (input_len >= sizeof(input)) {
        fprintf(stderr, "Input string is not null-terminated\n");
        return EXIT_FAILURE;
    }

    buffer = malloc(input_len + 1);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, input_len + 1, "%s", input);
    if (written < 0 || (size_t)written > input_len) {
        fprintf(stderr, "Input too long or encoding error\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Before: %s\n", buffer);

    remove_non_alnum(buffer);

    printf("After:  %s\n", buffer);

    free(buffer);

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weakness:**

1. **CWE-126 (buffer over-read via `strlen`)**: Replaced `strlen()` with `strnlen()`, which takes a maximum bound and cannot read past the buffer even if the string were not null-terminated.

2. **Bounded input declaration**: Changed `const char *input` to `static const char input[]` so that `sizeof(input)` yields the actual array size at compile time, giving `strnlen()` a legitimate upper bound to enforce.

3. **Termination check**: Added an explicit check that `input_len < sizeof(input)`, which detects the (theoretical) case where no null terminator exists within the buffer and fails safely instead of over-reading.

The remaining defensive measures were already correct in the original: `isalnum()` is called with an `unsigned char` cast to avoid undefined behavior on negative `char` values, `snprintf()` is bounds-checked and its return value validated, `malloc()` failure is handled, and the buffer is freed on all exit paths.
