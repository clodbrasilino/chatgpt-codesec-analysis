```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_non_alnum(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len;
    size_t result_capacity;
    size_t input_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result_capacity = input_len + 1;
    result = malloc(result_capacity);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        {
            size_t match_len = (size_t)(match.rm_eo - match.rm_so);
            size_t offset = (size_t)(cursor - input);
            size_t remaining;

            if (offset > input_len ||
                (size_t)match.rm_so > input_len - offset ||
                (size_t)match.rm_eo > input_len - offset ||
                match_len > input_len - offset) {
                free(result);
                regfree(&regex);
                return NULL;
            }

            if (result_len >= result_capacity) {
                free(result);
                regfree(&regex);
                return NULL;
            }

            remaining = result_capacity - 1 - result_len;
            if (match_len > remaining) {
                free(result);
                regfree(&regex);
                return NULL;
            }

            if (match_len > 0) {
                memcpy(result + result_len, cursor + match.rm_so, match_len);
                result_len += match_len;
            }
        }
        if (match.rm_eo == 0) {
            cursor += 1;
        } else {
            cursor += match.rm_eo;
        }
    }

    if (result_len >= result_capacity) {
        free(result);
        regfree(&regex);
        return NULL;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a-b_c=d+e",
        "!!!@@@###",
        "NoSpecialChars2024"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alnum(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: %s\nCleaned:  %s\n\n",
                   test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended against the Flawfinder-flagged `memcpy` issue (CWE-120). The warning is a heuristic hit on the function name, not an actual vulnerability, because:

1. **Destination sizing**: `result` is allocated with `input_len + 1` bytes, and the total copied alphanumeric characters can never exceed `input_len`.
2. **Explicit bounds checks**: Before every `memcpy`, the code verifies `match_len > result_capacity - 1 - result_len` (guarded against underflow by first checking `result_len >= result_capacity`), guaranteeing the copy fits within the allocation.
3. **Input validation**: `strnlen` with `MAX_INPUT_LEN` caps input size, `regmatch_t` offsets are validated for negative/inverted values, and match offsets are checked against remaining input length before use.
4. **No infinite loop**: The `match.rm_eo == 0` guard ensures forward progress even on a degenerate zero-length match.

In the rewrite I restructured the pre-copy checks into separate, ordered guards (overflow-safe `remaining` calculation before comparison) to make the safety invariant explicit and trivially auditable, which is the standard way to satisfy static analyzers on this pattern. Memory and regex resources are released on every exit path, so there are no leaks even on failure.
