#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static int remove_multiple_spaces(const char *input, char *output, size_t output_size);

static int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    size_t remaining;
    size_t input_len;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    input_len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (input_len > MAX_INPUT_LEN) {
        return -1;
    }

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len;
        size_t space_left;

        if ((match.rm_so < 0) || (match.rm_eo <= match.rm_so)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        prefix_len = (size_t)match.rm_so;

        if (out_pos >= (output_size - 1U)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        space_left = output_size - 1U - out_pos;

        if (prefix_len >= space_left) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        if ((out_pos + prefix_len + 1U) >= output_size) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        memcpy(&output[out_pos], cursor, prefix_len);
        out_pos += prefix_len;
        output[out_pos] = ' ';
        out_pos++;

        cursor += match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    remaining = strnlen(cursor, MAX_INPUT_LEN);

    if (remaining > (output_size - 1U - out_pos)) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    if ((out_pos + remaining) >= output_size) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    memcpy(&output[out_pos], cursor, remaining);
    out_pos += remaining;
    output[out_pos] = '\0';

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256] = {0};

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**Analysis notes:**

The two `memcpy` warnings flagged by Flawfinder (CWE-120) are false positives in this code — both copies are already guarded by explicit bounds checks:

1. **First `memcpy`** (`prefix_len` bytes): preceded by `prefix_len >= space_left` and `(out_pos + prefix_len + 1U) >= output_size` checks, guaranteeing room for the copied prefix plus the inserted space and the eventual NUL terminator.

2. **Second `memcpy`** (`remaining` bytes): guarded by `remaining > (output_size - 1U - out_pos)` and `(out_pos + remaining) >= output_size`, ensuring space for the tail plus the NUL terminator.

The **statically-sized array warning** (CWE-119/CWE-120) on `result[256]` is mitigated because `sizeof(result)` is passed explicitly to the function, which enforces all writes against `output_size` — no unbounded writes occur.

Other defensive measures already present and preserved:

- NULL-pointer and zero-size parameter validation.
- `strnlen` with a hard cap (`MAX_INPUT_LEN`) instead of `strlen`, preventing unbounded reads on non-terminated input.
- `regfree` called on every exit path after `regcomp` succeeds — no resource leaks.
- Output buffer is reset to an empty string on every failure path, so callers never see partial/stale data.
- Sanity check on `rm_so`/`rm_eo` guards against pathological regex results and guarantees loop progress (the pattern `"  +"` always matches ≥ 2 chars, so `cursor` strictly advances — no infinite loop).

The code is safe as written; no functional changes were required beyond removing the comments as requested.
