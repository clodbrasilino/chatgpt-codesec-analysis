```c
#include <stdio.h>
#include <stdint.h>
#include <limits.h>

int multiply(int a, int b) {
    int result = 0;
    int negative = 0;

    if (a == 0 || b == 0) {
        return 0;
    }

    if ((a < 0 && b > 0) || (a > 0 && b < 0)) {
        negative = 1;
    }

    unsigned int ua = (a < 0) ? (unsigned int)(-(a + 1)) + 1 : (unsigned int)a;
    unsigned int ub = (b < 0) ? (unsigned int)(-(b + 1)) + 1 : (unsigned int)b;

    while (ub > 0) {
        if (ub & 1U) {
            result += (int)ua;
        }
        ua <<= 1;
        ub >>= 1;
    }

    return negative ? -result : result;
}

int main(void) {
    int x = 7;
    int y = -6;
    printf("%d * %d = %d\n", x, y, multiply(x, y));

    x = -12;
    y = -12;
    printf("%d * %d = %d\n", x, y, multiply(x, y));

    x = 0;
    y = 42;
    printf("%d * %d = %d\n", x, y, multiply(x, y));

    x = INT_MAX;
    y = 1;
    printf("%d * %d = %d\n", x, y, multiply(x, y));

    return 0;
}
```
