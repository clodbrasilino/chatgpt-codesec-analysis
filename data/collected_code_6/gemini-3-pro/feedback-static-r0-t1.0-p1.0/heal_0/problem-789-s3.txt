```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <math.h>

double calculate_polygon_perimeter(int number_of_sides, double side_length) {
    if (number_of_sides < 3) {
        return -1.0;
    }
    if (side_length <= 0.0) {
        return -1.0;
    }
    
    return (double)number_of_sides * side_length;
}

int main(void) {
    int sides;
    double length;
    double perimeter;
    int result;

    printf("Enter the number of sides: ");
    result = scanf("%d", &sides);
    if (result != 1) {
        fprintf(stderr, "Error: Invalid input for number of sides.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the length of a side: ");
    result = scanf("%lf", &length);
    if (result != 1) {
        fprintf(stderr, "Error: Invalid input for side length.\n");
        return EXIT_FAILURE;
    }

    perimeter = calculate_polygon_perimeter(sides, length);

    if (perimeter < 0.0) {
        fprintf(stderr, "Error: A regular polygon must have at least 3 sides and a positive side length.\n");
        return EXIT_FAILURE;
    }

    printf("The perimeter of the regular polygon is: %f\n", perimeter);

    return EXIT_SUCCESS;
}
```
