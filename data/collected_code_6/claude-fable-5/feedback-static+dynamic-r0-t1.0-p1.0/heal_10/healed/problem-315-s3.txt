```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <ctype.h>

#define MAX_INPUT 1024

int find_first_max_even_word(const char *str, size_t str_max, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len = 0;
    size_t max_len = 0;
    size_t best_start = 0;
    int found = 0;

    if (str == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(str, str_max);

    while (i < len) {
        while (i < len && isspace((unsigned char)str[i])) {
            i++;
        }
        if (i >= len) {
            break;
        }
        {
            size_t start = i;
            size_t word_len = 0;

            while (i < len && !isspace((unsigned char)str[i])) {
                i++;
            }
            word_len = i - start;

            if ((word_len % 2u) == 0u && word_len > max_len) {
                max_len = word_len;
                best_start = start;
                found = 1;
            }
        }
    }

    if (found == 0) {
        return -1;
    }

    if (max_len >= result_size) {
        return -1;
    }

    if (best_start > len || max_len > len - best_start) {
        return -1;
    }

    memcpy(result, &str[best_start], max_len);
    result[max_len] = '\0';

    return (int)max_len;
}

int main(void)
{
    char input[MAX_INPUT];
    char result[MAX_INPUT];
    int ret = 0;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    ret = find_first_max_even_word(input, sizeof(input), result, sizeof(result));

    if (ret < 0) {
        printf("No even length word found\n");
    } else {
        printf("First maximum even length word: %s (length %d)\n", result, ret);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses flagged by Flawfinder were already mitigated in the original code; the rewrite preserves those mitigations:

1. **`memcpy` (CWE-120)** — Safe because the copy is guarded by `max_len >= result_size` (guaranteeing room for the data plus the null terminator) and by `best_start > len || max_len > len - best_start` (guaranteeing the source range is in bounds, written to avoid integer overflow).

2. **Statically-sized `char` arrays (CWE-119/CWE-120)** — Safe because:
   - `fgets` limits input to `sizeof(input)`, never overflowing the buffer.
   - Explicit null termination (`input[sizeof(input) - 1] = '\0'`) is defensive redundancy since `fgets` already terminates.
   - `strnlen` with `str_max` bounds the scan even if the string were not terminated.
   - `strcspn` safely strips the newline without risking out-of-bounds access.
   - Both buffers are zero-initialized with `memset` before use.

3. **General robustness** — NULL-pointer and zero-size parameter checks at function entry, `isspace` called with an `unsigned char` cast to avoid undefined behavior on negative `char` values, and unsigned arithmetic used consistently for lengths and indices.
