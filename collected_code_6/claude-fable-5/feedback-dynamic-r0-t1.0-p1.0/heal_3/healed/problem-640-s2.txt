```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *tests[] = {
        "Hello (remove me) World",
        "Nested (outer (inner) text) example",
        "No parentheses here"
    };
    size_t num_tests;
    size_t i;
    char *output;

    num_tests = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < num_tests; i++) {
        output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        printf("%s\n", output);
        free(output);
        output = NULL;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The core logic of the original `remove_parentheses` function was actually sound. The garbage output described in the comments (`�AX`, `` `AX ``) is the classic signature of reading **uninitialized or freed heap memory**, which typically stems from one of these weaknesses:

1. **Uninitialized buffer** — using `malloc` without null-terminating. Fixed here by using `calloc` (zero-initialized) *and* explicitly writing `result[j] = '\0'` — a belt-and-suspenders approach.
2. **Use-after-free** — printing `output` after `free()`. The rewrite prints *before* freeing and sets the pointer to `NULL` afterward to prevent dangling-pointer reuse.
3. **Missing allocation check** — a failed `calloc` returning `NULL` would cause undefined behavior when dereferenced; both the function and the caller now check for `NULL`.
4. **Buffer overflow safety** — the output can never exceed the input length, so `len + 1` guarantees room for the terminator; `j` is bounded by `i < len`.

Additionally, unmatched `)` characters are handled safely by the `depth > 0` guard, preventing negative depth from corrupting the filtering logic.
