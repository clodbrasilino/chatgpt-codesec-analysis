```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strlen(input);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            {
                size_t match_len = j - start;

                if (match_len >= MAX_INPUT) {
                    match_len = MAX_INPUT - 1;
                }

                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    len = strlen(input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

The only weakness flagged was the `variableScope` issue for `i` in `main`. The fix moves the declaration of `i` into the `for` loop initializer, reducing its scope to the block where it is actually used. This requires C99 or later, which is safe given the code already uses the `%zu` format specifier. The unused file-scope declaration was removed accordingly, and all other logic — the NULL checks, bounds checking on `found < max_matches`, the `match_len` clamp against `MAX_INPUT`, and the explicit NUL termination after `memcpy` — remains intact since it was already defensively correct.
