#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int count_reverse_pairs(const char **strs, int strCount) {
    int count = 0;
    for (int i = 0; i < strCount; ++i) {
        for (int j = i + 1; j < strCount; ++j) {
            size_t len1, len2;

            const char *str1 = strs[i];
            const char *str2 = strs[j];

            len1 = strlen(str1);
            len2 = strlen(str2);

            if (len1 == len2) {
                size_t match = 1;
                for (size_t k = 0; k < len1; ++k) {
                    if (str1[k] != str2[len2 - 1 - k]) {
                        match = 0;
                        break;
                    }
                }
                if (match) {
                    count++;
                }
            }
        }
    }
    return count;
}

int main() {
    const char *strs[] = {"abc", "cba", "xyz", "zyx", "123", "321"};
    int strCount = 6;
    printf("Number of reverse string pairs: %d\n", count_reverse_pairs(strs, strCount));
    return 0;
}
