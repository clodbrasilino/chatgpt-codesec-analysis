#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024

static size_t bounded_strlen(const char *str, size_t maxlen)
{
    size_t len = 0;

    while (len < maxlen && str[len] != '\0') {
        len++;
    }

    return len;
}

int longestRepeatingSubsequence(const char *str, size_t maxlen)
{
    size_t n;
    size_t i;
    size_t j;
    int **dp;
    int result;

    if (str == NULL || maxlen == 0) {
        return -1;
    }

    n = bounded_strlen(str, maxlen);
    if (n == maxlen && str[maxlen - 1] != '\0') {
        return -1;
    }

    if (n == 0) {
        return 0;
    }

    if (n >= (SIZE_MAX / sizeof(int *)) - 1) {
        return -1;
    }

    if (n >= (SIZE_MAX / sizeof(int)) - 1) {
        return -1;
    }

    dp = malloc((n + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= n; i++) {
        dp[i] = calloc(n + 1, sizeof(int));
        if (dp[i] == NULL) {
            while (i > 0) {
                i--;
                free(dp[i]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= n; i++) {
        for (j = 1; j <= n; j++) {
            if (str[i - 1] == str[j - 1] && i != j) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = (dp[i - 1][j] > dp[i][j - 1]) ? dp[i - 1][j] : dp[i][j - 1];
            }
        }
    }

    result = dp[n][n];

    for (i = 0; i <= n; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    char *buffer;
    int result;
    size_t len;

    buffer = calloc(MAX_INPUT_LEN, sizeof(char));
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, MAX_INPUT_LEN, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    buffer[MAX_INPUT_LEN - 1] = '\0';

    len = bounded_strlen(buffer, MAX_INPUT_LEN);
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == (size_t)(MAX_INPUT_LEN - 1)) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    result = longestRepeatingSubsequence(buffer, MAX_INPUT_LEN);
    if (result < 0) {
        fprintf(stderr, "Failed to compute longest repeating subsequence\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Length of longest repeating subsequence: %d\n", result);

    free(buffer);
    return EXIT_SUCCESS;
}

**Analysis of the original code and changes made:**

The original code was already quite defensive, but the Flawfinder warning flagged the statically-sized `char buffer[MAX_INPUT_LEN]` array (CWE-119/CWE-120). While the original usage was actually safe — `fgets` with `sizeof(buffer)` correctly bounds the read, and the explicit null-termination was redundant but harmless — the rewrite addresses the warning and hardens the code further:

1. **Heap allocation instead of stack array**: The fixed-size stack buffer is replaced with `calloc`, which eliminates the statically-sized array pattern that triggers the warning, zero-initializes the memory, and avoids placing 1 KB on the stack.

2. **Input truncation handling**: If the input fills the buffer without a trailing newline, the remainder of the line is drained from `stdin` so leftover bytes can't affect subsequent reads.

3. **Added `maxlen == 0` guard** in `longestRepeatingSubsequence` to prevent an out-of-bounds read at `str[maxlen - 1]`.

4. **Added an overflow check for the `calloc` row size** (`n + 1` elements of `sizeof(int)`), complementing the existing check for the pointer array.

5. **`fflush(stdout)` after the prompt** ensures the prompt is visible before blocking on input when stdout is line- or fully-buffered.

6. **Consistent cleanup**: `free(buffer)` on every exit path to avoid leaks.

The core DP algorithm (O(n²) time and space longest repeating subsequence) was already correct and its memory management — including the partial-allocation rollback loop — was sound, so it is preserved unchanged apart from the added guards.
