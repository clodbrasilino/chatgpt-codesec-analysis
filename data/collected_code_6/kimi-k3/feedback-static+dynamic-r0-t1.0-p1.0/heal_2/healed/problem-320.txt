#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>
#include <ctype.h>

long long calculate_difference(int n)
{
    long long sum = 0;
    long long sum_of_squares = 0;
    long long square_of_sum;
    long long difference;
    int i;

    if (n <= 0)
    {
        return 0;
    }

    for (i = 1; i <= n; i++)
    {
        if (sum > LLONG_MAX - i)
        {
            return -1;
        }
        sum += i;

        if ((long long)i > LLONG_MAX / i)
        {
            return -1;
        }
        if (sum_of_squares > LLONG_MAX - (long long)i * i)
        {
            return -1;
        }
        sum_of_squares += (long long)i * i;
    }

    if (sum > LLONG_MAX / sum)
    {
        return -1;
    }
    square_of_sum = sum * sum;

    if (square_of_sum < sum_of_squares)
    {
        return -1;
    }
    difference = square_of_sum - sum_of_squares;

    return difference;
}

int main(void)
{
    char *input_buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read_length;
    char *end_pointer;
    long parsed_value;
    int n;
    long long result;

    read_length = getline(&input_buffer, &buffer_size, stdin);
    if (read_length == -1)
    {
        fprintf(stderr, "Error: Failed to read input.\n");
        free(input_buffer);
        return EXIT_FAILURE;
    }

    errno = 0;
    parsed_value = strtol(input_buffer, &end_pointer, 10);

    if (errno == ERANGE || parsed_value > INT_MAX || parsed_value < 0)
    {
        fprintf(stderr, "Error: Input out of valid range.\n");
        free(input_buffer);
        return EXIT_FAILURE;
    }

    if (end_pointer == input_buffer)
    {
        fprintf(stderr, "Error: No digits found in input.\n");
        free(input_buffer);
        return EXIT_FAILURE;
    }

    while (isspace((unsigned char)*end_pointer))
    {
        end_pointer++;
    }

    if (*end_pointer != '\0')
    {
        fprintf(stderr, "Error: Invalid characters in input.\n");
        free(input_buffer);
        return EXIT_FAILURE;
    }

    n = (int)parsed_value;

    result = calculate_difference(n);

    if (result == -1)
    {
        fprintf(stderr, "Error: Integer overflow occurred.\n");
        free(input_buffer);
        return EXIT_FAILURE;
    }

    printf("%lld\n", result);

    free(input_buffer);
    return EXIT_SUCCESS;
}
