#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_WORDS 1024
#define WORD_LEN 64

char* find_first_repeated_word(const char *str) {
    char **words = (char **)malloc(MAX_WORDS * sizeof(char *));
    for (int i = 0; i < MAX_WORDS; i++) {
        words[i] = (char *)malloc(WORD_LEN * sizeof(char));
    }
    int word_count = 0;
    const char *p = str;
    bool found = false;

    while (*p) {
        if (word_count >= MAX_WORDS) break;
        if (*p == ' ' || *p == '\t' || *p == '\n') {
            p++;
            continue;
        }
        int i = 0;
        while (*p && *p != ' ' && *p != '\t' && *p != '\n' && i < WORD_LEN - 1) {
            words[word_count][i++] = *p++;
        }
        words[word_count][i] = '\0';
        for (int j = 0; j < word_count; j++) {
            if (strcmp(words[j], words[word_count]) == 0) {
                found = true;
                break;
            }
        }
        if (found) break;
        word_count++;
    }

    for (int i = 0; i < MAX_WORDS; i++) {
        free(words[i]);
    }
    free(words);

    return found ? strdup(words[word_count]) : NULL;
}

int main() {
    const char *input = "This is a test string to find the first repeated word in the string";
    char *result = find_first_repeated_word(input);
    if (result) {
        printf("First repeated word: %s\n", result);
        free(result);
    } else {
        printf("No repeated word found.\n");
    }
    return 0;
}
