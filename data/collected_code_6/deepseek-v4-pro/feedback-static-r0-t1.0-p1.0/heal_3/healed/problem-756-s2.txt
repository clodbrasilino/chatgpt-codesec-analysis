#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool match_pattern(const char *str) {
    if (str == NULL) {
        return false;
    }
    
    size_t len = 0;
    while (len < 256 && str[len] != '\0') {
        len++;
    }
    
    if (len == 0 || len >= 256) {
        return false;
    }
    
    if (str[0] != 'a') {
        return false;
    }
    
    if (len == 1) {
        return true;
    }
    
    if (len == 2) {
        return str[1] == 'b';
    }
    
    return false;
}

int main(void) {
    char input[257];
    
    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }
    
    size_t newline_pos = 0;
    while (newline_pos < sizeof(input) && input[newline_pos] != '\0') {
        if (input[newline_pos] == '\n') {
            input[newline_pos] = '\0';
            break;
        }
        newline_pos++;
    }
    
    if (newline_pos == sizeof(input) && input[sizeof(input) - 1] != '\0') {
        int c;
        while ((c = getchar()) != '\n' && c != EOF);
    }
    
    size_t length = 0;
    while (length < sizeof(input) && input[length] != '\0') {
        length++;
    }
    
    if (length >= 256) {
        printf("No match\n");
        return 0;
    }
    
    if (match_pattern(input)) {
        printf("Match\n");
    } else {
        printf("No match\n");
    }
    
    return 0;
}
