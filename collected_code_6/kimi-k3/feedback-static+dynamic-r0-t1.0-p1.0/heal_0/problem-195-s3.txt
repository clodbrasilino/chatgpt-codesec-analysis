#include <stdio.h>
#include <stdlib.h>

int find_first_position(const int *arr, size_t size, int target) {
    if (arr == NULL) {
        return -1;
    }
    
    size_t left = 0;
    size_t right = size;
    
    while (left < right) {
        size_t mid = left + (right - left) / 2;
        
        if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }
    
    if (left < size && arr[left] == target) {
        return (int)left;
    }
    
    return -1;
}

int main(void) {
    int arr[] = {1, 2, 2, 2, 3, 4, 5, 5, 6};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int target = 2;
    
    int result = find_first_position(arr, size, target);
    
    if (result != -1) {
        printf("First position of %d: %d\n", target, result);
    } else {
        printf("Element %d not found\n", target);
    }
    
    return 0;
}
