```c
#include <stdio.h>

double calculate_volume(double length, double width, double height) {
    return length * width * height;
}

int main(void) {
    double length = 5.0;
    double width = 3.0;
    double height = 2.0;
    double volume = calculate_volume(length, width, height);
    printf("Volume of cuboid: %.2f\n", volume);
    return 0;
}
```
