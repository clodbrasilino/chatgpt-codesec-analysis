#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool is_undulating(int num) {
    if (num < 100) {
        return false;
    }

    char str[32];
    snprintf(str, sizeof(str), "%d", num);
    int len = strlen(str);

    if (str[0] == str[1]) {
        return false;
    }

    for (int i = 0; i < len - 2; i++) {
        if (str[i] != str[i + 2]) {
            return false;
        }
    }

    return true;
}
