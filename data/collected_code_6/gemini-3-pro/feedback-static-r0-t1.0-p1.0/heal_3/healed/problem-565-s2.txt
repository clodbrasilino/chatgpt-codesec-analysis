#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char** split_string_into_characters(const char* str, size_t* count) {
    if (str == NULL || count == NULL) {
        return NULL;
    }

    size_t len = strlen(str);
    *count = len;

    char** chars = (char**)malloc(len * sizeof(char*));
    if (chars == NULL && len > 0) {
        return NULL;
    }

    for (size_t i = 0; i < len; i++) {
        chars[i] = (char*)malloc(2 * sizeof(char));
        if (chars[i] == NULL) {
            for (size_t j = 0; j < i; j++) {
                free(chars[j]);
            }
            free(chars);
            return NULL;
        }
        chars[i][0] = str[i];
        chars[i][1] = '\0';
    }

    return chars;
}

void free_characters_array(char** chars, size_t count) {
    if (chars != NULL) {
        for (size_t i = 0; i < count; i++) {
            free(chars[i]);
        }
        free(chars);
    }
}

int main(void) {
    const char* input_string = "Hello, World!";
    size_t char_count = 0;
    
    char** characters = split_string_into_characters(input_string, &char_count);
    
    if (characters != NULL) {
        for (size_t i = 0; i < char_count; i++) {
            printf("Character %zu: '%s'\n", i, characters[i]);
        }
        
        free_characters_array(characters, char_count);
    } else {
        fprintf(stderr, "Failed to split the string.\n");
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
