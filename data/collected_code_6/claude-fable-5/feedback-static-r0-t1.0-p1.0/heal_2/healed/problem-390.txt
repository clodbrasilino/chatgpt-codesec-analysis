```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096

static void cleanup_partial(char **list, size_t filled)
{
    while (filled > 0) {
        filled--;
        free(list[filled]);
    }
    free(list);
}

char **prepend_to_all(const char *prefix, const char **items, size_t count)
{
    char **result;
    size_t i;
    size_t prefix_len;

    if (prefix == NULL || items == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    prefix_len = strnlen(prefix, MAX_STRING_LEN);
    if (prefix_len >= MAX_STRING_LEN) {
        return NULL;
    }

    result = malloc(count * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t item_len;
        size_t total_len;

        if (items[i] == NULL) {
            cleanup_partial(result, i);
            return NULL;
        }

        item_len = strnlen(items[i], MAX_STRING_LEN);
        if (item_len >= MAX_STRING_LEN) {
            cleanup_partial(result, i);
            return NULL;
        }

        if (prefix_len > SIZE_MAX - item_len - 1) {
            cleanup_partial(result, i);
            return NULL;
        }

        total_len = prefix_len + item_len + 1;

        result[i] = malloc(total_len);
        if (result[i] == NULL) {
            cleanup_partial(result, i);
            return NULL;
        }

        if (prefix_len > total_len || item_len > total_len - prefix_len) {
            free(result[i]);
            cleanup_partial(result, i);
            return NULL;
        }

        memcpy(result[i], prefix, prefix_len);
        memcpy(result[i] + prefix_len, items[i], item_len);
        result[i][total_len - 1] = '\0';
    }

    return result;
}

void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *items[] = { "apple", "banana", "cherry" };
    size_t count = sizeof(items) / sizeof(items[0]);
    char **result;
    size_t i;

    result = prepend_to_all("fruit_", items, count);
    if (result == NULL) {
        fprintf(stderr, "Failed to build list\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", result[i]) < 0) {
            free_list(result, count);
            return EXIT_FAILURE;
        }
    }

    free_list(result, count);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already sound with respect to the two Flawfinder `memcpy` warnings — they are false positives typical of Flawfinder's pattern matching, since it cannot verify that the destination buffer was sized correctly. The rewrite hardens and clarifies the code:

1. **Both `memcpy` calls are provably safe**: the destination `result[i]` is allocated with exactly `total_len = prefix_len + item_len + 1` bytes, so writing `prefix_len` bytes at offset 0 and `item_len` bytes at offset `prefix_len` fits precisely, with one byte remaining for the NUL terminator. An explicit guard (`prefix_len > total_len || item_len > total_len - prefix_len`) was added before the copies to make the invariant machine-checkable and to satisfy static analyzers.

2. **Bounded string measurement**: `strnlen` with `MAX_STRING_LEN` prevents unbounded reads of non-terminated input (mitigating CWE-126), and inputs at or above the cap are rejected.

3. **Integer overflow protection**: both the `count * sizeof(char *)` allocation and the `prefix_len + item_len + 1` sum are checked against `SIZE_MAX` before use (mitigating CWE-190).

4. **Leak-free error paths**: the repeated unwind loops were factored into `cleanup_partial`, which frees all previously allocated strings and the pointer array on any failure, eliminating duplication and reducing the chance of a maintenance error introducing a leak or double-free.

5. **Explicit NUL termination**: the terminator is written manually rather than relying on `strcpy`/`strcat`, avoiding classic CWE-120 risks entirely.
