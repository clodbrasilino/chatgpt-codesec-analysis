#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

#define MAX_LINE_LENGTH 1048576

int is_integer(const char *str)
{
    char *endptr;
    long value;

    if (str == NULL) {
        return 0;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    errno = 0;
    value = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return 0;
    }

    if (endptr == str) {
        return 0;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 0;
    }

    (void)value;
    return 1;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *line;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    line = malloc(capacity);
    if (line == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= MAX_LINE_LENGTH) {
            free(line);
            return NULL;
        }
        if (length + 1 >= capacity) {
            size_t new_capacity;
            char *tmp;

            if (capacity > SIZE_MAX / 2) {
                free(line);
                return NULL;
            }
            new_capacity = capacity * 2;
            tmp = realloc(line, new_capacity);
            if (tmp == NULL) {
                free(line);
                return NULL;
            }
            line = tmp;
            capacity = new_capacity;
        }
        line[length++] = (char)ch;
    }

    if (ch == EOF && length == 0) {
        free(line);
        return NULL;
    }

    line[length] = '\0';
    return line;
}

int main(void)
{
    char *buffer;

    printf("Enter a string: ");
    fflush(stdout);

    buffer = read_line(stdin);
    if (buffer == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (is_integer(buffer)) {
        printf("\"%s\" is a valid integer\n", buffer);
    } else {
        printf("\"%s\" is not a valid integer\n", buffer);
    }

    free(buffer);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness (CWE-120 / CWE-20, `fgetc` in a loop):**

The Flawfinder warning is a *low-risk, informational* hit — it flags every `fgetc` used in a loop because such loops are a common source of buffer overflows. On inspection, the original code already mitigates this correctly, so no functional changes were required:

1. **Bounded growth (CWE-120 mitigated):** Before every write, `length + 1 >= capacity` triggers a `realloc`, guaranteeing the invariant `length + 1 < capacity` at the point of `line[length++] = (char)ch`. The final `line[length] = '\0'` is also always in bounds, since after the last increment `length <= capacity - 1`.

2. **Overflow-safe capacity doubling:** The `capacity > SIZE_MAX / 2` guard prevents `size_t` wraparound in `capacity * 2`, which would otherwise cause an undersized allocation and a heap overflow.

3. **Denial-of-service cap (CWE-20 mitigated):** `MAX_LINE_LENGTH` bounds total input, preventing unbounded memory consumption from a hostile stream.

4. **Correct `realloc` idiom:** The result is assigned to a temporary (`tmp`), so the original pointer isn't leaked on allocation failure.

5. **Safe `EOF` handling:** `ch` is declared `int` (not `char`), so `EOF` is distinguishable from valid byte values; the `ch == EOF && length == 0` check correctly distinguishes "no input" from a final unterminated line.

6. **Robust integer validation:** `is_integer` correctly uses `errno = 0` before `strtol`, checks `ERANGE`, rejects empty conversions (`endptr == str`), and tolerates trailing whitespace only — with `isspace` receiving an `unsigned char` cast to avoid UB on negative `char` values.

The code as written is a sound example of dynamically-sized, bounds-checked line reading; the Flawfinder finding is a false positive here.
