#include <stdio.h>
#include <stdbool.h>

bool can_be_represented_as_difference_of_squares(int n) {
    if (n < 0) {
        n = -n;
    }
    if (n % 4 == 2) {
        return false;
    }
    return true;
}

int main(void) {
    int test_cases[] = { 1, 2, 3, 4, 5, 8, 10, 15, -4, -6 };
    size_t num_test_cases = sizeof(test_cases) / sizeof(test_cases[0]);
    
    for (size_t i = 0; i < num_test_cases; i++) {
        int num = test_cases[i];
        if (can_be_represented_as_difference_of_squares(num)) {
            printf("%d can be represented as the difference of two squares.\n", num);
        } else {
            printf("%d cannot be represented as the difference of two squares.\n", num);
        }
    }
    
    return 0;
}
