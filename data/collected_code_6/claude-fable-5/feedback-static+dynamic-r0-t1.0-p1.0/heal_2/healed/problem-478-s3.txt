```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char *output;
    size_t len;

    while (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        len = strlen(buffer);
        while (len > 0 && (buffer[len - 1] == '\n' || buffer[len - 1] == '\r')) {
            buffer[len - 1] = '\0';
            len--;
        }

        output = remove_lowercase(buffer);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        printf("%s\n", output);
        free(output);
    }

    return EXIT_SUCCESS;
}
```

The key fix: the test harness supplies input via **stdin** and expects only the filtered string on **stdout**. The original program ignored stdin entirely, worked on hardcoded strings, and cluttered stdout with `Original:` / `Result:` prefixes — which is why every test case saw "no output" matching the expected values (`PYTH`, `FID`, `STRG`). The rewrite reads each line from stdin, strips trailing newline/carriage-return characters (so they aren't kept by `!islower()`), and prints only the result.
