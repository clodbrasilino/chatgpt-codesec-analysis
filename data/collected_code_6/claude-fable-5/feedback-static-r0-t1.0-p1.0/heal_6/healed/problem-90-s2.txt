#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define INITIAL_SIZE 1024

size_t longest_word_length(const char *str)
{
    size_t max_len = 0;
    size_t current_len = 0;
    size_t i = 0;

    if (str == NULL) {
        return 0;
    }

    while (str[i] != '\0') {
        if (isspace((unsigned char)str[i])) {
            if (current_len > max_len) {
                max_len = current_len;
            }
            current_len = 0;
        } else {
            current_len++;
        }
        i++;
    }

    if (current_len > max_len) {
        max_len = current_len;
    }

    return max_len;
}

static char *read_line(FILE *stream)
{
    size_t capacity = INITIAL_SIZE;
    size_t length = 0;
    char *buffer;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length + 1 >= capacity) {
            char *new_buffer;
            if (capacity > (size_t)-1 / 2) {
                free(buffer);
                return NULL;
            }
            capacity *= 2;
            new_buffer = realloc(buffer, capacity);
            if (new_buffer == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = new_buffer;
        }
        buffer[length++] = (char)ch;
    }

    if (length == 0 && ch == EOF) {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';
    return buffer;
}

        return EXIT_FAILURE;

    free(buffer);

Analysis of the flagged weakness and its mitigation:

**Flawfinder `fgetc` warning (CWE-120, CWE-20):** The concern is buffer overflow when reading characters in a loop. The code already handles this correctly, and the rewrite preserves those protections:

1. **Bounds check before every write:** The condition `length + 1 >= capacity` guarantees there is always room for both the incoming character and the eventual null terminator before `buffer[length++]` executes.

2. **Integer overflow protection:** The check `capacity > (size_t)-1 / 2` prevents `capacity *= 2` from wrapping around, which would otherwise cause an undersized allocation followed by out-of-bounds writes.

3. **Safe reallocation pattern:** `realloc` is assigned to a temporary pointer (`new_buffer`) so the original buffer is not leaked or lost on allocation failure.

4. **Character handling:** `fgetc` returns `int`, correctly compared against `EOF` before being cast to `char`, and `isspace` receives an `unsigned char` cast to avoid undefined behavior with negative `char` values.

5. **Guaranteed termination:** `buffer[length] = '\0'` is always safe because of the `length + 1 >= capacity` invariant maintained throughout the loop.

The original code was already sound; no functional changes were required—only comment removal as requested.
