#include <stdio.h>
#include <string.h>

int match(char *str)
{
    while (*str != '\0')
    {
        if (*str == 'a')
        {
            str++;
            if (*str == 'b')
                str++;
            return 1;
        }
        str++;
    }
    return 0;
}

int main()
{
    char *str1 = "abc";
    char *str2 = "acb";
    char *str3 = "bca";
    
    printf("%s: %d\n", str1, match(str1));
    printf("%s: %d\n", str2, match(str2));
    printf("%s: %d\n", str3, match(str3));

    return 0;
}
