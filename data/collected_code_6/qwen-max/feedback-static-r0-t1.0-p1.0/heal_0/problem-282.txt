#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode *next;
};

struct ListNode* createNode(int value) {
    struct ListNode *newNode = (struct ListNode*)malloc(sizeof(struct ListNode));
    newNode->val = value;
    newNode->next = NULL;
    return newNode;
}

struct ListNode* subtractLists(struct ListNode* l1, struct ListNode* l2) {
    if (l1 == NULL || l2 == NULL) {
        return NULL;
    }

    struct ListNode *result = NULL, *current = NULL;

    while (l1 != NULL && l2 != NULL) {
        int v1 = l1->val, v2 = l2->val;
        if (v1 > v2) {
            struct ListNode *newNode = createNode(v1);
            if (result == NULL) {
                result = newNode;
                current = newNode;
            } else {
                current->next = newNode;
                current = newNode;
            }
        } else if (v1 < v2) {
            l1 = l1->next;
        } else {
            l1 = l1->next;
            l2 = l2->next;
        }
    }

    while (l1 != NULL) {
        struct ListNode *newNode = createNode(l1->val);
        if (result == NULL) {
            result = newNode;
            current = newNode;
        } else {
            current->next = newNode;
            current = newNode;
        }
        l1 = l1->next;
    }

    return result;
}

void printList(struct ListNode* head) {
    while (head != NULL) {
        printf("%d -> ", head->val);
        head = head->next;
    }
    printf("NULL\n");
}

int main() {
    struct ListNode *l1 = createNode(5);
    l1->next = createNode(9);
    l1->next->next = createNode(12);

    struct ListNode *l2 = createNode(9);
    l2->next = createNode(10);

    struct ListNode *result = subtractLists(l1, l2);

    printList(result);

    return 0;
}
