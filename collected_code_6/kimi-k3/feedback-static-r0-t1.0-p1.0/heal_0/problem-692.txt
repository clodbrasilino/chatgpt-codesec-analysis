```c
#include <stdio.h>
#include <stdlib.h>

int last_two_digits_factorial(int n) {
    if (n < 0) {
        return -1;
    }
    if (n >= 10) {
        return 0;
    }
    
    int result = 1;
    for (int i = 2; i <= n; i++) {
        result = (result * i) % 100;
    }
    return result;
}

int main(void) {
    int n;
    char buffer[100];
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }
    
    if (sscanf(buffer, "%d", &n) != 1) {
        fprintf(stderr, "Invalid input format\n");
        return EXIT_FAILURE;
    }
    
    if (n < 0) {
        fprintf(stderr, "Input must be non-negative\n");
        return EXIT_FAILURE;
    }
    
    int result = last_two_digits_factorial(n);
    printf("%02d\n", result);
    
    return EXIT_SUCCESS;
}
```
