#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_BUFFER_SIZE 11

static int safe_copy(const char *input, int start, int len, char *output, size_t output_size, size_t *pos) {
    if (len <= 0) return -1;
    if (*pos + (size_t)len >= output_size) return -1;
    memcpy(output + *pos, input + start, (size_t)len);
    *pos += (size_t)len;
    return 0;
}

int convert_date_format(const char *input, char *output, size_t output_size) {
    regex_t regex;
    regmatch_t matches[4];
    const char *pattern = "^([0-9]{4})-([0-9]{2})-([0-9]{2})$";
    size_t pos = 0;
    
    if (output == NULL || input == NULL) {
        return -1;
    }
    
    if (output_size < DATE_BUFFER_SIZE) {
        return -1;
    }
    
    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        return -1;
    }
    
    if (regexec(&regex, input, 4, matches, 0) != 0) {
        regfree(&regex);
        return -1;
    }
    
    if (matches[1].rm_so == -1 || matches[2].rm_so == -1 || matches[3].rm_so == -1) {
        regfree(&regex);
        return -1;
    }
    
    int year_len = (int)(matches[1].rm_eo - matches[1].rm_so);
    int month_len = (int)(matches[2].rm_eo - matches[2].rm_so);
    int day_len = (int)(matches[3].rm_eo - matches[3].rm_so);
    
    if (year_len != 4 || month_len != 2 || day_len != 2) {
        regfree(&regex);
        return -1;
    }
    
    size_t input_len = strlen(input);
    if ((size_t)matches[3].rm_so + (size_t)day_len > input_len ||
        (size_t)matches[2].rm_so + (size_t)month_len > input_len ||
        (size_t)matches[1].rm_so + (size_t)year_len > input_len) {
        regfree(&regex);
        return -1;
    }
    
    if (safe_copy(input, matches[3].rm_so, day_len, output, output_size, &pos) != 0) {
        regfree(&regex);
        return -1;
    }
    
    if (pos >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[pos++] = '-';
    
    if (safe_copy(input, matches[2].rm_so, month_len, output, output_size, &pos) != 0) {
        regfree(&regex);
        return -1;
    }
    
    if (pos >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[pos++] = '-';
    
    if (safe_copy(input, matches[1].rm_so, year_len, output, output_size, &pos) != 0) {
        regfree(&regex);
        return -1;
    }
    
    if (pos >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[pos] = '\0';
    
    regfree(&regex);
    return 0;
}

int main(void) {
    const char *input_date = "2024-03-15";
    char output_date[DATE_BUFFER_SIZE] = {0};
    
    if (convert_date_format(input_date, output_date, sizeof(output_date)) == 0) {
        printf("%s\n", output_date);
    } else {
        fprintf(stderr, "Invalid date format\n");
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
