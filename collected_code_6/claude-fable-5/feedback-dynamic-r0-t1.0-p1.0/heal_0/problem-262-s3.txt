#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *create_node(int data)
{
    struct Node *node = malloc(sizeof(struct Node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

struct Node *build_list(const int *values, size_t count)
{
    struct Node *head = NULL;
    struct Node *tail = NULL;
    size_t i;

    if (values == NULL || count == 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        struct Node *node = create_node(values[i]);
        if (node == NULL) {
            while (head != NULL) {
                struct Node *tmp = head;
                head = head->next;
                free(tmp);
            }
            return NULL;
        }
        if (head == NULL) {
            head = node;
            tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
    }
    return head;
}

int split_list(struct Node *head, size_t first_len,
               struct Node **first, struct Node **second)
{
    struct Node *current = head;
    size_t i;

    if (first == NULL || second == NULL) {
        return -1;
    }

    *first = NULL;
    *second = NULL;

    if (first_len == 0) {
        *second = head;
        return 0;
    }

    if (head == NULL) {
        return -1;
    }

    for (i = 1; i < first_len; i++) {
        current = current->next;
        if (current == NULL) {
            return -1;
        }
    }

    *first = head;
    *second = current->next;
    current->next = NULL;
    return 0;
}

void print_list(const struct Node *head)
{
    const struct Node *current = head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

void free_list(struct Node *head)
{
    while (head != NULL) {
        struct Node *tmp = head;
        head = head->next;
        free(tmp);
    }
}

int main(void)
{
    int values[] = {1, 2, 3, 4, 5, 6, 7};
    size_t count = sizeof(values) / sizeof(values[0]);
    struct Node *head = NULL;
    struct Node *first = NULL;
    struct Node *second = NULL;
    size_t first_len = 3;

    head = build_list(values, count);
    if (head == NULL) {
        fprintf(stderr, "Failed to build list\n");
        return EXIT_FAILURE;
    }

    printf("Original list: ");
    print_list(head);

    if (split_list(head, first_len, &first, &second) != 0) {
        fprintf(stderr, "Failed to split list\n");
        free_list(head);
        return EXIT_FAILURE;
    }

    printf("First part: ");
    print_list(first);
    printf("Second part: ");
    print_list(second);

    free_list(first);
    free_list(second);

    return EXIT_SUCCESS;
}
