#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int min_swaps_to_equal(const char *s1, const char *s2) {
    size_t len1 = strlen(s1);
    size_t len2 = strlen(s2);
    
    if (len1 != len2) {
        return -1;
    }
    
    int count_10 = 0;
    int count_01 = 0;
    
    for (size_t i = 0; i < len1; i++) {
        if (s1[i] != s2[i]) {
            if (s1[i] == '1' && s2[i] == '0') {
                count_10++;
            } else if (s1[i] == '0' && s2[i] == '1') {
                count_01++;
            }
        }
    }
    
    if (count_10 != count_01) {
        return -1;
    }
    
    return count_10;
}

int main(void) {
    char s1[1001];
    char s2[1001];
    
    if (fgets(s1, sizeof(s1), stdin) == NULL) {
        return EXIT_FAILURE;
    }
    
    if (fgets(s2, sizeof(s2), stdin) == NULL) {
        return EXIT_FAILURE;
    }
    
    s1[strcspn(s1, "\n")] = '\0';
    s2[strcspn(s2, "\n")] = '\0';
    
    for (size_t i = 0; s1[i] != '\0'; i++) {
        if (s1[i] != '0' && s1[i] != '1') {
            return EXIT_FAILURE;
        }
    }
    
    for (size_t i = 0; s2[i] != '\0'; i++) {
        if (s2[i] != '0' && s2[i] != '1') {
            return EXIT_FAILURE;
        }
    }
    
    int result = min_swaps_to_equal(s1, s2);
    
    if (result == -1) {
        printf("-1\n");
    } else {
        printf("%d\n", result);
    }
    
    return EXIT_SUCCESS;
}
