#include <stdio.h>
#include <stdlib.h>

void rearrangeNumbers(int arr[], int size) {
    if (arr == NULL || size <= 0) {
        return;
    }

    int j = 0;
    for (int i = 0; i < size; i++) {
        if (arr[i] < 0) {
            if (i != j) {
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
            j++;
        }
    }
}

void printArray(int arr[], int size) {
    if (arr == NULL || size <= 0) {
        return;
    }
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main(void) {
    int arr[] = {-1, 2, -3, 4, 5, 6, -7, 8, -9};
    int size = sizeof(arr) / sizeof(arr[0]);

    printArray(arr, size);
    rearrangeNumbers(arr, size);
    printArray(arr, size);

    return 0;
}
