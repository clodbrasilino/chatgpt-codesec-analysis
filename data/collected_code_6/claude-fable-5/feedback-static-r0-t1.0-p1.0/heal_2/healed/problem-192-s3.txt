#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define BUFFER_SIZE 256

bool has_letter_and_number(const char *str)
{
    bool has_letter = false;
    bool has_digit = false;
    size_t i;

    if (str == NULL) {
        return false;
    }

    for (i = 0U; str[i] != '\0'; i++) {
        unsigned char ch = (unsigned char)str[i];
        if (isalpha(ch)) {
            has_letter = true;
        } else if (isdigit(ch)) {
            has_digit = true;
        }
        if (has_letter && has_digit) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (has_letter_and_number(buffer)) {
        printf("The string contains at least one letter and one number.\n");
    } else {
        printf("The string does not contain both a letter and a number.\n");
    }

    return EXIT_SUCCESS;
}

The rewritten code addresses each of the identified weaknesses:

1. **CWE-119/CWE-120 (statically-sized buffer)** — The buffer is zero-initialized with `memset`, `fgets` is used with `sizeof(buffer)` to strictly limit input length, and an explicit null terminator is written at the last position as a defensive guarantee. All indexing uses bounds-safe expressions (`len - 1U` only after checking `len > 0U`).

2. **CWE-126 (`strlen` over-read)** — Replaced `strlen` with `strnlen(buffer, sizeof(buffer))`, which bounds the scan to the buffer size and cannot over-read even if the string were somehow not null-terminated.

3. **CWE-120/CWE-20 (`getchar` in a loop)** — The `getchar` loop writes nothing to any buffer; it only discards excess characters until a newline or `EOF`, so no boundary can be violated. The loop terminates correctly on both conditions.

Additionally, `isalpha`/`isdigit` receive values cast to `unsigned char`, avoiding undefined behavior for negative `char` values, and `NULL` input is rejected in `has_letter_and_number`.
