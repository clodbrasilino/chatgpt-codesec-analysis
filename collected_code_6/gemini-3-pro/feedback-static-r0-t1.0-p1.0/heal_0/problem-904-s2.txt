```c
#include <stdio.h>
#include <stdbool.h>

bool is_even(int number) {
    return (number % 2 == 0);
}

int main(void) {
    int num;
    
    if (printf("Enter an integer: ") < 0) {
        return 1;
    }
    
    if (scanf("%d", &num) != 1) {
        if (printf("Invalid input.\n") < 0) {
            return 1;
        }
        return 1;
    }
    
    if (is_even(num)) {
        if (printf("%d is even.\n", num) < 0) {
            return 1;
        }
    } else {
        if (printf("%d is false (odd).\n", num) < 0) {
            return 1;
        }
    }
    
    return 0;
}
```
