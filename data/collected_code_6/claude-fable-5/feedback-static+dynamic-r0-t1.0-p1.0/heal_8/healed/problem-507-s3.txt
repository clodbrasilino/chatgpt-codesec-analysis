#include <stdlib.h>
#include <string.h>

char **remove_words(char **words, size_t count, char **targets, size_t target_count, size_t *new_count)
{
    char **result;
    size_t i;
    size_t j;
    size_t kept;
    int match;
    size_t len;

    if (new_count == NULL) {
        return NULL;
    }

    *new_count = 0;

    if (words == NULL && count > 0) {
        return NULL;
    }

    if (count > ((size_t)-1) / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count > 0 ? count : 1, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    kept = 0;
    for (i = 0; i < count; i++) {
        if (words[i] == NULL) {
            continue;
        }
        match = 0;
        if (targets != NULL) {
            for (j = 0; j < target_count; j++) {
                if (targets[j] != NULL && strcmp(words[i], targets[j]) == 0) {
                    match = 1;
                    break;
                }
            }
        }
        if (match == 0) {
            len = strlen(words[i]);
            if (len == (size_t)-1) {
                free_word_list(result, kept);
                return NULL;
            }
            result[kept] = malloc(len + 1);
            if (result[kept] == NULL) {
                free_word_list(result, kept);
                return NULL;
            }
            memcpy(result[kept], words[i], len);
            result[kept][len] = '\0';
            kept++;
        }
    }

    *new_count = kept;
    return result;
}

void free_word_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char *read_line(FILE *fp)
{
    char *buf;
    size_t capacity;
    size_t length;
    int c;
    char *tmp;

    if (fp == NULL) {
        return NULL;
    }

    capacity = 128;
    length = 0;
    buf = malloc(capacity);
    if (buf == NULL) {
        return NULL;
    }

    c = fgetc(fp);
    if (c == EOF) {
        free(buf);
        return NULL;
    }

    while (c != EOF && c != '\n') {
        if (length + 2 >= capacity) {
            if (capacity > ((size_t)-1) / 2) {
                free(buf);
                return NULL;
            }
            capacity *= 2;
            tmp = realloc(buf, capacity);
            if (tmp == NULL) {
                free(buf);
                return NULL;
            }
            buf = tmp;
        }
        buf[length] = (char)c;
        length++;
        c = fgetc(fp);
    }

    buf[length] = '\0';
    return buf;
}

static char **split_line(char *line, size_t *count)
{
    char **tokens;
    char *tok;
    char *saveptr;
    size_t capacity;
    size_t n;
    char **tmp;
    const char *delims = " \t\r\n,[]'\"";

    if (line == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;
    capacity = 8;
    n = 0;
    tokens = malloc(sizeof(char *) * capacity);
    if (tokens == NULL) {
        return NULL;
    }

    saveptr = NULL;
    tok = strtok_r(line, delims, &saveptr);
    while (tok != NULL) {
        if (n >= capacity) {
            if (capacity > ((size_t)-1) / (2 * sizeof(char *))) {
                free(tokens);
                return NULL;
            }
            capacity *= 2;
            tmp = realloc(tokens, sizeof(char *) * capacity);
            if (tmp == NULL) {
                free(tokens);
                return NULL;
            }
            tokens = tmp;
        }
        tokens[n] = tok;
        n++;
        tok = strtok_r(NULL, delims, &saveptr);
    }

    *count = n;
    return tokens;
}

static void print_list(char **list, size_t count)
{
    size_t i;

    printf("[");
    if (list != NULL) {
        for (i = 0; i < count; i++) {
            if (list[i] != NULL) {
                printf("'%s'", list[i]);
            } else {
                printf("''");
            }
            if (i + 1 < count) {
                printf(", ");
            }
        }
    }
    printf("]\n");
}

int main(void)
{
    char *word_line;
    char *target_line;
    char **words;
    char **targets;
    char **filtered;
    size_t word_count;
    size_t target_count;
    size_t new_count;

    word_count = 0;
    target_count = 0;
    new_count = 0;

    word_line = read_line(stdin);
    if (word_line == NULL) {
        word_line = malloc(1);
        if (word_line == NULL) {
            fprintf(stderr, "Out of memory\n");
            return EXIT_FAILURE;
        }
        word_line[0] = '\0';
    }

    target_line = read_line(stdin);
    if (target_line == NULL) {
        target_line = malloc(1);
        if (target_line == NULL) {
            free(word_line);
            fprintf(stderr, "Out of memory\n");
            return EXIT_FAILURE;
        }
        target_line[0] = '\0';
    }

    words = split_line(word_line, &word_count);
    if (words == NULL) {
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to parse words\n");
        return EXIT_FAILURE;
    }

    targets = split_line(target_line, &target_count);
    if (targets == NULL) {
        free(words);
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to parse targets\n");
        return EXIT_FAILURE;
    }

    filtered = remove_words(words, word_count, targets, target_count, &new_count);
    if (filtered == NULL) {
        free(words);
        free(targets);
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to remove words\n");
        return EXIT_FAILURE;
    }

    print_list(filtered, new_count);
    fflush(stdout);

    free_word_list(filtered, new_count);
    free(words);
    free(targets);
    free(word_line);
    free(target_line);
    return EXIT_SUCCESS;
}

Key changes made to address the identified weaknesses:

1. **`<no output>` test failures**: The most likely cause was `read_line` returning `NULL` (e.g., when input lacked a trailing newline on the last line, or when only one line was supplied), causing `main` to exit with an error before printing anything. The rewrite treats a missing/empty first line the same way as a missing target line — substituting an empty string instead of aborting — so the program always produces output on valid input.

2. **Buffer boundary in `read_line` (CWE-120/CWE-20)**: The growth check was tightened to `length + 2 >= capacity`, guaranteeing room for both the character being stored and the terminating `'\0'` written after the loop, eliminating any off-by-one risk.

3. **`strlen`/`memcpy` concerns (CWE-126/CWE-120)**: The strings passed to `strlen` originate from `read_line`, which always NUL-terminates, and the `memcpy` destination is allocated as `len + 1` immediately before the copy, so the destination is always large enough; the explicit `len == (size_t)-1` guard is retained defensively.

4. **Robustness hardening**: `split_line` and `remove_words` now initialize `*count`/`*new_count` before any early return, `print_list` tolerates `NULL` entries, the duplicated manual cleanup loop in `remove_words` was replaced with `free_word_list`, and `read_line` guards against a `NULL` `FILE *`.
