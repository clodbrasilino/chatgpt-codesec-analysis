```c
#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_STRING_LENGTH 100000

int shortest_supersequence_length(const char *str1, const char *str2)
{
    size_t len1;
    size_t len2;
    size_t i;
    size_t j;
    size_t *dp;
    size_t lcs_length;
    size_t result;

    if (str1 == NULL || str2 == NULL)
    {
        return -1;
    }

    len1 = strnlen(str1, MAX_STRING_LENGTH);
    if (len1 >= MAX_STRING_LENGTH)
    {
        return -1;
    }

    len2 = strnlen(str2, MAX_STRING_LENGTH);
    if (len2 >= MAX_STRING_LENGTH)
    {
        return -1;
    }

    dp = malloc((len2 + 1) * sizeof(*dp));
    if (dp == NULL)
    {
        return -1;
    }

    for (j = 0; j <= len2; j++)
    {
        dp[j] = 0;
    }

    for (i = 1; i <= len1; i++)
    {
        size_t prev_diagonal = 0;
        for (j = 1; j <= len2; j++)
        {
            size_t temp = dp[j];
            if (str1[i - 1] == str2[j - 1])
            {
                dp[j] = prev_diagonal + 1;
            }
            else
            {
                dp[j] = (dp[j - 1] > dp[j]) ? dp[j - 1] : dp[j];
            }
            prev_diagonal = temp;
        }
    }

    lcs_length = dp[len2];
    free(dp);
    dp = NULL;

    result = len1 + len2 - lcs_length;
    if (result > (size_t)INT_MAX)
    {
        return -1;
    }

    return (int)result;
}

int main(void)
{
    const char *str1 = "AGGTAB";
    const char *str2 = "GXTXAYB";
    int result;

    result = shortest_supersequence_length(str1, str2);
    if (result < 0)
    {
        fprintf(stderr, "Error: invalid input or memory allocation failure\n");
        return EXIT_FAILURE;
    }

    printf("str1: %s\n", str1);
    printf("str2: %s\n", str2);
    printf("Length of shortest common supersequence: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **Replaced `strlen` with `strnlen`** — Both flagged calls now use `strnlen(str, MAX_STRING_LENGTH)`, which bounds the scan to a maximum number of bytes. This eliminates the CWE-126 buffer over-read risk when a non-`\0`-terminated string is passed in, since the function will never read past the specified limit.

2. **Added termination validation** — If `strnlen` returns a length equal to `MAX_STRING_LENGTH`, the string is not null-terminated within the allowed bound, so the function returns `-1` to signal invalid input instead of proceeding with potentially unsafe data.

3. **Added `_POSIX_C_SOURCE 200809L` feature test macro** — This ensures `strnlen` is properly declared by `<string.h>` on POSIX-compliant systems (it is not part of strict C99/C11), preventing implicit-declaration warnings/errors.
