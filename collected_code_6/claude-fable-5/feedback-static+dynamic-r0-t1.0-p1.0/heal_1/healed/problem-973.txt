```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

int left_rotate(char *str, size_t buf_size, size_t d)
{
    size_t len;
    char *temp;

    if (str == NULL || buf_size == 0) {
        return -1;
    }

    len = strnlen(str, buf_size);
    if (len == buf_size) {
        return -1;
    }

    if (len == 0) {
        return 0;
    }

    d = d % len;
    if (d == 0) {
        return 0;
    }

    temp = malloc(len + 1);
    if (temp == NULL) {
        return -1;
    }

    if (len - d > len || d > len || len + 1 > buf_size) {
        free(temp);
        return -1;
    }

    memcpy(temp, str + d, len - d);
    memcpy(temp + (len - d), str, d);
    temp[len] = '\0';

    memcpy(str, temp, len + 1);
    free(temp);

    return 0;
}

int main(void)
{
    char str[] = "HelloWorld";
    size_t d = 3;

    printf("Original string: %s\n", str);

    if (left_rotate(str, sizeof(str), d) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    printf("Rotated string:  %s\n", str);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen`, bounded by an explicit buffer size passed by the caller (`buf_size`). If no null terminator is found within the buffer (`len == buf_size`), the function fails safely instead of over-reading.

2. **CWE-120 (memcpy overflow):** The function signature now takes `buf_size` so the destination capacity is known. Before any `memcpy`, the code verifies that `len + 1 <= buf_size` and that the partial copy lengths (`d`, `len - d`) are within bounds, guaranteeing the destination can always hold the source data.

3. **Caller update:** `main` passes `sizeof(str)` so the rotation function has an accurate bound on the writable buffer.
