#include <stdio.h>
#include <stdlib.h>

void sort012(int *arr, int size) {
    if (arr == NULL || size <= 0) {
        return;
    }
    
    int count0 = 0, count1 = 0, count2 = 0;
    
    for (int i = 0; i < size; i++) {
        if (arr[i] == 0) {
            count0++;
        } else if (arr[i] == 1) {
            count1++;
        } else if (arr[i] == 2) {
            count2++;
        }
    }
    
    int index = 0;
    
    while (count0 > 0) {
        arr[index++] = 0;
        count0--;
    }
    
    while (count1 > 0) {
        arr[index++] = 1;
        count1--;
    }
    
    while (count2 > 0) {
        arr[index++] = 2;
        count2--;
    }
}

int main(void) {
    int arr[] = {0, 1, 2, 1, 0, 2, 1, 0, 2, 1};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    printf("Original array: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    sort012(arr, size);
    
    printf("Sorted array: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
