#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <stdint.h>

long long sum_common_divisors(long long a, long long b) {
    if (a == 0 && b == 0) {
        return 0;
    }
    
    unsigned long long ua, ub;
    if (a == LLONG_MIN) {
        ua = (unsigned long long)LLONG_MAX + 1;
    } else if (a < 0) {
        ua = (unsigned long long)(-a);
    } else {
        ua = (unsigned long long)a;
    }
    
    if (b == LLONG_MIN) {
        ub = (unsigned long long)LLONG_MAX + 1;
    } else if (b < 0) {
        ub = (unsigned long long)(-b);
    } else {
        ub = (unsigned long long)b;
    }
    
    unsigned long long gcd_val = ua;
    unsigned long long temp = ub;
    
    while (temp != 0) {
        unsigned long long remainder = gcd_val % temp;
        gcd_val = temp;
        temp = remainder;
    }
    
    unsigned long long sum = 0;
    for (unsigned long long i = 1; i <= gcd_val / i; i++) {
        if (gcd_val % i == 0) {
            sum += i;
            if (i != gcd_val / i) {
                sum += gcd_val / i;
            }
        }
    }
    
    return (long long)sum;
}

int main(void) {
    long long num1, num2;
    
    if (scanf("%lld %lld", &num1, &num2) != 2) {
        return EXIT_FAILURE;
    }
    
    long long result = sum_common_divisors(num1, num2);
    printf("%lld\n", result);
    
    return EXIT_SUCCESS;
}
