#include <stdio.h>
#include <limits.h>
#include <stddef.h>
#include <string.h>

#define INPUT_BUFFER_SIZE 256

int find_max_occurring_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    unsigned char max_char = 0;
    int found = 0;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
            max_char = c;
            found = 1;
        }
    }

    if (found == 0) {
        return -1;
    }

    *result = (char)max_char;
    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    char max_char;
    char *newline;

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    newline = memchr(input, '\n', sizeof(input));
    if (newline != NULL) {
        *newline = '\0';
    }

    if (find_max_occurring_char(input, &max_char) != 0) {
        fprintf(stderr, "Empty string or invalid input\n");
        return 1;
    }

    if (printf("Maximum occurring character: %c\n", max_char) < 0) {
        return 1;
    }

    return 0;
}

This is a heuristic warning — Flawfinder flags all fixed-size character buffers because they *can* be misused. On inspection, the original code already mitigates this correctly, so no functional changes were required:

1. **Bounded read**: `fgets(input, (int)sizeof(input), stdin)` limits input to the buffer size, writing at most `sizeof(input) - 1` characters plus a null terminator. There is no overflow path here, unlike `gets()` or an unbounded `scanf("%s", ...)`.

2. **Guaranteed termination**: `input[sizeof(input) - 1] = '\0';` is defensive (redundant with a successful `fgets`, but harmless) and guarantees the buffer is null-terminated before any string operations.

5. **NULL-pointer checks** in `find_max_occurring_char` and checked return values of `fgets` and `printf` round out the defensive posture.
