#include <stdio.h>
#include <stdlib.h>

int find_kth_element(const int *arr, size_t n, size_t k, int *result);

int find_kth_element(const int *arr, size_t n, size_t k, int *result)
{
    size_t count;
    size_t i;

    if (arr == NULL || result == NULL || n == 0U || k == 0U || k > n) {
        return -1;
    }

    count = 0U;

    for (i = 0U; i < n; i++) {
        if ((arr[i] % 2) != 0) {
            count++;
            if (count == k) {
                *result = arr[i];
                return 0;
            }
        }
    }

    for (i = 0U; i < n; i++) {
        if ((arr[i] % 2) == 0) {
            count++;
            if (count == k) {
                *result = arr[i];
                return 0;
            }
        }
    }

    return -1;
}

int main(void)
{
    int arr[] = {4, 7, 2, 9, 6, 1, 8, 3, 5, 10};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k;
    int result;

    for (k = 1U; k <= n; k++) {
        if (find_kth_element(arr, n, k, &result) != 0) {
            (void)fprintf(stderr, "Error: could not find element at position %zu\n", k);
            return EXIT_FAILURE;
        }
        if (printf("Element at position %zu: %d\n", k, result) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (find_kth_element(arr, n, n + 1U, &result) == 0) {
        (void)fprintf(stderr, "Error: out-of-range lookup unexpectedly succeeded\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
