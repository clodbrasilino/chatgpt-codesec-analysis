```c
#include <stdio.h>
#include <stdlib.h>

long long sum_non_repeated(const int *arr, size_t n)
{
    long long sum = 0;

    if (arr == NULL || n == 0) {
        return 0;
    }

    for (size_t i = 0; i < n; i++) {
        int repeated = 0;
        for (size_t j = 0; j < n; j++) {
            if (i != j && arr[i] == arr[j]) {
                repeated = 1;
                break;
            }
        }
        if (!repeated) {
            sum += (long long)arr[i];
        }
    }

    return sum;
}

int main(void)
{
    int data[] = {1, 2, 3, 2, 4, 5, 1, 6};
    size_t n = sizeof(data) / sizeof(data[0]);
    long long result;

    result = sum_non_repeated(data, n);

    if (printf("Sum of non-repeated elements: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The key fix applied: the variable `repeated` is now declared inside the outer `for` loop body, reducing its scope to the smallest block where it is actually used. This addresses the `[variableScope]` weakness — it eliminates the risk of stale values leaking between iterations, makes the code easier to reason about, and lets the compiler optimize register allocation more effectively. Additionally, the loop counters `i` and `j` were moved into their respective `for` statements (C99 style), applying the same scope-reduction principle consistently. Functional behavior is unchanged: the program still sums only the elements that appear exactly once (3 + 4 + 5 + 6 = 18).
