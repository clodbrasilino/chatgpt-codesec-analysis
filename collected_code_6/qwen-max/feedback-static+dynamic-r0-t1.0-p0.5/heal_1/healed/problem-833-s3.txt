```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} DictionaryEntry;

typedef struct {
    DictionaryEntry *entries;
    int size;
    int capacity;
} Dictionary;

Dictionary* create_dictionary(int initial_capacity) {
    Dictionary *dict = (Dictionary*)malloc(sizeof(Dictionary));
    dict->capacity = initial_capacity;
    dict->size = 0;
    dict->entries = (DictionaryEntry*)calloc(initial_capacity, sizeof(DictionaryEntry));
    return dict;
}

void add_entry(Dictionary *dict, const char *key, const char *value) {
    if (dict->size >= dict->capacity) {
        dict->capacity *= 2;
        dict->entries = (DictionaryEntry*)realloc(dict->entries, dict->capacity * sizeof(DictionaryEntry));
    }
    dict->entries[dict->size].key = strdup(key);
    dict->entries[dict->size].value = strdup(value);
    dict->size++;
}

char** get_keys(Dictionary *dict) {
    char **keys = (char**)malloc(dict->size * sizeof(char*));
    for (int i = 0; i < dict->size; i++) {
        keys[i] = dict->entries[i].key;
    }
    return keys;
}

void free_dictionary(Dictionary *dict) {
    for (int i = 0; i < dict->size; i++) {
        free(dict->entries[i].key);
        free(dict->entries[i].value);
    }
    free(dict->entries);
    free(dict);
}

int main() {
    Dictionary *dict = create_dictionary(4);
    add_entry(dict, "one", "1");
    add_entry(dict, "two", "2");
    add_entry(dict, "three", "3");

    char **keys = get_keys(dict);
    for (int i = 0; i < dict->size; i++) {
        printf("%s\n", keys[i]);
    }

    free(keys);
    free_dictionary(dict);
    return 0;
}
```
