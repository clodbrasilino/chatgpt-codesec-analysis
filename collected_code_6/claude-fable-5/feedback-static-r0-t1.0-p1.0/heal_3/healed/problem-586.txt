```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    total_bytes = n * sizeof(int);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > n || (n - k) > n) {
        return -1;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already substantially safe — the Flawfinder warnings on `memcpy` are informational (risk level 2), flagging the *use* of the function rather than an actual defect. Here's why the copies were and remain safe, and what the rewrite adds:

1. **`memcpy(temp, arr, head_bytes)`** — `temp` is allocated with exactly `head_bytes`, and `arr` holds at least `n * sizeof(int) >= head_bytes` bytes since `k < n` after the modulo. Safe.

2. **`memmove(arr, arr + k, tail_bytes)`** — correctly uses `memmove` for potentially overlapping regions; `arr + k` plus `tail_bytes` ends exactly at `arr + n`. Safe.

3. **`memcpy(arr + (n - k), temp, head_bytes)`** — destination region spans `arr + (n - k)` to `arr + n`, exactly `head_bytes`. No overlap with `temp` (separate heap allocation). Safe.

**Hardening added in the rewrite:**

- An explicit invariant check `head_bytes + tail_bytes == total_bytes` guards against any arithmetic inconsistency, making the buffer-size guarantee locally verifiable at the call sites — which is precisely what CWE-120 auditing asks for ("make sure the destination can always hold the source data").
- Redundant but defensive bounds checks (`k > n`, `(n - k) > n`) document the preconditions the copies rely on.
- The existing overflow guard `n > SIZE_MAX / sizeof(int)` is retained, preventing wraparound in `n * sizeof(int)` and all derived byte counts.

The remaining structure (null checks, `malloc` failure handling, `free` before return, checked `printf` return values) was already correct and is preserved.
