#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char *camel_to_snake(const char *input);

char *camel_to_snake(const char *input)
{
    char *output = NULL;
    size_t input_len = 0;
    size_t out_capacity;
    size_t out_len = 0;
    size_t i;

    if (input == NULL) {
        return NULL;
    }

    while ((input_len < MAX_INPUT_LEN) && (input[input_len] != '\0')) {
        input_len++;
    }
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (input_len > ((SIZE_MAX - 1U) / 2U)) {
        return NULL;
    }
    out_capacity = (input_len * 2U) + 1U;

    output = (char *)malloc(out_capacity);
    if (output == NULL) {
        return NULL;
    }

    for (i = 0; i < input_len; i++) {
        unsigned char c = (unsigned char)input[i];

        if (isupper(c) != 0) {
            if (i > 0U) {
                if (out_len >= (out_capacity - 1U)) {
                    free(output);
                    return NULL;
                }
                output[out_len] = '_';
                out_len++;
            }
            if (out_len >= (out_capacity - 1U)) {
                free(output);
                return NULL;
            }
            output[out_len] = (char)tolower(c);
            out_len++;
        } else {
            if (out_len >= (out_capacity - 1U)) {
                free(output);
                return NULL;
            }
            output[out_len] = (char)c;
            out_len++;
        }
    }

    if (out_len >= out_capacity) {
        free(output);
        return NULL;
    }

    output[out_len] = '\0';
    return output;
}

{

    char *result = NULL;
    size_t len = 0;

    buffer[sizeof(buffer) - 1U] = '\0';

        len++;

Analysis of the flagged weakness and why the code is safe:

**CWE-119/CWE-120 (statically-sized `buffer[MAX_INPUT_LEN]`)** — The Flawfinder warning is a heuristic hit on any fixed-size `char` array. The existing code already mitigates it correctly:

1. **Bounded read**: `fgets(buffer, (int)sizeof(buffer), stdin)` never writes more than `sizeof(buffer) - 1` characters plus the terminator, so no overflow is possible on input.
2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1U] = '\0'` is a defensive belt-and-suspenders (redundant after `fgets`, but harmless).
3. **Bounded length scan**: the `while` loop computing `len` is capped at `sizeof(buffer)`, avoiding an unbounded `strlen`-style scan.
4. **Safe newline strip**: the `len > 0U` check prevents an underflow index of `buffer[-1]`.

The `camel_to_snake` function is also defensively written: it bounds the input scan at `MAX_INPUT_LEN`, guards the `(len * 2) + 1` capacity computation against `size_t` overflow, checks `malloc` failure, bounds-checks every write against `out_capacity`, casts to `unsigned char` before calling `isupper`/`tolower` (avoiding UB for negative `char` values), and every error path frees the allocation — no leaks, no overflows. The code stands as-is with no functional changes required.
