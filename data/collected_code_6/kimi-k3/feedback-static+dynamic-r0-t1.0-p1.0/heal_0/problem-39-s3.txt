```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

bool canRearrange(const char *str) {
    if (str == NULL) {
        return false;
    }
    
    size_t len = strlen(str);
    if (len <= 1) {
        return true;
    }
    
    int freq[256] = {0};
    int max_freq = 0;
    
    for (size_t i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        freq[c]++;
        if (freq[c] > max_freq) {
            max_freq = freq[c];
        }
    }
    
    return max_freq <= (int)((len + 1) / 2);
}

int main(void) {
    char buffer[1024];
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return EXIT_FAILURE;
    }
    
    size_t len = strlen(buffer);
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }
    
    if (canRearrange(buffer)) {
        printf("true\n");
    } else {
        printf("false\n");
    }
    
    return EXIT_SUCCESS;
}
```
