#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

long long product_of_non_repeated(const int arr[], int n) {
    if (n <= 0) {
        return 0;
    }
    
    int *temp = (int*)malloc(n * sizeof(int));
    if (temp == NULL) {
        return -1;
    }
    
    for (int i = 0; i < n; i++) {
        temp[i] = arr[i];
    }
    
    qsort(temp, n, sizeof(int), compare);
    
    long long product = 1;
    int has_non_repeated = 0;
    
    for (int i = 0; i < n; i++) {
        int count = 1;
        while (i + 1 < n && temp[i] == temp[i + 1]) {
            count++;
            i++;
        }
        if (count == 1) {
            product *= temp[i];
            has_non_repeated = 1;
        }
    }
    
    free(temp);
    
    if (!has_non_repeated) {
        return 0;
    }
    
    return product;
}

int main() {
    int arr1[] = {2, 3, 4, 3, 5, 4, 6};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    printf("%lld\n", product_of_non_repeated(arr1, n1));
    
    int arr2[] = {1, 1, 2, 2, 3, 3};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    printf("%lld\n", product_of_non_repeated(arr2, n2));
    
    int arr3[] = {7};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    printf("%lld\n", product_of_non_repeated(arr3, n3));
    
    int arr4[] = {};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);
    printf("%lld\n", product_of_non_repeated(arr4, n4));
    
    return 0;
}
