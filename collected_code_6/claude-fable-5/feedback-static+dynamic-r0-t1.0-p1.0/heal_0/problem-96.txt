#include <stdlib.h>
#include <limits.h>

int count_divisors(int n)
{
    int count = 0;
    long long v;
    long long i;

    if (n == 0) {
        return -1;
    }

    v = (long long)n;
    if (v < 0) {
        v = -v;
    }

    for (i = 1; i * i <= v; i++) {
        if (v % i == 0) {
            count++;
            if (i != v / i) {
                count++;
            }
        }
    }

    return count;
}

int main(void)
{

    if (scanf("%d", &n) != 1) {
