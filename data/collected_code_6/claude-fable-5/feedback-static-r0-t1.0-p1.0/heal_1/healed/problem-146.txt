```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        sum += (unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long result;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weaknesses:

1. **CWE-126 (over-read via `strlen`)**: Replaced `strlen` with `strnlen`, which takes a maximum length parameter. The function now requires the caller to pass the buffer's capacity, guaranteeing the scan never reads past the buffer even if the string is not null-terminated. If no terminator is found within `max_len` bytes, the function rejects the input by returning `-1`.

2. **CWE-119/CWE-120 (statically-sized buffer)**: The buffer size is defined via a named constant (`MAX_INPUT_LEN`), input is bounded with `fgets(buffer, sizeof(buffer), ...)` which prevents overflow, and `sizeof(buffer)` is used consistently rather than a hard-coded magic number, so the size can't silently desynchronize.

Note that `strnlen` is POSIX (and C23); if strict C90/C99 portability is required, a manual bounded loop can substitute for it.
