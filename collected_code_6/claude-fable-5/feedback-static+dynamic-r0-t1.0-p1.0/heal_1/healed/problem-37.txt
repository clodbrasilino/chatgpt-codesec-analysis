```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef enum {
    ITEM_INT,
    ITEM_STR
} ItemType;

typedef struct {
    ItemType type;
    union {
        long ival;
        char *sval;
    } data;
} Item;

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }
    len = strlen(src) + 1U;
    copy = malloc(len);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, src, len);
    return copy;
}

static int compare_items(const void *a, const void *b)
{
    const Item *ia = (const Item *)a;
    const Item *ib = (const Item *)b;

    if (ia->type != ib->type) {
        return (ia->type == ITEM_INT) ? -1 : 1;
    }
    if (ia->type == ITEM_INT) {
        if (ia->data.ival < ib->data.ival) {
            return -1;
        }
        if (ia->data.ival > ib->data.ival) {
            return 1;
        }
        return 0;
    }
    return strcmp(ia->data.sval, ib->data.sval);
}

static void sort_mixed_list(Item *items, size_t count)
{
    if (items == NULL || count < 2U) {
        return;
    }
    qsort(items, count, sizeof(Item), compare_items);
}

static void free_items(Item *items, size_t count)
{
    size_t i;

    if (items == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        if (items[i].type == ITEM_STR) {
            free(items[i].data.sval);
            items[i].data.sval = NULL;
        }
    }
    free(items);
}

static void add_int(Item *items, size_t index, long value)
{
    items[index].type = ITEM_INT;
    items[index].data.ival = value;
}

static int add_str(Item *items, size_t index, const char *value)
{
    char *copy = duplicate_string(value);

    if (copy == NULL) {
        return -1;
    }
    items[index].type = ITEM_STR;
    items[index].data.sval = copy;
    return 0;
}

static void print_items(const Item *items, size_t count)
{
    size_t i;

    for (i = 0U; i < count; i++) {
        if (items[i].type == ITEM_INT) {
            if (printf("%ld\n", items[i].data.ival) < 0) {
                return;
            }
        } else {
            if (printf("%s\n", items[i].data.sval) < 0) {
                return;
            }
        }
    }
}

int main(void)
{
    const size_t count = 7U;
    size_t filled = 0U;
    Item *items;

    items = calloc(count, sizeof(Item));
    if (items == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    add_int(items, filled, 42L);
    filled++;

    if (add_str(items, filled, "banana") != 0) {
        free_items(items, filled);
        return EXIT_FAILURE;
    }
    filled++;

    add_int(items, filled, -7L);
    filled++;

    if (add_str(items, filled, "apple") != 0) {
        free_items(items, filled);
        return EXIT_FAILURE;
    }
    filled++;

    add_int(items, filled, 13L);
    filled++;

    if (add_str(items, filled, "cherry") != 0) {
        free_items(items, filled);
        return EXIT_FAILURE;
    }
    filled++;

    add_int(items, filled, 0L);
    filled++;

    sort_mixed_list(items, filled);
    print_items(items, filled);
    free_items(items, filled);

    return EXIT_SUCCESS;
}
```

The static-analysis findings (`knownConditionTrueFalse`) were caused by `add_int` unconditionally returning `0`, making every `add_int(...) != 0` check dead code. The fix changes `add_int` to return `void` since it cannot fail, and removes the meaningless error-handling branches. The `add_str` checks are retained because `duplicate_string` can genuinely fail on allocation, so its return value remains meaningful. All other behavior — allocation checks, sorting, printing, and cleanup with proper freeing of string members — is preserved unchanged.
