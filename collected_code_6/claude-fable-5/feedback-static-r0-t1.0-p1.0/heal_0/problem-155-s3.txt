```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static unsigned int toggle_even_bits(unsigned int num)
{
    unsigned int mask = 0U;
    unsigned int bit = 1U;
    size_t i;

    for (i = 0U; i < (sizeof(unsigned int) * CHAR_BIT); i += 2U) {
        mask |= bit;
        if (i + 2U < (sizeof(unsigned int) * CHAR_BIT)) {
            bit <<= 2U;
        }
    }

    return num ^ mask;
}

int main(void)
{
    unsigned int number = 0U;
    unsigned int result = 0U;

    printf("Enter an unsigned integer: ");
    if (scanf("%u", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = toggle_even_bits(number);

    printf("Original number: %u\n", number);
    printf("After toggling even bits: %u\n", result);

    return EXIT_SUCCESS;
}
```

This implementation treats bit position 0 as an even bit position (0, 2, 4, ...), builds the mask portably regardless of the platform's integer width using `CHAR_BIT`, validates the `scanf` return value before using the input, avoids any undefined behavior from shifting beyond the type width, and uses `unsigned int` throughout to guarantee well-defined bitwise semantics.
