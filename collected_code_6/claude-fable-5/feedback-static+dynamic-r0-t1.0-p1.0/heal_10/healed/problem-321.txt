#include <stdio.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 10
#define MAX_OUTPUT_LEN (2 * MAX_INPUT_LEN)

static int findDemlo(const char *str, char *result, size_t result_size);

static int findDemlo(const char *str, char *result, size_t result_size)
{
    size_t len;
    size_t i;
    size_t pos;

    if (str == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);

    if (len == 0 || len > 9) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (str[i] != '1') {
            return -1;
        }
    }

    if (result_size < (2 * len)) {
        return -1;
    }

    pos = 0;
    for (i = 1; i <= len && pos < result_size - 1; i++) {
        result[pos] = (char)('0' + (int)i);
        pos++;
    }
    for (i = len; i > 1 && pos < result_size - 1; i--) {
        result[pos] = (char)('0' + (int)(i - 1));
        pos++;
    }
    result[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN + 2];
    char output[MAX_OUTPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(output, 0, sizeof(output));

    printf("Enter a repunit number (only 1s, max 9 digits): ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Invalid input: too long\n");
        return EXIT_FAILURE;
    }

    if (findDemlo(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid input: must be a repunit of at most 9 ones\n");
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", output);

    return EXIT_SUCCESS;

**1. Statically-sized `input` buffer (CWE-119/CWE-120):** The buffer is sized `MAX_INPUT_LEN + 2` to hold up to 10 digits plus a newline and null terminator. All reads use `fgets` with `sizeof(input)` as the bound, which guarantees no overflow. A forced null terminator is written at `sizeof(input) - 1` as defense in depth, and overly long input is detected and rejected rather than silently truncated.

**2. Statically-sized `output` buffer (CWE-119/CWE-120):** `findDemlo` receives `result_size` explicitly and validates it against the required length (`2 * len`) before writing. Every write in both loops is guarded by `pos < result_size - 1`, and the null terminator is always written within bounds. Since `MAX_OUTPUT_LEN` is 20 and the maximum Demlo string is 17 characters plus terminator, the sizing is correct with margin.

**3. `getchar()` in a loop (CWE-120/CWE-20):** The loop only discards characters until newline or EOF to drain stdin after detecting oversized input—it never stores the characters, so no buffer boundary can be violated. The `int` type for `ch` correctly distinguishes `EOF` from valid characters.
