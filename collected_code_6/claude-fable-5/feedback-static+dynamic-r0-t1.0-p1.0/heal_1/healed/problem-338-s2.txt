```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

long count_substrings(const char *str, size_t max_len)
{
    long count;
    size_t len;
    size_t i;
    size_t j;

    if (str == NULL) {
        return -1L;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        return -1L;
    }

    count = 0L;

    for (i = 0U; i < len; i++) {
        for (j = i; j < len; j++) {
            if (str[i] == str[j]) {
                count++;
            }
        }
    }

    return count;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    }

    result = count_substrings(buffer, sizeof(buffer));
    if (result < 0L) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (printf("%ld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced both calls to `strlen` with `strnlen`, which takes a maximum length bound and can never read past the end of the buffer, even if the string were not null-terminated.

2. **CWE-119/CWE-120 (statically-sized array):** 
   - The buffer size is now defined via a named constant (`BUFFER_SIZE`) for maintainability.
   - The buffer is zero-initialized with `memset` before use, guaranteeing null-termination from the start.
   - An explicit null terminator is forced at `buffer[sizeof(buffer) - 1]` after `fgets` as defense-in-depth.
   - `count_substrings` now receives the buffer's capacity as a parameter and validates the string length against it, so the function no longer trusts the caller to provide a properly terminated string. If `strnlen` hits the bound (meaning no terminator was found within the buffer), it returns an error rather than processing potentially invalid data.

These changes preserve the original program semantics (counting matching character pairs `(i, j)` with `i <= j`) while eliminating the possibility of buffer over-reads flagged by Flawfinder.
