#include <stdio.h>
#include <string.h>

int match_string(const char *str) {
    if (str == NULL) {
        return 0;
    }
    size_t len = strnlen(str, 5);
    if (len < 4 || len > 4) {
        return 0;
    }
    return (str[0] == 'a' && str[1] == 'b' && str[2] == 'b' && str[3] == 'b');
}

int main() {
    const char *test_strings[] = {"abbb", "aabb", "abbbb", "abc", "aabbb"};
    for (size_t i = 0; i < sizeof(test_strings) / sizeof(test_strings[0]); ++i) {
        if (match_string(test_strings[i])) {
            printf("Match: %s\n", test_strings[i]);
        } else {
            printf("No match: %s\n", test_strings[i]);
        }
    }
    return 0;
}
