```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

static void free_tokens(char **tokens, size_t count)
{
    if (tokens == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

static int append_token(char ***tokens, size_t *n, const char *start, size_t len)
{
    if (tokens == NULL || n == NULL || start == NULL) {
        return -1;
    }
    if (len >= SIZE_MAX) {
        return -1;
    }

    char *tok = malloc(len + 1);
    if (tok == NULL) {
        return -1;
    }
    if (len > 0) {
        memcpy(tok, start, len);
    }
    tok[len] = '\0';

    if (*n >= SIZE_MAX / sizeof(**tokens) - 1) {
        free(tok);
        return -1;
    }

    char **tmp = realloc(*tokens, (*n + 1) * sizeof(**tokens));
    if (tmp == NULL) {
        free(tok);
        return -1;
    }
    *tokens = tmp;
    (*tokens)[*n] = tok;
    (*n)++;
    return 0;
}

static char **split_at_uppercase(const char *input, size_t *count)
{
    regex_t re;
    regmatch_t match;
    char **tokens = NULL;
    size_t n = 0;
    const char *cursor;
    const char *end;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    cursor = input;
    end = input + strlen(input);

    if (regcomp(&re, "[A-Z]", REG_EXTENDED) != 0) {
        return NULL;
    }

    while (regexec(&re, cursor, 1, &match, 0) == 0) {
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            goto error;
        }

        size_t len = (size_t)match.rm_so;
        if (len > (size_t)(end - cursor)) {
            goto error;
        }

        if (append_token(&tokens, &n, cursor, len) != 0) {
            goto error;
        }

        if ((size_t)match.rm_eo > (size_t)(end - cursor)) {
            goto error;
        }
        cursor += match.rm_eo;
    }

    {
        size_t len = (size_t)(end - cursor);
        if (append_token(&tokens, &n, cursor, len) != 0) {
            goto error;
        }
    }

    regfree(&re);
    *count = n;
    return tokens;

error:
    free_tokens(tokens, n);
    regfree(&re);
    return NULL;
}

int main(void)
{
    const char *text = "helloWorldThisIsRegexSplit";
    size_t count = 0;
    char **tokens = split_at_uppercase(text, &count);

    if (tokens == NULL) {
        if (fprintf(stderr, "Failed to split string\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < count; i++) {
        if (printf("Token %zu: \"%s\"\n", i, tokens[i]) < 0) {
            free_tokens(tokens, count);
            return EXIT_FAILURE;
        }
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Missing include fixed**: The original code used `SIZE_MAX` without including `<stdint.h>`, which is a compilation defect. This header is now included.

2. **Flawfinder memcpy (CWE-120)**: The `memcpy` call is safe because the destination buffer `tok` is allocated with `malloc(len + 1)` immediately before the copy of exactly `len` bytes, and the `len >= SIZE_MAX` guard prevents integer overflow in the allocation size. The destination is therefore always guaranteed to hold the source data, so the call is retained with its existing bounds guarantees.

3. **Unbounded memchr replaced**: The original `memchr(input, '\0', SIZE_MAX)` scanned up to `SIZE_MAX` bytes, which is undefined behavior territory if applied to a non-terminated buffer and flagged as suspicious by analyzers. It has been replaced with `strlen`, which carries the same precondition (a NUL-terminated string) but expresses intent clearly and avoids the artificial huge bound. All subsequent bounds checks against `end - cursor` remain intact.

4. **Existing defensive measures preserved**: NULL-parameter validation, `regmatch_t` offset sanity checks, overflow guard on the `realloc` size computation, complete cleanup on all error paths via `goto error`, `regfree` on every exit path, and checked return values of `printf`/`fprintf` are all kept, so the program remains leak-free and robust against malformed regex results.
