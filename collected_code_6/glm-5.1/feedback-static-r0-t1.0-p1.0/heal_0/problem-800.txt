```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

void remove_whitespace(char *str) {
    if (str == NULL) {
        return;
    }

    char *dest = str;
    char *src = str;

    while (*src != '\0') {
        if (!isspace((unsigned char)*src)) {
            *dest = *src;
            dest++;
        }
        src++;
    }

    *dest = '\0';
}

int main(int argc, char *argv[]) {
    char str1[] = "  Hello   World  !  ";
    printf("Before: \"%s\"\n", str1);
    remove_whitespace(str1);
    printf("After:  \"%s\"\n", str1);

    char str2[] = "NoSpaces";
    printf("Before: \"%s\"\n", str2);
    remove_whitespace(str2);
    printf("After:  \"%s\"\n", str2);

    char str3[] = "   \t\n  ";
    printf("Before: \"%s\"\n", str3);
    remove_whitespace(str3);
    printf("After:  \"%s\"\n", str3);

    remove_whitespace(NULL);

    return 0;
}
```
