```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

static int binary_to_decimal(const char *binary, size_t max_len, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_line(int *was_truncated)
{
    int ch;
    unsigned long guard = 0UL;

    if (was_truncated == NULL) {
        return -1;
    }

    *was_truncated = 0;

    while ((ch = getchar()) != '\n' && ch != EOF) {
        *was_truncated = 1;
        guard++;
        if (guard == ULONG_MAX) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error: failed to flush output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int truncated = 0;

        if (flush_stdin_line(&truncated) != 0) {
            fprintf(stderr, "Error: internal failure.\n");
            return EXIT_FAILURE;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** in `flush_stdin_line`: This is a false positive in practice — the loop never stores characters into a buffer, it merely discards them. The code additionally guards against unbounded looping with the `guard` counter capped at `ULONG_MAX`, so no fix beyond what exists is needed.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** on `input[INPUT_SIZE]`: The buffer is properly protected. `fgets` is called with `sizeof(input)`, limiting the read; the buffer is zero-initialized with `memset`; a defensive NUL terminator is written at `input[sizeof(input) - 1U]`; and `strnlen` (rather than `strlen`) bounds all length computations. Truncated input is detected via `flush_stdin_line` and rejected rather than silently processed.

3. **Integer overflow protection**: `binary_to_decimal` checks `value > (ULLONG_MAX >> 1)` before shifting, preventing overflow of the `unsigned long long` accumulator, and rejects empty or unterminated strings via the `len == 0U || len >= max_len` check.

All identified weaknesses were already correctly mitigated, so the rewritten code preserves the defensive logic with the comments removed.
