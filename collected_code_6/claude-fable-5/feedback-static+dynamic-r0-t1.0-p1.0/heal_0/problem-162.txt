```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static long long sum_series(int n)
{
    long long sum = 0;
    int current = n;

    while (current > 0) {
        sum += (long long)current;
        current -= 2;
    }

    return sum;
}

int main(void)
{
    int n = 0;
    long long result = 0;

    printf("Enter a positive integer: ");

    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (n <= 0) {
        fprintf(stderr, "Error: input must be a positive integer.\n");
        return EXIT_FAILURE;
    }

    result = sum_series(n);

    if (printf("Sum: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
