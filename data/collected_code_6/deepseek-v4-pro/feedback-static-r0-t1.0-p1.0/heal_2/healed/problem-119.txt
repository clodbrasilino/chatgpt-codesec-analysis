#include <stdio.h>
#include <stdlib.h>

int findSingleElement(int arr[], int size) {
    if (size <= 0) {
        return -1;
    }
    
    if (size == 1) {
        return arr[0];
    }
    
    int left = 0;
    int right = size - 1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (left == right) {
            return arr[left];
        }
        
        if (mid % 2 == 0) {
            if (mid + 1 < size && arr[mid] == arr[mid + 1]) {
                left = mid + 2;
            } else {
                right = mid;
            }
        } else {
            if (arr[mid] == arr[mid - 1]) {
                left = mid + 1;
            } else {
                right = mid - 1;
                if (right < left) {
                    right = left;
                }
            }
        }
    }
    
    return -1;
}
