#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_multiple_spaces(const char *input) {
    if (input == NULL) {
        return NULL;
    }
    
    size_t input_len = strlen(input);
    if (input_len >= 65536) {
        return NULL;
    }
    
    char *result = (char *)malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }
    
    const char *src = input;
    char *dst = result;
    int in_space = 0;
    
    while (*src != '\0') {
        if (isspace((unsigned char)*src)) {
            if (!in_space && dst > result) {
                *dst++ = ' ';
                in_space = 1;
            } else if (dst == result) {
                in_space = 1;
            }
        } else {
            *dst++ = *src;
            in_space = 0;
        }
        src++;
    }
    
    if (dst > result && *(dst - 1) == ' ') {
        dst--;
    }
    
    *dst = '\0';
    
    char *trimmed = (char *)realloc(result, (dst - result) + 1);
    if (trimmed != NULL) {
        return trimmed;
    }
    return result;
}

int main(void) {
    const char *test_cases[] = {
        "Google  Assistant",
        "Quad   Core",
        "ChromeCast    Built-in"
    };
    int num_tests = sizeof(test_cases) / sizeof(test_cases[0]);
    
    for (int i = 0; i < num_tests; i++) {
        char *cleaned = remove_multiple_spaces(test_cases[i]);
        
        if (cleaned != NULL) {
            printf("%s\n", cleaned);
            free(cleaned);
        } else {
            fprintf(stderr, "Error processing string\n");
            return EXIT_FAILURE;
        }
    }
    
    return EXIT_SUCCESS;
}
