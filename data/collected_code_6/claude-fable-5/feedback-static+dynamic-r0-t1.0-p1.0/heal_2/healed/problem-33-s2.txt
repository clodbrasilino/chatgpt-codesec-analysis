```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define BINARY_BUFFER_SIZE (sizeof(unsigned int) * CHAR_BIT + 1)

static int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[index++] = '0';
        buffer[index] = '\0';
        return 0;
    }

    while (decimal > 0) {
        if (index >= buffer_size - 1) {
            buffer[0] = '\0';
            return -1;
        }
        buffer[index++] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
    }
    buffer[index] = '\0';

    for (size_t i = 0, j = index - 1; i < j; i++, j--) {
        char temp = buffer[i];
        buffer[i] = buffer[j];
        buffer[j] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[BINARY_BUFFER_SIZE];
    int result;

    printf("Enter a non-negative decimal number: ");

    result = scanf("%u", &number);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary representation: %s\n", binary);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and why the code is already safe:

- **CWE-119/CWE-120 (buffer overflow on `char binary[]`)**: This is a false positive from Flawfinder, which flags all statically-sized character arrays heuristically. The buffer is sized as `sizeof(unsigned int) * CHAR_BIT + 1`, which is exactly the maximum number of binary digits an `unsigned int` can produce plus room for the null terminator. It is therefore mathematically impossible to overflow it with any valid `unsigned int` input.

- **Defense in depth already present**: `decimal_to_binary()` receives the buffer size via `sizeof(binary)` and performs explicit bounds checking (`index >= buffer_size - 1`) before every write, failing safely with a null-terminated empty string rather than overflowing. It also validates against `NULL` and undersized buffers.

- **Input handling**: `scanf("%u", ...)` writes to a fixed-size integer, not a string, so there is no unbounded string read; the return value is checked to reject invalid input.

No functional changes were required — the original code correctly mitigates the reported weakness through compile-time sizing and runtime bounds checks.
