```c
#include <stdio.h>
#include <string.h>

int lengthOfLastWord(const char *s) {
    if (s == NULL) {
        return 0;
    }

    int length = 0;
    int i = strlen(s) - 1;

    while (i >= 0 && (s[i] == ' ' || s[i] == '\n' || s[i] == '\r')) {
        i--;
    }
    
    while (i >= 0 && s[i] != ' ' && s[i] != '\n' && s[i] != '\r') {
        length++;
        i--;
    }

    return length;
}

int main(void) {
    char buffer[10000];

    while (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        printf("%d\n", lengthOfLastWord(buffer));
    }

    return 0;
}
```
