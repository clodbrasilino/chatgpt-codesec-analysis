#include <stdio.h>
#include <stdlib.h>

int gcd(int a, int b) {
    while (b != 0) {
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

int minOpsToEqual(int x, int y) {
    if (x == y) return 0;
    int result = 0;
    int factor = gcd(x, y);
    x /= factor;
    y /= factor;
    result += abs(x - y);
    return result;
}

int main() {
    int a = 10, b = 15;
    int operations = minOpsToEqual(a, b);
    printf("Minimum operations required: %d\n", operations);
    return 0;
}
