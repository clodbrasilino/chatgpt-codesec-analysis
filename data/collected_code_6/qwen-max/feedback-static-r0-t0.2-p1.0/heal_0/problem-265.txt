#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node* create_node(int value) {
    Node *new_node = (Node*)malloc(sizeof(Node));
    new_node->value = value;
    new_node->next = NULL;
    return new_node;
}

void append_node(Node **head, int value) {
    Node *new_node = create_node(value);
    if (*head == NULL) {
        *head = new_node;
    } else {
        Node *current = *head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
}

void print_list(Node *head) {
    while (head != NULL) {
        printf("%d -> ", head->value);
        head = head->next;
    }
    printf("NULL\n");
}

Node** split_list(Node *head, int n, int *list_count) {
    if (head == NULL || n <= 0) {
        *list_count = 0;
        return NULL;
    }

    Node **lists = (Node**)malloc(sizeof(Node*) * (n + 1));
    for (int i = 0; i < n; ++i) {
        lists[i] = NULL;
    }

    int index = 0;
    while (head != NULL) {
        append_node(&lists[index], head->value);
        head = head->next;
        index = (index + 1) % n;
    }

    *list_count = n;
    return lists;
}

void free_list(Node *head) {
    while (head != NULL) {
        Node *temp = head;
        head = head->next;
        free(temp);
    }
}

void free_lists(Node **lists, int list_count) {
    for (int i = 0; i < list_count; ++i) {
        free_list(lists[i]);
    }
    free(lists);
}

int main() {
    Node *head = NULL;
    for (int i = 1; i <= 10; ++i) {
        append_node(&head, i);
    }

    print_list(head);

    int list_count;
    Node **lists = split_list(head, 3, &list_count);

    for (int i = 0; i < list_count; ++i) {
        print_list(lists[i]);
    }

    free_lists(lists, list_count);
    free_list(head);

    return 0;
}
