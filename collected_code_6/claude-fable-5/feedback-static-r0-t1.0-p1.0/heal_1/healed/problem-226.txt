```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096U

char *remove_odd_index_chars(const char *input);

char *remove_odd_index_chars(const char *input)
{
    size_t len;
    size_t result_len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result_len = (len / 2U) + (len % 2U);

    result = malloc(result_len + 1U);
    if (result == NULL) {
        return NULL;
    }

    j = 0U;
    for (i = 0U; i < len; i += 2U) {
        result[j] = input[i];
        j++;
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "abcdef",
        "hello world",
        "a",
        ""
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0U; i < count; i++) {
        char *output = remove_odd_index_chars(test_strings[i]);
        if (output == NULL) {
            fprintf(stderr, "Error: failed to process string\n");
            return EXIT_FAILURE;
        }
        if (printf("Input: \"%s\" -> Output: \"%s\"\n",
                   test_strings[i], output) < 0) {
            free(output);
            return EXIT_FAILURE;
        }
        free(output);
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. If the input is not null-terminated within the limit, the function safely returns `NULL` instead of reading past the buffer.

2. **Variable scope [variableScope]:** Moved the declaration of `output` inside the `for` loop body, reducing its scope to exactly where it is used. This also makes the `output = NULL;` reset after `free()` unnecessary, since the pointer goes out of scope each iteration.

**Note:** `strnlen` is specified in POSIX.1-2008 and C23; if strict C89/C99 portability is required, a small local bounded-length helper can be substituted with identical semantics.
