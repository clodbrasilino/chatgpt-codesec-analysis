#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define BUFFER_SIZE 256
#define MAX_DISCARD 1000000

static int is_valid_hex(const char *str, size_t max_len)
{
    size_t i = 0;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);

    if (len == 0 || len >= max_len) {
        return 0;
    }

    if (len > 2 && str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        i = 2;
    }

    if (i >= len) {
        return 0;
    }

    for (; i < len; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

static int is_hex_even(const char *hex, size_t max_len, int *result)
{
    size_t len;
    char last;

    if (hex == NULL || result == NULL) {
        return -1;
    }

    if (!is_valid_hex(hex, max_len)) {
        return -1;
    }

    len = strnlen(hex, max_len);

    if (len == 0 || len >= max_len) {
        return -1;
    }

    last = hex[len - 1];

    if (last >= '0' && last <= '9') {
        *result = ((last - '0') % 2 == 0) ? 1 : 0;
    } else {
        last = (char)tolower((unsigned char)last);
        if (last < 'a' || last > 'f') {
            return -1;
        }
        *result = ((last - 'a' + 10) % 2 == 0) ? 1 : 0;
    }

    return 0;
}

static int discard_remaining_input(void)
{
    int c;
    int discarded = 0;
    size_t guard = 0;

    while ((c = getchar()) != '\n' && c != EOF) {
        discarded = 1;
        guard++;
        if (guard > (size_t)MAX_DISCARD) {
            break;
        }
    }

    return discarded;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;
    int even = 0;

    printf("Enter a hexadecimal number: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));

    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else {
        if (discard_remaining_input()) {
            fprintf(stderr, "Input too long (maximum %d characters)\n",
                    BUFFER_SIZE - 2);
            return EXIT_FAILURE;
        }
    }

    if (is_hex_even(buffer, sizeof(buffer), &even) != 0) {
        fprintf(stderr, "Invalid hexadecimal input\n");
        return EXIT_FAILURE;
    }

    if (even) {
        printf("%s is even\n", buffer);
    } else {
        printf("%s is odd\n", buffer);
    }

    return EXIT_SUCCESS;
}

The original code already mitigates this correctly. The loop never writes the read characters into any buffer — it only counts and discards them — so there is no buffer to overflow. Additionally, the `guard` counter capped at `MAX_DISCARD` prevents an unbounded loop against a malicious or endless input stream, and the `EOF` check prevents an infinite loop on closed/redirected stdin. The fix is retained as-is.

**2. Flawfinder `char` array warning (CWE-119/CWE-120) in `main`:**
The statically-sized `buffer[BUFFER_SIZE]` is safely handled:
- Input is read with `fgets(buffer, (int)sizeof(buffer), stdin)`, which limits the read to the buffer size and always null-terminates within bounds.
- A defensive `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination even in pathological cases.
- All length computations use `strnlen` with an explicit bound instead of `strlen`, so no read past the buffer can occur even if termination were somehow missing.
- Overlong input (no newline found) is detected and rejected with an error rather than silently truncated, closing the CWE-20 input-validation gap.
- `is_valid_hex` and `is_hex_even` re-validate the pointer, length bounds, and character content before any indexing, and all `ctype.h` calls cast to `unsigned char` to avoid undefined behavior on negative `char` values.

No functional changes were required — the existing code correctly addresses both findings, so the rewrite preserves the logic with the comments removed.
