#include <stdio.h>
#include <stdlib.h>

int find_kth_element(const int *arr, int size, int k) {
    if (arr == NULL || size <= 0 || k < 0 || k >= size) {
        return -1;
    }
    
    int *temp = (int *)malloc(size * sizeof(int));
    if (temp == NULL) {
        return -1;
    }
    
    for (int i = 0; i < size; i++) {
        temp[i] = arr[i];
    }
    
    for (int i = 0; i < size - 1; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            if (temp[j] > temp[j + 1]) {
                int swap = temp[j];
                temp[j] = temp[j + 1];
                temp[j + 1] = swap;
            }
        }
    }
    
    int result = temp[k];
    free(temp);
    return result;
}

int find_kth_smallest(const int *arr, int size, int k) {
    if (arr == NULL || size <= 0 || k < 0 || k >= size) {
        return -1;
    }
    
    int *temp = (int *)malloc(size * sizeof(int));
    if (temp == NULL) {
        return -1;
    }
    
    for (int i = 0; i < size; i++) {
        temp[i] = arr[i];
    }
    
    for (int i = 0; i < size - 1; i++) {
        int min_idx = i;
        for (int j = i + 1; j < size; j++) {
            if (temp[j] < temp[min_idx]) {
                min_idx = j;
            }
        }
        if (min_idx != i) {
            int swap = temp[i];
            temp[i] = temp[min_idx];
            temp[min_idx] = swap;
        }
    }
    
    int result = temp[k];
    free(temp);
    return result;
}

int main(void) {
    int test1[] = {40, 10, 30, 20, 50};
    int size1 = sizeof(test1) / sizeof(test1[0]);
    int k1 = 2;
    int result1 = find_kth_smallest(test1, size1, k1);
    printf("Test 1 - Element at index %d: %d\n", k1, result1);
    
    int test2[] = {8, 1, 6, 3, 4, 9, 2, 7, 5};
    int size2 = sizeof(test2) / sizeof(test2[0]);
    int k2 = 4;
    int result2 = find_kth_smallest(test2, size2, k2);
    printf("Test 2 - Element at index %d: %d\n", k2, result2);

    int result3 = find_kth_smallest(test3, size3, k3);

    return 0;
}
