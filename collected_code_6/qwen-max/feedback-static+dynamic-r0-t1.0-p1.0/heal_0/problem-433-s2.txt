```c
#include <stdio.h>
#include <stdbool.h>

bool is_greater_than_all(int num, int arr[], int size) {
    for (int i = 0; i < size; i++) {
        if (num <= arr[i]) {
            return false;
        }
    }
    return true;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    int num;

    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    if (is_greater_than_all(num, arr, size)) {
        printf("%d is greater than all elements in the array.\n", num);
    } else {
        printf("%d is not greater than all elements in the array.\n", num);
    }

    return 0;
}
```
