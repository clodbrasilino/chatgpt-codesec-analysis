```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        sum += (long)(unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Total ASCII value: %ld\n", result);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen`, which takes a maximum length bound and cannot read past the end of the buffer even if the string is not null-terminated. The `ascii_sum` function now receives an explicit `max_len` parameter so it never trusts the caller to provide a terminated string.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is defined via `BUFFER_SIZE`, all reads are bounded by `sizeof(buffer)` (already done via `fgets`), and an explicit `buffer[sizeof(buffer) - 1] = '\0';` guarantees null-termination defensively before any string operation occurs.

Note: `strnlen` is POSIX/C23; if you need strict C89/C99 portability, compile with `-D_POSIX_C_SOURCE=200809L` or provide a small fallback implementation.
