```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size);
void free_tuple(Tuple *t);
void print_tuple(const Tuple *t);

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;
    size_t list_bytes;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    list_bytes = list_size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(len + 1);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->str, str, len);
    t->str[len] = '\0';

    if (list_size > 0) {
        t->list = malloc(list_bytes);
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_bytes);
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->str == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *s = "WEB";
    int values[] = {1, 2};
    Tuple *t;

    t = create_tuple(s, values, sizeof(values) / sizeof(values[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

Both Flawfinder warnings on the `memcpy` calls are **false positives**, and the code as written is already safe. Here's why no functional changes were required:

1. **First `memcpy` (string copy):** The destination buffer is allocated with exactly `len + 1` bytes, and precisely `len` bytes are copied, followed by explicit null termination. The length is bounded by `strnlen(str, MAX_STR_LEN)` with a rejection of strings at or above the limit, so the destination can always hold the source data.

2. **Second `memcpy` (list copy):** The destination is allocated with `list_bytes` and exactly `list_bytes` are copied. The multiplication `list_size * sizeof(int)` is guarded against integer overflow via the `SIZE_MAX / sizeof(int)` check before the allocation, eliminating the classic CWE-190 → CWE-120 chain.

The code also correctly handles:
- NULL input validation (including the `list == NULL && list_size > 0` inconsistency case)
- All allocation failure paths with proper cleanup (no leaks)
- The zero-length list case (`t->list = NULL`, and `free(NULL)` is well-defined)
- Safe iteration in `print_tuple` bounded by the stored `list_size`

Flawfinder flags every `memcpy` by pattern matching regardless of context; since the sizes are provably correct here, the appropriate resolution is to verify and document rather than alter the logic.
