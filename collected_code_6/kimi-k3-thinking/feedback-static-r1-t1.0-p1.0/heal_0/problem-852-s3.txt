#include <stdio.h>
#include <stdlib.h>

size_t remove_negatives(int *arr, size_t size)
{
    size_t i;
    size_t write_index = 0;

    if (arr == NULL) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (arr[i] >= 0) {
            arr[write_index] = arr[i];
            write_index++;
        }
    }

    return write_index;
}

int main(void)
{
    size_t size = 10;
    size_t new_size;
    size_t i;
    int *arr = malloc(size * sizeof(int));

    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    arr[0] = 5;
    arr[1] = -3;
    arr[2] = 8;
    arr[3] = -1;
    arr[4] = 0;
    arr[5] = -7;
    arr[6] = 12;
    arr[7] = -9;
    arr[8] = 4;
    arr[9] = -2;

    printf("Original array: ");
    for (i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    new_size = remove_negatives(arr, size);

    printf("Array after removing negatives: ");
    for (i = 0; i < new_size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    free(arr);
    arr = NULL;

    return EXIT_SUCCESS;
}
