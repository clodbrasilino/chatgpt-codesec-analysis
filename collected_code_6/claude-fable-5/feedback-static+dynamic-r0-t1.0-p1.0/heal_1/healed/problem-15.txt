```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t num_tokens = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len = 0;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;

                if (seg_len >= MAX_INPUT_LEN) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                char *segment = malloc(seg_len + 1);
                if (segment == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                memcpy(segment, &input[start], seg_len);
                segment[seg_len] = '\0';

                if (num_tokens >= (SIZE_MAX / sizeof(char *)) - 1) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                temp = realloc(tokens, (num_tokens + 1) * sizeof(char *));
                if (temp == NULL) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }
                tokens = temp;
                tokens[num_tokens] = segment;
                num_tokens++;
            }
            start = i + 1;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "HELLOworldFOObarBAZ123";
    char **tokens = NULL;
    size_t count = 0;
    size_t i;

    tokens = split_at_lowercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens: %zu\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu]: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, rejecting inputs that are not terminated within the limit. This prevents unbounded reads on non-null-terminated buffers.

2. **CWE-120 (memcpy overflow)**: The destination buffer is allocated as `seg_len + 1` immediately before the `memcpy` of exactly `seg_len` bytes, and `seg_len` is now explicitly validated against the bound, so the destination is guaranteed to hold the source data.

3. **Integer overflow in realloc sizing**: Added a check on `num_tokens` against `SIZE_MAX / sizeof(char *)` before computing `(num_tokens + 1) * sizeof(char *)`, preventing a wrap-around that could lead to an undersized allocation.

4. **knownConditionTrueFalse**: Removed the always-true `strlen(test) > 0` check; the error path now simply reports failure and returns `EXIT_FAILURE`, which also fixes the original logic flaw where a `NULL` token array was still dereferenced in the print loop after a failure.

5. Requires `<stdint.h>` semantics for `SIZE_MAX` — it is available via `<stdlib.h>` on many platforms, but for strict portability you may add `#include <stdint.h>`.
