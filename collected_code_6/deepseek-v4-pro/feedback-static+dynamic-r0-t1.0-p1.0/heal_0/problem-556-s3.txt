```c
#include <stdio.h>

int countPairsWithOddXor(int arr[], int n) {
    int oddCount = 0, evenCount = 0;
    
    for (int i = 0; i < n; i++) {
        if (arr[i] % 2 == 0) {
            evenCount++;
        } else {
            oddCount++;
        }
    }
    
    return oddCount * evenCount;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5, 6, 7, 8};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = countPairsWithOddXor(arr, n);
    
    printf("Number of pairs with odd XOR: %d\n", result);
    
    return 0;
}
```
