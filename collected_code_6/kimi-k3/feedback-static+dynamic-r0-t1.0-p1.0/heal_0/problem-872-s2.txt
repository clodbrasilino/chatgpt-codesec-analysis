#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct NestedList {
    int is_integer;
    int integer;
    struct NestedList **list;
    int size;
    int capacity;
} NestedList;

NestedList* create_integer(int value) {
    NestedList *node = (NestedList*)malloc(sizeof(NestedList));
    if (node == NULL) {
        return NULL;
    }
    node->is_integer = 1;
    node->integer = value;
    node->list = NULL;
    node->size = 0;
    node->capacity = 0;
    return node;
}

NestedList* create_list(void) {
    NestedList *node = (NestedList*)malloc(sizeof(NestedList));
    if (node == NULL) {
        return NULL;
    }
    node->is_integer = 0;
    node->integer = 0;
    node->list = NULL;
    node->size = 0;
    node->capacity = 0;
    return node;
}

int add_to_list(NestedList *parent, NestedList *child) {
    if (parent == NULL || child == NULL || parent->is_integer) {
        return 0;
    }
    
    if (parent->size >= parent->capacity) {
        int new_capacity = parent->capacity == 0 ? 4 : parent->capacity * 2;
        NestedList **new_list = (NestedList**)realloc(parent->list, 
                                                       new_capacity * sizeof(NestedList*));
        if (new_list == NULL) {
            return 0;
        }
        parent->list = new_list;
        parent->capacity = new_capacity;
    }
    
    parent->list[parent->size] = child;
    parent->size++;
    return 1;
}

int is_equal(NestedList *a, NestedList *b) {
    if (a == NULL || b == NULL) {
        return 0;
    }
    
    if (a->is_integer != b->is_integer) {
        return 0;
    }
    
    if (a->is_integer) {
        return a->integer == b->integer;
    }
    
    if (a->size != b->size) {
        return 0;
    }
    
    for (int i = 0; i < a->size; i++) {
        if (!is_equal(a->list[i], b->list[i])) {
            return 0;
        }
    }
    
    return 1;
}

int is_subset(NestedList *subset, NestedList *superset) {
    if (subset == NULL || superset == NULL) {
        return 0;
    }
    
    if (subset->is_integer && superset->is_integer) {
        return subset->integer == superset->integer;
    }
    
    if (subset->is_integer) {
        return 0;
    }
    
    if (!superset->is_integer) {
        for (int i = 0; i < subset->size; i++) {
            int found = 0;
            for (int j = 0; j < superset->size; j++) {
                if (is_equal(subset->list[i], superset->list[j]) ||
                    is_subset(subset->list[i], superset->list[j])) {
                    found = 1;
                    break;
                }
            }
            if (!found) {
                return 0;
            }
        }
        return 1;
    }
    
    return 0;
}

void free_nested_list(NestedList *node) {
    if (node == NULL) {
        return;
    }
    
    if (!node->is_integer && node->list != NULL) {
        for (int i = 0; i < node->size; i++) {
            free_nested_list(node->list[i]);
        }
        free(node->list);
    }
    
    free(node);
}
