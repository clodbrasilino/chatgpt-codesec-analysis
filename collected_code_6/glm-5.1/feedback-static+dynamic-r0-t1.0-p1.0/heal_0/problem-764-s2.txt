```c
#include <stdio.h>

int count_numeric(const char *str) {
    if (str == NULL) {
        return 0;
    }

    int count = 0;
    for (size_t i = 0; str[i] != '\0'; i++) {
        if (str[i] >= '0' && str[i] <= '9') {
            count++;
        }
    }
    return count;
}

int main(void) {
    const char *text = "Hello 123 World 456!";
    int result = count_numeric(text);
    printf("%d\n", result);
    return 0;
}
```
