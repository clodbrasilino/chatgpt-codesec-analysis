#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stddef.h>

void swap(char *a, char *b) {
    char temp = *a;
    *a = *b;
    *b = temp;
}

void permute(char *str, int left, int right) {
    if (left == right) {
        printf("%s\n", str);
        return;
    }

    for (int i = left; i <= right; i++) {
        swap(&str[left], &str[i]);
        permute(str, left + 1, right);
        swap(&str[left], &str[i]);
    }
}

int main(void) {
    char input[256] = {0};

    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }

    size_t newline_pos = strcspn(input, "\n");
    if (newline_pos < sizeof(input)) {
        input[newline_pos] = '\0';
    } else {
        input[sizeof(input) - 1] = '\0';
    }

    size_t len = strlen(input);

    if (len == 0) {
        return 0;
    }

    if (len >= sizeof(input)) {
        return 1;
    }

    permute(input, 0, (int)(len - 1));

    return 0;
}
