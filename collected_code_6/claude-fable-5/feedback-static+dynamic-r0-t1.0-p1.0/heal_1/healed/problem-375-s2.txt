```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long round_to_nearest_multiple(long number, long multiple)
{
    long remainder;
    long half;
    long up;
    long down;

    if (multiple == 0) {
        return number;
    }

    if (multiple == LONG_MIN) {
        if (number == LONG_MIN) {
            return number;
        }
        if (number <= LONG_MIN / 2) {
            return LONG_MIN;
        }
        return 0;
    }

    if (multiple < 0) {
        multiple = -multiple;
    }

    remainder = number % multiple;

    if (remainder == 0) {
        return number;
    }

    half = multiple / 2;

    if (number >= 0) {
        if (remainder >= multiple - half) {
            up = multiple - remainder;
            if (number > LONG_MAX - up) {
                return number - remainder;
            }
            return number + up;
        }
        return number - remainder;
    }

    if (-remainder >= multiple - half) {
        down = multiple + remainder;
        if (number < LONG_MIN + down) {
            return number - remainder;
        }
        return number - down;
    }
    return number - remainder;
}

int main(void)
{
    long number;
    long multiple;
    long result;

    printf("Enter the number: ");
    if (scanf("%ld", &number) != 1) {
        fprintf(stderr, "Invalid input for number\n");
        return EXIT_FAILURE;
    }

    printf("Enter the multiple: ");
    if (scanf("%ld", &multiple) != 1) {
        fprintf(stderr, "Invalid input for multiple\n");
        return EXIT_FAILURE;
    }

    if (multiple == 0) {
        fprintf(stderr, "Multiple must not be zero\n");
        return EXIT_FAILURE;
    }

    result = round_to_nearest_multiple(number, multiple);

    printf("%ld rounded to the nearest multiple of %ld is %ld\n", number, multiple, result);

    return EXIT_SUCCESS;
}
```

Key changes made to eliminate the undefined behavior reported by UBSan/AFL:

1. **`LONG_MIN` negation** — `multiple == LONG_MIN` is now handled as a dedicated special case before the `-multiple` negation, since `-LONG_MIN` cannot be represented in a `long`. The only representable multiples of `|LONG_MIN|` are `0` and `LONG_MIN` itself, so the function rounds to whichever is nearer.

2. **Positive overflow on round-up** — before computing `number + (multiple - remainder)`, the code checks `number > LONG_MAX - up`. If rounding up would overflow, it safely falls back to rounding down (toward zero), which is always representable.

3. **Negative overflow on round-down** — before computing `number - (multiple + remainder)`, the code checks `number < LONG_MIN + down`. If rounding away from zero would underflow, it falls back to rounding toward zero, which is always safe.

Note that `-remainder` in the negative branch is safe because `|remainder| < multiple <= LONG_MAX`, and `multiple + remainder` cannot overflow since `remainder` is negative and strictly greater than `-multiple`.
