#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *merge_sorted_lists(const Node *list1, const Node *list2);
void free_list(Node *head);
static Node *create_node(int data);
static Node *build_list(const int *values, size_t count);
static void print_list(const Node *head);

static Node *create_node(int data)
{
    Node *node = malloc(sizeof(*node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

static Node *build_list(const int *values, size_t count)
{
    Node *head = NULL;
    Node *tail = NULL;
    size_t i;

    for (i = 0; i < count; i++) {
        Node *node = create_node(values[i]);
        if (node == NULL) {
            free_list(head);
            return NULL;
        }
        if (tail == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }

    return head;
}

Node *merge_sorted_lists(const Node *list1, const Node *list2)
{
    Node *head = NULL;
    Node *tail = NULL;

    while (list1 != NULL || list2 != NULL) {
        int value;
        Node *node;

        if (list2 == NULL || (list1 != NULL && list1->data <= list2->data)) {
            value = list1->data;
            list1 = list1->next;
        } else {
            value = list2->data;
            list2 = list2->next;
        }

        node = create_node(value);
        if (node == NULL) {
            free_list(head);
            return NULL;
        }

        if (tail == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }

    return head;
}

static void print_list(const Node *head)
{
    const Node *current = head;

    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

int main(void)
{
    int values1[] = {1, 3, 5, 7, 9};
    int values2[] = {2, 4, 6, 8, 10, 12};
    Node *list1;
    Node *list2;
    Node *merged;

    list1 = build_list(values1, sizeof(values1) / sizeof(values1[0]));
    if (list1 == NULL) {
        fprintf(stderr, "Failed to build first list\n");
        return EXIT_FAILURE;
    }

    list2 = build_list(values2, sizeof(values2) / sizeof(values2[0]));
    if (list2 == NULL) {
        fprintf(stderr, "Failed to build second list\n");
        free_list(list1);
        return EXIT_FAILURE;
    }

    merged = merge_sorted_lists(list1, list2);
    if (merged == NULL) {
        fprintf(stderr, "Failed to merge lists\n");
        free_list(list1);
        free_list(list2);
        return EXIT_FAILURE;
    }

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);
    printf("Merged: ");
    print_list(merged);

    free_list(list1);
    free_list(list2);
    free_list(merged);

    return EXIT_SUCCESS;
}
