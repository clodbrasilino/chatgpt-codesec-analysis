```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 1024U

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static int permute(char *str, size_t left, size_t right)
{
    size_t i;

    if (left == right) {
        if (puts(str) == EOF) {
            return -1;
        }
        return 0;
    }

    for (i = left; i <= right; i++) {
        swap_chars(&str[left], &str[i]);
        if (permute(str, left + 1U, right) != 0) {
            swap_chars(&str[left], &str[i]);
            return -1;
        }
        swap_chars(&str[left], &str[i]);
    }

    return 0;
}

static int print_permutations(const char *input)
{
    char *buffer;
    size_t len;
    size_t buf_size;
    int result;

    if (input == NULL) {
        return -1;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == 0U || len >= MAX_INPUT_LEN) {
        return -1;
    }

    buf_size = len + 1U;

    buffer = malloc(buf_size);
    if (buffer == NULL) {
        return -1;
    }

    memcpy(buffer, input, len);
    buffer[len] = '\0';

    result = permute(buffer, 0U, len - 1U);

    free(buffer);

    return result;
}

int main(void)
{
    const char *test = "ABC";

    if (print_permutations(test) != 0) {
        (void)fprintf(stderr, "Failed to print permutations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Removed the dead condition `len >= buf_size`**: Since `buf_size = len + 1U` and `len < MAX_INPUT_LEN` guarantees no overflow of the `size_t` arithmetic, this condition was provably always false (as cppcheck's `knownConditionTrueFalse` flagged). Dead code removed.

2. **`memcpy` (CWE-120, Flawfinder)**: This is a false positive in this context, but the code is now structured so safety is provable: `len` is bounded by `strnlen(input, MAX_INPUT_LEN)` and validated (`len < MAX_INPUT_LEN`), the destination is allocated with exactly `len + 1U` bytes, and exactly `len` bytes are copied followed by explicit NUL termination. The destination can always hold the source data by construction.

3. **`malloc` NULL check retained**: The static analyzer noted "assuming condition is false," but the check is correct defensive practice and remains — allocation failure is a genuine runtime possibility.

4. **`buf_size = len + 1U` overflow concern**: The prior validation `len >= MAX_INPUT_LEN → return -1` ensures `len <= MAX_INPUT_LEN - 1`, so `len + 1U` cannot wrap; the assignment is safe.

The permutation logic itself (recursive backtracking with swap/restore, error propagation from `puts`) was sound and left intact.
