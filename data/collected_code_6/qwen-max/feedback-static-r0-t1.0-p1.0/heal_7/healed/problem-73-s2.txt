#include <stdio.h>
#include <stdlib.h>
#include <regex.h>
#include <string.h>

int split_string(const char *input, const char *delimiters, char ***result) {
    regex_t regex;
    regmatch_t match[1];
    size_t nmatch = 1;
    char *str;
    char **parts = NULL;
    int count = 0, part_idx = 0;

    if (regcomp(&regex, delimiters, REG_EXTENDED | REG_NEWLINE) != 0) {
        return -1;
    }

    str = strdup(input);
    if (str == NULL) {
        regfree(&regex);
        return -1;
    }

    while (1) {
        int status = regexec(&regex, str, nmatch, match, 0);
        if (status != 0) break;

        char **temp = realloc(parts, (count + 2) * sizeof(char*));
        if (temp == NULL) {
            free(str);
            for (int i = 0; i < part_idx; i++) {
                free(parts[i]);
            }
            free(parts);
            regfree(&regex);
            return -1;
        }
        parts = temp;

        if (match[0].rm_so > 0) {
            parts[part_idx] = strndup(str, match[0].rm_so);
            if (parts[part_idx] == NULL) {
                free(str);
                for (int i = 0; i < part_idx; i++) {
                    free(parts[i]);
                }
                free(parts);
                regfree(&regex);
                return -1;
            }
            part_idx++;
            count++;
        }

        str += match[0].rm_eo;
    }

    if (*str != '\0') {
        char **temp = realloc(parts, (count + 1) * sizeof(char*));
        if (temp == NULL) {
            free(str);
            for (int i = 0; i < part_idx; i++) {
                free(parts[i]);
            }
            free(parts);
            regfree(&regex);
            return -1;
        }
        parts = temp;

        parts[part_idx] = strdup(str);
        if (parts[part_idx] == NULL) {
            free(str);
            for (int i = 0; i < part_idx; i++) {
                free(parts[i]);
            }
            free(parts);
            regfree(&regex);
            return -1;
        }
        part_idx++;
        count++;
    }

    char **temp = realloc(parts, (count + 1) * sizeof(char*));
    if (temp == NULL) {
        free(str);
        for (int i = 0; i < part_idx; i++) {
            free(parts[i]);
        }
        free(parts);
        regfree(&regex);
        return -1;
    }
    parts = temp;

    parts[count] = NULL;
    *result = parts;

    free(str);
    regfree(&regex);
    return count;
}

int main() {
    const char *input = "This,is a test;split string:with multiple,delimiters";
    const char *delimiters = ",|;|:|\\s+";
    char **result = NULL;
    int count = split_string(input, delimiters, &result);

    if (count >= 0) {
        for (int i = 0; result[i] != NULL; i++) {
            printf("%s\n", result[i]);
            free(result[i]);
        }
        free(result);
    } else {
        fprintf(stderr, "Split failed\n");
    }

    return 0;
}
