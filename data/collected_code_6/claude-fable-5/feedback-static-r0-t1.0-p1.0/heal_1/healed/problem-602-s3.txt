```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define MAX_INPUT_LEN 256

int find_first_repeated_char(const char *str, size_t max_len, char *result)
{
    int seen[UCHAR_MAX + 1] = {0};
    size_t i;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char repeated = '\0';
    int status;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    status = find_first_repeated_char(buffer, sizeof(buffer), &repeated);

    if (status < 0) {
        fprintf(stderr, "Invalid arguments\n");
        return 1;
    }

    if (status == 0) {
        if (printf("First repeated character: %c\n", repeated) < 0) {
            return 1;
        }
    } else {
        if (printf("No repeated character found\n") < 0) {
            return 1;
        }
    }

    return 0;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (strlen over-read):** Removed the `strlen` call entirely. The function now takes an explicit `max_len` parameter and the loop is bounded by both `max_len` and the null terminator (`i < max_len && str[i] != '\0'`), so even a non-terminated buffer cannot cause an over-read beyond the stated size.

2. **CWE-119/CWE-120 (statically-sized array):** The buffer size is defined once as `MAX_INPUT_LEN` and consistently referenced via `sizeof(buffer)`. The buffer is zero-initialized with `memset`, and null termination is explicitly guaranteed with `buffer[sizeof(buffer) - 1] = '\0'` as a defensive measure before any string operations. `fgets` already limits input to the buffer size, preventing overflow.
