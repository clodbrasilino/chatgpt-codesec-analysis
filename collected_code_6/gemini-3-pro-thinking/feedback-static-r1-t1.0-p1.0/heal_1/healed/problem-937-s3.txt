#include <stdio.h>
#include <stddef.h>

char get_most_common_character(const char *str) {
    if (str == NULL || *str == '\0') {
        return '\0';
    }

    unsigned int frequencies[256] = {0};
    unsigned int max_frequency = 0;
    char most_common = '\0';

    while (*str != '\0') {
        unsigned char current_char = (unsigned char)(*str);
        frequencies[current_char]++;
        
        if (frequencies[current_char] > max_frequency) {
            max_frequency = frequencies[current_char];
            most_common = (char)current_char;
        }
        str++;
    }

    return most_common;
}

int main(void) {
    const char *test_strings[] = {
        "hello world",
        NULL,
        ""
    };
    
    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);

    for (size_t i = 0; i < num_tests; i++) {
        const char *current_str = test_strings[i];
        char result = get_most_common_character(current_str);

        if (result != '\0') {
            printf("%c\n", result);
        } else if (current_str == NULL) {
            printf("Error: Invalid string\n");
        } else if (*current_str == '\0') {
            printf("Error: Empty string\n");
        }
    }

    return 0;
}
