#include <stdio.h>
#include <stdlib.h>

long long sum_of_odd_factors(long long n) {
    if (n == 0) {
        return 0;
    }

    unsigned long long un = n < 0 ? -(unsigned long long)n : (unsigned long long)n;

    while (un % 2 == 0) {
        un /= 2;
    }

    unsigned long long sum = 1;
    unsigned long long i = 3;
    
    while (i * i <= un) {
        unsigned long long current_sum = 1;
        unsigned long long current_term = 1;
        while (un % i == 0) {
            current_term *= i;
            current_sum += current_term;
            un /= i;
        }
        sum *= current_sum;
        i += 2;
    }

    if (un > 1) {
        sum *= (1 + un);
    }

    return (long long)sum;
}

int main(void) {
    long long n;
    
    if (scanf("%lld", &n) != 1) {
        return EXIT_FAILURE;
    }

    long long result = sum_of_odd_factors(n);
    printf("%lld\n", result);

    return EXIT_SUCCESS;
}
