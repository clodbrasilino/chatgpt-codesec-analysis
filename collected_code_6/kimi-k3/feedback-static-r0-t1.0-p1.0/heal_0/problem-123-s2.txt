#include <stdio.h>
#include <stdlib.h>

int sum_of_divisors(int n) {
    int sum = 1;
    int i;
    if (n <= 1) {
        return 0;
    }
    for (i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i != n / i) {
                sum += n / i;
            }
        }
    }
    return sum;
}

long long sum_amicable_numbers(int limit) {
    long long total = 0;
    int i;
    if (limit < 1) {
        return 0;
    }
    for (i = 1; i <= limit; i++) {
        int partner = sum_of_divisors(i);
        if (partner > i && partner <= limit) {
            if (sum_of_divisors(partner) == i) {
                total += i + partner;
            }
        }
    }
    return total;
}

int main(void) {
    int limit;
    long long result;
    if (scanf("%d", &limit) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    if (limit < 1) {
        fprintf(stderr, "Limit must be positive\n");
        return EXIT_FAILURE;
    }
    result = sum_amicable_numbers(limit);
    printf("%lld\n", result);
    return EXIT_SUCCESS;
}
