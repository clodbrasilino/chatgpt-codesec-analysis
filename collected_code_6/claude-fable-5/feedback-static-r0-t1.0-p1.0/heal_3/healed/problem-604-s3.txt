```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT 256

static void reverse_range(char *str, size_t start, size_t end)
{
    while (start < end) {
        char tmp = str[start];
        str[start] = str[end];
        str[end] = tmp;
        start++;
        end--;
    }
}

int reverse_words(char *str, size_t max_len)
{
    size_t len;
    size_t i;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }
    if (len == 0) {
        return 0;
    }

    reverse_range(str, 0, len - 1);

    i = 0;
    while (i < len) {
        while (i < len && str[i] == ' ') {
            i++;
        }
        if (i >= len) {
            break;
        }
        size_t word_start = i;
        while (i < len && str[i] != ' ') {
            i++;
        }
        reverse_range(str, word_start, i - 1);
    }

    return 0;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char *input;
    char *newline;

    input = calloc(MAX_INPUT, sizeof(char));
    if (input == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter a string: ");
    if (fgets(input, MAX_INPUT, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(input);
        return EXIT_FAILURE;
    }

    input[MAX_INPUT - 1] = '\0';

    newline = strchr(input, '\n');
    if (newline != NULL) {
        *newline = '\0';
    } else {
        discard_remaining_input();
    }

    if (reverse_words(input, MAX_INPUT) != 0) {
        fprintf(stderr, "Error reversing words\n");
        free(input);
        return EXIT_FAILURE;
    }

    printf("Reversed: %s\n", input);

    free(input);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended; the Flawfinder hit on `getchar` (CWE-120/CWE-20) is a low-risk, largely informational warning. The key points verified and preserved in the rewrite:

1. **`getchar` drain loop** — the flagged weakness. The loop reads into an `int` and discards each character without storing it in any buffer, so no buffer overflow is possible. It correctly terminates on both `'\n'` and `EOF`, preventing an infinite loop if stdin closes. I extracted it into `discard_remaining_input()` for clarity and bounded, auditable behavior.

2. **Bounded input** — `fgets(input, MAX_INPUT, stdin)` already prevents overflow, and the explicit `input[MAX_INPUT - 1] = '\0'` guarantees termination.

3. **`strnlen` guard** — `reverse_words` rejects unterminated buffers (`len == max_len`) and NULL/zero-length arguments, avoiding out-of-bounds reads.

4. **Memory hygiene** — `calloc` result checked, `free` called on every exit path; no leaks or double-frees.

No functional changes were required — the rewrite is a structural cleanup confirming the flagged pattern is safe.
