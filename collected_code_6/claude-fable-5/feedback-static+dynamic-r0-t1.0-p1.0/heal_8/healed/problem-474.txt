```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_SOURCE_LEN 1024

size_t replace_chars(char *str, char old_char, char new_char);

size_t replace_chars(char *str, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    if (old_char == '\0' || new_char == '\0') {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    const char *source = "hello world, hello everyone";
    size_t len;
    size_t buffer_size;
    char *buffer;
    size_t replaced;
    int written;

    len = strnlen(source, MAX_SOURCE_LEN);
    if (len >= MAX_SOURCE_LEN) {
        fprintf(stderr, "Source string too long or not null-terminated\n");
        return EXIT_FAILURE;
    }

    if (len > SIZE_MAX - 1) {
        fprintf(stderr, "Source length overflow\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    buffer = calloc(buffer_size, sizeof(char));
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, buffer_size, "%s", source);
    if (written < 0 || (size_t)written >= buffer_size) {
        fprintf(stderr, "Buffer too small for source data\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", buffer);

    replaced = replace_chars(buffer, 'l', 'L');

    printf("Modified: %s\n", buffer);
    printf("Replacements made: %zu\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

**Summary of changes made:**

1. **Removed the always-false condition** (`len + 1 > buffer_size`): Since `buffer_size` was assigned `len + 1` immediately before, the check was dead code flagged by `knownConditionTrueFalse`. It has been removed entirely.

2. **Replaced `memcpy` with `snprintf`**: This addresses the Flawfinder CWE-120 warning. `snprintf` is bounds-aware, guarantees null-termination within `buffer_size`, and its return value is checked to detect truncation — providing a meaningful runtime safety check instead of the removed dead-code check.

3. **Added an explicit overflow guard** (`len > SIZE_MAX - 1`): This protects the `len + 1` computation against unsigned wraparound. While it can never trigger given the `MAX_SOURCE_LEN` check, it makes the safety of the arithmetic explicit and robust against future changes to the length-validation logic.

4. **Removed the now-redundant manual null-termination** (`buffer[buffer_size - 1] = '\0'`): `snprintf` handles termination, and `calloc` already zero-initializes the buffer as a defense-in-depth measure.

The `replace_chars` function was left unchanged, as it already correctly guards against `NULL` pointers and null-character arguments, and iterates strictly within the string bounds.
