```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(const char *input, char *output, size_t output_size);

int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    size_t remaining;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len = (size_t)match.rm_so;

        if ((prefix_len >= output_size) ||
            (out_pos >= (output_size - prefix_len - 1U))) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        memcpy(&output[out_pos], cursor, prefix_len);
        out_pos += prefix_len;
        output[out_pos] = ' ';
        out_pos++;

        cursor += match.rm_eo;
    }

    remaining = strnlen(cursor, output_size);

    if ((remaining >= output_size) ||
        (out_pos >= (output_size - remaining))) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    memcpy(&output[out_pos], cursor, remaining);
    out_pos += remaining;
    output[out_pos] = '\0';

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256] = {0};

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes from the original:

- **`strlen` → `strnlen`**: bounds the read of the remaining input to `output_size`, so a non-NUL-terminated string can no longer cause an unbounded over-read (CWE-126).
- **Overflow-safe bounds checks**: the checks `(out_pos + prefix_len + 1U) >= output_size` and `(out_pos + remaining) >= output_size` could wrap around on `size_t` overflow. They are rewritten as subtraction-based comparisons (`out_pos >= output_size - prefix_len - 1U`) with a preliminary guard (`prefix_len >= output_size`), which cannot overflow. This makes every `memcpy` provably within bounds (CWE-120/CWE-119).
- **Defensive initialization**: `output[0] = '\0'` at entry and on every error path guarantees the caller never reads an uninitialized buffer after a failure; `result` in `main` is zero-initialized for the same reason.
- The statically sized `result[256]` is retained but is safe because its exact size is passed via `sizeof(result)` and enforced by the checked copy logic.
