#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

char *concatenate(const char *list[], size_t count)
{
    size_t total_length = 0;
    size_t offset = 0;
    size_t i;
    char *result = NULL;
    size_t *lengths = NULL;

    if (list == NULL && count > 0)
    {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(size_t))
    {
        return NULL;
    }

    if (count > 0)
    {
        lengths = (size_t *)malloc(count * sizeof(size_t));
        if (lengths == NULL)
        {
            return NULL;
        }
    }

    for (i = 0; i < count; i++)
    {
        size_t len;

        if (list[i] == NULL)
        {
            free(lengths);
            return NULL;
        }

        len = strnlen(list[i], MAX_STRING_LENGTH);
        if (len >= MAX_STRING_LENGTH)
        {
            free(lengths);
            return NULL;
        }

        if (len >= SIZE_MAX - total_length)
        {
            free(lengths);
            return NULL;
        }

        lengths[i] = len;
        total_length += len;
    }

    if (total_length >= SIZE_MAX)
    {
        free(lengths);
        return NULL;
    }

    result = (char *)calloc(total_length + 1, sizeof(char));
    if (result == NULL)
    {
        free(lengths);
        return NULL;
    }

    for (i = 0; i < count; i++)
    {
        size_t len = lengths[i];

        if (len > 0)
        {
            if (offset > total_length || len > total_length - offset)
            {
                free(lengths);
                free(result);
                return NULL;
            }

            if (strnlen(list[i], len + 1) != len)
            {
                free(lengths);
                free(result);
                return NULL;
            }

            memcpy(result + offset, list[i], len);
            offset += len;
        }
    }

    if (offset != total_length)
    {
        free(lengths);
        free(result);
        return NULL;
    }

    result[offset] = '\0';
    free(lengths);

    return result;
}

int main(void)
{
    const char *list[] = {"Hello", " ", "World", "!"};
    size_t count = sizeof(list) / sizeof(list[0]);
    char *result = concatenate(list, count);

    if (result == NULL)
    {
        fprintf(stderr, "Error: concatenation failed\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", result);
    free(result);

    return EXIT_SUCCESS;
}
