#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int *data;
    size_t size;
} IntList;

int modulo_lists(const IntList *list1, const IntList *list2, IntList *result);
void free_list(IntList *list);

int modulo_lists(const IntList *list1, const IntList *list2, IntList *result) {
    if (list1 == NULL || list2 == NULL || result == NULL) {
        return -1;
    }

    if (list1->data == NULL || list2->data == NULL) {
        return -1;
    }

    if (list1->size != list2->size) {
        return -2;
    }

    for (size_t i = 0; i < list2->size; i++) {
        if (list2->data[i] == 0) {
            return -3;
        }
        if (list1->data[i] == INT_MIN && list2->data[i] == -1) {
            return -4;
        }
    }

    result->data = (int *)malloc(list1->size * sizeof(int));
    if (result->data == NULL) {
        return -5;
    }
    
    result->size = list1->size;

    for (size_t i = 0; i < list1->size; i++) {
        result->data[i] = list1->data[i] % list2->data[i];
    }

    return 0;
}

void free_list(IntList *list) {
    if (list != NULL) {
        if (list->data != NULL) {
            free(list->data);
            list->data = NULL;
        }
        list->size = 0;
    }
}

int main(void) {
    int arr1[] = {10, 25, -33, 42, INT_MIN};
    int arr2[] = {3, 7, 5, 8, 2};
    
    IntList list1 = {arr1, 5};
    IntList list2 = {arr2, 5};
    IntList result = {NULL, 0};

    int status = modulo_lists(&list1, &list2, &result);
    
    if (status == 0) {
        for (size_t i = 0; i < result.size; i++) {
            printf("%d\n", result.data[i]);
        }
        free_list(&result);
    } else {
        printf("Error code: %d\n", status);
    }

    return status;
}
