```c
#include <stdio.h>
#include <stdbool.h>

bool check_elements_greater(const int first[], const int second[], size_t size) {
    if (first == NULL || second == NULL) {
        return false;
    }
    
    for (size_t i = 0; i < size; i++) {
        if (second[i] <= first[i]) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int first[] = {1, 2, 3, 4, 5};
    int second[] = {2, 3, 4, 5, 6};
    size_t size = sizeof(first) / sizeof(first[0]);
    
    if (size != sizeof(second) / sizeof(second[0])) {
        fprintf(stderr, "Array size mismatch\n");
        return 1;
    }
    
    if (check_elements_greater(first, second, size)) {
        printf("All elements in second tuple are greater than corresponding elements in first tuple\n");
    } else {
        printf("Not all elements in second tuple are greater than corresponding elements in first tuple\n");
    }
    
    return 0;
}
```
