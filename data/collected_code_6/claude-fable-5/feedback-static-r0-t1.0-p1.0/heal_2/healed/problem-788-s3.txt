```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_NAME_LEN 4096

typedef struct {
    char *name;
    int *values;
    size_t count;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t count)
{
    Tuple *t;
    size_t len;

    if (str == NULL || (list == NULL && count > 0)) {
        return NULL;
    }

    len = strnlen(str, MAX_NAME_LEN);
    if (len == MAX_NAME_LEN) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->name = malloc(len + 1);
    if (t->name == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->name, str, len);
    t->name[len] = '\0';

    if (count > 0) {
        t->values = malloc(count * sizeof(int));
        if (t->values == NULL) {
            free(t->name);
            free(t);
            return NULL;
        }
        memcpy(t->values, list, count * sizeof(int));
    } else {
        t->values = NULL;
    }

    t->count = count;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->name);
        free(t->values);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL) {
        return;
    }

    printf("(\"%s\", [", t->name);
    for (i = 0; i < t->count; i++) {
        printf("%d", t->values[i]);
        if (i + 1 < t->count) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    int numbers[] = {10, 20, 30, 40};
    Tuple *t;

    t = create_tuple("example", numbers, sizeof(numbers) / sizeof(numbers[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}

The size arithmetic is also guarded against overflow via the `count > SIZE_MAX / sizeof(int)` check, and the string length is bounded by `strnlen` with `MAX_NAME_LEN` rejecting unterminated or oversized input. All allocation failures are handled with correct cleanup (no leaks on partial construction), and `free_tuple` safely tolerates NULL.
