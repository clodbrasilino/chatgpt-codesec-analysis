#include <stdio.h>

int is_prime(int n) {
    if (n < 2) return 0;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return 0;
    }
    return 1;
}

int sum_prime_divisors(int n) {
    int sum = 0;
    if (n < 0) n = -n;
    
    for (int i = 2; i <= n; i++) {
        if (n % i == 0 && is_prime(i)) {
            sum += i;
        }
    }
    
    return sum;
}

int main(void) {
    int num;
    
    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    int result = sum_prime_divisors(num);
    printf("Sum of prime divisors: %d\n", result);
    
    return 0;
}
