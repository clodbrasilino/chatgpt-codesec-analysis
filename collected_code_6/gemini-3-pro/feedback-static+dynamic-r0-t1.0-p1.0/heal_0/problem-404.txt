#include <stdio.h>

int get_min(int a, int b);

int get_min(int a, int b) {
    return (a < b) ? a : b;
}

int main(void) {
    int num1 = 0;
    int num2 = 0;
    int min_value = 0;

    if (scanf("%d %d", &num1, &num2) != 2) {
        return 1;
    }

    min_value = get_min(num1, num2);

    if (printf("%d\n", min_value) < 0) {
        return 1;
    }

    return 0;
}
