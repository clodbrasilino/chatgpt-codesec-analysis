```c
#include <stdio.h>
#include <stdlib.h>

double average_of_list(const int *list, size_t size) {
    if (list == NULL || size == 0) {
        return 0.0;
    }

    double sum = 0.0;
    for (size_t i = 0; i < size; i++) {
        sum += list[i];
    }

    return sum / (double)size;
}

int main(void) {
    int numbers[] = {10, 20, 30, 40, 50};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);

    double avg = average_of_list(numbers, count);
    printf("Average: %.2f\n", avg);

    return 0;
}
```
