```c
#include <stdio.h>
#include <math.h>

double geometric_nth_term(double a, double r, int n) {
    return a * pow(r, n - 1);
}

int main() {
    double a = 2.0;
    double r = 3.0;
    int n = 5;
    double result = geometric_nth_term(a, r, n);
    printf("The %d-th term of the geometric series is: %f\n", n, result);
    return 0;
}
```
