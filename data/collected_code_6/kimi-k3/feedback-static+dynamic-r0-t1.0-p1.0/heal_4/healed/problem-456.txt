#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void reverse_string(char *str, size_t len)
{
    size_t left;
    size_t right;

    if (str == NULL)
    {
        return;
    }

    left = 0;
    right = len;

    if (right == 0)
    {
        return;
    }

    right--;

    while (left < right)
    {
        char temp = str[left];
        str[left] = str[right];
        str[right] = temp;
        left++;
        right--;
    }
}

int reverse_string_list(char **list, size_t count, const size_t *lengths)
{
    size_t i;

    if (list == NULL || lengths == NULL)
    {
        return -1;
    }

    for (i = 0; i < count; i++)
    {
        if (list[i] == NULL)
        {
            return -1;
        }
    }

    for (i = 0; i < count; i++)
    {
        reverse_string(list[i], lengths[i]);
    }

    return 0;
}

int main(void)
{
    char string1[] = "hello";
    char string2[] = "world";
    char string3[] = "programming";
    char string4[] = "language";
    char string5[] = "secure";

    char *strings[] = {
        string1,
        string2,
        string3,
        string4,
        string5
    };

    const size_t lengths[] = {
        sizeof(string1) - 1,
        sizeof(string2) - 1,
        sizeof(string3) - 1,
        sizeof(string4) - 1,
        sizeof(string5) - 1
    };

    size_t count;
    size_t i;
    int result;

    count = sizeof(strings) / sizeof(strings[0]);

    result = reverse_string_list(strings, count, lengths);

    if (result != 0)
    {
        fprintf(stderr, "Error: failed to reverse string list\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++)
    {
        if (printf("%s\n", strings[i]) < 0)
        {
            fprintf(stderr, "Error: failed to print output\n");
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
