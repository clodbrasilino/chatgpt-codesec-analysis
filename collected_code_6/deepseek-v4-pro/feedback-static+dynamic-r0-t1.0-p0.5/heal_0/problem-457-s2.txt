#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
} Sublist;

Sublist find_minimum_length_sublist(int *array, size_t array_length, int target_sum) {
    Sublist result = {NULL, 0};
    size_t min_length = array_length + 1;
    size_t current_sum = 0;
    size_t start = 0;
    size_t end = 0;

    if (array == NULL || array_length == 0) {
        return result;
    }

    while (end < array_length) {
        current_sum += array[end];

        while (current_sum >= (size_t)target_sum && start <= end) {
            size_t current_length = end - start + 1;
            if (current_length < min_length) {
                min_length = current_length;
                result.data = &array[start];
                result.length = current_length;
            }
            current_sum -= array[start];
            start++;
        }
        end++;
    }

    if (min_length == array_length + 1) {
        result.data = NULL;
        result.length = 0;
    }

    return result;
}

int main(void) {
    int test_array[] = {2, 3, 1, 2, 4, 3};
    size_t array_length = sizeof(test_array) / sizeof(test_array[0]);
    int target_sum = 7;
    Sublist result;

    result = find_minimum_length_sublist(test_array, array_length, target_sum);

    if (result.data != NULL) {
        printf("Minimum length sublist: [");
        for (size_t i = 0; i < result.length; i++) {
            printf("%d", result.data[i]);
            if (i < result.length - 1) {
                printf(", ");
            }
        }
        printf("]\n");
        printf("Length: %zu\n", result.length);
    } else {
        printf("No sublist found with sum >= %d\n", target_sum);
    }

    return 0;
}
