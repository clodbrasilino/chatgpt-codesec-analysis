```c
#include <stdio.h>
#include <stdbool.h>
#include <limits.h>
#include <math.h>

bool isDifferenceOfSquares(long n) {
    if (n < 0) return false;
    long x = 0, y = 1;
    long max_val = (long)sqrt((double)LONG_MAX);
    while (x <= y && y <= max_val) {
        long diff = y * y - x * x;
        if (diff == n) {
            return true;
        } else if (diff < n) {
            y++;
        } else {
            x++;
        }
    }
    return false;
}

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);
    if (isDifferenceOfSquares(num)) {
        printf("Can be represented as difference of two squares.\n");
    } else {
        printf("Cannot be represented as difference of two squares.\n");
    }
    return 0;
}
```
