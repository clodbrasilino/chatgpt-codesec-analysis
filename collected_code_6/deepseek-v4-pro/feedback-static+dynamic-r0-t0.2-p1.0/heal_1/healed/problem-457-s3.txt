#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
    size_t start;
} Sublist;

Sublist find_minimum_length_sublist(const int *arr, size_t n, int target) {
    Sublist result = {NULL, 0, 0};
    size_t min_length = n + 1;
    size_t current_sum = 0;
    size_t left = 0;

    for (size_t right = 0; right < n; right++) {
        current_sum += arr[right];

        while (current_sum >= target && left <= right) {
            size_t current_length = right - left + 1;
            if (current_length < min_length) {
                min_length = current_length;
                result.start = left;
                result.length = current_length;
            }
            current_sum -= arr[left];
            left++;
        }
    }

    if (min_length <= n) {
        result.data = (int *)malloc(result.length * sizeof(int));
        if (result.data != NULL) {
            for (size_t i = 0; i < result.length; i++) {
                result.data[i] = arr[result.start + i];
            }
        }
    }

    return result;
}

int main(void) {
    int arr[] = {2, 3, 1, 2, 4, 3};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int target = 7;

    Sublist result = find_minimum_length_sublist(arr, n, target);

    if (result.data != NULL) {
        printf("Minimum length sublist: ");
        for (size_t i = 0; i < result.length; i++) {
            printf("%d ", result.data[i]);
        }
        printf("\n");
        free(result.data);
    } else {
        printf("No sublist found\n");
    }

    return 0;
}
