```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t capacity;
    size_t num_tokens;
    size_t start;
    size_t i;
    char **tokens;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return NULL;
    }

    capacity = 8;
    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    num_tokens = 0;
    start = 0;

    for (i = 0; i <= len; i++) {
        if (i == len || (isupper((unsigned char)input[i]) && i > start)) {
            size_t token_len = i - start;

            if (token_len > 0) {
                char *token;
                size_t alloc_size;

                if (num_tokens == capacity) {
                    size_t new_capacity;
                    char **resized;

                    if (capacity > SIZE_MAX / 2 ||
                        (capacity * 2) > SIZE_MAX / sizeof(char *)) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    new_capacity = capacity * 2;
                    resized = realloc(tokens, new_capacity * sizeof(char *));
                    if (resized == NULL) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    tokens = resized;
                    capacity = new_capacity;
                }

                if (start >= len || token_len > len - start) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                alloc_size = token_len + 1;

                token = malloc(alloc_size);
                if (token == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                memcpy(token, &input[start], token_len);
                token[token_len] = '\0';
                tokens[num_tokens] = token;
                num_tokens++;
            }
            start = i;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "helloWorldThisIsATest";
    size_t count = 0;
    size_t i;
    char **tokens;

    tokens = split_at_uppercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens (%zu):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **Removed the `alloc_size <= token_len` check** — flagged by cppcheck as `knownConditionTrueFalse`. Since `token_len` is strictly bounded by `MAX_INPUT_LEN` (4096), `token_len + 1` can never wrap around `SIZE_MAX`, making the overflow check dead code.

2. **Removed the redundant post-malloc `token_len >= alloc_size` check** — same reasoning; `alloc_size` was just assigned `token_len + 1` with no possibility of overflow, so this condition was provably always false.

3. **Removed the `capacity > SIZE_MAX / sizeof(char *)` check on the initial constant capacity of 8** — trivially always false and dead code.

4. **`memcpy` retained but provably safe** — the Flawfinder warning (CWE-120) is a risk-2 informational hit. The destination buffer is allocated with exactly `token_len + 1` bytes immediately before the copy of `token_len` bytes, and the source bounds are validated (`start < len`, `token_len <= len - start`), so the destination always holds the source data plus the NUL terminator. This is the correct mitigation rather than switching to `strncpy`, which would add cost without safety benefit for non-string binary-safe copies.

5. **Tightened the bounds guard** from `start > len` to `start >= len`, which is the stronger invariant that must hold before indexing `input[start]`.

The defensive structure (cleanup on every error path via `free_tokens`, `realloc` result checked into a temporary to avoid leaking the original pointer, `strnlen` to cap unbounded input) was preserved as it is correct.
