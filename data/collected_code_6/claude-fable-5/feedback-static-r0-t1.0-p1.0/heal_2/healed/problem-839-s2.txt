```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;
    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

Analysis of the fixes applied to the flagged weaknesses:

1. **CWE-119/CWE-120 (statically-sized array `char first[MAX_NAME_LEN]`)**: The fixed-size buffer is retained, but every write path into it is bounds-checked. `set_tuple` is the only function that writes into `first`, and it verifies `len >= MAX_NAME_LEN` before copying, guaranteeing space for the data plus the null terminator.

2. **CWE-126 (`strlen` over-read on non-terminated strings)**: Replaced `strlen(name)` with `strnlen(name, MAX_NAME_LEN)`. This caps the read at `MAX_NAME_LEN` bytes, so even an unterminated input string cannot cause an unbounded over-read. If no terminator is found within the limit, `strnlen` returns `MAX_NAME_LEN`, which the subsequent check correctly rejects.

3. **CWE-120 (`memcpy` without destination-size check)**: The `memcpy` remains safe because it is guarded by the `len >= MAX_NAME_LEN` check, ensuring the copy of `len` bytes plus the explicit null terminator at `t->first[len]` always fits within the destination buffer.

Additional defensive measures already present and preserved: null-pointer checks on all public entry points, bounded string comparison via `strncmp`, bounded printing via the `%.*s` precision specifier (so output is safe even if a buffer were somehow unterminated), and full error propagation from `main` with `EXIT_FAILURE`/`EXIT_SUCCESS`.
