#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int a;
    int b;
} Tuple;

int compare_tuples(const void *t1, const void *t2) {
    Tuple *tuple1 = (Tuple *)t1;
    Tuple *tuple2 = (Tuple *)t2;
    if (tuple1->a == tuple2->a) {
        return tuple1->b - tuple2->b;
    }
    return tuple1->a - tuple2->a;
}

int find_intersection(Tuple *list1, int size1, Tuple *list2, int size2, Tuple **result) {
    qsort(list1, size1, sizeof(Tuple), compare_tuples);
    qsort(list2, size2, sizeof(Tuple), compare_tuples);

    int i = 0, j = 0, k = 0;
    while (i < size1 && j < size2) {
        if (list1[i].a == list2[j].a && list1[i].b == list2[j].b) {
            (*result)[k++] = list1[i];
            i++;
            j++;
        } else if (compare_tuples(&list1[i], &list2[j]) < 0) {
            i++;
        } else {
            j++;
        }
    }
    return k;
}

int main() {
    Tuple list1[] = {{1, 2}, {3, 4}, {5, 6}};
    Tuple list2[] = {{3, 4}, {7, 8}, {1, 2}};
    int size1 = sizeof(list1) / sizeof(list1[0]);
    int size2 = sizeof(list2) / sizeof(list2[0]);

    Tuple *result = (Tuple *)malloc(size1 * sizeof(Tuple));
    if (result == NULL) {
        return 1;
    }

    int result_size = find_intersection(list1, size1, list2, size2, &result);

    for (int i = 0; i < result_size; i++) {
        printf("(%d, %d)\n", result[i].a, result[i].b);
    }

    free(result);
    return 0;
}
