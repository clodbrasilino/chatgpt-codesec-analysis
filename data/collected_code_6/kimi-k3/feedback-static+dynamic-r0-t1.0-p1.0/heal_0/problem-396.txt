#include <stdio.h>
#include <string.h>
#include <regex.h>

int check_start_end_same(const char *str) {
    regex_t regex;
    int result;
    int ret = 0;
    
    if (str == NULL) {
        return -1;
    }
    
    if (strlen(str) == 0) {
        return 0;
    }
    
    result = regcomp(&regex, "^(.).*\\1$|^.$", REG_EXTENDED);
    if (result != 0) {
        return -1;
    }
    
    result = regexec(&regex, str, 0, NULL, 0);
    if (result == 0) {
        ret = 1;
    } else if (result == REG_NOMATCH) {
        ret = 0;
    } else {
        ret = -1;
    }
    
    regfree(&regex);
    
    return ret;
}

int main(void) {
    char str1[] = "helloh";
    char str2[] = "world";
    char str3[] = "a";
    char str4[] = "";
    
    printf("\"%s\": %s\n", str1, check_start_end_same(str1) == 1 ? "Match" : "No Match");
    printf("\"%s\": %s\n", str2, check_start_end_same(str2) == 1 ? "Match" : "No Match");
    printf("\"%s\": %s\n", str3, check_start_end_same(str3) == 1 ? "Match" : "No Match");
    printf("\"%s\": %s\n", str4, check_start_end_same(str4) == 1 ? "Match" : "No Match");
    
    return 0;
}
