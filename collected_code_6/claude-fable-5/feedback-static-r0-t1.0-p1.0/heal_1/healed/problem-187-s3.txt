```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static char *longest_common_subsequence(const char *s1, const char *s2)
{
    size_t m;
    size_t n;
    size_t i;
    size_t j;
    size_t idx;
    int **dp;
    char *result;

    if (s1 == NULL || s2 == NULL) {
        return NULL;
    }

    m = strnlen(s1, MAX_INPUT_LEN);
    n = strnlen(s2, MAX_INPUT_LEN);

    if (m >= MAX_INPUT_LEN || n >= MAX_INPUT_LEN) {
        return NULL;
    }

    dp = malloc((m + 1) * sizeof(int *));
    if (dp == NULL) {
        return NULL;
    }

    for (i = 0; i <= m; i++) {
        dp[i] = calloc(n + 1, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return NULL;
        }
    }

    for (i = 1; i <= m; i++) {
        for (j = 1; j <= n; j++) {
            if (s1[i - 1] == s2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else if (dp[i - 1][j] >= dp[i][j - 1]) {
                dp[i][j] = dp[i - 1][j];
            } else {
                dp[i][j] = dp[i][j - 1];
            }
        }
    }

    idx = (size_t)dp[m][n];
    result = malloc(idx + 1);
    if (result == NULL) {
        for (i = 0; i <= m; i++) {
            free(dp[i]);
        }
        free(dp);
        return NULL;
    }

    result[idx] = '\0';
    i = m;
    j = n;
    while (i > 0 && j > 0) {
        if (s1[i - 1] == s2[j - 1]) {
            idx--;
            result[idx] = s1[i - 1];
            i--;
            j--;
        } else if (dp[i - 1][j] >= dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }

    for (i = 0; i <= m; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    const char *seq1 = "AGGTAB";
    const char *seq2 = "GXTXAYB";
    char *lcs;

    lcs = longest_common_subsequence(seq1, seq2);
    if (lcs == NULL) {
        fprintf(stderr, "Failed to compute longest common subsequence\n");
        return EXIT_FAILURE;
    }

    if (printf("Sequence 1: %s\n", seq1) < 0 ||
        printf("Sequence 2: %s\n", seq2) < 0 ||
        printf("LCS: %s\n", lcs) < 0 ||
        printf("Length: %zu\n", strnlen(lcs, MAX_INPUT_LEN)) < 0) {
        free(lcs);
        return EXIT_FAILURE;
    }

    free(lcs);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Replaced `strlen` with `strnlen`** (CWE-126 mitigation): All three flagged `strlen` calls now use `strnlen` with an explicit upper bound (`MAX_INPUT_LEN`), guaranteeing the read is bounded even if a non-null-terminated string is passed.

2. **Added length validation**: If either input reaches the maximum bound (meaning it may be unterminated or excessively long), the function safely returns `NULL` instead of proceeding. This also serves as a defense against integer overflow in the `malloc((m + 1) * sizeof(int *))` allocation on pathologically large inputs.

3. **Preserved existing robustness**: The original code already handled `NULL` inputs, allocation failures (with proper cleanup of partially allocated `dp` rows), and freed all memory on every path — these were retained as-is since they were correct.

Note that in this specific program the inputs are string literals, so the original `strlen` calls were technically safe; the `strnlen` changes harden the function for reuse in contexts where input provenance isn't guaranteed.
