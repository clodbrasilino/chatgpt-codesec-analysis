#include <stdio.h>
#include <stdlib.h>

int is_perfect_number(int num) {
    if (num <= 1) return 0;
    int sum = 1;
    for (int i = 2; i * i <= num; i++) {
        if (num % i == 0) {
            sum += i;
            if (i * i != num) {
                sum += num / i;
            }
        }
    }
    return sum == num;
}

int find_square_root(int num) {
    if (num < 0) {
        return -1;
    }
    if (num == 0 || num == 1) {
        return num;
    }
    int left = 0, right = num;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        long long square = (long long)mid * mid;
        if (square == num) {
            return mid;
        } else if (square < num) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return -1;
}

int main(void) {
    int num;
    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    if (!is_perfect_number(num)) {
        fprintf(stderr, "Error: %d is not a perfect number\n", num);
        return EXIT_FAILURE;
    }
    int root = find_square_root(num);
    if (root == -1) {
        fprintf(stderr, "Error: could not find a perfect square root\n");
        return EXIT_FAILURE;
    }
    printf("Square root of %d is %d\n", num, root);
    return EXIT_SUCCESS;
}
