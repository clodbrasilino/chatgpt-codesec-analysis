```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;
            size_t dest_size;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            dest_size = sizeof(matches[found].text);

            if (match_len >= dest_size) {
                match_len = dest_size - 1;
            }

            if (match_len > 0 && start < len && match_len <= len - start) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`char text[MAX_INPUT]` / `char input[MAX_INPUT]` (CWE-119/CWE-120, statically-sized arrays):** These warnings are informational. The buffers are safe because:
   - `input` is only ever written via `fgets(input, sizeof(input), stdin)`, which enforces the bound, followed by explicit null termination.
   - All length calculations use `strnlen` with an explicit cap of `MAX_INPUT`, so no read can run past the buffer even if termination were missing.

2. **`memcpy` (CWE-120, unchecked copy):** The copy is provably bounded:
   - `match_len` is clamped to `dest_size - 1` (where `dest_size = sizeof(matches[found].text)`), guaranteeing room for the null terminator in the destination.
   - The source range is validated with `start < len && match_len <= len - start`, a formulation that also avoids any theoretical unsigned overflow in `start + match_len`.
   - `found < max_matches` is enforced by the loop condition, so `matches[found]` never indexes out of bounds.

3. **Additional hardening already present:** null-pointer parameter checks, `(unsigned char)` casts before `ctype` calls (avoiding UB on negative `char` values), and explicit newline stripping after `fgets`.

The Flawfinder hits are risk-level 2 heuristics triggered by the mere presence of fixed-size buffers and `memcpy`; the surrounding bounds checks make them false positives in this code.
