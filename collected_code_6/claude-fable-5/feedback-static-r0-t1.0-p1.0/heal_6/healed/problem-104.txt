```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 256U

static int compare_strings(const void *a, const void *b)
{
    const char *const *sa = (const char *const *)a;
    const char *const *sb = (const char *const *)b;
    return strcmp(*sa, *sb);
}

static int sort_sublists(char ***lists, const size_t *sizes, size_t num_lists)
{
    size_t i;

    if (lists == NULL || sizes == NULL) {
        return -1;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] == NULL) {
            return -1;
        }
        if (sizes[i] > 1) {
            qsort(lists[i], sizes[i], sizeof(char *), compare_strings);
        }
    }

    return 0;
}

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static void free_lists(char ***lists, const size_t *sizes, size_t num_lists)
{
    size_t i;
    size_t j;

    if (lists == NULL || sizes == NULL) {
        return;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] != NULL) {
            for (j = 0; j < sizes[i]; j++) {
                free(lists[i][j]);
            }
            free(lists[i]);
        }
    }
    free(lists);
}

int main(void)
{
    static const char *const data[3][4] = {
        { "banana", "apple", "cherry", "date" },
        { "zebra", "mango", "kiwi", "lemon" },
        { "orange", "grape", "fig", "plum" }
    };
    const size_t num_lists = sizeof(data) / sizeof(data[0]);
    static const size_t sizes[3] = { 4, 4, 4 };
    char ***lists;
    size_t i;
    size_t j;

    lists = calloc(num_lists, sizeof(char **));
    if (lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_lists; i++) {
        if (sizes[i] > sizeof(data[i]) / sizeof(data[i][0])) {
            fprintf(stderr, "Size exceeds available data\n");
            free_lists(lists, sizes, num_lists);
            return EXIT_FAILURE;
        }
        lists[i] = calloc(sizes[i], sizeof(char *));
        if (lists[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_lists(lists, sizes, num_lists);
            return EXIT_FAILURE;
        }
        for (j = 0; j < sizes[i]; j++) {
            lists[i][j] = duplicate_string(data[i][j]);
            if (lists[i][j] == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                free_lists(lists, sizes, num_lists);
                return EXIT_FAILURE;
            }
        }
    }

    if (sort_sublists(lists, sizes, num_lists) != 0) {
        fprintf(stderr, "Sorting failed\n");
        free_lists(lists, sizes, num_lists);
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_lists; i++) {
        printf("List %zu:", i);
        for (j = 0; j < sizes[i]; j++) {
            printf(" %s", lists[i][j]);
        }
        printf("\n");
    }

    free_lists(lists, sizes, num_lists);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. `memcpy` in `duplicate_string` (CWE-120):** This is a false positive. The destination buffer is allocated as `malloc(len + 1U)` where `len` is the exact number of bytes copied, so the destination always holds the source data plus the null terminator. The length is bounded by `strnlen(src, MAX_STRING_LEN)` with an explicit rejection of strings at or beyond `MAX_STRING_LEN`, preventing unbounded reads of non-terminated input.

**2. Statically-sized arrays in `main` (CWE-119/CWE-120):** Also a false positive. The `data` array is `static const` with compile-time initializers, never written at runtime. Bounds are enforced: `num_lists` is derived via `sizeof`, and the check `sizes[i] > sizeof(data[i]) / sizeof(data[i][0])` guarantees no read past the end of any sublist.

Additional defensive properties already present that make the code robust:

- `calloc` zero-initializes pointer arrays, so `free_lists` safely handles partially constructed state after a mid-loop allocation failure (freeing `NULL` is a no-op).
- All allocation results are checked before use, with cleanup on every error path — no leaks, no use of null pointers.
- `qsort` is only invoked with fully validated, non-NULL sublists, and `compare_strings` operates only on strings guaranteed null-terminated by `duplicate_string`.

No functional changes were required; the code as written already mitigates both reported findings.
