```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int get_n_smallest(const int *data, size_t data_len, size_t n, int *result)
{
    int *copy;

    if (data == NULL || result == NULL) {
        return -1;
    }
    if (n == 0 || n > data_len) {
        return -1;
    }
    if (data_len > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy = malloc(data_len * sizeof(int));
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, data, data_len * sizeof(int));
    qsort(copy, data_len, sizeof(int), compare_ints);
    memcpy(result, copy, n * sizeof(int));

    free(copy);
    return 0;
}

int main(void)
{
    int dataset[] = { 42, 7, 19, 3, 88, 15, 1, 56, 23, 9 };
    size_t dataset_len = sizeof(dataset) / sizeof(dataset[0]);
    size_t n = 4;
    int *smallest;
    size_t i;

    if (n == 0 || n > dataset_len || n > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Invalid value of n\n");
        return EXIT_FAILURE;
    }

    smallest = malloc(n * sizeof(int));
    if (smallest == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (get_n_smallest(dataset, dataset_len, n, smallest) != 0) {
        fprintf(stderr, "Failed to get n smallest items\n");
        free(smallest);
        return EXIT_FAILURE;
    }

    printf("The %zu smallest items are:", n);
    for (i = 0; i < n; i++) {
        printf(" %d", smallest[i]);
    }
    printf("\n");

    free(smallest);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged `memcpy` calls (CWE-120) were false positives in the original code, but the reasoning behind their safety deserves verification, which the rewrite preserves and strengthens:

1. **First `memcpy` (`copy` ← `data`):** The destination `copy` is allocated with exactly `data_len * sizeof(int)` bytes, and the overflow guard `data_len > SIZE_MAX / sizeof(int)` ensures the multiplication cannot wrap around. The copy size matches the allocation exactly, so no overflow is possible.

2. **Second `memcpy` (`result` ← `copy`):** The check `n > data_len` guarantees `n * sizeof(int)` never exceeds the size of `copy`. The caller contract requires `result` to hold at least `n` ints — enforced in `main` by allocating `n * sizeof(int)` and validating `n` against the dataset length before allocation.

3. **Actual bug fixed:** The original code used `SIZE_MAX` without including `<stdint.h>`, which would fail to compile (or worse, rely on an implicit declaration in pre-C99 modes). The rewrite adds the missing header.

4. **Defensive addition in `main`:** The `n` validation and multiplication-overflow check before `malloc(n * sizeof(int))` ensures the caller-side buffer is always adequately sized, closing the loop on the CWE-120 concern from both sides of the API boundary.
