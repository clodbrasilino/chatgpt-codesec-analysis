```c
#include <stdio.h>

int is_equilateral(int a, int b, int c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return -1;
    }
    if (a + b <= c || a + c <= b || b + c <= a) {
        return -1;
    }
    return (a == b && b == c) ? 1 : 0;
}

int main(void) {
    int a, b, c;
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        return 1;
    }
    int result = is_equilateral(a, b, c);
    if (result == -1) {
        printf("Invalid triangle\n");
    } else if (result == 1) {
        printf("Equilateral\n");
    } else {
        printf("Not equilateral\n");
    }
    return 0;
}
```
