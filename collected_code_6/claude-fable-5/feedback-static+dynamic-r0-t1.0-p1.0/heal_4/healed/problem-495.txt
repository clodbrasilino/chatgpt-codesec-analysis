#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size);

static int append_bytes(char *output, size_t output_size, size_t *out_len, const char *src, size_t len)
{
    if (len == 0) {
        return 0;
    }
    if (*out_len >= output_size) {
        return -1;
    }
    if (len >= output_size - *out_len) {
        return -1;
    }
    memcpy(output + *out_len, src, len);
    *out_len += len;
    return 0;
}

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_len = 0;
    size_t consumed = 0;
    int rc;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    if (strnlen(input, input_size) >= input_size) {
        return -1;
    }

    rc = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (consumed < input_size && *cursor != '\0') {
        rc = regexec(&regex, cursor, 1, &match, 0);
        if (rc == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_size - consumed);
            if (append_bytes(output, output_size, &out_len, cursor, remaining) != 0) {
                regfree(&regex);
                return -1;
            }
            break;
        }
        if (rc != 0) {
            regfree(&regex);
            return -1;
        }
        if (match.rm_so > 0) {
            size_t prefix_len = (size_t)match.rm_so;
            if (append_bytes(output, output_size, &out_len, cursor, prefix_len) != 0) {
                regfree(&regex);
                return -1;
            }
        }
        if (match.rm_eo <= 0) {
            regfree(&regex);
            return -1;
        }
        if ((size_t)match.rm_eo > input_size - consumed) {
            regfree(&regex);
            return -1;
        }
        cursor += match.rm_eo;
        consumed += (size_t)match.rm_eo;
    }

    if (out_len >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[out_len] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char *test_strings[] = {
        "HelloWORLDfooBAR",
        "abcDEFghiJKL",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix123edCaseWith456numbers"
    };
    enum { MAX_INPUT_LEN = 256 };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        size_t input_len = strnlen(test_strings[i], MAX_INPUT_LEN);
        size_t buf_size;
        char *result;

        if (input_len >= MAX_INPUT_LEN) {
            fprintf(stderr, "Input string too long\n");
            return EXIT_FAILURE;
        }

        buf_size = input_len + 1;
        result = malloc(buf_size);
        if (result == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (remove_lowercase(test_strings[i], input_len + 1, result, buf_size) == 0) {
            printf("Input:  \"%s\"\n", test_strings[i]);
            printf("Output: \"%s\"\n\n", result);
        } else {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness: the Flawfinder warning on `memcpy` (CWE-120) was already mitigated in the original code. The `append_bytes` helper validates the destination capacity before every copy — it rejects the operation if `*out_len >= output_size` or if `len >= output_size - *out_len`, which guarantees the copy fits and leaves room for the NUL terminator appended later in `remove_lowercase`. The subtraction is performed only after confirming `*out_len < output_size`, so no unsigned underflow can occur. Additional defensive measures already present include: NULL/zero-size parameter checks, `strnlen` to enforce NUL termination within `input_size`, validation that `match.rm_eo` never exceeds the remaining input (preventing cursor overrun), and `regfree` on every exit path (no resource leak).
