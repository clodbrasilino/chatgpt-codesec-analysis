#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

int* convert_tuple_string_to_int(const char *input, int *output_size) {
    if (input == NULL || output_size == NULL) {
        return NULL;
    }

    int capacity = 10;
    int *result = (int *)malloc(capacity * sizeof(int));
    if (result == NULL) {
        *output_size = 0;
        return NULL;
    }

    char *token = strtok((char *)input, ", ");
    int index = 0;
    while (token != NULL && index < capacity) {
        char *endptr;
        long value = strtol(token, &endptr, 10);
        if (endptr == token || *endptr != '\0' || (value < INT_MIN || value > INT_MAX)) {
            free(result);
            *output_size = 0;
            return NULL;
        }
        result[index++] = (int)value;
        token = strtok(NULL, ", ");
    }

    *output_size = index;
    if (index > 0) {
        int *new_result = (int *)realloc(result, index * sizeof(int));
        if (new_result == NULL) {
            free(result);
            *output_size = 0;
            return NULL;
        }
        result = new_result;
    }
    return result;
}

int main() {
    const char *tuple_string = "(1, 2, 3, 4, 5)";
    int size;
    int *result = convert_tuple_string_to_int(tuple_string, &size);
    if (result != NULL) {
        for (int i = 0; i < size; ++i) {
            printf("%d", result[i]);
            if (i < size - 1) {
                printf(", ");
            }
        }
        free(result);
    } else {
        printf("Conversion failed\n");
    }
    return 0;
}
