#include <stdio.h>
#include <stdlib.h>

void multiply_adjacent(const int *arr, int n, int *result) {
    if (n < 2) return;
    for (int i = 0; i < n - 1; ++i) {
        result[i] = arr[i] * arr[i + 1];
    }
}

int main() {
    int tuple[] = {1, 2, 3, 4, 5};
    int n = sizeof(tuple) / sizeof(tuple[0]);
    int *result = (int *)malloc((n - 1) * sizeof(int));
    if (!result) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    multiply_adjacent(tuple, n, result);

    for (int i = 0; i < n - 1; ++i) {
        printf("%d ", result[i]);
    }
    free(result);
    return EXIT_SUCCESS;
}
