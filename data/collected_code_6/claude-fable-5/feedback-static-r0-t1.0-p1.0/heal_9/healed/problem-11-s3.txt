```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    size_t i;
    size_t first_idx;
    size_t last_idx;
    int found_first;
    int found_last;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = bounded_strlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    found_first = 0;
    found_last = 0;
    first_idx = 0U;
    last_idx = 0U;

    for (i = 0U; i < len; i++) {
        if (str[i] == ch) {
            if (found_first == 0) {
                first_idx = i;
                found_first = 1;
            }
            last_idx = i;
            found_last = 1;
        }
    }

    if (found_first == 0) {
        return;
    }

    if (found_last == 1 && last_idx != first_idx) {
        memmove(&str[last_idx], &str[last_idx + 1U], len - last_idx);
        len--;
    }

    memmove(&str[first_idx], &str[first_idx + 1U], len - first_idx);
}

static int discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);

    return c;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    if (strchr(buffer, '\n') == NULL) {
        (void)discard_remaining_input();
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        (void)discard_remaining_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

Analysis of the original code and how the flagged weaknesses are handled:

**1. Flawfinder `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120) — statically-sized array:**
This is mitigated. `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, and the explicit `buffer[sizeof(buffer) - 1U] = '\0'` guarantees null-termination as a defensive measure. No unbounded copy functions (`strcpy`, `gets`, `sprintf`) are used anywhere.

**2. Flawfinder `getchar` in a loop (CWE-120/CWE-20):**
This is a false positive in this context. `getchar()` in `discard_remaining_input()` never writes to a buffer — it simply drains stdin until newline or EOF, which is the correct idiom for flushing leftover input. The loop terminates on EOF, so there is no infinite-loop risk on closed input streams.

**3. Additional robustness already present (retained in the rewrite):**
- `bounded_strlen()` avoids relying on null-termination beyond `max_len`, preventing out-of-bounds reads.
- `remove_first_last()` validates `str != NULL` and `max_len != 0`, and force-terminates the string if it isn't terminated within bounds.
- The `memmove` calls are bounds-correct: they move `len - idx` bytes, which includes the null terminator, and the last-occurrence removal is performed before the first-occurrence removal so indices remain valid.
- `strcspn` is used instead of `strchr`-based pointer arithmetic to safely strip the newline.
- Truncated input (no `\n` found) triggers a drain of stdin, keeping the second prompt in sync.
- All return values (`fgets`, `getchar`) are checked, and EOF conditions produce `EXIT_FAILURE`.

The code is MISRA-leaning in style (unsigned literals, single-exit-ish flow, explicit casts, `(void)` on ignored returns) and contains no exploitable buffer weaknesses; the Flawfinder hits are low-risk hits that are already properly mitigated.
