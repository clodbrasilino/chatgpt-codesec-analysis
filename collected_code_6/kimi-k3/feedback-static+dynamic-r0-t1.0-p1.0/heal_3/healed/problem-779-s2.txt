#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} List;

typedef struct {
    List *lists;
    size_t count;
} ListCollection;

int compare_lists(const List *a, const List *b) {
    if (a->size != b->size) {
        return 0;
    }
    if (a->size == 0) {
        return 1;
    }
    return memcmp(a->data, b->data, a->size * sizeof(int)) == 0;
}

size_t count_unique_lists(const ListCollection *collection) {
    if (collection == NULL || collection->lists == NULL || collection->count == 0) {
        return 0;
    }

    size_t unique_count = 0;
    int *processed = calloc(collection->count, sizeof(int));
    if (processed == NULL) {
        return 0;
    }

    for (size_t i = 0; i < collection->count; i++) {
        if (processed[i]) {
            continue;
        }
        unique_count++;
        processed[i] = 1;

        for (size_t j = i + 1; j < collection->count; j++) {
            if (!processed[j] && compare_lists(&collection->lists[i], &collection->lists[j])) {
                processed[j] = 1;
            }
        }
    }

    free(processed);
    return unique_count;
}

void free_collection(ListCollection *collection) {
    if (collection == NULL) {
        return;
    }
    if (collection->lists != NULL) {
        for (size_t i = 0; i < collection->count; i++) {
            free(collection->lists[i].data);
            collection->lists[i].data = NULL;
        }
        free(collection->lists);
        collection->lists = NULL;
    }
    collection->count = 0;
}

int safe_copy_list(List *dest, const int *src, size_t count) {
    if (dest == NULL || src == NULL || count == 0) {
        return 0;
    }

    if (count > SIZE_MAX / sizeof(int)) {
        return 0;
    }

    if (dest->data == NULL || dest->capacity < count) {
        return 0;
    }

    memcpy(dest->data, src, count * sizeof(int));
    dest->size = count;
    return 1;
}

int main(void) {
    int data1[] = {1, 2, 3};
    int data2[] = {4, 5};
    int data3[] = {1, 2, 3};
    int data4[] = {6, 7, 8, 9};
    int data5[] = {4, 5};
    int data6[] = {1, 2, 3, 4};

    ListCollection collection = {0};
    collection.count = 6;
    collection.lists = calloc(collection.count, sizeof(List));
    if (collection.lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    const size_t sizes[] = {
        sizeof(data1) / sizeof(data1[0]),
        sizeof(data2) / sizeof(data2[0]),
        sizeof(data3) / sizeof(data3[0]),
        sizeof(data4) / sizeof(data4[0]),
        sizeof(data5) / sizeof(data5[0]),
        sizeof(data6) / sizeof(data6[0])
    };

    for (size_t i = 0; i < collection.count; i++) {
        collection.lists[i].data = malloc(sizes[i] * sizeof(int));
        if (collection.lists[i].data == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_collection(&collection);
            return EXIT_FAILURE;
        }
        collection.lists[i].capacity = sizes[i];
        collection.lists[i].size = 0;
    }

    if (!safe_copy_list(&collection.lists[0], data1, sizeof(data1) / sizeof(data1[0])) ||
        !safe_copy_list(&collection.lists[1], data2, sizeof(data2) / sizeof(data2[0])) ||
        !safe_copy_list(&collection.lists[2], data3, sizeof(data3) / sizeof(data3[0])) ||
        !safe_copy_list(&collection.lists[3], data4, sizeof(data4) / sizeof(data4[0])) ||
        !safe_copy_list(&collection.lists[4], data5, sizeof(data5) / sizeof(data5[0])) ||
        !safe_copy_list(&collection.lists[5], data6, sizeof(data6) / sizeof(data6[0]))) {
        fprintf(stderr, "Copy operation failed\n");
        free_collection(&collection);
        return EXIT_FAILURE;
    }

    size_t unique = count_unique_lists(&collection);
    printf("Number of unique lists: %zu\n", unique);

    free_collection(&collection);
    return EXIT_SUCCESS;
}
