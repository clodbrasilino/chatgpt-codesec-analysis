```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

static int safe_copy(char *dst, size_t dst_size, size_t pos,
                     const char *src, size_t copy_len)
{
    if (dst == NULL || src == NULL) {
        return -1;
    }
    if (copy_len > dst_size) {
        return -1;
    }
    if (pos > dst_size - copy_len) {
        return -1;
    }
    if (pos + copy_len >= dst_size) {
        return -1;
    }
    memcpy(dst + pos, src, copy_len);
    return 0;
}

char *remove_non_alnum(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t len;
    size_t pos;
    size_t offset;
    size_t alloc_size;
    size_t copy_len;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[^a-zA-Z0-9]+", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        regfree(&regex);
        return NULL;
    }

    alloc_size = len + 1;
    result = malloc(alloc_size);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    pos = 0;
    offset = 0;

    while (offset < len) {
        ret = regexec(&regex, input + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            copy_len = len - offset;
            if (safe_copy(result, alloc_size, pos, input + offset, copy_len) != 0) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            pos += copy_len;
            break;
        }
        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        copy_len = (size_t)match.rm_so;
        if (copy_len > len - offset) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (safe_copy(result, alloc_size, pos, input + offset, copy_len) != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        pos += copy_len;
        if ((size_t)match.rm_eo > len - offset) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        offset += (size_t)match.rm_eo;
    }

    if (pos >= alloc_size) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    result[pos] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test = "Hello, World! 123 #$% Test_456";
    char *cleaned;

    cleaned = remove_non_alnum(test);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }
    if (printf("Cleaned:  %s\n", cleaned) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    free(cleaned);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely safe, but the Flawfinder warnings on both `memcpy` calls pointed to the fact that the bounds-checking logic was duplicated inline and subtly fragile. Key points addressed in the rewrite:

1. **Centralized bounds validation** — Both `memcpy` sites now go through `safe_copy()`, which validates that `copy_len` fits within the destination, that `pos` cannot cause an overflow when added to `copy_len` (checked via subtraction to avoid integer overflow), and that the write always leaves room for the NUL terminator (`pos + copy_len < dst_size`).

2. **Integer overflow safety** — The check `pos > dst_size - copy_len` is performed only after confirming `copy_len <= dst_size`, so the subtraction cannot wrap around.

3. **Preserved invariants** — Total copied bytes can never exceed `len`, and `alloc_size = len + 1` guarantees space for the terminator; the explicit checks make this provable locally rather than relying on global reasoning, which satisfies the CWE-120 concern.

4. **Resource management** — Every error path still frees `result` and calls `regfree()`, so no leaks were introduced.

The functional behavior is unchanged: the program prints `HelloWorld123Test456` for the test input.
