```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_MONTH_LEN 32

static int has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char normalized[MAX_MONTH_LEN];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_MONTH_LEN);
    if (len == 0 || len >= MAX_MONTH_LEN) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(normalized, months_31[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_MONTH_LEN];
    int result;

    printf("Enter a month name: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\r\n")] = '\0';

    result = has_31_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name provided\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 31 days\n", input);
    } else {
        printf("%s does not have 31 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-written from a security standpoint — the Flawfinder hits were largely false positives, but here is how each is addressed in the rewrite:

1. **`char normalized[MAX_MONTH_LEN]` (CWE-119/CWE-120):** The original already validated `len >= MAX_MONTH_LEN` before writing, guaranteeing space for the null terminator. This check is preserved, so no overflow is possible.

2. **`strlen` on possibly non-terminated string (CWE-126):** Replaced with `strnlen(month, MAX_MONTH_LEN)`, which bounds the read to `MAX_MONTH_LEN` bytes. A non-terminated buffer of that size returns `MAX_MONTH_LEN`, which the subsequent `len >= MAX_MONTH_LEN` check rejects — eliminating the over-read risk entirely.

3. **`char input[MAX_MONTH_LEN]` in `main` (CWE-119/CWE-120):** Safe as-is, because `fgets(input, sizeof(input), stdin)` limits input to the buffer size and guarantees null termination. Kept unchanged.

**Additional hardening added:**

- **Input validation:** Rejects non-alphabetic characters early, tightening the input domain and avoiding surprising behavior with punctuation, digits, or embedded control characters.
- **`fflush(stdout)`:** Ensures the prompt is visible before blocking on `fgets`, since `stdout` may be fully buffered when not attached to a terminal.
- **`tolower`/`isalpha` casts to `unsigned char`:** Retained from the original — this correctly avoids undefined behavior when `char` is signed and holds negative values.

Note: if targeting strict C99 without POSIX, `strnlen` can be replaced with `memchr(month, '\0', MAX_MONTH_LEN)` logic to achieve the same bounded scan.
