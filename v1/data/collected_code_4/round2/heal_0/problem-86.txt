```c
#include <stdio.h>

long long int nthCenteredHexagonalNumber(int n) {
    if(n < 1) {
        printf("Invalid input. Please enter a positive integer.\n");
        return -1;
    }

    return 3LL*n*(n-1) + 1;
}

int main() {
    int n;
    printf("Enter the position of the centered hexagonal number: ");
    scanf("%d", &n);
    long long int hexNum = nthCenteredHexagonalNumber(n);    
    if(hexNum != -1) {
        printf("The %dth centered hexagonal number is: %lld\n", n, hexNum);
    }

    return 0;
}
```
