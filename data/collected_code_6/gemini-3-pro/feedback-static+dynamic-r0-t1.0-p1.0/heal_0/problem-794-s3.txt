#include <stdio.h>
#include <string.h>
#include <stdbool.h>

bool match_string(const char *str) {
    if (str == NULL) {
        return false;
    }

    size_t len = strlen(str);
    if (len < 2) {
        return false;
    }

    if (str[0] == 'a' && str[len - 1] == 'b') {
        return true;
    }

    return false;
}

int main(void) {
    const char *test_strings[] = {
        "ab",
        "a123b",
        "a_b",
        "axb",
        "b",
        "a",
        "abc",
        "cab",
        NULL
    };

    for (int i = 0; test_strings[i] != NULL; ++i) {
        if (match_string(test_strings[i])) {
            printf("\"%s\" matches.\n", test_strings[i]);
        } else {
            printf("\"%s\" does not match.\n", test_strings[i]);
        }
    }

    return 0;
}
