```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

char *string_list_to_tuple(const char *list[], size_t count)
{
    size_t total_length = 3;
    size_t i;
    char *tuple;
    char *cursor;

    if (list == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t len;

        if (list[i] == NULL) {
            return NULL;
        }

        len = strnlen(list[i], MAX_STRING_LENGTH);

        if (total_length > SIZE_MAX - (len + 4)) {
            return NULL;
        }

        total_length += len + 2;

        if (i > 0) {
            total_length += 2;
        }
    }

    tuple = (char *)malloc(total_length);
    if (tuple == NULL) {
        return NULL;
    }

    cursor = tuple;
    *cursor++ = '(';

    for (i = 0; i < count; i++) {
        size_t len = strnlen(list[i], MAX_STRING_LENGTH);
        size_t space_left = total_length - (size_t)(cursor - tuple);

        if (i > 0) {
            *cursor++ = ',';
            *cursor++ = ' ';
            space_left -= 2;
        }

        if (space_left < len + 4) {
            free(tuple);
            return NULL;
        }

        *cursor++ = '\'';
        memcpy(cursor, list[i], len);
        cursor += len;
        *cursor++ = '\'';
    }

    *cursor++ = ')';
    *cursor = '\0';

    return tuple;
}

int main(void)
{
    const char *fruits[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(fruits) / sizeof(fruits[0]);
    char *result = string_list_to_tuple(fruits, count);

    if (result == NULL) {
        fprintf(stderr, "Error: failed to convert string list to tuple\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", result);

    free(result);
    result = NULL;

    return EXIT_SUCCESS;
}
```

Fixes applied:

- **variableScope**: `len` is now declared inside each loop body, reducing its scope to where it is actually used.
- **CWE-126 (strlen over-read)**: `strlen` replaced with `strnlen(list[i], MAX_STRING_LENGTH)`, bounding the scan so non-`'\0'`-terminated strings cannot cause an over-read.
- **CWE-120 (memcpy overflow)**: the remaining buffer capacity (`space_left`) is computed from the actual allocation before every copy, and `memcpy` only executes after verifying the data plus the enclosing quotes, closing paren, and terminator fit; on failure the allocation is freed and `NULL` returned.
- **Hardening**: an integer-overflow guard was added to the size accumulation loop, and the length pass and copy pass both use the same bounded `strnlen` measurement so the allocation size always matches the bytes written.
