#include <stdio.h>
#include <stdint.h>

int check_odd_parity(uint32_t num) {
    uint32_t count = 0;
    while (num > 0) {
        count += (num & 1);
        num >>= 1;
    }
    return (count % 2) != 0;
}

int main(void) {
    uint32_t test_num = 11; 
    
    if (check_odd_parity(test_num)) {
        printf("The number has odd parity.\n");
    } else {
        printf("The number has even parity.\n");
    }

    return 0;
}
