#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
    struct List* next;
} List;

Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (!newNode) {
        exit(EXIT_FAILURE);
    }
    newNode->data = data;
    newNode->next = NULL;
    return newNode;
}

List* createList() {
    List* newList = (List*)malloc(sizeof(List));
    if (!newList) {
        exit(EXIT_FAILURE);
    }
    newList->head = NULL;
    newList->next = NULL;
    return newList;
}

void appendNode(List* list, int data) {
    if (!list) return;
    Node* newNode = createNode(data);
    if (!list->head) {
        list->head = newNode;
    } else {
        Node* current = list->head;
        while (current->next) {
            current = current->next;
        }
        current->next = newNode;
    }
}

void appendList(List** head, List* newList) {
    if (!head || !newList) return;
    if (!*head) {
        *head = newList;
    } else {
        List* current = *head;
        while (current->next) {
            current = current->next;
        }
        current->next = newList;
    }
}

bool areListsEqual(List* list1, List* list2) {
    if (!list1 && !list2) return true;
    if (!list1 || !list2) return false;

    Node* curr1 = list1->head;
    Node* curr2 = list2->head;

    while (curr1 && curr2) {
        if (curr1->data != curr2->data) {
            return false;
        }
        curr1 = curr1->next;
        curr2 = curr2->next;
    }

    return curr1 == NULL && curr2 == NULL;
}

void freeNodeList(Node* head) {
    while (head) {
        Node* temp = head;
        head = head->next;
        free(temp);
    }
}

void freeList(List* list) {
    if (list) {
        freeNodeList(list->head);
        free(list);
    }
}

void freeListOfLists(List* head) {
    while (head) {
        List* temp = head;
        head = head->next;
        freeList(temp);
    }
}

void removeDuplicates(List** head) {
    if (!head || !*head) return;

    List* current = *head;
    while (current) {
        List* runner = current;
        while (runner->next) {
            if (areListsEqual(current, runner->next)) {
                List* duplicate = runner->next;
                runner->next = runner->next->next;
                freeList(duplicate);
            } else {
                runner = runner->next;
            }
        }
        current = current->next;
    }
}

void printListOfLists(List* head) {
    List* currentList = head;
    while (currentList) {
        Node* currentNode = currentList->head;
        printf("[ ");
        while (currentNode) {
            printf("%d ", currentNode->data);
            currentNode = currentNode->next;
        }
        printf("]\n");
        currentList = currentList->next;
    }
}

int main() {
    List* listOfLists = NULL;

    List* list1 = createList();
    appendNode(list1, 1);
    appendNode(list1, 2);
    appendNode(list1, 3);

    List* list2 = createList();
    appendNode(list2, 4);
    appendNode(list2, 5);

    List* list3 = createList();
    appendNode(list3, 1);
    appendNode(list3, 2);
    appendNode(list3, 3);

    List* list4 = createList();
    appendNode(list4, 6);

    appendList(&listOfLists, list1);
    appendList(&listOfLists, list2);
    appendList(&listOfLists, list3);
    appendList(&listOfLists, list4);

    removeDuplicates(&listOfLists);

    printListOfLists(listOfLists);

    freeListOfLists(listOfLists);

    return 0;
}
