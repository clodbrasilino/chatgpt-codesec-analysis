```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static void print_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1U);
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (*offset + (size_t)end) <= input_len) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN &&
                (*offset + (size_t)start + len) <= input_len) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    *offset += (size_t)matches[0].rm_eo;
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "Regex execution failed");
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

1. **`errbuf` (CWE-119/CWE-120, statically-sized array):** The buffer is zero-initialized with `memset`, `regerror` is called with `sizeof(errbuf) - 1U` to reserve space for the terminator, and the last byte is explicitly set to `'\0'`. No overflow is possible.

2. **`memcpy` (CWE-120, unchecked copy):** The destination is allocated with `malloc(len + 1U)`, guaranteeing it can hold exactly `len` bytes plus the null terminator. Before the copy, the code validates:
   - `start >= 0 && end >= start` — match offsets are sane,
   - `(*offset + (size_t)end) <= input_len` — the match lies within the input,
   - `len < MAX_INPUT_LEN` and `(*offset + (size_t)start + len) <= input_len` — the source range is fully in bounds,
   - the `malloc` return value is checked before use.

Additional defensive measures already present: `strnlen` bounds the input scan, NULL-pointer parameters are rejected, `regcomp`/`regexec` errors are reported via `regerror`, and `regfree` is always called to avoid leaks. The caller frees each returned string, so there are no memory leaks in `main` either. The code stands as-is with the comments removed.
