#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} Array;

typedef struct {
    int value;
    int index;
    int array_index;
} HeapNode;

typedef struct {
    HeapNode *nodes;
    size_t size;
    size_t capacity;
} MinHeap;

void array_init(Array *arr) {
    arr->data = NULL;
    arr->size = 0;
    arr->capacity = 0;
}

void array_push(Array *arr, int value) {
    if (arr->size == arr->capacity) {
        arr->capacity = (arr->capacity > 0) ? 2 * arr->capacity : 1;
        arr->data = realloc(arr->data, arr->capacity * sizeof(int));
    }
    arr->data[arr->size++] = value;
}

void min_heap_init(MinHeap *heap) {
    heap->nodes = NULL;
    heap->size = 0;
    heap->capacity = 0;
}

void min_heap_push(MinHeap *heap, HeapNode node) {
    if (heap->size == heap->capacity) {
        heap->capacity = (heap->capacity > 0) ? 2 * heap->capacity : 1;
        heap->nodes = realloc(heap->nodes, heap->capacity * sizeof(HeapNode));
    }
    heap->nodes[heap->size++] = node;
}

void min_heap_pop(MinHeap *heap) {
    if (heap->size == 0) return;
    heap->nodes[0] = heap->nodes[--heap->size];
    for (int i = 0; 2 * i + 1 < heap->size;) {
        int smallest = i;
        if (2 * i + 1 < heap->size && heap->nodes[2 * i + 1].value < heap->nodes[smallest].value)
            smallest = 2 * i + 1;
        if (2 * i + 2 < heap->size && heap->nodes[2 * i + 2].value < heap->nodes[smallest].value)
            smallest = 2 * i + 2;
        if (smallest != i) {
            HeapNode temp = heap->nodes[i];
            heap->nodes[i] = heap->nodes[smallest];
            heap->nodes[smallest] = temp;
            i = smallest;
        } else {
            break;
        }
    }
}

int min_heap_top(MinHeap *heap) {
    return (heap->size > 0) ? heap->nodes[0].value : 0;
}

void merge_sorted_arrays(const Array *arrays, size_t count, Array *result) {
    MinHeap heap;
    min_heap_init(&heap);

    for (size_t i = 0; i < count; ++i) {
        if (arrays[i].size > 0) {
            min_heap_push(&heap, (HeapNode){.value = arrays[i].data[0], .index = 0, .array_index = i});
        }
    }

    while (heap.size > 0) {
        HeapNode node = heap.nodes[0];
        array_push(result, node.value);
        min_heap_pop(&heap);

        if (node.index + 1 < arrays[node.array_index].size) {
            min_heap_push(&heap, (HeapNode){.value = arrays[node.array_index].data[node.index + 1], .index = node.index + 1, .array_index = node.array_index});
        }
    }
}

int main() {
    Array arrays[3];
    array_init(&arrays[0]);
    array_init(&arrays[1]);
    array_init(&arrays[2]);

    array_push(&arrays[0], 1);
    array_push(&arrays[0], 4);
    array_push(&arrays[0], 5);

    array_push(&arrays[1], 1);
    array_push(&arrays[1], 3);
    array_push(&arrays[1], 4);

    array_push(&arrays[2], 2);
    array_push(&arrays[2], 6);

    Array result;
    array_init(&result);

    merge_sorted_arrays(arrays, 3, &result);

    for (size_t i = 0; i < result.size; ++i) {
        printf("%d ", result.data[i]);
    }
    printf("\n");

    free(arrays[0].data);
    free(arrays[1].data);
    free(arrays[2].data);
    free(result.data);

    return 0;
}
