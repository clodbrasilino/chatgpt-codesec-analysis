```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t capacity = 8;
    size_t num = 0;
    size_t start = 0;
    size_t i;
    char **result;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(capacity * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || (i > 0 && isupper((unsigned char)input[i]))) {
            size_t seglen = i - start;
            char *segment;

            if (seglen == 0) {
                start = i;
                continue;
            }

            if (start > len || seglen > len - start || seglen >= SIZE_MAX) {
                for (size_t j = 0; j < num; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }

            segment = malloc(seglen + 1);
            if (segment == NULL) {
                for (size_t j = 0; j < num; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }

            memcpy(segment, input + start, seglen);
            segment[seglen] = '\0';

            if (num >= capacity) {
                size_t new_capacity;
                char **tmp;

                if (capacity > SIZE_MAX / 2 ||
                    (capacity * 2) > SIZE_MAX / sizeof(char *)) {
                    free(segment);
                    for (size_t j = 0; j < num; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }

                new_capacity = capacity * 2;
                tmp = realloc(result, new_capacity * sizeof(char *));
                if (tmp == NULL) {
                    free(segment);
                    for (size_t j = 0; j < num; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }
                result = tmp;
                capacity = new_capacity;
            }

            result[num] = segment;
            num++;
            start = i;
        }
    }

    *count = num;
    return result;
}

void free_split(char **parts, size_t count)
{
    size_t i;

    if (parts == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(parts[i]);
    }
    free(parts);
}

int main(void)
{
    const char *text = "helloWorldThisIsATest";
    size_t count = 0;
    size_t i;
    char **parts;

    parts = split_at_uppercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", parts[i]) < 0) {
            free_split(parts, count);
            return EXIT_FAILURE;
        }
    }

    free_split(parts, count);
    return EXIT_SUCCESS;
}
```

The `memcpy` flagged by Flawfinder is safe in this implementation: the destination buffer `segment` is allocated with exactly `seglen + 1` bytes immediately before the copy, and the source range `input + start` through `input + start + seglen` is validated against `len` (itself bounded by `MAX_INPUT_LEN` via `strnlen`) before the allocation. The bounds checks `start > len`, `seglen > len - start`, and `seglen >= SIZE_MAX` guarantee no overflow can occur in either the size arithmetic or the copy itself, so the warning is a false positive and no functional change was required.
