#include <stdio.h>
#include <math.h>

double square_pyramid_surface_area(double base_edge, double slant_height) {
    double base_area = base_edge * base_edge;
    double lateral_area = 2.0 * base_edge * slant_height;
    return base_area + lateral_area;
}

int main(void) {
    double base_edge, slant_height;
    
    printf("Enter base edge length: ");
    if (scanf("%lf", &base_edge) != 1 || base_edge <= 0) {
        printf("Invalid input for base edge length\n");
        return 1;
    }
    
    printf("Enter slant height: ");
    if (scanf("%lf", &slant_height) != 1 || slant_height <= 0) {
        printf("Invalid input for slant height\n");
        return 1;
    }
    
    double surface_area = square_pyramid_surface_area(base_edge, slant_height);
    printf("Surface area of square pyramid: %.2f\n", surface_area);
    
    return 0;
}
