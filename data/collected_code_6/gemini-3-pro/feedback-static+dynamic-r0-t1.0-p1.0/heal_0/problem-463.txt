#include <stdio.h>
#include <stdlib.h>

long long maxProduct(int* nums, int numsSize) {
    if (nums == NULL || numsSize == 0) {
        return 0;
    }

    long long max_so_far = nums[0];
    long long min_so_far = nums[0];
    long long result = nums[0];

    for (int i = 1; i < numsSize; i++) {
        long long curr = nums[i];
        long long temp_max = max_so_far * curr;
        long long temp_min = min_so_far * curr;

        max_so_far = curr > temp_max ? curr : temp_max;
        max_so_far = max_so_far > temp_min ? max_so_far : temp_min;

        min_so_far = curr < temp_min ? curr : temp_min;
        min_so_far = min_so_far < temp_max ? min_so_far : temp_max;

        if (max_so_far > result) {
            result = max_so_far;
        }
    }

    return result;
}

int main(void) {
    int arr[] = {2, 3, -2, 4};
    int size = sizeof(arr) / sizeof(arr[0]);

    long long max_prod = maxProduct(arr, size);
    
    printf("Maximum product subarray is: %lld\n", max_prod);

    return 0;
}
