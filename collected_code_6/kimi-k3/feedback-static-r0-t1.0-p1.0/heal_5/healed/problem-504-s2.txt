#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <stdlib.h>
#include <limits.h>
#include <ctype.h>

uint64_t cube_sum(uint32_t n)
{
    uint64_t sum = 0;
    for (uint32_t i = 1; i <= n; i++)
    {
        sum += (uint64_t)i * i * i;
    }
    return sum;
}

int main(void)
{
    char buffer[64];
    char *endptr;
    unsigned long value;
    uint32_t n;
    size_t len;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL)
    {
        return 1;
    }

    len = 0;
    while (len < sizeof(buffer) && buffer[len] != '\0')
    {
        len++;
    }

    if (len == sizeof(buffer))
    {
        return 1;
    }

    if (len > 0 && buffer[len - 1] != '\n' && !feof(stdin))
    {
        int c;
        while ((c = getchar()) != '\n' && c != EOF)
        {
        }
        return 1;
    }

    errno = 0;
    value = strtoul(buffer, &endptr, 10);

    if (errno == ERANGE || endptr == buffer || value > UINT32_MAX)
    {
        return 1;
    }

    while (*endptr != '\0' && isspace((unsigned char)*endptr))
    {
        endptr++;
    }

    if (*endptr != '\0')
    {
        return 1;
    }

    n = (uint32_t)value;
    printf("%" PRIu64 "\n", cube_sum(n));
    return 0;
}
