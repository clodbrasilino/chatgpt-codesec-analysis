```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int best;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            best = max_int(best, dp[i - 1][j] - prices[j]);
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The algorithm itself is the standard O(k·n) dynamic programming solution for "Best Time to Buy and Sell Stock IV," and after tracing it against the three test cases, it produces the correct results:

- `{3, 2, 6, 5, 0, 3}`, k=2 → **7** (buy at 2, sell at 6; buy at 0, sell at 3)
- `{1, 2, 3, 4, 5}`, k=2 → **4** (handled by the greedy fast path since k ≥ n/2)
- `{2, 4, 1}`, k=2 → **2** (also greedy fast path)

The "expected" values in the weakness comments (10, 7, 2 with results 5, 0, 0) don't correspond to this input data with k=2 — they appear to come from a broken earlier revision (likely one where the inner DP loop failed to carry `best` correctly or where the transaction loop bounds were off-by-one, yielding zero or half the profit). The current recurrence:

- `best = max(best, dp[i-1][j] - prices[j])` correctly tracks the best "buy state,"
- `dp[i][j] = max(dp[i][j-1], prices[j] + best)` correctly extends or sells,

is sound. Memory handling is also robust: allocation failures unwind partial allocations, all rows plus the pointer array are freed on success, and degenerate inputs (`NULL` prices, `n <= 1`, `k <= 0`) return 0 safely. The `k >= n/2` greedy shortcut both prevents oversized allocations and is mathematically equivalent to unlimited transactions in that regime.
