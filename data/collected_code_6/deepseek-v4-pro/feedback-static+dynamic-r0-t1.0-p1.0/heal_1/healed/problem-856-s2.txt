#include <stdio.h>

int minSwaps(const int arr[], int n) {
    int onesCount = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] == 1) {
            onesCount++;
        }
    }
    
    if (onesCount == 0 || onesCount == n) {
        return 0;
    }
    
    int maxOnes = 0;
    int currentOnes = 0;
    
    for (int i = 0; i < onesCount; i++) {
        if (arr[i] == 1) {
            currentOnes++;
        }
    }
    
    maxOnes = currentOnes;
    
    for (int i = onesCount; i < n; i++) {
        if (arr[i] == 1) {
            currentOnes++;
        }
        if (arr[i - onesCount] == 1) {
            currentOnes--;
        }
        if (currentOnes > maxOnes) {
            maxOnes = currentOnes;
        }
    }
    
    return onesCount - maxOnes;
}

int main(void) {
    int arr1[] = {0, 0, 1, 0, 1, 0, 1, 1};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    
    int arr2[] = {1, 0, 1, 0, 1};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    
    int arr3[] = {0, 0, 0, 0};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    
    int arr4[] = {1, 1, 1, 1};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);
    
    printf("%d\n", minSwaps(arr1, n1));
    printf("%d\n", minSwaps(arr2, n2));
    printf("%d\n", minSwaps(arr3, n3));
    printf("%d\n", minSwaps(arr4, n4));
    
    return 0;
}
