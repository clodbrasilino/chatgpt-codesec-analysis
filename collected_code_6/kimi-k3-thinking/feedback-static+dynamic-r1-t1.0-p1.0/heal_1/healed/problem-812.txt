```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static size_t bounded_strlen(const char *s, size_t max)
{
    size_t n = 0u;

    while (n < max && s[n] != '\0') {
        n++;
    }
    return n;
}

char *abbreviate_road(const char *input, size_t input_max)
{
    const char word[] = "road";
    const char replacement[] = "rd.";
    const size_t word_len = sizeof(word) - 1u;
    const size_t rep_len = sizeof(replacement) - 1u;
    const char *p;
    const char *src;
    char *dst;
    char *result;
    size_t count = 0u;
    size_t len;
    size_t new_size;
    size_t remaining;

    if (input == NULL || input_max == 0u) {
        return NULL;
    }

    len = bounded_strlen(input, input_max);
    if (len == input_max) {
        return NULL;
    }

    p = input;
    while ((p = strstr(p, word)) != NULL) {
        count++;
        p += word_len;
    }

    if (rep_len > word_len) {
        size_t growth = rep_len - word_len;
        if (count > (SIZE_MAX - len - 1u) / growth) {
            return NULL;
        }
        new_size = len + count * growth + 1u;
    } else {
        new_size = len - count * (word_len - rep_len) + 1u;
    }

    result = malloc(new_size);
    if (result == NULL) {
        return NULL;
    }

    src = input;
    dst = result;
    remaining = new_size;

    while ((p = strstr(src, word)) != NULL) {
        size_t prefix_len = (size_t)(p - src);

        if (prefix_len >= remaining) {
            free(result);
            return NULL;
        }
        memcpy(dst, src, prefix_len);
        dst += prefix_len;
        remaining -= prefix_len;

        if (rep_len >= remaining) {
            free(result);
            return NULL;
        }
        memcpy(dst, replacement, rep_len);
        dst += rep_len;
        remaining -= rep_len;

        src = p + word_len;
    }

    {
        size_t tail_len = strlen(src);

        if (tail_len >= remaining) {
            free(result);
            return NULL;
        }
        memcpy(dst, src, tail_len + 1u);
    }

    return result;
}

int main(void)
{
    const char input[] = "Take the road and follow the road to the end of the road.";
    char *output = abbreviate_road(input, sizeof(input));

    if (output == NULL) {
        fprintf(stderr, "Error: abbreviation failed\n");
        return EXIT_FAILURE;
    }

    printf("Original:    %s\n", input);
    printf("Abbreviated: %s\n", output);

    free(output);
    output = NULL;

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **CWE-126 (strlen over-read):** The two `strlen` calls on the literals `"road"`/`"rd."` were replaced with `sizeof(...) - 1`, eliminating them entirely. For `input`, the function now takes an `input_max` bound and uses a bounded length scan; if no NUL terminator is found within the bound, the function rejects the input instead of over-reading.

2. **CWE-120 (memcpy overflows):** A `remaining` capacity tracker is maintained for the destination buffer; every `memcpy` is preceded by an explicit bounds check, and the failure path frees the partially built buffer.

3. **CWE-120 (strcpy):** Replaced with a length-checked `memcpy` of the tail (including the NUL terminator), guaranteeing the copy stays within the allocated buffer.

4. **Hardening:** Added an overflow-safe `new_size` computation (guards `count * growth` against `SIZE_MAX` wrap), and `main` now uses a character array so `sizeof(input)` supplies the exact bounded size.
