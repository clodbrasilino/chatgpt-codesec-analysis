```c
#include <stdio.h>
#include <stdlib.h>

void square_elements(int *arr, int n) {
    for (int i = 0; i < n; ++i) {
        arr[i] = arr[i] * arr[i];
    }
}

int main() {
    int numbers[] = {1, 2, 3, 4, 5};
    int n = sizeof(numbers) / sizeof(numbers[0]);

    square_elements(numbers, n);

    for (int i = 0; i < n; ++i) {
        printf("%d ", numbers[i]);
    }

    return 0;
}
```
