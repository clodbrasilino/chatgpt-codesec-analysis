#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void remove_chars(char *str1, const char *str2) {
    size_t len = strlen(str1);
    for (size_t i = 0; i < len; i++) {
        for (size_t j = 0; str2[j] != '\0'; j++) {
            if (str1[i] == str2[j]) {
                memmove(&str1[i], &str1[i + 1], len - i);
                i--;
                len--;
                break;
            }
        }
    }
}

int main() {
    char str1[100] = "Hello, World!";
    const char *str2 = ",!";

    remove_chars(str1, str2);
    printf("%s\n", str1);

    return 0;
}
