#include <stdio.h>
#include <string.h>
#include <regex.h>
#include <stdlib.h>

void swap_whitespace_underscore(char *str, size_t len) {
    regex_t re;
    if (regcomp(&re, " ", REG_EXTENDED) != 0) return;
    regmatch_t match;
    char *new_str = malloc(len + 1);
    if (new_str == NULL) return;
    strncpy(new_str, str, len + 1);
    new_str[len] = '\0';
    while (regexec(&re, new_str, 1, &match, 0) == 0 && match.rm_so < len) {
        memmove(new_str + match.rm_so + 1, new_str + match.rm_so, len - match.rm_so);
        new_str[match.rm_so] = '_';
        len++;
    }
    strncpy(str, new_str, len + 1);
    str[len] = '\0';
    free(new_str);
    regfree(&re);

    if (regcomp(&re, "_", REG_EXTENDED) != 0) return;
    new_str = malloc(len + 1);
    if (new_str == NULL) return;
    strncpy(new_str, str, len + 1);
    new_str[len] = '\0';
    while (regexec(&re, new_str, 1, &match, 0) == 0 && match.rm_so < len) {
        memmove(new_str + match.rm_so + 1, new_str + match.rm_so, len - match.rm_so);
        new_str[match.rm_so] = ' ';
        len++;
    }
    strncpy(str, new_str, len + 1);
    str[len] = '\0';
    free(new_str);
    regfree(&re);
}

int main() {
    char input[1024];
    if (fgets(input, sizeof(input), stdin) != NULL) {
        size_t len = strcspn(input, "\n");
        if (len >= 1023) {
            input[1023] = '\0';
            len = 1023;
        } else {
            input[len] = '\0';
        }
        swap_whitespace_underscore(input, len);
        printf("%s\n", input);
    }
    return 0;
}
