#include <stdio.h>
#include <stdlib.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

int goldMine(int gold[][10], int m, int n) {
    int dp[10][10];
    for (int col = n - 1; col >= 0; --col) {
        for (int row = 0; row < m; ++row) {
            int right = (col == n - 1) ? 0 : dp[row][col + 1];
            int right_up = (row == 0 || col == n - 1) ? 0 : dp[row - 1][col + 1];
            int right_down = (row == m - 1 || col == n - 1) ? 0 : dp[row + 1][col + 1];
            dp[row][col] = gold[row][col] + max(right, max(right_up, right_down));
        }
    }
    int result = 0;
    for (int i = 0; i < m; ++i) {
        result = max(result, dp[i][0]);
    }
    return result;
}

int main() {
    int gold[4][10] = {
        {1, 3, 1, 5},
        {2, 2, 4, 1},
        {5, 0, 2, 3},
        {0, 6, 1, 2}
    };
    int m = 4;
    int n = 4;
    printf("Maximum gold: %d\n", goldMine(gold, m, n));
    return 0;
}
