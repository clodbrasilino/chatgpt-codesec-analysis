#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <errno.h>

static int binomial_probability(long n, long k, double p, double *result)
{
    double log_prob;

    if (result == NULL) {
        return -1;
    }
    if (n < 0 || k < 0 || k > n) {
        return -1;
    }
    if (p < 0.0 || p > 1.0 || isnan(p)) {
        return -1;
    }

    if (p == 0.0) {
        *result = (k == 0) ? 1.0 : 0.0;
        return 0;
    }
    if (p == 1.0) {
        *result = (k == n) ? 1.0 : 0.0;
        return 0;
    }

    log_prob = lgamma((double)n + 1.0)
             - lgamma((double)k + 1.0)
             - lgamma((double)(n - k) + 1.0)
             + (double)k * log(p)
             + (double)(n - k) * log(1.0 - p);

    *result = exp(log_prob);

    if (isnan(*result) || isinf(*result)) {
        return -1;
    }

    return 0;
}

int main(void)
{
    long n;
    long k;
    double p;
    double probability;

    printf("Enter number of trials (n): ");
    if (scanf("%ld", &n) != 1) {
        fprintf(stderr, "Error: invalid input for n\n");
        return EXIT_FAILURE;
    }

    printf("Enter number of successes (k): ");
    if (scanf("%ld", &k) != 1) {
        fprintf(stderr, "Error: invalid input for k\n");
        return EXIT_FAILURE;
    }

    printf("Enter probability of success (p): ");
    if (scanf("%lf", &p) != 1) {
        fprintf(stderr, "Error: invalid input for p\n");
        return EXIT_FAILURE;
    }

    if (binomial_probability(n, k, p, &probability) != 0) {
        fprintf(stderr, "Error: invalid parameters for binomial probability\n");
        return EXIT_FAILURE;
    }

    if (printf("P(X = %ld) = %.10f\n", k, probability) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
