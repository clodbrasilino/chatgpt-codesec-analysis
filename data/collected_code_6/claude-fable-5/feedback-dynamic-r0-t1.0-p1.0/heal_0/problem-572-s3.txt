#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

static bool contains(const int *arr, size_t count, int value)
{
    size_t i;

    for (i = 0; i < count; i++) {
        if (arr[i] == value) {
            return true;
        }
    }
    return false;
}

static size_t remove_duplicates(const int *src, size_t src_len, int *dst, size_t dst_cap)
{
    size_t i;
    size_t count = 0;

    if ((src == NULL) || (dst == NULL)) {
        return 0;
    }

    for (i = 0; i < src_len; i++) {
        if (!contains(dst, count, src[i])) {
            if (count >= dst_cap) {
                break;
            }
            dst[count] = src[i];
            count++;
        }
    }
    return count;
}

static int process_list(const int *list, size_t len)
{
    int *result;
    size_t new_len;
    size_t i;

    if ((list == NULL) || (len == 0U)) {
        return -1;
    }

    result = malloc(len * sizeof(int));
    if (result == NULL) {
        return -1;
    }

    new_len = remove_duplicates(list, len, result, len);

    printf("Result: ");
    for (i = 0; i < new_len; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);
    return 0;
}

int main(void)
{
    const int list1[] = {1, 2, 2, 3, 4, 4, 5};
    const int list2[] = {7, 7, 8, 9, 9, 10};
    int status;

    status = process_list(list1, sizeof(list1) / sizeof(list1[0]));
    if (status != 0) {
        fprintf(stderr, "Failed to process list 1\n");
        return EXIT_FAILURE;
    }

    status = process_list(list2, sizeof(list2) / sizeof(list2[0]));
    if (status != 0) {
        fprintf(stderr, "Failed to process list 2\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
