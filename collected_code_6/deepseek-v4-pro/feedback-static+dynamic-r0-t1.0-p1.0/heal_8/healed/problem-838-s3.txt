#include <stdio.h>
#include <string.h>

int minSwaps(const char *s1, const char *s2, size_t len) {
    int count1 = 0, count2 = 0;

    for (size_t i = 0; i < len; i++) {
        if (s1[i] == '1') count1++;
        if (s2[i] == '1') count2++;
    }

    if (count1 != count2) {
        return -1;
    }

    int pos1 = 0, pos2 = 0;
    for (size_t i = 0; i < len; i++) {
        if (s1[i] != s2[i]) {
            if (s1[i] == '1') pos1++;
            else pos2++;
        }
    }

    if (pos1 % 2 == 0 && pos2 % 2 == 0) {
        return pos1 / 2 + pos2 / 2;
    } else if (pos1 % 2 == 1 && pos2 % 2 == 1) {
        return pos1 / 2 + pos2 / 2 + 2;
    }

    return -1;
}

int main(void) {
    char s1[102] = {0};
    char s2[102] = {0};

    printf("Enter first binary string: ");
    if (fgets(s1, sizeof(s1), stdin) == NULL) {
        printf("-1\n");
        return 0;
    }
    s1[strcspn(s1, "\n")] = '\0';

    printf("Enter second binary string: ");
    if (fgets(s2, sizeof(s2), stdin) == NULL) {
        printf("-1\n");
        return 0;
    }
    s2[strcspn(s2, "\n")] = '\0';

    size_t len1 = strnlen(s1, sizeof(s1));
    size_t len2 = strnlen(s2, sizeof(s2));

    if (len1 != len2 || len1 > 100 || len1 == 0) {
        printf("-1\n");
        return 0;
    }

    for (size_t i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') || (s2[i] != '0' && s2[i] != '1')) {
            printf("-1\n");
            return 0;
        }
    }

    int result = minSwaps(s1, s2, len1);
    printf("%d\n", result);

    return 0;
}
