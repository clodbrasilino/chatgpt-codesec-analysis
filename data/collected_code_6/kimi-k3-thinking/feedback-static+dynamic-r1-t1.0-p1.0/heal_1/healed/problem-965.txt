```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 65536

static size_t bounded_strlen(const char *s, size_t max)
{
    size_t n = 0;

    if (s == NULL) {
        return 0;
    }

    while (n < max && s[n] != '\0') {
        n++;
    }

    return n;
}

char *camel_to_snake(const char *input)
{
    if (input == NULL) {
        return NULL;
    }

    size_t len = bounded_strlen(input, MAX_INPUT_LEN);

    if (len == MAX_INPUT_LEN && input[len] != '\0') {
        return NULL;
    }

    if (len > (SIZE_MAX - 1) / 2) {
        return NULL;
    }

    size_t max_len = (len * 2) + 1;

    char *output = malloc(max_len);
    if (output == NULL) {
        return NULL;
    }

    size_t out_idx = 0;
    for (size_t i = 0; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (isupper(c)) {
            if (i > 0) {
                output[out_idx++] = '_';
            }
            output[out_idx++] = (char)tolower(c);
        } else {
            output[out_idx++] = (char)c;
        }
    }
    output[out_idx] = '\0';

    return output;
}

int main(void)
{
    const char *tests[] = {
        "helloWorld",
        "camelCaseString",
        "myVeryLongVariableName",
        "simple",
        "A",
        "already_snake"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = camel_to_snake(tests[i]);
        if (result == NULL) {
            fprintf(stderr, "Error: conversion failed for input \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("Input:  %s\n", tests[i]);
        printf("Output: %s\n\n", result);
        free(result);
        result = NULL;
    }

    return EXIT_SUCCESS;
}
```

**Key fixes applied:**

1. **CWE-126 (buffer over-read via `strlen`):** Replaced the unbounded `strlen()` call with a `bounded_strlen()` helper that scans at most `MAX_INPUT_LEN` bytes. If no NUL terminator is found within that bound, the function safely returns `NULL` instead of performing an over-read.

2. **Integer overflow hardening:** Added a check that `len > (SIZE_MAX - 1) / 2` before computing `(len * 2) + 1`, guarding against allocation-size wraparound, with `<stdint.h>` included for `SIZE_MAX`.
