```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *demlo_number(int n) {
    if (n <= 0) {
        return NULL;
    }

    size_t size = 2 * n;
    char *result = malloc(size);
    if (!result) {
        return NULL;
    }

    char *p = result;

    for (int i = 1; i <= n; i++) {
        *p++ = '0' + i;
    }
    for (int i = n - 1; i >= 1; i--) {
        *p++ = '0' + i;
    }
    *p = '\0';

    return result;
}

int main(void) {
    int n = 5;
    char *demlo = demlo_number(n);

    if (demlo) {
        printf("%s\n", demlo);
        free(demlo);
    } else {
        printf("Invalid input or allocation failure\n");
    }

    return 0;
}
```
