```c
#include <stdio.h>
#include <stdlib.h>

size_t count_true(const int *list, size_t size) {
    if (list == NULL) {
        return 0;
    }

    size_t count = 0;
    for (size_t i = 0; i < size; i++) {
        if (list[i] != 0) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int list[] = {0, 1, 1, 0, 1, 1, 0, 1};
    size_t size = sizeof(list) / sizeof(list[0]);

    size_t true_count = count_true(list, size);
    printf("%zu\n", true_count);

    return 0;
}
```
