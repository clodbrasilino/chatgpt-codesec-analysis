#include <stdio.h>
#include <stdlib.h>

void sort_array(int *arr, int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

int find_triplet(int *arr, int n, int sum, int *result) {
    if (arr == NULL || n < 3 || result == NULL) {
        return -1;
    }
    sort_array(arr, n);
    for (int i = 0; i < n - 2; i++) {
        if (i > 0 && arr[i] == arr[i - 1]) {
            continue;
        }
        int left = i + 1;
        int right = n - 1;
        while (left < right) {
            int current_sum = arr[i] + arr[left] + arr[right];
            if (current_sum == sum) {
                result[0] = arr[i];
                result[1] = arr[left];
                result[2] = arr[right];
                return 0;
            } else if (current_sum < sum) {
                left++;
            } else {
                right--;
            }
        }
    }
    return -1;
}

int main() {
    int arr[] = {12, 3, 4, 1, 6, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    int sum = 24;
    int result[3];
    int status = find_triplet(arr, n, sum, result);
    if (status == 0) {
        printf("%d %d %d\n", result[0], result[1], result[2]);
    } else {
        printf("No triplet found\n");
    }
    return 0;
}
