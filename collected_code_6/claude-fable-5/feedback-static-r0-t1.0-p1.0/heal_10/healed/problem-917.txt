```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

static int copy_match(Match *dest, const char *src, size_t src_len)
{
    size_t copy_len;

    if (dest == NULL || src == NULL || src_len == 0) {
        return -1;
    }

    if (src_len >= sizeof(dest->text)) {
        copy_len = sizeof(dest->text) - 1;
    } else {
        copy_len = src_len;
    }

    if (copy_len == 0 || copy_len >= sizeof(dest->text)) {
        return -1;
    }

    memcpy(dest->text, src, copy_len);
    dest->text[copy_len] = '\0';
    return 0;
}

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);

    if (len >= MAX_INPUT) {
        return -1;
    }

    i = 0;

    while (i + 1 < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;

            if (match_len > 0 && match_len <= len - start) {
                if (copy_match(&matches[found], &input[start], match_len) == 0) {
                    found++;
                }
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

1. **`char text[MAX_INPUT]` in `Match` (CWE-119/CWE-120):** This is a false positive in practice. The array is only ever written through `copy_match()`, which clamps the copy length to `sizeof(dest->text) - 1` and always null-terminates. The static sizing is safe because all writes are bounds-checked.

2. **`memcpy` in `copy_match` (CWE-120):** Mitigated. Before the `memcpy` executes, `copy_len` is guaranteed to be strictly less than `sizeof(dest->text)`, leaving room for the null terminator written immediately after. The destination can always hold the source data.

3. **`char input[MAX_INPUT]` in `main` (CWE-119/CWE-120):** Mitigated. Input is read exclusively via `fgets(input, sizeof(input), stdin)`, which limits the read to the buffer size. A defensive explicit null terminator is set at `input[sizeof(input) - 1]`, and `strnlen` (rather than `strlen`) is used to bound the length scan.

Additional defensive measures already present and preserved: `isupper`/`islower` receive `unsigned char` casts to avoid undefined behavior with negative `char` values, all pointer parameters are NULL-checked, and `strnlen` bounds every length computation. The code as written contains no exploitable buffer overflow; the Flawfinder hits are risk-2 informational findings that are properly mitigated by the surrounding bounds checks.
