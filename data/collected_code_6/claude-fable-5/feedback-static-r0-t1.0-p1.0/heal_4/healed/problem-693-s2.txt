#include <stdlib.h>
#include <string.h>

char *remove_multiple_spaces(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *output;
    const char *cursor;
    size_t input_len;
    size_t out_len;
    size_t remainder_len;
    size_t output_size;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    input_len = strnlen(input, (size_t)-1 / 2U);

    if (input_len >= ((size_t)-1 / 2U)) {
        regfree(&regex);
        return NULL;
    }

    output_size = input_len + 1U;

    output = malloc(output_size);
    if (output == NULL) {
        regfree(&regex);
        return NULL;
    }

    out_len = 0U;
    cursor = input;

    while ((ret = regexec(&regex, cursor, 1U, &match, 0)) == 0) {
        size_t prefix_len;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            free(output);
            regfree(&regex);
            return NULL;
        }

        prefix_len = (size_t)match.rm_so;

        if (prefix_len > (output_size - 1U) ||
            out_len > ((output_size - 1U) - prefix_len)) {
            free(output);
            regfree(&regex);
            return NULL;
        }

        if ((out_len + prefix_len + 1U) > output_size) {
            free(output);
            regfree(&regex);
            return NULL;
        }

        memcpy(output + out_len, cursor, prefix_len);
        out_len += prefix_len;
        output[out_len] = ' ';
        out_len += 1U;
        cursor += match.rm_eo;
    }

    if (ret != REG_NOMATCH) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    if ((size_t)(cursor - input) > input_len) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    remainder_len = strnlen(cursor, input_len - (size_t)(cursor - input));

    if (remainder_len > (output_size - 1U) ||
        out_len > ((output_size - 1U) - remainder_len)) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    if ((out_len + remainder_len + 1U) > output_size) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    memcpy(output + out_len, cursor, remainder_len);
    out_len += remainder_len;
    output[out_len] = '\0';

    regfree(&regex);
    return output;
}

int main(void)
{
    const char *test = "This   is  a    string   with     multiple spaces.";
    char *result;

    result = remove_multiple_spaces(test);

        fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;

    if (printf("Original: %s\n", test) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}

**Analysis summary:** The two Flawfinder warnings on the `memcpy` calls (CWE-120) are false positives in this code, and the rewritten version preserves the mitigations that make them safe:

1. **First `memcpy` (prefix copy):** Before copying, the code verifies that `prefix_len` and `out_len` combined cannot exceed `output_size - 1`, using subtraction-based comparisons (`out_len > (output_size - 1U) - prefix_len`) that avoid integer overflow, plus a redundant defense-in-depth check `(out_len + prefix_len + 1U) > output_size`. Since the output can never grow larger than the input (each run of multiple spaces collapses to a single space), the buffer of `input_len + 1` bytes is always sufficient.

2. **Second `memcpy` (remainder copy):** The same overflow-safe bounds validation is applied to `remainder_len`, and `strnlen` is bounded by the remaining input length rather than scanning unbounded memory.

Additional hardening already present and retained: `NULL` input handling, `regcomp`/`regexec` return-value checks, validation of `rm_so`/`rm_eo` match offsets before casting to `size_t`, capping `input_len` at `SIZE_MAX / 2` to prevent allocation-size overflow, and consistent cleanup (`free` + `regfree`) on every error path to prevent memory and regex-handle leaks.
