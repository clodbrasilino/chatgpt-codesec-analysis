```C
#include <stdio.h>
#include <stdlib.h>

void flip(int arr[], int i) {
    int start = 0;
    while (start < i) {
        int temp = arr[start];
        arr[start] = arr[i];
        arr[i] = temp;
        start++;
        i--;
    }
}

int findMax(int arr[], int n){
    int mi, i;
    for (mi = 0, i = 0; i < n; ++i)
        if (arr[i] > arr[mi])
                mi = i;
    return mi;
}

void pancakeSort(int *arr, int n) {
    for (int curr_size = n; curr_size > 1; --curr_size) {
        int mi = findMax(arr, curr_size);
        if (mi != curr_size-1) {
            flip(arr, mi);
            flip(arr, curr_size-1);
        }
    }
}

int main(){
    int arr[] = {23, 10, 20, 11, 12, 6, 7};
    int n = sizeof(arr)/sizeof(arr[0]);

    pancakeSort(arr, n);

    printf("Sorted Array: ");
    for (int i=0; i<n; i++)
        printf("%d ", arr[i]);

    return 0;
}
```
The algorithm runs as follows:

1. Find the index of the maximum number in the unsorted part of the list,
2. If the maximum number is not in the correct place, flip the numbers from the first element to the maximum number's place,
3. Flip the whole unsorted part of the list to put the previously maximum number at the beginning.
