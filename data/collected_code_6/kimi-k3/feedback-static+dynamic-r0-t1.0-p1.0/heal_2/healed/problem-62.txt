```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_smallest(const int *arr, size_t size, int *out) {
    if (arr == NULL || size == 0 || out == NULL) {
        return -1;
    }
    
    int smallest = arr[0];
    for (size_t i = 1; i < size; i++) {
        if (arr[i] < smallest) {
            smallest = arr[i];
        }
    }
    *out = smallest;
    return 0;
}

int main(void) {
    int numbers[] = {34, 15, 88, 2, 47, 19};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);
    
    int result;
    if (find_smallest(numbers, size, &result) != 0) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    printf("Smallest number: %d\n", result);
    return EXIT_SUCCESS;
}
```
