```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_STR_LEN 4096U

bool is_alternating(const char *str)
{
    size_t len;
    size_t i;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return false;
    }

    if (len < 2U) {
        return false;
    }

    if (str[0] == str[1]) {
        return false;
    }

    for (i = 2U; i < len; i++) {
        if (str[i] != str[i - 2U]) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *tests[] = {
        "ababab",
        "xyxyxyxy",
        "aaaa",
        "abcabc",
        "ab",
        "a",
        ""
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        if (is_alternating(tests[i])) {
            if (printf("\"%s\": yes\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("\"%s\": no\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Key change:** Replaced `strlen` with `strnlen` bounded by `MAX_STR_LEN` to address the CWE-126 over-read risk flagged by Flawfinder. If the string is not properly NUL-terminated within the bound, `strnlen` returns `MAX_STR_LEN` and the function safely rejects the input instead of reading past the buffer. All other logic (NULL check, minimum length check, alternation verification, and `printf` return-value checks in `main`) is preserved unchanged.
