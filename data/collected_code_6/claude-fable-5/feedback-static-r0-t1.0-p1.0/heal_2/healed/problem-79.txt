```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define MAX_WORD_LENGTH 256

bool is_word_length_odd(const char *word, size_t max_length)
{
    size_t length;

    if (word == NULL) {
        return false;
    }

    length = strnlen(word, max_length);
    return (length % 2U) != 0U;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[MAX_WORD_LENGTH];
    size_t length;

    printf("Enter a word: ");

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    length = strnlen(buffer, sizeof(buffer));

    if (length > 0U && buffer[length - 1U] == '\n') {
        buffer[length - 1U] = '\0';
        length--;
    } else if (length == sizeof(buffer) - 1U) {
        discard_remaining_input();
    }

    if (length == 0U || buffer[0] == '\0') {
        fprintf(stderr, "Error: empty input.\n");
        return 1;
    }

    if (is_word_length_odd(buffer, sizeof(buffer))) {
        printf("The length of the word \"%s\" is odd.\n", buffer);
    } else {
        printf("The length of the word \"%s\" is even.\n", buffer);
    }

    return 0;
}
```

**Analysis of the identified weaknesses and their mitigations:**

1. **CWE-119/CWE-120 (statically-sized `buffer`)** — The risk is mitigated because:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, never overflowing.
   - Explicit null termination at `buffer[sizeof(buffer) - 1U]` guarantees a terminated string even in edge cases.
   - `strnlen` is used instead of `strlen`, bounding all length computations to `sizeof(buffer)`.

2. **CWE-120/CWE-20 (`getchar` in a loop)** — The loop only *discards* excess characters when the input line exceeded the buffer; it never writes to memory, so no overflow is possible. It correctly terminates on both `'\n'` and `EOF`, preventing an infinite loop on truncated streams. I extracted it into `discard_remaining_input()` for clarity and reuse.

The remaining logic (newline stripping, empty-input rejection, bounded parity check) was already sound, so it is preserved as-is.
