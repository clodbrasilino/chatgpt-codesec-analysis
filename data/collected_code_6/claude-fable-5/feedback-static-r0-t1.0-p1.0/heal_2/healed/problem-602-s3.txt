#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define MAX_INPUT_LEN 256

int find_first_repeated_char(const char *str, size_t max_len, char *result)
{
    int seen[UCHAR_MAX + 1] = {0};
    size_t i;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char repeated = '\0';
    int status;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    status = find_first_repeated_char(buffer, sizeof(buffer), &repeated);

    if (status < 0) {
        fprintf(stderr, "Invalid arguments\n");
        return 1;
    }

    if (status == 0) {
        if (printf("First repeated character: %c\n", repeated) < 0) {
            return 1;
        }
    } else {
        if (printf("No repeated character found\n") < 0) {
            return 1;
        }
    }

    return 0;
}

**Analysis of the flagged weakness (CWE-119/CWE-120):**

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits reads to at most 255 characters plus the null terminator, so no overflow is possible.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure (fgets already null-terminates, but this protects against future refactoring).
3. **Bounded scanning:** `find_first_repeated_char` receives `max_len` and never iterates past it, so even a non-terminated buffer could not cause an out-of-bounds read.
4. **Safe indexing:** The cast to `unsigned char` before indexing `seen[]` prevents negative array indices on platforms where `char` is signed — an often-missed subtle bug that this code handles correctly.
5. **NULL checks and return-value checks** on `fgets` and `printf` are already in place.
