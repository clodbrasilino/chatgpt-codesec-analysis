#include <stdio.h>
#include <stdlib.h>

int bitwise_xor_tuples(const int *tup1, const int *tup2, int *result, size_t len)
{
    size_t i;

    if (tup1 == NULL || tup2 == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        result[i] = tup1[i] ^ tup2[i];
    }

    return 0;
}

int main(void)
{
    const int tuple1[] = {10, 4, 6, 9};
    const int tuple2[] = {5, 2, 3, 3};
    size_t len = sizeof(tuple1) / sizeof(tuple1[0]);
    int *result;
    size_t i;
    int status;

    result = malloc(len * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    status = bitwise_xor_tuples(tuple1, tuple2, result, len);
    if (status != 0) {
        fprintf(stderr, "XOR operation failed\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("(");
    for (i = 0; i < len; i++) {
        printf("%d", result[i]);
        if (i < len - 1) {
            printf(", ");
        }
    }
    printf(")\n");

    free(result);
    return EXIT_SUCCESS;
}
