#include <stdio.h>
#include <stdlib.h>

int *filter_odd_numbers(const int *input, size_t length, size_t *output_length) {
    if (input == NULL || output_length == NULL) {
        return NULL;
    }

    size_t odd_count = 0;
    for (size_t i = 0; i < length; i++) {
        if (input[i] % 2 != 0) {
            odd_count++;
        }
    }

    if (odd_count == 0) {
        *output_length = 0;
        return NULL;
    }

    int *result = (int *)malloc(odd_count * sizeof(int));
    if (result == NULL) {
        *output_length = 0;
        return NULL;
    }

    size_t result_index = 0;
    for (size_t i = 0; i < length; i++) {
        if (input[i] % 2 != 0) {
            result[result_index++] = input[i];
        }
    }

    *output_length = odd_count;
    return result;
}

int main(void) {
    int test_cases[][4] = {
        {1, 3, 5, 7},
        {45, 67, 93, 0},
        {5, 7, 9, 3}
    };
    size_t test_lengths[] = {4, 3, 4};
    size_t num_tests = sizeof(test_lengths) / sizeof(test_lengths[0]);

    for (size_t t = 0; t < num_tests; t++) {
        size_t odd_count = 0;
        int *odd_numbers = filter_odd_numbers(test_cases[t], test_lengths[t], &odd_count);

        if (odd_numbers == NULL) {
            if (odd_count == 0) {
                printf("[]\n");
            } else {
                printf("Memory allocation failed\n");
            }
            continue;
        }

        printf("[");
        for (size_t i = 0; i < odd_count; i++) {
            printf("%d", odd_numbers[i]);
            if (i < odd_count - 1) {
                printf(", ");
            }
        }
        printf("]\n");

        free(odd_numbers);
    }

    return 0;
}
