#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCHES 100
#define MAX_ADVERB_LEN 50

typedef struct {
    char adverb[MAX_ADVERB_LEN];
    int position;
} AdverbMatch;

static char *safe_strdup(const char *str, size_t max_len) {
    char *copy;
    size_t len;
    
    if (str == NULL || max_len == 0) {
        return NULL;
    }
    
    len = strnlen(str, max_len);
    if (len >= max_len) {
        len = max_len - 1;
    }
    
    copy = malloc(len + 1);
    if (copy != NULL) {
        if (len > 0) {
            memcpy(copy, str, len);
        }
        copy[len] = '\0';
    }
    return copy;
}

int find_adverbs(const char *sentence, AdverbMatch matches[], int max_matches) {
    regex_t regex;
    regmatch_t pmatch[2];
    const char *pattern = "\\b[A-Za-z]+ly\\b";
    int match_count = 0;
    int offset = 0;
    int ret;
    char *remaining;
    size_t sentence_len;
    size_t remaining_len;
    size_t max_sentence_len;
    
    if (sentence == NULL || matches == NULL || max_matches <= 0) {
        return 0;
    }
    
    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return 0;
    }
    
    max_sentence_len = (size_t)MAX_ADVERB_LEN * (size_t)MAX_MATCHES;
    sentence_len = strnlen(sentence, max_sentence_len);
    remaining = safe_strdup(sentence, max_sentence_len);
    if (remaining == NULL) {
        regfree(&regex);
        return 0;
    }
    
    remaining_len = sentence_len;
    
    while (match_count < max_matches && offset < (int)sentence_len && remaining_len > 0) {
        ret = regexec(&regex, remaining, 2, pmatch, 0);
        
        if (ret == REG_NOMATCH) {
            break;
        }
        
        if (ret != 0) {
            break;
        }
        
        if (pmatch[0].rm_so >= 0 && pmatch[0].rm_eo > pmatch[0].rm_so) {
            size_t match_start = (size_t)pmatch[0].rm_so;
            size_t match_end = (size_t)pmatch[0].rm_eo;
            size_t match_len = match_end - match_start;
            
            if (match_len > 0 && match_len < MAX_ADVERB_LEN) {
                size_t copy_len = match_len;
                if (copy_len >= sizeof(matches[match_count].adverb)) {
                    copy_len = sizeof(matches[match_count].adverb) - 1;
                }
                
                if (match_start < remaining_len && match_start + copy_len <= remaining_len) {
                    memcpy(matches[match_count].adverb, remaining + match_start, copy_len);
                    matches[match_count].adverb[copy_len] = '\0';
                    matches[match_count].position = offset + (int)match_start;
                    match_count++;
                }
            }
        }
        
        if ((size_t)pmatch[0].rm_eo >= remaining_len) {
            break;
        }
        
        offset += (int)pmatch[0].rm_eo;
        
        {
            size_t move_len = remaining_len - (size_t)pmatch[0].rm_eo;
            if (move_len < remaining_len && move_len > 0) {
                memmove(remaining, remaining + pmatch[0].rm_eo, move_len);
                remaining[move_len] = '\0';
                remaining_len = move_len;
            } else {
                break;
            }
        }
    }
    
    free(remaining);
    regfree(&regex);
    
    return match_count;
}

int main(void) {
    const char *sentence = "The quickly running fox silently jumped over the lazy dog quietly.";
    AdverbMatch matches[MAX_MATCHES];
    int count;
    int i;
    
    count = find_adverbs(sentence, matches, MAX_MATCHES);
    
    printf("Sentence: %s\n", sentence);
    printf("Found %d adverbs:\n", count);
    
    for (i = 0; i < count; i++) {
        printf("  Adverb: '%s' at position %d\n", matches[i].adverb, matches[i].position);
    }
    
    return 0;
}
