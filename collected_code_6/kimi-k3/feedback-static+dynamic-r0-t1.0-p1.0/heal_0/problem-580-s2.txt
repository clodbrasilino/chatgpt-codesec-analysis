#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_TUPLES 100
#define MAX_ELEMENTS 10
#define MAX_STRING_LEN 256

typedef enum {
    TYPE_INT,
    TYPE_STRING,
    TYPE_TUPLE
} ElementType;

typedef struct Element Element;
typedef struct Tuple Tuple;

struct Element {
    ElementType type;
    union {
        int int_val;
        char *str_val;
        Tuple *tuple_val;
    } data;
};

struct Tuple {
    Element *elements;
    size_t count;
    size_t capacity;
};

typedef struct {
    int *values;
    size_t count;
    size_t capacity;
} IntArray;

Tuple *tuple_create(size_t capacity) {
    Tuple *t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }
    t->elements = malloc(capacity * sizeof(Element));
    if (t->elements == NULL) {
        free(t);
        return NULL;
    }
    t->count = 0;
    t->capacity = capacity;
    return t;
}

Element *element_create_int(int val) {
    Element *e = malloc(sizeof(Element));
    if (e == NULL) {
        return NULL;
    }
    e->type = TYPE_INT;
    e->data.int_val = val;
    return e;
}

Element *element_create_string(const char *str) {
    if (str == NULL) {
        return NULL;
    }
    Element *e = malloc(sizeof(Element));
    if (e == NULL) {
        return NULL;
    }
    e->type = TYPE_STRING;
    e->data.str_val = malloc(strlen(str) + 1);
    if (e->data.str_val == NULL) {
        free(e);
        return NULL;
    }
    strcpy(e->data.str_val, str);
    return e;
}

Element *element_create_tuple(Tuple *t) {
    if (t == NULL) {
        return NULL;
    }
    Element *e = malloc(sizeof(Element));
    if (e == NULL) {
        return NULL;
    }
    e->type = TYPE_TUPLE;
    e->data.tuple_val = t;
    return e;
}

int tuple_add_element(Tuple *t, Element *e) {
    if (t == NULL || e == NULL) {
        return -1;
    }
    if (t->count >= t->capacity) {
        return -1;
    }
    t->elements[t->count] = *e;
    t->count++;
    free(e);
    return 0;
}

void element_free(Element *e) {
    if (e == NULL) {
        return;
    }
    if (e->type == TYPE_STRING) {
        free(e->data.str_val);
    } else if (e->type == TYPE_TUPLE) {
        if (e->data.tuple_val != NULL) {
            for (size_t i = 0; i < e->data.tuple_val->count; i++) {
                element_free(&e->data.tuple_val->elements[i]);
            }
            free(e->data.tuple_val->elements);
            free(e->data.tuple_val);
        }
    }
}

void tuple_free(Tuple *t) {
    if (t == NULL) {
        return;
    }
    for (size_t i = 0; i < t->count; i++) {
        element_free(&t->elements[i]);
    }
    free(t->elements);
    free(t);
}

IntArray *int_array_create(size_t capacity) {
    IntArray *arr = malloc(sizeof(IntArray));
    if (arr == NULL) {
        return NULL;
    }
    arr->values = malloc(capacity * sizeof(int));
    if (arr->values == NULL) {
        free(arr);
        return NULL;
    }
    arr->count = 0;
    arr->capacity = capacity;
    return arr;
}

int int_array_add(IntArray *arr, int val) {
    if (arr == NULL) {
        return -1;
    }
    if (arr->count >= arr->capacity) {
        size_t new_capacity = arr->capacity * 2;
        int *new_values = realloc(arr->values, new_capacity * sizeof(int));
        if (new_values == NULL) {
            return -1;
        }
        arr->values = new_values;
        arr->capacity = new_capacity;
    }
    arr->values[arr->count] = val;
    arr->count++;
    return 0;
}

void int_array_free(IntArray *arr) {
    if (arr == NULL) {
        return;
    }
    free(arr->values);
    free(arr);
}

int extract_even_elements(Tuple *t, IntArray *result) {
    if (t == NULL || result == NULL) {
        return -1;
    }
    for (size_t i = 0; i < t->count; i++) {
        Element *e = &t->elements[i];
        if (e->type == TYPE_INT) {
            if (e->data.int_val % 2 == 0) {
                if (int_array_add(result, e->data.int_val) != 0) {
                    return -1;
                }
            }
        } else if (e->type == TYPE_TUPLE) {
            if (extract_even_elements(e->data.tuple_val, result) != 0) {
                return -1;
            }
        }
    }
    return 0;
}

void print_int_array(IntArray *arr) {
    if (arr == NULL) {
        return;
    }
    printf("Even elements: ");
    for (size_t i = 0; i < arr->count; i++) {
        printf("%d ", arr->values[i]);
    }
    printf("\n");
}

int main(void) {
    Tuple *outer = tuple_create(5);
    if (outer == NULL) {
        fprintf(stderr, "Failed to create outer tuple\n");
        return EXIT_FAILURE;
    }

    Tuple *inner1 = tuple_create(3);
    if (inner1 == NULL) {
        fprintf(stderr, "Failed to create inner tuple 1\n");
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    Tuple *inner2 = tuple_create(3);
    if (inner2 == NULL) {
        fprintf(stderr, "Failed to create inner tuple 2\n");
        tuple_free(inner1);
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    if (tuple_add_element(inner1, element_create_int(1)) != 0 ||
        tuple_add_element(inner1, element_create_int(2)) != 0 ||
        tuple_add_element(inner1, element_create_int(3)) != 0) {
        fprintf(stderr, "Failed to add elements to inner tuple 1\n");
        tuple_free(inner2);
        tuple_free(inner1);
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    if (tuple_add_element(inner2, element_create_int(4)) != 0 ||
        tuple_add_element(inner2, element_create_string("hello")) != 0 ||
        tuple_add_element(inner2, element_create_int(6)) != 0) {
        fprintf(stderr, "Failed to add elements to inner tuple 2\n");
        tuple_free(inner2);
        tuple_free(inner1);
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    if (tuple_add_element(outer, element_create_int(8)) != 0 ||
        tuple_add_element(outer, element_create_tuple(inner1)) != 0 ||
        tuple_add_element(outer, element_create_string("world")) != 0 ||
        tuple_add_element(outer, element_create_tuple(inner2)) != 0 ||
        tuple_add_element(outer, element_create_int(10)) != 0) {
        fprintf(stderr, "Failed to add elements to outer tuple\n");
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    IntArray *result = int_array_create(10);
    if (result == NULL) {
        fprintf(stderr, "Failed to create result array\n");
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    if (extract_even_elements(outer, result) != 0) {
        fprintf(stderr, "Failed to extract even elements\n");
        int_array_free(result);
        tuple_free(outer);
        return EXIT_FAILURE;
    }

    print_int_array(result);

    int_array_free(result);
    tuple_free(outer);

    return EXIT_SUCCESS;
}
