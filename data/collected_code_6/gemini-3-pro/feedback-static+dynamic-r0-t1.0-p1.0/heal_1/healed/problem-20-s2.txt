#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool isWoodall(int n) {
    if (n < 1) {
        return false;
    }
    
    n = n + 1;
    int x = 0;
    while (n % 2 == 0) {
        n /= 2;
        x++;
        if (x == n) {
            return true;
        }
    }
    
    return x == n;
}

int main() {
    int num;
    int result;

    result = scanf("%d", &num);
    if (result != 1) {
        return EXIT_FAILURE;
    }

    if (isWoodall(num)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }

    return EXIT_SUCCESS;
}
