#include <stdio.h>
#include <stdbool.h>

bool isMajorityElement(const int arr[], int size, int target) {
    if (arr == NULL || size <= 0) {
        return false;
    }
    
    int left = 0;
    int right = size - 1;
    int firstOccurrence = -1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) {
            firstOccurrence = mid;
            right = mid - 1;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    if (firstOccurrence == -1) {
        return false;
    }
    
    int lastIndex = firstOccurrence + size / 2;
    if (lastIndex < size && arr[lastIndex] == target) {
        return true;
    }
    
    return false;
}

int main(void) {
    int arr[] = {1, 2, 2, 2, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    int target = 2;
    
    if (isMajorityElement(arr, size, target)) {
        printf("%d is a majority element\n", target);
    } else {
        printf("%d is not a majority element\n", target);
    }
    
    return 0;
}
