```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

double find_third_side(double side1, double side2, int is_hypotenuse_known, int *error)
{
    double result;

    *error = 0;

    if (side1 <= 0.0 || side2 <= 0.0) {
        *error = 1;
        return 0.0;
    }

    if (is_hypotenuse_known) {
        if (side2 >= side1) {
            *error = 1;
            return 0.0;
        }
        result = sqrt((side1 * side1) - (side2 * side2));
    } else {
        result = sqrt((side1 * side1) + (side2 * side2));
    }

    if (!isfinite(result)) {
        *error = 1;
        return 0.0;
    }

    return result;
}

int main(void)
{
    double a;
    double b;
    double third;
    int choice;
    int error;

    printf("Enter 1 if hypotenuse is known, 0 if both legs are known: ");
    if (scanf("%d", &choice) != 1) {
        fprintf(stderr, "Invalid input for choice.\n");
        return EXIT_FAILURE;
    }

    if (choice != 0 && choice != 1) {
        fprintf(stderr, "Choice must be 0 or 1.\n");
        return EXIT_FAILURE;
    }

    if (choice == 1) {
        printf("Enter hypotenuse: ");
    } else {
        printf("Enter first leg: ");
    }
    if (scanf("%lf", &a) != 1) {
        fprintf(stderr, "Invalid input for first side.\n");
        return EXIT_FAILURE;
    }

    if (choice == 1) {
        printf("Enter known leg: ");
    } else {
        printf("Enter second leg: ");
    }
    if (scanf("%lf", &b) != 1) {
        fprintf(stderr, "Invalid input for second side.\n");
        return EXIT_FAILURE;
    }

    third = find_third_side(a, b, choice, &error);
    if (error != 0) {
        fprintf(stderr, "Invalid triangle sides provided.\n");
        return EXIT_FAILURE;
    }

    printf("The third side is: %f\n", third);

    return EXIT_SUCCESS;
}
```

This implementation validates all inputs, checks every `scanf` return value, guards against invalid triangle configurations (such as a leg longer than the hypotenuse or non-positive sides), and verifies the computed result is finite before use.
