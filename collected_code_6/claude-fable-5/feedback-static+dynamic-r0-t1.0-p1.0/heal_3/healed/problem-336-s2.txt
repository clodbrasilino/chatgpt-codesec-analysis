```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`char normalized[MAX_INPUT]` in `has_28_days` (CWE-119/CWE-120):** The array is protected by validating `len` with `strnlen(month, MAX_INPUT)` and rejecting anything where `len >= MAX_INPUT`. This guarantees `normalized[len] = '\0'` writes at most to index `MAX_INPUT - 1`, so no overflow is possible.

2. **`getchar()` in a loop (CWE-120/CWE-20):** The `flush_stdin` function never stores the characters it reads — it only discards them until newline or EOF. Since nothing is written to a buffer, no boundary issue exists. The EOF check also prevents an infinite loop if the stream closes.

3. **`char input[MAX_INPUT]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits the read to the buffer size including the terminator. The explicit `input[sizeof(input) - 1] = '\0'` is defensive redundancy, and `strnlen` bounds the length scan. Oversized input lines are drained via `flush_stdin()` so leftover bytes cannot pollute subsequent reads.

The flawfinder hits are informational (statically-sized buffers inherently trigger risk-level warnings); the mitigations — bounded reads (`fgets`, `strnlen`), length validation before writes, and character-class validation with `unsigned char` casts before `isalpha`/`tolower` (avoiding UB on negative `char` values) — are already correctly in place, so no functional changes were required.
