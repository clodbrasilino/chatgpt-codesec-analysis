#include <stdio.h>

void move_zeroes_to_end(int arr[], int n) {
    int write_index = 0;

    for (int read_index = 0; read_index < n; read_index++) {
        if (arr[read_index] != 0) {
            arr[write_index] = arr[read_index];
            write_index++;
        }
    }

    while (write_index < n) {
        arr[write_index] = 0;
        write_index++;
    }
}

int main(void) {
    int arr[] = {0, 1, 0, 3, 12, 0, 5};
    int n = sizeof(arr) / sizeof(arr[0]);

    move_zeroes_to_end(arr, n);

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
