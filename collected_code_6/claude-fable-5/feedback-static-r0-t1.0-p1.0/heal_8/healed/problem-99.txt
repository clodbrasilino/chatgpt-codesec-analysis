```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

static int decimal_to_binary(unsigned int number, char *buffer, size_t buffer_size)
{
    size_t index = 0U;
    size_t start;
    size_t end;

    if (buffer == NULL || buffer_size < 2U) {
        return -1;
    }

    if (number == 0U) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (number > 0U && index < (buffer_size - 1U)) {
        buffer[index] = (char)('0' + (number % 2U));
        number /= 2U;
        index++;
    }

    if (number > 0U) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[index] = '\0';

    start = 0U;
    end = index - 1U;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    unsigned int value = 0U;
    char binary[MAX_BITS + 1U] = {0};
    int scan_result;

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &value);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(value, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary equivalent: %s\n", binary);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and why the code is safe:

**CWE-119/CWE-120 (statically-sized array `binary`):** The Flawfinder hit is a low-confidence heuristic triggered by any fixed-size `char` array. In this code the risk is fully mitigated:

1. **Correct sizing:** `binary` is sized `MAX_BITS + 1U`, i.e., `sizeof(unsigned int) * CHAR_BIT + 1`, which is exactly enough for the maximum possible binary representation of an `unsigned int` plus the null terminator, on any conforming platform.

2. **Size propagation:** The buffer size is passed via `sizeof(binary)` rather than a hardcoded constant, so the callee always knows the true capacity.

3. **Bounds checking in the callee:** The write loop is guarded by `index < (buffer_size - 1U)`, guaranteeing space for the terminator. If the number cannot fit (impossible here, but defensively handled), the function returns `-1` instead of overflowing. I added `buffer[0] = '\0'` on that error path so the buffer is never left non-terminated.

4. **Null/size validation:** `buffer == NULL || buffer_size < 2U` rejects degenerate inputs before any write occurs.

5. **Input handling:** `scanf("%u", ...)` writes to a fixed-size scalar, not a string, so no `%s`-style overflow is possible; its return value is checked.

The warning is therefore a false positive, and the rewritten code retains all defensive measures.
