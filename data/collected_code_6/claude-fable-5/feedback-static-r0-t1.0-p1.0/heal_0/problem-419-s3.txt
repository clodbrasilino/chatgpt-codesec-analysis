#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long long round_and_sum(const double *numbers, size_t length)
{
    long long sum = 0;
    size_t i;

    if (numbers == NULL || length == 0) {
        return 0;
    }

    for (i = 0; i < length; i++) {
        sum += (long long)llround(numbers[i]);
    }

    return sum * (long long)length;
}

int main(void)
{
    double numbers[] = { 1.4, 2.6, 3.5, -4.2, 5.9 };
    size_t length = sizeof(numbers) / sizeof(numbers[0]);
    long long result;

    result = round_and_sum(numbers, length);

    if (printf("%lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
