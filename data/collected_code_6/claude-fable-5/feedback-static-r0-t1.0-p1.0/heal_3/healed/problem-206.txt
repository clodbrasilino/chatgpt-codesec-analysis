#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096U

static void cleanup_partial(char **result, size_t filled)
{
    for (size_t j = 0U; j < filled; j++) {
        free(result[j]);
    }
    free(result);
}

char **concatenate_adjacent(const char **tuple, size_t count, size_t *result_count)
{
    if (tuple == NULL || result_count == NULL || count < 2U) {
        if (result_count != NULL) {
            *result_count = 0U;
        }
        return NULL;
    }

    size_t out_count = count - 1U;
    char **result = calloc(out_count, sizeof(char *));
    if (result == NULL) {
        *result_count = 0U;
        return NULL;
    }

    for (size_t i = 0U; i < out_count; i++) {
        if (tuple[i] == NULL || tuple[i + 1U] == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t len1 = strnlen(tuple[i], MAX_STRING_LENGTH);
        size_t len2 = strnlen(tuple[i + 1U], MAX_STRING_LENGTH);

        if (len1 >= MAX_STRING_LENGTH || len2 >= MAX_STRING_LENGTH) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > SIZE_MAX - len2 - 1U) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t total = len1 + len2 + 1U;
        char *combined = malloc(total);
        if (combined == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > total - 1U || len2 > total - 1U - len1) {
            free(combined);
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        memcpy(combined, tuple[i], len1);
        memcpy(combined + len1, tuple[i + 1U], len2);
        combined[len1 + len2] = '\0';
        result[i] = combined;
    }

    *result_count = out_count;
    return result;
}

void free_result(char **result, size_t count)
{
    if (result == NULL) {
        return;
    }
    for (size_t i = 0U; i < count; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void)
{
    const char *tuple[] = { "DSP", "IS", "BEST", "FOR", "ALL", "UTS" };
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    size_t result_count = 0U;

    char **result = concatenate_adjacent(tuple, count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Concatenation failed\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (size_t i = 0U; i < result_count; i++) {
        printf("'%s'", result[i]);
        if (i + 1U < result_count) {
            printf(", ");
        }
    }
    printf(")\n");

    free_result(result, result_count);
    return EXIT_SUCCESS;
}

Analysis of the original code and the changes made:

**The flagged `memcpy` warnings were false positives.** Flawfinder flags every `memcpy` call because it cannot statically verify destination capacity (CWE-120). In the original code, the destination buffer `combined` was allocated with exactly `len1 + len2 + 1` bytes, and the copies write precisely `len1` and `len2` bytes respectively — so no overflow was actually possible.

**What was already done correctly:**
- `strnlen` with `MAX_STRING_LENGTH` bounds the length computation, protecting against unterminated strings.
- The `len1 >= MAX_STRING_LENGTH` check rejects strings at or beyond the cap, guaranteeing the measured lengths reflect real, terminated string lengths.
- The `SIZE_MAX` overflow guard prevents integer wraparound in `len1 + len2 + 1` (technically unreachable given the 4096 cap, but sound defensive practice).
- All allocation failures and `NULL` inputs are handled with full cleanup of partial results — no leaks on any error path.

**Change made to satisfy the analyzer:** An explicit runtime guard was added immediately before the copies, verifying `len1 <= total - 1` and `len2 <= total - 1 - len1`. This makes the capacity invariant locally checkable rather than relying on the reader (or tool) tracing the allocation arithmetic, addressing the CWE-120 concern directly at the copy site.

The program's behavior is unchanged: for the input tuple it prints `('DSPIS', 'ISBEST', 'BESTFOR', 'FORALL', 'ALLUTS')`.
