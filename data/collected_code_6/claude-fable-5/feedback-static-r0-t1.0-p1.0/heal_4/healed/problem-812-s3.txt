```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

static char *abbreviate_road(const char *input)
{
    static const char pattern[] = "road";
    static const char replacement[] = "rd.";
    const size_t pattern_len = sizeof(pattern) - 1;
    const size_t replacement_len = sizeof(replacement) - 1;
    size_t input_len;
    size_t count;
    size_t result_len;
    size_t remaining;
    const char *p;
    char *result;
    char *dest;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    count = 0;
    p = input;
    while ((p = strstr(p, pattern)) != NULL) {
        count++;
        p += pattern_len;
    }

    if (count > input_len / pattern_len) {
        return NULL;
    }

    if (pattern_len < replacement_len) {
        return NULL;
    }

    result_len = input_len - count * (pattern_len - replacement_len);

    if (result_len >= SIZE_MAX) {
        return NULL;
    }

    result = malloc(result_len + 1);
    if (result == NULL) {
        return NULL;
    }

    dest = result;
    remaining = result_len;
    p = input;
    while (*p != '\0') {
        if (strncmp(p, pattern, pattern_len) == 0) {
            if (remaining < replacement_len) {
                free(result);
                return NULL;
            }
            memcpy(dest, replacement, replacement_len);
            dest += replacement_len;
            remaining -= replacement_len;
            p += pattern_len;
        } else {
            if (remaining < 1) {
                free(result);
                return NULL;
            }
            *dest = *p;
            dest++;
            remaining--;
            p++;
        }
    }
    *dest = '\0';

    return result;
}

int main(void)
{
    const char *text = "Take the main road until the road forks near Elm road.";
    char *abbreviated;

    abbreviated = abbreviate_road(text);
    if (abbreviated == NULL) {
        fprintf(stderr, "Failed to abbreviate string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original:    %s\n", text) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    if (printf("Abbreviated: %s\n", abbreviated) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    free(abbreviated);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and how the code handles it:

**Flawfinder `memcpy` warning (CWE-120):** The warning is a heuristic hit on any `memcpy` call. In this code the risk is already mitigated because:

1. **Exact destination sizing:** `result_len` is computed precisely from the input length and the number of pattern occurrences, so the buffer is allocated to the exact required size plus the NUL terminator.
2. **Runtime bounds tracking:** The `remaining` counter tracks free space in the destination, and the check `if (remaining < replacement_len)` before the `memcpy` guarantees the copy can never overflow, even if the earlier length arithmetic were somehow wrong. The single-character copy path has the analogous `remaining < 1` check.
3. **Integer overflow protection:** The `count > input_len / pattern_len` check prevents `count * (pattern_len - replacement_len)` from underflowing `result_len`, and the rewrite adds an explicit `pattern_len < replacement_len` guard so the subtraction in the size calculation can never wrap even if the pattern/replacement constants are later changed to a growing replacement.
4. **Input length capping:** `strnlen` with `MAX_INPUT_LEN` bounds the work done on untrusted input and rejects unterminated or oversized strings.

The remaining `memcpy` is safe by construction; the checks surrounding it make the Flawfinder finding a false positive.
