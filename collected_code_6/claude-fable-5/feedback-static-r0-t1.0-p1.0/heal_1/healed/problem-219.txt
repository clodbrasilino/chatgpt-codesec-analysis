```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int extract_min_max_k(const int *arr, size_t n, size_t k,
                      int *min_out, int *max_out)
{
    int *copy;
    size_t i;
    size_t bytes;

    if (arr == NULL || min_out == NULL || max_out == NULL) {
        return -1;
    }
    if (n == 0U || k == 0U || k > n) {
        return -1;
    }
    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    bytes = n * sizeof(int);

    copy = malloc(bytes);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, bytes);
    qsort(copy, n, sizeof(int), compare_ints);

    for (i = 0U; i < k; i++) {
        min_out[i] = copy[i];
        max_out[i] = copy[n - k + i];
    }

    free(copy);
    return 0;
}

int main(void)
{
    int data[] = { 5, 20, 3, 7, 6, 8, 100, -2, 45, 9 };
    size_t n = sizeof(data) / sizeof(data[0]);
    size_t k = 3U;
    int *min_vals;
    int *max_vals;
    size_t i;
    int status;

    if (k == 0U || k > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Invalid k\n");
        return EXIT_FAILURE;
    }

    min_vals = malloc(k * sizeof(int));
    if (min_vals == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    max_vals = malloc(k * sizeof(int));
    if (max_vals == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(min_vals);
        return EXIT_FAILURE;
    }

    status = extract_min_max_k(data, n, k, min_vals, max_vals);
    if (status != 0) {
        fprintf(stderr, "Extraction failed\n");
        free(min_vals);
        free(max_vals);
        return EXIT_FAILURE;
    }

    printf("Minimum %zu elements: ", k);
    for (i = 0U; i < k; i++) {
        printf("%d ", min_vals[i]);
    }
    printf("\n");

    printf("Maximum %zu elements: ", k);
    for (i = 0U; i < k; i++) {
        printf("%d ", max_vals[i]);
    }
    printf("\n");

    free(min_vals);
    free(max_vals);
    return EXIT_SUCCESS;
}
```

Key changes from the original:

1. **Added `#include <stdint.h>`** — the original code used `SIZE_MAX` without including the header that defines it, which would fail to compile (or rely on transitive includes) on a strictly conforming implementation.

2. **Addressed the Flawfinder `memcpy` finding (CWE-120)** — the copy size is computed once into `bytes` after the overflow guard (`n > SIZE_MAX / sizeof(int)`), and the exact same value is used for both `malloc` and `memcpy`. This guarantees the destination buffer is always exactly as large as the amount of data copied, eliminating any possibility of the allocation size and copy size drifting apart.

3. **Added defensive validation of `k` in `main`** before computing `k * sizeof(int)` for the output buffers, preventing a potential size overflow in the allocation and ensuring the caller upholds the contract that `min_vals`/`max_vals` can hold `k` integers — which is the real precondition backing the safety of the writes inside `extract_min_max_k`.

The rest of the logic (null checks, `k <= n` validation, error propagation, and memory cleanup on all exit paths) was already sound and has been preserved.
