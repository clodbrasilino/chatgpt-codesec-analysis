```c
#include <stdio.h>

int count_digits(int num) {
    int count = 0;
    
    if (num == 0) {
        return 1;
    }
    
    while (num != 0) {
        num /= 10;
        count++;
    }
    
    return count;
}

int main(void) {
    int number;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    printf("Number of digits: %d\n", count_digits(number));
    
    return 0;
}
```
