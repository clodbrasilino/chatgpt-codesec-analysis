#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

static int safe_multiply(int a, int b, long long *result) {
    long long temp = (long long)a * b;
    if (temp > LLONG_MAX || temp < LLONG_MIN) {
        return 0;
    }
    *result = temp;
    return 1;
}

static int safe_add(long long a, long long b, long long *result) {
    if ((b > 0 && a > LLONG_MAX - b) || (b < 0 && a < LLONG_MIN - b)) {
        return 0;
    }
    *result = a + b;
    return 1;
}

static int int_pow5(int base, long long *result) {
    long long temp = base;
    long long final_result = 1;
    long long mult_result;
    int i;

    for (i = 0; i < 5; i++) {
        if (!safe_multiply((int)final_result, (int)temp, &mult_result)) {
            return 0;
        }
        final_result = mult_result;
    }

    *result = final_result;
    return 1;
}

static long long sum_of_fifth_powers(int n, int *error) {
    long long total = 0;
    long long power_result;
    long long add_result;
    int i;

    if (n < 0) {
        *error = EINVAL;
        return 0;
    }

    for (i = 1; i <= n; i++) {
        if (!int_pow5(i, &power_result)) {
            *error = EOVERFLOW;
            return 0;
        }

        if (!safe_add(total, power_result, &add_result)) {
            *error = EOVERFLOW;
            return 0;
        }
        total = add_result;
    }

    *error = 0;
    return total;
}

int main(void) {
    int n;
    long long result;
    int error;
    char input_buffer[64];
    char *endptr;

    printf("Enter a non-negative integer n: ");
    if (fgets(input_buffer, sizeof(input_buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    n = (int)strtol(input_buffer, &endptr, 10);

    if (errno == ERANGE || endptr == input_buffer || *endptr != '\n') {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0) {
        fprintf(stderr, "n must be non-negative\n");
        return EXIT_FAILURE;
    }

    result = sum_of_fifth_powers(n, &error);

    if (error != 0) {
        fprintf(stderr, "Overflow occurred while calculating the sum\n");
        return EXIT_FAILURE;
    }

    printf("Sum of fifth powers from 1 to %d is: %lld\n", n, result);
    return EXIT_SUCCESS;
}
