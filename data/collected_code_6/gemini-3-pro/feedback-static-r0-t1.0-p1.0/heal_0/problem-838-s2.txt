#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int min_swaps(const char *s1, const char *s2) {
    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    size_t len1 = strlen(s1);
    size_t len2 = strlen(s2);

    if (len1 != len2) {
        return -1;
    }

    int mismatches_01 = 0;
    int mismatches_10 = 0;

    for (size_t i = 0; i < len1; i++) {
        if (s1[i] != '0' && s1[i] != '1') {
            return -1;
        }
        if (s2[i] != '0' && s2[i] != '1') {
            return -1;
        }

        if (s1[i] == '0' && s2[i] == '1') {
            mismatches_01++;
        } else if (s1[i] == '1' && s2[i] == '0') {
            mismatches_10++;
        }
    }

    if ((mismatches_01 + mismatches_10) % 2 != 0) {
        return -1;
    }

    int swaps = (mismatches_01 / 2) + (mismatches_10 / 2);
    
    if (mismatches_01 % 2 != 0) {
        swaps += 2;
    }

    return swaps;
}

int main(void) {
    const char *s1 = "0100101";
    const char *s2 = "0110001";

    int result = min_swaps(s1, s2);
    
    if (result != -1) {
        printf("%d\n", result);
    } else {
        printf("Error: Invalid or incompatible strings.\n");
    }

    return 0;
}
