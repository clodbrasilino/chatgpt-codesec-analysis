#include <stdio.h>
#include <stdlib.h>

int check_common_elements_order(const int *list1, size_t size1, const int *list2, size_t size2) {
    if (list1 == NULL || list2 == NULL) {
        return -1;
    }

    size_t i = 0;
    size_t j = 0;

    while (i < size1 && j < size2) {
        if (list1[i] == list2[j]) {
            i++;
            j++;
        } else {
            i++;
        }
    }

    if (j == size2) {
        return 1;
    }

    return 0;
}

int main(void) {
    int list1[] = {1, 2, 3, 4, 5};
    int list2[] = {2, 4, 5};
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    size_t size2 = sizeof(list2) / sizeof(list2[0]);

    int result = check_common_elements_order(list1, size1, list2, size2);

    if (result == 1) {
        printf("Common elements are in the same order.\n");
    } else if (result == 0) {
        printf("Common elements are not in the same order.\n");
    } else {
        printf("Invalid input.\n");
    }

    return 0;
}
