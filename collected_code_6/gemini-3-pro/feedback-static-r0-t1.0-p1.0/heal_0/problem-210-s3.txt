```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>
#include <string.h>

int check_string(const char *str) {
    if (str == NULL) {
        return 0;
    }

    regex_t regex;
    int reti;
    int result = 0;

    reti = regcomp(&regex, "^[a-zA-Z0-9]+$", REG_EXTENDED);
    if (reti != 0) {
        return 0;
    }

    reti = regexec(&regex, str, 0, NULL, 0);
    if (!reti) {
        result = 1;
    } else if (reti == REG_NOMATCH) {
        result = 0;
    } else {
        result = 0;
    }

    regfree(&regex);
    return result;
}

int main(void) {
    const char *test1 = "ValidString123";
    const char *test2 = "Invalid String!";

    if (check_string(test1)) {
        printf("'%s' is valid.\n", test1);
    } else {
        printf("'%s' is invalid.\n", test1);
    }

    if (check_string(test2)) {
        printf("'%s' is valid.\n", test2);
    } else {
        printf("'%s' is invalid.\n", test2);
    }

    return 0;
}
```
