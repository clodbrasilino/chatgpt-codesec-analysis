```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parenthesis_area(const char *str)
{
    char *result;
    size_t i;
    size_t j;
    size_t len;
    int depth;

    if (str == NULL) {
        return NULL;
    }

    len = strlen(str);
    result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (str[i] == '(') {
            depth++;
        } else if (str[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = str[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(int argc, char *argv[])
{
    char input[256];
    const char *source;
    char *cleaned;

    if (argc > 1) {
        source = argv[1];
    } else {
        if (fgets(input, sizeof(input), stdin) == NULL) {
            fprintf(stderr, "Error reading input\n");
            return EXIT_FAILURE;
        }
        input[strcspn(input, "\n")] = '\0';
        source = input;
    }

    cleaned = remove_parenthesis_area(source);
    if (cleaned == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", cleaned);

    free(cleaned);
    cleaned = NULL;

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **Input source flexibility** — `main` now accepts the string either as a command-line argument (`argv[1]`) or from `stdin`. The original version blocked on `fgets()` when the test harness supplied input via arguments, which produced `<no output>` in all three failing test cases.

2. **Clean output format** — removed the `"Enter a string: "` prompt and the `"Result: "` prefix. The program now prints only the processed string (e.g., `python`, `string`, `alpha`), matching the expected test output exactly.

3. **Preserved robustness** — NULL checks, balanced/unbalanced parenthesis handling (unmatched `)` ignored, unmatched `(` still consumes to end), correct allocation of `len + 1`, explicit null-termination, and proper memory cleanup are all retained.
