```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    size_t errbuf_size;
    char *errbuf;

    errbuf_size = regerror(errcode, regex, NULL, 0);

    errbuf = (char *)malloc(errbuf_size);
    if (errbuf == NULL)
    {
        fprintf(stderr, "%s: error code %d\n", prefix, errcode);
        return;
    }

    regerror(errcode, regex, errbuf, errbuf_size);
    fprintf(stderr, "%s: %s\n", prefix, errbuf);

    free(errbuf);
}

int find_sequences(const char *str)
{
    regex_t regex;
    regmatch_t match;
    size_t offset = 0;
    size_t len;
    int ret;
    int count = 0;

    if (str == NULL)
    {
        fprintf(stderr, "Invalid input: string is NULL\n");
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LENGTH);
    if (len >= MAX_INPUT_LENGTH)
    {
        fprintf(stderr, "Invalid input: string is not null-terminated within the allowed limit\n");
        return -1;
    }

    ret = regcomp(&regex, "[A-Z][a-z]+", REG_EXTENDED);
    if (ret != 0)
    {
        print_regex_error(ret, &regex, "Could not compile regex");
        return -1;
    }

    while (offset <= len)
    {
        ret = regexec(&regex, str + offset, 1, &match, 0);

        if (ret == REG_NOMATCH)
        {
            break;
        }
        else if (ret != 0)
        {
            print_regex_error(ret, &regex, "Regex execution failed");
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so)
        {
            break;
        }

        printf("Found sequence: %.*s at position %lld\n",
               (int)(match.rm_eo - match.rm_so),
               str + offset + match.rm_so,
               (long long)(offset + (size_t)match.rm_so));

        count++;
        offset += (size_t)match.rm_eo;
    }

    regfree(&regex);
    return count;
}

int main(void)
{
    const char *input = "Hello World this Is a Test String With some Sequences Like Abc and Xyz";
    int result;

    printf("Input string: %s\n", input);

    result = find_sequences(input);

    if (result < 0)
    {
        fprintf(stderr, "Failed to find sequences\n");
        return EXIT_FAILURE;
    }

    printf("Total sequences found: %d\n", result);

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **CWE-119/CWE-120 (static `char errbuf[128]`)**: Replaced the fixed-size stack buffer with a dynamically allocated buffer. `regerror()` is first called with `NULL`/`0` to query the exact size needed, then the buffer is allocated with `malloc()` and freed after use, eliminating any risk of an improperly restricted static buffer.

2. **CWE-126 (`strlen` over-read)**: Replaced `strlen(str)` with `strnlen(str, MAX_INPUT_LENGTH)`. If no null terminator is found within the bounded limit, the input is rejected, preventing an unbounded over-read on non-terminated strings.
