```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int count;
} Tuple;

int has_k_elements(Tuple t, int k) {
    return t.count == k;
}

int main() {
    Tuple t = {4};
    int k = 4;
    if (has_k_elements(t, k)) {
        printf("Tuple has all %d elements.\n", k);
    } else {
        printf("Tuple does not have all %d elements.\n", k);
    }
    return 0;
}
```
