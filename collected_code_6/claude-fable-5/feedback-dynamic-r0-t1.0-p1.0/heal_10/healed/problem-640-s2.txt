```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *tests[] = {
        "py(remove)thon",
        "str(nested (inner))ing",
        "(leading)alpha"
    };
    const char *expected[] = {
        "python",
        "string",
        "alpha"
    };
    size_t num_tests;
    size_t i;
    char *output;
    int status;

    num_tests = sizeof(tests) / sizeof(tests[0]);
    status = EXIT_SUCCESS;

    for (i = 0; i < num_tests; i++) {
        output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (strcmp(output, expected[i]) == 0) {
            printf("test case %zu passed: '%s'\n", i, output);
        } else {
            fprintf(stderr, "test case %zu failed: expected '%s', got '%s'\n",
                    i, expected[i], output);
            status = EXIT_FAILURE;
        }
        free(output);
        output = NULL;
    }

    return status;
}
```

**Analysis notes:**

The symptoms described in the original comments (garbage output like `` `S: ``, `�R:`) are classic signs of reading uninitialized or freed heap memory. The likely weaknesses in a broken variant of this code would have been:

1. **Missing null termination** — using `malloc(len)` instead of `calloc(len + 1, ...)` and never writing `result[j] = '\0'`, causing `strcmp`/`printf` to read past the intended string into uninitialized heap bytes.
2. **Off-by-one allocation** — allocating `len` bytes instead of `len + 1`, leaving no room for the terminator (heap buffer overflow when writing `'\0'`).
3. **Use-after-free** — printing `output` after `free(output)` would also produce garbage from the allocator's freed-block metadata.

The version above defends against all three: `calloc(len + 1, sizeof(char))` guarantees a zero-initialized buffer with room for the terminator, `result[j] = '\0'` explicitly terminates, `free` happens only after the string is used, and the pointer is nulled after freeing to prevent accidental reuse. `NULL` input and allocation failure are both handled, and the caller checks the return value before dereferencing.
