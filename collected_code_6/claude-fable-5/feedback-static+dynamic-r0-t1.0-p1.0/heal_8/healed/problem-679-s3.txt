```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dict;

static char *safe_strdup(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(src, max_len);
    if (len >= max_len) {
        return NULL;
    }

    if (len >= SIZE_MAX) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static Dict *dict_create(size_t capacity)
{
    Dict *d;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(DictEntry)) {
        return NULL;
    }

    d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }

    d->entries = malloc(capacity * sizeof(DictEntry));
    if (d->entries == NULL) {
        free(d);
        return NULL;
    }

    d->size = 0;
    d->capacity = capacity;
    return d;
}

static int dict_add(Dict *d, const char *key, const char *value)
{
    char *key_copy;
    char *value_copy;

    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }

    if (d->size >= d->capacity) {
        return -1;
    }

    key_copy = safe_strdup(key, MAX_STR_LEN);
    if (key_copy == NULL) {
        return -1;
    }

    value_copy = safe_strdup(value, MAX_STR_LEN);
    if (value_copy == NULL) {
        free(key_copy);
        return -1;
    }

    d->entries[d->size].key = key_copy;
    d->entries[d->size].value = value_copy;
    d->size++;
    return 0;
}

static const DictEntry *dict_get_by_index(const Dict *d, size_t index)
{
    if (d == NULL || index >= d->size) {
        return NULL;
    }
    return &d->entries[index];
}

static void dict_destroy(Dict *d)
{
    size_t i;

    if (d == NULL) {
        return;
    }

    for (i = 0; i < d->size; i++) {
        free(d->entries[i].key);
        free(d->entries[i].value);
    }
    free(d->entries);
    free(d);
}

int main(void)
{
    Dict *d;
    const DictEntry *entry;
    size_t i;

    d = dict_create(8);
    if (d == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }

    if (dict_add(d, "apple", "fruit") != 0 ||
        dict_add(d, "carrot", "vegetable") != 0 ||
        dict_add(d, "salmon", "fish") != 0) {
        fprintf(stderr, "Failed to add entries\n");
        dict_destroy(d);
        return EXIT_FAILURE;
    }

    for (i = 0; i < d->size; i++) {
        entry = dict_get_by_index(d, i);
        if (entry != NULL) {
            printf("Index %zu: key=%s, value=%s\n", i, entry->key, entry->value);
        }
    }

    entry = dict_get_by_index(d, 100);
    if (entry == NULL) {
        printf("Index 100 is out of range\n");
    }

    dict_destroy(d);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) is a false positive in this code. The original implementation was already safe, and the rewrite preserves that safety for the following reasons:

1. **Destination size is guaranteed sufficient**: The buffer is allocated with `malloc(len + 1)` immediately before the copy, and exactly `len` bytes are copied, leaving one byte for the null terminator. The destination can always hold the source data by construction.

2. **Bounded length measurement**: `strnlen(src, max_len)` is used instead of `strlen`, so the copy length can never exceed `max_len - 1` (4096-byte cap), even if `src` is not null-terminated within that range — the `len >= max_len` check rejects such input.

3. **Integer overflow protection**: The `len >= SIZE_MAX` check ensures `len + 1` cannot wrap around to zero in the `malloc` call. (I changed `==` to `>=` for defensive clarity, though it's unreachable given the `max_len` check.)

4. **Other robustness already present**: `dict_create` guards against multiplication overflow (`capacity > SIZE_MAX / sizeof(DictEntry)`), all allocations are checked, cleanup paths avoid leaks on partial failure, and `dict_get_by_index` bounds-checks the index.

No functional changes were required — the code follows the correct "allocate exact size, then copy exact size" pattern that static analyzers cannot always verify automatically.
