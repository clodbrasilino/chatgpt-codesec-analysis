```c
#include <stdio.h>
#include <math.h>

typedef struct {
    double real;
    double imag;
} Complex;

typedef struct {
    double magnitude;
    double angle;
} Polar;

Polar complex_to_polar(Complex c) {
    Polar p;
    p.magnitude = sqrt(c.real * c.real + c.imag * c.imag);
    p.angle = atan2(c.imag, c.real);
    return p;
}

int main(void) {
    Complex c;
    Polar p;

    c.real = 3.0;
    c.imag = 4.0;

    p = complex_to_polar(c);

    printf("Complex: %.2f + %.2fi\n", c.real, c.imag);
    printf("Polar: Magnitude = %.2f, Angle = %.2f radians\n", p.magnitude, p.angle);

    return 0;
}
```
