#include <stdio.h>
#include <limits.h>

int findSmallestValueFrequency(const int arr[], int n) {
    int min = INT_MAX;
    int frequency = 0;

    for (int i = 0; i < n; ++i) {
        if (arr[i] < min) {
            min = arr[i];
            frequency = 1;
        } else if (arr[i] == min) {
            ++frequency;
        }
    }

    return frequency;
}

int main() {
    const int arr[] = {3, 1, 2, 1, 4, 1, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int frequency = findSmallestValueFrequency(arr, n);
    printf("Frequency of the smallest value: %d\n", frequency);
    return 0;
}
