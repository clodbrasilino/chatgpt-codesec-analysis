#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

static char *copy_string(const char *src)
{
    size_t len;
    char *dest;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LENGTH);
    dest = malloc(len + 1);
    if (dest == NULL) {
        return NULL;
    }

    memcpy(dest, src, len);
    dest[len] = '\0';
    return dest;
}

char *second_most_frequent(const char *sequence[], size_t count)
{
    const char **unique;
    size_t *frequencies;
    size_t unique_count = 0;
    size_t i;
    size_t j;
    char *result = NULL;

    if (sequence == NULL || count == 0) {
        return NULL;
    }

    unique = malloc(count * sizeof(*unique));
    if (unique == NULL) {
        return NULL;
    }

    frequencies = calloc(count, sizeof(*frequencies));
    if (frequencies == NULL) {
        free(unique);
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;

        if (sequence[i] == NULL) {
            continue;
        }

        for (j = 0; j < unique_count; j++) {
            if (strcmp(unique[j], sequence[i]) == 0) {
                frequencies[j]++;
                found = 1;
                break;
            }
        }

        if (!found) {
            unique[unique_count] = sequence[i];
            frequencies[unique_count] = 1;
            unique_count++;
        }
    }

    if (unique_count >= 2) {
        size_t first_idx = 0;
        size_t second_idx = 0;
        int have_first = 0;
        int have_second = 0;

        for (i = 0; i < unique_count; i++) {
            if (!have_first || frequencies[i] > frequencies[first_idx]) {
                if (have_first) {
                    second_idx = first_idx;
                    have_second = 1;
                }
                first_idx = i;
                have_first = 1;
            } else if (frequencies[i] < frequencies[first_idx] &&
                       (!have_second || frequencies[i] > frequencies[second_idx])) {
                second_idx = i;
                have_second = 1;
            }
        }

        if (have_second) {
            result = copy_string(unique[second_idx]);
        }
    }

    free(unique);
    free(frequencies);

    return result;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "cherry",
        "banana", "apple", "banana", "apple",
        "cherry"
    };
    size_t count = sizeof(sequence) / sizeof(sequence[0]);
    char *second = second_most_frequent(sequence, count);

    if (second != NULL) {
        printf("Second most frequent string: %s\n", second);
        free(second);
    } else {
        printf("No second most frequent string exists.\n");
    }

    return 0;
}
