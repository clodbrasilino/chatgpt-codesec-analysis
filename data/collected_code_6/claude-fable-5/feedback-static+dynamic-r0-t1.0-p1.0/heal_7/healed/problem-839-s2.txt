```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(t->first));
    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, sizeof(ta->first));
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

1. **Statically-sized array `first[MAX_NAME_LEN]` (CWE-119/CWE-120):** The original code already mitigated this correctly — `set_tuple()` uses `strnlen()` bounded by the buffer size and rejects any name that would not fit including the null terminator. In the rewrite I tightened this by bounding `strnlen()` directly with `sizeof(t->first)` rather than the macro, so the check stays correct even if the struct definition changes independently of the macro.

2. **`memcpy` without bounds check (CWE-120):** This is a false positive in practice. The copy is guarded: `len < sizeof(t->first)` is guaranteed before the `memcpy`, so the destination can always hold `len` bytes plus the explicit null terminator. The guard is preserved in the rewrite.

Additional defensive measures already present and retained: `strncmp` with a length bound instead of `strcmp`, bounded `%.*s` printing so output never depends on termination beyond the buffer, NULL checks on all pointer parameters, and checked return values for every fallible call.
