```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
```

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning (CWE-119/CWE-120) about the statically-sized `char buffer[MAX_INPUT_LEN]` is a low-risk, largely informational finding. The original code already mitigates it properly:

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits reads to the buffer size, preventing overflow (unlike `gets` or unbounded `scanf("%s", ...)`).
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant since `fgets` always null-terminates on success, but harmless and explicit).
3. **Newline stripping:** `strcspn(buffer, "\n")` safely removes the trailing newline without risk of out-of-bounds access.
4. **Bounded string traversal:** `strnlen(str, max_len)` in `count_chars_with_vowel_neighbors` guards against non-terminated input, and the `NULL` check prevents dereferencing a null pointer.
5. **Correct `ctype` usage:** casting to `unsigned char` before `tolower` avoids undefined behavior with negative `char` values.

The only substantive change made is adding `#define _POSIX_C_SOURCE 200809L`, since `strnlen` is a POSIX function (not standard C89/C99) and may not be declared by `<string.h>` in strict compilation modes — an implicit declaration would itself be a latent defect. The rest of the code is retained as it already follows secure coding practice.
