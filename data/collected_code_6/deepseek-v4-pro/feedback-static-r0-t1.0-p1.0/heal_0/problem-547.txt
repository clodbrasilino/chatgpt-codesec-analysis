#include <stdio.h>
#include <stdlib.h>

int hamming_distance(int a, int b) {
    int xor_result = a ^ b;
    int distance = 0;
    while (xor_result != 0) {
        distance += xor_result & 1;
        xor_result >>= 1;
    }
    return distance;
}

int sum_hamming_distances(int n) {
    if (n < 0) {
        return -1;
    }
    
    int total = 0;
    for (int i = 0; i < n; i++) {
        total += hamming_distance(i, i + 1);
    }
    return total;
}

int main(void) {
    int n;
    
    printf("Enter a non-negative integer n: ");
    if (scanf("%d", &n) != 1) {
        printf("Invalid input. Please enter an integer.\n");
        return EXIT_FAILURE;
    }
    
    if (n < 0) {
        printf("Error: n must be non-negative.\n");
        return EXIT_FAILURE;
    }
    
    int result = sum_hamming_distances(n);
    printf("Sum of Hamming distances of consecutive numbers from 0 to %d: %d\n", n, result);
    
    return EXIT_SUCCESS;
}
