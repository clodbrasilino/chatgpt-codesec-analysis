#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0U) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0U; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char **tuple_to_list(const char *const *tuple, size_t tuple_len,
                            const char *insert, size_t *out_len)
{
    char **result;
    size_t total;
    size_t i;
    size_t pos;

    if (tuple == NULL || insert == NULL || out_len == NULL) {
        return NULL;
    }

    if (tuple_len == 0U) {
        return NULL;
    }

    if (tuple_len > (SIZE_MAX / sizeof(char *)) / 2U) {
        return NULL;
    }

    total = tuple_len * 2U;
    result = malloc(total * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    pos = 0U;
    for (i = 0U; i < tuple_len; i++) {
        result[pos] = duplicate_string(tuple[i]);
        if (result[pos] == NULL) {
            free_string_list(result, pos);
            return NULL;
        }
        pos++;

        result[pos] = duplicate_string(insert);
        if (result[pos] == NULL) {
            free_string_list(result, pos);
            return NULL;
        }
        pos++;
    }

    *out_len = total;
    return result;
}

int main(void)
{
    const char *tuple[] = { "apple", "banana", "cherry" };
    size_t tuple_len = sizeof(tuple) / sizeof(tuple[0]);
    const char *insert = "FRUIT";
    char **list;
    size_t list_len;
    size_t i;

    list = tuple_to_list(tuple, tuple_len, insert, &list_len);
    if (list == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (i = 0U; i < list_len; i++) {
        printf("'%s'", list[i]);
        if (i + 1U < list_len) {
            printf(", ");
        }
    }
    printf("]\n");

    free_string_list(list, list_len);
    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness (CWE-120, `memcpy` in `duplicate_string`):**

The Flawfinder warning is a heuristic hit, not an actual vulnerability, and no functional change is required. The code already satisfies the conditions the warning asks you to verify:

1. **Bounded length measurement** — `strnlen(src, MAX_STRING_LEN)` guarantees `len` never exceeds `MAX_STRING_LEN`, and the subsequent `len >= MAX_STRING_LEN` check rejects unterminated or oversized inputs before any allocation occurs.
2. **Destination sized from the same `len`** — `malloc(len + 1U)` allocates exactly the number of bytes that `memcpy(copy, src, len)` writes, plus one byte for the terminator. The destination can therefore always hold the source data, which is precisely the fix CWE-120 calls for.
3. **No integer overflow** — `len + 1U` cannot wrap because `len < MAX_STRING_LEN` (4096), far below `SIZE_MAX`.
4. **Guaranteed termination** — `copy[len] = '\0'` is written explicitly rather than relying on the source being terminated.

Other defensive properties already present and preserved: NULL-parameter guards throughout, the `SIZE_MAX / sizeof(char *) / 2U` check preventing multiplication overflow in the allocation for `result`, and `free_string_list(result, pos)` correctly freeing only the `pos` successfully-allocated entries on partial failure, avoiding both leaks and frees of uninitialized pointers.
