```c
#include <stdio.h>
#include <stdlib.h>

int findLongestSubsequence(int arr[], int n) {
    if (n <= 0) {
        return 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    int maxLength = 1;
    int i, j;

    for (i = 0; i < n; i++) {
        dp[i] = 1;
    }

    for (i = 1; i < n; i++) {
        for (j = 0; j < i; j++) {
            if (abs(arr[i] - arr[j]) == 1 && dp[i] < dp[j] + 1) {
                dp[i] = dp[j] + 1;
            }
        }
        if (maxLength < dp[i]) {
            maxLength = dp[i];
        }
    }

    free(dp);
    return maxLength;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5, 3, 2};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = findLongestSubsequence(arr, n);

    if (result >= 0) {
        printf("Length of longest subsequence: %d\n", result);
    } else {
        printf("Error occurred\n");
    }

    return 0;
}
```
