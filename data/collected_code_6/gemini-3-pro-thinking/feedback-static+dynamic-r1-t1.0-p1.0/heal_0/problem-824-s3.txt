#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

size_t remove_even_numbers(int *array, size_t length);

size_t remove_even_numbers(int *array, size_t length) {
    if (array == NULL || length == 0) {
        return 0;
    }

    size_t new_length = 0;
    for (size_t i = 0; i < length; i++) {
        if (array[i] % 2 != 0) {
            array[new_length] = array[i];
            new_length++;
        }
    }

    return new_length;
}

int main(void) {
    size_t initial_length = 10;
    int *numbers = (int *)malloc(initial_length * sizeof(int));

    if (numbers == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < initial_length; i++) {
        numbers[i] = (int)i;
    }

    size_t new_length = remove_even_numbers(numbers, initial_length);

    for (size_t i = 0; i < new_length; i++) {
        printf("%d\n", numbers[i]);
    }

    free(numbers);
    numbers = NULL;

    return EXIT_SUCCESS;
}
