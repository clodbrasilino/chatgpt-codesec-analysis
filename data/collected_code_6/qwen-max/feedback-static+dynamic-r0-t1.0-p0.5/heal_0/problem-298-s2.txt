#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node* create_node(int value) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    new_node->value = value;
    new_node->next = NULL;
    return new_node;
}

void append(Node **head, int value) {
    Node *new_node = create_node(value);
    if (*head == NULL) {
        *head = new_node;
    } else {
        Node *current = *head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
}

int is_in_list(Node *list, int value) {
    while (list != NULL) {
        if (list->value == value) {
            return 1;
        }
        list = list->next;
    }
    return 0;
}

void find_common_elements(Node *list1, Node *list2) {
    Node *current = list1;
    while (current != NULL) {
        if (is_in_list(list2, current->value)) {
            printf("%d ", current->value);
        }
        current = current->next;
    }
    printf("\n");
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int main() {
    Node *list1 = NULL;
    Node *list2 = NULL;

    append(&list1, 1);
    append(&list1, 2);
    append(&list1, 3);
    append(&list1, 4);

    append(&list2, 3);
    append(&list2, 4);
    append(&list2, 5);
    append(&list2, 6);

    find_common_elements(list1, list2);

    free_list(list1);
    free_list(list2);

    return 0;
}
