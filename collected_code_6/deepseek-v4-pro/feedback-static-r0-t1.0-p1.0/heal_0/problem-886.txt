```c
#include <stdio.h>
#include <stdlib.h>

double average_of_list(int *list, int length) {
    if (list == NULL || length <= 0) {
        return 0.0;
    }
    
    int sum = 0;
    for (int i = 0; i < length; i++) {
        sum += list[i];
    }
    
    return (double)sum / length;
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5};
    int length = sizeof(list) / sizeof(list[0]);
    
    double avg = average_of_list(list, length);
    printf("Average: %.2f\n", avg);
    
    return 0;
}
```
