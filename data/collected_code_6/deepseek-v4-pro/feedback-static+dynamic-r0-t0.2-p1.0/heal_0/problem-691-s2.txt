#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int first;
    int second;
} Tuple;

typedef struct {
    int key;
    int *values;
    int count;
} Group;

Group *group_by_second(Tuple *tuples, int tuple_count, int *group_count) {
    if (tuples == NULL || tuple_count <= 0 || group_count == NULL) {
        *group_count = 0;
        return NULL;
    }

    Group *groups = NULL;
    int group_capacity = 0;
    *group_count = 0;

    for (int i = 0; i < tuple_count; i++) {
        int found = 0;
        for (int j = 0; j < *group_count; j++) {
            if (groups[j].key == tuples[i].second) {
                groups[j].values = (int *)realloc(groups[j].values, 
                                                  (groups[j].count + 1) * sizeof(int));
                if (groups[j].values == NULL) {
                    for (int k = 0; k < *group_count; k++) {
                        free(groups[k].values);
                    }
                    free(groups);
                    *group_count = 0;
                    return NULL;
                }
                groups[j].values[groups[j].count] = tuples[i].first;
                groups[j].count++;
                found = 1;
                break;
            }
        }

        if (!found) {
            if (*group_count >= group_capacity) {
                group_capacity = (group_capacity == 0) ? 4 : group_capacity * 2;
                Group *temp = (Group *)realloc(groups, group_capacity * sizeof(Group));
                if (temp == NULL) {
                    for (int k = 0; k < *group_count; k++) {
                        free(groups[k].values);
                    }
                    free(groups);
                    *group_count = 0;
                    return NULL;
                }
                groups = temp;
            }

            groups[*group_count].key = tuples[i].second;
            groups[*group_count].values = (int *)malloc(sizeof(int));
            if (groups[*group_count].values == NULL) {
                for (int k = 0; k < *group_count; k++) {
                    free(groups[k].values);
                }
                free(groups);
                *group_count = 0;
                return NULL;
            }
            groups[*group_count].values[0] = tuples[i].first;
            groups[*group_count].count = 1;
            (*group_count)++;
        }
    }

    return groups;
}

void free_groups(Group *groups, int group_count) {
    if (groups == NULL) {
        return;
    }
    for (int i = 0; i < group_count; i++) {
        free(groups[i].values);
    }
    free(groups);
}

int main(void) {
    Tuple tuples[] = {
        {1, 2},
        {3, 1},
        {4, 2},
        {5, 3},
        {6, 1},
        {7, 2},
        {8, 3},
        {9, 1}
    };
    int tuple_count = sizeof(tuples) / sizeof(tuples[0]);
    int group_count = 0;

    Group *groups = group_by_second(tuples, tuple_count, &group_count);

    if (groups == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    for (int i = 0; i < group_count; i++) {
        printf("Key %d: ", groups[i].key);
        for (int j = 0; j < groups[i].count; j++) {
            printf("%d ", groups[i].values[j]);
        }
        printf("\n");
    }

    free_groups(groups, group_count);

    return 0;
}
