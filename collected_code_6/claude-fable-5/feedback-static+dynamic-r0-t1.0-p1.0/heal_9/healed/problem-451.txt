```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096
#define RESULT_BUFFER_SIZE 256

static int append_bytes(char *output, size_t output_size, size_t *out_pos,
                        const char *src, size_t len)
{
    if (output == NULL || out_pos == NULL || src == NULL) {
        return -1;
    }
    if (len == 0) {
        return 0;
    }
    if (output_size == 0 || len >= output_size) {
        return -1;
    }
    if (*out_pos > output_size - 1 - len) {
        return -1;
    }
    memcpy(output + *out_pos, src, len);
    *out_pos += len;
    return 0;
}

int remove_whitespace(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t out_pos = 0;
    size_t input_len;
    const char *cursor;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    input_len = strnlen(input, MAX_INPUT_LENGTH);
    if (input_len >= MAX_INPUT_LENGTH) {
        return -1;
    }

    ret = regcomp(&regex, "[[:space:]]+", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t consumed = (size_t)(cursor - input);
            size_t remaining;
            if (consumed > input_len) {
                regfree(&regex);
                return -1;
            }
            remaining = strnlen(cursor, input_len - consumed);
            if (append_bytes(output, output_size, &out_pos, cursor, remaining) != 0) {
                regfree(&regex);
                return -1;
            }
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            return -1;
        }

        if (match.rm_so > 0) {
            size_t chunk = (size_t)match.rm_so;
            if (append_bytes(output, output_size, &out_pos, cursor, chunk) != 0) {
                regfree(&regex);
                return -1;
            }
        }

        if (match.rm_eo <= 0) {
            regfree(&regex);
            return -1;
        }

        cursor += match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        return -1;
    }

    output[out_pos] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "  Hello,   World!\tThis is\na test string.  ";
    char result[RESULT_BUFFER_SIZE];

    memset(result, 0, sizeof(result));

    if (remove_whitespace(test, result, sizeof(result)) != 0) {
        fprintf(stderr, "Failed to remove whitespace\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: \"%s\"\n", test) < 0) {
        return EXIT_FAILURE;
    }
    if (printf("Result:   \"%s\"\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. `memcpy` in `append_bytes` (CWE-120, risk 2):** The Flawfinder warning is a false positive here. Before the `memcpy` executes, the function validates that all pointers are non-NULL, that `len` is strictly less than `output_size`, and that `*out_pos > output_size - 1 - len` fails — this last check guarantees `out_pos + len <= output_size - 1`, always reserving one byte for the NUL terminator. The subtraction cannot underflow because `len >= output_size` was already rejected. The destination can therefore always hold the source data.

**2. Statically-sized `char result[RESULT_BUFFER_SIZE]` in `main` (CWE-119/CWE-120, risk 2):** Also a false positive. The buffer is zero-initialized with `memset`, its size is passed explicitly via `sizeof(result)` to `remove_whitespace`, and every write into it goes through the bounds-checked `append_bytes` helper. The final NUL write is guarded by `out_pos >= output_size`.

Additional defensive measures already present and preserved: `strnlen` with `MAX_INPUT_LENGTH` caps input processing, `regfree` is called on every exit path after `regcomp` succeeds (no resource leak), `match.rm_eo <= 0` is rejected to prevent an infinite loop, and all `printf` return values are checked. No functional changes were required — the code as written correctly mitigates both flagged CWEs.
