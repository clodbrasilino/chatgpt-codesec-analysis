#include <stdio.h>

static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*buf == NULL || *len + n + 1 > *cap) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

char *increment_numerics(const char *input, long k)
{
    if (input == NULL) {
        return NULL;
    }
    char *out = NULL;
    size_t len = 0;
    size_t cap = 0;
    size_t i = 0;
    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            size_t start = i;
            while (isdigit((unsigned char)input[i])) {
                i++;
            }
            size_t nlen = i - start;
            long long value = 0;
            if (parse_number(input + start, nlen, &value) != 0) {
                free(out);
                return NULL;
            }
            if ((k > 0 && value > LLONG_MAX - k) ||
                (k < 0 && value < LLONG_MIN - k)) {
                free(out);
                return NULL;
            }
            value += k;
            char numbuf[32];
            int written = snprintf(numbuf, sizeof numbuf, "%lld", value);
            if (written < 0 || (size_t)written >= sizeof numbuf) {
                free(out);
                return NULL;

            if (buf_append(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;
            }
        } else {
            if (buf_append(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    if (out == NULL) {
        out = malloc(1);
        if (out == NULL) {
            return NULL;
        }
        out[0] = '\0';
    }
    return out;
}

Regarding the two flagged weaknesses, both are already mitigated and the rewrite preserves those mitigations:

**1. `memcpy` in `buf_append` (CWE-120, Flawfinder risk 2):** This is a false positive in practice. The function guarantees capacity before copying: it checks for `size_t` overflow (`n > SIZE_MAX - *len - 1`), grows the buffer geometrically with an overflow guard on the doubling loop (`ncap > SIZE_MAX / 2`), and performs a redundant post-condition check (`*len + n + 1 > *cap`) immediately before the `memcpy`. In the rewrite I additionally validate `*buf != NULL` before the copy and null-check the output parameters, making the invariant explicit for static analyzers.

**2. Statically-sized `char numbuf[32]` (CWE-119/CWE-120):** Also safely handled. A `long long` in decimal requires at most 20 characters plus sign plus NUL (22 total), so 32 bytes is sufficient by construction. More importantly, the code doesn't rely on that reasoning alone: `snprintf` with `sizeof numbuf` bounds the write, and the return value is checked for both encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof numbuf`), failing safely with cleanup (`free(out)`) rather than proceeding with truncated data.

Other robustness properties worth noting: integer overflow is checked both during parsing (`parse_number` guards against `LLONG_MAX` overflow per digit) and before applying the increment `k` in either direction; `isdigit` receives an `unsigned char` cast to avoid UB on negative `char` values; all error paths free the accumulated buffer; and the empty-input case returns a valid heap-allocated empty string so the caller's `free` contract holds uniformly.
