```c
#include <stdio.h>
#include <stdlib.h>

long long multiply_consecutive(int *arr, int size) {
    if (size <= 0) return 1;
    long long product = 1;
    for (int i = 0; i < size; ++i) {
        product *= arr[i];
    }
    return product;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    long long result = multiply_consecutive(arr, size);
    printf("Product of the consecutive numbers: %lld\n", result);
    return 0;
}
```
