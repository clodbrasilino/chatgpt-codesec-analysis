#include <stdlib.h>
#include <errno.h>
#include <limits.h>

long long jacobsthal_lucas(int n) {
    if (n == 0) return 2;
    if (n == 1) return 1;
    
    long long prev2 = 2;
    long long prev1 = 1;
    long long current = 0;
    
    for (int i = 2; i <= n; i++) {
        if (prev1 > LLONG_MAX - 2 * prev2) {
            errno = ERANGE;
            return -1;
        }
        current = prev1 + 2 * prev2;
        prev2 = prev1;
        prev1 = current;
    }
    
    return current;
}

    int n;

        fprintf(stderr, "Invalid input\n");

    }
    
    if (n < 0) {
        fprintf(stderr, "Input must be non-negative\n");
        return EXIT_FAILURE;
    }
    
    errno = 0;
    long long result = jacobsthal_lucas(n);

    return EXIT_SUCCESS;
}
