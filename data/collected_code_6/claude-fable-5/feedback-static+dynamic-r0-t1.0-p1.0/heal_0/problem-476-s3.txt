#include <stdio.h>
#include <stdlib.h>

int sum_min_max(const int *arr, size_t size, long long *result)
{
    size_t i;
    int min;
    int max;

    if (arr == NULL || size == 0 || result == NULL) {
        return -1;
    }

    min = arr[0];
    max = arr[0];

    for (i = 1; i < size; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
        if (arr[i] > max) {
            max = arr[i];
        }
    }

    *result = (long long)min + (long long)max;
    return 0;
}

int main(void)
{
    int numbers[] = { 12, -7, 45, 3, 0, 99, -23, 8 };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    long long sum = 0;

    if (sum_min_max(numbers, count, &sum) != 0) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    printf("Sum of largest and smallest: %lld\n", sum);
    return EXIT_SUCCESS;
}
