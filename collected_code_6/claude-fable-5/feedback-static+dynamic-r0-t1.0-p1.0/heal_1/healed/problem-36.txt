```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int nth_digit(long long numerator, long long denominator, long long n, int *digit)
{
    unsigned long long num;
    unsigned long long den;
    unsigned long long remainder;
    long long i;

    if (digit == NULL) {
        return -1;
    }

    if (denominator == 0) {
        return -1;
    }

    if (n < 1) {
        return -1;
    }

    if (numerator < 0) {
        num = (unsigned long long)(-(numerator + 1)) + 1ULL;
    } else {
        num = (unsigned long long)numerator;
    }

    if (denominator < 0) {
        den = (unsigned long long)(-(denominator + 1)) + 1ULL;
    } else {
        den = (unsigned long long)denominator;
    }

    if (num >= den) {
        return -1;
    }

    if (den > ULLONG_MAX / 10ULL) {
        return -1;
    }

    remainder = num % den;

    for (i = 0; i < n - 1; i++) {
        remainder = (remainder * 10ULL) % den;
    }

    *digit = (int)((remainder * 10ULL) / den);

    return 0;
}

int main(void)
{
    long long numerator;
    long long denominator;
    long long n;
    int digit;
    int result;

    printf("Enter numerator: ");
    if (scanf("%lld", &numerator) != 1) {
        fprintf(stderr, "Invalid input for numerator.\n");
        return EXIT_FAILURE;
    }

    printf("Enter denominator: ");
    if (scanf("%lld", &denominator) != 1) {
        fprintf(stderr, "Invalid input for denominator.\n");
        return EXIT_FAILURE;
    }

    printf("Enter n: ");
    if (scanf("%lld", &n) != 1) {
        fprintf(stderr, "Invalid input for n.\n");
        return EXIT_FAILURE;
    }

    result = nth_digit(numerator, denominator, n, &digit);

    if (result != 0) {
        fprintf(stderr, "Error: invalid arguments (denominator must be nonzero, n >= 1, and fraction must be proper).\n");
        return EXIT_FAILURE;
    }

    printf("The digit at position %lld is: %d\n", n, digit);

    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weaknesses:

1. **Negation UB on `LLONG_MIN`**: Both `numerator = -numerator` and `denominator = -denominator` invoked undefined behavior when the value was `LLONG_MIN` (since `-LLONG_MIN` cannot be represented in `long long`). The fix computes the absolute value into an `unsigned long long` using `-(x + 1) + 1`, which avoids overflow at every step and correctly handles `LLONG_MIN`.

2. **Signed multiplication overflow in `remainder * 10`**: All arithmetic now uses `unsigned long long`, and a guard `den > ULLONG_MAX / 10` rejects denominators large enough to cause overflow. Since `remainder < den` is an invariant of the modulo operation, `remainder * 10` can never exceed `ULLONG_MAX` after that guard, eliminating both flagged overflow sites in the loop and in the final digit computation.
