```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_SIZE 1024

static char *snake_to_camel(const char *snake, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, max_len);
    if (len >= max_len) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = (char)tolower((unsigned char)snake[i]);
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT_SIZE];
    char *result;
    size_t len;

    memset(input, 0, sizeof(input));

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "No input provided\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input, sizeof(input));
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (`strlen` over-read)**: Replaced both `strlen` calls with `strnlen`, which takes an explicit maximum length. The `snake_to_camel` function now receives a `max_len` parameter so it never reads beyond the caller's known buffer bounds, even if the string were not null-terminated.

2. **CWE-119/CWE-120 (statically-sized array)**: The fixed-size buffer is retained (it is safe when used with `fgets` bounded by `sizeof(input)`), but a named constant `MAX_INPUT_SIZE` is used, the buffer is zero-initialized, explicitly null-terminated after `fgets`, and all subsequent operations are bounded by `sizeof(input)`.

3. **Removed dead code**: The original check `len == (size_t)-1` was meaningless (a `size_t` returned by `strlen` can never meaningfully be compared that way); it was replaced with a real bounds check (`len >= max_len`) that detects an unterminated or oversized input.

4. **Allocation safety**: `malloc(len + 1)` can no longer overflow since `len` is strictly bounded by `max_len`.
