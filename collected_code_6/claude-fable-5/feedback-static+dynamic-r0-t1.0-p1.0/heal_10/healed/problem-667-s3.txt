```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT 1024

size_t count_vowels(const char *str)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        int c = tolower((unsigned char)str[i]);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
            count++;
        }
    }

    return count;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = fgetc(stdin);
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[MAX_INPUT];
    size_t len;
    size_t vowels;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    vowels = count_vowels(buffer);
    printf("Number of vowels: %zu\n", vowels);

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses:**

1. **`fgetc` in a loop (CWE-120, CWE-20, risk 1):** This is a false positive in this context. The `flush_stdin` function reads characters one at a time and discards them — no data is stored into any buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop if the stream closes.

2. **Statically-sized `char` array (CWE-119/CWE-120, risk 2):** This is mitigated correctly in the original code:
   - `fgets(buffer, sizeof(buffer), stdin)` limits the read to the buffer size, guaranteeing no overflow.
   - The explicit `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders null termination (technically redundant since `fgets` always null-terminates, but harmless).
   - `strnlen` is used instead of `strlen`, bounding the length scan.
   - The truncation case (`len == sizeof(buffer) - 1` with no newline) is handled by draining the remainder of the line via `flush_stdin`, preventing leftover input from polluting subsequent reads.

3. **`tolower` usage:** Correctly casts to `unsigned char` before passing to `tolower`, avoiding undefined behavior on platforms where `char` is signed and the value is negative.
