```c
#include <stdio.h>
#include <stdlib.h>

typedef enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_DOUBLE
} ValueType;

typedef struct {
    ValueType type;
    union {
        int i;
        float f;
        double d;
    } value;
} HeterogeneousItem;

static double item_as_double(const HeterogeneousItem *item)
{
    switch (item->type) {
    case TYPE_INT:
        return (double)item->value.i;
    case TYPE_FLOAT:
        return (double)item->value.f;
    case TYPE_DOUBLE:
        return item->value.d;
    default:
        return 0.0;
    }
}

int find_max(const HeterogeneousItem *list, size_t count, double *result)
{
    size_t i;
    double max;

    if (list == NULL || result == NULL || count == 0U) {
        return -1;
    }

    max = item_as_double(&list[0]);

    for (i = 1U; i < count; i++) {
        double current = item_as_double(&list[i]);
        if (current > max) {
            max = current;
        }
    }

    *result = max;
    return 0;
}

int main(void)
{
    HeterogeneousItem list[5];
    double max_value = 0.0;
    size_t count = 5U;

    list[0].type = TYPE_INT;
    list[0].value.i = 42;

    list[1].type = TYPE_FLOAT;
    list[1].value.f = 3.14f;

    list[2].type = TYPE_DOUBLE;
    list[2].value.d = 99.5;

    list[3].type = TYPE_INT;
    list[3].value.i = 7;

    list[4].type = TYPE_DOUBLE;
    list[4].value.d = 56.78;

    if (find_max(list, count, &max_value) != 0) {
        (void)fprintf(stderr, "Error: could not find maximum value\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum value: %f\n", max_value) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
