```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_words_with_length(const char *str, int k) {
    if (str == NULL || k <= 0) {
        return NULL;
    }

    size_t len = strlen(str);
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t i = 0, j = 0;
    while (i < len) {
        while (i < len && isspace((unsigned char)str[i])) {
            result[j++] = str[i++];
        }

        size_t word_start = i;
        while (i < len && !isspace((unsigned char)str[i])) {
            i++;
        }
        size_t word_length = i - word_start;

        if (word_length != (size_t)k) {
            for (size_t m = word_start; m < i; m++) {
                result[j++] = str[m];
            }
        }
    }

    result[j] = '\0';
    return result;
}

int main(void) {
    const char *input = "This is a test string with words of varying lengths";
    int k = 2;

    char *output = remove_words_with_length(input, k);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed or invalid input\n");
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", input);
    printf("After removing words of length %d: %s\n", k, output);

    free(output);
    return EXIT_SUCCESS;
}
```
