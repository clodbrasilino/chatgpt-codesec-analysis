#include <stdio.h>
#include <stdlib.h>

int find_missing_number(int *array, size_t size) {
    if (size == 0) {
        return -1;
    }

    size_t low = 0;
    size_t high = size - 1;

    while (low <= high) {
        size_t mid = low + (high - low) / 2;

        if (array[mid] == (int)mid + array[0]) {
            low = mid + 1;
        } else {
            if (mid == 0) {
                return array[0] + (int)mid;
            }
            high = mid - 1;
        }
    }

    return array[0] + (int)low;
}

int main(void) {
    int arr1[] = {1, 2, 3, 4, 6, 7, 8};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    int result1 = find_missing_number(arr1, size1);
    printf("%d\n", result1);

    int arr2[] = {2, 3, 5, 6, 7};
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);
    int result2 = find_missing_number(arr2, size2);
    printf("%d\n", result2);

    int arr3[] = {1, 2, 3, 4};
    size_t size3 = sizeof(arr3) / sizeof(arr3[0]);
    int result3 = find_missing_number(arr3, size3);
    printf("%d\n", result3);

    int *arr4 = NULL;
    size_t size4 = 0;
    int result4 = find_missing_number(arr4, size4);
    printf("%d\n", result4);

    return 0;
}
