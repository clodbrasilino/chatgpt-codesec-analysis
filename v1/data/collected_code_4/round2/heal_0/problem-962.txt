```C
#include<stdio.h>

int sum_even_numbers(int l, int r) {
    int sum = 0;
    for (int i = l; i <= r; i++) {
        if (i % 2 == 0) {
            sum += i;
        }
    }
    return sum;
}

int main() {
    int l, r;
    printf("Enter the lower range: ");
    scanf("%d", &l);
    printf("Enter the upper range: ");
    scanf("%d", &r);

    if (l > r){
        printf("Invalid range. Lower limit is greater than upper limit.\n");
        return -1;
    }

    int sum = sum_even_numbers(l, r);
    printf("The sum of even numbers between %d and %d is: %d\n", l, r, sum);
    return 0;
}
```
