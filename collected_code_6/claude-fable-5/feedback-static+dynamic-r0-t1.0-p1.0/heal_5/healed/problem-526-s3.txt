```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define BUFFER_SIZE 256

static void capitalize_first_last(char *str, size_t max_len)
{
    size_t i;
    size_t len;

    if (str == NULL || max_len == 0) {
        return;
    }

    len = strnlen(str, max_len);
    i = 0;

    while (i < len) {
        while (i < len && isspace((unsigned char)str[i])) {
            i++;
        }
        if (i >= len) {
            break;
        }
        {
            size_t start = i;
            size_t end;

            while (i < len && !isspace((unsigned char)str[i])) {
                i++;
            }
            end = i - 1;
            str[start] = (char)toupper((unsigned char)str[start]);
            str[end] = (char)toupper((unsigned char)str[end]);
        }
    }
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char *newline;

    if (printf("Enter a string: ") < 0) {
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    newline = strchr(buffer, '\n');
    if (newline != NULL) {
        *newline = '\0';
    }

    capitalize_first_last(buffer, sizeof(buffer));

    if (printf("Result: %s\n", buffer) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120). The rewritten version preserves all of the existing mitigations, since they correctly address the identified weakness:

1. **Bounded input reading** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing the classic `gets()`-style overflow.
2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` acts as a defensive backstop, even though `fgets` already null-terminates on success.
3. **Length-limited string scan** — `strnlen(str, max_len)` instead of `strlen()` ensures the function never reads past the buffer even if passed a non-terminated string.
4. **NULL and zero-length guards** — `capitalize_first_last` validates its arguments before any pointer dereference.
5. **Safe `ctype` usage** — all characters are cast to `unsigned char` before being passed to `isspace()`/`toupper()`, avoiding undefined behavior with negative `char` values.
6. **Checked I/O return values** — `printf`, `fflush`, and `fgets` failures are all handled, so errors don't silently propagate.

The only change made was extracting the magic number `256` into a `BUFFER_SIZE` macro for maintainability; no functional or security-relevant behavior was altered because the original mitigations were already correct and complete.
