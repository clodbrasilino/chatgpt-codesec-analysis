#include <stdio.h>
#include <math.h>
#include <stdlib.h>

double tetrahedron_area(double a, double b, double c, double d, double e, double f) {
    double a2 = a * a;
    double b2 = b * b;
    double c2 = c * c;
    double d2 = d * d;
    double e2 = e * e;
    double f2 = f * f;
    
    double term1 = a2 * d2 * (b2 + c2 + e2 + f2 - a2 - d2);
    double term2 = b2 * e2 * (a2 + c2 + d2 + f2 - b2 - e2);
    double term3 = c2 * f2 * (a2 + b2 + d2 + e2 - c2 - f2);
    double term4 = a2 * b2 * c2 + a2 * e2 * f2 + b2 * d2 * f2 + c2 * d2 * e2;
    
    double volume_sq = (term1 + term2 + term3 - term4) / 144.0;
    if (volume_sq < 0.0) {
        return -1.0;
    }
    return sqrt(volume_sq);
}

int main(void) {
    double edges[6];
    const char *labels[] = {"AB", "AC", "AD", "BC", "BD", "CD"};
    
    for (int i = 0; i < 6; i++) {
        printf("Enter edge %s: ", labels[i]);
        if (scanf("%lf", &edges[i]) != 1) {
            fprintf(stderr, "Invalid input\n");
            return EXIT_FAILURE;
        }
        if (edges[i] <= 0.0) {
            fprintf(stderr, "Edge lengths must be positive\n");
            return EXIT_FAILURE;
        }
    }
    
    double area = tetrahedron_area(edges[0], edges[1], edges[2], edges[3], edges[4], edges[5]);
    
    if (area < 0.0) {
        fprintf(stderr, "Invalid tetrahedron dimensions\n");
        return EXIT_FAILURE;
    }
    
    printf("Surface area: %.6f\n", area);
    return EXIT_SUCCESS;
}
