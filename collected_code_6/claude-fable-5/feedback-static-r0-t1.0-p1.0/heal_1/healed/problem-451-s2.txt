```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_whitespace(const char *input);

char *remove_whitespace(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result = NULL;
    size_t result_len = 0;
    size_t input_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:space:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_len - (size_t)(cursor - input));
            if (result_len + remaining > input_len) {
                regfree(&regex);
                free(result);
                return NULL;
            }
            memcpy(result + result_len, cursor, remaining);
            result_len += remaining;
            break;
        } else if (ret != 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if (match.rm_so < 0 || match.rm_eo <= 0 ||
            (size_t)match.rm_eo > input_len - (size_t)(cursor - input)) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if (result_len + (size_t)match.rm_so > input_len) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(result + result_len, cursor, (size_t)match.rm_so);
        result_len += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test = "  Hello \t World \n from  C  ";
    char *cleaned;

    cleaned = remove_whitespace(test);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to remove whitespace\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: \"%s\"\n", test) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  \"%s\"\n", cleaned) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    free(cleaned);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (over-read via `strlen`)**: Replaced both `strlen` calls with `strnlen` bounded by `MAX_INPUT_LEN` (for the initial input) and by the remaining input length (inside the loop). If the input is not null-terminated within the bound, the function fails safely rather than over-reading.

2. **CWE-120 (unchecked `memcpy`)**: Before every `memcpy`, the code now verifies that `result_len` plus the copy size cannot exceed the allocated capacity (`input_len`), returning `NULL` on any violation instead of overflowing the buffer.

3. **Defensive validation of regex offsets**: `match.rm_so` and `match.rm_eo` are validated (non-negative, `rm_eo > 0` to prevent a zero-progress infinite loop, and bounded by the remaining input) before being used in pointer arithmetic or as copy lengths.
