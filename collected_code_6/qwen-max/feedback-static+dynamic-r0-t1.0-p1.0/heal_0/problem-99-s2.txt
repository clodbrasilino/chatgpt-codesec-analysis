#include <stdio.h>
#include <stdlib.h>

void decimalToBinary(int n) {
    int binaryNum[32];
    int i = 0;
    while (n > 0) {
        binaryNum[i] = n % 2;
        n = n / 2;
        i++;
    }

    for (int j = i - 1; j >= 0; j--) {
        printf("%d", binaryNum[j]);
    }
}

int main() {
    int number;
    printf("Enter a decimal number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input. Please enter an integer value.\n");
        return EXIT_FAILURE;
    }

    printf("Binary: ");
    if (number == 0) {
        printf("0");
    } else {
        decimalToBinary(number);
    }
    printf("\n");

    return EXIT_SUCCESS;
}
