```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char *output;
    size_t len;

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    output = remove_lowercase(buffer);
    if (output == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", output);
    fflush(stdout);
    free(output);

    return EXIT_SUCCESS;
}
```

The key change addressing the failed test cases: the original `remove_lowercase` returned `NULL` whenever `strnlen(input, MAX_INPUT_LEN)` equaled `MAX_INPUT_LEN`. Since `main` passes a buffer of exactly `MAX_INPUT_LEN` bytes, an input that filled the buffer (or edge conditions around the boundary) caused the function to bail out and print nothing — hence `<no output>` for the test cases. Removing that rejection (the string is already guaranteed NUL-terminated by `main`, and `strnlen` bounds the scan safely) allows valid inputs like `PYTHon`, `FIDdle`, and `STRinG` to produce `PYTH`, `FID`, and `STRG` as expected. The buffer in `main` is safely bounded by `fgets(buffer, sizeof(buffer), stdin)` plus explicit NUL termination, mitigating the CWE-119/CWE-120 concern, and `fflush(stdout)` guarantees the result is emitted before exit.
