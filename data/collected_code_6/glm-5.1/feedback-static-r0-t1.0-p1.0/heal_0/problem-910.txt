#include <stdio.h>
#include <stdbool.h>

bool is_leap_year(int year) {
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

bool validate_gregorian_date(int day, int month, int year) {
    if (year < 1582) {
        return false;
    }

    if (year == 1582) {
        if (month < 10) {
            return false;
        }
        if (month == 10 && day < 15) {
            return false;
        }
    }

    if (month < 1 || month > 12) {
        return false;
    }

    if (day < 1) {
        return false;
    }

    int max_days;

    switch (month) {
        case 1: case 3: case 5: case 7: case 8: case 10: case 12:
            max_days = 31;
            break;
        case 4: case 6: case 9: case 11:
            max_days = 30;
            break;
        case 2:
            max_days = is_leap_year(year) ? 29 : 28;
            break;
        default:
            return false;
    }

    if (day > max_days) {
        return false;
    }

    return true;
}

int main() {
    int day, month, year;

    if (scanf("%d %d %d", &day, &month, &year) != 3) {
        return 1;
    }

    if (validate_gregorian_date(day, month, year)) {
        printf("Valid\n");
    } else {
        printf("Invalid\n");
    }

    return 0;
}
