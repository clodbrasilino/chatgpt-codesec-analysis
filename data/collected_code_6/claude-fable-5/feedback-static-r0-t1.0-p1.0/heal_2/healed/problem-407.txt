```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged weakness (CWE-119/CWE-120, statically-sized buffer `digits[32]`) was already effectively mitigated in the original code, and those mitigations are preserved here:

1. **Bounded write:** `snprintf(digits, sizeof(digits), ...)` is used instead of `sprintf`, so the write can never exceed the buffer.
2. **Truncation check:** The return value of `snprintf` is validated (`len < 0 || (size_t)len >= sizeof(digits)`), rejecting any truncated or failed conversion. Since `LLONG_MAX` has at most 19 digits, a 32-byte buffer is more than sufficient, but the check guards against any surprise.
3. **Zero-initialization:** `memset` guarantees the buffer is fully initialized and NUL-terminated before use.
4. **Index safety:** All loop indices (`i`, `j`) are derived from the validated `len` and checked against bounds (`i >= 0`, `j > i`) before any array access, so `reverse_range` and `swap_chars` never receive out-of-range indices.
5. **Overflow safety on reconstruction:** The `result > (LLONG_MAX - d) / 10` guard prevents signed integer overflow when converting the permuted digit string back to `long long`.

The only change made is extracting the buffer size into a named constant (`DIGITS_BUF_SIZE`), which keeps the size definition in one place and makes future bounds reasoning explicit — the logic itself was already sound.
