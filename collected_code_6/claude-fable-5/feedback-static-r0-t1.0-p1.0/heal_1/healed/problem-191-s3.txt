#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_thirty_days(const char *month);
static void to_lower_string(char *dest, const char *src, size_t dest_size);

static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static int has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april",
        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN];
    size_t i;
    size_t count;

    if (month == NULL) {
        return -1;
    }

    if (strnlen(month, MAX_INPUT_LEN) >= MAX_INPUT_LEN) {
        return -1;
    }

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    for (i = 0U; i < count; i++) {
        if (strcmp(lowered, thirty_day_months[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    printf("Enter a month name: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (input[0] == '\0') {
        fprintf(stderr, "Empty input provided.\n");
        return EXIT_FAILURE;
    }

    result = has_thirty_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name provided.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the fixes applied:**

1. **CWE-119/CWE-120 (statically-sized arrays `lowered` and `input`):** These were already mitigated in the original code — `fgets(input, sizeof(input), stdin)` limits input to the buffer size, and `to_lower_string` receives `sizeof(lowered)` and performs explicit bounds checking (`i < dest_size - 1U`) before every write, always NUL-terminating. I added an explicit `input[sizeof(input) - 1U] = '\0'` as defense-in-depth to guarantee termination regardless of stream state.

2. **CWE-126 (`strlen` on potentially non-terminated strings):** Replaced both `strlen` calls with `strnlen`, bounding the read to the buffer capacity. This prevents any over-read even if a caller ever passes a non-terminated buffer to `has_thirty_days`, and makes the length check `>= MAX_INPUT_LEN` still behave correctly since `strnlen` caps at exactly that bound.

3. **CWE-120/CWE-20 (`getchar` in a loop):** This is a false positive in context — the loop discards excess characters until newline or EOF and writes nothing to any buffer, so no boundary can be violated. It is retained as-is because it is the correct idiom for draining stdin after a truncated `fgets` read.

The remaining logic (NULL checks, empty-input rejection, unsigned-char cast before `tolower`, `sizeof`-derived array count) was already defensive and correct, so it is preserved unchanged.
