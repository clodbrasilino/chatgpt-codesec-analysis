#include <stdlib.h>
#include <stddef.h>

typedef struct Dictionary {
    void *entries;
    size_t size;
    size_t capacity;
} Dictionary;

typedef struct DictionaryList {
    Dictionary **items;
    size_t count;
} DictionaryList;

Dictionary *dictionary_create(void) {
    Dictionary *d = malloc(sizeof(*d));
    if (d == NULL) {
        return NULL;
    }
    d->entries = NULL;
    d->size = 0;
    d->capacity = 0;
    return d;
}

void dictionary_destroy(Dictionary *d) {
    if (d == NULL) {
        return;
    }
    free(d->entries);
    free(d);
}

DictionaryList *create_list_of_empty_dictionaries(size_t count) {
    DictionaryList *list = malloc(sizeof(*list));
    if (list == NULL) {
        return NULL;
    }
    list->items = calloc(count, sizeof(*list->items));
    if (list->items == NULL && count > 0) {
        free(list);
        return NULL;
    }
    list->count = count;
    for (size_t i = 0; i < count; ++i) {
        list->items[i] = dictionary_create();
        if (list->items[i] == NULL) {
            for (size_t j = 0; j < i; ++j) {
                dictionary_destroy(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
    }
    return list;
}

void dictionary_list_destroy(DictionaryList *list) {
    if (list == NULL) {
        return;
    }
    for (size_t i = 0; i < list->count; ++i) {
        dictionary_destroy(list->items[i]);
    }
    free(list->items);
    free(list);
}

int main(void) {
    DictionaryList *list = create_list_of_empty_dictionaries(5);
    if (list == NULL) {
        return EXIT_FAILURE;
    }
    dictionary_list_destroy(list);
    return EXIT_SUCCESS;
}
