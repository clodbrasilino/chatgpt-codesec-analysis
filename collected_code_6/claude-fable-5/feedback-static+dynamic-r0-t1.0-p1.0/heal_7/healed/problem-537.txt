```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 1024
#define MAX_WORD_LEN 256
#define MAX_INPUT_LEN 65536

static char *duplicate_string(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(src, max_len - 1);
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static int extract_word(const char *str, size_t str_len, size_t *pos,
                        char *word, size_t word_size)
{
    size_t i;
    size_t w = 0;

    if (str == NULL || pos == NULL || word == NULL || word_size == 0) {
        return 0;
    }

    i = *pos;

    while (i < str_len && str[i] != '\0' && !isalnum((unsigned char)str[i])) {
        i++;
    }

    if (i >= str_len || str[i] == '\0') {
        *pos = i;
        word[0] = '\0';
        return 0;
    }

    while (i < str_len && str[i] != '\0' && isalnum((unsigned char)str[i])) {
        if (w + 1 < word_size) {
            word[w] = str[i];
            w++;
        }
        i++;
    }

    word[w] = '\0';
    *pos = i;
    return 1;
}

char *first_repeated_word(const char *str)
{
    char (*words)[MAX_WORD_LEN];
    char current[MAX_WORD_LEN];
    size_t pos = 0;
    size_t count = 0;
    size_t i;
    size_t len;
    size_t str_len;
    char *result;

    if (str == NULL) {
        return duplicate_string("None", MAX_WORD_LEN);
    }

    str_len = strnlen(str, MAX_INPUT_LEN);

    words = calloc((size_t)MAX_WORDS, sizeof(*words));
    if (words == NULL) {
        return NULL;
    }

    while (extract_word(str, str_len, &pos, current, sizeof(current)) == 1) {
        for (i = 0; i < count; i++) {
            if (strncmp(words[i], current, MAX_WORD_LEN) == 0) {
                result = duplicate_string(current, MAX_WORD_LEN);
                free(words);
                return result;
            }
        }
        if (count < MAX_WORDS) {
            len = strnlen(current, sizeof(current));
            if (len > (size_t)(MAX_WORD_LEN - 1)) {
                len = (size_t)(MAX_WORD_LEN - 1);
            }
            if (len > 0) {
                memcpy(words[count], current, len);
            }
            words[count][len] = '\0';
            count++;
        } else {
            break;
        }
    }

    free(words);
    return duplicate_string("None", MAX_WORD_LEN);
}

static int run_test(const char *input)
{
    char *repeated;

    repeated = first_repeated_word(input);
    if (repeated == NULL) {
        if (printf("Memory allocation failure\n") < 0) {
            return -1;
        }
        return -1;
    }

    if (strncmp(repeated, "None", MAX_WORD_LEN) == 0) {
        if (printf("No repeated word found\n") < 0) {
            free(repeated);
            return -1;
        }
    } else {
        if (printf("First repeated word: %s\n", repeated) < 0) {
            free(repeated);
            return -1;
        }
    }

    free(repeated);
    return 0;
}

int main(void)
{
    const char *test1 = "the quick brown fox jumps over the lazy dog";
    const char *test2 = "hello world foo bar";
    const char *test3 = "one two two three";

    if (run_test(test1) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test2) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test3) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`memcpy` in `duplicate_string` (CWE-120):** The destination is allocated with `malloc(len + 1)` immediately before the copy of exactly `len` bytes, so the destination is always large enough. The length is bounded by `strnlen(src, max_len - 1)`, preventing reads past unterminated input.

2. **`memcpy` in `first_repeated_word` (CWE-120):** The destination `words[count]` is exactly `MAX_WORD_LEN` bytes. The copy length is clamped with `strnlen(current, sizeof(current))` and additionally capped at `MAX_WORD_LEN - 1`, guaranteeing room for the null terminator written afterward.

3. **Statically-sized arrays (CWE-119/CWE-120):** Both `current[MAX_WORD_LEN]` and the heap-allocated `words` array are only ever written through bounds-checked paths: `extract_word` enforces `w + 1 < word_size` before writing and always null-terminates, and the `count < MAX_WORDS` guard prevents overflowing the word table.

Additional defensive properties already present: `str_len` is capped at `MAX_INPUT_LEN` via `strnlen`, all `ctype` calls cast to `unsigned char` to avoid UB on negative `char` values, `calloc` zero-initializes the word table, allocation failures are propagated and reported, and all heap memory is freed on every return path (no leaks on the early-return match case). The flawfinder hits are low-risk informational findings; no functional changes were required beyond explicit casts on the length clamp to silence signed/unsigned comparison warnings.
