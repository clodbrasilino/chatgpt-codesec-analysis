```c
#include <stdio.h>

int findLargestNumber(int arr[], int size) {
    if(size <= 0){
        return -1;
    }
    int i, max = arr[0];

    for(i = 1; i < size; i++) {
        if(arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

int main() {
    int arr[] = {15, 1200, 500, 100, 1250, 50};
    int size = sizeof(arr) / sizeof(arr[0]);

    int largest = findLargestNumber(arr, size);

    if(largest == -1){
        printf("Invalid array size...\n");
    } else {
        printf("Largest number in given array is %d\n", largest);
    }

    return 0;
}
```
