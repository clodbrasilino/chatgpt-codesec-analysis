```c
#include <stdio.h>
#include <stddef.h>
#include <stdbool.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096U
#define MAX_WORDS 7U

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t len = 0U;

    if (str == NULL) {
        return 0U;
    }

    while ((len < max_len) && (str[len] != '\0')) {
        len++;
    }

    return len;
}

static bool is_palindrome(const char *str)
{
    size_t left;
    size_t right;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = bounded_length(str, MAX_STRING_LENGTH);
    if ((len == 0U) || (len >= MAX_STRING_LENGTH)) {
        return false;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        if (tolower((unsigned char)str[left]) != tolower((unsigned char)str[right])) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

static size_t find_palindromes(const char *const *strings, size_t count,
                               const char **results, size_t results_capacity)
{
    size_t found;

    if ((strings == NULL) || (results == NULL) || (results_capacity == 0U)) {
        return 0U;
    }

    if (count > results_capacity) {
        count = results_capacity;
    }

    found = 0U;
    for (size_t i = 0U; i < count; i++) {
        if (found >= results_capacity) {
            break;
        }
        if (is_palindrome(strings[i])) {
            results[found] = strings[i];
            found++;
        }
    }

    return found;
}

int main(void)
{
    static const char *const words[MAX_WORDS] = {
        "racecar",
        "hello",
        "Level",
        "world",
        "noon",
        "example",
        "madam"
    };
    const size_t word_count = sizeof(words) / sizeof(words[0]);
    const char *palindromes[MAX_WORDS] = { NULL };
    const size_t palindromes_capacity = sizeof(palindromes) / sizeof(palindromes[0]);
    size_t found;

    if (word_count > palindromes_capacity) {
        return 1;
    }

    found = find_palindromes(words, word_count, palindromes, palindromes_capacity);

    if (found > palindromes_capacity) {
        return 1;
    }

    if (found == 0U) {
        if (puts("No palindromes found.") == EOF) {
            return 1;
        }
    } else {
        if (printf("Found %zu palindrome(s):\n", found) < 0) {
            return 1;
        }
        for (size_t i = 0U; i < found; i++) {
            if (palindromes[i] == NULL) {
                return 1;
            }
            if (printf("%s\n", palindromes[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

Analysis notes on the flagged weaknesses:

- **CWE-119/CWE-120 (statically-sized arrays)**: Both flagged arrays (`words` and `palindromes`) were already safe in the original code. They are fixed-size, initialized with compile-time constants, and never written beyond their bounds. `find_palindromes` receives the capacity explicitly and checks `found >= results_capacity` before every write, so no overflow is possible. These are Flawfinder false positives triggered simply by the presence of array declarations.

- **Hardening applied in the rewrite**: I added a defensive clamp (`count > results_capacity`) in `find_palindromes` so the loop bound itself can never exceed the destination capacity, making the bounds guarantee explicit at the point of entry rather than relying solely on the per-iteration check.

- **Existing safe practices preserved**: `bounded_length` replaces `strlen` to guarantee termination within `MAX_STRING_LENGTH`; `tolower` receives values cast to `unsigned char` to avoid undefined behavior with negative `char` values; all pointers are NULL-checked; all I/O return values are verified; sizes are computed with `sizeof` rather than hard-coded constants.
