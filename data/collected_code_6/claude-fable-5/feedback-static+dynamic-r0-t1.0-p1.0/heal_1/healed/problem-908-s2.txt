```c
#include <stdio.h>
#include <stdlib.h>

int find_fixed_point(const int *arr, size_t size)
{
    size_t low;
    size_t high;

    if (arr == NULL || size == 0U) {
        return -1;
    }

    low = 0U;
    high = size - 1U;

    while (low <= high) {
        size_t mid = low + ((high - low) / 2U);

        if (arr[mid] == (int)mid) {
            return (int)mid;
        }

        if (arr[mid] < (int)mid) {
            low = mid + 1U;
        } else {
            if (mid == 0U) {
                break;
            }
            high = mid - 1U;
        }
    }

    return -1;
}

int main(void)
{
    int arr[] = { -10, -5, 0, 3, 7 };
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int result;

    result = find_fixed_point(arr, size);

    if (result >= 0) {
        if (printf("Fixed point found at index: %d\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No fixed point found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

The change made addresses the identified weakness: the variable `mid` was previously declared at function scope but is only used inside the `while` loop. Its declaration has been moved into the loop body, reducing its scope to where it is actually needed. This improves readability, limits the variable's lifetime, and prevents accidental use of a stale value outside the loop.

The rest of the code was already sound: the NULL/zero-size guard prevents the `size - 1U` underflow, the `mid == 0U` check prevents the unsigned wraparound when decrementing `high`, the midpoint calculation avoids overflow, and the return values of `printf` are checked.
