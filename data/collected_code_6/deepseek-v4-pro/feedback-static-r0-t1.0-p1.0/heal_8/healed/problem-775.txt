#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

bool check_odd_indices_odd(const int *arr, size_t len) {
    if (arr == NULL || len < 2) {
        return false;
    }
    
    for (size_t i = 1; i < len; i += 2) {
        if (arr[i] % 2 == 0) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int arr1[] = {2, 3, 6, 5, 10};
    size_t len1 = sizeof(arr1) / sizeof(arr1[0]);
    
    int arr2[] = {2, 4, 6, 5, 10};
    size_t len2 = sizeof(arr2) / sizeof(arr2[0]);
    
    int arr3[] = {1, 3, 5};
    size_t len3 = sizeof(arr3) / sizeof(arr3[0]);
    
    int arr4[] = {2};
    size_t len4 = sizeof(arr4) / sizeof(arr4[0]);
    
    printf("Test 1: %s\n", check_odd_indices_odd(arr1, len1) ? "true" : "false");
    printf("Test 2: %s\n", check_odd_indices_odd(arr2, len2) ? "true" : "false");
    printf("Test 3: %s\n", check_odd_indices_odd(arr3, len3) ? "true" : "false");
    printf("Test 4: %s\n", check_odd_indices_odd(arr4, len4) ? "true" : "false");
    
    const int *null_arr = NULL;
    size_t null_len = 0;
    printf("Test 5: %s\n", check_odd_indices_odd(null_arr, null_len) ? "true" : "false");
    
    return 0;
}
