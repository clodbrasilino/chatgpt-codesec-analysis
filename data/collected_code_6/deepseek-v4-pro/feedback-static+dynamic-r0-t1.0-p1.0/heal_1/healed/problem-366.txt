#include <stdio.h>
#include <limits.h>

int largest_adjacent_product(const int *arr, int size) {
    if (arr == NULL || size < 2) {
        return INT_MIN;
    }
    
    int max_product = arr[0] * arr[1];
    
    for (int i = 1; i < size - 1; i++) {
        int product = arr[i] * arr[i + 1];
        if (product > max_product) {
            max_product = product;
        }
    }
    
    return max_product;
}

int main(void) {
    int arr1[] = {3, 6, -2, -5, 7, 3};
    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    printf("%d\n", largest_adjacent_product(arr1, size1));
    
    int arr2[] = {-1, -2, -3, -4};
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    printf("%d\n", largest_adjacent_product(arr2, size2));
    
    int arr3[] = {5, 2};
    int size3 = sizeof(arr3) / sizeof(arr3[0]);
    printf("%d\n", largest_adjacent_product(arr3, size3));
    
    int arr4[] = {1};
    int size4 = sizeof(arr4) / sizeof(arr4[0]);
    printf("%d\n", largest_adjacent_product(arr4, size4));
    
    return 0;
}
