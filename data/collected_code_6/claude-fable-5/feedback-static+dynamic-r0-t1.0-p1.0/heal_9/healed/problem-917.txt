```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if ((i + 1 < len) &&
            isupper((unsigned char)input[i]) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;
            size_t dest_size;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            dest_size = sizeof(matches[found].text);

            if (match_len >= dest_size) {
                match_len = dest_size - 1;
            }

            if (match_len > 0 && match_len <= len - start && match_len < dest_size) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`char text[MAX_INPUT]` and `char input[MAX_INPUT]` (CWE-119/CWE-120, statically-sized arrays):** These warnings are informational. The buffers are safe because every write into them is bounds-checked. `fgets(input, sizeof(input), stdin)` limits input to the buffer size, and the explicit `input[sizeof(input) - 1] = '\0'` guarantees NUL-termination. Using `strnlen` instead of `strlen` bounds the length computation even if termination were somehow missing.

2. **`memcpy` (CWE-120, unchecked copy):** This is a false positive here. Before the copy, `match_len` is clamped with `if (match_len >= dest_size) match_len = dest_size - 1;`, and the copy only executes when `match_len < dest_size` and `match_len <= len - start`, ensuring both that the destination can hold the data plus the NUL terminator and that the source read stays within `input`. The terminator write at `text[match_len]` is guaranteed in-bounds since `match_len <= dest_size - 1`.

3. **Additional defensive measures already in place:** NULL-pointer and zero-size validation in `find_sequences`, casts to `unsigned char` before `isupper`/`islower` (avoiding undefined behavior with negative `char` values), and a bounded loop capped by `max_matches` preventing overflow of the `matches` array.

The code as written contains the appropriate mitigations, so the rewrite preserves the logic verbatim with the comments removed.
