#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int* data;
    size_t length;
} List;

typedef struct {
    List* lists;
    size_t count;
} ListOfLists;

ListOfLists* find_min_length_lists(const ListOfLists* input) {
    if (!input || input->count == 0) {
        return NULL;
    }

    size_t min_length = input->lists[0].length;
    for (size_t i = 1; i < input->count; ++i) {
        if (input->lists[i].length < min_length) {
            min_length = input->lists[i].length;
        }
    }

    size_t min_count = 0;
    for (size_t i = 0; i < input->count; ++i) {
        if (input->lists[i].length == min_length) {
            min_count++;
        }
    }

    ListOfLists* result = malloc(sizeof(ListOfLists));
    if (!result) {
        return NULL;
    }

    result->lists = malloc(min_count * sizeof(List));
    if (!result->lists) {
        free(result);
        return NULL;
    }

    result->count = min_count;
    size_t index = 0;
    for (size_t i = 0; i < input->count; ++i) {
        if (input->lists[i].length == min_length) {
            result->lists[index].length = input->lists[i].length;
            if (min_length > 0) {
                result->lists[index].data = malloc(min_length * sizeof(int));
                if (!result->lists[index].data) {
                    for (size_t j = 0; j < index; ++j) {
                        free(result->lists[j].data);
                    }
                    free(result->lists);
                    free(result);
                    return NULL;
                }
                for (size_t k = 0; k < min_length; ++k) {
                    result->lists[index].data[k] = input->lists[i].data[k];
                }
            } else {
                result->lists[index].data = NULL;
            }
            index++;
        }
    }

    return result;
}

void free_list_of_lists(ListOfLists* lol) {
    if (lol) {
        for (size_t i = 0; i < lol->count; ++i) {
            free(lol->lists[i].data);
        }
        free(lol->lists);
        free(lol);
    }
}

int main(void) {
    int data1[] = {1, 2, 3};
    int data2[] = {4};
    int data3[] = {5, 6};
    int data4[] = {7};

    List lists[] = {
        {data1, 3},
        {data2, 1},
        {data3, 2},
        {data4, 1}
    };

    ListOfLists input = {lists, 4};

    ListOfLists* mins = find_min_length_lists(&input);

    if (mins) {
        for (size_t i = 0; i < mins->count; ++i) {
            printf("List %zu (length %zu): ", i, mins->lists[i].length);
            for (size_t j = 0; j < mins->lists[i].length; ++j) {
                printf("%d ", mins->lists[i].data[j]);
            }
            printf("\n");
        }
        free_list_of_lists(mins);
    }

    return 0;
}
