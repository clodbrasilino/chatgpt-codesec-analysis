#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
            best = max_int(best, dp[i - 1][j] - prices[j]);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}

**Analysis notes:**

The DP recurrence in the rewritten version is the correct formulation for the "at most k stock transactions" problem, and it resolves the failures listed in the comments:

1. **Test case 0** (`{3, 2, 6, 5, 0, 3}`, k=2): The buggy version returned 5, which is symptomatic of only capturing a single buy/sell pass or resetting `best` incorrectly between transaction levels. The correct recurrence — `dp[i][j] = max(dp[i][j-1], prices[j] + best)` with `best = max(best, dp[i-1][j] - prices[j])` — properly chains profit from the previous transaction level, yielding 7 (buy 2/sell 6, buy 0/sell 3).

2. **Test case 1** (`{1, 2, 3, 4, 5}`, k=2): Returning 0 suggests the earlier code skipped the `k >= n/2` fast path or mishandled monotonically increasing sequences. The greedy accumulation branch now correctly returns 4.

3. **Test case 2** (`{2, 4, 1}`, k=2): Same fast-path fix applies (k=2 ≥ n/2=1), correctly returning 2.

Other robustness points preserved: guard clauses for `NULL`/trivial inputs, full cleanup on partial `calloc` failure (no leaks on the error path), `calloc` guaranteeing zero-initialized base cases for `dp[0][*]` and `dp[*][0]`, and a distinct `-1` sentinel for allocation failure checked by the caller.
