#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* remove_chars(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return NULL;
    }

    size_t len1 = strlen(str1);
    size_t len2 = strlen(str2);
    char *result = (char*)malloc(len1 + 1);
    
    if (result == NULL) {
        return NULL;
    }

    size_t res_index = 0;
    for (size_t i = 0; i < len1; i++) {
        int found = 0;
        for (size_t j = 0; j < len2; j++) {
            if (str1[i] == str2[j]) {
                found = 1;
                break;
            }
        }
        if (!found) {
            result[res_index++] = str1[i];
        }
    }
    result[res_index] = '\0';

    char *shrunk = (char*)realloc(result, res_index + 1);
    if (shrunk != NULL) {
        result = shrunk;
    }

    return result;
}

int main() {
    const char *s1 = "hello world";
    const char *s2 = "old";
    
    char *filtered = remove_chars(s1, s2);
    if (filtered != NULL) {
        printf("%s\n", filtered);
        free(filtered);
    }

    return 0;
}
