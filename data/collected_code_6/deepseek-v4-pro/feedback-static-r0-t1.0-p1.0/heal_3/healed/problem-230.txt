#include <stdio.h>
#include <string.h>

void replace_spaces(char *str, char replacement) {
    if (str == NULL) {
        return;
    }
    
    while (*str != '\0') {
        if (*str == ' ') {
            *str = replacement;
        }
        str++;
    }
}

    char str[256];
    char replacement;
    size_t len;
