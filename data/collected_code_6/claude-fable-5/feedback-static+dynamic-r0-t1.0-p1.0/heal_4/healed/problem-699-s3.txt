#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static void strip_newline(char *s)
{
    size_t len;

    if (s == NULL) {
        return;
    }

    len = strnlen(s, MAX_INPUT_LEN + 2);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r')) {
        s[len - 1] = '\0';
        len--;
    }
}

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count01 = 0;
    int count10 = 0;
    int result;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN + 1);
    len2 = strnlen(s2, MAX_INPUT_LEN + 1);

    if (len1 > MAX_INPUT_LEN || len2 > MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2 || len1 == 0) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] != s2[i]) {
            if (s1[i] == '0') {
                count01++;
            } else {
                count10++;
            }
        }
    }

    if ((count01 + count10) % 2 != 0) {
        return -1;
    }

    result = count01 / 2 + count10 / 2;
    if (count01 % 2 != 0) {
        result += 2;
    }

    return result;
}

{
    char s1[MAX_INPUT_LEN + 2];
    char s2[MAX_INPUT_LEN + 2];

**Key changes made:**

1. **Fixed the "no output" test failures (test cases 0, 1, 2):** The original code used `fgets` and silently returned `1` with no output whenever a read failed (e.g., when both strings appear on one line separated by whitespace, or when the last line lacks a trailing newline). The rewrite uses whitespace-tolerant `scanf` with an explicit field width, and always emits `"Not Possible"` on read failure rather than exiting silently.

2. **Addressed the Flawfinder CWE-119/CWE-120 buffer concerns:** The `scanf` calls use a field width (`%4097s`) strictly smaller than the buffer size (`MAX_INPUT_LEN + 2`), guaranteeing space for the NUL terminator and preventing overflow. Buffers are also zero-initialized so they are always valid NUL-terminated strings.

3. **Fixed an off-by-one length check:** The original `len1 >= MAX_INPUT_LEN` incorrectly rejected valid inputs of exactly `MAX_INPUT_LEN` characters; the bound is now `len > MAX_INPUT_LEN` with `strnlen` capped at `MAX_INPUT_LEN + 1` to detect oversize input safely.

4. **Rejected empty strings** explicitly (`len1 == 0`), since an empty comparison is not a meaningful swap problem and avoids ambiguity with failed reads.
