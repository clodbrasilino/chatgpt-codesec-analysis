#include <stdio.h>
#include <stdlib.h>

long long calculate_unique_product(int *arr, int n) {
    int *unique = (int *)calloc(n, sizeof(int));
    if (unique == NULL) {
        return -1;
    }
    
    int unique_count = 0;
    for (int i = 0; i < n; ++i) {
        int j;
        for (j = 0; j < unique_count; ++j) {
            if (arr[i] == unique[j]) {
                break;
            }
        }
        if (j == unique_count) {
            unique[unique_count++] = arr[i];
        }
    }

    long long product = 1;
    for (int i = 0; i < unique_count; ++i) {
        product *= unique[i];
    }

    free(unique);
    return product;
}

int main() {
    int arr[] = {2, 3, 5, 2, 4, 3, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    long long product = calculate_unique_product(arr, n);
    if (product != -1) {
        printf("Product of unique numbers: %lld\n", product);
    } else {
        printf("Error calculating product.\n");
    }
    return 0;
}
