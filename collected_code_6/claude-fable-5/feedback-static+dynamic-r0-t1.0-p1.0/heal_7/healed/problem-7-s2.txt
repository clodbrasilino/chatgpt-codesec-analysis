```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;
    size_t total_len;

    if (text == NULL) {
        return -1;
    }

    total_len = strnlen(text, MAX_INPUT_LEN);
    if (total_len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not terminated\n");
        return -1;
    }

    ret = regcomp(&regex, "[[:alpha:]]{4,}", REG_EXTENDED);
    if (ret != 0) {
        size_t errlen = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errlen);
        if (errbuf != NULL) {
            regerror(ret, &regex, errbuf, errlen);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Regex compilation failed\n");
        }
        return -1;
    }

    cursor = text;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            break;
        }

        size_t remaining = strnlen(cursor, MAX_INPUT_LEN);
        if ((size_t)match.rm_eo > remaining) {
            break;
        }

        size_t length = (size_t)(match.rm_eo - match.rm_so);
        if (length >= MAX_INPUT_LEN || length > SIZE_MAX - 1) {
            break;
        }

        if ((size_t)match.rm_so > remaining ||
            length > remaining - (size_t)match.rm_so) {
            break;
        }

        size_t bufsize = length + 1;
        char *word = malloc(bufsize);
        if (word == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(word, cursor + match.rm_so, length);
        word[length] = '\0';
        printf("%s\n", word);

        free(word);

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *input = "The quick brown fox jumps over a lazy dog near it";

    if (find_long_words(input) != 0) {
        fprintf(stderr, "Failed to process input string\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **Redundant condition removed** — `length < bufsize` was always true since `bufsize = length + 1`; the check is gone, which eliminates the `knownConditionTrueFalse` warning.

2. **Bounds validated before allocation** — the check `(size_t)match.rm_so + length <= remaining` was rewritten as `length > remaining - (size_t)match.rm_so` (after verifying `match.rm_so <= remaining`), which avoids potential integer overflow in the addition and guarantees the source range is valid *before* `memcpy` is reached. This resolves the Flawfinder CWE-120 concern: the destination is exactly `length + 1` bytes, exactly `length` bytes are copied, and the source range is proven in-bounds.

3. **Overflow guard on `bufsize`** — added `length > SIZE_MAX - 1` check so `length + 1` can never wrap around, making the allocation size provably correct.

4. **Fail-fast structure** — invalid ranges now `break` out of the loop before any allocation occurs, so `malloc`/`memcpy` only execute on fully validated input.
