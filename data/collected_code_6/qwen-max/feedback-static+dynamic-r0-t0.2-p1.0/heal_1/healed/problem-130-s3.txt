#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int value;
    int count;
} Item;

int compare(const void *a, const void *b) {
    return ((Item *)a)->count - ((Item *)b)->count;
}

int findMaxFrequency(const int *arr, int n, int *maxValue) {
    if (n <= 0) return -1;

    Item *items = (Item *)malloc(n * sizeof(Item));
    if (items == NULL) return -1;

    for (int i = 0; i < n; ++i) {
        items[i].value = arr[i];
        items[i].count = 1;
    }

    for (int i = 0; i < n; ++i) {
        if (items[i].value != -1) {
            for (int j = i + 1; j < n; ++j) {
                if (items[i].value == items[j].value) {
                    items[i].count++;
                    items[j].value = -1;
                }
            }
        }
    }

    qsort(items, n, sizeof(Item), compare);

    int result = -1;
    for (int i = n - 1; i >= 0; --i) {
        if (items[i].value != -1) {
            *maxValue = items[i].value;
            result = items[i].count;
            break;
        }
    }

    free(items);
    return result;
}

int main() {
    int arr[] = {1, 2, 3, 2, 4, 2, 5, 1, 1};
    int n = sizeof(arr) / sizeof(arr[0]);
    int maxValue;

    int maxCount = findMaxFrequency(arr, n, &maxValue);
    if (maxCount != -1) {
        printf("Item with maximum frequency: %d, Frequency: %d\n", maxValue, maxCount);
    } else {
        printf("Error in finding the item with maximum frequency.\n");
    }

    return 0;
}
