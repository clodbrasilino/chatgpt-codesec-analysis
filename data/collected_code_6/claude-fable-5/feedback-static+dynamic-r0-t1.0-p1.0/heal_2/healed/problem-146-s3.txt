long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        sum += (long)(unsigned char)str[i];
    }

    return sum;
}

{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Total ASCII value: %ld\n", result);

    return EXIT_SUCCESS;
}

2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1] = '\0'` is defensive redundancy — `fgets` already null-terminates, but this guards against any misuse or future refactoring.

3. **Bounded length calculation**: `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so even an unterminated buffer could not cause an out-of-bounds read.

4. **Safe summation**: `ascii_sum` checks for `NULL`, uses `strnlen` with an explicit `max_len` bound, and casts through `unsigned char` before widening to `long`, avoiding sign-extension issues with characters above 0x7F.

5. **Newline stripping** is done with an explicit `len > 0` check, avoiding an out-of-bounds access at index `-1` on an empty string.
