```c
#include <stdio.h>

struct Node {
    int data;
    struct Node *next;
};

int list_sum(const struct Node *node) {
    if (node == NULL) {
        return 0;
    }
    return node->data + list_sum(node->next);
}

int main(void) {
    struct Node n1 = {1, NULL};
    struct Node n2 = {2, NULL};
    struct Node n3 = {3, NULL};

    n1.next = &n2;
    n2.next = &n3;

    printf("%d\n", list_sum(&n1));

    return 0;
}
```
