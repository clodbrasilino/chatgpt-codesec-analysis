```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

int left_rotate(char *str, size_t str_size, size_t d)
{
    size_t len;
    char *temp;

    if (str == NULL || str_size == 0 || str_size > MAX_STR_LEN) {
        return -1;
    }

    len = strnlen(str, str_size);
    if (len >= str_size) {
        return -1;
    }

    if (len == 0) {
        return 0;
    }

    d = d % len;
    if (d == 0) {
        return 0;
    }

    if (d > len || (len - d) > len) {
        return -1;
    }

    temp = malloc(len + 1);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, str + d, len - d);
    memcpy(temp + (len - d), str, d);
    temp[len] = '\0';

    if (len + 1 > str_size) {
        free(temp);
        return -1;
    }

    memcpy(str, temp, len + 1);
    free(temp);

    return 0;
}

int main(void)
{
    char str[] = "GeeksforGeeks";
    size_t d = 2;

    if (left_rotate(str, sizeof(str), d) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", str) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fixes applied:**

The original code was already quite defensively written — the flawfinder warnings on `memcpy` are heuristic (risk level 2) and flag every use of the function regardless of context. Here is why each call was already safe, and what the rewrite adds to make the safety explicit:

1. **`memcpy(temp, str + d, len - d)`** — `temp` holds `len + 1` bytes and `d < len` is guaranteed after the modulo and the `d == 0` early return, so `len - d < len + 1`. Safe by construction.

2. **`memcpy(temp + (len - d), str, d)`** — writes `d` bytes starting at offset `len - d`, ending exactly at offset `len`, within the `len + 1` allocation. Safe.

3. **`memcpy(str, temp, len + 1)`** — since `strnlen` returned `len < str_size`, we know `len + 1 <= str_size`, so the destination buffer can hold the copy. Safe.

The rewrite adds two explicit guard checks (`d > len` invariant check and `len + 1 > str_size` before copying back) so the bounds reasoning is enforced in code rather than only implied by control flow. These checks are effectively unreachable given the earlier logic, but they satisfy static analyzers, protect against future refactoring mistakes, and document the invariants at the point of use — a common practice when hardening code flagged by tools like Flawfinder.

Other correctness points preserved: `strnlen` prevents over-reads on unterminated buffers, the `len >= str_size` check rejects non-NUL-terminated input, `malloc` failure is handled, `temp` is freed on all paths after allocation, and `main` checks both the rotation result and the `printf` return value.
