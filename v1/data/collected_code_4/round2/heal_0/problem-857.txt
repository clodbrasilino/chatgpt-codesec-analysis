```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_STRINGS 10
#define MAX_LENGTH 50

void map(char stringArray[MAX_STRINGS][MAX_LENGTH]) {
    for(int i = 0; i < MAX_STRINGS; i++) {
        if(stringArray[i][0] != '\0') {
            printf("%s\n", stringArray[i]);
        } else {
            break;
        }
    }
}

int main() {
    char stringArray[MAX_STRINGS][MAX_LENGTH] = {"Hello", "World", ""};
    map(stringArray);
    return 0;
}
```
