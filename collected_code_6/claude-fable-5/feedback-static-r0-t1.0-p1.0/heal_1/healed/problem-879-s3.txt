```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

#define ERRBUF_SIZE 256

static int match_pattern(const char *text)
{
    regex_t regex;
    int ret;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "a.*b$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        char errbuf[ERRBUF_SIZE];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, text, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex execution failed\n");
        return -1;
    }
}

int main(void)
{
    const char *tests[] = {
        "ab",
        "acb",
        "a123b",
        "abc",
        "xyz",
        "aXXXXb"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = match_pattern(tests[i]);
        if (result < 0) {
            fprintf(stderr, "Error matching string: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (result == 1) {
            printf("\"%s\" matches\n", tests[i]);
        } else {
            printf("\"%s\" does not match\n", tests[i]);
        }
    }

    return EXIT_SUCCESS;
}
```

The key changes made to address the identified weaknesses:

1. **Buffer safety (CWE-119/CWE-120)**: The `errbuf` array now uses a named constant `ERRBUF_SIZE`, the call to `regerror` continues to pass `sizeof(errbuf)` to bound the write, and an explicit null-terminator is enforced at the last index as defensive hardening against any truncation edge case.

2. **Variable scope reduction**: The `result` variable is now declared inside the `for` loop body, limiting its scope to exactly where it is used, which resolves the cppcheck `variableScope` warning and improves maintainability.
