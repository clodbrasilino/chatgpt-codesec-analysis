#include <stdio.h>
#include <stdlib.h>

int check_last_element_p_times(int *array, int len, int p) {
    if (!array || len < 1 || p < 1)
        return -1;

    int last_element = array[len - 1];

    for(int i=0; i<p; i++) {
        if(last_element%2 == 0) {
            printf("The last element is even\n");
        } else {
            printf("The last element is odd\n");
        }
    }
    return 0;
}

int main() {
    int myarray[] = {1, 2, 3, 4, 5, 6};
    int len = sizeof(myarray) / sizeof(myarray[0]);

    if(check_last_element_p_times(myarray, len, 3) == -1) {
        printf("An error occurred!\n");
        return 1;
    }
    return 0;
}
