```c
#include <stdio.h>
#include <stdlib.h>

int sequential_search(const int *array, size_t size, int target) {
    if (array == NULL) {
        return -1;
    }

    for (size_t i = 0; i < size; ++i) {
        if (array[i] == target) {
            return (int)i;
        }
    }

    return -1;
}

int main(void) {
    int array[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3};
    size_t size = sizeof(array) / sizeof(array[0]);
    int target = 9;

    int result = sequential_search(array, size, target);

    if (result != -1) {
        printf("Element %d found at index %d.\n", target, result);
    } else {
        printf("Element %d not found in the array.\n", target);
    }

    target = 7;
    result = sequential_search(array, size, target);

    if (result != -1) {
        printf("Element %d found at index %d.\n", target, result);
    } else {
        printf("Element %d not found in the array.\n", target);
    }

    return 0;
}
```
