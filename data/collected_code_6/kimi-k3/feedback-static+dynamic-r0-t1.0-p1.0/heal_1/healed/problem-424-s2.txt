#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void extract_rear_elements(char *arr[], int size, char *result, size_t max_len) {
    if (arr == NULL || result == NULL || size <= 0 || max_len == 0) {
        return;
    }
    
    for (int i = 0; i < size; i++) {
        if (arr[i] != NULL) {
            size_t len = strnlen(arr[i], max_len);
            if (len > 0) {
                result[i] = arr[i][len - 1];
            } else {
                result[i] = '\0';
            }
        } else {
            result[i] = '\0';
        }
    }
    result[size] = '\0';
}

int main() {
    char *strings[] = {"hello", "world", "programming", "code"};
    int size = sizeof(strings) / sizeof(strings[0]);
    size_t max_len = 1024;
    char *result = malloc((size + 1) * sizeof(char));
    
    if (result == NULL) {
        return 1;
    }
    
    extract_rear_elements(strings, size, result, max_len);
    
    printf("Rear elements: ");
    for (int i = 0; i < size; i++) {
        if (result[i] != '\0') {
            printf("%c ", result[i]);
        }
    }
    printf("\n");
    
    free(result);
    return 0;
}
