#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month);
static void discard_remaining_input(void);

static bool has_31_days(const char *month)
{
    static const char *const months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    memset(lowered, 0, sizeof(lowered));

    for (i = 0; i < len && i < sizeof(lowered) - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strncmp(lowered, months_31[i], sizeof(lowered)) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No input provided.\n");
        return EXIT_FAILURE;
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

1. **`char lowered[MAX_INPUT_LEN]` (CWE-119/CWE-120, risk 2):** This is a false positive in practice. The buffer is fully protected: `strnlen` bounds the length check, inputs of length `>= MAX_INPUT_LEN` are rejected early, the copy loop is doubly bounded by `i < len && i < sizeof(lowered) - 1`, and the buffer is zero-initialized with explicit null termination afterward. No overflow is possible.

2. **`getchar` in a loop (CWE-120/CWE-20, risk 1):** Also a false positive. The characters read are discarded, never stored into a buffer, so no boundary can be violated. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed streams.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120, risk 2):** Safe as written. `fgets` is given `sizeof(input)` so it can never overflow, the buffer is pre-zeroed, a defensive null terminator is written at the last index, and leftover input beyond the buffer is drained via `discard_remaining_input()` so it cannot pollute subsequent reads.

Additional defensive qualities already present: `tolower` receives an `unsigned char` cast (avoiding UB on negative `char` values), `strncmp` is bounded, `fflush` and `fgets` return values are checked, and empty input is rejected. The code is portable C99 and requires no functional changes to remediate the reported findings.
