#include <stdio.h>
#include <string.h>

size_t safe_strlen(const char *str, size_t max_len) {
    size_t len = 0;
    while (len < max_len && str[len] != '\0') {
        len++;
    }
    return len;
}

void remove_first_last_occurrence(char *str, char ch) {
    if (str == NULL || *str == '\0') {
        return;
    }

    size_t len = safe_strlen(str, 4096);
    int first = -1;
    int last = -1;

    for (size_t i = 0; i < len; i++) {
        if (str[i] == ch) {
            if (first == -1) {
                first = (int)i;
            }
            last = (int)i;
        }
    }

    if (first == -1) {
        return;
    }

    if (first == last) {
        memmove(&str[first], &str[first + 1], len - (size_t)first);
        return;
    }

    memmove(&str[last], &str[last + 1], len - (size_t)last);
    memmove(&str[first], &str[first + 1], len - (size_t)first - 1);
}

int main(void) {
    char str1[] = "hello world";
    remove_first_last_occurrence(str1, 'o');
    printf("%s\n", str1);

    char str2[] = "aaaa";
    remove_first_last_occurrence(str2, 'a');
    printf("%s\n", str2);

    char str3[] = "character";
    remove_first_last_occurrence(str3, 'a');
    printf("%s\n", str3);

    char str4[] = "test";
    remove_first_last_occurrence(str4, 'x');
    printf("%s\n", str4);

    char str5[] = "single";
    remove_first_last_occurrence(str5, 'g');
    printf("%s\n", str5);

    return 0;
}
