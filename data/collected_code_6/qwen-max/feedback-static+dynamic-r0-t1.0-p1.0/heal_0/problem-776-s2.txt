#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_vowel(char c) {
    c = tolower(c);
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

int count_vowel_neighbors(const char *str) {
    int count = 0;
    if (str == NULL || *str == '\0') return 0;
    while (*str != '\0') {
        if (is_vowel(*(str - 1)) || is_vowel(*(str + 1))) {
            count++;
        }
        str++;
    }
    return count;
}

int main() {
    const char *test_str = "example";
    int result = count_vowel_neighbors(test_str);
    printf("Count: %d\n", result);
    return 0;
}
