```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *left_rotate(const char *str, size_t d)
{
    size_t len;
    char *result;

    if (str == NULL) {
        return NULL;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && str[len] != '\0') {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    if (len == 0) {
        result[0] = '\0';
        return result;
    }

    d = d % len;

    if (d > len || (len - d) > len) {
        free(result);
        return NULL;
    }

    memcpy(result, str + d, len - d);
    memcpy(result + (len - d), str, d);
    result[len] = '\0';

    return result;
}

int main(void)
{
    const char *original = "HelloWorld";
    size_t shift = 3;
    char *rotated;

    rotated = left_rotate(original, shift);
    if (rotated == NULL) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", original) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    if (printf("Rotated:  %s\n", rotated) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    free(rotated);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. This guarantees the scan never reads beyond a fixed limit, so a non-NUL-terminated buffer cannot cause an unbounded over-read. If the string reaches the limit without termination, the function safely rejects it by returning `NULL`.

2. **CWE-120 (`memcpy` buffer overflow):** The destination buffer is allocated as exactly `len + 1` bytes, and an explicit sanity check verifies that `d` and `len - d` cannot exceed `len` before either `memcpy` executes. Since `d = d % len` ensures `d < len`, the two copies (`len - d` bytes plus `d` bytes) total exactly `len` bytes, always fitting within the allocation with room for the terminator.

3. All existing defensive practices were preserved: `NULL` input handling, `malloc` failure checking, empty-string short-circuit, `printf` return-value checks, and proper `free` on every exit path to avoid leaks.
