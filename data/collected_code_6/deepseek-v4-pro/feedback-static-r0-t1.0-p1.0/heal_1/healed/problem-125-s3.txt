#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int maxDifference(const char *str) {
    if (str == NULL) return 0;
    
    int len = 0;
    while (str[len] != '\0') {
        if (len >= 1000) return -1;
        len++;
    }
    
    if (len == 0) return 0;
    
    int current_sum = 0;
    int max_sum = -1;
    int zeros = 0;
    int ones = 0;
    
    for (int i = 0; i < len; i++) {
        if (str[i] == '0') {
            current_sum++;
            zeros++;
        } else if (str[i] == '1') {
            current_sum--;
            ones++;
        } else {
            return -1;
        }
        
        if (current_sum > max_sum) {
            max_sum = current_sum;
        }
        
        if (current_sum < 0) {
            current_sum = 0;
        }
    }
    
    if (ones == len && zeros == 0) {
        return -1;
    }
    
    if (ones < len) {
        return max_sum;
    }
    
    return -1;
}

int main() {
    char input[1001];
    
    printf("Enter a binary string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }
    
    size_t len = 0;
    while (len < sizeof(input) && input[len] != '\0') {
        len++;
    }
    
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }
    
    if (len == 0) {
        printf("Maximum difference: 0\n");
        return 0;
    }
    
    int result = maxDifference(input);
    
    if (result == -1) {
        printf("Invalid input or no sub-string with positive difference\n");
    } else {
        printf("Maximum difference: %d\n", result);
    }
    
    return 0;
}
