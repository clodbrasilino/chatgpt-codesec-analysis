```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <regex.h>

#define NUM_BUFFER_SIZE 64

static int copy_match(char *dest, size_t dest_size, const char *src, size_t len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_size == 0 || len >= dest_size) {
        return -1;
    }
    memcpy(dest, src, len);
    dest[len] = '\0';
    return 0;
}

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[NUM_BUFFER_SIZE];
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    memset(buffer, 0, sizeof(buffer));

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;
        long value;
        char *endptr;

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            regfree(&regex);
            return -1;
        }

        len = (size_t)(match.rm_eo - match.rm_so);

        if (len >= sizeof(buffer)) {
            regfree(&regex);
            return -1;
        }

        if (copy_match(buffer, sizeof(buffer), cursor + match.rm_so, len) != 0) {
            regfree(&regex);
            return -1;
        }

        errno = 0;
        value = strtol(buffer, &endptr, 10);
        if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test1 = "abc 42 def -17 ghi 305 jkl 7";
    const char *test2 = "no numbers here";
    long max = 0;

    if (extract_max_number(test1, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test1, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test1);
    }

    if (extract_max_number(test2, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test2, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test2);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`memcpy` (CWE-120, buffer overflow risk):** The Flawfinder warning is a false positive here because `copy_match` validates all preconditions before copying — it rejects `NULL` pointers, a zero-size destination, and any length where `len >= dest_size`, guaranteeing room for both the data and the null terminator. The caller additionally performs a redundant `len >= sizeof(buffer)` check before invoking `copy_match`, providing defense in depth.

2. **`char buffer[NUM_BUFFER_SIZE]` (CWE-119/CWE-120, statically-sized array):** The fixed-size buffer is safe because every write path is bounds-checked. The match length is computed from validated `regmatch_t` offsets (`rm_so >= 0`, `rm_eo > rm_so`) and rejected if it cannot fit. The buffer is zero-initialized via `memset` before use.

3. **Additional robustness already present and preserved:** `strtol` overflow detection via `errno == ERANGE`, full-token validation via `*endptr != '\0'`, proper `regfree` on every exit path to avoid resource leaks, and loop termination when the cursor reaches the end of the string, preventing an infinite loop on empty matches.

The code is functionally correct and the identified weaknesses are properly mitigated by the existing bounds checks.
