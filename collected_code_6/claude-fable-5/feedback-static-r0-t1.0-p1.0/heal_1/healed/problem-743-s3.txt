```c
#include <stdio.h>
#include <stdlib.h>

static void reverse_range(int *arr, size_t start, size_t end)
{
    int tmp;

    while (start < end) {
        tmp = arr[start];
        arr[start] = arr[end];
        arr[end] = tmp;
        start++;
        end--;
    }
}

int rotate_right(int *arr, size_t n, size_t k)
{
    if (arr == NULL || n == 0) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    reverse_range(arr, 0, n - 1);
    reverse_range(arr, 0, k - 1);
    reverse_range(arr, k, n - 1);

    return 0;
}

int main(void)
{
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 3;
    size_t i;

    printf("Original list: ");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    if (rotate_right(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    printf("Rotated list:  ");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was functionally correct — the `memcpy` was safe because `temp` and `arr` both hold exactly `n` elements of the same type. However, the Flawfinder warning (CWE-120) flags `memcpy` because it cannot statically verify the destination size.

The rewrite eliminates the weakness entirely rather than merely suppressing it:

1. **No `memcpy`, no `malloc`** — the rotation is now done in-place using the classic three-reversal algorithm (reverse the whole array, then reverse the first `k` elements, then the remaining `n - k`). This removes the flagged call, the heap allocation, and the associated failure path (`malloc` returning `NULL`).

2. **Reduced attack surface** — with no dynamic memory, there is no risk of allocation failure, memory leaks, or size-calculation overflow in `n * sizeof(*temp)`.

3. **Same complexity, better space** — time remains O(n), but auxiliary space drops from O(n) to O(1).

4. **Preserved contract** — the function still returns `-1` for `NULL`/empty input and `0` on success, and `k % n` handling for `k >= n` is retained, so `main` and any other callers work unchanged.
