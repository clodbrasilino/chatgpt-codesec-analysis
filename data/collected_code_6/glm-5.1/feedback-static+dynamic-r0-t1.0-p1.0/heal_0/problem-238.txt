```c
#include <stdio.h>
#include <string.h>

size_t count_substrings(const char *str) {
    if (str == NULL) {
        return 0;
    }

    size_t n = strlen(str);
    return n * (n + 1) / 2;
}

int main(void) {
    const char *s = "abc";
    size_t count = count_substrings(s);
    printf("%zu\n", count);
    return 0;
}
```
