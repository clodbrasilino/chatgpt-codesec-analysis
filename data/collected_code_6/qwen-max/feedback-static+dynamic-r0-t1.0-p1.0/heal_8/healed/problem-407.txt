#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

int compare(const void *a, const void *b) {
    return (*(char *)a - *(char *)b);
}

int nextBiggerNumber(long num) {
    char str[21];
    snprintf(str, sizeof(str), "%ld", num);
    if (strlen(str) >= sizeof(str) - 1) {
        return -1;
    }
    size_t len = strlen(str);
    int i;
    for (i = len - 2; i >= 0 && str[i] >= str[i + 1]; --i) {}

    if (i == -1) {
        return -1;
    }

    for (int j = len - 1; j > i; --j) {
        if (str[j] > str[i]) {
            char temp = str[i];
            str[i] = str[j];
            str[j] = temp;
            break;
        }
    }

    qsort(str + i + 1, len - i - 1, sizeof(char), compare);

    char *endptr;
    errno = 0;
    long result = strtol(str, &endptr, 10);
    if (*endptr != '\0' || result > INT_MAX || result < INT_MIN || errno == ERANGE) {
        return -1;
    }
    return (int)result;
}

int main() {
    long num = 12345;
    int result = nextBiggerNumber(num);
    if (result != -1) {
        printf("Next bigger number: %d\n", result);
    } else {
        printf("No bigger number can be formed.\n");
    }
    return 0;
}
