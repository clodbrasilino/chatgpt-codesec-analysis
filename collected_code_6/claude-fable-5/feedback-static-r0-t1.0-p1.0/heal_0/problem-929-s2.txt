#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

size_t count_repeated_items(const int *tuple, size_t length);

size_t count_repeated_items(const int *tuple, size_t length)
{
    size_t repeated_count = 0;
    size_t i;
    size_t j;

    if (tuple == NULL || length == 0U) {
        return 0U;
    }

    for (i = 0U; i < length; i++) {
        bool seen_before = false;

        for (j = 0U; j < i; j++) {
            if (tuple[j] == tuple[i]) {
                seen_before = true;
                break;
            }
        }

        if (seen_before) {
            continue;
        }

        size_t occurrences = 1U;

        for (j = i + 1U; j < length; j++) {
            if (tuple[j] == tuple[i]) {
                occurrences++;
            }
        }

        if (occurrences > 1U) {
            repeated_count++;
        }
    }

    return repeated_count;
}

int main(void)
{
    const int tuple[] = {1, 2, 3, 2, 4, 5, 1, 2, 6};
    const size_t length = sizeof(tuple) / sizeof(tuple[0]);
    size_t result;

    result = count_repeated_items(tuple, length);

    if (printf("Number of repeated items: %zu\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
