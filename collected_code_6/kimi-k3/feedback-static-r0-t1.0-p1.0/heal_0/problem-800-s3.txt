#include <stdio.h>
#include <ctype.h>
#include <string.h>

void remove_whitespace(char *str) {
    if (str == NULL) {
        return;
    }
    
    char *read_ptr = str;
    char *write_ptr = str;
    
    while (*read_ptr != '\0') {
        if (!isspace((unsigned char)*read_ptr)) {
            *write_ptr = *read_ptr;
            write_ptr++;
        }
        read_ptr++;
    }
    *write_ptr = '\0';
}

int main(void) {
    char test_string[] = "Hello World This Is A Test";
    
    printf("Original: %s\n", test_string);
    remove_whitespace(test_string);
    printf("Modified: %s\n", test_string);
    
    return 0;
}
