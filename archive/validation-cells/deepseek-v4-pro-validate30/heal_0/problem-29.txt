```c
#include <stdio.h>
#include <stdlib.h>

int findOddOccurrence(const int *arr, int size) {
    int result = 0;
    for (int i = 0; i < size; i++) {
        result ^= arr[i];
    }
    return result;
}

int main(void) {
    int arr[] = {1, 2, 3, 2, 3, 1, 3};
    int size = sizeof(arr) / sizeof(arr[0]);
    int odd = findOddOccurrence(arr, size);
    printf("%d\n", odd);
    return 0;
}
```
