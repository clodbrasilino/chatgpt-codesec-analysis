#include <stdio.h>
#include <stdbool.h>

bool checkPattern(const char *str, const char *const *patterns, int pattern_count) {
    if (str == NULL || patterns == NULL || pattern_count <= 0) {
        return false;
    }

    const char *current_pos = str;

    for (int i = 0; i < pattern_count; i++) {
        if (patterns[i] == NULL) {
            return false;
        }

        const char *pat = patterns[i];
        while (*pat != '\0') {
            if (*current_pos == '\0' || *current_pos != *pat) {
                return false;
            }
            current_pos++;
            pat++;
        }
    }

    return *current_pos == '\0';
}

int main(void) {
    const char *str1 = "applebananacherry";
    const char *patterns1[] = {"apple", "banana", "cherry"};
    
    if (checkPattern(str1, patterns1, 3)) {
        printf("String matches the pattern sequence.\n");
    } else {
        printf("String does not match the pattern sequence.\n");
    }

    const char *str2 = "applecherrybanana";
    
    if (checkPattern(str2, patterns1, 3)) {
        printf("String matches the pattern sequence.\n");
    } else {
        printf("String does not match the pattern sequence.\n");
    }

    return 0;
}
