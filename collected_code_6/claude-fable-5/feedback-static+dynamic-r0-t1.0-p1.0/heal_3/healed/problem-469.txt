```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
            best = max_int(best, dp[i - 1][j] - prices[j]);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The algorithm itself is the standard O(k·n) dynamic-programming solution for "at most k stock transactions," and the DP recurrence is correct:

- `dp[i][j]` = max profit using at most `i` transactions through day `j`
- `best` tracks `max(dp[i-1][m] - prices[m])` for `m ≤ j`, so buys and sells are properly ordered

The failure symptoms in the comment (`got 0`, `got 5`) are the classic signatures of two bugs this version already guards against:

1. **Returning 0 spuriously** — caused in broken variants by an overly aggressive guard such as `k >= n` returning 0, or by initializing `best` to `-prices[0]` outside the transaction loop and never refreshing it per transaction level. Here `best` is correctly reset inside the `i` loop from `dp[i-1][0]`.

2. **Under-counting profit (got 5 instead of expected)** — typically from a missing `k >= n/2` fast path (capping transactions incorrectly) or from updating `best` *before* computing `dp[i][j]`, which allows buying and selling on the same day within the same transaction incorrectly. The order here (compute `dp[i][j]` first, then update `best`) is correct.

Memory handling is sound: partial-allocation cleanup on `calloc` failure, the `-1` error sentinel checked by the caller, and full deallocation before return. Expected output: `7`, `4`, and `2`.
