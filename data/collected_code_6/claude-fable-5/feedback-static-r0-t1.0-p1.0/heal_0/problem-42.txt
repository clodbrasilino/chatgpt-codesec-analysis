```c
#include <stdio.h>
#include <stdlib.h>

long sum_of_repeated_elements(const int *arr, size_t n)
{
    long sum = 0;
    size_t i;
    size_t j;

    if (arr == NULL || n == 0) {
        return 0;
    }

    for (i = 0; i < n; i++) {
        int is_repeated = 0;
        for (j = 0; j < n; j++) {
            if (i != j && arr[i] == arr[j]) {
                is_repeated = 1;
                break;
            }
        }
        if (is_repeated) {
            sum += (long)arr[i];
        }
    }

    return sum;
}

int main(void)
{
    int arr[] = {1, 2, 3, 2, 4, 3, 5, 3};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long result;

    result = sum_of_repeated_elements(arr, n);

    if (printf("Sum of repeated elements: %ld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
