#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    float *values;
    size_t size;
} FloatTuple;

FloatTuple parse_float_tuple(const char *str) {
    FloatTuple tuple = {NULL, 0};
    
    if (str == NULL) {
        return tuple;
    }
    
    const char *start = strchr(str, '(');
    const char *end = strrchr(str, ')');
    
    if (start == NULL || end == NULL || end <= start) {
        return tuple;
    }
    
    size_t src_len = (size_t)(end - start - 1);
    if (src_len == 0) {
        return tuple;
    }
    
    if (src_len >= SIZE_MAX) {
        return tuple;
    }
    
    char *buffer = (char *)malloc(src_len + 1);
    if (buffer == NULL) {
        return tuple;
    }
    
    if (src_len > SIZE_MAX - 1) {
        free(buffer);
        tuple.values = NULL;
        tuple.size = 0;
        return tuple;
    }
    
    memcpy(buffer, start + 1, src_len);
    buffer[src_len] = '\0';
    
    size_t capacity = 4;
    if (capacity > SIZE_MAX / sizeof(float)) {
        free(buffer);
        return tuple;
    }
    
    tuple.values = (float *)malloc(capacity * sizeof(float));
    if (tuple.values == NULL) {
        free(buffer);
        return tuple;
    }
    
    char *saveptr = NULL;
    char *token = strtok_r(buffer, ",", &saveptr);
    while (token != NULL) {
        while (*token == ' ') token++;
        
        if (*token == '\0') {
            free(tuple.values);
            free(buffer);
            tuple.values = NULL;
            tuple.size = 0;
            return tuple;
        }
        
        char *endptr = NULL;
        float val = strtof(token, &endptr);
        
        if (endptr == token) {
            free(tuple.values);
            free(buffer);
            tuple.values = NULL;
            tuple.size = 0;
            return tuple;
        }
        
        while (*endptr == ' ') endptr++;
        
        if (*endptr != '\0') {
            free(tuple.values);
            free(buffer);
            tuple.values = NULL;
            tuple.size = 0;
            return tuple;
        }
        
        if (tuple.size >= capacity) {
            if (capacity > SIZE_MAX / 2) {
                free(tuple.values);
                free(buffer);
                tuple.values = NULL;
                tuple.size = 0;
                return tuple;
            }
            size_t new_capacity = capacity * 2;
            if (new_capacity > SIZE_MAX / sizeof(float)) {
                free(tuple.values);
                free(buffer);
                tuple.values = NULL;
                tuple.size = 0;
                return tuple;
            }
            float *temp = (float *)realloc(tuple.values, new_capacity * sizeof(float));
            if (temp == NULL) {
                free(tuple.values);
                free(buffer);
                tuple.values = NULL;
                tuple.size = 0;
                return tuple;
            }
            tuple.values = temp;
            capacity = new_capacity;
        }
        
        tuple.values[tuple.size++] = val;
        token = strtok_r(NULL, ",", &saveptr);
    }
    
    free(buffer);
    return tuple;
}

int main(void) {
    const char *test = "(3.14, 2.718, 1.414, 0.577)";
    FloatTuple result = parse_float_tuple(test);
    
    if (result.values == NULL) {
        fprintf(stderr, "Failed to parse tuple\n");
        return EXIT_FAILURE;
    }
    
    printf("Parsed %zu values:\n", result.size);
    for (size_t i = 0; i < result.size; i++) {
        printf("%.3f ", result.values[i]);
    }
    printf("\n");
    
    free(result.values);
    return EXIT_SUCCESS;
}
