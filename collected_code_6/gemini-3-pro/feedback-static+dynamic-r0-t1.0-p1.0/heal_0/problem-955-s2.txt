```c
#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool is_abundant(int number) {
    if (number <= 0) {
        return false;
    }

    int sum = 0;
    for (int i = 1; i <= number / 2; i++) {
        if (number % i == 0) {
            sum += i;
        }
    }

    return sum > number;
}

int main(void) {
    int input_number = 0;

    if (scanf("%d", &input_number) != 1) {
        return EXIT_FAILURE;
    }

    if (is_abundant(input_number)) {
        printf("The number %d is abundant.\n", input_number);
    } else {
        printf("The number %d is not abundant.\n", input_number);
    }

    return EXIT_SUCCESS;
}
```
