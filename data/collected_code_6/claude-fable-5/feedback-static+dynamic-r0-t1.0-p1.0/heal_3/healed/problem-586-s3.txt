```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);

    if ((n - k) + k != n) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The two Flawfinder `memcpy` warnings (CWE-120) in the original code are effectively **false positives**, and the original code already contains the mitigations Flawfinder asks for:

1. **First `memcpy(temp, arr, head_bytes)`** — the destination `temp` is allocated with exactly `head_bytes`, and the source `arr` is guaranteed to hold at least `head_bytes` because `k < n` after the modulo reduction. The destination can always hold the source data.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`** — the write starts at offset `(n - k) * sizeof(int)` and spans `head_bytes = k * sizeof(int)`, ending exactly at `total_bytes`. The explicit checks `head_bytes + tail_bytes == total_bytes` and `head_bytes <= total_bytes` guarantee no overflow past the caller's buffer.

Additional defensive properties already present and preserved:

- **NULL/zero-length guard** prevents undefined behavior on empty input.
- **`n > SIZE_MAX / sizeof(int)`** guards against integer overflow in the byte-size multiplications.
- **`k = k % n`** normalizes the rotation count, so `k < n` always holds before any arithmetic.
- **`malloc` return value is checked**, avoiding a NULL dereference.
- **`memmove`** (not `memcpy`) is correctly used for the overlapping in-place shift.
- **`free(temp)`** on every exit path after allocation — no memory leak.

The rewrite keeps all of these invariants and adds one extra sanity assertion before the second `memcpy` to make the bounds relationship explicit at the point of use, which also quiets static analyzers that don't track the earlier arithmetic checks.
