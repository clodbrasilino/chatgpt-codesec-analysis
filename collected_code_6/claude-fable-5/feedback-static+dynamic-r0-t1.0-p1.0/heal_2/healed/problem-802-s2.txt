```c
#include <stdio.h>
#include <stdlib.h>

int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        size_t mid;

        if (arr[low] <= arr[high]) {
            return (int)low;
        }

        mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else {
            high = mid;
        }
    }

    return (int)low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    int arr3[] = {1, 2, 3, 4, 5};
    int result;

    result = count_rotations(arr1, sizeof(arr1) / sizeof(arr1[0]));
    if (result != 2) {
        fprintf(stderr, "Error computing rotations for arr1: expected 2, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr1: %d\n", result);

    result = count_rotations(arr2, sizeof(arr2) / sizeof(arr2[0]));
    if (result != 4) {
        fprintf(stderr, "Error computing rotations for arr2: expected 4, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr2: %d\n", result);

    result = count_rotations(arr3, sizeof(arr3) / sizeof(arr3[0]));
    if (result != 0) {
        fprintf(stderr, "Error computing rotations for arr3: expected 0, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr3: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis summary:** The original implementation mixed two strategies — a circular neighbor check (`next`/`prev` with modulo wrap-around) and a half-interval binary search — which introduced fragile edge conditions. The modulo-based `prev`/`next` comparisons can incorrectly identify a pivot when `mid` sits at an array boundary, and the `high = mid - 1U` narrowing combined with `low <= high` risks skipping past the true minimum (the reported failure of returning 3 instead of 2) as well as a `size_t` underflow when `mid == 0`.

The rewrite uses the canonical, provably correct binary search for the rotation point: if `arr[mid] > arr[high]` the minimum lies strictly to the right (`low = mid + 1`), otherwise it lies at `mid` or to the left (`high = mid`, never excluding `mid` itself). The loop invariant guarantees the minimum always remains within `[low, high]`, the `low < high` condition eliminates any unsigned underflow, and the early `arr[low] <= arr[high]` check keeps the O(log n) fast path for already-sorted (unrotated) segments.
