```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS 64
#define BUFFER_SIZE (MAX_BITS + 1)

static int decimal_to_binary(unsigned long long value, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;
    char temp[BUFFER_SIZE];

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));
    memset(buffer, 0, buffer_size);

    if (value == 0ULL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0ULL && index < (sizeof(temp) - 1)) {
        temp[index] = (char)('0' + (value & 1ULL));
        value >>= 1ULL;
        index++;
    }

    if (value > 0ULL) {
        return -1;
    }

    if (index >= buffer_size) {
        return -1;
    }

    for (i = 0; i < index; i++) {
        buffer[i] = temp[index - 1 - i];
    }
    buffer[index] = '\0';

    return 0;
}

int main(void)
{
    long long input = 0;
    unsigned long long magnitude = 0ULL;
    char binary[BUFFER_SIZE];
    int is_negative = 0;

    memset(binary, 0, sizeof(binary));

    printf("Enter a decimal number: ");

    if (scanf("%lld", &input) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (input < 0) {
        is_negative = 1;
        if (input == LLONG_MIN) {
            magnitude = (unsigned long long)LLONG_MAX + 1ULL;
        } else {
            magnitude = (unsigned long long)(-input);
        }
    } else {
        magnitude = (unsigned long long)input;
    }

    if (decimal_to_binary(magnitude, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    if (is_negative) {
        printf("Binary equivalent: -%s\n", binary);
    } else {
        printf("Binary equivalent: %s\n", binary);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warnings (CWE-119/CWE-120) flagged on `temp` and `binary` are false positives in this code — the warnings are triggered simply because statically-sized `char` arrays exist, not because of an actual defect. The original code already contains the mitigations Flawfinder recommends:

1. **Bounds checking on writes to `temp`:** The conversion loop condition `index < (sizeof(temp) - 1)` guarantees the loop never writes past the array, always leaving room for the NUL terminator.

2. **Overflow detection:** The post-loop check `if (value > 0ULL) return -1;` detects the (theoretically impossible for 64-bit values) case where the buffer was exhausted before conversion completed.

3. **Caller-supplied size validation:** `decimal_to_binary` takes `buffer_size` explicitly, rejects `NULL` or undersized buffers (`buffer_size < 2`), and verifies `index >= buffer_size` before copying — so the reversal loop and terminator write can never overflow the destination.

4. **Sufficient sizing:** `BUFFER_SIZE` is 65, which covers the maximum 64 binary digits of an `unsigned long long` plus the terminator.

5. **`LLONG_MIN` handling:** The special case avoids undefined behavior from negating `LLONG_MIN`, correctly computing the magnitude as `LLONG_MAX + 1ULL`.

6. **Initialization:** Both buffers are zeroed with `memset` before use, preventing reads of indeterminate memory.

Since the code is already defensively written and no genuine buffer overflow path exists, the rewrite preserves the logic as-is with the audit comments removed.
