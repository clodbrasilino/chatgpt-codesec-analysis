#include <stdio.h>

int findSmallestMissing(const int arr[], int size) {
    int present = 1;

    for (int i = 0; i < size; i++) {
        if (arr[i] == present) {
            present++;
            i = -1;
        }
    }
    return present;
}

int main(void) {
    int arr1[] = {1, 2, 3, 4, 5, 6, 7, 8};
    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    int missing1 = findSmallestMissing(arr1, size1);
    printf("Smallest missing: %d\n", missing1);

    int arr2[] = {2, 3, 4, 5, 6, 7, 8, 10};
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    int missing2 = findSmallestMissing(arr2, size2);
    printf("Smallest missing: %d\n", missing2);

    int arr3[] = {5, 4, 2, 1, 7, 8, 9, 3};
    int size3 = sizeof(arr3) / sizeof(arr3[0]);
    int missing3 = findSmallestMissing(arr3, size3);
    printf("Smallest missing: %d\n", missing3);

    int arr4[] = {2, 1, 3, 4, 5, 6, 7, 8, 10, 9, 11, 12, 14};
    int size4 = sizeof(arr4) / sizeof(arr4[0]);
    int missing4 = findSmallestMissing(arr4, size4);
    printf("Smallest missing: %d\n", missing4);

    int arr5[] = {1};
    int size5 = sizeof(arr5) / sizeof(arr5[0]);
    int missing5 = findSmallestMissing(arr5, size5);
    printf("Smallest missing: %d\n", missing5);

    int arr6[] = {3, 2, 1};
    int size6 = sizeof(arr6) / sizeof(arr6[0]);
    int missing6 = findSmallestMissing(arr6, size6);
    printf("Smallest missing: %d\n", missing6);

    return 0;
}
