#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int min_jumps(const int *arr, size_t n, int *result)
{
    size_t jumps, max_reach, steps, i;

    if (result == NULL) {
        return -1;
    }

    *result = 0;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n == 1) {
        *result = 0;
        return 0;
    }

    if (arr[0] <= 0) {
        return -1;
    }

    jumps = 1;
    max_reach = (size_t)arr[0];
    steps = (size_t)arr[0];

    for (i = 1; i < n; ++i) {
        if (i == n - 1) {
            *result = (int)jumps;
            return 0;
        }

        if (arr[i] > 0) {
            size_t candidate = i + (size_t)arr[i];
            if (candidate > max_reach) {
                max_reach = candidate;
            }
        }

        if (steps == 0) {
            return -1;
        }

        --steps;

        if (steps == 0) {
            if (max_reach <= i) {
                return -1;
            }
            ++jumps;
            steps = max_reach - i;
        }
    }

    return -1;
}

int main(void)
{
    int arr[] = {1, 3, 5, 8, 9, 2, 6, 7, 6, 8, 9};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result = 0;

    if (min_jumps(arr, n, &result) == 0) {
        printf("Minimum number of jumps: %d\n", result);
    } else {
        printf("End is not reachable\n");
    }

    return 0;
}
