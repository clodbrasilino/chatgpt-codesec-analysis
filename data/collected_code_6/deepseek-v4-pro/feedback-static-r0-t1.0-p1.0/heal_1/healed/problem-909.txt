#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

static void reverse_string(char *str) {
    int len = strlen(str);
    for (int i = 0; i < len / 2; i++) {
        char temp = str[i];
        str[i] = str[len - 1 - i];
        str[len - 1 - i] = temp;
    }
}

static bool is_palindrome(const char *str) {
    int len = strlen(str);
    for (int i = 0; i < len / 2; i++) {
        if (str[i] != str[len - 1 - i]) {
            return false;
        }
    }
    return true;
}

static void decrement_string(char *str) {
    int len = strlen(str);
    int i = len - 1;
    while (i >= 0 && str[i] == '0') {
        str[i] = '9';
        i--;
    }
    if (i >= 0) {
        str[i]--;
    }
    if (str[0] == '0' && len > 1) {
        memmove(str, str + 1, len);
        str[len - 1] = '\0';
    }
}

static void make_palindrome_from_left(char *str) {
    int len = strlen(str);
    for (int i = 0; i < len / 2; i++) {
        str[len - 1 - i] = str[i];
    }
}

long long previous_palindrome(long long num) {
    if (num <= 0) {
        return 0;
    }
    if (num <= 10) {
        return num - 1;
    }

    char str[64];
    int written = snprintf(str, sizeof(str), "%lld", num);
    if (written < 0 || written >= (int)sizeof(str)) {
        return 0;
    }
    
    if (is_palindrome(str)) {
        decrement_string(str);
    }

    int len = strlen(str);
    char original[64];
    if (len >= (int)sizeof(original)) {
        return 0;
    }
    memcpy(original, str, len + 1);

    make_palindrome_from_left(str);
    
    if (strcmp(str, original) >= 0) {
        int mid = (len - 1) / 2;
        int i = mid;
        while (i >= 0 && str[i] == '0') {
            str[i] = '9';
            if (len - 1 - i != i) {
                str[len - 1 - i] = '9';
            }
            i--;
        }
        if (i >= 0) {
            str[i]--;
            if (len - 1 - i != i) {
                str[len - 1 - i] = str[i];
            }
        } else {
            len--;
            if (len < 0) len = 0;
            if (len >= (int)sizeof(str)) {
                return 0;
            }
            memset(str, '9', len);
            str[len] = '\0';
        }
        
        if (str[0] == '0' && len > 1) {
            memmove(str, str + 1, len);
            str[len - 1] = '\0';
        }
    }

    return atoll(str);
}

int main(void) {
    long long test_cases[] = {100, 99, 12345, 123456, 99999, 100000, 1, 11, 10001, 12321};
    int num_tests = sizeof(test_cases) / sizeof(test_cases[0]);

    for (int i = 0; i < num_tests; i++) {
        long long result = previous_palindrome(test_cases[i]);
        printf("Previous palindrome of %lld is %lld\n", test_cases[i], result);
    }

    return 0;
}
