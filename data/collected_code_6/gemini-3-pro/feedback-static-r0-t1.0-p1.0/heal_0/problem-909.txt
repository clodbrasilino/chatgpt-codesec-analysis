#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int is_palindrome(long long num) {
    long long reversed = 0;
    long long original = num;
    
    if (num < 0) {
        return 0;
    }
    
    while (num > 0) {
        long long remainder = num % 10;
        reversed = reversed * 10 + remainder;
        num /= 10;
    }
    
    return original == reversed;
}

long long find_previous_palindrome(long long num) {
    if (num <= 0) {
        return -1;
    }
    
    num--;
    
    while (num >= 0) {
        if (is_palindrome(num)) {
            return num;
        }
        num--;
    }
    
    return -1;
}

int main() {
    long long num;
    long long prev_palindrome;

    if (scanf("%lld", &num) != 1) {
        return EXIT_FAILURE;
    }

    prev_palindrome = find_previous_palindrome(num);

    if (prev_palindrome != -1) {
        printf("%lld\n", prev_palindrome);
    } else {
        printf("-1\n");
    }

    return EXIT_SUCCESS;
}
