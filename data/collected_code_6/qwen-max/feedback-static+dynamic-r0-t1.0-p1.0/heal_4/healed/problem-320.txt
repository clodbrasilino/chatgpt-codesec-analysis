#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

bool is_safe_add(long long a, long long b) {
    if (b > 0 && a > LLONG_MAX - b) return false;
    if (b < 0 && a < LLONG_MIN - b) return false;
    return true;
}

bool is_safe_mult(long long a, long long b) {
    if (a == 0 || b == 0) return true;
    if (a > 0) {
        if (b > 0) {
            if (a > LLONG_MAX / b) return false;
        } else {
            if (b == LLONG_MIN && a > -1) return false;
            if (a > (LLONG_MIN - (b > LLONG_MIN ? 1 : 0)) / b) return false;
        }
    } else {
        if (b > 0) {
            if (a < LLONG_MIN / b) return false;
        } else {
            if (a < LLONG_MAX / b) return false;
        }
    }
    return true;
}

long long square_sum(long long n) {
    if (!is_safe_add(n, 1)) return 0;
    if (!is_safe_mult(n, n + 1)) return 0;
    long long term = n * (n + 1) / 2;
    if (!is_safe_mult(term, term)) return 0;
    return term * term;
}

long long sum_of_squares(long long n) {
    if (!is_safe_add(n, 1)) return 0;
    if (!is_safe_mult(n, n + 1)) return 0;
    if (!is_safe_add(2 * n, 1)) return 0;
    if (!is_safe_mult(2 * n, 2 * n + 1)) return 0;
    if (!is_safe_mult(n, (n + 1) * (2 * n + 1))) return 0;
    return n * (n + 1) * (2 * n + 1) / 6;
}

long long difference_between_sums(long long n) {
    long long ss = square_sum(n);
    long long sos = sum_of_squares(n);
    if (!is_safe_add(ss, -sos)) return 0;
    return ss - sos;
}

int main() {
    long long n;
    printf("Enter a number: ");
    if (scanf("%lld", &n) != 1) return 1;
    printf("Difference: %lld\n", difference_between_sums(n));
    return 0;
}
