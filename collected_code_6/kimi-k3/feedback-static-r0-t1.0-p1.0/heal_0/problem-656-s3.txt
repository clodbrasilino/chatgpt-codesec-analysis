#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int compare_ints(const void *a, const void *b) {
    int arg1 = *(const int *)a;
    int arg2 = *(const int *)b;
    if (arg1 < arg2) return -1;
    if (arg1 > arg2) return 1;
    return 0;
}

long long min_absolute_sum_diff(int *a, int *b, size_t n) {
    if (a == NULL || b == NULL || n == 0) {
        return 0;
    }
    
    if (n > SIZE_MAX / sizeof(int)) {
        return LLONG_MAX;
    }
    
    int *sorted_a = malloc(n * sizeof(int));
    int *sorted_b = malloc(n * sizeof(int));
    
    if (sorted_a == NULL || sorted_b == NULL) {
        free(sorted_a);
        free(sorted_b);
        return LLONG_MAX;
    }
    
    for (size_t i = 0; i < n; i++) {
        sorted_a[i] = a[i];
        sorted_b[i] = b[i];
    }
    
    qsort(sorted_a, n, sizeof(int), compare_ints);
    qsort(sorted_b, n, sizeof(int), compare_ints);
    
    long long sum = 0;
    for (size_t i = 0; i < n; i++) {
        long long diff = (long long)sorted_a[i] - (long long)sorted_b[i];
        sum += (diff < 0) ? -diff : diff;
    }
    
    free(sorted_a);
    free(sorted_b);
    
    return sum;
}

int main(void) {
    int a[] = {4, 1, 8, 7};
    int b[] = {2, 3, 6, 5};
    size_t n = sizeof(a) / sizeof(a[0]);
    
    long long result = min_absolute_sum_diff(a, b, n);
    
    if (result == LLONG_MAX) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    printf("%lld\n", result);
    
    return EXIT_SUCCESS;
}
