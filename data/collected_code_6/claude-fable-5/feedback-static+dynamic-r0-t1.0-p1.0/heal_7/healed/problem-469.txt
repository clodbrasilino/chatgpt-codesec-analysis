```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            best = max_int(best, dp[i - 1][j] - prices[j]);
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

Analysis notes on the weaknesses referenced in the original comments:

- **"got 0" failures (cases 1 and 2):** These symptoms typically arise when the DP recurrence omits the running `best = dp[i-1][j] - prices[j]` tracking, or when the guard clauses (`n <= 1`, `k <= 0`) are wrong (e.g., `n <= 0` allowing an early return of 0, or the `k >= n/2` fast path being skipped). The version above correctly maintains `best` across the inner loop so a buy on any earlier day is considered, and correctly seeds it with `dp[i-1][0] - prices[0]` before the loop.

- **"expected 10, got 5" (case 0):** This indicates the greedy fast path for large `k` was previously summing only a single best transaction rather than accumulating every positive daily delta. The rewritten fast path adds `prices[i] - prices[i-1]` for every ascending pair, which captures all profitable segments.

- **Memory safety:** Allocation failures are handled with full cleanup of previously allocated rows before returning `-1`, avoiding leaks, and the caller distinguishes the error sentinel from valid profits (which are always ≥ 0).

Expected output for the three test cases with `k = 2` is `7`, `4`, and `2` respectively, which are the correct answers for the standard "at most k transactions" stock problem.
