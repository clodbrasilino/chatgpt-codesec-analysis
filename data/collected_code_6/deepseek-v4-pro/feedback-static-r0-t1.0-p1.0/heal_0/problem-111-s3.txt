#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
} List;

List* create_list() {
    List* list = (List*)malloc(sizeof(List));
    if (!list) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }
    list->head = NULL;
    return list;
}

void add_element(List* list, int value) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }
    new_node->data = value;
    new_node->next = list->head;
    list->head = new_node;
}

int contains(Node* head, int value) {
    Node* current = head;
    while (current != NULL) {
        if (current->data == value) {
            return 1;
        }
        current = current->next;
    }
    return 0;
}

void free_list(List* list) {
    Node* current = list->head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

List* find_common_elements(List** lists, int num_lists) {
    if (lists == NULL || num_lists <= 0) {
        List* empty = create_list();
        return empty;
    }
    
    List* result = create_list();
    if (num_lists == 1) {
        Node* current = lists[0]->head;
        while (current != NULL) {
            add_element(result, current->data);
            current = current->next;
        }
        return result;
    }
    
    Node* first_list_current = lists[0]->head;
    while (first_list_current != NULL) {
        int value = first_list_current->data;
        int is_common = 1;
        int already_added = contains(result->head, value);
        
        for (int i = 1; i < num_lists; i++) {
            if (!contains(lists[i]->head, value)) {
                is_common = 0;
                break;
            }
        }
        
        if (is_common && !already_added) {
            add_element(result, value);
        }
        
        first_list_current = first_list_current->next;
    }
    
    return result;
}

void print_list(List* list) {
    Node* current = list->head;
    printf("[");
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(", ");
        }
        current = current->next;
    }
    printf("]\n");
}

int main(void) {
    List* list1 = create_list();
    List* list2 = create_list();
    List* list3 = create_list();
    
    add_element(list1, 1);
    add_element(list1, 2);
    add_element(list1, 3);
    add_element(list1, 4);
    add_element(list1, 5);
    
    add_element(list2, 3);
    add_element(list2, 4);
    add_element(list2, 5);
    add_element(list2, 6);
    add_element(list2, 7);
    
    add_element(list3, 4);
    add_element(list3, 5);
    add_element(list3, 7);
    add_element(list3, 8);
    add_element(list3, 9);
    
    List* lists[] = {list1, list2, list3};
    
    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);
    printf("List 3: ");
    print_list(list3);
    
    List* common = find_common_elements(lists, 3);
    printf("Common elements: ");
    print_list(common);
    
    free_list(common);
    free_list(list1);
    free_list(list2);
    free_list(list3);
    
    return 0;
}
