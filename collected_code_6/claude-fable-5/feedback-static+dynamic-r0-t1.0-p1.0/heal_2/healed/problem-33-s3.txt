```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (decimal > 0 && index < buffer_size - 1) {
        buffer[index] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
        index++;
    }

    if (decimal > 0) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[index] = '\0';

    for (i = 0; i < index / 2; i++) {
        char temp = buffer[i];
        buffer[i] = buffer[index - 1 - i];
        buffer[index - 1 - i] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[MAX_BITS + 1] = {0};
    int scan_result;

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &number);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal: %u\n", number);
    printf("Binary: %s\n", binary);

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-written from a security standpoint. The Flawfinder warning about the statically-sized array (`CWE-119`/`CWE-120`) is a false positive in this case, and here is why:

1. **Buffer sizing is correct**: `binary[MAX_BITS + 1]` is computed from `sizeof(unsigned int) * CHAR_BIT`, guaranteeing enough space for every bit of any `unsigned int` plus the null terminator, regardless of platform.

2. **Bounds checking is enforced**: The conversion loop condition `index < buffer_size - 1` prevents writing past the buffer, and the caller passes `sizeof(binary)` rather than a hardcoded size, so the function cannot overflow even if the buffer size changes.

3. **Overflow is detected, not ignored**: If the buffer were too small, the residual check `if (decimal > 0)` returns an error instead of truncating silently.

4. **Input validation**: `scanf("%u", ...)` return value is checked, and `%u` with a bounded integer target carries no buffer-overflow risk (unlike `%s`).

The only defensive improvement I added is null-terminating the buffer (`buffer[0] = '\0'`) on the error path where conversion fails partway, ensuring the caller never receives a non-terminated string even if it ignores the return code.
