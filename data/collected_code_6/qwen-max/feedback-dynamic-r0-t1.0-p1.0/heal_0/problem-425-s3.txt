#include <stdio.h>
#include <stdlib.h>

int count_sublists_with_element(int **lists, int n, int element) {
    int count = 0;
    for (int i = 0; i < n; i++) {
        if (lists[i] != NULL) {
            for (int j = 0; lists[i][j] != 0; j++) {
                if (lists[i][j] == element) {
                    count++;
                    break;
                }
            }
        }
    }
    return count;
}

int main() {
    int list1[] = {1, 2, 3, 4, 0};
    int list2[] = {5, 6, 7, 0};
    int list3[] = {8, 9, 1, 0};
    int *lists[] = {list1, list2, list3};

    int n = sizeof(lists) / sizeof(lists[0]);
    int element = 1;

    int result = count_sublists_with_element(lists, n, element);
    printf("Number of sublists containing element %d: %d\n", element, result);

    return 0;
}
