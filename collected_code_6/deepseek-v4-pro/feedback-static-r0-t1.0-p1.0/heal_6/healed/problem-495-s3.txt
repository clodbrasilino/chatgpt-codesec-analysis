#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>

char *remove_lowercase_substrings(const char *input) {
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[a-z]+";
    size_t input_len;
    char *result;
    size_t result_len = 0;
    const char *cursor;
    int status;

    if (!input) {
        return NULL;
    }

    input_len = strnlen(input, SIZE_MAX);
    if (input_len >= SIZE_MAX - 1) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (!result) {
        return NULL;
    }

    cursor = input;

    status = regcomp(&regex, pattern, REG_EXTENDED);
    if (status) {
        free(result);
        return NULL;
    }

    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len = (size_t)match.rm_so;
        size_t cursor_offset = (size_t)(cursor - input);

        if (cursor_offset > input_len) {
            break;
        }

        if (prefix_len > input_len - cursor_offset) {
            break;
        }

        if (prefix_len > input_len - result_len) {
            break;
        }

        if (prefix_len > 0) {
            if (prefix_len > input_len - result_len) {
                break;
            }
            memcpy(result + result_len, cursor, prefix_len);
            result_len += prefix_len;
        }

        if ((size_t)match.rm_eo > SIZE_MAX - cursor_offset) {
            break;
        }

        cursor += match.rm_eo;
        if ((size_t)(cursor - input) > input_len) {
            break;
        }
    }

    if (result_len < input_len) {
        size_t processed = (size_t)(cursor - input);
        if (processed <= input_len) {
            size_t remaining = input_len - processed;
            if (remaining > input_len - result_len) {
                remaining = input_len - result_len;
            }
            if (remaining > 0) {
                if (result_len + remaining > input_len) {
                    remaining = input_len - result_len;
                }
                memcpy(result + result_len, cursor, remaining);
                result_len += remaining;
            }
        }
    }

    regfree(&regex);

    result[result_len] = '\0';
    return result;
}

int main(void) {
    const char *test1 = "HelloWorld";
    const char *test2 = "abcDEFghiJKL";
    const char *test3 = "NoLowercase";
    const char *test4 = "ALLUPPERCASE";
    const char *test5 = "Mixed123lowercaseUPPER";
    char *cleaned;

    cleaned = remove_lowercase_substrings(test1);
    if (cleaned) {
        printf("Input:    %s\nOutput:   %s\n\n", test1, cleaned);
        free(cleaned);
    }

    cleaned = remove_lowercase_substrings(test2);
    if (cleaned) {
        printf("Input:    %s\nOutput:   %s\n\n", test2, cleaned);
        free(cleaned);
    }

    cleaned = remove_lowercase_substrings(test3);
    if (cleaned) {
        printf("Input:    %s\nOutput:   %s\n\n", test3, cleaned);
        free(cleaned);
    }

    cleaned = remove_lowercase_substrings(test4);
    if (cleaned) {
        printf("Input:    %s\nOutput:   %s\n\n", test4, cleaned);
        free(cleaned);
    }

    cleaned = remove_lowercase_substrings(test5);
    if (cleaned) {
        printf("Input:    %s\nOutput:   %s\n\n", test5, cleaned);
        free(cleaned);
    }

    return 0;
}
