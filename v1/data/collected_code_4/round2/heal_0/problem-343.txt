```C
#include <ctype.h>
#include <stdio.h>

void count_digits_letters(char *str, int *num_digits, int *num_letters) {
    *num_digits = 0;
    *num_letters = 0;

    for(int i = 0; str[i]; i++)
    {
        if(isdigit(str[i]))
            (*num_digits)++;
        else if(isalpha(str[i]))
            (*num_letters)++;
    }
}

int main() {
    char str[] = "Hello 1234";
    int num_digits, num_letters;

    count_digits_letters(str, &num_digits, &num_letters);

    printf("Digits: %d\n", num_digits);
    printf("Letters: %d\n", num_letters);

    return 0;
}
```
