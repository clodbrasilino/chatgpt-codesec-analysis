```c
#include <stdio.h>
#include <stdlib.h>

#define ROWS 4
#define COLS 3

static long row_sum(const int *row, size_t cols)
{
    long sum = 0;
    size_t j;

    for (j = 0; j < cols; j++) {
        sum += row[j];
    }
    return sum;
}

static void swap_rows(int matrix[][COLS], size_t a, size_t b)
{
    size_t j;

    for (j = 0; j < COLS; j++) {
        const int tmp = matrix[a][j];
        matrix[a][j] = matrix[b][j];
        matrix[b][j] = tmp;
    }
}

static void sort_matrix_by_row_sum(int matrix[][COLS], size_t rows)
{
    size_t i;
    size_t k;

    if (rows < 2U) {
        return;
    }

    for (i = 0; i < rows - 1U; i++) {
        size_t min_idx = i;
        for (k = i + 1U; k < rows; k++) {
            if (row_sum(matrix[k], COLS) < row_sum(matrix[min_idx], COLS)) {
                min_idx = k;
            }
        }
        if (min_idx != i) {
            swap_rows(matrix, i, min_idx);
        }
    }
}

static void print_matrix(const int matrix[][COLS], size_t rows)
{
    size_t i;
    size_t j;

    for (i = 0; i < rows; i++) {
        for (j = 0; j < COLS; j++) {
            if (printf("%d ", matrix[i][j]) < 0) {
                exit(EXIT_FAILURE);
            }
        }
        if (printf("\n") < 0) {
            exit(EXIT_FAILURE);
        }
    }
}

int main(void)
{
    int matrix[ROWS][COLS] = {
        {9, 8, 7},
        {1, 2, 3},
        {4, 5, 6},
        {0, 0, 1}
    };

    if (printf("Original matrix:\n") < 0) {
        return EXIT_FAILURE;
    }
    print_matrix(matrix, ROWS);

    sort_matrix_by_row_sum(matrix, ROWS);

    if (printf("Sorted matrix:\n") < 0) {
        return EXIT_FAILURE;
    }
    print_matrix(matrix, ROWS);

    return EXIT_SUCCESS;
}
```

The changes address both `variableScope` findings: `tmp` in `swap_rows` is now declared inside the loop body (and made `const` since it's assigned once per iteration), and `min_idx` in `sort_matrix_by_row_sum` is declared inside the outer loop where it's actually used. This limits each variable's lifetime to the smallest necessary scope, improving readability and reducing the chance of accidental reuse of stale values. All other behavior — selection sort by row sum, checked `printf` return values, and array-decay-safe function signatures — remains identical.
