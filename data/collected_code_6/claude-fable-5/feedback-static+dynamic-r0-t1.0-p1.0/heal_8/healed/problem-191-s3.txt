#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32U
#define FLUSH_GUARD_LIMIT 4096U

static int has_thirty_days(const char *month);
static void to_lower_string(char *dest, const char *src, size_t dest_size);

static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < FLUSH_GUARD_LIMIT);
}

static int has_thirty_days(const char *month)
{
    static const char *const thirty_day_months[] = {
        "april",
        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN] = {0};
    size_t i;
    size_t count;

    if (month == NULL) {
        return -1;
    }

    if (bounded_strlen(month, MAX_INPUT_LEN) >= MAX_INPUT_LEN) {
        return -1;
    }

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    for (i = 0U; i < count; i++) {
        if (strncmp(lowered, thirty_day_months[i], sizeof(lowered)) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;
    int result;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = bounded_strlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        flush_stdin();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "Empty input provided.\n");
        return EXIT_FAILURE;
    }

    result = has_thirty_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name provided.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`getchar` in a loop (CWE-120, CWE-20)** — The `flush_stdin` function reads into no buffer at all; it simply discards characters. The `guard` counter bounds the loop at 4096 iterations, and it terminates on `\n` or `EOF`, so there is no unbounded consumption or overflow risk. This is a false positive, mitigated defensively.

2. **Statically-sized `char lowered[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — All writes to `lowered` go through `to_lower_string`, which receives `sizeof(lowered)` and never writes past `dest_size - 1`, always null-terminating. The input length is also pre-validated with `bounded_strlen`. No overflow is possible.

3. **Statically-sized `char input[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits writes to the buffer size, followed by an explicit forced null terminator. Overlong lines are drained via `flush_stdin`, preventing stale data from leaking into subsequent reads.

All three Flawfinder hits are false positives in this implementation; the rewritten code preserves the existing mitigations (bounded copies, explicit null termination, length validation, guarded flush loop) with comments removed as requested.
