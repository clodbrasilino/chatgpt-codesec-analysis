#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

int binary_to_decimal(const char *binary, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, 128U);
    if (len == 0U || len >= 128U) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[128];
    unsigned long long decimal = 0ULL;
    size_t len;

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int ch;
        int truncated = 0;
        while ((ch = getchar()) != '\n' && ch != EOF) {
            truncated = 1;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}

1. **`strlen` over-read (CWE-126)** — Both occurrences of `strlen` were replaced with `strnlen`, which bounds the scan to the buffer size. In practice, the original code was already safe because `fgets` guarantees null-termination, but `strnlen` makes this defense explicit. An additional `input[sizeof(input) - 1U] = '\0';` guarantees termination regardless of input source. In `binary_to_decimal`, a length of `>= 128` is rejected, ensuring the string was properly terminated within the expected bound.

2. **Statically-sized array (CWE-119/CWE-120)** — The 128-byte buffer is safe because `fgets(input, sizeof(input), stdin)` limits the read to the buffer size, so no overflow is possible. The `sizeof` idiom keeps the size and the bound in sync.

3. **`getchar` loop (CWE-120, CWE-20)** — The loop only consumes and discards characters; nothing is written to a buffer, so no bounds issue exists. It correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on stream closure, and properly reports oversized input as an error.
