```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 256
#define MAX_DISCARD_ITERATIONS 65536

bool is_valid_password(const char *str, size_t required_length);
static void discard_remaining_input(void);
static void secure_clear(void *ptr, size_t len);

bool is_valid_password(const char *str, size_t required_length)
{
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    size_t length = 0;
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (required_length == 0 || required_length >= MAX_INPUT_LENGTH) {
        return false;
    }

    length = strnlen(str, MAX_INPUT_LENGTH);

    if (length >= MAX_INPUT_LENGTH) {
        return false;
    }

    if (length != required_length) {
        return false;
    }

    for (i = 0; i < length; i++) {
        unsigned char c = (unsigned char)str[i];

        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        }
    }

    return (has_upper && has_lower && has_digit);
}

static void discard_remaining_input(void)
{
    int ch;
    size_t count = 0;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < MAX_DISCARD_ITERATIONS);
}

static void secure_clear(void *ptr, size_t len)
{
    volatile unsigned char *p;
    size_t i;

    if (ptr == NULL || len == 0) {
        return;
    }

    p = (volatile unsigned char *)ptr;

    for (i = 0; i < len; i++) {
        p[i] = 0;
    }
}

int main(void)
{
    char buffer[MAX_INPUT_LENGTH];
    size_t required_length = 8;
    size_t input_length;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string to validate (required length %zu): ", required_length);

    if (fflush(stdout) != 0) {
        secure_clear(buffer, sizeof(buffer));
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        secure_clear(buffer, sizeof(buffer));
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    input_length = strnlen(buffer, sizeof(buffer));

    if (input_length > 0 && buffer[input_length - 1] == '\n') {
        buffer[input_length - 1] = '\0';
    } else if (input_length == sizeof(buffer) - 1) {
        discard_remaining_input();
    }

    if (is_valid_password(buffer, required_length)) {
        printf("The string is valid.\n");
    } else {
        printf("The string is invalid.\n");
    }

    secure_clear(buffer, sizeof(buffer));

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** This is a low-risk informational warning. The `discard_remaining_input()` function is safe because it never stores the read characters into a buffer — it only discards them. It also correctly terminates on `\n` and `EOF`, and the `MAX_DISCARD_ITERATIONS` cap prevents an unbounded loop on a hostile/pathological stream. Additionally, `ch` is correctly declared as `int` (not `char`), so the `EOF` comparison is reliable.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer usage is safe as written:
   - Input is read with `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces the bound and always NUL-terminates within the given size.
   - A defensive `buffer[sizeof(buffer) - 1] = '\0';` guarantees termination regardless.
   - Length inspection uses `strnlen()` with an explicit bound rather than `strlen()`.
   - Overflow input (line longer than the buffer) is detected and the excess is drained via `discard_remaining_input()`, preventing stale input from leaking into subsequent reads.

3. **Additional good practices already present:** `secure_clear()` uses a `volatile` pointer to prevent the compiler from optimizing away the wipe of sensitive password data on all exit paths; `isupper`/`islower`/`isdigit` receive values cast to `unsigned char`, avoiding undefined behavior for negative `char` values.

No functional changes were required — the flagged findings are false positives given the existing bounds checking, so the rewrite preserves the logic with the advisory comments removed.
