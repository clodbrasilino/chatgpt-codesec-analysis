#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U

long long count_substrings(const char *s, size_t max_len);
static int read_line(char *buf, size_t buf_size);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n == max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;
    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        long long idx;

        prefix_sum += (long long)(s[i] - '0') - 1;
        idx = prefix_sum + offset;
        if (idx < 0 || (size_t)idx >= table_size) {
            free(freq);
            return -1;
        }
        count += freq[idx];
        freq[idx]++;
    }

    free(freq);
    return count;
}

static int read_line(char *buf, size_t buf_size)
{
    size_t len;

    if (buf == NULL || buf_size < 2U || buf_size > (size_t)INT32_MAX) {
        return -1;
    }

    if (fgets(buf, (int)buf_size, stdin) == NULL) {
        return -1;
    }

    buf[buf_size - 1U] = '\0';

    len = strcspn(buf, "\r\n");
    if (buf[len] == '\0' && len == buf_size - 1U) {
        int c;
        do {
            c = getchar();
        } while (c != '\n' && c != EOF);
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    memset(buffer, 0, sizeof buffer);

    printf("Enter a digit string: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (read_line(buffer, sizeof buffer) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long.\n");
        return EXIT_FAILURE;
    }

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("Number of substrings: %lld\n", result);
    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code above is safe as written:

**Flawfinder finding 1 — `getchar` in a loop (CWE-120, CWE-20):** This is a false positive in this context. The `getchar()` loop in `read_line` never writes into a buffer; it only discards excess characters from `stdin` until a newline or `EOF` is reached. There is no buffer to overflow. The loop terminates correctly on both `\n` and `EOF`, so no unbounded behavior exists. No change required.

**Flawfinder finding 2 — statically-sized `char` array (CWE-119/CWE-120):** Also mitigated by the existing design:

- `read_line` receives `sizeof buffer` and passes it to `fgets`, which enforces the bound, so no write past the end is possible.
- The explicit `buf[buf_size - 1U] = '\0'` guarantees termination even in edge cases.
- The truncation-detection logic (`len == buf_size - 1U` with no newline found) correctly rejects over-long input rather than silently truncating it.
- `count_substrings` uses `strnlen` with the buffer capacity and rejects strings that fill the entire buffer, so it never scans out of bounds.

Additional robustness already present that is worth noting:

- Integer-overflow guards before computing `table_size` (`n > (SIZE_MAX - 1U) / 10U` and the multiplication check against `SIZE_MAX / sizeof *freq`) prevent wrap-around in the allocation size.
- `calloc` return value is checked, and `freq` is freed on every exit path — no leaks or NULL dereferences.
- The defensive bounds check on `idx` inside the loop is mathematically unreachable given the prefix-sum range (−n … 9n shifted by +n fits in `[0, 10n]`), but it is harmless hardening.
- `isdigit` is called with an `unsigned char` cast, avoiding undefined behavior for negative `char` values.

The code is already correct with respect to both reported findings; the rewrite preserves the safe structure verbatim with comments removed.
