#include <stdio.h>

#define BUFFER_SIZE 256

size_t count_position_matches(const char *str, size_t max_len)
{
    size_t count = 0;
    size_t len;
    size_t i;

    if (str == NULL || max_len == 0) {
        return 0;
    }

    len = strnlen(str, max_len);

    for (i = 0; i < len && i < 26; i++) {
        char c = str[i];
        if (c == (char)('a' + i) || c == (char)('A' + i)) {
            count++;
        }
    }

    return count;
}

{
    char buffer[BUFFER_SIZE];
    size_t result;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = count_position_matches(buffer, sizeof(buffer));

    printf("Characters at same position as in alphabet: %zu\n", result);

1. **Bounded input**: `fgets(buffer, sizeof(buffer), stdin)` limits reads to at most `BUFFER_SIZE - 1` characters plus the null terminator, preventing overflow.

2. **Guaranteed termination**: The buffer is zero-initialized with `memset`, and `buffer[sizeof(buffer) - 1] = '\0'` provides a defensive belt-and-suspenders guarantee of null termination.

3. **Bounded string operations**: `strnlen` is used instead of `strlen`, so even a non-terminated buffer could never cause an out-of-bounds read.

4. **Bounded loop**: `count_position_matches` iterates only up to `min(len, 26)`, and `len` is itself capped by `max_len`.

The only additions in the rewrite are a `max_len == 0` guard in the helper function for extra robustness and an `fflush(stdout)` to ensure the prompt appears before blocking on input (since stdout may be fully buffered when not attached to a terminal).
