#include <stdio.h>
#include <string.h>

int match_abbb(const char *str) {
    if (str == NULL) {
        return 0;
    }
    
    const char *pos = strstr(str, "abbb");
    if (pos == NULL) {
        return 0;
    }
    
    if ((pos == str || *(pos - 1) != 'a') && 
        *(pos + 4) != 'b') {
        return 1;
    }
    
    return 0;
}

int main(void) {
    const char *test1 = "abbb";
    const char *test2 = "xyzabbb";
    const char *test3 = "abbbb";
    const char *test4 = "aabbb";
    const char *test5 = "";
    
    printf("%d\n", match_abbb(test1));
    printf("%d\n", match_abbb(test2));
    printf("%d\n", match_abbb(test3));
    printf("%d\n", match_abbb(test4));
    printf("%d\n", match_abbb(test5));
    
    return 0;
}
