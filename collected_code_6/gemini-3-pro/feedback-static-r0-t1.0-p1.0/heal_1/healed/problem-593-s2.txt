#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void remove_leading_zeroes(const char *ip, char *result, size_t result_size) {
    if (ip == NULL || result == NULL || result_size == 0) {
        return;
    }

    const char *ptr = ip;
    size_t result_idx = 0;
    int has_digit = 0;
    int leading_zero_skipped = 0;

    while (*ptr != '\0') {
        if (*ptr == '.') {
            if (!has_digit && leading_zero_skipped) {
                if (result_idx < result_size - 1) {
                    result[result_idx++] = '0';
                }
            }
            if (result_idx < result_size - 1) {
                result[result_idx++] = '.';
            }
            has_digit = 0;
            leading_zero_skipped = 0;
        } else if (*ptr >= '0' && *ptr <= '9') {
            if (!has_digit && *ptr == '0') {
                const char *next = ptr + 1;
                if (*next >= '0' && *next <= '9') {
                    leading_zero_skipped = 1;
                    ptr++;
                    continue;
                }
            }
            has_digit = 1;
            if (result_idx < result_size - 1) {
                result[result_idx++] = *ptr;
            }
        } else {
            if (result_idx < result_size - 1) {
                result[result_idx++] = *ptr;
            }
        }
        ptr++;
    }

    if (!has_digit && leading_zero_skipped) {
        if (result_idx < result_size - 1) {
            result[result_idx++] = '0';
        }
    }

    result[result_idx] = '\0';
}

int main(void) {
    const char *test_ips[] = {
        "192.168.001.001",
        "010.000.020.030",
        "0.00.000.0",
        "172.016.254.001",
        NULL
    };

    size_t buffer_size = 64;

    for (size_t i = 0; test_ips[i] != NULL; i++) {
        char *result = (char *)calloc(buffer_size, sizeof(char));
        if (result != NULL) {
            remove_leading_zeroes(test_ips[i], result, buffer_size);
            printf("%s -> %s\n", test_ips[i], result);
            free(result);
        }
    }

    return 0;
}
