```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_BUFFER_SIZE 128U
#define MAX_BINARY_DIGITS (sizeof(unsigned long) * 8U)

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == sizeof(input) - 1U) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the original code and the fix applied:**

The Flawfinder warning flagged the statically-sized `char input[INPUT_BUFFER_SIZE]` array (CWE-119/CWE-120). The original code already mitigated the core overflow risk correctly:

1. **Bounded read** — `fgets(input, (int)sizeof(input), stdin)` limits input to the buffer size, preventing a classic `gets()`-style overflow.
2. **Guaranteed termination** — explicitly writing `input[sizeof(input) - 1U] = '\0'` ensures the string is always NUL-terminated even in edge cases.
3. **Bounded length calculation** — `strnlen()` with an explicit maximum instead of `strlen()` prevents reads past the buffer.
4. **Validated conversion** — `binary_to_decimal()` checks for `NULL` pointers, zero/oversized lengths, invalid characters, and shift overflow against `ULONG_MAX >> 1` before every left shift.

The rewrite adds one remaining hardening improvement: **truncation detection**. In the original, if the user entered more than 127 characters, `fgets` would silently truncate and the leftover characters would remain in `stdin`. The rewritten version detects the missing newline on a full buffer, drains the remaining input via `flush_stdin()`, and reports an explicit error instead of silently processing a truncated (and therefore incorrect) value.
