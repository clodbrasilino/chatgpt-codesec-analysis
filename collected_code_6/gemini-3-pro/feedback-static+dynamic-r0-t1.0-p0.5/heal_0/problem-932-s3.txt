#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void remove_duplicate_words(char **list, int *count) {
    if (list == NULL || count == NULL || *count <= 0) {
        return;
    }

    int new_count = 0;
    for (int i = 0; i < *count; i++) {
        if (list[i] == NULL) {
            continue;
        }

        int is_duplicate = 0;
        for (int j = 0; j < new_count; j++) {
            if (strcmp(list[i], list[j]) == 0) {
                is_duplicate = 1;
                break;
            }
        }

        if (!is_duplicate) {
            if (i != new_count) {
                list[new_count] = list[i];
            }
            new_count++;
        } else {
            free(list[i]);
            list[i] = NULL;
        }
    }
    *count = new_count;
}

int main(void) {
    int count = 5;
    char **list = malloc(count * sizeof(char *));
    if (list == NULL) {
        return 1;
    }

    list[0] = strdup("apple");
    list[1] = strdup("banana");
    list[2] = strdup("apple");
    list[3] = strdup("orange");
    list[4] = strdup("banana");

    for (int i = 0; i < count; i++) {
        if (list[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(list[j]);
            }
            free(list);
            return 1;
        }
    }

    remove_duplicate_words(list, &count);

    for (int i = 0; i < count; i++) {
        printf("%s\n", list[i]);
    }

    for (int i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);

    return 0;
}
