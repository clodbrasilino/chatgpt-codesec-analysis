#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void removeLowercaseSubstrings(char *str){   
    regex_t regex;
    int index = 0;

    if(regcomp(&regex, "[a-z]+", REG_EXTENDED) == 0){
        regmatch_t match;
        char *result = (char *)malloc(sizeof(char) * strlen(str) + 1);

        if(result){
            while(regexec(&regex, str, 1, &match, 0) == 0){
                for(int i = 0; i < match.rm_so; i++, index++){
                    result[index] = str[i];
                }
                str += match.rm_eo;
            }

            for(int i = 0; i < strlen(str) + 1; i++, index++){
                result[index] = str[i];
            }

            result[index] = '\0';
            strcpy(str, result);
            free(result);
        }

        regfree(&regex);
    }
}

int main(){
    char str[] = "Hello, I'm a LowerCase STRING";
    removeLowercaseSubstrings(str);
    printf("%s\n", str);
    return 0;
}
