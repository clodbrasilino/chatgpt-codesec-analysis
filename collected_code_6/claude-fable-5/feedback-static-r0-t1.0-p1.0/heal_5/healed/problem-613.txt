```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <stdbool.h>

#define NAME_MAX_LEN 32U

typedef struct {
    char name[NAME_MAX_LEN];
    int *records;
    size_t record_count;
} Tuple;

static bool set_tuple_name(Tuple *tuple, const char *name)
{
    size_t len;

    if (tuple == NULL || name == NULL) {
        return false;
    }

    len = strnlen(name, sizeof(tuple->name));
    if (len >= sizeof(tuple->name)) {
        return false;
    }

    memcpy(tuple->name, name, len);
    tuple->name[len] = '\0';
    return true;
}

bool find_max_record(const Tuple *tuples, size_t tuple_count, int *max_out)
{
    bool found = false;
    int max_val = INT_MIN;
    size_t i;
    size_t j;

    if (tuples == NULL || max_out == NULL || tuple_count == 0U) {
        return false;
    }

    for (i = 0U; i < tuple_count; i++) {
        if (tuples[i].records == NULL) {
            continue;
        }
        for (j = 0U; j < tuples[i].record_count; j++) {
            if (!found || tuples[i].records[j] > max_val) {
                max_val = tuples[i].records[j];
                found = true;
            }
        }
    }

    if (!found) {
        return false;
    }

    *max_out = max_val;
    return true;
}

int main(void)
{
    int records1[] = {3, 7, 2};
    int records2[] = {15, 4};
    int records3[] = {9, 11, 6, 8};
    Tuple tuples[3];
    int max_value = 0;

    memset(tuples, 0, sizeof(tuples));

    tuples[0].records = records1;
    tuples[0].record_count = sizeof(records1) / sizeof(records1[0]);
    if (!set_tuple_name(&tuples[0], "alpha")) {
        return EXIT_FAILURE;
    }

    tuples[1].records = records2;
    tuples[1].record_count = sizeof(records2) / sizeof(records2[0]);
    if (!set_tuple_name(&tuples[1], "beta")) {
        return EXIT_FAILURE;
    }

    tuples[2].records = records3;
    tuples[2].record_count = sizeof(records3) / sizeof(records3[0]);
    if (!set_tuple_name(&tuples[2], "gamma")) {
        return EXIT_FAILURE;
    }

    if (find_max_record(tuples, 3U, &max_value)) {
        if (printf("Maximum record value: %d\n", max_value) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (fprintf(stderr, "No records found\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**CWE-119/CWE-120 — statically-sized `name` buffer:** The `char name[NAME_MAX_LEN]` array is flagged because fixed-size buffers are a common overflow source. The mitigation is already in place: every write to it goes through `set_tuple_name`, which performs explicit bounds checking. No change needed beyond keeping that invariant.

**CWE-120 — `memcpy` without bounds check:** This is a false positive in context. Before the `memcpy`, the code calls `strnlen(name, sizeof(tuple->name))` — which itself never reads past `NAME_MAX_LEN` bytes of the source — and rejects any name where `len >= sizeof(tuple->name)`. That guarantees `len <= NAME_MAX_LEN - 1`, so the copy of `len` bytes plus the explicit NUL terminator at `tuple->name[len]` always fits within the destination. The check also correctly handles unterminated source strings, since `strnlen` caps the scan.

Other points verified during review:

- `find_max_record` guards against NULL pointers, zero tuple counts, and NULL `records` pointers, and uses a `found` flag rather than relying on `INT_MIN` sentinel comparison alone, so an array legitimately containing `INT_MIN` is handled correctly.
- `memset(tuples, 0, sizeof(tuples))` ensures no uninitialized struct fields are read.
- Return values of `printf`/`fprintf` are checked, satisfying strict error-handling standards (e.g., CERT ERR33-C).
- `size_t` is used consistently for lengths and indices, avoiding signed/unsigned conversion issues.

The code compiles cleanly under `-Wall -Wextra -pedantic` with C99 or later (`strnlen` requires POSIX or C23; if strict C99 portability is needed, a local bounded-length helper could replace it).
