```C
#include <stdio.h>

int check_parity(int num) {
    int result = num ^ (num >> 1);
    result ^= result >> 2;
    result ^= result >> 4;
    result ^= result >> 8;
    result ^= result >> 16;
    return result & 1;
}

int main() {
    int num;
    printf("Enter a number: ");
    // Checking if input is a number
    if (scanf("%d", &num) == 1){
        if (check_parity(num)) {
            printf("Number has odd parity.\n");
        } else {
            printf("Number has even parity.\n");
        }
    } else {
        printf("Invalid input.\n");
    }
    return 0;
}
```
