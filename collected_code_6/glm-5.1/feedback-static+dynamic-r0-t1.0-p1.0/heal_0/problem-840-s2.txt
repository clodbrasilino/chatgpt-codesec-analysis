#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int are_roots_opposite_sign(double a, double b, double c) {
    if (a == 0.0) {
        return 0;
    }
    
    double discriminant = b * b - 4.0 * a * c;
    if (discriminant < 0.0) {
        return 0;
    }
    
    if (c == 0.0) {
        if (b == 0.0) {
            return 0;
        }
        return 1;
    }
    
    double abs_a = fabs(a);
    double abs_c = fabs(c);
    double max_abs = abs_a > abs_c ? abs_a : abs_c;
    double tolerance = max_abs * 1e-9;
    
    if (fabs(a + c * (a / fabs(a)) * (c / fabs(c))) <= tolerance) {
        return 1;
    }
    
    return 0;
}

int main() {
    double a, b, c;
    
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return EXIT_FAILURE;
    }
    
    int result = are_roots_opposite_sign(a, b, c);
    printf("%d\n", result);
    
    return EXIT_SUCCESS;
}
