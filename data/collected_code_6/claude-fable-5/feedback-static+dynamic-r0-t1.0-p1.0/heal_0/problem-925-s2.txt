#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static int product_of_tuple(const long *tuple, size_t count, long *result)
{
    size_t i;
    long product;

    if (tuple == NULL || result == NULL || count == 0U) {
        return -1;
    }

    product = 1L;

    for (i = 0U; i < count; i++) {
        long value = tuple[i];

        if (value != 0L) {
            if (product > 0L) {
                if (value > 0L) {
                    if (product > LONG_MAX / value) {
                        return -2;
                    }
                } else {
                    if (value < LONG_MIN / product) {
                        return -2;
                    }
                }
            } else if (product < 0L) {
                if (value > 0L) {
                    if (product < LONG_MIN / value) {
                        return -2;
                    }
                } else {
                    if (product < LONG_MAX / value) {
                        return -2;
                    }
                }
            }
        }

        product *= value;
    }

    *result = product;
    return 0;
}

int main(void)
{
    long tuple[] = {2L, 3L, 4L, 5L};
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    long result = 0L;
    int status;

    status = product_of_tuple(tuple, count, &result);

    if (status == 0) {
        if (printf("Product: %ld\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else if (status == -2) {
        if (fprintf(stderr, "Error: overflow detected\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    } else {
        if (fprintf(stderr, "Error: invalid input\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
