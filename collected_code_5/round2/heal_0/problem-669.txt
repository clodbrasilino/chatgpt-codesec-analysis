#include <stdio.h>
#include <string.h>
#include <regex.h>

#define MAX_IP_LENGTH 16

int isValidIp(char* ipAddress) {
    regex_t regex;
    int ret;
    char pattern[] = "^([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])$";

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if(ret) {
        printf("Compilation of regex failed\n");
        return 0;
    }
  
    ret = regexec(&regex, ipAddress, 0, NULL, 0);
    regfree(&regex);
    if(!ret) {
        return 1;
    }
    else if(ret == REG_NOMATCH) {
        return 0;
    }
    else {
        printf("Regex match failed\n");
        return 0;
    }
}

int main() {
    char ipAddress[MAX_IP_LENGTH];
    printf("Enter an IP address: ");
    fgets(ipAddress, MAX_IP_LENGTH, stdin);
    ipAddress[strcspn(ipAddress, "\n")] = 0;

    if(isValidIp(ipAddress)) {
        printf("The IP address is valid.\n");
    } else {
        printf("The IP address is not valid.\n");
    }

    return 0;
}
