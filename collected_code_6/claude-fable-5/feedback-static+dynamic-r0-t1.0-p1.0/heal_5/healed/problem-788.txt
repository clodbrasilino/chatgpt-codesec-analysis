```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;
    size_t list_bytes;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    list_bytes = list_size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(len + 1);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->str, str, len);
    t->str[len] = '\0';

    if (list_size > 0) {
        t->list = malloc(list_bytes);
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_bytes);
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->str == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *s = "WEB";
    int values[] = {1, 2};
    Tuple *t;

    t = create_tuple(s, values, sizeof(values) / sizeof(values[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) are **false positives** in this code, and no functional changes were required. Here's why each is already safe:

1. **String copy (`memcpy(t->str, str, len)`):** The destination buffer is allocated with `malloc(len + 1)` immediately before the copy, using the exact same `len` value obtained from `strnlen(str, MAX_STR_LEN)`. The bounded `strnlen` prevents reading past unterminated input, the `len >= MAX_STR_LEN` check rejects oversized strings, and null-termination is written explicitly afterward. The destination is guaranteed to hold exactly the source data plus the terminator.

2. **List copy (`memcpy(t->list, list, list_bytes)`):** The destination is allocated with `malloc(list_bytes)` using the same byte count as the copy. The `list_size > SIZE_MAX / sizeof(int)` guard prevents integer overflow in the `list_size * sizeof(int)` multiplication, which is the classic vulnerability pattern in this kind of code (CWE-190 leading to a heap overflow).

Other correctness properties preserved:

- All `malloc` return values are checked, with proper cleanup of previously allocated memory on each failure path (no leaks on error).
- `free_tuple` tolerates `NULL` input and relies on `free(NULL)` being a no-op for the `list` member when `list_size == 0`.
- `create_tuple` validates that a non-zero `list_size` is never paired with a `NULL` list pointer.
- `print_tuple` guards against `NULL` tuple and string pointers before dereferencing.

The code is defensively written and the Flawfinder findings are advisory pattern matches rather than actual vulnerabilities, so the rewrite is functionally identical with the audit comments removed.
