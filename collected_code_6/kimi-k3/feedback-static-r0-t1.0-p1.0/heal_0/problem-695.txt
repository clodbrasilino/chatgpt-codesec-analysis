```c
#include <stdio.h>
#include <stdbool.h>

bool check_elements_greater(const int tuple1[], const int tuple2[], size_t size) {
    if (tuple1 == NULL || tuple2 == NULL || size == 0) {
        return false;
    }
    
    for (size_t i = 0; i < size; i++) {
        if (tuple2[i] <= tuple1[i]) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int tuple1[] = {1, 2, 3, 4, 5};
    int tuple2[] = {2, 3, 4, 5, 6};
    size_t size = sizeof(tuple1) / sizeof(tuple1[0]);
    
    if (check_elements_greater(tuple1, tuple2, size)) {
        printf("All elements in tuple2 are greater than corresponding elements in tuple1\n");
    } else {
        printf("Not all elements in tuple2 are greater than corresponding elements in tuple1\n");
    }
    
    return 0;
}
```
