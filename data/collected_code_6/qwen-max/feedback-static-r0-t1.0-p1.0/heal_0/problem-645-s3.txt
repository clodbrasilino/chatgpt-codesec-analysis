```c
#include <stdio.h>
#include <stdlib.h>

int product_of_kth(int tuples[][2], int len, int k) {
    int product = 1;
    for (int i = 0; i < len; ++i) {
        if (k >= 0 && k < 2)
            product *= tuples[i][k];
    }
    return product;
}

int main() {
    int tuples[3][2] = {{1, 2}, {3, 4}, {5, 6}};
    int k = 1;
    int result = product_of_kth(tuples, 3, k);
    printf("Product of %dth index: %d\n", k, result);
    return 0;
}
```
