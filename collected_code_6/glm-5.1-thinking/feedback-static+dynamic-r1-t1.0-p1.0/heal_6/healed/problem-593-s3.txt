#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

int remove_ip_leading_zeroes(const char *src, size_t src_len, char *dest, size_t dest_size) {
    if (src == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }

    size_t j = 0;
    int at_octet_start = 1;
    size_t i;

    for (i = 0; i < src_len; ++i) {
        if (at_octet_start && src[i] == '0' && (i + 1 < src_len) && isdigit((unsigned char)src[i + 1])) {
            continue;
        }

        if (j >= dest_size - 1) {
            break;
        }

        dest[j++] = src[i];
        at_octet_start = (src[i] == '.');
    }

    dest[j] = '\0';

    if (i < src_len) {
        return -1;
    }

    return 0;
}

int main(void) {
    const char *addresses[] = {
        "192.168.001.010",
        "010.000.000.001",
        "0.0.0.0",
        "255.255.255.255",
        "001.002.003.004"
    };
    size_t count = sizeof(addresses) / sizeof(addresses[0]);

    for (size_t i = 0; i < count; ++i) {
        size_t src_len = strnlen(addresses[i], 16);
        size_t buffer_size = src_len + 1;
        char *buffer = malloc(buffer_size);
        
        if (buffer == NULL) {
            printf("Error processing: %s\n", addresses[i]);
            continue;
        }
        
        if (remove_ip_leading_zeroes(addresses[i], src_len, buffer, buffer_size) == 0) {
            printf("%s -> %s\n", addresses[i], buffer);
        } else {
            printf("Error processing: %s\n", addresses[i]);
        }
        
        free(buffer);
    }

    return 0;
}
