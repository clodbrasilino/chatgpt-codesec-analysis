```c
#include <stdio.h>

double calculateSurfaceArea(double side) {
    return 6 * side * side;
}

int main() {
    double side;
    printf("Enter the length of a side of the cube: ");
    if (scanf("%lf", &side) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    double surfaceArea = calculateSurfaceArea(side);
    printf("The surface area of the cube is: %f\n", surfaceArea);
    return 0;
}
```
