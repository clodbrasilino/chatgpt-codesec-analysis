```c
#include <stdio.h>
#include <stdint.h>

uint32_t toggle_odd_bits(uint32_t num) {
    return num ^ 0xAAAAAAAAu;
}

int main(void) {
    uint32_t num;
    int scan_result;
    
    printf("Enter a number: ");
    scan_result = scanf("%u", &num);
    
    if (scan_result != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    printf("Result: %u\n", toggle_odd_bits(num));
    
    return 0;
}
```
