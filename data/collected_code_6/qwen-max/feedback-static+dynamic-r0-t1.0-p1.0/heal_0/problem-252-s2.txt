```c
#include <stdio.h>
#include <math.h>

typedef struct {
    double real;
    double imag;
} Complex;

typedef struct {
    double magnitude;
    double angle;
} Polar;

Polar complexToPolar(Complex c) {
    Polar p;
    p.magnitude = sqrt(c.real * c.real + c.imag * c.imag);
    p.angle = atan2(c.imag, c.real);
    return p;
}

int main() {
    Complex c = {3.0, 4.0};
    Polar p = complexToPolar(c);
    printf("Magnitude: %f, Angle: %f\n", p.magnitude, p.angle);
    return 0;
}
```
