```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void increment_string_numeric_values(char *str, int k) {
    for (char *p = str; *p; ++p) {
        if (*p >= '0' && *p <= '9') {
            *p = ((*p - '0' + k) % 10) + '0';
        }
    }
}

int main() {
    char str[] = "abc123def456";
    int k = 2;
    increment_string_numeric_values(str, k);
    printf("%s\n", str);
    return 0;
}
```
