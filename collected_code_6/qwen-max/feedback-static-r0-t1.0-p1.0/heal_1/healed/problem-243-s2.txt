```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
} Tuple;

int compare(const void *a, const void *b) {
    const Tuple *tupleA = (const Tuple *)a;
    const Tuple *tupleB = (const Tuple *)b;
    return (tupleA->first - tupleB->first);
}

void sortTuplesByFirstElement(Tuple *tuples, size_t count) {
    qsort(tuples, count, sizeof(Tuple), compare);
}

int main() {
    Tuple tuples[] = {{3, 5}, {1, 2}, {3, 4}, {2, 9}, {1, 8}};
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    sortTuplesByFirstElement(tuples, count);

    for (size_t i = 0; i < count; ++i) {
        printf("(%d, %d) ", tuples[i].first, tuples[i].second);
    }

    return 0;
}
```
