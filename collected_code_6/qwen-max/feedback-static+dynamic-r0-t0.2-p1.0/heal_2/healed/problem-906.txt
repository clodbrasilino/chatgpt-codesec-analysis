#include <stdio.h>
#include <stdlib.h>
#include <regex.h>
#include <limits.h>

int valid_date(int year, int month, int day) {
    if (year < 2000 || year > 2100 || month < 1 || month > 12 || day < 1 || day > 31) {
        return 0;
    }
    if ((month == 4 || month == 6 || month == 9 || month == 11) && day == 31) {
        return 0;
    }
    if (month == 2) {
        if (day > 29 || (day == 29 && (year % 4 != 0 || (year % 100 == 0 && year % 400 != 0)))) {
            return 0;
        }
    }
    return 1;
}

void extract_date(const char *url, int *year, int *month, int *day) {
    regex_t regex;
    regmatch_t matches[4];
    const char *pattern = ".*\\/([0-9]{4})\\/([0-9]{2})\\/([0-9]{2}).*";

    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        return;
    }

    if (regexec(&regex, url, 4, matches, 0) == 0) {
        char str_year[5], str_month[3], str_day[3];
        size_t n;

        n = matches[1].rm_eo - matches[1].rm_so;
        for (size_t i = 0; i < (n < 4 ? n : 4); ++i) {
            str_year[i] = url[matches[1].rm_so + i];
        }
        str_year[n < 4 ? n : 4] = '\0';

        n = matches[2].rm_eo - matches[2].rm_so;
        for (size_t i = 0; i < (n < 2 ? n : 2); ++i) {
            str_month[i] = url[matches[2].rm_so + i];
        }
        str_month[n < 2 ? n : 2] = '\0';

        n = matches[3].rm_eo - matches[3].rm_so;
        for (size_t i = 0; i < (n < 2 ? n : 2); ++i) {
            str_day[i] = url[matches[3].rm_so + i];
        }
        str_day[n < 2 ? n : 2] = '\0';

        long y, m, d;
        y = strtol(str_year, NULL, 10);
        if (y < 2000 || y > 2100) *year = 0;
        else *year = (int)y;

        m = strtol(str_month, NULL, 10);
        if (m < 1 || m > 12) *month = 0;
        else *month = (int)m;

        d = strtol(str_day, NULL, 10);
        if (d < 1 || d > 31) *day = 0;
        else *day = (int)d;

        if (!valid_date(*year, *month, *day)) {
            *year = 0;
            *month = 0;
            *day = 0;
        }
    }

    regfree(&regex);
}

int main() {
    const char *url = "http://example.com/2023/10/05";
    int year, month, day;

    extract_date(url, &year, &month, &day);

    if (year && month && day) {
        printf("Year: %d, Month: %02d, Day: %02d\n", year, month, day);
    } else {
        printf("Date not found in URL.\n");
    }

    return 0;
}
