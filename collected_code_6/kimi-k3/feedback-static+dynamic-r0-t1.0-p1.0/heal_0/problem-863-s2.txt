#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;
    if (x < y) return -1;
    if (x > y) return 1;
    return 0;
}

int longestConsecutiveSubsequence(int arr[], int n)
{
    if (arr == NULL || n <= 0)
        return 0;

    qsort(arr, (size_t)n, sizeof(int), compare);

    int longest = 1;
    int current = 1;

    for (int i = 1; i < n; i++) {
        if (arr[i] == arr[i - 1])
            continue;
        else if (arr[i] == arr[i - 1] + 1)
            current++;
        else {
            if (current > longest)
                longest = current;
            current = 1;
        }
    }

    if (current > longest)
        longest = current;

    return longest;
}

int main(void)
{
    int arr[] = {1, 9, 3, 10, 4, 20, 2};
    int n = (int)(sizeof(arr) / sizeof(arr[0]));

    int result = longestConsecutiveSubsequence(arr, n);
    printf("Length of longest consecutive subsequence: %d\n", result);

    return 0;
}
