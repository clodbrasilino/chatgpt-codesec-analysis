```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size);

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_len = 0;
    size_t consumed = 0;
    int rc;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    if (strnlen(input, input_size) >= input_size) {
        return -1;
    }

    rc = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (consumed < input_size && *cursor != '\0') {
        rc = regexec(&regex, cursor, 1, &match, 0);
        if (rc == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_size - consumed);
            if (out_len >= output_size || remaining >= output_size - out_len) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_len, cursor, remaining);
            out_len += remaining;
            break;
        }
        if (rc != 0) {
            regfree(&regex);
            return -1;
        }
        if (match.rm_so > 0) {
            size_t prefix_len = (size_t)match.rm_so;
            if (out_len >= output_size || prefix_len >= output_size - out_len) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_len, cursor, prefix_len);
            out_len += prefix_len;
        }
        if (match.rm_eo <= 0) {
            regfree(&regex);
            return -1;
        }
        cursor += match.rm_eo;
        consumed += (size_t)match.rm_eo;
    }

    if (out_len >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[out_len] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char *test_strings[] = {
        "HelloWORLDfooBAR",
        "abcDEFghiJKL",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix123edCaseWith456numbers"
    };
    enum { MAX_INPUT_LEN = 256 };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        size_t input_len = strnlen(test_strings[i], MAX_INPUT_LEN);
        size_t buf_size;
        char *result;

        if (input_len >= MAX_INPUT_LEN) {
            fprintf(stderr, "Input string too long\n");
            return EXIT_FAILURE;
        }

        buf_size = input_len + 1;
        result = malloc(buf_size);
        if (result == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (remove_lowercase(test_strings[i], input_len + 1, result, buf_size) == 0) {
            printf("Input:  \"%s\"\n", test_strings[i]);
            printf("Output: \"%s\"\n\n", result);
        } else {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** All `strlen` calls were replaced with `strnlen` bounded by an explicit size. The `remove_lowercase` function now takes an `input_size` parameter and validates that the input is properly NUL-terminated within that bound before processing. In `main`, inputs are bounded by `MAX_INPUT_LEN`.

2. **CWE-120 (`memcpy` buffer overflow):** The bounds checks preceding each `memcpy` were rewritten to avoid potential integer overflow in expressions like `out_len + remaining >= output_size` — they now use the subtraction form (`remaining >= output_size - out_len`) after first verifying `out_len < output_size`, guaranteeing the destination can always hold the copied data.

3. **Defensive hardening:** Added a guard against a zero-length regex match (`match.rm_eo <= 0`) to prevent a potential infinite loop, tracked consumed input bytes so the cursor can never advance past the validated input region, and re-verified `out_len < output_size` before writing the final NUL terminator.
