#define BUFFER_SIZE 64

static int find_demlo(const char *input, size_t input_max, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, input_max);

    if (len == 0U || len >= input_max || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    i = len - 1U;

    while (i >= 1U)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
        i--;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;

        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, sizeof(input), result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;

The code was already well-defended against the flagged weaknesses, so the rewrite preserves the existing mitigations rather than restructuring them:

**CWE-119/CWE-120 (statically-sized buffers `input` and `result`):** Both buffers are zero-initialized with `memset` before use. Input is read exclusively via `fgets` with `sizeof(input)` as the limit, guaranteeing no write past the buffer. A forced null terminator at `input[sizeof(input) - 1U]` ensures the string is always terminated. Inside `find_demlo`, every write to `output` is guarded by `pos >= (output_size - 1U)` bounds checks, an upfront capacity check (`output_size < 2U * len`) rejects insufficient buffers before any writing begins, and `strnlen` (rather than `strlen`) caps length measurement at `input_max`.

**CWE-120/CWE-20 (`getchar` in a loop):** The `getchar` loop performs no buffer writes at all — it only discards excess characters from stdin until newline or EOF, so there is no boundary to violate. The loop correctly terminates on `EOF` (which is why `c` is declared `int`, not `char`), preventing an infinite loop on closed input streams.

Additional robustness already present and retained: `NULL`/zero-size parameter validation, rejection of unterminated input (`len >= input_max`), strict input validation (only `'1'` characters, length 1–9), and the `size_t` decrement loop is safe since `i` starts at `len - 1U` with `len >= 1U` guaranteed, so it never underflows.
