#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int x;
    int y;
} Tuple;

typedef struct {
    Tuple *data;
    size_t size;
} TupleArray;

bool are_tuples_equal(const Tuple *t1, const Tuple *t2) {
    if (t1 == NULL || t2 == NULL) {
        return false;
    }
    return (t1->x == t2->x) && (t1->y == t2->y);
}

int remove_matching_tuples(TupleArray *arr1, TupleArray *arr2) {
    if (arr1 == NULL || arr2 == NULL) {
        return -1;
    }
    if ((arr1->size > 0 && arr1->data == NULL) || (arr2->size > 0 && arr2->data == NULL)) {
        return -1;
    }
    if (arr1->size == 0 || arr2->size == 0) {
        return 0;
    }

    bool *keep1 = (bool *)malloc(arr1->size * sizeof(bool));
    if (keep1 == NULL) {
        return -1;
    }

    bool *keep2 = (bool *)malloc(arr2->size * sizeof(bool));
    if (keep2 == NULL) {
        free(keep1);
        return -1;
    }

    for (size_t i = 0; i < arr1->size; i++) {
        keep1[i] = true;
    }
    for (size_t i = 0; i < arr2->size; i++) {
        keep2[i] = true;
    }

    for (size_t i = 0; i < arr1->size; i++) {
        for (size_t j = 0; j < arr2->size; j++) {
            if (are_tuples_equal(&(arr1->data[i]), &(arr2->data[j]))) {
                keep1[i] = false;
                keep2[j] = false;
            }
        }
    }

    size_t new_size1 = 0;
    for (size_t i = 0; i < arr1->size; i++) {
        if (keep1[i]) {
            arr1->data[new_size1] = arr1->data[i];
            new_size1++;
        }
    }
    arr1->size = new_size1;

    size_t new_size2 = 0;
    for (size_t i = 0; i < arr2->size; i++) {
        if (keep2[i]) {
            arr2->data[new_size2] = arr2->data[i];
            new_size2++;
        }
    }
    arr2->size = new_size2;

    free(keep1);
    free(keep2);

    return 0;
}

TupleArray create_tuple_array(size_t size) {
    TupleArray arr;
    arr.size = size;
    if (size > 0) {
        arr.data = (Tuple *)malloc(size * sizeof(Tuple));
        if (arr.data == NULL) {
            arr.size = 0;
        }
    } else {
        arr.data = NULL;
    }
    return arr;
}

void free_tuple_array(TupleArray *arr) {
    if (arr != NULL) {
        if (arr->data != NULL) {
            free(arr->data);
            arr->data = NULL;
        }
        arr->size = 0;
    }
}

void print_tuple_array(const TupleArray *arr) {
    if (arr == NULL || arr->data == NULL) {
        printf("\n");
        return;
    }
    for (size_t i = 0; i < arr->size; i++) {
        printf("(%d, %d) ", arr->data[i].x, arr->data[i].y);
    }
    printf("\n");
}

int main(void) {
    TupleArray arr1 = create_tuple_array(4);
    if (arr1.data == NULL && arr1.size > 0) {
        return EXIT_FAILURE;
    }

    arr1.data[0] = (Tuple){1, 2};
    arr1.data[1] = (Tuple){3, 4};
    arr1.data[2] = (Tuple){5, 6};
    arr1.data[3] = (Tuple){7, 8};

    TupleArray arr2 = create_tuple_array(3);
    if (arr2.data == NULL && arr2.size > 0) {
        free_tuple_array(&arr1);
        return EXIT_FAILURE;
    }

    arr2.data[0] = (Tuple){9, 10};
    arr2.data[1] = (Tuple){3, 4};
    arr2.data[2] = (Tuple){1, 2};

    int result = remove_matching_tuples(&arr1, &arr2);
    
    if (result == 0) {
        print_tuple_array(&arr1);
        print_tuple_array(&arr2);
    }

    free_tuple_array(&arr1);
    free_tuple_array(&arr2);

    return result == 0 ? EXIT_SUCCESS : EXIT_FAILURE;
}
