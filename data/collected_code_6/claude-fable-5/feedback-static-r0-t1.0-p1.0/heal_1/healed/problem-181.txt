```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

char *longest_common_prefix(const char *const *strs, size_t count)
{
    char *prefix = NULL;
    size_t prefix_len = 0;
    size_t i;

    if (strs == NULL || count == 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        if (strs[i] == NULL) {
            return NULL;
        }
    }

    prefix_len = strnlen(strs[0], MAX_STR_LEN);
    if (prefix_len == MAX_STR_LEN) {
        return NULL;
    }

    for (i = 1; i < count; i++) {
        size_t j = 0;
        size_t cur_len = strnlen(strs[i], MAX_STR_LEN);
        if (cur_len == MAX_STR_LEN) {
            return NULL;
        }
        while (j < prefix_len && j < cur_len && strs[i][j] == strs[0][j]) {
            j++;
        }
        prefix_len = j;
        if (prefix_len == 0) {
            break;
        }
    }

    if (prefix_len >= MAX_STR_LEN) {
        return NULL;
    }

    prefix = malloc(prefix_len + 1);
    if (prefix == NULL) {
        return NULL;
    }

    memcpy(prefix, strs[0], prefix_len);
    prefix[prefix_len] = '\0';

    return prefix;
}

int main(void)
{
    const char *strings[] = {
        "flowering",
        "flower",
        "flow",
        "flourish"
    };
    size_t count = sizeof(strings) / sizeof(strings[0]);
    char *result;

    result = longest_common_prefix(strings, count);
    if (result == NULL) {
        fprintf(stderr, "Failed to compute longest common prefix\n");
        return EXIT_FAILURE;
    }

    if (printf("Longest common prefix: \"%s\"\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_STR_LEN`. If a string is not null-terminated within that bound, `strnlen` returns `MAX_STR_LEN` and the function safely bails out with `NULL` instead of reading past the buffer. The inner comparison loop is also bounded by the current string's measured length (`cur_len`), so it can never index past an unterminated string.

2. **CWE-120 (`memcpy` overflow):** The destination buffer is allocated as `prefix_len + 1`, and `prefix_len` is now explicitly validated against `MAX_STR_LEN` before allocation, guaranteeing the destination can always hold exactly `prefix_len` bytes plus the terminator. The copy length can never exceed the allocation, and the `malloc` return value is checked before use.

The logic remains correct: for the given input the program prints `Longest common prefix: "flo"`, and all allocated memory is freed on every exit path.
