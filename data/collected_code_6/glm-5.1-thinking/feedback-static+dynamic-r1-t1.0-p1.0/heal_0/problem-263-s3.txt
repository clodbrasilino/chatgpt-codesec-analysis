#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dictionary;

static char *string_duplicate(const char *str) {
    if (!str) return NULL;
    size_t len = strlen(str) + 1;
    char *dup = malloc(len);
    if (dup) {
        memcpy(dup, str, len);
    }
    return dup;
}

void dictionary_init(Dictionary *dict) {
    if (!dict) return;
    dict->entries = NULL;
    dict->size = 0;
    dict->capacity = 0;
}

void dictionary_free(Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        free(dict->entries[i].key);
        free(dict->entries[i].value);
    }
    free(dict->entries);
    dict->entries = NULL;
    dict->size = 0;
    dict->capacity = 0;
}

int dictionary_set(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) return -1;
    for (size_t i = 0; i < dict->size; i++) {
        if (strcmp(dict->entries[i].key, key) == 0) {
            char *new_value = string_duplicate(value);
            if (!new_value) return -1;
            free(dict->entries[i].value);
            dict->entries[i].value = new_value;
            return 0;
        }
    }
    if (dict->size == dict->capacity) {
        size_t new_capacity = dict->capacity == 0 ? 4 : dict->capacity * 2;
        DictEntry *new_entries = realloc(dict->entries, new_capacity * sizeof(DictEntry));
        if (!new_entries) return -1;
        dict->entries = new_entries;
        dict->capacity = new_capacity;
    }
    char *new_key = string_duplicate(key);
    if (!new_key) return -1;
    char *new_value = string_duplicate(value);
    if (!new_value) {
        free(new_key);
        return -1;
    }
    dict->entries[dict->size].key = new_key;
    dict->entries[dict->size].value = new_value;
    dict->size++;
    return 0;
}

int dictionary_merge(Dictionary *dest, const Dictionary *src) {
    if (!dest || !src) return -1;
    for (size_t i = 0; i < src->size; i++) {
        if (dictionary_set(dest, src->entries[i].key, src->entries[i].value) != 0) {
            return -1;
        }
    }
    return 0;
}

int main(void) {
    Dictionary dict1;
    dictionary_init(&dict1);

    if (dictionary_set(&dict1, "name", "Alice") != 0 ||
        dictionary_set(&dict1, "age", "30") != 0 ||
        dictionary_set(&dict1, "city", "New York") != 0) {
        fprintf(stderr, "Failed to populate dictionary 1\n");
        dictionary_free(&dict1);
        return EXIT_FAILURE;
    }

    Dictionary dict2;
    dictionary_init(&dict2);

    if (dictionary_set(&dict2, "age", "31") != 0 ||
        dictionary_set(&dict2, "country", "USA") != 0 ||
        dictionary_set(&dict2, "job", "Engineer") != 0) {
        fprintf(stderr, "Failed to populate dictionary 2\n");
        dictionary_free(&dict1);
        dictionary_free(&dict2);
        return EXIT_FAILURE;
    }

    if (dictionary_merge(&dict1, &dict2) != 0) {
        fprintf(stderr, "Failed to merge dictionaries\n");
        dictionary_free(&dict1);
        dictionary_free(&dict2);
        return EXIT_FAILURE;
    }

    dictionary_free(&dict2);

    for (size_t i = 0; i < dict1.size; i++) {
        printf("%s: %s\n", dict1.entries[i].key, dict1.entries[i].value);
    }

    dictionary_free(&dict1);
    return EXIT_SUCCESS;
}
