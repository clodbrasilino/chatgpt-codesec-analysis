```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS 64
#define BUFFER_SIZE (MAX_BITS + 1)

static int decimal_to_binary(unsigned long long value, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;
    char temp[BUFFER_SIZE];

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));
    memset(buffer, 0, buffer_size);

    if (value == 0ULL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0ULL && index < (sizeof(temp) - 1)) {
        temp[index] = (char)('0' + (value & 1ULL));
        value >>= 1ULL;
        index++;
    }

    if (value > 0ULL) {
        return -1;
    }

    if (index >= buffer_size) {
        return -1;
    }

    for (i = 0; i < index; i++) {
        buffer[i] = temp[index - 1 - i];
    }
    buffer[index] = '\0';

    return 0;
}

int main(void)
{
    long long input = 0;
    unsigned long long magnitude = 0ULL;
    char binary[BUFFER_SIZE];
    int is_negative = 0;

    memset(binary, 0, sizeof(binary));

    printf("Enter a decimal number: ");

    if (scanf("%lld", &input) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (input < 0) {
        is_negative = 1;
        if (input == LLONG_MIN) {
            magnitude = (unsigned long long)LLONG_MAX + 1ULL;
        } else {
            magnitude = (unsigned long long)(-input);
        }
    } else {
        magnitude = (unsigned long long)input;
    }

    if (decimal_to_binary(magnitude, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    if (is_negative) {
        printf("Binary equivalent: -%s\n", binary);
    } else {
        printf("Binary equivalent: %s\n", binary);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended against the Flawfinder warnings, which are largely informational (risk level 2) and flag *any* statically-sized array by pattern matching, not by actual data-flow analysis. The key observations:

1. **`temp[MAX_BITS + 1]` in `decimal_to_binary`**: The write loop was bounded by `index < MAX_BITS`, guaranteeing no overflow. In the rewrite, I tied the bound directly to `sizeof(temp) - 1` so the check can never silently desynchronize from the array size if the constant changes.

2. **`binary[MAX_BITS + 1]` in `main`**: Safe because `sizeof(binary)` is passed explicitly to the conversion function, which validates `index >= buffer_size` before copying and null-terminates within bounds. I introduced a `BUFFER_SIZE` macro so both the caller's array and the callee's temporary derive from a single definition.

3. **`LLONG_MIN` handling**: Correctly avoids the undefined behavior of `-LLONG_MIN` by computing the magnitude as `(unsigned long long)LLONG_MAX + 1ULL`. Preserved as-is.

4. **Input validation**: `scanf` return value is checked; `buffer == NULL` and minimum-size guards are retained.

No genuine vulnerabilities existed; the rewrite hardens the code structurally (size constants derived from `sizeof`, single source of truth for buffer sizing) so the bounds checks remain correct under future maintenance.
