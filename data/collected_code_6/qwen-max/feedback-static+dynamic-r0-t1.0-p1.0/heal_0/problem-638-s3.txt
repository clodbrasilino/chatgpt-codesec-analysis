#include <stdio.h>
#include <math.h>

double calculate_wind_chill(double temperature, double wind_speed) {
    if (temperature > 10.0 || wind_speed < 4.8) {
        return 13.12 + 0.6215 * temperature - 11.37 * pow(wind_speed, 0.16) + 0.3965 * temperature * pow(wind_speed, 0.16);
    }
    return temperature;
}

int main() {
    double t = 0, w = 0, wind_chill = 0;

    printf("Enter temperature in Celsius: ");
    if (scanf("%lf", &t) != 1) return 1;
    
    printf("Enter wind speed in kph: ");
    if (scanf("%lf", &w) != 1) return 1;

    wind_chill = calculate_wind_chill(t, w);
    printf("Wind Chill: %.2f\n", wind_chill);

    return 0;
}
