```c
#include <stdio.h>
#include <stdlib.h>

#define PI 3.14159265358979323846

double radians_to_degrees(double radians);

double radians_to_degrees(double radians) {
    return radians * (180.0 / PI);
}

int main(void) {
    double radians = 1.57079632679489661923;
    double degrees = radians_to_degrees(radians);
    
    if (printf("%f radians is equal to %f degrees.\n", radians, degrees) < 0) {
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
```
