```C
#include <stdio.h>

int sumOddLengthSubarrays(int* arr, int arrSize) {
    int sum = 0, i, j, k;
    for(i = 0; i < arrSize; i++) {
        for(j = i; j < arrSize; j++) {
            if((j - i + 1) % 2 != 0) {
                for(k = i; k <= j; k++) {
                    sum += arr[k];
                }
            }
        }
    }
    return sum;
}

int main() {
    int arr[5] = {1, 4, 2, 5, 3};
    int res = sumOddLengthSubarrays(arr, 5);
    printf("%d", res);
    return 0;
}
```
