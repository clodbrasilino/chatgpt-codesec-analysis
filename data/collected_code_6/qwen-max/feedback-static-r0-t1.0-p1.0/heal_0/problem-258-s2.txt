```c
#include <stdio.h>

int count_odds(int *arr, int n) {
    int count = 0;
    for (int i = 0; i < n; ++i) {
        if (arr[i] % 2 != 0) {
            ++count;
        }
    }
    return count;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5, 6, 7, 8, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    int odds = count_odds(arr, n);
    printf("Number of odd elements: %d\n", odds);
    return 0;
}
```
