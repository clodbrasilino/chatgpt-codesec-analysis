```c
#include <stdio.h>
#include <stdlib.h>

unsigned long long sum_of_fifth_powers(unsigned int n) {
    unsigned long long sum = 0;
    unsigned long long i;
    for (i = 1; i <= n; i++) {
        sum += i * i * i * i * i;
    }
    return sum;
}

int main(void) {
    unsigned int n;
    int result;

    result = scanf("%u", &n);
    if (result != 1) {
        return EXIT_FAILURE;
    }

    printf("%llu\n", sum_of_fifth_powers(n));

    return EXIT_SUCCESS;
}
```
