#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int x;
    int y;
} Point;

typedef struct {
    Point points[6];
} Hexagon;

typedef struct {
    Hexagon *hexagons;
    size_t count;
} HexagonGrid;

HexagonGrid create_hexagon_grid(int centerX, int centerY, int radius, int rows, int cols) {
    HexagonGrid grid;
    grid.hexagons = (Hexagon *)malloc(rows * cols * sizeof(Hexagon));
    if (grid.hexagons == NULL) {
        exit(EXIT_FAILURE);
    }
    grid.count = rows * cols;

    for (int row = 0; row < rows; ++row) {
        for (int col = 0; col < cols; ++col) {
            int index = row * cols + col;
            int x_offset = (col % 2 == 0) ? 0 : radius;
            int y_offset = row * (radius * 3 / 2);

            grid.hexagons[index].points[0] = (Point){centerX + x_offset, centerY + y_offset - radius};
            grid.hexagons[index].points[1] = (Point){centerX + x_offset + radius, centerY + y_offset - radius / 2};
            grid.hexagons[index].points[2] = (Point){centerX + x_offset + radius, centerY + y_offset + radius / 2};
            grid.hexagons[index].points[3] = (Point){centerX + x_offset, centerY + y_offset + radius};
            grid.hexagons[index].points[4] = (Point){centerX + x_offset - radius, centerY + y_offset + radius / 2};
            grid.hexagons[index].points[5] = (Point){centerX + x_offset - radius, centerY + y_offset - radius / 2};
        }
    }

    return grid;
}

void print_hexagon_grid(const HexagonGrid *grid) {
    for (size_t i = 0; i < grid->count; ++i) {
        for (int j = 0; j < 6; ++j) {
            printf("(%d, %d) ", grid->hexagons[i].points[j].x, grid->hexagons[i].points[j].y);
        }
        printf("\n");
    }
}

void free_hexagon_grid(HexagonGrid *grid) {
    free(grid->hexagons);
    grid->hexagons = NULL;
    grid->count = 0;
}

int main() {
    int centerX = 0;
    int centerY = 0;
    int radius = 10;
    int rows = 3;
    int cols = 3;

    HexagonGrid grid = create_hexagon_grid(centerX, centerY, radius, rows, cols);
    print_hexagon_grid(&grid);
    free_hexagon_grid(&grid);

    return 0;
}
