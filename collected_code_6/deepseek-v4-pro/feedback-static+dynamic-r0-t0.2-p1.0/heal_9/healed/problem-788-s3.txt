#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    char *str;
    int *list;
    size_t list_len;
} Tuple;

Tuple create_tuple(const char *str, const int *list, size_t list_len) {
    Tuple t;

    if (str == NULL) {
        t.str = NULL;
    } else {
        size_t str_len = strnlen(str, SIZE_MAX);
        if (str_len == SIZE_MAX) {
            fprintf(stderr, "String too long or not null-terminated\n");
            exit(EXIT_FAILURE);
        }

        t.str = malloc(str_len + 1);
        if (t.str == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            exit(EXIT_FAILURE);
        }

        if (str_len > 0) {
            if (strnlen(str, str_len) != str_len) {
                free(t.str);
                fprintf(stderr, "String length mismatch detected\n");
                exit(EXIT_FAILURE);
            }
            memcpy(t.str, str, str_len);
        }
        t.str[str_len] = '\0';
    }

    if (list_len > 0 && list != NULL) {
        if (list_len > SIZE_MAX / sizeof(int)) {
            free(t.str);
            fprintf(stderr, "Integer overflow detected\n");
            exit(EXIT_FAILURE);
        }

        size_t alloc_size = list_len * sizeof(int);
        t.list = malloc(alloc_size);
        if (t.list == NULL) {
            free(t.str);
            fprintf(stderr, "Memory allocation failed\n");
            exit(EXIT_FAILURE);
        }

        if (alloc_size > 0) {
            memcpy(t.list, list, alloc_size);
        }
    } else {
        t.list = NULL;
    }

    t.list_len = (list != NULL) ? list_len : 0;
    return t;
}

void free_tuple(Tuple *t) {
    if (t != NULL) {
        free(t->str);
        free(t->list);
        t->str = NULL;
        t->list = NULL;
        t->list_len = 0;
    }
}

int main(void) {
    const char *str = "hello";
    int list[] = {1, 2, 3, 4, 5};
    size_t list_len = sizeof(list) / sizeof(list[0]);
    Tuple t;

    t = create_tuple(str, list, list_len);

    if (t.str != NULL) {
        printf("String: %s\n", t.str);
    }
    printf("List: ");
    for (size_t i = 0; i < t.list_len; i++) {
        printf("%d ", t.list[i]);
    }
    printf("\n");

    free_tuple(&t);

    return 0;
}
