#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int first;
    int second;
} Tuple;

int count_bidirectional_pairs(Tuple* tuples, size_t size) {
    if (tuples == NULL || size == 0) {
        return 0;
    }

    int count = 0;
    int* matched = (int*)calloc(size, sizeof(int));
    if (matched == NULL) {
        return -1;
    }

    for (size_t i = 0; i < size; ++i) {
        if (matched[i]) {
            continue;
        }

        for (size_t j = i + 1; j < size; ++j) {
            if (!matched[j] && tuples[i].first == tuples[j].second && tuples[i].second == tuples[j].first) {
                count++;
                matched[i] = 1;
                matched[j] = 1;
                break;
            }
        }
    }

    free(matched);
    return count;
}

int main(void) {
    Tuple tuples[] = {
        {1, 2},
        {2, 1},
        {3, 4},
        {4, 3},
        {5, 6},
        {1, 2},
        {2, 1}
    };
    
    size_t size = sizeof(tuples) / sizeof(tuples[0]);
    int result = count_bidirectional_pairs(tuples, size);
    
    if (result != -1) {
        printf("%d\n", result);
    }
    
    return 0;
}
