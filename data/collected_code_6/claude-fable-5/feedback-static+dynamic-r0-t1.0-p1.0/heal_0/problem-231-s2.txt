#include <stdio.h>
#include <stdlib.h>

static int max_triangle_sum(int **triangle, int n)
{
    int i;
    int j;
    int result;
    int *dp;

    dp = malloc((size_t)n * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    for (j = 0; j < n; j++) {
        dp[j] = triangle[n - 1][j];
    }

    for (i = n - 2; i >= 0; i--) {
        for (j = 0; j <= i; j++) {
            if (dp[j] > dp[j + 1]) {
                dp[j] = triangle[i][j] + dp[j];
            } else {
                dp[j] = triangle[i][j] + dp[j + 1];
            }
        }
    }

    result = dp[0];
    free(dp);
    return result;
}

static void free_triangle(int **triangle, int rows)
{
    int i;

    for (i = 0; i < rows; i++) {
        free(triangle[i]);
    }
    free(triangle);
}

int main(void)
{
    int n;
    int i;
    int j;
    int **triangle;
    int result;

    printf("Enter number of rows: ");
    if (scanf("%d", &n) != 1 || n <= 0 || n > 1000) {
        fprintf(stderr, "Invalid input for number of rows\n");
        return EXIT_FAILURE;
    }

    triangle = malloc((size_t)n * sizeof(int *));
    if (triangle == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        triangle[i] = malloc((size_t)(i + 1) * sizeof(int));
        if (triangle[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_triangle(triangle, i);
            return EXIT_FAILURE;
        }
    }

    printf("Enter triangle values row by row:\n");
    for (i = 0; i < n; i++) {
        for (j = 0; j <= i; j++) {
            if (scanf("%d", &triangle[i][j]) != 1) {
                fprintf(stderr, "Invalid triangle value\n");
                free_triangle(triangle, n);
                return EXIT_FAILURE;
            }
        }
    }

    result = max_triangle_sum(triangle, n);
    if (result == -1) {
        fprintf(stderr, "Computation failed\n");
        free_triangle(triangle, n);
        return EXIT_FAILURE;
    }

    printf("Maximum sum: %d\n", result);

    free_triangle(triangle, n);
    return EXIT_SUCCESS;
}
