#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 1000
#define MAX_WORD_LENGTH 50

char* remove_duplicates(const char *input) {
    char words[MAX_WORDS][MAX_WORD_LENGTH] = {0};
    int word_count = 0;
    const char *token;
    char *output, *temp;
    int i;

    token = strtok((char *)input, " ");
    while (token != NULL && word_count < MAX_WORDS) {
        int found = 0;
        for (i = 0; i < word_count; ++i) {
            if (strcmp(words[i], token) == 0) {
                found = 1;
                break;
            }
        }
        if (!found) {
            size_t token_length = strnlen(token, MAX_WORD_LENGTH);
            if (token_length >= MAX_WORD_LENGTH) {
                token_length = MAX_WORD_LENGTH - 1;
            }
            memcpy(words[word_count], token, token_length);
            words[word_count][token_length] = '\0';
            word_count++;
        }
        token = strtok(NULL, " ");
    }

    size_t output_length = 0;
    for (i = 0; i < word_count; ++i) {
        output_length += strlen(words[i]) + 1;
    }
    output = malloc(output_length);
    if (output == NULL) {
        return NULL;
    }
    temp = output;

    for (i = 0; i < word_count; ++i) {
        size_t len = strlen(words[i]);
        memcpy(temp, words[i], len);
        temp += len;
        if (i < word_count - 1) {
            *temp++ = ' ';
        }
    }
    *temp = '\0';

    return output;
}

int main() {
    const char *input = "this is a test this is only a test";
    char *result = remove_duplicates(input);
    if (result != NULL) {
        printf("%s\n", result);
        free(result);
    } else {
        fprintf(stderr, "Memory allocation failed\n");
    }
    return 0;
}
