#include <stdio.h>
#include <stdbool.h>

bool is_in_sequence(long long A, long long B, long long C) {
    if (C == 0) {
        return A == B;
    }
    
    if ((B - A) % C == 0 && (B - A) / C >= 0) {
        return true;
    }
    
    return false;
}

int main(void) {
    long long A, B, C;
    
    if (scanf("%lld %lld %lld", &A, &B, &C) != 3) {
        return 1;
    }
    
    if (is_in_sequence(A, B, C)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }
    
    return 0;
}
