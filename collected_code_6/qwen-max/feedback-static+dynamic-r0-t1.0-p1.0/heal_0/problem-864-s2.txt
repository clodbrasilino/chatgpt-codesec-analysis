#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

bool is_palindrome(const char *str) {
    int len = strlen(str);
    for (int i = 0; i < len / 2; ++i) {
        if (str[i] != str[len - 1 - i]) {
            return false;
        }
    }
    return true;
}

void find_palindromes(char **strings, int count, char **result, int *result_count) {
    *result_count = 0;
    for (int i = 0; i < count; ++i) {
        if (is_palindrome(strings[i])) {
            result[*result_count] = strings[i];
            (*result_count)++;
        }
    }
}

int main() {
    char *strings[] = {"雷达", "level", "world", "madam", "deified", "hello"};
    int count = sizeof(strings) / sizeof(strings[0]);
    char **result = (char **)malloc(count * sizeof(char *));
    if (result == NULL) {
        return 1;
    }
    int result_count;
    
    find_palindromes(strings, count, result, &result_count);

    for (int i = 0; i < result_count; ++i) {
        printf("%s\n", result[i]);
    }

    free(result);
    return 0;
}
