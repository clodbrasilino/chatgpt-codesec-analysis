```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int min_abs_diff_sum(const int *arr1, const int *arr2, size_t n, long long *result)
{
    int *sorted1 = NULL;
    int *sorted2 = NULL;
    long long sum = 0;
    size_t i;
    size_t bytes;

    if (arr1 == NULL || arr2 == NULL || result == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*sorted1)) {
        return -1;
    }

    bytes = n * sizeof(*sorted1);

    sorted1 = malloc(bytes);
    if (sorted1 == NULL) {
        return -1;
    }

    sorted2 = malloc(bytes);
    if (sorted2 == NULL) {
        free(sorted1);
        return -1;
    }

    memcpy(sorted1, arr1, bytes);
    memcpy(sorted2, arr2, bytes);

    qsort(sorted1, n, sizeof(*sorted1), compare_ints);
    qsort(sorted2, n, sizeof(*sorted2), compare_ints);

    for (i = 0; i < n; i++) {
        long long diff = (long long)sorted1[i] - (long long)sorted2[i];
        if (diff < 0) {
            diff = -diff;
        }
        sum += diff;
    }

    free(sorted1);
    free(sorted2);

    *result = sum;
    return 0;
}

int main(void)
{
    const int arr1[] = {4, 1, 8, 7};
    const int arr2[] = {2, 3, 6, 5};
    size_t n = sizeof(arr1) / sizeof(arr1[0]);
    long long result = 0;

    if (min_abs_diff_sum(arr1, arr2, n, &result) != 0) {
        fprintf(stderr, "Error computing minimum sum of absolute differences\n");
        return EXIT_FAILURE;
    }

    printf("Minimum sum of absolute differences: %lld\n", result);
    return EXIT_SUCCESS;
}
```

**Summary of the changes made:**

1. **CWE-120 / memcpy buffer overflow (Flawfinder, risk 2):** The core concern is that the destination buffer might not be large enough for the copy. The rewrite computes the byte count once (`bytes = n * sizeof(*sorted1)`) and uses that exact same value for both the `malloc` allocations and the `memcpy` calls, guaranteeing destination and copy sizes always match. Additionally, an explicit integer-overflow guard (`n > SIZE_MAX / sizeof(*sorted1)`) was added before the multiplication, preventing a wrapped-around undersized allocation followed by an oversized copy — the classic path to this vulnerability.

2. **cppcheck `constVariable` on `arr2`:** Both `arr1` and `arr2` are now declared `const`, since they are never modified in `main` and are passed to a function taking `const int *` parameters. This lets the compiler enforce immutability and may allow the arrays to be placed in read-only storage.

3. **Preserved correctness properties:** NULL/zero-size argument validation, cleanup of `sorted1` on the second allocation failure, use of `long long` arithmetic to avoid overflow when computing differences of `int` values, and proper freeing of both buffers on the success path all remain intact.
