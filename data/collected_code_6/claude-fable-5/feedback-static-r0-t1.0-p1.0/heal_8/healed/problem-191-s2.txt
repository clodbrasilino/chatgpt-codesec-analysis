#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool normalize_month(const char *month, char *out, size_t out_size)
{
    size_t len;
    size_t i;

    if (month == NULL || out == NULL || out_size == 0) {
        return false;
    }

    len = strnlen(month, out_size);
    if (len == 0 || len >= out_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        out[i] = (char)tolower((unsigned char)month[i]);
    }
    out[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strcmp(normalized, all_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %s\n", input);
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`char normalized[MAX_INPUT_LEN]` / `char input[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — These are false positives. Every write into these buffers is bounds-checked: `normalize_month` receives `sizeof(normalized)` explicitly, uses `strnlen` capped at `out_size`, and rejects any input where `len >= out_size`, guaranteeing room for the null terminator. `fgets` in `main` is called with `sizeof(input)` and can never overflow, and the explicit `input[sizeof(input) - 1] = '\0'` is a defensive guarantee of termination.

2. **`getchar` in a loop (CWE-120/CWE-20)** — Also a false positive. `discard_remaining_input` never stores the characters read; it merely drains `stdin` until newline or `EOF`, so no buffer is involved and no overflow is possible. The `EOF` check prevents an infinite loop on closed input streams.

3. **Input validation (CWE-20)** — Input is validated against a whitelist of exact month names after normalization, truncated input is detected (the leftover-data path drains the stream so subsequent reads aren't polluted), and case handling uses `tolower((unsigned char)c)` to avoid undefined behavior with negative `char` values.

The code follows defense-in-depth practices already; no functional changes were required to remediate the reported findings — only verification that each flagged pattern is properly bounded.
