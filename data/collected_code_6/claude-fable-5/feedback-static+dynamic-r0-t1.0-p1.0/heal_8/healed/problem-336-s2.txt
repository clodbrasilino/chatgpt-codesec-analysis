```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32
#define FLUSH_GUARD_LIMIT 4096

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len && i < (sizeof(normalized) - 1); i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < FLUSH_GUARD_LIMIT);
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code already mitigates them:**

1. **`char normalized[MAX_INPUT]` in `has_28_days` (CWE-119/CWE-120):** The buffer cannot overflow because the input length is validated with `strnlen(month, MAX_INPUT)` and rejected if `len >= MAX_INPUT`. Additionally, the copy loop enforces a second bound `i < (sizeof(normalized) - 1)`, and the null terminator is written at index `i`, which is guaranteed to be within bounds. This is defense-in-depth — the warning is a false positive.

2. **`getchar()` in a loop within `flush_stdin` (CWE-120/CWE-20):** No buffer is written to, so no overflow is possible. The `FLUSH_GUARD_LIMIT` counter also prevents an unbounded loop against a hostile stream that never delivers `'\n'` or `EOF`, mitigating a potential denial-of-service.

3. **`char input[MAX_INPUT]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits writes to the buffer size. The explicit `input[sizeof(input) - 1] = '\0'` guarantees termination, `strnlen` is used instead of `strlen` to bound the scan, and any excess characters left on the stream are drained via `flush_stdin` so they can't leak into subsequent reads.

The Flawfinder hits are all risk level 1–2 informational warnings about the *presence* of fixed-size buffers and loop-based character input, not actual exploitable defects. The code as written already applies the recommended mitigations (bounded reads, `strnlen`, explicit termination, dual bounds checks, input validation via `isalpha`), so the rewrite preserves the logic unchanged with the comments removed.
