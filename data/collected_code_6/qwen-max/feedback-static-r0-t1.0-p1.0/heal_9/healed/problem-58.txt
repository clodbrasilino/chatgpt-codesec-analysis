```c
#include <stdio.h>

int have_opposite_signs(int x, int y) {
    return (x ^ y) < 0;
}

int main() {
    int a = -10;
    int b = 20;

    if (have_opposite_signs(a, b)) {
        printf("The numbers have opposite signs.\n");
    } else {
        printf("The numbers do not have opposite signs.\n");
    }

    a = 10;
    b = 20;

    if (have_opposite_signs(a, b)) {
        printf("The numbers have opposite signs.\n");
    } else {
        printf("The numbers do not have opposite signs.\n");
    }

    return 0;
}
```
