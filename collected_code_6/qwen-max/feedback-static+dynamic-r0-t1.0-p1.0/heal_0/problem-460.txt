#include <stdio.h>
#include <stdlib.h>

int* get_first_elements(int** sublists, int sublist_count, int* result_size) {
    *result_size = 0;
    for (int i = 0; i < sublist_count; i++) {
        if (sublists[i] == NULL) {
            return NULL;
        }
    }

    int* first_elements = malloc(sublist_count * sizeof(int));
    if (first_elements == NULL) {
        return NULL;
    }

    for (int i = 0; i < sublist_count; i++) {
        first_elements[*result_size] = sublists[i][0];
        (*result_size)++;
    }

    return first_elements;
}

int main() {
    int list1[] = {1, 2, 3};
    int list2[] = {4, 5, 6};
    int list3[] = {7, 8, 9};

    int* sublists[3] = {list1, list2, list3};

    int result_size;
    int* result = get_first_elements(sublists, 3, &result_size);

    if (result != NULL) {
        for (int i = 0; i < result_size; i++) {
            printf("%d ", result[i]);
        }
        free(result);
    } else {
        printf("Error: Failed to get first elements.\n");
    }

    return 0;
}
