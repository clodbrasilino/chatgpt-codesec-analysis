```C
#include <stdio.h>
#include <stdbool.h>

bool isPrime(int num) {
    if(num<=1) return false;
    if(num==2) return true;
    for(int i=2;i*i<=num;i++) {
        if(num%i==0) return false;
    }
    return true;
}

int sumPrimeDivisors(int num) {
    int sum = 0;
    for(int i=1;i<=num;i++) {
        if(num%i == 0 && isPrime(i)) {
            sum += i;
        }
    }
    return sum;
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d",&n);
    printf("Sum of Prime divisors of %d is %d\n", n, sumPrimeDivisors(n));
    return 0;
}
```
