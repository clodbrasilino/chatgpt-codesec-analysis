#include <stdio.h>
#include <stdlib.h>

#define MAX_MATRIX_SIZE 46340

int **generate_spiral_matrix(int n);
void print_matrix(const int *const *matrix, int n);
void free_matrix(int **matrix, int n);

int **generate_spiral_matrix(int n)
{
    int **matrix;
    int i;
    int top;
    int bottom;
    int left;
    int right;
    int value;

    if (n <= 0 || n > MAX_MATRIX_SIZE) {
        return NULL;
    }

    matrix = malloc((size_t)n * sizeof(*matrix));
    if (matrix == NULL) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        matrix[i] = malloc((size_t)n * sizeof(**matrix));
        if (matrix[i] == NULL) {
            int j;
            for (j = 0; j < i; j++) {
                free(matrix[j]);
                matrix[j] = NULL;
            }
            free(matrix);
            matrix = NULL;
            return NULL;
        }
    }

    top = 0;
    bottom = n - 1;
    left = 0;
    right = n - 1;
    value = 1;

    while (top <= bottom && left <= right) {
        int k;

        for (k = left; k <= right; k++) {
            matrix[top][k] = value++;
        }
        top++;

        for (k = top; k <= bottom; k++) {
            matrix[k][right] = value++;
        }
        right--;

        if (top <= bottom) {
            for (k = right; k >= left; k--) {
                matrix[bottom][k] = value++;
            }
            bottom--;
        }

        if (left <= right) {
            for (k = bottom; k >= top; k--) {
                matrix[k][left] = value++;
            }
            left++;
        }
    }

    return matrix;
}

void print_matrix(const int *const *matrix, int n)
{
    int i;
    int j;

    if (matrix == NULL || n <= 0) {
        return;
    }

    for (i = 0; i < n; i++) {
        if (matrix[i] == NULL) {
            return;
        }
        for (j = 0; j < n; j++) {
            printf("%6d", matrix[i][j]);
        }
        printf("\n");
    }
}

void free_matrix(int **matrix, int n)
{
    int i;

    if (matrix == NULL || n <= 0) {
        return;
    }

    for (i = 0; i < n; i++) {
        free(matrix[i]);
        matrix[i] = NULL;
    }
    free(matrix);
}

int main(void)
{
    int n;
    int **matrix;

    printf("Enter the matrix size n: ");
    fflush(stdout);

    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (n <= 0 || n > MAX_MATRIX_SIZE) {
        fprintf(stderr, "Error: n must be between 1 and %d.\n", MAX_MATRIX_SIZE);
        return EXIT_FAILURE;
    }

    matrix = generate_spiral_matrix(n);
    if (matrix == NULL) {
        fprintf(stderr, "Error: failed to allocate matrix.\n");
        return EXIT_FAILURE;
    }

    print_matrix(matrix, n);
    free_matrix(matrix, n);
    matrix = NULL;

    return EXIT_SUCCESS;
}
