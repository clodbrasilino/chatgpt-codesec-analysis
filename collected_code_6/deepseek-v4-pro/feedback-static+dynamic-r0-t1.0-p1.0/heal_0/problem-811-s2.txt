#include <stdio.h>
#include <stdlib.h>

typedef struct Tuple {
    int first;
    int second;
    struct Tuple *next;
} Tuple;

void addTuple(Tuple **head, int first, int second) {
    Tuple *newTuple = malloc(sizeof(Tuple));
    if (!newTuple) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }
    newTuple->first = first;
    newTuple->second = second;
    newTuple->next = *head;
    *head = newTuple;
}

void freeTuples(Tuple *head) {
    Tuple *current = head;
    while (current) {
        Tuple *temp = current;
        current = current->next;
        free(temp);
    }
}

int areTuplesEqual(Tuple *list1, Tuple *list2) {
    while (list1 && list2) {
        if (list1->first != list2->first || list1->second != list2->second) {
            return 0;
        }
        list1 = list1->next;
        list2 = list2->next;
    }
    return list1 == NULL && list2 == NULL;
}

int main(void) {
    Tuple *list1 = NULL;
    Tuple *list2 = NULL;
    
    addTuple(&list1, 1, 2);
    addTuple(&list1, 3, 4);
    
    addTuple(&list2, 1, 2);
    addTuple(&list2, 3, 4);
    
    if (areTuplesEqual(list1, list2)) {
        printf("Lists are identical\n");
    } else {
        printf("Lists are not identical\n");
    }
    
    freeTuples(list1);
    freeTuples(list2);
    
    return 0;
}
