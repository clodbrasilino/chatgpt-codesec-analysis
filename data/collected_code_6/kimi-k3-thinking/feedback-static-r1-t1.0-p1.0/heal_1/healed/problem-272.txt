```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TUPLE_NAME_CAPACITY 32U

typedef struct {
    int id;
    double value;
    char name[TUPLE_NAME_CAPACITY];
} Tuple;

typedef struct Node {
    Tuple data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    Node *tail;
    size_t size;
} TupleList;

int tuple_init(Tuple *tuple, int id, double value, const char *name)
{
    int written;

    if (tuple == NULL) {
        return -1;
    }

    tuple->id = id;
    tuple->value = value;
    tuple->name[0] = '\0';

    if (name == NULL) {
        return 0;
    }

    written = snprintf(tuple->name, TUPLE_NAME_CAPACITY, "%s", name);
    if (written < 0) {
        tuple->name[0] = '\0';
        return -1;
    }
    if ((size_t)written >= TUPLE_NAME_CAPACITY) {
        tuple->name[TUPLE_NAME_CAPACITY - 1U] = '\0';
        return -1;
    }

    return 0;
}

void tuple_list_init(TupleList *list)
{
    if (list == NULL) {
        return;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0U;
}

int tuple_list_append(TupleList *list, const Tuple *tuple)
{
    Node *node;

    if (list == NULL || tuple == NULL) {
        return -1;
    }

    node = malloc(sizeof(*node));
    if (node == NULL) {
        return -1;
    }

    node->data = *tuple;
    node->data.name[TUPLE_NAME_CAPACITY - 1U] = '\0';
    node->next = NULL;

    if (list->tail != NULL) {
        list->tail->next = node;
    } else {
        list->head = node;
    }

    list->tail = node;
    list->size++;

    return 0;
}

int tuple_list_extract_rear(TupleList *list, Tuple *out)
{
    Node *prev;
    Node *rear;

    if (list == NULL || out == NULL) {
        return -1;
    }

    if (list->tail == NULL) {
        return 1;
    }

    rear = list->tail;

    if (list->head == rear) {
        list->head = NULL;
        list->tail = NULL;
    } else {
        prev = list->head;
        while (prev->next != rear) {
            prev = prev->next;
        }
        prev->next = NULL;
        list->tail = prev;
    }

    *out = rear->data;
    out->name[TUPLE_NAME_CAPACITY - 1U] = '\0';
    free(rear);
    rear = NULL;

    if (list->size > 0U) {
        list->size--;
    }

    return 0;
}

void tuple_list_destroy(TupleList *list)
{
    Node *current;
    Node *next;

    if (list == NULL) {
        return;
    }

    current = list->head;
    while (current != NULL) {
        next = current->next;
        free(current);
        current = next;
    }

    list->head = NULL;
    list->tail = NULL;
    list->size = 0U;
}

int main(void)
{
    TupleList list;
    Tuple extracted;
    Tuple samples[4];
    size_t i;
    size_t count;
    const int ids[] = {1, 2, 3, 4};
    const double values[] = {10.5, 20.25, 30.75, 40.0};
    const char *names[] = {"alpha", "beta", "gamma", "delta"};

    count = sizeof(samples) / sizeof(samples[0]);

    tuple_list_init(&list);

    for (i = 0U; i < count; i++) {
        if (tuple_init(&samples[i], ids[i], values[i], names[i]) != 0) {
            fprintf(stderr, "Failed to initialize tuple at index %zu\n", i);
            tuple_list_destroy(&list);
            return EXIT_FAILURE;
        }
        if (tuple_list_append(&list, &samples[i]) != 0) {
            fprintf(stderr, "Failed to append tuple at index %zu\n", i);
            tuple_list_destroy(&list);
            return EXIT_FAILURE;
        }
    }

    while (tuple_list_extract_rear(&list, &extracted) == 0) {
        printf("Extracted rear: id=%d value=%.2f name=%s\n",
               extracted.id, extracted.value, extracted.name);
    }

    if (list.size != 0U) {
        fprintf(stderr, "List size inconsistency detected\n");
        tuple_list_destroy(&list);
        return EXIT_FAILURE;
    }

    tuple_list_destroy(&list);

    return EXIT_SUCCESS;
}
```

Key fixes applied for the flagged CWE-119/CWE-120 weakness:

1. **`tuple_init()` with bounded copy**: All writes into `name[32]` now go through `snprintf()`, which limits the write to the buffer capacity and always NUL-terminates. Truncation is detected via the return value and reported as an error.
2. **NULL `name` handling**: A NULL source string safely produces an empty name instead of crashing.
3. **Defensive termination on struct copies**: Since `node->data = *tuple` copies the raw array, the code forces `name[TUPLE_NAME_CAPACITY - 1] = '\0'` after copying in both `tuple_list_append()` and `tuple_list_extract_rear()`, guaranteeing termination even if a caller passes a `Tuple` built by hand with an unterminated array. This also makes the `%s` `printf` safe.
4. **Named capacity constant**: `TUPLE_NAME_CAPACITY` replaces the magic number `32`, so bounds checks and the buffer size can never drift apart.
