```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < (size_t)INT_MAX);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size == 0 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = strnlen(buffer, size);
    if (len > 0 && buffer[len - 1] != '\n' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    return 0;
}

int main(void)
{
    char buffer[64];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and how they are addressed:**

1. **`getchar` in a loop (CWE-120, CWE-20):** The original loop was already safe since `getchar()` returns an `int` and no buffer is written to. Nevertheless, a `guard` counter was added to guarantee the loop terminates even on a pathological stream that never yields `\n` or `EOF`, hardening against unbounded input consumption.

2. **Statically-sized `char buffer[64]` (CWE-119/CWE-120):** The buffer is safe because:
   - `read_line` receives the size via `sizeof(buffer)` and passes it to `fgets`, which limits the number of bytes written.
   - Explicit null termination at `buffer[size - 1]` guarantees the string is bounded.
   - `strnlen` (rather than `strlen`) is used, so length computation itself is bounds-limited.
   - Truncated input (a full buffer without a trailing newline) is detected, the remaining stream is drained, and an error is returned rather than silently processing partial data.

3. **Additional robustness already present and preserved:** `strtol` with full `errno`/`ERANGE` checking, `endptr` validation, explicit range checks against `INT_MAX`/`INT_MIN`, the `INT_MIN` underflow guard in `closest_smaller`, NULL-pointer checks on all function parameters, and checked return values for `printf` and `fgets`. The code compiles cleanly with `-Wall -Wextra -pedantic` and has no remaining exploitable overflow or integer-wrap conditions.
