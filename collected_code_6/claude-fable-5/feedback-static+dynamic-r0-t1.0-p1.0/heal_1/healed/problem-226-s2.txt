```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096U

char *remove_odd_index_chars(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LENGTH);
    if (len == MAX_INPUT_LENGTH) {
        return NULL;
    }

    result = malloc((len / 2U) + 2U);
    if (result == NULL) {
        return NULL;
    }

    j = 0U;
    for (i = 0U; i < len; i++) {
        if ((i % 2U) == 0U) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "abcdef",
        "hello world",
        "a",
        ""
    };
    size_t count;
    size_t i;
    int status;

    status = EXIT_SUCCESS;
    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0U; i < count; i++) {
        char *output = remove_odd_index_chars(test_strings[i]);
        if (output == NULL) {
            if (fprintf(stderr, "Memory allocation failed\n") < 0) {
                status = EXIT_FAILURE;
                break;
            }
            status = EXIT_FAILURE;
            break;
        }
        if (printf("Input: \"%s\" -> Output: \"%s\"\n", test_strings[i], output) < 0) {
            free(output);
            status = EXIT_FAILURE;
            break;
        }
        free(output);
    }

    return status;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LENGTH`. If the input reaches the maximum bound (indicating a possibly unterminated string), the function safely returns `NULL` instead of over-reading.

2. **Variable scope**: The `output` pointer is now declared inside the `for` loop body, reducing its scope to where it is actually used. This also removes the need for the redundant `output = NULL;` reset after `free`, since the variable goes out of scope each iteration.

The rest of the logic — allocation sizing (`len / 2 + 2` covers the worst case plus the terminator), NULL checks, error handling on `printf`/`fprintf`, and memory cleanup on all paths — was already sound and is preserved.
