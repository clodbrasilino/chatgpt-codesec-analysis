```c
#include <stdio.h>
#include <stdlib.h>

size_t remove_odd_numbers(int *array, size_t size)
{
    size_t i;
    size_t write_index = 0;

    if (array == NULL) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (array[i] % 2 == 0) {
            array[write_index] = array[i];
            write_index++;
        }
    }

    return write_index;
}

int main(void)
{
    size_t size = 10;
    size_t new_size;
    size_t i;
    int *numbers = malloc(size * sizeof(int));

    if (numbers == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    numbers[0] = 1;
    numbers[1] = 2;
    numbers[2] = 3;
    numbers[3] = 4;
    numbers[4] = 5;
    numbers[5] = 6;
    numbers[6] = 7;
    numbers[7] = 8;
    numbers[8] = 9;
    numbers[9] = 10;

    printf("Original list: ");
    for (i = 0; i < size; i++) {
        printf("%d ", numbers[i]);
    }
    printf("\n");

    new_size = remove_odd_numbers(numbers, size);

    printf("List after removing odd numbers: ");
    for (i = 0; i < new_size; i++) {
        printf("%d ", numbers[i]);
    }
    printf("\n");

    free(numbers);
    numbers = NULL;

    return EXIT_SUCCESS;
}
```

Key safety measures in this code:

1. **CWE-664 (Resource Lifetime)**: The dynamically allocated memory from `malloc` is always freed with `free()` before program exit, and the pointer is set to `NULL` afterward to prevent accidental use-after-free.

2. **CWE-703 (Exceptional Conditions)**: The return value of `malloc` is checked against `NULL` before use, and the function itself validates its pointer argument before dereferencing it. Errors are reported to `stderr` with a proper failure exit code.

3. **CWE-710 (Coding Standards)**: Uses `int main(void)` with an explicit return value, `size_t` for sizes and indices to avoid signed/unsigned mismatches, `EXIT_SUCCESS`/`EXIT_FAILURE` macros instead of magic numbers, and a proper function signature with validated parameters.
