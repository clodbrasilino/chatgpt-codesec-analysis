```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U

long long count_substrings(const char *s, size_t max_len);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n == max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;
    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        prefix_sum += (long long)(s[i] - '0') - 1;
        count += freq[prefix_sum + offset];
        freq[prefix_sum + offset]++;
    }

    free(freq);
    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    printf("Enter a digit string: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof buffer, stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof buffer - 1U] = '\0';
    buffer[strcspn(buffer, "\r\n")] = '\0';

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("Number of substrings: %lld\n", result);
    return EXIT_SUCCESS;
}
```

**Summary of hardening changes:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by an explicit `max_len` parameter passed from the caller. If the string is not NUL-terminated within the given bound, the function returns `-1` instead of reading past the buffer.

2. **CWE-119/CWE-120 (statically-sized `char` array):** The buffer size is defined via a single named constant (`MAX_INPUT_LEN`), reads are bounded by `fgets(buffer, sizeof buffer, ...)`, and a defensive `buffer[sizeof buffer - 1U] = '\0'` guarantees NUL-termination before any string operations. The buffer's capacity is also propagated to `count_substrings` so the callee cannot over-read.

3. **Additional defense:** Added an overflow guard on `table_size * sizeof *freq` before `calloc`, since some legacy allocators do not check the multiplication internally.

The core algorithm (counting substrings whose digit sum equals their length, via prefix sums and a frequency table) is preserved unchanged; index bounds remain valid since `prefix_sum + offset` always lies in `[0, 9n]`, within the `10n + 1` table.
