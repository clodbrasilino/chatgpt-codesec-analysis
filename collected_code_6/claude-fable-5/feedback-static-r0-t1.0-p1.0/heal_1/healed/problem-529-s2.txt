#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static int jacobsthal_lucas(unsigned int n, unsigned long long *result)
{
    unsigned long long prev;
    unsigned long long curr;
    unsigned int i;

    if (result == NULL) {
        return -1;
    }

    if (n == 0U) {
        *result = 2ULL;
        return 0;
    }

    if (n == 1U) {
        *result = 1ULL;
        return 0;
    }

    prev = 2ULL;
    curr = 1ULL;

    for (i = 2U; i <= n; i++) {
        unsigned long long next;

        if (prev > (ULLONG_MAX / 2ULL)) {
            return -1;
        }
        if (curr > (ULLONG_MAX - (2ULL * prev))) {
            return -1;
        }
        next = curr + (2ULL * prev);
        prev = curr;
        curr = next;
    }

    *result = curr;
    return 0;
}

int main(void)
{
    unsigned int n;
    unsigned long long value;
    int scan_status;

    printf("Enter n: ");

    scan_status = scanf("%u", &n);
    if (scan_status != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (jacobsthal_lucas(n, &value) != 0) {
        fprintf(stderr, "Computation failed (overflow or invalid argument)\n");
        return EXIT_FAILURE;
    }

    printf("Jacobsthal-Lucas number %u is %llu\n", n, value);

    return EXIT_SUCCESS;
}

The fix applied addresses the `[variableScope]` weakness: the variable `next` is now declared inside the `for` loop body, reducing its scope to the smallest block where it is actually used. This improves readability and prevents accidental use of a stale value outside the loop. The rest of the code was already sound — it properly guards against integer overflow before each multiplication and addition, validates the output pointer, handles the base cases `n == 0` and `n == 1`, and checks the return value of `scanf`.
