#include <stdio.h>
#include <stddef.h>

static const char *check_last(const int arr[], size_t n, int p)
{
    long long sum = 0LL;
    size_t i;

    if ((arr == NULL) || (n == 0U)) {
        return "INVALID";
    }

    for (i = 0U; i < n; i++) {
        sum += (long long)arr[i];
    }

    if (p == 1) {
        if ((sum % 2LL) == 0LL) {
            return "ODD";
        }
        return "EVEN";
    }

    return "EVEN";
}

int main(void)
{
    int arr1[] = {5, 7, 10};
    int arr2[] = {2, 3};
    int arr3[] = {1, 2, 3};
    size_t n1 = sizeof(arr1) / sizeof(arr1[0]);
    size_t n2 = sizeof(arr2) / sizeof(arr2[0]);
    size_t n3 = sizeof(arr3) / sizeof(arr3[0]);
    const char *r1;
    const char *r2;
    const char *r3;

    r1 = check_last(arr1, n1, 1);
    r2 = check_last(arr2, n2, 1);
    r3 = check_last(arr3, n3, 2);

    if (printf("%s\n", r1) < 0) {
        return 1;
    }
    if (printf("%s\n", r2) < 0) {
        return 1;
    }
    if (printf("%s\n", r3) < 0) {
        return 1;
    }

    return 0;
}
