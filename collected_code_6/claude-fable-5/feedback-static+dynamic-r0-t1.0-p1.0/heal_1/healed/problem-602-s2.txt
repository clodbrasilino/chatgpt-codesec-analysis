```c
#include <stdio.h>
#include <string.h>

#define MAX_INPUT 256

int find_first_repeated(const char *str, size_t max_len, char *result)
{
    int seen[256] = {0};
    size_t i;
    size_t len;

    if (str == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c]) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[MAX_INPUT];
    char repeated;
    int status;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    status = find_first_repeated(buffer, sizeof(buffer), &repeated);
    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    return 0;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen`, which takes an explicit maximum length and cannot read past the buffer bounds even if the string is not null-terminated. The `find_first_repeated` function now accepts a `max_len` parameter so the caller communicates the buffer size.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is defined once via `MAX_INPUT` and consistently referenced with `sizeof(buffer)`. Additionally, `buffer[sizeof(buffer) - 1] = '\0';` guarantees null-termination as a defensive measure, even though `fgets` normally terminates the string.

3. **Existing protections preserved**: `fgets` with `sizeof(buffer)` already prevents input overflow, the NULL checks remain, and the `unsigned char` cast prevents negative indexing into the `seen` array on platforms where `char` is signed.
