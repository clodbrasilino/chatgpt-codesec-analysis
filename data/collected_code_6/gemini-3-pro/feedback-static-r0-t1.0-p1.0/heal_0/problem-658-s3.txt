#include <stdio.h>
#include <stdlib.h>

struct ElementCount {
    int value;
    int count;
};

int findMaxOccurrence(const int *list, size_t size, int *result) {
    if (list == NULL || size == 0 || result == NULL) {
        return -1;
    }

    struct ElementCount *counts = malloc(size * sizeof(struct ElementCount));
    if (counts == NULL) {
        return -1;
    }

    size_t unique_elements = 0;

    for (size_t i = 0; i < size; i++) {
        int found = 0;
        for (size_t j = 0; j < unique_elements; j++) {
            if (counts[j].value == list[i]) {
                counts[j].count++;
                found = 1;
                break;
            }
        }
        if (!found) {
            counts[unique_elements].value = list[i];
            counts[unique_elements].count = 1;
            unique_elements++;
        }
    }

    int max_count = 0;
    for (size_t i = 0; i < unique_elements; i++) {
        if (counts[i].count > max_count) {
            max_count = counts[i].count;
            *result = counts[i].value;
        }
    }

    free(counts);
    return 0;
}

int main(void) {
    int list[] = {1, 3, 2, 3, 4, 1, 3, 2, 3, 5};
    size_t size = sizeof(list) / sizeof(list[0]);
    int result = 0;

    if (findMaxOccurrence(list, size, &result) == 0) {
        printf("Item with maximum occurrences: %d\n", result);
    } else {
        printf("Error finding maximum occurrence or list is empty.\n");
    }

    return 0;
}
