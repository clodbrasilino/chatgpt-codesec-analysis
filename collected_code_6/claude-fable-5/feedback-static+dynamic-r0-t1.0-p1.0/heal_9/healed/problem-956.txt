#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

static void free_tokens(char **tokens, size_t count)
{
    if (tokens == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

static char *copy_token(const char *src, size_t len)
{
    char *tok;
    size_t dest_size;

    if (src == NULL || len >= SIZE_MAX) {
        return NULL;
    }
    dest_size = len + 1;
    tok = malloc(dest_size);
    if (tok == NULL) {
        return NULL;
    }
    if (len > dest_size - 1) {
        free(tok);
        return NULL;
    }
    memcpy(tok, src, len);
    tok[len] = '\0';
    return tok;
}

static int append_token(char ***tokens, size_t *n, char *tok)
{
    char **tmp;

    if (tokens == NULL || n == NULL || tok == NULL) {
        return -1;
    }
    if (*n >= SIZE_MAX / sizeof(**tokens)) {
        return -1;
    }
    tmp = realloc(*tokens, (*n + 1) * sizeof(**tokens));
    if (tmp == NULL) {
        return -1;
    }
    *tokens = tmp;
    tmp[*n] = tok;
    (*n)++;
    return 0;
}

static char **split_at_uppercase(const char *input, size_t input_max, size_t *count)
{
    regex_t re;
    regmatch_t match;
    char **tokens = NULL;
    size_t n = 0;
    const char *cursor;
    size_t remaining;

    if (input == NULL || count == NULL || input_max == 0) {
        return NULL;
    }

    remaining = strnlen(input, input_max);
    if (remaining >= input_max) {
        return NULL;
    }
    cursor = input;

    if (regcomp(&re, "[A-Z]", REG_EXTENDED) != 0) {
        return NULL;
    }

    while (regexec(&re, cursor, 1, &match, 0) == 0) {
        size_t len;
        char *tok;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so ||
            (size_t)match.rm_eo > remaining) {
            goto error;
        }

        if (match.rm_eo == match.rm_so) {
            goto error;
        }

        len = (size_t)match.rm_so;

        if (len > remaining) {
            goto error;
        }

        tok = copy_token(cursor, len);
        if (tok == NULL) {
            goto error;
        }

        if (append_token(&tokens, &n, tok) != 0) {
            free(tok);
            goto error;
        }

        cursor += match.rm_eo;
        remaining -= (size_t)match.rm_eo;
    }

    {
        char *tok = copy_token(cursor, remaining);
        if (tok == NULL) {
            goto error;
        }
        if (append_token(&tokens, &n, tok) != 0) {
            free(tok);
            goto error;
        }
    }

    regfree(&re);
    *count = n;
    return tokens;

error:
    free_tokens(tokens, n);
    regfree(&re);
    return NULL;
}

int main(void)
{
    const char *text = "helloWorldThisIsRegexSplit";
    size_t count = 0;
    char **tokens = split_at_uppercase(text, 4096, &count);

    if (tokens == NULL) {
        if (fprintf(stderr, "Failed to split string\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < count; i++) {
        if (printf("Token %zu: \"%s\"\n", i, tokens[i]) < 0) {
            free_tokens(tokens, count);
            return EXIT_FAILURE;
        }
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe:

**The `memcpy` finding (CWE-120):** Flawfinder flags every `memcpy` call because it cannot statically verify the destination size. In `copy_token`, the call is already safe by construction:

1. `len >= SIZE_MAX` is rejected upfront, so `len + 1` cannot overflow when computing `dest_size`.
2. The destination buffer is allocated with exactly `dest_size = len + 1` bytes, and `malloc` failure is handled.
3. The guard `len > dest_size - 1` is a redundant defensive check (it can never trigger given the arithmetic, but it satisfies static analyzers and protects against future refactoring mistakes).
4. Exactly `len` bytes are copied into a buffer of `len + 1`, leaving room for the NUL terminator written afterward.

**Other defensive measures already present:**

- `strnlen` with an explicit `input_max` bound prevents reading unterminated input; the `remaining >= input_max` check rejects strings without a NUL inside the bound.
- `append_token` guards the `realloc` size computation against `size_t` overflow (`*n >= SIZE_MAX / sizeof(**tokens)`) and preserves the original pointer on failure, avoiding a leak.
- Regex match offsets (`rm_so`, `rm_eo`) are validated against negative values, inversion, and the remaining buffer length before any pointer arithmetic.
- Zero-width match detection (`rm_eo == rm_so`) prevents an infinite loop, though `[A-Z]` cannot produce one.
- All error paths funnel through a single `error:` label that frees accumulated tokens and calls `regfree`, so no resources leak.
- Return values of `printf`/`fprintf` are checked in `main`.
