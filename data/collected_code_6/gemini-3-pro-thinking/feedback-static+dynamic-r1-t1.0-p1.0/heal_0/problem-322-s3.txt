#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

size_t *find_minimum_indices(const int *array, size_t length, size_t *out_count);

size_t *find_minimum_indices(const int *array, size_t length, size_t *out_count) {
    if (array == NULL || out_count == NULL) {
        return NULL;
    }

    if (length == 0) {
        *out_count = 0;
        return NULL;
    }

    int min_value = array[0];
    size_t count = 1;

    for (size_t i = 1; i < length; i++) {
        if (array[i] < min_value) {
            min_value = array[i];
            count = 1;
        } else if (array[i] == min_value) {
            count++;
        }
    }

    size_t *indices = (size_t *)malloc(count * sizeof(size_t));
    if (indices == NULL) {
        *out_count = 0;
        return NULL;
    }

    size_t current_index = 0;
    for (size_t i = 0; i < length; i++) {
        if (array[i] == min_value) {
            indices[current_index] = i;
            current_index++;
        }
    }

    *out_count = count;
    return indices;
}

int main(void) {
    const int data[] = {7, 3, 5, 3, 9, 3, 8};
    const size_t length = sizeof(data) / sizeof(data[0]);
    size_t count = 0;

    size_t *indices = find_minimum_indices(data, length, &count);

    if (indices != NULL) {
        for (size_t i = 0; i < count; i++) {
            printf("%zu\n", indices[i]);
        }
        free(indices);
        indices = NULL;
    }

    return EXIT_SUCCESS;
}
