#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 9
#define BUFFER_SIZE 64

static int find_demlo(const char *input, size_t input_max, char *output, size_t output_size);

static int find_demlo(const char *input, size_t input_max, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, input_max);

    if (len == 0U || len >= input_max || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    i = len - 1U;

    while (i >= 1U)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
        i--;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;

        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, sizeof(input), result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already well-hardened, and the Flawfinder hits are informational rather than actual defects:

1. **CWE-119/CWE-120 on `input` and `result` buffers** — These are false positives in context. Both buffers are zero-initialized with `memset`, `fgets` is bounded by `sizeof(input)`, an explicit null terminator is forced at `sizeof(input) - 1U`, and `strnlen` (not `strlen`) is used everywhere, guaranteeing bounded reads.

2. **CWE-120/CWE-20 on `getchar` loop** — Also a false positive: the loop only consumes and discards excess characters until newline or EOF to flush stdin; nothing is stored, so no buffer can overflow. The `EOF` check prevents an infinite loop on closed input.

3. **`find_demlo` bounds safety** — The function validates all pointers, checks `output_size` against the required `2 * len - 1` characters plus terminator up front, and additionally re-checks `pos >= output_size - 1U` before every single write, making an out-of-bounds write impossible even if the earlier size check were wrong.

4. **Integer safety** — `len` is capped at `MAX_INPUT_LEN` (9), so `'0' + i` never exceeds `'9'` and no `char` overflow occurs; `len - 1U` cannot underflow because `len == 0` is rejected first.

Since every flagged risk is already mitigated by existing bounds checking, the rewrite preserves the logic unchanged with the advisory comments removed.
