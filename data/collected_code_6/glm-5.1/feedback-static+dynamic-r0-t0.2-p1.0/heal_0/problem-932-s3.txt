#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **remove_duplicate_words(char **words, int count, int *new_count) {
    if (words == NULL || new_count == NULL || count <= 0) {
        if (new_count != NULL) {
            *new_count = 0;
        }
        return NULL;
    }

    char **result = (char **)malloc(count * sizeof(char *));
    if (result == NULL) {
        *new_count = 0;
        return NULL;
    }

    int unique_count = 0;

    for (int i = 0; i < count; i++) {
        if (words[i] == NULL) {
            continue;
        }

        int is_duplicate = 0;
        for (int j = 0; j < unique_count; j++) {
            if (strcmp(words[i], result[j]) == 0) {
                is_duplicate = 1;
                break;
            }
        }

        if (!is_duplicate) {
            result[unique_count] = words[i];
            unique_count++;
        }
    }

    char **shrunk_result = (char **)realloc(result, unique_count * sizeof(char *));
    if (shrunk_result != NULL) {
        result = shrunk_result;
    }

    *new_count = unique_count;
    return result;
}

int main(void) {
    char *words[] = {"apple", "banana", "apple", "orange", "banana", "grape", "orange"};
    int count = sizeof(words) / sizeof(words[0]);
    int new_count = 0;

    char **unique_words = remove_duplicate_words(words, count, &new_count);

    if (unique_words != NULL) {
        for (int i = 0; i < new_count; i++) {
            printf("%s\n", unique_words[i]);
        }
        free(unique_words);
    }

    return 0;
}
