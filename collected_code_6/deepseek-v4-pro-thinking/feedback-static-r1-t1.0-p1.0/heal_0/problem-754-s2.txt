#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <stdint.h>

int find_common_indices(const int *a, size_t len_a,
                        const int *b, size_t len_b,
                        const int *c, size_t len_c,
                        int **out_indices, size_t *out_count)
{
    size_t len;
    size_t i;
    size_t count;
    size_t index;
    int *result;

    if (out_indices == NULL || out_count == NULL) {
        return -1;
    }

    *out_indices = NULL;
    *out_count = 0;

    len = len_a < len_b ? len_a : len_b;
    if (len_c < len) {
        len = len_c;
    }

    if (len == 0) {
        return 0;
    }

    if (a == NULL || b == NULL || c == NULL) {
        return -1;
    }

    if (len > INT_MAX) {
        return -1;
    }

    count = 0;
    for (i = 0; i < len; ++i) {
        if (a[i] == b[i] && b[i] == c[i]) {
            ++count;
        }
    }

    if (count == 0) {
        return 0;
    }

    if (count > SIZE_MAX / sizeof(*result)) {
        return -1;
    }

    result = malloc(count * sizeof(*result));
    if (result == NULL) {
        return -1;
    }

    index = 0;
    for (i = 0; i < len; ++i) {
        if (a[i] == b[i] && b[i] == c[i]) {
            result[index] = (int)i;
            ++index;
        }
    }

    *out_indices = result;
    *out_count = count;
    return 0;
}

int main(void)
{
    int a[] = {1, 2, 3, 4, 5};
    int b[] = {0, 2, 3, 9, 5};
    int c[] = {1, 2, 4, 4, 5};
    int *indices = NULL;
    size_t count = 0;
    size_t i;
    int status;

    status = find_common_indices(a, sizeof(a) / sizeof(a[0]),
                                 b, sizeof(b) / sizeof(b[0]),
                                 c, sizeof(c) / sizeof(c[0]),
                                 &indices, &count);
    if (status != 0) {
        return 1;
    }

    for (i = 0; i < count; ++i) {
        printf("%d\n", indices[i]);
    }

    free(indices);
    return 0;
}
