#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>
#include <ctype.h>

long long largest_prime_factor(long long n)
{
    long long largest = -1;
    long long i;

    if (n <= 1)
        return -1;

    while (n % 2 == 0) {
        largest = 2;
        n /= 2;
    }

    for (i = 3; i <= n / i; i += 2) {
        while (n % i == 0) {
            largest = i;
            n /= i;
        }
    }

    if (n > 2)
        largest = n;

    return largest;
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t len;
    char *endptr;
    long long number;
    long long result;

    len = getline(&buffer, &buffer_size, stdin);
    if (len == -1) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    errno = 0;
    number = strtoll(buffer, &endptr, 10);

    if (errno == ERANGE) {
        fprintf(stderr, "Number out of range\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (endptr == buffer) {
        fprintf(stderr, "Invalid input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    while (isspace((unsigned char)*endptr))
        endptr++;

    if (*endptr != '\0') {
        fprintf(stderr, "Invalid input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (number == LLONG_MIN) {
        fprintf(stderr, "Number out of range\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (number < 0)
        number = -number;

    result = largest_prime_factor(number);

    if (result == -1)
        printf("No prime factors\n");
    else
        printf("%lld\n", result);

    free(buffer);
    return EXIT_SUCCESS;
}
