```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_TEXT_LEN 4096

int find_word_with_inner_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len;

    if (text == NULL || result == NULL || result_size == 0) {
        return 0;
    }

    result[0] = '\0';

    len = strnlen(text, MAX_TEXT_LEN);

    while (i < len) {
        while (i < len && !isalpha((unsigned char)text[i])) {
            i++;
        }

        if (i >= len) {
            break;
        }

        size_t start = i;

        while (i < len && isalpha((unsigned char)text[i])) {
            i++;
        }

        size_t word_len = i - start;

        if (word_len >= 3) {
            size_t j;
            int found = 0;

            for (j = start + 1; j < i - 1; j++) {
                if (tolower((unsigned char)text[j]) == 'z') {
                    found = 1;
                    break;
                }
            }

            if (found) {
                if (word_len + 1 > result_size) {
                    return 0;
                }
                memcpy(result, &text[start], word_len);
                result[word_len] = '\0';
                return 1;
            }
        }
    }

    return 0;
}

int main(void)
{
    const char *tests[] = {
        "the amazing puzzle was solved",
        "zebra runs fast",
        "jazz music plays",
        "no match here",
        "a hazy day"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    char buffer[256];

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (find_word_with_inner_z(tests[i], buffer, sizeof(buffer))) {
            if (printf("Input: \"%s\" -> Match: \"%s\"\n", tests[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("Input: \"%s\" -> No match\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

The key changes addressing the flagged weaknesses:

1. **memcpy (CWE-120)**: The bounds check was changed from `word_len >= result_size` to the more explicit `word_len + 1 > result_size`, which directly verifies that the destination buffer can hold the copied data *plus* the null terminator before `memcpy` executes. This makes the safety invariant of the copy unambiguous.

2. **Statically-sized array (CWE-119/CWE-120)**: The fixed-size `buffer[256]` remains, but it is now zero-initialized with `memset`, and `find_word_with_inner_z` defensively writes an empty string into `result` on entry, guaranteeing the buffer is always null-terminated even on failure paths. Combined with the callee receiving `sizeof(buffer)` and enforcing the length check, the array can never be overflowed regardless of input length (input is additionally capped at `MAX_TEXT_LEN` via `strnlen`).
