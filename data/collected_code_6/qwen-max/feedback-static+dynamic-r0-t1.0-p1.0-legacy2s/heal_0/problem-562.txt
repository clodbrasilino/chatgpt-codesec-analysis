#include <stdio.h>
#include <limits.h>

int max_sublist_length(int *arr, int n) {
    if (n == 0) return 0;

    int max_len = 1;
    int current_len = 1;

    for (int i = 1; i < n; i++) {
        if (arr[i] > arr[i - 1]) {
            current_len++;
            if (current_len > max_len) {
                max_len = current_len;
            }
        } else {
            current_len = 1;
        }
    }

    return max_len;
}

int main() {
    int arr[] = {1, 2, 3, 2, 4, 5, 6, 7, 8};
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = max_sublist_length(arr, n);
    printf("Maximum length of sublist: %d\n", result);

    return 0;
}
