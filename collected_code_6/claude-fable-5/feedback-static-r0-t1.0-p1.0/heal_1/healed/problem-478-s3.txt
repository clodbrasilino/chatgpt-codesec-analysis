```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test1 = "HelloWorldABC";
    const char *test2 = "KDeoALOklOOHserfLoAJSIskdsf";
    char *output1;
    char *output2;

    output1 = remove_lowercase(test1);
    if (output1 == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    output2 = remove_lowercase(test2);
    if (output2 == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(output1);
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", test1);
    printf("Result:   %s\n", output1);
    printf("Original: %s\n", test2);
    printf("Result:   %s\n", output2);

    free(output1);
    free(output2);

    return EXIT_SUCCESS;
}
```

The key change addresses the flagged CWE-126 weakness: `strlen` was replaced with `strnlen` bounded by `MAX_INPUT_LEN`, which prevents an over-read on strings that are not properly NUL-terminated. If the input reaches the bound (meaning no terminator was found within the limit), the function safely rejects it by returning `NULL` rather than processing potentially unterminated data. The existing good practices were preserved: NULL input checks, `malloc` failure handling, casting to `unsigned char` before calling `islower` (avoiding undefined behavior with negative `char` values), explicit NUL-termination of the result, and proper cleanup of allocated memory on all exit paths in `main`.
