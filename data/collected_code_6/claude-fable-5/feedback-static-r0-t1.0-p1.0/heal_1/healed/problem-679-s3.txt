```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DICT_MAX_STRING_LEN 4096

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dict;

static char *dict_strdup_bounded(const char *s)
{
    size_t len;
    char *copy;

    if (s == NULL) {
        return NULL;
    }

    len = strnlen(s, DICT_MAX_STRING_LEN);
    if (len >= DICT_MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, s, len);
    copy[len] = '\0';
    return copy;
}

static Dict *dict_create(size_t capacity)
{
    Dict *d;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(DictEntry)) {
        return NULL;
    }

    d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }

    d->entries = malloc(capacity * sizeof(DictEntry));
    if (d->entries == NULL) {
        free(d);
        return NULL;
    }

    d->size = 0;
    d->capacity = capacity;
    return d;
}

static int dict_add(Dict *d, const char *key, const char *value)
{
    char *key_copy;
    char *value_copy;

    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }

    if (d->size >= d->capacity) {
        return -1;
    }

    key_copy = dict_strdup_bounded(key);
    if (key_copy == NULL) {
        return -1;
    }

    value_copy = dict_strdup_bounded(value);
    if (value_copy == NULL) {
        free(key_copy);
        return -1;
    }

    d->entries[d->size].key = key_copy;
    d->entries[d->size].value = value_copy;
    d->size++;
    return 0;
}

static const DictEntry *dict_get_by_index(const Dict *d, size_t index)
{
    if (d == NULL || index >= d->size) {
        return NULL;
    }
    return &d->entries[index];
}

static void dict_destroy(Dict *d)
{
    size_t i;

    if (d == NULL) {
        return;
    }

    for (i = 0; i < d->size; i++) {
        free(d->entries[i].key);
        free(d->entries[i].value);
    }
    free(d->entries);
    free(d);
}

int main(void)
{
    Dict *d;
    const DictEntry *entry;
    size_t i;

    d = dict_create(8);
    if (d == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }

    if (dict_add(d, "apple", "fruit") != 0 ||
        dict_add(d, "carrot", "vegetable") != 0 ||
        dict_add(d, "salmon", "fish") != 0) {
        fprintf(stderr, "Failed to add entries\n");
        dict_destroy(d);
        return EXIT_FAILURE;
    }

    for (i = 0; i < d->size; i++) {
        entry = dict_get_by_index(d, i);
        if (entry != NULL) {
            printf("Index %zu: key=%s, value=%s\n", i, entry->key, entry->value);
        }
    }

    entry = dict_get_by_index(d, 100);
    if (entry == NULL) {
        printf("Index 100 is out of range\n");
    }

    dict_destroy(d);
    return EXIT_SUCCESS;
}
```

**Summary of changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `DICT_MAX_STRING_LEN`, so a non-terminated string can no longer cause an unbounded read. Strings at or beyond the limit are rejected rather than truncated silently.

2. **CWE-120 (`strcpy` buffer overflow):** Replaced `strcpy` with `memcpy` using the exact, already-validated length, followed by explicit null termination. Since the destination buffer is allocated as `len + 1`, the copy can never overflow.

3. **Refactoring:** Both duplication sites were consolidated into a single helper, `dict_strdup_bounded`, eliminating duplicated risky code paths and centralizing the bounds checking.

4. **Defensive hardening (bonus):** Added an overflow guard in `dict_create` (`capacity > SIZE_MAX / sizeof(DictEntry)`) to prevent integer overflow in the allocation size calculation on very large capacity requests.

The program's observable behavior is unchanged: it prints the three entries by index, reports that index 100 is out of range, and cleans up all allocations before exiting.
