#include <stdio.h>
#include <stdbool.h>
#include <ctype.h>
#include <string.h>

bool contains_all_vowels(const char *str) {
    bool vowels[5] = {false, false, false, false, false};
    size_t i;
    
    if (str == NULL) {
        return false;
    }
    
    for (i = 0; str[i] != '\0'; i++) {
        char c = tolower((unsigned char)str[i]);
        
        switch (c) {
            case 'a':
                vowels[0] = true;
                break;
            case 'e':
                vowels[1] = true;
                break;
            case 'i':
                vowels[2] = true;
                break;
            case 'o':
                vowels[3] = true;
                break;
            case 'u':
                vowels[4] = true;
                break;
            default:
                break;
        }
    }
    
    for (int j = 0; j < 5; j++) {
        if (!vowels[j]) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    char input[256];
    size_t len;
    bool newline_found = false;
    
    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }
    
    len = strlen(input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        newline_found = true;
    }
    
    if (!newline_found) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
    }
    
    if (contains_all_vowels(input)) {
        printf("The string contains all vowels.\n");
    } else {
        printf("The string does not contain all vowels.\n");
    }
    
    return 0;
}
