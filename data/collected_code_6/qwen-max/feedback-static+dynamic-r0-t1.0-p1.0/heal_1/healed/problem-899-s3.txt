#include <stdio.h>
#include <stdbool.h>

bool canSortByCorners(const int arr[], int n) {
    if (n <= 1) return true;

    int start = 0, end = n - 1;
    bool isLeftSmaller = arr[start] < arr[end];

    while (start < end) {
        if (isLeftSmaller && arr[start] > arr[start + 1]) return false;
        if (!isLeftSmaller && arr[end] < arr[end - 1]) return false;

        if (isLeftSmaller)
            ++start;
        else
            --end;

        isLeftSmaller = arr[start] < arr[end];
    }

    return true;
}

int main() {
    const int arr1[] = {3, 2, 1, 4, 5};
    const int arr2[] = {1, 3, 2, 4, 5};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int n2 = sizeof(arr2) / sizeof(arr2[0]);

    printf("Array 1 can be sorted by corners: %s\n", canSortByCorners(arr1, n1) ? "true" : "false");
    printf("Array 2 can be sorted by corners: %s\n", canSortByCorners(arr2, n2) ? "true" : "false");

    return 0;
}
