#include <stdio.h>
#include <stdlib.h>
#include <math.h>

void findPerfectSquares(int start, int end) {
    if (start > end) {
        return;
    }
    
    int current = start;
    if (current < 0) {
        current = 0;
    }
    
    double root = sqrt((double)current);
    int int_root = (int)ceil(root);
    
    while (1) {
        long long square = (long long)int_root * int_root;
        if (square > end) {
            break;
        }
        printf("%lld\n", square);
        int_root++;
    }
}

int main(void) {
    int start = 1;
    int end = 100;
    
    findPerfectSquares(start, end);
    
    return 0;
}
