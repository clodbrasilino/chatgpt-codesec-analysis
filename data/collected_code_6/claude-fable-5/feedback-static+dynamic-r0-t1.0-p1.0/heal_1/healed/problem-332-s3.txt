```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define CHARSET_SIZE (UCHAR_MAX + 1)
#define INPUT_SIZE 256

int count_char_frequency(const char *str, size_t str_size, size_t freq[CHARSET_SIZE])
{
    size_t i;

    if (str == NULL || freq == NULL) {
        return -1;
    }

    for (i = 0; i < CHARSET_SIZE; i++) {
        freq[i] = 0;
    }

    for (i = 0; i < str_size && str[i] != '\0'; i++) {
        freq[(unsigned char)str[i]]++;
    }

    if (i == str_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    size_t freq[CHARSET_SIZE];
    size_t i;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (count_char_frequency(input, sizeof(input), freq) != 0) {
        fprintf(stderr, "Failed to count character frequency\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < CHARSET_SIZE; i++) {
        if (freq[i] > 0) {
            if (i >= 32 && i < 127) {
                printf("'%c' : %zu\n", (int)i, freq[i]);
            } else {
                printf("0x%02zX : %zu\n", i, freq[i]);
            }
        }
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-119/CWE-120 (buffer bounds)**: `count_char_frequency` now takes an explicit `str_size` parameter and bounds every read with `i < str_size`, so it can never over-read even if passed a non-terminated buffer. It returns an error if no terminator is found within bounds.

2. **CWE-126 (`strlen` over-read)**: Replaced `strlen` with `strnlen(input, sizeof(input))`, which is bounded, and added a defensive `input[sizeof(input) - 1] = '\0'` to guarantee termination regardless of what `fgets` did.

3. **Minor hardening**: Added `fflush(stdout)` so the prompt is guaranteed to appear before blocking on input, and replaced the magic number `256` with a named constant `INPUT_SIZE` to keep the buffer size and its uses consistent.

The original code was already largely safe (`fgets` bounds the read and null-terminates on success), so these changes are defense-in-depth measures that make the safety invariants explicit rather than implicit.
