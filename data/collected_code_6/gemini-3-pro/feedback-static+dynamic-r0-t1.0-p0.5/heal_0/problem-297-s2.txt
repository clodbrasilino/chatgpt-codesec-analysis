#include <stdio.h>
#include <stdlib.h>

typedef struct NestedList {
    int is_integer;
    int value;
    struct NestedList** list;
    size_t list_size;
} NestedList;

typedef struct FlatList {
    int* data;
    size_t size;
    size_t capacity;
} FlatList;

FlatList* create_flat_list(size_t initial_capacity) {
    FlatList* flat = (FlatList*)malloc(sizeof(FlatList));
    if (!flat) return NULL;
    
    flat->data = (int*)malloc(initial_capacity * sizeof(int));
    if (!flat->data) {
        free(flat);
        return NULL;
    }
    
    flat->size = 0;
    flat->capacity = initial_capacity;
    return flat;
}

void free_flat_list(FlatList* flat) {
    if (flat) {
        free(flat->data);
        free(flat);
    }
}

int append_to_flat_list(FlatList* flat, int value) {
    if (!flat) return 0;
    
    if (flat->size >= flat->capacity) {
        size_t new_capacity = flat->capacity == 0 ? 4 : flat->capacity * 2;
        int* new_data = (int*)realloc(flat->data, new_capacity * sizeof(int));
        if (!new_data) return 0;
        
        flat->data = new_data;
        flat->capacity = new_capacity;
    }
    
    flat->data[flat->size++] = value;
    return 1;
}

int flatten_helper(const NestedList* nested, FlatList* flat) {
    if (!nested || !flat) return 0;
    
    if (nested->is_integer) {
        return append_to_flat_list(flat, nested->value);
    } else {
        for (size_t i = 0; i < nested->list_size; ++i) {
            if (!flatten_helper(nested->list[i], flat)) {
                return 0;
            }
        }
    }
    return 1;
}

FlatList* flatten(const NestedList* nested) {
    if (!nested) return NULL;
    
    FlatList* flat = create_flat_list(16);
    if (!flat) return NULL;
    
    if (!flatten_helper(nested, flat)) {
        free_flat_list(flat);
        return NULL;
    }
    
    return flat;
}

NestedList* create_integer(int value) {
    NestedList* node = (NestedList*)malloc(sizeof(NestedList));
    if (!node) return NULL;
    
    node->is_integer = 1;
    node->value = value;
    node->list = NULL;
    node->list_size = 0;
    return node;
}

NestedList* create_list(NestedList** list, size_t size) {
    NestedList* node = (NestedList*)malloc(sizeof(NestedList));
    if (!node) return NULL;
    
    node->is_integer = 0;
    node->value = 0;
    node->list = list;
    node->list_size = size;
    return node;
}

void free_nested_list(NestedList* nested) {
    if (!nested) return;
    
    if (!nested->is_integer && nested->list) {
        for (size_t i = 0; i < nested->list_size; ++i) {
            free_nested_list(nested->list[i]);
        }
        free(nested->list);
    }
    free(nested);
}

int main(void) {
    NestedList* val1 = create_integer(1);
    NestedList* val2 = create_integer(2);
    NestedList* val3 = create_integer(3);
    
    if (!val1 || !val2 || !val3) {
        free_nested_list(val1);
        free_nested_list(val2);
        free_nested_list(val3);
        return 1;
    }
    
    NestedList** inner_arr = (NestedList**)malloc(2 * sizeof(NestedList*));
    if (!inner_arr) {
        free_nested_list(val1);
        free_nested_list(val2);
        free_nested_list(val3);
        return 1;
    }
    inner_arr[0] = val1;
    inner_arr[1] = val2;
    
    NestedList* inner_list = create_list(inner_arr, 2);
    if (!inner_list) {
        free(inner_arr);
        free_nested_list(val1);
        free_nested_list(val2);
        free_nested_list(val3);
        return 1;
    }
    
    NestedList** outer_arr = (NestedList**)malloc(2 * sizeof(NestedList*));
    if (!outer_arr) {
        free_nested_list(inner_list);
        free_nested_list(val3);
        return 1;
    }
    outer_arr[0] = inner_list;
    outer_arr[1] = val3;
    
    NestedList* root = create_list(outer_arr, 2);
    if (!root) {
        free(outer_arr);
        free_nested_list(inner_list);
        free_nested_list(val3);
        return 1;
    }
    
    FlatList* flat = flatten(root);
    if (flat) {
        for (size_t i = 0; i < flat->size; ++i) {
            printf("%d ", flat->data[i]);
        }
        printf("\n");
        free_flat_list(flat);
    }
    
    free_nested_list(root);
    return 0;
}
