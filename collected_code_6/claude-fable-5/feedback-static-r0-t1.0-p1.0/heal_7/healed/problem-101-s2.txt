#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int find_kth_element(const int *arr, size_t n, size_t k, int *result)
{
    int *copy;
    size_t total_size;

    if (arr == NULL || result == NULL || n == 0 || k == 0 || k > n) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*copy)) {
        return -1;
    }

    total_size = n * sizeof(*copy);

    copy = malloc(total_size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, total_size);
    qsort(copy, n, sizeof(*copy), compare_ints);

    *result = copy[k - 1];
    free(copy);

    return 0;
}

int main(void)
{
    int arr[] = { 12, 3, 5, 7, 4, 19, 26 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 3;
    int result = 0;

    if (find_kth_element(arr, n, k, &result) != 0) {
        fprintf(stderr, "Error: invalid input or allocation failure\n");
        return EXIT_FAILURE;
    }

    if (printf("The %zu-th smallest element is %d\n", k, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) is a heuristic hit rather than an actual vulnerability in this code. The original source already contains the correct mitigations, so the rewritten code preserves them unchanged:

1. **Destination size is guaranteed to hold the source data** — `copy` is allocated with exactly `total_size` bytes, and `memcpy` copies exactly `total_size` bytes. The destination cannot be smaller than the copy length.

2. **Integer overflow in the size calculation is prevented** — the check `n > SIZE_MAX / sizeof(*copy)` ensures `n * sizeof(*copy)` cannot wrap around, which would otherwise cause a short allocation followed by a heap overflow.

3. **Allocation failure is handled** — `malloc` is checked for `NULL` before use, so `memcpy` never writes through a null pointer.

4. **Input validation is complete** — `NULL` pointers, `n == 0`, `k == 0`, and `k > n` are all rejected, preventing out-of-bounds reads on `copy[k - 1]`.

5. **No memory leaks** — `copy` is freed on the success path, and no allocation exists on the error paths.
