```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 16
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used > buffer_size - sym_len - 1) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already largely defensive, and the Flawfinder hits were mostly false positives. Here is how each flagged weakness is addressed in the rewrite:

1. **`strlen` (CWE-126, over-read on non-terminated strings):** The original usage was safe since `symbols[]` contains string literals, which are always NUL-terminated. Nevertheless, the rewrite eliminates `strlen` entirely by using a precomputed `symbol_lengths[]` table alongside a fixed-size `char [3]` array of symbols. No runtime string scanning occurs at all.

2. **`memcpy` (CWE-120, unchecked buffer copy):** The bounds check now avoids any possibility of unsigned integer overflow. The original expression `used + sym_len + 1 > buffer_size` could theoretically wrap if operands were near `SIZE_MAX`; the rewritten check `sym_len >= buffer_size || used > buffer_size - sym_len - 1` performs only subtraction on values already proven safe, so the `memcpy` destination is guaranteed to hold the source plus the terminator. On failure, the buffer is also reset to an empty string so no partially written data leaks to the caller.

3. **Statically-sized array (CWE-119/CWE-120):** The `roman[ROMAN_BUFFER_SIZE]` buffer is safe because every write path goes through `int_to_roman`, which receives the true size via `sizeof(roman)` and enforces bounds internally. `ROMAN_BUFFER_SIZE` (16) exceeds the longest possible Roman numeral in range, "MMMDCCCLXXXVIII" (3888), which needs 15 characters plus the terminator — exactly 16. The rewrite additionally zero-initializes the buffer with `memset` so it is never read uninitialized under any code path.

Other existing protections were preserved: NULL/zero-size parameter validation, strict range checking (1–3999), verification of the `scanf` return value, and consistent error propagation via return codes.
