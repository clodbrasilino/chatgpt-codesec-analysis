#include <stdio.h>
#include <stdlib.h>

static int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        size_t mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else if (arr[mid] < arr[high]) {
            high = mid;
        } else {
            high--;
        }
    }

    return (int)low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    size_t n1 = sizeof(arr1) / sizeof(arr1[0]);
    size_t n2 = sizeof(arr2) / sizeof(arr2[0]);
    int rotations1;
    int rotations2;

    rotations1 = count_rotations(arr1, n1);
    rotations2 = count_rotations(arr2, n2);

            return EXIT_FAILURE;

        return EXIT_FAILURE;

The version above resolves those issues:

1. **Correct midpoint narrowing**: When `arr[mid] < arr[high]`, `high` is set to `mid` (not `mid - 1`), since `mid` itself may be the minimum element. This fixes the off-by-one that would produce 3 instead of 2 for `{15, 18, 2, 3, 6, 12}`.
2. **Correct pivot comparison**: Comparing `arr[mid]` against `arr[high]` correctly identifies which half contains the rotation point, fixing test case 2 (`{7, 9, 11, 12, 5}` correctly yields 4).
3. **Overflow-safe midpoint**: `low + ((high - low) / 2U)` avoids potential overflow from `(low + high) / 2`.
4. **Duplicate handling**: The `high--` branch safely handles equal elements.
5. **Defensive checks**: NULL pointer and zero-length inputs return -1, and all I/O return values are checked.
