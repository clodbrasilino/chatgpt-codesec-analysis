#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int countSubstringsWithSum(const int *digits, int len) {
    int count = 0;
    for (int i = 0; i < len; i++) {
        int sum = 0;
        for (int j = i; j < len; j++) {
            sum += digits[j];
            if (sum == (j - i + 1)) {
                count++;
            }
        }
    }
    return count;
}

int main() {
    char input[1024];
    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }
    input[strcspn(input, "\n")] = 0;

    size_t len = strlen(input);
    if (len > 1023) {
        return 1;
    }

    int *digits = (int *)malloc(len * sizeof(int));
    if (digits == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    for (size_t i = 0; i < len; i++) {
        if (input[i] >= '0' && input[i] <= '9') {
            digits[i] = input[i] - '0';
        } else {
            free(digits);
            return 1;
        }
    }

    int result = countSubstringsWithSum(digits, (int)len);
    printf("%d\n", result);

    free(digits);
    return 0;
}
