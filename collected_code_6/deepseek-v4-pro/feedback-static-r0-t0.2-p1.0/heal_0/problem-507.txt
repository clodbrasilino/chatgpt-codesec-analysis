#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct WordNode {
    char *word;
    struct WordNode *next;
} WordNode;

typedef struct {
    WordNode *head;
    size_t size;
} WordList;

WordList *create_word_list(void) {
    WordList *list = (WordList *)malloc(sizeof(WordList));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->size = 0;
    return list;
}

int add_word(WordList *list, const char *word) {
    if (list == NULL || word == NULL) {
        return -1;
    }
    
    WordNode *new_node = (WordNode *)malloc(sizeof(WordNode));
    if (new_node == NULL) {
        return -1;
    }
    
    new_node->word = (char *)malloc(strlen(word) + 1);
    if (new_node->word == NULL) {
        free(new_node);
        return -1;
    }
    
    strcpy(new_node->word, word);
    new_node->next = NULL;
    
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        WordNode *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    
    list->size++;
    return 0;
}

void remove_words(WordList *list, const char **words_to_remove, size_t remove_count) {
    if (list == NULL || words_to_remove == NULL || remove_count == 0) {
        return;
    }
    
    WordNode *current = list->head;
    WordNode *prev = NULL;
    
    while (current != NULL) {
        int should_remove = 0;
        
        for (size_t i = 0; i < remove_count; i++) {
            if (words_to_remove[i] != NULL && strcmp(current->word, words_to_remove[i]) == 0) {
                should_remove = 1;
                break;
            }
        }
        
        if (should_remove) {
            WordNode *to_delete = current;
            
            if (prev == NULL) {
                list->head = current->next;
                current = list->head;
            } else {
                prev->next = current->next;
                current = current->next;
            }
            
            free(to_delete->word);
            free(to_delete);
            list->size--;
        } else {
            prev = current;
            current = current->next;
        }
    }
}

void print_word_list(const WordList *list) {
    if (list == NULL) {
        return;
    }
    
    WordNode *current = list->head;
    while (current != NULL) {
        printf("%s ", current->word);
        current = current->next;
    }
    printf("\n");
}

void free_word_list(WordList *list) {
    if (list == NULL) {
        return;
    }
    
    WordNode *current = list->head;
    while (current != NULL) {
        WordNode *next = current->next;
        free(current->word);
        free(current);
        current = next;
    }
    
    free(list);
}

int main(void) {
    WordList *list = create_word_list();
    if (list == NULL) {
        return 1;
    }
    
    add_word(list, "apple");
    add_word(list, "banana");
    add_word(list, "cherry");
    add_word(list, "date");
    add_word(list, "elderberry");
    add_word(list, "fig");
    add_word(list, "grape");
    
    printf("Original list: ");
    print_word_list(list);
    
    const char *words_to_remove[] = {"banana", "date", "grape"};
    size_t remove_count = sizeof(words_to_remove) / sizeof(words_to_remove[0]);
    
    remove_words(list, words_to_remove, remove_count);
    
    printf("List after removal: ");
    print_word_list(list);
    
    free_word_list(list);
    
    return 0;
}
