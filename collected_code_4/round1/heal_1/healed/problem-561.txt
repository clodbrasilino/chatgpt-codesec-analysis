#include <stdio.h>

typedef struct {
  int first;
  int second;
} pair;

void assign_pairs(pair *p1, pair *p2, size_t n) {
  if (p1 != NULL && p2 != NULL) {
    for (size_t i = 0; i < n; ++i) {
      p1[i].first = p2[i].first;
      p1[i].second = p2[i].second;
    }
  }
}

int main() {
  pair p1[3] = {{1, 2}, {3, 4}, {5, 6}};
  pair p2[3] = {{7, 8}, {9, 10}, {11, 12}};
  
  assign_pairs(p1, p2, 3);
  
  for (int i = 0; i < 3; ++i) {
      printf("Pair %d: (%d, %d)\n", i+1, p1[i].first, p1[i].second);
  }
  
  return 0;
}
