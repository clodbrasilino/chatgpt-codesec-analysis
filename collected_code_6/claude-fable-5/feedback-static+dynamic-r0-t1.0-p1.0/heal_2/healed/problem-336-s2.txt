```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32

int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-hardened; the Flawfinder hits were largely informational:

1. **`char normalized[MAX_INPUT]` / `char input[MAX_INPUT]` (CWE-119/CWE-120):** Both buffers are properly protected. `has_28_days()` uses `strnlen()` bounded by `MAX_INPUT` and rejects inputs where `len >= MAX_INPUT`, guaranteeing `normalized[len] = '\0'` never writes out of bounds. In `main()`, `fgets()` with `sizeof(input)` prevents overflow, and the explicit NUL-termination at `sizeof(input) - 1` is defensive belt-and-braces.

2. **`getchar()` in a loop (CWE-120/CWE-20):** This is a false positive — the loop discards excess characters until newline or EOF and writes nothing to any buffer. I factored it into a `flush_stdin()` helper for clarity, and the `EOF` check prevents an infinite loop on closed stdin.

3. **Minor improvements made:** the return value of `fflush(stdout)` is now checked, and all `ctype.h` calls retain the `(unsigned char)` casts, which correctly avoid undefined behavior for negative `char` values.
