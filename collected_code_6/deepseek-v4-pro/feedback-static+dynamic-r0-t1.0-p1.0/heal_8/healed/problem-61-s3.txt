#include <stdio.h>
#include <string.h>
#include <stddef.h>

int countSubstrings(const char *str, size_t max_len) {
    size_t n = 0;
    while (n < max_len && str[n] != '\0') {
        n++;
    }
    
    int count = 0;
    
    for (size_t i = 0; i < n; i++) {
        int sum = 0;
        for (size_t j = i; j < n; j++) {
            if (str[j] < '0' || str[j] > '9') {
                return -1;
            }
            sum += str[j] - '0';
            size_t len = j - i + 1;
            if (sum == (int)len) {
                count++;
            }
        }
    }
    
    return count;
}

int main(void) {
    char str[1001];
    
    if (fgets(str, sizeof(str), stdin) != NULL) {
        size_t actual_len = strnlen(str, sizeof(str));
        if (actual_len > 0 && actual_len < sizeof(str) && str[actual_len - 1] == '\n') {
            str[actual_len - 1] = '\0';
            actual_len--;
        } else if (actual_len == sizeof(str) - 1 && str[actual_len] == '\0') {
            actual_len = sizeof(str) - 1;
        }
        printf("%d\n", countSubstrings(str, actual_len));
    }
    
    return 0;
}
