#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <errno.h>

uint64_t find_largest_prime_factor(uint64_t n) {
    uint64_t largest_factor = 0;

    if (n <= 1) {
        return 0;
    }

    while (n % 2 == 0) {
        largest_factor = 2;
        n /= 2;
    }

    for (uint64_t i = 3; i * i <= n; i += 2) {
        while (n % i == 0) {
            largest_factor = i;
            n /= i;
        }
    }

    if (n > 2) {
        largest_factor = n;
    }

    return largest_factor;
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <number>\n", argv[0]);
        return EXIT_FAILURE;
    }

    char *endptr;
    errno = 0;
    unsigned long long input_val = strtoull(argv[1], &endptr, 10);

    if (errno != 0 || *endptr != '\0' || argv[1] == endptr) {
        fprintf(stderr, "Invalid input. Please provide a valid positive integer.\n");
        return EXIT_FAILURE;
    }

    if (input_val <= 1) {
        fprintf(stderr, "Input must be greater than 1.\n");
        return EXIT_FAILURE;
    }

    uint64_t n = (uint64_t)input_val;
    uint64_t largest_prime = find_largest_prime_factor(n);

    if (largest_prime == 0) {
        fprintf(stderr, "Could not find a prime factor.\n");
        return EXIT_FAILURE;
    }

    printf("%llu\n", (unsigned long long)largest_prime);

    return EXIT_SUCCESS;
}
