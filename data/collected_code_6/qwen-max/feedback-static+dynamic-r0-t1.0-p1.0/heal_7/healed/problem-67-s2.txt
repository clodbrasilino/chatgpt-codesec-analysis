#include <stdio.h>
#include <stdlib.h>
#include <inttypes.h>
#include <stdint.h>

uint64_t bell_number(int n) {
    if (n < 0 || n > 63) {
        fprintf(stderr, "Input out of range\n");
        exit(EXIT_FAILURE);
    }
    uint64_t *bell = (uint64_t*)calloc((n+1)*(n+2)/2, sizeof(uint64_t));
    if (bell == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    bell[0] = 1;
    int index = 1;
    for (int i = 1; i <= n; i++) {
        bell[index + i] = bell[index - 1];
        for (int j = 1; j < i; j++) {
            bell[index + j] += bell[index - i + j - 1];
        }
        index += i + 1;
    }

    uint64_t result = bell[(n+1)*(n+2)/2 - 1];
    free(bell);
    return result;
}

int main() {
    int n;
    printf("Enter the number of elements in the set: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    uint64_t result = bell_number(n);
    printf("The number of ways to partition a set of %d elements is: %" PRIu64 "\n", n, result);

    return EXIT_SUCCESS;
}
