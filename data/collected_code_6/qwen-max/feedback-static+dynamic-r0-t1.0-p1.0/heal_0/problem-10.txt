```c
#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int *)a - *(int *)b);
}

void get_n_smallest(int *data, int size, int n, int *result) {
    qsort(data, size, sizeof(int), compare);
    for (int i = 0; i < n; ++i) {
        result[i] = data[i];
    }
}

int main() {
    int data[] = {5, 3, 8, 1, 9, 4, 6, 7, 2};
    int size = sizeof(data) / sizeof(data[0]);
    int n = 3;
    int *result = (int *)malloc(n * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    get_n_smallest(data, size, n, result);
    for (int i = 0; i < n; ++i) {
        printf("%d ", result[i]);
    }
    free(result);
    return 0;
}
```
