#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <ctype.h>

#define MAX_STR_LEN 4096

char *snake_to_camel(const char *snake_case) {
    if (snake_case == NULL) {
        return NULL;
    }

    size_t len = strnlen(snake_case, MAX_STR_LEN);
    if (len == MAX_STR_LEN) {
        return NULL;
    }

    regex_t regex;
    int reti = regcomp(&regex, "_([a-z])", REG_EXTENDED);
    if (reti != 0) {
        return NULL;
    }

    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    char *dest = result;
    const char *src = snake_case;
    regmatch_t pmatch[2];
    size_t dest_size = len + 1;

    while (1) {
        reti = regexec(&regex, src, 2, pmatch, 0);
        
        if (reti == REG_NOMATCH) {
            snprintf(dest, dest_size, "%s", src);
            break;
        }
        
        if (reti != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        size_t prefix_len = (size_t)pmatch[0].rm_so;
        
        if (prefix_len >= dest_size) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        for (size_t i = 0; i < prefix_len; i++) {
            dest[i] = src[i];
        }
        
        dest += prefix_len;
        dest_size -= prefix_len;

        if (dest_size <= 1) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        *dest++ = toupper((unsigned char)src[pmatch[1].rm_so]);
        dest_size--;

        src += pmatch[0].rm_eo;
    }

    regfree(&regex);
    return result;
}

int main(void) {
    const char *snake_str = "this_is_a_snake_case_string";
    
    char *camel_str = snake_to_camel(snake_str);
    
    if (camel_str != NULL) {
        printf("%s\n", camel_str);
        free(camel_str);
    } else {
        fprintf(stderr, "Error during conversion\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
