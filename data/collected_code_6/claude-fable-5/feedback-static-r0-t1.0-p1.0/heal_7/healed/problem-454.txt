```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

int match_word_with_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return 0;
    }

    result[0] = '\0';

    while (text[i] != '\0') {
        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }
        if (text[i] == '\0') {
            break;
        }
        {
            size_t start = i;
            int has_z = 0;

            while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
                if (tolower((unsigned char)text[i]) == 'z') {
                    has_z = 1;
                }
                i++;
            }
            if (has_z) {
                size_t len = i - start;

                if (len >= result_size) {
                    result[0] = '\0';
                    return 0;
                }
                memcpy(result, &text[start], len);
                result[len] = '\0';
                return 1;
            }
        }
    }
    return 0;
}

int main(void)
{
    const char *text = "The lazy dog jumped over the fence";
    char result[256];

    memset(result, 0, sizeof(result));

    if (match_word_with_z(text, result, sizeof(result))) {
        if (printf("Found word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No word containing 'z' found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and their mitigations:**

1. **`memcpy` (CWE-120, buffer overflow risk):** The original code already guards this correctly — `len >= result_size` is checked before copying, guaranteeing space for both the `len` bytes and the terminating `'\0'`. The rewrite preserves this check and additionally resets `result[0] = '\0'` on the failure path so the caller never sees stale or partial data when the function returns `0`.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The fixed-size buffer `result[256]` is safe because its size is always passed via `sizeof(result)` rather than a hard-coded constant, and the callee performs bounds checking against that size. The rewrite adds `memset` initialization as defense-in-depth so the buffer is never read uninitialized under any code path.

3. **General robustness already present and retained:** NULL-pointer and zero-size parameter validation, casting to `unsigned char` before calling `isalpha`/`tolower` (avoiding undefined behavior for negative `char` values), and checking the return value of `printf`.

The code compiles cleanly with `-Wall -Wextra -Wpedantic` and contains no exploitable buffer overflow: every write into `result` is provably within bounds.
