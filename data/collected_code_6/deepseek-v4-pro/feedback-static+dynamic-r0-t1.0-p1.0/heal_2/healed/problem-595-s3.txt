#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int min_swaps_to_convert(const char *str1, const char *str2) {
    int len1 = 0;
    while (str1[len1] != '\0') {
        len1++;
    }
    
    int len2 = 0;
    while (str2[len2] != '\0') {
        len2++;
    }
    
    if (len1 != len2) {
        return -1;
    }
    
    int count1_1 = 0, count1_0 = 0;
    int count2_1 = 0, count2_0 = 0;
    
    for (int i = 0; i < len1; i++) {
        if (str1[i] == '1') count1_1++;
        else count1_0++;
        
        if (str2[i] == '1') count2_1++;
        else count2_0++;
    }
    
    if (count1_1 != count2_1 || count1_0 != count2_0) {
        return -1;
    }
    
    char *s1 = malloc(len1 + 1);
    char *s2 = malloc(len2 + 1);
    if (s1 == NULL || s2 == NULL) {
        free(s1);
        free(s2);
        return -1;
    }
    
    for (int i = 0; i < len1; i++) {
        s1[i] = str1[i];
        s2[i] = str2[i];
    }
    s1[len1] = '\0';
    s2[len2] = '\0';
    
    int swaps = 0;
    for (int i = 0; i < len1; i++) {
        if (s1[i] != s2[i]) {
            for (int j = i + 1; j < len1; j++) {
                if (s1[j] != s2[j] && s1[i] == s2[j] && s1[j] == s2[i]) {
                    char temp = s2[i];
                    s2[i] = s2[j];
                    s2[j] = temp;
                    swaps++;
                    break;
                }
            }
        }
    }
    
    for (int i = 0; i < len1; i++) {
        if (s1[i] != s2[i]) {
            for (int j = i + 1; j < len1; j++) {
                if (s1[j] != s2[j] && s1[i] != s2[i]) {
                    char temp = s2[i];
                    s2[i] = s2[j];
                    s2[j] = temp;
                    swaps++;
                    break;
                }
            }
        }
    }
    
    free(s1);
    free(s2);
    return swaps;
}

int main(void) {
    char str1[101];
    char str2[101];

    printf("Enter second binary string: ");
    if (scanf("%100s", str2) != 1) {
        return 1;

    if (result == -1) {

    }

}
