```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int findSum(int arr[], int n)
{
    int i, smallest = INT_MAX, largest = INT_MIN;

    for(i = 0; i < n; i++)
    {
        if(arr[i] < smallest)
            smallest = arr[i];
        if(arr[i] > largest)
            largest = arr[i];
    }

    return smallest+largest;
}

int main()
{
    int n;
    int *arr;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    arr = (int*) malloc(sizeof(int) * n);
    if(arr == NULL)
    {
        printf("Memory allocation failed\n");
        return 1;
    }

    printf("Enter the elements of the array: ");
    for(int i=0; i<n; i++)
          scanf("%d", &arr[i]);

    printf("Sum of smallest and largest numbers in array is: %d\n", findSum(arr, n));

    free(arr);
    return 0;
}
```
