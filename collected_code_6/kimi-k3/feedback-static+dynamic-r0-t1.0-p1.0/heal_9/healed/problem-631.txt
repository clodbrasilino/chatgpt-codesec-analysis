#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>

int replace_whitespace_underscore(const char *input, size_t input_len, char *output, size_t output_size) {
    regex_t regex;
    regmatch_t match;
    size_t cursor = 0;
    size_t output_len = 0;
    int ret;
    char *temp_output;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    if (input_len > SIZE_MAX - 1) {
        return -1;
    }

    temp_output = malloc(input_len + 1);
    if (temp_output == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[ \t\n\r]|_", REG_EXTENDED);
    if (ret != 0) {
        free(temp_output);
        return -1;
    }

    while (cursor < input_len) {
        ret = regexec(&regex, input + cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = input_len - cursor;
            if (remaining > output_size - output_len - 1) {
                regfree(&regex);
                free(temp_output);
                return -1;
            }
            if (remaining > input_len - output_len) {
                regfree(&regex);
                free(temp_output);
                return -1;
            }
            memcpy(temp_output + output_len, input + cursor, remaining);
            output_len += remaining;
            break;
        } else if (ret != 0) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < 0 || match.rm_so > match.rm_eo || (size_t)match.rm_eo > input_len - cursor) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        if ((size_t)match.rm_so > output_size - output_len - 2) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        if ((size_t)match.rm_so > input_len - output_len) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        memcpy(temp_output + output_len, input + cursor, (size_t)match.rm_so);
        output_len += (size_t)match.rm_so;

        if (input[cursor + match.rm_so] == '_') {
            temp_output[output_len] = ' ';
        } else {
            temp_output[output_len] = '_';
        }
        output_len++;

        cursor += (size_t)match.rm_eo;
    }

    if (output_len >= output_size) {
        regfree(&regex);
        free(temp_output);
        return -1;
    }
    temp_output[output_len] = '\0';

    memcpy(output, temp_output, output_len + 1);

    regfree(&regex);
    free(temp_output);
    return 0;
}

int main(void) {
    char *input = NULL;
    char *output = NULL;
    size_t input_size = 256;
    size_t len;
    int result;

    input = malloc(input_size);
    if (input == NULL) {
        return 1;
    }

    if (fgets(input, input_size, stdin) == NULL) {
        free(input);
        return 1;
    }

    len = strnlen(input, input_size);
    if (len == input_size) {
        fprintf(stderr, "Error: input too long\n");
        free(input);
        return 1;
    }
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }

    output = malloc(len + 1);
    if (output == NULL) {
        free(input);
        return 1;
    }

    result = replace_whitespace_underscore(input, len, output, len + 1);
    if (result != 0) {
        fprintf(stderr, "Error: replacement failed\n");
        free(input);
        free(output);
        return 1;
    }

    printf("%s\n", output);
    free(input);
    free(output);
    return 0;
}
