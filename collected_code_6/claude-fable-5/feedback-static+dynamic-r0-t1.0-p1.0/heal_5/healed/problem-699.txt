```c
#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static size_t bounded_len(const char *s, size_t max_len)
{
    size_t i;

    for (i = 0; i < max_len; i++) {
        if (s[i] == '\0') {
            break;
        }
    }
    return i;
}

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count10 = 0;
    int count01 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = bounded_len(s1, MAX_INPUT_LEN);
    len2 = bounded_len(s2, MAX_INPUT_LEN);

    if (len1 >= MAX_INPUT_LEN || len2 >= MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2 || len1 == 0) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') || (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] != s2[i]) {
            if (s1[i] == '1') {
                count10++;
            } else {
                count01++;
            }
        }
    }

    if (count10 != count01) {
        return -1;
    }

    return count10;
}

static void strip_newline(char *s)
{
    size_t len;

    if (s == NULL) {
        return;
    }

    len = bounded_len(s, MAX_INPUT_LEN);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r')) {
        s[len - 1] = '\0';
        len--;
    }
}

int main(void)
{
    char s1[MAX_INPUT_LEN];
    char s2[MAX_INPUT_LEN];
    int result;

    memset(s1, 0, sizeof(s1));
    memset(s2, 0, sizeof(s2));

    if (fgets(s1, (int)sizeof(s1), stdin) == NULL) {
        return 1;
    }
    if (fgets(s2, (int)sizeof(s2), stdin) == NULL) {
        return 1;
    }

    s1[sizeof(s1) - 1] = '\0';
    s2[sizeof(s2) - 1] = '\0';

    strip_newline(s1);
    strip_newline(s2);

    result = min_swaps(s1, s2);

    if (result < 0) {
        if (printf("Not Possible\n") < 0) {
            return 1;
        }
    } else {
        if (printf("%d\n", result) < 0) {
            return 1;
        }
    }

    if (fflush(stdout) != 0) {
        return 1;
    }

    return 0;
}
```

**Summary of the fixes applied:**

1. **Logic defect (failing test cases):** The original `min_swaps` implemented the wrong algorithm. The failing tests (`expected 1, got 2`, `expected 3, got 4`, and a missed `Not Possible` case) indicate the intended problem is the minimum number of swaps *within* `s1` to transform it into `s2`. The correct feasibility condition is `count10 == count01` (both strings must contain the same number of `1`s), not merely that the total mismatch count is even. The correct answer is simply `count10` (each swap pairs one `1→0` mismatch with one `0→1` mismatch, fixing both). The old formula `(count10/2) + (count01/2) + ((count10 % 2) * 2)` solved a different problem (swaps *between* two strings) and also wrongly accepted infeasible inputs such as `count10 = 2, count01 = 0`.

2. **Buffer-safety warnings (CWE-119/CWE-120):** The Flawfinder warnings on the statically sized arrays are already mitigated: `fgets` is bounded by `sizeof(buffer)`, buffers are zero-initialized with `memset`, explicit NUL termination is forced at the last index, and all string scanning uses `bounded_len` capped at `MAX_INPUT_LEN` instead of unbounded `strlen`. These defensive measures were retained; the warnings are informational (risk 2) and no actual overflow path exists.

3. **Robustness retained:** NULL checks, length-equality and non-empty validation, strict `0`/`1` character validation, newline/CR stripping, and checked return values for `printf`/`fflush` are all preserved.
