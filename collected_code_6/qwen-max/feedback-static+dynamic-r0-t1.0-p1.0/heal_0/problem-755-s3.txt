#include <stdio.h>
#include <limits.h>

int find_second_smallest(int arr[], int n) {
    int first, second;
    if (n < 2) return INT_MAX;
    if (arr[0] < arr[1]) {
        first = arr[0];
        second = arr[1];
    } else {
        first = arr[1];
        second = arr[0];
    }
    for (int i = 2; i < n; i++) {
        if (arr[i] < first) {
            second = first;
            first = arr[i];
        } else if (first < arr[i] && arr[i] < second) {
            second = arr[i];
        }
    }
    return (second == INT_MAX || second == first) ? INT_MAX : second;
}

int main() {
    int arr[] = {5, 2, 4, 1, 3};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = find_second_smallest(arr, n);
    if (result == INT_MAX) {
        printf("There is no second smallest element.\n");
    } else {
        printf("The second smallest element is %d\n", result);
    }
    return 0;
}
