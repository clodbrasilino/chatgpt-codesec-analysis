```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);

    if (len >= MAX_INPUT) {
        return -1;
    }

    i = 0;

    while (i + 1 < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            if (match_len >= sizeof(matches[found].text)) {
                match_len = sizeof(matches[found].text) - 1;
            }

            if (match_len > 0 && match_len <= len - start) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **Out-of-bounds read (the real bug):** The original loop condition `while (i < len ...)` allowed `i` to equal `len - 1`, at which point `input[i + 1]` accessed `input[len]`. If the input was not null-terminated within `MAX_INPUT` bytes (i.e., `strnlen` returned `MAX_INPUT`), this read one byte past the buffer. The rewrite changes the loop condition to `i + 1 < len`, guaranteeing both `input[i]` and `input[i + 1]` are always in bounds.

2. **Unterminated input guard:** Added an explicit `len >= MAX_INPUT` check in `find_sequences`, rejecting inputs where `strnlen` found no terminator. This hardens the function against callers passing non-terminated buffers, addressing the CWE-119/CWE-120 concern on the statically-sized `text` array.

3. **`memcpy` bounds (CWE-120):** The copy remains safe because `match_len` is clamped to `sizeof(matches[found].text) - 1` before copying, and the terminator is written explicitly afterward. The `match_len <= len - start` check ensures the source read never exceeds the validated input length.

4. **`main` input handling:** Retained the defensive pattern — `fgets` bounded by `sizeof(input)`, forced null termination of the last byte, and newline stripping via `strnlen` — which keeps the fixed-size `input` array safe from overflow.
