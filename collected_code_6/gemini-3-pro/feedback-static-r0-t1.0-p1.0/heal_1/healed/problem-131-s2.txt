#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <ctype.h>
#include <stdlib.h>

bool is_vowel(char c) {
    c = (char)tolower((unsigned char)c);
    return (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u');
}

void reverse_vowels(char *str, size_t max_len) {
    if (str == NULL || max_len == 0) {
        return;
    }

    size_t len = 0;
    while (len < max_len && str[len] != '\0') {
        len++;
    }

    if (len == 0) {
        return;
    }

    size_t left = 0;
    size_t right = len - 1;

    while (left < right) {
        while (left < right && !is_vowel(str[left])) {
            left++;
        }
        while (left < right && !is_vowel(str[right])) {
            right--;
        }

        if (left < right) {
            char temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            if (right > 0) {
                right--;
            }
        }
    }
}

int main(void) {
    char test_str[] = "hello world";
    reverse_vowels(test_str, sizeof(test_str));
    printf("%s\n", test_str);

    char test_str2[] = "Programming is fun";
    reverse_vowels(test_str2, sizeof(test_str2));
    printf("%s\n", test_str2);

    return 0;
}
