#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *replace_chars(const char *str, char old_char, char new_char) {
    if (str == NULL) {
        return NULL;
    }

    size_t len = strlen(str);
    char *result = (char *)malloc(len + 1);

    if (result == NULL) {
        return NULL;
    }

    for (size_t i = 0; i < len; i++) {
        if (str[i] == old_char) {
            result[i] = new_char;
        } else {
            result[i] = str[i];
        }
    }

    result[len] = '\0';

    return result;
}

int main(void) {
    const char *original = "hello world";
    char *modified = replace_chars(original, 'o', 'x');

    if (modified == NULL) {
        return EXIT_FAILURE;
    }

    printf("%s\n", modified);

    free(modified);
    modified = NULL;

    return EXIT_SUCCESS;
}
