```c
#include <stdio.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

double cone_volume(double radius, double height) {
    if (radius < 0 || height < 0) {
        return -1.0;
    }
    return (1.0 / 3.0) * M_PI * radius * radius * height;
}

int main(void) {
    double radius;
    double height;
    double volume;

    printf("Enter radius: ");
    if (scanf("%lf", &radius) != 1) {
        fprintf(stderr, "Invalid input for radius.\n");
        return 1;
    }

    printf("Enter height: ");
    if (scanf("%lf", &height) != 1) {
        fprintf(stderr, "Invalid input for height.\n");
        return 1;
    }

    volume = cone_volume(radius, height);
    if (volume < 0) {
        fprintf(stderr, "Radius and height must be non-negative.\n");
        return 1;
    }

    printf("Volume of cone: %.2f\n", volume);
    return 0;
}
```
