```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

double pentagon_area(double side)
{
    return (5.0 * side * side) / (4.0 * tan(M_PI / 5.0));
}

int main(void)
{
    double side = 0.0;
    double area = 0.0;
    int result = 0;

    printf("Enter the side length of the pentagon: ");

    result = scanf("%lf", &side);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (side <= 0.0) {
        fprintf(stderr, "Error: side length must be positive.\n");
        return EXIT_FAILURE;
    }

    if (!isfinite(side)) {
        fprintf(stderr, "Error: side length must be a finite number.\n");
        return EXIT_FAILURE;
    }

    area = pentagon_area(side);

    if (!isfinite(area)) {
        fprintf(stderr, "Error: computation overflow.\n");
        return EXIT_FAILURE;
    }

    printf("The area of the pentagon is: %.6f\n", area);

    return EXIT_SUCCESS;
}
```

This program computes the area of a regular pentagon using the formula A = (5 × s²) / (4 × tan(π/5)). It validates the `scanf` return value to handle malformed input, rejects non-positive and non-finite side lengths, and verifies the computed result is finite before printing, ensuring exceptional conditions are properly checked and resources (stdin/stdout) are used within their normal lifetimes.
