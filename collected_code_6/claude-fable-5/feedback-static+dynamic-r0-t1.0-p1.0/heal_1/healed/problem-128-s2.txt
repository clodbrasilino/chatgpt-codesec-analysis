```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

static size_t bounded_length(const char *s)
{
    size_t len = 0;

    while (len < MAX_WORD_LEN && s[len] != '\0') {
        len++;
    }
    return len;
}

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count)
{
    char **result;
    size_t matched;
    size_t i;

    if (result_count == NULL) {
        return NULL;
    }

    *result_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    matched = 0;
    for (i = 0; i < count; i++) {
        if (words[i] != NULL && bounded_length(words[i]) > n) {
            matched++;
        }
    }

    if (matched == 0) {
        return NULL;
    }

    if (matched > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(matched * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    matched = 0;
    for (i = 0; i < count; i++) {
        if (words[i] != NULL) {
            size_t len = bounded_length(words[i]);
            if (len > n) {
                char *copy;

                if (len >= SIZE_MAX) {
                    size_t j;
                    for (j = 0; j < matched; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }

                copy = malloc(len + 1);
                if (copy == NULL) {
                    size_t j;
                    for (j = 0; j < matched; j++) {
                        free(result[j]);
                    }
                    free(result);
                    return NULL;
                }

                memcpy(copy, words[i], len);
                copy[len] = '\0';
                result[matched] = copy;
                matched++;
            }
        }
    }

    *result_count = matched;
    return result;
}

void free_shortlist(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *words[] = { "apple", "cat", "banana", "dog", "elephant", "sun" };
    size_t count = sizeof(words) / sizeof(words[0]);
    size_t n = 3;
    size_t result_count = 0;
    size_t i;
    char **shortlisted;

    shortlisted = shortlist_words(words, count, n, &result_count);

    if (shortlisted == NULL) {
        if (printf("No words longer than %zu found.\n", n) < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("Words longer than %zu:\n", n) < 0) {
        free_shortlist(shortlisted, result_count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_count; i++) {
        if (printf("%s\n", shortlisted[i]) < 0) {
            free_shortlist(shortlisted, result_count);
            return EXIT_FAILURE;
        }
    }

    free_shortlist(shortlisted, result_count);
    return EXIT_SUCCESS;
}
```

The key fixes applied:

1. **CWE-126 (strlen over-read)**: Replaced all `strlen` calls with a `bounded_length` function that never reads beyond `MAX_WORD_LEN` bytes, so a non-terminated string cannot cause an unbounded over-read. The length is also computed only once per word in the copy loop, eliminating the TOCTOU-style double `strlen`.

2. **CWE-120 (memcpy overflow)**: The destination buffer is allocated with exactly `len + 1` bytes, `memcpy` copies exactly `len` bytes, and the terminator is written explicitly, guaranteeing the destination always holds the source data.

3. **Integer overflow hardening**: Added a check that `matched * sizeof(char *)` and `len + 1` cannot overflow `size_t` before the corresponding `malloc` calls.
