```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_IP_LEN 16

static int remove_leading_zeroes(const char *input, char *output, size_t output_size)
{
    unsigned int octets[4];
    size_t pos = 0;
    int octet_index;
    int written;

    if (input == NULL || output == NULL || output_size < MAX_IP_LEN) {
        return -1;
    }

    for (octet_index = 0; octet_index < 4; octet_index++) {
        unsigned long value = 0;
        int digit_count = 0;

        while (input[pos] != '\0' && isdigit((unsigned char)input[pos])) {
            value = value * 10 + (unsigned long)(input[pos] - '0');
            digit_count++;
            pos++;
            if (digit_count > 3) {
                return -1;
            }
        }

        if (digit_count == 0 || value > 255) {
            return -1;
        }

        octets[octet_index] = (unsigned int)value;

        if (octet_index < 3) {
            if (input[pos] != '.') {
                return -1;
            }
            pos++;
        }
    }

    if (input[pos] != '\0') {
        return -1;
    }

    written = snprintf(output, output_size, "%u.%u.%u.%u",
                       octets[0], octets[1], octets[2], octets[3]);

    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *tests[] = {
        "192.068.001.010",
        "001.002.003.004",
        "255.255.255.255",
        "000.000.000.000",
        "10.0.0.1",
        "256.1.1.1",
        "1.2.3",
        "abc.def.ghi.jkl"
    };
    char result[MAX_IP_LEN];
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    memset(result, 0, sizeof(result));

    for (i = 0; i < count; i++) {
        if (remove_leading_zeroes(tests[i], result, sizeof(result)) == 0) {
            if (printf("%s -> %s\n", tests[i], result) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Invalid IP address: %s\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

The identified weakness (CWE-119/CWE-120, statically-sized buffer) was already properly mitigated in the original code, and those mitigations are preserved here: the `result` buffer is fixed at `MAX_IP_LEN` (16 bytes, the exact maximum for a dotted-quad IPv4 string including the null terminator), its size is always passed explicitly via `sizeof(result)`, the function validates `output_size` before any write, all writing goes through `snprintf` with the caller-supplied bound, and the return value of `snprintf` is checked for both errors and truncation. Input parsing is also bounds-safe: digit counts are capped at 3 per octet, values are range-checked against 255, and the string must terminate exactly after the fourth octet, so no out-of-bounds read or write is possible.
