#include <stdio.h>
#include <stdlib.h>

typedef struct {
    unsigned char r;
    unsigned char g;
    unsigned char b;
} RGB;

typedef struct {
    double h;
    double s;
    double v;
} HSV;

static HSV rgb_to_hsv(RGB in)
{
    HSV out;
    double r = (double)in.r / 255.0;
    double g = (double)in.g / 255.0;
    double b = (double)in.b / 255.0;

    double max = r;
    if (g > max) {
        max = g;
    }
    if (b > max) {
        max = b;
    }

    double min = r;
    if (g < min) {
        min = g;
    }
    if (b < min) {
        min = b;
    }

    double delta = max - min;

    out.v = max;

    if (max > 0.0) {
        out.s = delta / max;
    } else {
        out.s = 0.0;
        out.h = 0.0;
        return out;
    }

    if (delta <= 0.0) {
        out.h = 0.0;
        return out;
    }

    if (max == r) {
        out.h = 60.0 * ((g - b) / delta);
    } else if (max == g) {
        out.h = 60.0 * (((b - r) / delta) + 2.0);
    } else {
        out.h = 60.0 * (((r - g) / delta) + 4.0);
    }

    if (out.h < 0.0) {
        out.h += 360.0;
    }

    return out;
}

int main(void)
{
    RGB colors[] = {
        { 255U, 0U, 0U },
        { 0U, 255U, 0U },
        { 0U, 0U, 255U },
        { 255U, 255U, 255U },
        { 0U, 0U, 0U },
        { 128U, 64U, 32U }
    };

    size_t count = sizeof(colors) / sizeof(colors[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        HSV result = rgb_to_hsv(colors[i]);
        int rc = printf("RGB(%u, %u, %u) -> HSV(%.2f, %.2f, %.2f)\n",
                        (unsigned int)colors[i].r,
                        (unsigned int)colors[i].g,
                        (unsigned int)colors[i].b,
                        result.h,
                        result.s,
                        result.v);
        if (rc < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
