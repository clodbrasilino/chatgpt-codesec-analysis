#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stddef.h>

typedef struct {
    int x;
    int y;
} Tuple;

typedef struct {
    Tuple *data;
    size_t size;
    size_t capacity;
} TupleList;

int list_init(TupleList *list, size_t initial_capacity) {
    if (list == NULL) {
        return -1;
    }
    if (initial_capacity > SIZE_MAX / sizeof(Tuple)) {
        list->data = NULL;
        list->size = 0;
        list->capacity = 0;
        return -1;
    }
    list->size = 0;
    list->capacity = initial_capacity;
    if (initial_capacity == 0) {
        list->data = NULL;
        return 0;
    }
    list->data = malloc(initial_capacity * sizeof(Tuple));
    if (list->data == NULL) {
        list->capacity = 0;
        return -1;
    }
    return 0;
}

void list_free(TupleList *list) {
    if (list == NULL) {
        return;
    }
    free(list->data);
    list->data = NULL;
    list->size = 0;
    list->capacity = 0;
}

int list_add(TupleList *list, Tuple tuple) {
    if (list == NULL) {
        return -1;
    }
    if (list->size == list->capacity) {
        size_t new_capacity;
        if (list->capacity == 0) {
            new_capacity = 1;
        } else {
            if (list->capacity > SIZE_MAX / 2) {
                return -1;
            }
            new_capacity = list->capacity * 2;
        }
        if (new_capacity > SIZE_MAX / sizeof(Tuple)) {
            return -1;
        }
        Tuple *new_data = realloc(list->data, new_capacity * sizeof(Tuple));
        if (new_data == NULL) {
            return -1;
        }
        list->data = new_data;
        list->capacity = new_capacity;
    }
    list->data[list->size] = tuple;
    list->size++;
    return 0;
}

int main(void) {
    TupleList list;
    if (list_init(&list, 2) != 0) {
        return EXIT_FAILURE;
    }

    Tuple t1 = {1, 2};
    Tuple t2 = {3, 4};
    Tuple t3 = {5, 6};

    if (list_add(&list, t1) != 0 ||
        list_add(&list, t2) != 0 ||
        list_add(&list, t3) != 0) {
        list_free(&list);
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < list.size; ++i) {
        if (printf("%d %d\n", list.data[i].x, list.data[i].y) < 0) {
            list_free(&list);
            return EXIT_FAILURE;
        }
    }

    list_free(&list);
    return EXIT_SUCCESS;
}
