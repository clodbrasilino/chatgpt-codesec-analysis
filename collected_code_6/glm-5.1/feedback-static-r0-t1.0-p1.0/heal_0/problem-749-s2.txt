#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int numeric_compare(const void *a, const void *b) {
    long val_a = strtol(*(const char **)a, NULL, 10);
    long val_b = strtol(*(const char **)b, NULL, 10);

    if (val_a < val_b) {
        return -1;
    }
    if (val_a > val_b) {
        return 1;
    }
    return 0;
}

void sort_strings_numerically(char **strings, size_t count) {
    if (strings == NULL || count == 0) {
        return;
    }
    qsort(strings, count, sizeof(char *), numeric_compare);
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        return 1;
    }

    size_t count = (size_t)(argc - 1);
    char **list = malloc(count * sizeof(char *));
    if (list == NULL) {
        return 1;
    }

    for (size_t i = 0; i < count; i++) {
        list[i] = argv[i + 1];
    }

    sort_strings_numerically(list, count);

    for (size_t i = 0; i < count; i++) {
        printf("%s\n", list[i]);
    }

    free(list);
    return 0;
}
