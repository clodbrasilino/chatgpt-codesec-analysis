#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    size_t count;
} Tuple;

int sum_tuple(const Tuple *t) {
    int sum = 0;
    size_t i;
    if (t == NULL || t->elements == NULL) {
        return 0;
    }
    for (i = 0; i < t->count; i++) {
        sum += t->elements[i];
    }
    return sum;
}

int sum_all_tuples(const Tuple *tuples, size_t num_tuples) {
    int total = 0;
    size_t i;
    if (tuples == NULL) {
        return 0;
    }
    for (i = 0; i < num_tuples; i++) {
        total += sum_tuple(&tuples[i]);
    }
    return total;
}

int main(void) {
    int data1[] = {1, 2, 3};
    int data2[] = {4, 5};
    int data3[] = {10, 20, 30, 40};

    Tuple tuples[3];
    tuples[0].elements = data1;
    tuples[0].count = 3;
    tuples[1].elements = data2;
    tuples[1].count = 2;
    tuples[2].elements = data3;
    tuples[2].count = 4;

    printf("Sum of tuple 0: %d\n", sum_tuple(&tuples[0]));
    printf("Sum of tuple 1: %d\n", sum_tuple(&tuples[1]));
    printf("Sum of tuple 2: %d\n", sum_tuple(&tuples[2]));
    printf("Sum of all tuples: %d\n", sum_all_tuples(tuples, 3));

    return 0;
}
