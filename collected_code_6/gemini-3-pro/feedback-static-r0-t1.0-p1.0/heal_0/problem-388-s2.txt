#include <stdio.h>
#include <stdlib.h>

unsigned int highest_power_of_2(unsigned int n) {
    if (n == 0) {
        return 0;
    }
    
    n |= (n >> 1);
    n |= (n >> 2);
    n |= (n >> 4);
    n |= (n >> 8);
    n |= (n >> 16);
    
    return n - (n >> 1);
}

int main(void) {
    unsigned int n = 0;
    
    if (scanf("%u", &n) != 1) {
        return EXIT_FAILURE;
    }
    
    unsigned int result = highest_power_of_2(n);
    printf("%u\n", result);
    
    return EXIT_SUCCESS;
}
