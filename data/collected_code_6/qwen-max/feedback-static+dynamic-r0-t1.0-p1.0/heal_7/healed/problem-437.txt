#include <stdio.h>
#include <stdlib.h>

char* remove_odd_chars(const char *input) {
    if (input == NULL) return NULL;
    size_t len = 0;
    while (input[len] && input[len] != '\0') len++;
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }
    int result_index = 0;
    for (size_t i = 0; i < len; ++i) {
        if (input[i] >= '0' && input[i] <= '9' && (input[i] - '0') % 2 == 0) {
            result[result_index++] = input[i];
        }
    }
    result[result_index] = '\0';
    return result;
}

int main() {
    const char *input = "1234567890";
    char *output = remove_odd_chars(input);
    if (output != NULL) {
        printf("%s\n", output);
        free(output);
    } else {
        fprintf(stderr, "Memory allocation failed\n");
    }
    return 0;
}
