```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    if (head_bytes > total_bytes) {
        free(temp);
        return -1;
    }
    memcpy(temp, arr, head_bytes);

    if (tail_bytes > total_bytes) {
        free(temp);
        return -1;
    }
    memmove(arr, arr + k, tail_bytes);

    if ((n - k) > n || head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) are false positives in this code, but the rewrite makes the bounds guarantees explicit and verifiable:

1. **First `memcpy(temp, arr, head_bytes)`** — `temp` is allocated with exactly `head_bytes`, and `arr` holds `total_bytes ≥ head_bytes` (guaranteed since `k < n` after the modulo). The destination always fits the source.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`** — the destination region spans from offset `(n - k) * sizeof(int)` to the end of the array, which is exactly `head_bytes` in size, since `head_bytes = total_bytes - tail_bytes`. The explicit check `head_bytes > total_bytes - tail_bytes` guards this.

3. **Overflow safety** — the `n > SIZE_MAX / sizeof(int)` check prevents integer overflow in `n * sizeof(int)`, so all derived byte counts are valid. The original tautological condition `tail_bytes <= total_bytes - head_bytes + head_bytes` was cleaned up to a straightforward `tail_bytes > total_bytes` guard before `memmove`.

4. **Resource handling** — `temp` is freed on every exit path after allocation, so there are no memory leaks, and `malloc` failure is handled.
