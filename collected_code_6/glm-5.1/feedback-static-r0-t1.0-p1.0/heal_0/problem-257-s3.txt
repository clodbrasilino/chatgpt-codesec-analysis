```c
#include <stdio.h>
#include <stdlib.h>

void swap(int *a, int *b) {
    if (a == NULL || b == NULL) {
        abort();
    }
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 10;
    int y = 20;

    printf("Before swap: x = %d, y = %d\n", x, y);

    swap(&x, &y);

    printf("After swap: x = %d, y = %d\n", x, y);

    return 0;
}
```
