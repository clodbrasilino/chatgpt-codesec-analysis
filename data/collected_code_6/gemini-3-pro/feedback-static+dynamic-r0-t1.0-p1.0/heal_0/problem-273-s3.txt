```c
#include <stdio.h>
#include <stdlib.h>

void subtract_tuples(const int *tuple1, const int *tuple2, int *result, size_t size) {
    if (tuple1 == NULL || tuple2 == NULL || result == NULL) {
        return;
    }

    for (size_t i = 0; i < size; ++i) {
        result[i] = tuple1[i] - tuple2[i];
    }
}

int main(void) {
    const size_t size = 5;
    
    int *tuple1 = malloc(size * sizeof(int));
    int *tuple2 = malloc(size * sizeof(int));
    int *result = malloc(size * sizeof(int));

    if (tuple1 == NULL || tuple2 == NULL || result == NULL) {
        free(tuple1);
        free(tuple2);
        free(result);
        return 1;
    }

    for (size_t i = 0; i < size; ++i) {
        tuple1[i] = (int)(10 + i);
        tuple2[i] = (int)(5 + i);
    }

    subtract_tuples(tuple1, tuple2, result, size);

    for (size_t i = 0; i < size; ++i) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(tuple1);
    free(tuple2);
    free(result);

    return 0;
}
```
