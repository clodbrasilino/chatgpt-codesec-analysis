```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 256

bool is_valid_password(const char *str, size_t required_length);
static void discard_remaining_input(void);
static void secure_clear(void *ptr, size_t len);

bool is_valid_password(const char *str, size_t required_length)
{
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    size_t length = 0;
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (required_length == 0 || required_length >= MAX_INPUT_LENGTH) {
        return false;
    }

    length = strnlen(str, MAX_INPUT_LENGTH);

    if (length >= MAX_INPUT_LENGTH) {
        return false;
    }

    if (length != required_length) {
        return false;
    }

    for (i = 0; i < length; i++) {
        unsigned char c = (unsigned char)str[i];

        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        }
    }

    return (has_upper && has_lower && has_digit);
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

static void secure_clear(void *ptr, size_t len)
{
    volatile unsigned char *p = (volatile unsigned char *)ptr;
    size_t i;

    if (ptr == NULL) {
        return;
    }

    for (i = 0; i < len; i++) {
        p[i] = 0;
    }
}

int main(void)
{
    char buffer[MAX_INPUT_LENGTH];
    size_t required_length = 8;
    size_t input_length;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string to validate (required length %zu): ", required_length);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        secure_clear(buffer, sizeof(buffer));
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    input_length = strnlen(buffer, sizeof(buffer));

    if (input_length > 0 && buffer[input_length - 1] == '\n') {
        buffer[input_length - 1] = '\0';
    } else if (input_length == sizeof(buffer) - 1) {
        discard_remaining_input();
    }

    if (is_valid_password(buffer, required_length)) {
        printf("The string is valid.\n");
    } else {
        printf("The string is invalid.\n");
    }

    secure_clear(buffer, sizeof(buffer));

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The original loop was already safe since `getchar()` returns an `int` and no buffer is written to. I simplified it by removing the pointless `discarded` counter (which served no protective purpose) and kept the correct `EOF`/newline termination conditions. Additionally, `discard_remaining_input()` is now only called when the buffer was actually filled to capacity (`input_length == sizeof(buffer) - 1`), avoiding an unnecessary blocking `getchar()` call when input ended via EOF without a newline.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer usage was already bounds-safe (`fgets` with `sizeof(buffer)`, forced NUL termination, `strnlen` capped at buffer size). I added a defensive `length >= MAX_INPUT_LENGTH` check in `is_valid_password` for robustness.

3. **Sensitive data clearing:** The original `memset(buffer, 0, ...)` before `return` can legally be optimized away by the compiler (dead-store elimination), leaving the password in memory. I replaced it with a `secure_clear()` function using a `volatile` pointer, which the compiler cannot elide. It's also now called on the early error path, ensuring cleanup on all exit paths.
