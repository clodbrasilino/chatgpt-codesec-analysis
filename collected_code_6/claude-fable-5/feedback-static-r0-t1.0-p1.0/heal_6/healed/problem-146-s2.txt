```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        sum += (long)(unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

The original code already addressed the flagged weakness (CWE-119/CWE-120) correctly, so the rewrite preserves those mitigations:

- **Bounded input reading**: `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing the classic `gets()`-style overflow the Flawfinder warning is concerned with.
- **Guaranteed null termination**: `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant after `fgets`, but harmless and protects against future refactoring).
- **Length-limited string operations**: `strnlen()` is used instead of `strlen()`, both in `main` and in `ascii_sum()`, so even a non-terminated buffer cannot cause an out-of-bounds read.
- **NULL pointer check**: `ascii_sum()` validates its input and signals failure via a negative return value, which the caller checks.
- **Safe character arithmetic**: the cast through `unsigned char` avoids sign-extension issues on platforms where plain `char` is signed, ensuring correct sums for bytes above 0x7F.
- **Newline stripping with bounds check**: `len > 0` guards the `buffer[len - 1]` access against an empty string.

No functional changes were needed — the statically-sized array is used safely throughout, since every access is bounds-checked or length-limited.
