{

    regmatch_t match;
    char *result;

    size_t result_len;
    size_t result_cap;
    const char *cursor;

    }

        return NULL;
    }

        return NULL;

        return NULL;

        if (consumed > input_len) {
            regfree(&regex);
            free(result);
            return NULL;

            if (result_len >= result_cap ||
                remaining >= result_cap - result_len) {
                regfree(&regex);
                free(result);
                return NULL;
            }

            free(result);
            return NULL;

            return NULL;

            free(result);
            return NULL;

            free(result);
            return NULL;

}
