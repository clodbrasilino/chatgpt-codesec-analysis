#include <stdio.h>
#include <stdlib.h>

int count_sequences(int n, int m) {
    if (n == 0 || m <= 0) return 1;
    int *dp = (int *)calloc((n + 1) * (m + 1), sizeof(int));
    if (dp == NULL) return -1;
    for (int j = 1; j <= m; ++j) dp[j] = 1;
    for (int i = 2; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            dp[(i-1)*(m+1) + j] = 0;
    for (int i = 2; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            for (int k = 1; 2 * k <= j; ++k)
                dp[i*(m+1) + j] += dp[(i-1)*(m+1) + k];
    int result = 0;
    for (int j = 1; j <= m; ++j)
        result += dp[n*(m+1) + j];
    free(dp);
    return result;
}

int main() {
    int n, m;
    printf("Enter the length of the sequence (n): ");
    if (scanf("%d", &n) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    printf("Enter the maximum value in the sequence (m): ");
    if (scanf("%d", &m) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    int result = count_sequences(n, m);
    if (result == -1) {
        printf("Memory allocation failed.\n");
        return 1;
    }
    printf("Number of sequences: %d\n", result);
    return 0;
}
