```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int min_abs_diff_sum(const int *arr1, const int *arr2, size_t n, long long *result)
{
    int *a;
    int *b;
    size_t i;
    size_t bytes;
    long long sum;

    if (arr1 == NULL || arr2 == NULL || result == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*a)) {
        return -1;
    }

    bytes = n * sizeof(*a);

    a = malloc(bytes);
    if (a == NULL) {
        return -1;
    }

    b = malloc(bytes);
    if (b == NULL) {
        free(a);
        return -1;
    }

    for (i = 0; i < n; i++) {
        a[i] = arr1[i];
        b[i] = arr2[i];
    }

    qsort(a, n, sizeof(*a), compare_ints);
    qsort(b, n, sizeof(*b), compare_ints);

    sum = 0;
    for (i = 0; i < n; i++) {
        long long diff = (long long)a[i] - (long long)b[i];
        if (diff < 0) {
            diff = -diff;
        }
        sum += diff;
    }

    free(a);
    free(b);

    *result = sum;
    return 0;
}

int main(void)
{
    const int arr1[] = {4, 1, 8, 7};
    const int arr2[] = {2, 3, 6, 5};
    size_t n = sizeof(arr1) / sizeof(arr1[0]);
    long long result;

    if (min_abs_diff_sum(arr1, arr2, n, &result) != 0) {
        fprintf(stderr, "Error computing minimum sum\n");
        return EXIT_FAILURE;
    }

    printf("Minimum sum of absolute differences: %lld\n", result);
    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The Flawfinder warnings flagged the two `memcpy` calls as potential CWE-120 buffer overflow risks. In the original code these were actually safe — the destination buffers `a` and `b` were allocated with exactly `n * sizeof(*a)` bytes, the same size used in the copy, and the multiplication was guarded against overflow via the `SIZE_MAX / sizeof(*a)` check. The warnings were false positives typical of pattern-based static analysis.

To silence the tool and make the bounds relationship explicit, I replaced both `memcpy` calls with element-by-element copy loops bounded by `n`. This guarantees the copy can never exceed the allocated element count, removes the flagged function entirely, and merges both copies into a single loop for clarity. I also hoisted the size computation into a single `bytes` variable so the allocation size and the overflow guard are visibly tied together.

All other defensive measures from the original were preserved: NULL/zero-length parameter validation, the `size_t` overflow check before allocation, cleanup of `a` if the second `malloc` fails, and the widening to `long long` before subtraction to avoid signed integer overflow when computing differences. The program's behavior is unchanged — it still prints `Minimum sum of absolute differences: 6` for the given inputs.
