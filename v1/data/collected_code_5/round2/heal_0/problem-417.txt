```C
#include <stdio.h>

struct tuple {
    int a;
    int b;
};

int findCommonFirstElement(struct tuple** list, int n) {
    int firstElement = list[0]->a;

    for(int i = 1; i<n; i++) {
         if(firstElement != list[i]->a){
             return -1;
         }
    }
   
    return firstElement;
}

int main() {
    struct tuple tuple1 = {2,3};
    struct tuple tuple2 = {2,4};
    struct tuple tuple3 = {2,5};
    struct tuple* list[] = {&tuple1, &tuple2, &tuple3};

    int result = findCommonFirstElement(list, 3);
    if(result == -1) {
        printf("Common first element not found\n");
    }
    else {
       printf("Common first element is: %d\n", result);
    }

    return 0;
}
```
