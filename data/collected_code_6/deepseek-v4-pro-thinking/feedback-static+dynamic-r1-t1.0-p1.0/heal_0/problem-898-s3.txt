#include <limits.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int extract_contiguous_elements(const int *list, size_t list_length, size_t start_index,
                                size_t element_count, int *destination, size_t destination_capacity)
{
    if (list == NULL) {
        return -1;
    }
    if (start_index > list_length) {
        return -2;
    }
    if (element_count > list_length - start_index) {
        return -3;
    }
    if (element_count > destination_capacity) {
        return -4;
    }
    if (element_count > INT_MAX) {
        return -5;
    }
    if (element_count == 0) {
        return 0;
    }
    if (destination == NULL) {
        return -6;
    }
    memmove(destination, list + start_index, element_count * sizeof(*destination));
    return (int)element_count;
}

int main(void)
{
    int list[] = {10, 20, 30, 40, 50, 60};
    size_t list_length = sizeof(list) / sizeof(list[0]);
    size_t start_index = 2;
    size_t element_count = 3;
    int *destination = malloc(element_count * sizeof(*destination));

    if (destination == NULL) {
        return 1;
    }

    int result = extract_contiguous_elements(list, list_length, start_index,
                                             element_count, destination, element_count);
    if (result < 0) {
        free(destination);
        return 1;
    }

    for (size_t i = 0; i < (size_t)result; i++) {
        printf("%d\n", destination[i]);
    }

    free(destination);
    return 0;
}
