#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char **items;
    size_t count;
} StringList;

typedef struct {
    StringList *lists;
    size_t count;
} ListOfLists;

static int compare_strings(const void *a, const void *b)
{
    const char *const *sa = (const char *const *)a;
    const char *const *sb = (const char *const *)b;

    if (*sa == NULL && *sb == NULL) {
        return 0;
    }
    if (*sa == NULL) {
        return 1;
    }
    if (*sb == NULL) {
        return -1;
    }
    return strcmp(*sa, *sb);
}

static int sort_sublists(ListOfLists *list_of_lists)
{
    size_t i;

    if (list_of_lists == NULL) {
        return -1;
    }
    if (list_of_lists->count > 0 && list_of_lists->lists == NULL) {
        return -1;
    }

    for (i = 0; i < list_of_lists->count; ++i) {
        StringList *sublist = &list_of_lists->lists[i];

        if (sublist->count > 0 && sublist->items == NULL) {
            return -1;
        }
        if (sublist->count > 1) {
            qsort(sublist->items, sublist->count, sizeof(sublist->items[0]), compare_strings);
        }
    }

    return 0;
}

static void free_list_of_lists(ListOfLists *list_of_lists)
{
    if (list_of_lists == NULL) {
        return;
    }

    if (list_of_lists->lists != NULL) {
        for (size_t i = 0; i < list_of_lists->count; ++i) {
            if (list_of_lists->lists[i].items != NULL) {
                for (size_t j = 0; j < list_of_lists->lists[i].count; ++j) {
                    free(list_of_lists->lists[i].items[j]);
                }
                free(list_of_lists->lists[i].items);
            }
        }
        free(list_of_lists->lists);
    }

    list_of_lists->lists = NULL;
    list_of_lists->count = 0;
}

static char *duplicate_string(const char *source)
{
    size_t length;
    char *copy;

    if (source == NULL) {
        return NULL;
    }

    length = strlen(source);
    if (length == (size_t)-1) {
        return NULL;
    }

    copy = (char *)malloc(length + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, source, length + 1);
    return copy;
}

static int init_list_of_lists(ListOfLists *list_of_lists,
                              const char *const *const *values,
                              const size_t *counts,
                              size_t list_count)
{
    if (list_of_lists == NULL) {
        return -1;
    }

    list_of_lists->lists = NULL;
    list_of_lists->count = 0;

    if (list_count == 0) {
        return 0;
    }

    if (values == NULL || counts == NULL) {
        return -1;
    }

    list_of_lists->lists = (StringList *)calloc(list_count, sizeof(*list_of_lists->lists));
    if (list_of_lists->lists == NULL) {
        return -1;
    }

    list_of_lists->count = list_count;

    for (size_t i = 0; i < list_count; ++i) {
        if (counts[i] == 0) {
            continue;
        }
        if (values[i] == NULL) {
            free_list_of_lists(list_of_lists);
            return -1;
        }

        list_of_lists->lists[i].items = (char **)calloc(counts[i], sizeof(*list_of_lists->lists[i].items));
        if (list_of_lists->lists[i].items == NULL) {
            free_list_of_lists(list_of_lists);
            return -1;
        }

        list_of_lists->lists[i].count = counts[i];

        for (size_t j = 0; j < counts[i]; ++j) {
            if (values[i][j] == NULL) {
                free_list_of_lists(list_of_lists);
                return -1;
            }

            list_of_lists->lists[i].items[j] = duplicate_string(values[i][j]);
            if (list_of_lists->lists[i].items[j] == NULL) {
                free_list_of_lists(list_of_lists);
                return -1;
            }
        }
    }

    return 0;
}

static void print_list_of_lists(const ListOfLists *list_of_lists)
{
    if (list_of_lists == NULL) {
        return;
    }

    for (size_t i = 0; i < list_of_lists->count; ++i) {
        printf("[");
        for (size_t j = 0; j < list_of_lists->lists[i].count; ++j) {
            if (j > 0) {
                printf(", ");
            }
            if (list_of_lists->lists[i].items[j] != NULL) {
                printf("\"%s\"", list_of_lists->lists[i].items[j]);
            } else {
                printf("NULL");
            }
        }
        printf("]\n");
    }
}

int main(void)
{
    const char *sublist1[] = {"banana", "apple", "cherry"};
    const char *sublist2[] = {"dog", "cat", "elephant", "ant"};
    const char *sublist3[] = {"red", "blue", "green"};
    const char *const *values[] = {sublist1, sublist2, sublist3};
    const size_t counts[] = {3, 4, 3};
    ListOfLists list_of_lists;

    if (init_list_of_lists(&list_of_lists, values, counts, 3) != 0) {
        fprintf(stderr, "Failed to initialize list of lists\n");
        return EXIT_FAILURE;
    }

    printf("Before sorting:\n");
    print_list_of_lists(&list_of_lists);

    if (sort_sublists(&list_of_lists) != 0) {
        fprintf(stderr, "Failed to sort sublists\n");
        free_list_of_lists(&list_of_lists);
        return EXIT_FAILURE;
    }

    printf("After sorting:\n");
    print_list_of_lists(&list_of_lists);

    free_list_of_lists(&list_of_lists);
    return EXIT_SUCCESS;
}
