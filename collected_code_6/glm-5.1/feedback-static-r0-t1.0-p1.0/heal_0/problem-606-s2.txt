```c
#include <stdio.h>
#include <math.h>

#define M_PI 3.14159265358979323846

double degrees_to_radians(double degrees) {
    return degrees * (M_PI / 180.0);
}

int main(void) {
    double degrees = 180.0;
    double radians = degrees_to_radians(degrees);
    
    if (isfinite(radians)) {
        printf("%f\n", radians);
    }
    
    return 0;
}
```
