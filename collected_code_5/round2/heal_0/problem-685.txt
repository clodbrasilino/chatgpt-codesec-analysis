#include<stdio.h>

int checkPrimeNumber(int number) {
    int i, isPrime = 1;

    if (number == 1) {
        isPrime = 0;
    }
    else {
        for(i = 2; i <= number/2; i++) {
            if( number%i == 0) {
                isPrime = 0;
                break;
            }
        }
    }
    return isPrime;
}

int sumOfPrimes(int n) {
    int sum = 0;
    for(int i = 2; i <= n; i++) {
        if(checkPrimeNumber(i)==1) {
            sum += i;
        }
    }
    return sum;
}

int main() {
    int n;
    printf("Enter a positive number: ");
    scanf("%d", &n);

    printf("Sum of prime numbers from 1 to %d = %d", n, sumOfPrimes(n));

    return 0;
}
