```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TUPLE_COUNT 5
#define NAME_LEN 32

typedef struct {
    char name[NAME_LEN];
    float value;
} Tuple;

static int set_tuple_name(Tuple *tuple, const char *name)
{
    size_t len;

    if (tuple == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, NAME_LEN);
    if (len >= NAME_LEN) {
        return -1;
    }

    memcpy(tuple->name, name, len);
    tuple->name[len] = '\0';
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static void sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }

    for (i = 0U; i < count; i++) {
        if (printf("(%.*s, %.2f)\n", NAME_LEN - 1, tuples[i].name,
                   (double)tuples[i].value) < 0) {
            break;
        }
    }
}

int main(void)
{
    Tuple tuples[TUPLE_COUNT];
    const char *names[TUPLE_COUNT] = { "alpha", "beta", "gamma", "delta", "epsilon" };
    const float values[TUPLE_COUNT] = { 3.14f, 1.41f, 2.72f, 0.58f, 1.62f };
    size_t i;

    memset(tuples, 0, sizeof(tuples));

    for (i = 0U; i < TUPLE_COUNT; i++) {
        if (set_tuple_name(&tuples[i], names[i]) != 0) {
            return EXIT_FAILURE;
        }
        tuples[i].value = values[i];
    }

    if (printf("Before sorting:\n") < 0) {
        return EXIT_FAILURE;
    }
    print_tuples(tuples, TUPLE_COUNT);

    sort_tuples(tuples, TUPLE_COUNT);

    if (printf("After sorting:\n") < 0) {
        return EXIT_FAILURE;
    }
    print_tuples(tuples, TUPLE_COUNT);

    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **CWE-126 (`strlen` over-read)**: Replaced `strlen(name)` with `strnlen(name, NAME_LEN)`, which bounds the read to `NAME_LEN` bytes. If the input is not null-terminated within the buffer size, `strnlen` returns `NAME_LEN`, which the subsequent `len >= NAME_LEN` check rejects — so an unterminated string can no longer cause an unbounded read.

2. **CWE-120 (`memcpy` overflow)**: The `memcpy` is now provably safe because `len` is guaranteed to be strictly less than `NAME_LEN` before the copy, and the null terminator write at `tuple->name[len]` stays within bounds.

3. **CWE-119/CWE-120 (statically-sized arrays)**: The fixed-size buffers remain, but every write path into them is bounds-checked (`set_tuple_name` validates length, `memset` uses `sizeof`), and every read path is bounded (`printf` uses the `%.*s` precision specifier with `NAME_LEN - 1` to prevent over-reads even if termination were somehow lost).

The remaining structure — null-pointer guards, `printf` return-value checks, and `qsort` with a total-order comparator — was already sound and is preserved.
