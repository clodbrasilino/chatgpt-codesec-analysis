```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096u

char *insert_spaces(const char *input)
{
    regex_t re;
    regmatch_t match;
    char *output;
    size_t len;
    size_t out_size;
    size_t out_pos;
    size_t offset;
    int rc;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    out_size = (len * 2u) + 1u;
    output = (char *)malloc(out_size);
    if (output == NULL) {
        return NULL;
    }

    rc = regcomp(&re, "[A-Z]", REG_EXTENDED);
    if (rc != 0) {
        free(output);
        return NULL;
    }

    out_pos = 0u;
    offset = 0u;

    while ((offset < len) &&
           (regexec(&re, input + offset, 1u, &match, 0) == 0)) {
        size_t start = offset + (size_t)match.rm_so;
        size_t chunk = (size_t)match.rm_so;

        if ((out_pos + chunk + 2u) > out_size) {
            regfree(&re);
            free(output);
            return NULL;
        }

        memcpy(output + out_pos, input + offset, chunk);
        out_pos += chunk;

        if ((start > 0u) && (input[start - 1u] != ' ')) {
            output[out_pos] = ' ';
            out_pos++;
        }

        output[out_pos] = input[start];
        out_pos++;
        offset = start + 1u;
    }

    if ((out_pos + (len - offset) + 1u) > out_size) {
        regfree(&re);
        free(output);
        return NULL;
    }

    memcpy(output + out_pos, input + offset, len - offset);
    out_pos += (len - offset);
    output[out_pos] = '\0';

    regfree(&re);
    return output;
}

int main(void)
{
    const char *text = "HelloWorldThisIsATestString";
    char *result;

    result = insert_spaces(text);
    if (result == NULL) {
        (void)fprintf(stderr, "Error: failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. If the input is not NUL-terminated within that limit, the function safely rejects it instead of reading past the buffer indefinitely.

2. **CWE-120 (`memcpy` buffer overflow):** Both `memcpy` calls are now guarded by explicit capacity checks against `out_size` before copying. Although the original allocation of `2*len + 1` is mathematically sufficient for the worst case (a space inserted before every character), the explicit checks make the invariant defensively verifiable and protect against future maintenance errors — if the insertion logic ever changes, the code fails cleanly rather than corrupting the heap.

3. **Resource safety preserved:** All error paths correctly release both the compiled regex (`regfree`) and the output buffer (`free`), avoiding leaks introduced by the new early-exit checks.
