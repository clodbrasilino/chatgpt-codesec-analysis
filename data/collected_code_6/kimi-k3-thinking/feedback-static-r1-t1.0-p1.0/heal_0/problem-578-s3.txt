#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

static Node *create_node(int data)
{
    Node *node = malloc(sizeof(*node));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

static int append_node(Node **head, Node **tail, int data)
{
    Node *node;

    if (head == NULL || tail == NULL) {
        return -1;
    }

    node = create_node(data);
    if (node == NULL) {
        return -1;
    }

    if (*head == NULL) {
        *head = node;
    } else {
        (*tail)->next = node;
    }
    *tail = node;
    return 0;
}

static void free_list(Node *head)
{
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
}

static size_t list_length(const Node *head)
{
    size_t length = 0;
    const Node *current = head;
    while (current != NULL) {
        length++;
        current = current->next;
    }
    return length;
}

static Node *build_list(const int *values, size_t count)
{
    Node *head = NULL;
    Node *tail = NULL;
    size_t i;

    if (values == NULL && count > 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        if (append_node(&head, &tail, values[i]) != 0) {
            free_list(head);
            return NULL;
        }
    }
    return head;
}

static Node *interleave_lists(const Node *list1, const Node *list2)
{
    Node *head = NULL;
    Node *tail = NULL;
    const Node *current1 = list1;
    const Node *current2 = list2;

    if (list1 == NULL || list2 == NULL) {
        return NULL;
    }

    if (list_length(list1) != list_length(list2)) {
        return NULL;
    }

    while (current1 != NULL && current2 != NULL) {
        if (append_node(&head, &tail, current1->data) != 0) {
            free_list(head);
            return NULL;
        }
        if (append_node(&head, &tail, current2->data) != 0) {
            free_list(head);
            return NULL;
        }
        current1 = current1->next;
        current2 = current2->next;
    }

    return head;
}

static void print_list(const Node *head)
{
    const Node *current = head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

int main(void)
{
    int values1[] = {1, 3, 5, 7};
    int values2[] = {2, 4, 6, 8};
    size_t count1 = sizeof(values1) / sizeof(values1[0]);
    size_t count2 = sizeof(values2) / sizeof(values2[0]);
    Node *list1;
    Node *list2;
    Node *interleaved;

    list1 = build_list(values1, count1);
    if (list1 == NULL) {
        fprintf(stderr, "Failed to build first list\n");
        return EXIT_FAILURE;
    }

    list2 = build_list(values2, count2);
    if (list2 == NULL) {
        fprintf(stderr, "Failed to build second list\n");
        free_list(list1);
        return EXIT_FAILURE;
    }

    interleaved = interleave_lists(list1, list2);
    if (interleaved == NULL) {
        fprintf(stderr, "Failed to interleave lists\n");
        free_list(list2);
        free_list(list1);
        return EXIT_FAILURE;
    }

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);
    printf("Interleaved: ");
    print_list(interleaved);

    free_list(interleaved);
    free_list(list2);
    free_list(list1);

    return EXIT_SUCCESS;
}
