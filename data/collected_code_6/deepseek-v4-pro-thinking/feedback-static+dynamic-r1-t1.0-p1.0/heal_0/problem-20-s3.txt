#include <stdio.h>
#include <limits.h>

int isWoodall(int num) {
    if (num < 1) {
        return 0;
    }

    unsigned long long target = (unsigned long long)num + 1ULL;
    int max_bits = (int)(sizeof(unsigned long long) * CHAR_BIT);

    for (int n = 1; n < max_bits; ++n) {
        unsigned long long power = 1ULL << n;

        if (power > target / (unsigned long long)n) {
            return 0;
        }

        if ((unsigned long long)n * power - 1ULL == (unsigned long long)num) {
            return 1;
        }
    }

    return 0;
}

int main(void) {
    int number;

    if (scanf("%d", &number) != 1) {
        return 1;
    }

    if (isWoodall(number)) {
        printf("%d is a Woodall number.\n", number);
    } else {
        printf("%d is not a Woodall number.\n", number);
    }

    return 0;
}
