```c
#include <stdio.h>
#include <stdlib.h>

void print_negative_numbers(const int *list, size_t size) {
    if (list == NULL || size == 0) {
        return;
    }

    for (size_t i = 0; i < size; i++) {
        if (list[i] < 0) {
            printf("%d\n", list[i]);
        }
    }
}

int main(void) {
    int numbers[] = {10, -5, 3, -1, 0, -8, 7};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);

    print_negative_numbers(numbers, size);

    return 0;
}
```
