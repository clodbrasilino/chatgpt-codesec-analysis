#include <stdio.h>
#include <stdlib.h>

int isPrime(int num) {
    if(num <= 1) {
        return 0;
    }
    for(int i = 2; i * i <= num; i++) {
        if(num % i == 0) {
            return 0;
        }
    }
    return 1;
}

int countPrimePowerFactors(int num) {
    int count = 0;
    for(int i = 2; i <= num; i++) {
        if(num % i == 0 && isPrime(i)){
            while(num % i == 0){
                num /= i;
            }
            count++;
        }
    }
    return count;
}

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);
    printf("Count of Distinct Prime Powers: %d\n", countPrimePowerFactors(num));
    return 0;
}
