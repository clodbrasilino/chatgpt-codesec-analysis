#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_digits(const char *input) {
    if (input == NULL) return NULL;
    size_t len = strlen(input);
    char *result = malloc(len + 1);
    if (result == NULL) return NULL;
    char *ptr = result;
    for (size_t i = 0; i < len; ++i) {
        if (!isdigit((unsigned char)input[i])) {
            *ptr++ = input[i];
        }
    }
    *ptr = '\0';
    return result;
}

void free_strings(char **strings, int count) {
    for (int i = 0; i < count; ++i) {
        free(strings[i]);
    }
    free(strings);
}

int main() {
    const char *inputs[] = {"hello123", "456world", "no7digits89", "123456"};
    int count = sizeof(inputs) / sizeof(inputs[0]);
    char **results = malloc(count * sizeof(char *));
    if (results == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    for (int i = 0; i < count; ++i) {
        results[i] = remove_digits(inputs[i]);
        if (results[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_strings(results, i);
            return 1;
        }
    }

    for (int i = 0; i < count; ++i) {
        printf("%s\n", results[i]);
    }

    free_strings(results, count);
    return 0;
}
