#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void findThreeClosest(int arr1[], int size1, int arr2[], int size2, int arr3[], int size3, int *res1, int *res2, int *res3) {
    int i = 0, j = 0, k = 0;
    int min_diff = INT_MAX;
    int diff;

    while (i < size1 && j < size2 && k < size3) {
        int minimum = arr1[i] < arr2[j] ? (arr1[i] < arr3[k] ? arr1[i] : arr3[k]) : (arr2[j] < arr3[k] ? arr2[j] : arr3[k]);
        int maximum = arr1[i] > arr2[j] ? (arr1[i] > arr3[k] ? arr1[i] : arr3[k]) : (arr2[j] > arr3[k] ? arr2[j] : arr3[k]);

        diff = maximum - minimum;

        if (diff < min_diff) {
            min_diff = diff;
            *res1 = arr1[i];
            *res2 = arr2[j];
            *res3 = arr3[k];
        }

        if (min_diff == 0) {
            break;
        }

        if (arr1[i] == minimum) {
            i++;
        } else if (arr2[j] == minimum) {
            j++;
        } else {
            k++;
        }
    }
}

int main(void) {
    int arr1[] = {1, 4, 10};
    int arr2[] = {2, 15, 20};
    int arr3[] = {10, 12};

    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    int size3 = sizeof(arr3) / sizeof(arr3[0]);

    int res1, res2, res3;

    findThreeClosest(arr1, size1, arr2, size2, arr3, size3, &res1, &res2, &res3);

    printf("%d %d %d\n", res1, res2, res3);

    return 0;
}
