```c
#include <stdio.h>
#include <stddef.h>
#include <stdbool.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096U

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t len = 0U;

    while ((len < max_len) && (str[len] != '\0')) {
        len++;
    }

    return len;
}

static bool is_palindrome(const char *str)
{
    size_t left;
    size_t right;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = bounded_length(str, MAX_STRING_LENGTH);
    if ((len == 0U) || (len >= MAX_STRING_LENGTH)) {
        return false;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        if (tolower((unsigned char)str[left]) != tolower((unsigned char)str[right])) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

static size_t find_palindromes(const char *const *strings, size_t count,
                               const char **results, size_t results_capacity)
{
    size_t found;

    if ((strings == NULL) || (results == NULL) || (results_capacity == 0U)) {
        return 0U;
    }

    found = 0U;
    for (size_t i = 0U; i < count; i++) {
        if (is_palindrome(strings[i])) {
            if (found < results_capacity) {
                results[found] = strings[i];
                found++;
            } else {
                break;
            }
        }
    }

    return found;
}

int main(void)
{
    static const char *const words[] = {
        "racecar",
        "hello",
        "Level",
        "world",
        "noon",
        "example",
        "madam"
    };
    const size_t word_count = sizeof(words) / sizeof(words[0]);
    const char *palindromes[sizeof(words) / sizeof(words[0])] = { NULL };
    size_t found;

    found = find_palindromes(words, word_count, palindromes,
                             sizeof(palindromes) / sizeof(palindromes[0]));

    if (found == 0U) {
        if (puts("No palindromes found.") == EOF) {
            return 1;
        }
    } else {
        if (printf("Found %zu palindrome(s):\n", found) < 0) {
            return 1;
        }
        for (size_t i = 0U; i < found; i++) {
            if (palindromes[i] == NULL) {
                return 1;
            }
            if (printf("%s\n", palindromes[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

The key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced `strlen()` with a custom `bounded_length()` function that scans at most `MAX_STRING_LENGTH` bytes, so a non-terminated string can never cause an unbounded over-read. Strings hitting the cap are rejected. This also removed the need for `<string.h>`.

2. **CWE-119/CWE-120 (statically-sized array)**: The `palindromes` array is now zero-initialized, its capacity is passed to `find_palindromes()` using `sizeof(palindromes) / sizeof(palindromes[0])` (derived directly from the array itself rather than a separate variable), and each element is null-checked before dereferencing. The existing `found < results_capacity` bounds check in `find_palindromes()` was preserved.

3. **Variable scope (cppcheck)**: The loop index `i` is now declared inside the `for` statements in both `main()` and `find_palindromes()`, reducing its scope to the minimum required.

4. Additionally, `words` was made `static const` since it never changes, avoiding stack initialization of the pointer array on each call.
