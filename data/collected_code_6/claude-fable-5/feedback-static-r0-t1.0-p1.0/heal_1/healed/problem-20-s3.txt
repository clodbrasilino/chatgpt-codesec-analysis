```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int is_woodall(unsigned long long x)
{
    unsigned long long n;

    if (x < 1ULL) {
        return 0;
    }

    for (n = 1ULL; n < 64ULL; n++) {
        unsigned long long value;

        if (n > (ULLONG_MAX >> n)) {
            break;
        }
        value = n << n;
        if (value - 1ULL == x) {
            return 1;
        }
        if (value - 1ULL > x) {
            break;
        }
    }

    return 0;
}

int main(void)
{
    unsigned long long num;
    int result;

    printf("Enter a number: ");
    if (scanf("%llu", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = is_woodall(num);

    if (result == 1) {
        printf("%llu is a Woodall number\n", num);
    } else {
        printf("%llu is not a Woodall number\n", num);
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the `[variableScope]` weakness: the variable `value` was declared at function scope but is only used inside the `for` loop, so its declaration has been moved into the loop body, reducing its scope to the minimum required. This improves readability and makes it clear the value doesn't persist across iterations. All other logic — the overflow guard (`n > (ULLONG_MAX >> n)`), the Woodall check (`n * 2^n - 1 == x`), the early-exit condition, and the input validation in `main` — remains unchanged, as it was already correct and safe.
