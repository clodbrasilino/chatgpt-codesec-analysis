#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void print_numbers(char *str) {
    int len = strlen(str);

    for(int i = 0; i < len; i++) {
        if(isdigit(str[i])) {
            printf("Number: %c at position: %d\n", str[i], i+1);
        }
    }
}

int main() {
    char str[100];

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    // Replace trailing newline with null character, if present
    if(str[strlen(str)-1] == '\n')
        str[strlen(str)-1] = '\0';

    print_numbers(str);

    return 0;
}
