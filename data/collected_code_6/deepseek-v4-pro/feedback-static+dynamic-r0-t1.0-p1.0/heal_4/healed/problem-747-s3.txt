#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

char* lcs_three(const char* s1, const char* s2, const char* s3) {
    if (s1 == NULL || s2 == NULL || s3 == NULL) return NULL;
    
    size_t m = strnlen(s1, SIZE_MAX);
    size_t n = strnlen(s2, SIZE_MAX);
    size_t p = strnlen(s3, SIZE_MAX);
    
    size_t dim1 = m + 1;
    size_t dim2 = n + 1;
    size_t dim3 = p + 1;
    
    int*** dp = (int***)calloc(dim1, sizeof(int**));
    if (!dp) return NULL;
    
    for (size_t i = 0; i < dim1; i++) {
        dp[i] = (int**)calloc(dim2, sizeof(int*));
        if (!dp[i]) {
            for (size_t x = 0; x < i; x++) {
                for (size_t y = 0; y < dim2; y++) free(dp[x][y]);
                free(dp[x]);
            }
            free(dp);
            return NULL;
        }
        for (size_t j = 0; j < dim2; j++) {
            dp[i][j] = (int*)calloc(dim3, sizeof(int));
            if (!dp[i][j]) {
                for (size_t y = 0; y < j; y++) free(dp[i][y]);
                free(dp[i]);
                for (size_t x = 0; x < i; x++) {
                    for (size_t y = 0; y < dim2; y++) free(dp[x][y]);
                    free(dp[x]);
                }
                free(dp);
                return NULL;
            }
        }
    }
    
    for (size_t i = 1; i < dim1; i++) {
        for (size_t j = 1; j < dim2; j++) {
            for (size_t k = 1; k < dim3; k++) {
                if (s1[i-1] == s2[j-1] && s1[i-1] == s3[k-1]) {
                    dp[i][j][k] = dp[i-1][j-1][k-1] + 1;
                } else {
                    int max = dp[i-1][j][k];
                    if (dp[i][j-1][k] > max) max = dp[i][j-1][k];
                    if (dp[i][j][k-1] > max) max = dp[i][j][k-1];
                    if (dp[i-1][j-1][k] > max) max = dp[i-1][j-1][k];
                    if (dp[i-1][j][k-1] > max) max = dp[i-1][j][k-1];
                    if (dp[i][j-1][k-1] > max) max = dp[i][j-1][k-1];
                    dp[i][j][k] = max;
                }
            }
        }
    }
    
    size_t len = (size_t)dp[m][n][p];
    char* result = (char*)malloc((len + 1) * sizeof(char));
    if (!result) {
        for (size_t i = 0; i < dim1; i++) {
            for (size_t j = 0; j < dim2; j++) free(dp[i][j]);
            free(dp[i]);
        }
        free(dp);
        return NULL;
    }
    
    result[len] = '\0';
    size_t i_pos = m;
    size_t j_pos = n;
    size_t k_pos = p;
    size_t index = len;
    
    while (i_pos > 0 && j_pos > 0 && k_pos > 0) {
        if (s1[i_pos-1] == s2[j_pos-1] && s1[i_pos-1] == s3[k_pos-1]) {
            index--;
            result[index] = s1[i_pos-1];
            i_pos--;
            j_pos--;
            k_pos--;
        } else if (dp[i_pos-1][j_pos][k_pos] >= dp[i_pos][j_pos-1][k_pos] && dp[i_pos-1][j_pos][k_pos] >= dp[i_pos][j_pos][k_pos-1]) {
            i_pos--;
        } else if (dp[i_pos][j_pos-1][k_pos] >= dp[i_pos-1][j_pos][k_pos] && dp[i_pos][j_pos-1][k_pos] >= dp[i_pos][j_pos][k_pos-1]) {
            j_pos--;
        } else {
            k_pos--;
        }
    }
    
    for (size_t i = 0; i < dim1; i++) {
        for (size_t j = 0; j < dim2; j++) free(dp[i][j]);
        free(dp[i]);
    }
    free(dp);
    
    return result;
}

int main(void) {
    const char* str1 = "ABCDGH";
    const char* str2 = "AEDFHR";
    const char* str3 = "ADH";
    
    char* result = lcs_three(str1, str2, str3);
    
    if (result) {
        printf("LCS of three strings: %s\n", result);
        printf("Length: %zu\n", strnlen(result, SIZE_MAX));
        free(result);
    } else {
        printf("Error computing LCS\n");
    }
    
    const char* test1_a = "AGGT12";
    const char* test1_b = "12TXAYB";
    const char* test1_c = "12XBA";
    result = lcs_three(test1_a, test1_b, test1_c);
    if (result) {
        printf("LCS: %s\n", result);
        free(result);
    }
    
    const char* test2_a = "";
    const char* test2_b = "ABC";
    const char* test2_c = "DEF";
    result = lcs_three(test2_a, test2_b, test2_c);
    if (result) {
        printf("LCS with empty string: '%s'\n", result);
        free(result);
    }
    
    return 0;
}
