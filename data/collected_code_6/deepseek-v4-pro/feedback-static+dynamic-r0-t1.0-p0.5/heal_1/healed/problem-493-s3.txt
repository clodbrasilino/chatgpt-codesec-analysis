#include <stdlib.h>
#include <math.h>
#include <stdio.h>

typedef struct {
    double x;
    double y;
} Point;

typedef struct {
    Point vertices[6];
} Hexagon;

typedef struct {
    Hexagon *hexagons;
    int count;
} HexagonGrid;

HexagonGrid calculate_hexagon_grid(int rows, int cols, double radius) {
    HexagonGrid grid;
    grid.hexagons = NULL;
    grid.count = 0;

    if (rows <= 0 || cols <= 0 || radius <= 0.0) {
        return grid;
    }

    grid.hexagons = (Hexagon *)malloc((size_t)(rows * cols) * sizeof(Hexagon));
    if (grid.hexagons == NULL) {
        return grid;
    }

    grid.count = rows * cols;
    double hex_height = radius * sqrt(3.0);
    double hex_width = radius * 2.0;
    double vert_spacing = hex_height;
    double horiz_spacing = hex_width * 0.75;

    for (int row = 0; row < rows; row++) {
        for (int col = 0; col < cols; col++) {
            double center_x = col * horiz_spacing;
            double center_y = row * vert_spacing;
            if (col % 2 == 1) {
                center_y += vert_spacing * 0.5;
            }

            int index = row * cols + col;
            for (int i = 0; i < 6; i++) {
                double angle = M_PI / 3.0 * (double)i + M_PI / 6.0;
                grid.hexagons[index].vertices[i].x = center_x + radius * cos(angle);
                grid.hexagons[index].vertices[i].y = center_y + radius * sin(angle);
            }
        }
    }

    return grid;
}

void free_hexagon_grid(HexagonGrid *grid) {
    if (grid == NULL) {
        return;
    }
    free(grid->hexagons);
    grid->hexagons = NULL;
    grid->count = 0;
}

int main(void) {
    HexagonGrid grid = calculate_hexagon_grid(3, 4, 1.0);
    if (grid.hexagons == NULL) {
        return 1;
    }

    for (int i = 0; i < grid.count; i++) {
        for (int j = 0; j < 6; j++) {
            if (j > 0) {
                printf(" ");
            }
            printf("(%.2f, %.2f)", grid.hexagons[i].vertices[j].x, grid.hexagons[i].vertices[j].y);
        }
        printf("\n");
    }

    free_hexagon_grid(&grid);
    return 0;
}
