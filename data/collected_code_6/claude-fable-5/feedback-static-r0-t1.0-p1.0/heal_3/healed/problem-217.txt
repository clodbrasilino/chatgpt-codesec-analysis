```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define BUFFER_SIZE 256

int find_first_repeated_char(const char *str, size_t max_len, char *result)
{
    unsigned char seen[UCHAR_MAX + 1] = {0};
    size_t i;

    if (str == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

static void discard_remaining_input(void)
{
    size_t guard = 0;
    const size_t guard_limit = (size_t)-1;

    while (guard < guard_limit) {
        int ch = getchar();
        if (ch == '\n' || ch == EOF) {
            break;
        }
        guard++;
    }
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char repeated = '\0';
    int status;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strcspn(buffer, "\n");
    if (len >= sizeof(buffer)) {
        len = sizeof(buffer) - 1;
    }
    buffer[len] = '\0';

    if (len == sizeof(buffer) - 1) {
        discard_remaining_input();
    }

    status = find_first_repeated_char(buffer, sizeof(buffer), &repeated);

    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input to function\n");
        return 1;
    }

    return 0;
}
```

**Analysis of the identified weaknesses and how the code addresses them:**

1. **Flawfinder `char` array warnings (CWE-119/CWE-120)** — Both statically-sized buffers are handled safely:
   - `seen[UCHAR_MAX + 1]` is indexed exclusively by an `unsigned char` value, which by definition cannot exceed `UCHAR_MAX`, so no out-of-bounds access is possible.
   - `buffer[BUFFER_SIZE]` is only written to via `fgets` with `sizeof(buffer)` as the limit, followed by explicit null termination at `sizeof(buffer) - 1`. All subsequent indexing is bounds-checked against `sizeof(buffer)`.

2. **Flawfinder `getchar` warning (CWE-120/CWE-20)** — The read loop in `discard_remaining_input` never stores characters into a buffer, and a `guard` counter with an upper limit prevents any theoretical unbounded looping. Termination on both `'\n'` and `EOF` is handled.

3. **Cppcheck `variableScope` on `ch`** — This was the one genuine actionable finding. The declaration of `ch` has been moved inside the `while` loop body, reducing its scope to the minimum required. It remains declared as `int` (not `char`), which is essential for correctly distinguishing `EOF` from valid character values.

4. **General robustness retained** — NULL-pointer and zero-length parameter validation in `find_first_repeated_char`, checking the `fgets` return value, clamping `strcspn` results defensively, and flushing stdin only when the buffer was completely filled (indicating possible truncation) are all preserved.
