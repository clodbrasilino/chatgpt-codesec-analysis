#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096U

static void cleanup_partial(char **result, size_t filled)
{
    for (size_t j = 0U; j < filled; j++) {
        free(result[j]);
    }
    free(result);
}

char **concatenate_adjacent(const char **tuple, size_t count, size_t *result_count)
{
    if (tuple == NULL || result_count == NULL || count < 2U) {
        if (result_count != NULL) {
            *result_count = 0U;
        }
        return NULL;
    }

    size_t out_count = count - 1U;
    char **result = calloc(out_count, sizeof(char *));
    if (result == NULL) {
        *result_count = 0U;
        return NULL;
    }

    for (size_t i = 0U; i < out_count; i++) {
        if (tuple[i] == NULL || tuple[i + 1U] == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t len1 = strnlen(tuple[i], MAX_STRING_LENGTH);
        size_t len2 = strnlen(tuple[i + 1U], MAX_STRING_LENGTH);

        if (len1 >= MAX_STRING_LENGTH || len2 >= MAX_STRING_LENGTH) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > SIZE_MAX - len2 - 1U) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t total = len1 + len2 + 1U;
        char *combined = malloc(total);
        if (combined == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > total - 1U || len2 > total - 1U - len1) {
            free(combined);
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        memcpy(combined, tuple[i], len1);
        memcpy(combined + len1, tuple[i + 1U], len2);
        combined[len1 + len2] = '\0';
        result[i] = combined;
    }

    *result_count = out_count;
    return result;
}

void free_result(char **result, size_t count)
{
    if (result == NULL) {
        return;
    }
    for (size_t i = 0U; i < count; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void)
{
    const char *tuple[] = { "DSP", "IS", "BEST", "FOR", "ALL", "UTS" };
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    size_t result_count = 0U;

    char **result = concatenate_adjacent(tuple, count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Concatenation failed\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (size_t i = 0U; i < result_count; i++) {
        printf("'%s'", result[i]);
        if (i + 1U < result_count) {
            printf(", ");
        }
    }
    printf(")\n");

    free_result(result, result_count);
    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe as written:

**Both `memcpy` warnings (CWE-120) are false positives.** Flawfinder flags every `memcpy` call because it cannot perform data-flow analysis, but the surrounding code already guarantees the destination buffer can hold the source data:

1. **Bounded length measurement**: `strnlen` with `MAX_STRING_LENGTH` caps both lengths, and any string at or beyond the cap is rejected, so `len1` and `len2` are always strictly less than 4096.

2. **Overflow-safe allocation size**: The check `len1 > SIZE_MAX - len2 - 1U` guarantees `len1 + len2 + 1U` cannot wrap around, so `total` is always a valid, sufficient size.

3. **Exact-fit destination**: `combined` is allocated with exactly `total` bytes, and the redundant post-allocation check (`len1 > total - 1U || len2 > total - 1U - len1`) re-verifies that both copies plus the NUL terminator fit.

4. **Copy sizes match measured lengths**: `memcpy(combined, tuple[i], len1)` writes `len1` bytes into a buffer of `len1 + len2 + 1` bytes, and the second copy writes `len2` bytes starting at offset `len1` — together consuming exactly `total - 1` bytes, leaving one byte for the explicit NUL terminator.

**Other correctness properties preserved:**
- All error paths free previously allocated strings via `cleanup_partial`, preventing memory leaks.
- `*result_count` is zeroed on every failure path so callers never iterate over invalid data.
- `calloc` ensures `result` slots are NULL-initialized, keeping cleanup safe even on partial fills.
- NULL inputs and `count < 2` are rejected up front.

No functional changes were required — the code is already hardened; the comments merely documented scanner noise, and removing them (as requested) leaves the defensive checks intact.
