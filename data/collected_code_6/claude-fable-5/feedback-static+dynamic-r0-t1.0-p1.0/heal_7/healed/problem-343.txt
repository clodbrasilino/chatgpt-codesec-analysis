```c
#include <stdio.h>
#include <ctype.h>
#include <stddef.h>
#include <string.h>

int count_digits_letters(const char *str, size_t *digits, size_t *letters)
{
    size_t i;

    if (str == NULL || digits == NULL || letters == NULL) {
        return -1;
    }

    *digits = 0;
    *letters = 0;

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isdigit(c)) {
            (*digits)++;
        } else if (isalpha(c)) {
            (*letters)++;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[256];
    size_t digits = 0;
    size_t letters = 0;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (count_digits_letters(buffer, &digits, &letters) != 0) {
        fprintf(stderr, "Error counting characters\n");
        return 1;
    }

    printf("Digits: %zu\n", digits);
    printf("Letters: %zu\n", letters);

    return 0;
}
```

1. **Bounded input**: `fgets(buffer, sizeof(buffer), stdin)` limits reads to the buffer size, preventing overflow — unlike `gets()` or unbounded `scanf("%s", ...)`.
2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant since `fgets` always null-terminates on success, but harmless).
3. **Safe length computation**: `strnlen` with an explicit bound avoids reading past the buffer even if termination were somehow missing.
4. **Safe `ctype` usage**: casting to `unsigned char` before `isdigit`/`isalpha` avoids undefined behavior for negative `char` values (e.g., extended/UTF-8 bytes on platforms with signed `char`).
5. **NULL-pointer validation** in `count_digits_letters` and error checking of `fgets` return value handle failure paths gracefully.
