```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_SOURCE_LEN 1024

size_t replace_chars(char *str, char old_char, char new_char);

size_t replace_chars(char *str, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    if (old_char == '\0' || new_char == '\0') {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    const char *source = "hello world, hello everyone";
    size_t len;
    size_t buffer_size;
    char *buffer;
    size_t replaced;

    len = strnlen(source, MAX_SOURCE_LEN);
    if (len >= MAX_SOURCE_LEN) {
        fprintf(stderr, "Source string too long or not null-terminated\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    buffer = calloc(buffer_size, sizeof(char));
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (len + 1 > buffer_size) {
        fprintf(stderr, "Buffer too small for source data\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    memcpy(buffer, source, len);
    buffer[buffer_size - 1] = '\0';

    printf("Original: %s\n", buffer);

    replaced = replace_chars(buffer, 'l', 'L');

    printf("Modified: %s\n", buffer);
    printf("Replacements made: %zu\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fix applied:**

The original code was already largely defensive, but the Flawfinder warning on `memcpy` (CWE-120) deserves scrutiny:

1. **The `memcpy` bounds check**: The original guard `if (len > buffer_size - 1)` works, but it is a fragile pattern — if `buffer_size` were ever `0`, the unsigned subtraction `buffer_size - 1` would wrap around to `SIZE_MAX` and the check would silently pass. The rewrite uses `len + 1 > buffer_size`, which avoids unsigned underflow entirely (overflow of `len + 1` is impossible here since `len < MAX_SOURCE_LEN`).

2. **Null termination**: Writing the terminator at `buffer[buffer_size - 1]` rather than `buffer[len]` ties the write to the actual allocation size, guaranteeing it stays in bounds even if the size logic changes later. Since `buffer_size == len + 1`, the result is identical, but the invariant is now expressed against the allocated capacity.

3. **What was already correct and preserved**:
   - `strnlen` with `MAX_SOURCE_LEN` bounds the length scan and rejects overly long/unterminated input.
   - `calloc` return value is checked, and `calloc` zero-initializes, providing defense-in-depth for termination.
   - `replace_chars` validates its pointer and rejects `'\0'` as either character, preventing terminator corruption or infinite-loop-style bugs.
   - `free(buffer)` followed by `buffer = NULL` prevents accidental use-after-free/double-free.
   - `%zu` is the correct format specifier for `size_t`.

The `memcpy` call itself is safe because the destination size is derived directly from `len` before the copy — the fix simply makes the guarding arithmetic robust against unsigned wraparound.
