```c
#include <stdio.h>
#include <stdlib.h>

int count_equal_pairs(const int *arr, size_t size) {
    if (arr == NULL || size < 2) {
        return 0;
    }

    int count = 0;
    for (size_t i = 0; i < size - 1; i++) {
        for (size_t j = i + 1; j < size; j++) {
            if (arr[i] == arr[j]) {
                count++;
            }
        }
    }
    return count;
}

int main(void) {
    int arr[] = {1, 2, 1, 3, 2, 1};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    int result = count_equal_pairs(arr, size);
    printf("Number of equal element pairs: %d\n", result);

    return 0;
}
```
