#include <stdio.h>
#include <stdlib.h>

int min_operations(int a, int b) {
    int count = 0;
    int xor_result = a ^ b;
    
    while (xor_result != 0) {
        count += xor_result & 1;
        xor_result >>= 1;
    }
    
    return count;
}

int main(void) {
    int num1, num2;
    
    printf("Enter first number: ");
    if (scanf("%d", &num1) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    printf("Enter second number: ");
    if (scanf("%d", &num2) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    printf("Minimum operations required: %d\n", min_operations(num1, num2));
    
    return 0;
}
