```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096

typedef struct {
    char **keys;
    char **values;
    size_t count;
    size_t capacity;
} Dict;

static Dict *dict_create(void)
{
    Dict *d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }
    d->count = 0;
    d->capacity = 8;
    d->keys = malloc(d->capacity * sizeof(char *));
    if (d->keys == NULL) {
        free(d);
        return NULL;
    }
    d->values = malloc(d->capacity * sizeof(char *));
    if (d->values == NULL) {
        free(d->keys);
        free(d);
        return NULL;
    }
    return d;
}

static void dict_destroy(Dict *d)
{
    size_t i;
    if (d == NULL) {
        return;
    }
    for (i = 0; i < d->count; i++) {
        free(d->keys[i]);
        free(d->values[i]);
    }
    free(d->keys);
    free(d->values);
    free(d);
}

static size_t bounded_length(const char *s, size_t max_len)
{
    size_t i;
    for (i = 0; i < max_len; i++) {
        if (s[i] == '\0') {
            return i;
        }
    }
    return max_len;
}

static char *string_duplicate(const char *s)
{
    size_t len;
    char *copy;
    if (s == NULL) {
        return NULL;
    }
    len = bounded_length(s, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, s, len);
    copy[len] = '\0';
    return copy;
}

static int dict_set(Dict *d, const char *key, const char *value)
{
    size_t i;
    char *new_value;
    char *new_key;

    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }

    for (i = 0; i < d->count; i++) {
        if (strcmp(d->keys[i], key) == 0) {
            new_value = string_duplicate(value);
            if (new_value == NULL) {
                return -1;
            }
            free(d->values[i]);
            d->values[i] = new_value;
            return 0;
        }
    }

    if (d->count == d->capacity) {
        size_t new_capacity;
        char **new_keys;
        char **new_values;
        if (d->capacity > SIZE_MAX / (2 * sizeof(char *))) {
            return -1;
        }
        new_capacity = d->capacity * 2;
        new_keys = realloc(d->keys, new_capacity * sizeof(char *));
        if (new_keys == NULL) {
            return -1;
        }
        d->keys = new_keys;
        new_values = realloc(d->values, new_capacity * sizeof(char *));
        if (new_values == NULL) {
            return -1;
        }
        d->values = new_values;
        d->capacity = new_capacity;
    }

    new_key = string_duplicate(key);
    if (new_key == NULL) {
        return -1;
    }
    new_value = string_duplicate(value);
    if (new_value == NULL) {
        free(new_key);
        return -1;
    }

    d->keys[d->count] = new_key;
    d->values[d->count] = new_value;
    d->count++;
    return 0;
}

static int dict_copy_into(Dict *dest, const Dict *src)
{
    size_t i;
    if (dest == NULL || src == NULL) {
        return -1;
    }
    for (i = 0; i < src->count; i++) {
        if (dict_set(dest, src->keys[i], src->values[i]) != 0) {
            return -1;
        }
    }
    return 0;
}

static Dict *dict_merge_three(const Dict *a, const Dict *b, const Dict *c)
{
    Dict *merged = dict_create();
    if (merged == NULL) {
        return NULL;
    }
    if ((dict_copy_into(merged, a) != 0) ||
        (dict_copy_into(merged, b) != 0) ||
        (dict_copy_into(merged, c) != 0)) {
        dict_destroy(merged);
        return NULL;
    }
    return merged;
}

static void dict_print(const Dict *d)
{
    size_t i;
    if (d == NULL) {
        return;
    }
    for (i = 0; i < d->count; i++) {
        printf("%s: %s\n", d->keys[i], d->values[i]);
    }
}

int main(void)
{
    Dict *a = NULL;
    Dict *b = NULL;
    Dict *c = NULL;
    Dict *merged = NULL;
    int status = EXIT_FAILURE;

    a = dict_create();
    b = dict_create();
    c = dict_create();

    if (a == NULL || b == NULL || c == NULL) {
        goto cleanup;
    }

    if (dict_set(a, "name", "Alice") != 0) {
        goto cleanup;
    }
    if (dict_set(a, "city", "Paris") != 0) {
        goto cleanup;
    }
    if (dict_set(b, "age", "30") != 0) {
        goto cleanup;
    }
    if (dict_set(b, "city", "London") != 0) {
        goto cleanup;
    }
    if (dict_set(c, "job", "Engineer") != 0) {
        goto cleanup;
    }
    if (dict_set(c, "name", "Bob") != 0) {
        goto cleanup;
    }

    merged = dict_merge_three(a, b, c);
    if (merged == NULL) {
        goto cleanup;
    }

    printf("Merged dictionary:\n");
    dict_print(merged);
    status = EXIT_SUCCESS;

cleanup:
    dict_destroy(a);
    dict_destroy(b);
    dict_destroy(c);
    dict_destroy(merged);
    if (status != EXIT_SUCCESS) {
        fprintf(stderr, "An error occurred\n");
    }
    return status;
}
```

Regarding the flagged `memcpy` weakness: the code is already safe as written. The destination buffer is allocated with `malloc(len + 1)` immediately before the copy, and exactly `len` bytes are copied followed by explicit null termination. The length itself is bounded by `bounded_length` against `MAX_STRING_LEN`, so no overflow is possible. The Flawfinder warning is a false positive here, so the `memcpy` call is retained unchanged.
