```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    char *key;
    int value;
} DictionaryEntry;

typedef struct {
    DictionaryEntry *entries;
    int size;
} Dictionary;

bool areAllValuesSame(Dictionary *dict) {
    if (dict->size == 0) return true;
    int firstValue = dict->entries[0].value;
    for (int i = 1; i < dict->size; ++i) {
        if (dict->entries[i].value != firstValue) {
            return false;
        }
    }
    return true;
}

int main() {
    Dictionary dict = { .size = 0, .entries = NULL };
    // Example dictionary with all same values
    dict.size = 3;
    dict.entries = (DictionaryEntry *)malloc(dict.size * sizeof(DictionaryEntry));
    dict.entries[0] = (DictionaryEntry){.key = "a", .value = 5};
    dict.entries[1] = (DictionaryEntry){.key = "b", .value = 5};
    dict.entries[2] = (DictionaryEntry){.key = "c", .value = 5};

    if (areAllValuesSame(&dict)) {
        printf("All values are the same.\n");
    } else {
        printf("Not all values are the same.\n");
    }

    free(dict.entries);
    return 0;
}
```
