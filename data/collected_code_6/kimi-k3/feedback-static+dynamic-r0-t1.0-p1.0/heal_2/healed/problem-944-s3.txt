#include <stdio.h>
#include <ctype.h>

void separate_and_print_numbers(const char *str) {
    if (str == NULL) {
        return;
    }

    size_t i = 0;

    while (str[i] != '\0') {
        if (isdigit((unsigned char)str[i])) {
            size_t start = i;
            while (str[i] != '\0' && isdigit((unsigned char)str[i])) {
                i++;
            }
            printf("Number: %.*s, Position: %zu\n", (int)(i - start), str + start, start);
        } else {
            i++;
        }
    }
}

int main(void) {
    const char *input = "abc123def456ghi789";
    separate_and_print_numbers(input);
    return 0;
}
