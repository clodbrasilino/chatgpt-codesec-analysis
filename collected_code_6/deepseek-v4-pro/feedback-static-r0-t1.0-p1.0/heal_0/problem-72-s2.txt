```c
#include <stdio.h>

int is_difference_of_squares(int n) {
    if (n < 0) {
        return 0;
    }
    if (n % 4 == 2) {
        return 0;
    }
    return 1;
}

int main(void) {
    int num;
    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input\n");
        return 1;
    }

    if (is_difference_of_squares(num)) {
        printf("%d can be represented as difference of two squares\n", num);
    } else {
        printf("%d cannot be represented as difference of two squares\n", num);
    }

    return 0;
}
```
