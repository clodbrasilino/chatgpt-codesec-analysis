#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdbool.h>

bool is_valid_ip(const char *ip_address) {
    if (ip_address == NULL) {
        return false;
    }

    regex_t regex;
    const char *pattern = "^((25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\\.){3}(25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$";
    int reti = regcomp(&regex, pattern, REG_EXTENDED);
    
    if (reti != 0) {
        return false;
    }

    reti = regexec(&regex, ip_address, 0, NULL, 0);
    regfree(&regex);

    return (reti == 0);
}

int main(void) {
    const char *test_ips[] = {
        "192.168.1.1",
        "255.255.255.255",
        "256.100.50.25",
        "192.168.1",
        "invalid_ip",
        "0.0.0.0",
        NULL
    };

    for (size_t i = 0; i < sizeof(test_ips) / sizeof(test_ips[0]); i++) {
        if (test_ips[i] != NULL) {
            printf("%s: %s\n", test_ips[i], is_valid_ip(test_ips[i]) ? "Valid" : "Invalid");
        }
    }

    return 0;
}
