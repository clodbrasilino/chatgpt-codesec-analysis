```c
#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <stdlib.h>
#include <limits.h>

uint32_t toggle_even_bits(uint32_t num)
{
    return num ^ 0xAAAAAAAAU;
}

int main(void)
{
    char buffer[64];
    char *endptr;
    unsigned long value;
    uint32_t num;
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL)
    {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    errno = 0;
    value = strtoul(buffer, &endptr, 10);
    
    if (errno != 0 || endptr == buffer || value > UINT32_MAX)
    {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    num = (uint32_t)value;
    
    printf("%" PRIu32 "\n", toggle_even_bits(num));
    
    return 0;
}
```
