#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static unsigned int to_unsigned_abs(int value)
{
    if (value < 0) {
        return (unsigned int)(-(long long)value);
    }
    return (unsigned int)value;
}

static unsigned int digit_distance(int a, int b)
{
    unsigned int ua = to_unsigned_abs(a);
    unsigned int ub = to_unsigned_abs(b);
    unsigned int distance = 0U;

    while ((ua > 0U) || (ub > 0U)) {
        unsigned int da = ua % 10U;
        unsigned int db = ub % 10U;

        if (da > db) {
            distance += (da - db);
        } else {
            distance += (db - da);
        }

        ua /= 10U;
        ub /= 10U;
    }

    return distance;
}

int main(void)
{
    int a = 0;
    int b = 0;

    printf("Enter first integer: ");
    if (scanf("%d", &a) != 1) {
        fprintf(stderr, "Invalid input for first integer.\n");
        return EXIT_FAILURE;
    }

    printf("Enter second integer: ");
    if (scanf("%d", &b) != 1) {
        fprintf(stderr, "Invalid input for second integer.\n");
        return EXIT_FAILURE;
    }

    printf("Digit distance: %u\n", digit_distance(a, b));

    return EXIT_SUCCESS;
}
