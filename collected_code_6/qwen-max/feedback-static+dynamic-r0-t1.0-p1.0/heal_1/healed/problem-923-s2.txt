#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int shortestCommonSupersequenceLength(const char *str1, const char *str2) {
    int len1 = (str1 != NULL) ? strlen(str1) : 0;
    int len2 = (str2 != NULL) ? strlen(str2) : 0;
    int dp[len1 + 1][len2 + 1];

    for (int i = 0; i <= len1; i++) {
        for (int j = 0; j <= len2; j++) {
            if (i == 0 || j == 0) {
                dp[i][j] = i + j;
            } else if (str1[i - 1] == str2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = 1 + (dp[i - 1][j] < dp[i][j - 1] ? dp[i - 1][j] : dp[i][j - 1]);
            }
        }
    }
    return dp[len1][len2];
}

int main() {
    const char* str1 = "ABCD";
    const char* str2 = "ACDF";
    int result = shortestCommonSupersequenceLength(str1, str2);
    printf("Length of the shortest common supersequence: %d\n", result);
    return 0;
}
