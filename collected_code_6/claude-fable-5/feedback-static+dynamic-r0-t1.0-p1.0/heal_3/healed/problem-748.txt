#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static int safe_copy(char *dest, size_t dest_capacity, size_t dest_offset,
                     const char *src, size_t count)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_offset > dest_capacity) {
        return -1;
    }
    if (count > (dest_capacity - dest_offset)) {
        return -1;
    }
    memcpy(dest + dest_offset, src, count);
    return 0;
}

char *add_spaces(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t len;
    size_t out;
    size_t offset;
    size_t capacity;
    int rc;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    rc = regcomp(&regex, "[A-Z]", REG_EXTENDED);
    if (rc != 0) {
        return NULL;
    }

    capacity = (2U * len) + 1U;
    result = malloc(capacity);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    out = 0U;
    offset = 0U;
    while ((offset < len) &&
           (regexec(&regex, input + offset, 1U, &match, 0) == 0)) {
        size_t start;
        size_t chunk;

        if (match.rm_so < 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        start = offset + (size_t)match.rm_so;
        chunk = (size_t)match.rm_so;

        if ((start >= len) || ((out + chunk + 2U) > capacity)) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        if (safe_copy(result, capacity, out, input + offset, chunk) != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        out += chunk;

        if ((start > 0U) && (input[start - 1U] != ' ')) {
            result[out] = ' ';
            out++;
        }

        result[out] = input[start];
        out++;
        offset = start + 1U;
    }

    if ((offset > len) || ((out + (len - offset) + 1U) > capacity)) {
        free(result);
        regfree(&regex);
        return NULL;
    }

    if (safe_copy(result, capacity, out, input + offset, len - offset) != 0) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    out += len - offset;

    if (out >= capacity) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    result[out] = '\0';

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "HelloWorldThisIsATestString";
    char *spaced;

    spaced = add_spaces(text);
    if (spaced == NULL) {
        fprintf(stderr, "Failed to process input string\n");
        return EXIT_FAILURE;
    }

    printf("Input : %s\n", text);
    printf("Output: %s\n", spaced);

    free(spaced);
    return EXIT_SUCCESS;
}

Analysis of the flagged weakness (CWE-120, `memcpy` in `safe_copy`): the Flawfinder warning is a false positive in this code. The `memcpy` call is already fully guarded — `safe_copy` validates that both pointers are non-NULL, that `dest_offset` does not exceed `dest_capacity`, and that `count` fits within the remaining space (`dest_capacity - dest_offset`), using subtraction rather than addition to avoid unsigned integer overflow. Every caller also pre-validates bounds (`out + chunk + 2U > capacity`, etc.) before writing, the input length is capped via `strnlen` against `MAX_INPUT_LEN`, and the output buffer is sized to the worst case (`2 * len + 1`, one inserted space per character plus the NUL terminator). All error paths correctly free the buffer and the compiled regex, so there are no leaks or overflows. The code stands as safe with no changes required beyond removing the comments as requested.
