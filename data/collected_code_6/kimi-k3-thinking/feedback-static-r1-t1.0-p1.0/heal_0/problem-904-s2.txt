```c
#include <stdio.h>
#include <stdbool.h>

bool isEven(int number)
{
    return (number % 2) == 0;
}

int main(void)
{
    int number;

    printf("Enter a number: ");

    if (scanf("%d", &number) != 1)
    {
        fprintf(stderr, "Invalid input.\n");
        return 1;
    }

    if (isEven(number))
    {
        printf("%d is even.\n", number);
    }
    else
    {
        printf("%d is odd.\n", number);
    }

    return 0;
}
```
