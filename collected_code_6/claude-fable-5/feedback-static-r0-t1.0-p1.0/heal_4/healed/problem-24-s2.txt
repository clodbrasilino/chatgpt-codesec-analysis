#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_LEN 128U

static int binary_to_decimal(const char *binary, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, MAX_INPUT_LEN);
    if (len == 0U || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_check_truncation(void)
{
    int ch;
    int truncated = 0;
    unsigned long guard = 0UL;

    while ((ch = getchar()) != '\n' && ch != EOF) {
        truncated = 1;
        guard++;
        if (guard >= ULONG_MAX) {
            break;
        }
    }

    return truncated;
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    unsigned long long decimal = 0ULL;
    size_t len;

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        if (flush_stdin_check_truncation() == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}

**Analysis of the original code and how the rewrite addresses the flagged weaknesses:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20):** The original loop in `flush_stdin_check_truncation` was already safe since it doesn't write to any buffer — it only discards characters. It correctly terminates on both `'\n'` and `EOF`, avoiding the classic infinite-loop bug when stdin is closed. The rewrite adds a guard counter as an extra defensive measure against pathological infinite streams, though the original was functionally sound.

2. **Flawfinder `char` array warning (CWE-119/CWE-120):** The statically-sized `input` buffer is properly protected: `fgets` is called with `sizeof(input)` limiting the read, a forced NUL terminator is written at the last index, and `strnlen` (rather than `strlen`) bounds all subsequent length calculations. The rewrite additionally zero-initializes the buffer so it is never read in an indeterminate state.

3. **Overflow safety in conversion:** The `value > (ULLONG_MAX >> 1)` check before each shift correctly prevents unsigned overflow (well-defined in C but semantically wrong here), and the digit validation rejects anything other than `'0'`/`'1'`. NULL-pointer checks on both parameters guard against misuse.

4. **Truncation detection:** If the newline is missing from the buffer, the code correctly distinguishes between an exact-length final line (EOF without newline, acceptable) and an over-long line (extra characters flushed, rejected) — a subtle correctness point many implementations get wrong.

The original code was already well-hardened; the flawfinder hits were low-risk informational findings, both mitigated by the existing bounds checking.
