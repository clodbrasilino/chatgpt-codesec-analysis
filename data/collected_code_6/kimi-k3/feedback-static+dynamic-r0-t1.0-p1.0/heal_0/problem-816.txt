#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    char name[50];
    double value;
} Tuple;

void clear_tuple(Tuple *t) {
    if (t != NULL) {
        t->id = 0;
        memset(t->name, 0, sizeof(t->name));
        t->value = 0.0;
    }
}

void clear_tuples(Tuple *tuples, size_t count) {
    size_t i;
    if (tuples != NULL && count > 0) {
        for (i = 0; i < count; i++) {
            clear_tuple(&tuples[i]);
        }
    }
}

int main(void) {
    Tuple tuples[3];
    size_t i;
    
    tuples[0].id = 1;
    strcpy(tuples[0].name, "First");
    tuples[0].value = 10.5;
    
    tuples[1].id = 2;
    strcpy(tuples[1].name, "Second");
    tuples[1].value = 20.75;
    
    tuples[2].id = 3;
    strcpy(tuples[2].name, "Third");
    tuples[2].value = 30.25;
    
    printf("Before clearing:\n");
    for (i = 0; i < 3; i++) {
        printf("Tuple %zu: id=%d, name=%s, value=%.2f\n", 
               i, tuples[i].id, tuples[i].name, tuples[i].value);
    }
    
    clear_tuples(tuples, 3);
    
    printf("\nAfter clearing:\n");
    for (i = 0; i < 3; i++) {
        printf("Tuple %zu: id=%d, name=%s, value=%.2f\n", 
               i, tuples[i].id, tuples[i].name, tuples[i].value);
    }
    
    return 0;
}
