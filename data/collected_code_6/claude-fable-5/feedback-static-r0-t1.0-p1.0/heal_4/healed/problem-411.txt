```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    const char *cursor;
    size_t out_pos;
    size_t len;
    size_t tail_len;
    size_t buf_size;
    int status;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    status = regcomp(&regex, "_([a-zA-Z])", REG_EXTENDED);
    if (status != 0) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    out_pos = 0U;
    cursor = input;

    while (regexec(&regex, cursor, 1U, &match, 0) == 0) {
        size_t chunk;
        size_t remaining;

        if (match.rm_so < 0 || match.rm_eo < 0 ||
            (size_t)match.rm_eo > strnlen(cursor, buf_size)) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        chunk = (size_t)match.rm_so;

        if (chunk > 0U) {
            if (out_pos >= buf_size) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            remaining = buf_size - out_pos;
            if (chunk >= remaining) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(result + out_pos, cursor, chunk);
            out_pos += chunk;
        }

        if (out_pos + 1U >= buf_size) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        result[out_pos] = (char)toupper((unsigned char)cursor[match.rm_so + 1]);
        out_pos++;
        cursor += match.rm_eo;
    }

    tail_len = strnlen(cursor, buf_size);
    if (out_pos >= buf_size) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    if (tail_len >= buf_size - out_pos) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    memcpy(result + out_pos, cursor, tail_len);
    out_pos += tail_len;

    if (out_pos >= buf_size) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    result[out_pos] = '\0';

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world_example",
        "convert_this_snake_case",
        "single",
        "a_b_c_d"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        char *camel = snake_to_camel(tests[i]);
        if (camel == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("%s -> %s\n", tests[i], camel) < 0) {
            free(camel);
            return EXIT_FAILURE;
        }
        free(camel);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) were already effectively mitigated in the original code, and the rewrite makes the guarantees more explicit:

1. **First `memcpy` (chunk copy):** Before copying, the code verifies `out_pos < buf_size` and `chunk < buf_size - out_pos`, guaranteeing `out_pos + chunk < buf_size`. The subtraction is safe from underflow because the `out_pos >= buf_size` check precedes it.

2. **Second `memcpy` (tail copy):** Same pattern — `tail_len < buf_size - out_pos` ensures the copy plus the subsequent NUL terminator fit within the allocation. An additional post-copy bounds check guards the terminator write.

Other defensive properties preserved:
- `strnlen` with `MAX_INPUT_LEN` prevents unbounded reads on non-terminated input.
- `regmatch_t` offsets are validated (`rm_so`/`rm_eo` non-negative, `rm_eo` within the cursor's remaining length) before use, protecting against pathological regex results.
- `toupper` receives an `unsigned char` cast, avoiding undefined behavior on negative `char` values.
- All error paths release both the regex and the heap buffer — no leaks.

Since camelCase output is always shorter than or equal to the snake_case input, `len + 1` is a provably sufficient buffer size, so the bounds checks can never fail on valid input; they exist purely as defense-in-depth.
