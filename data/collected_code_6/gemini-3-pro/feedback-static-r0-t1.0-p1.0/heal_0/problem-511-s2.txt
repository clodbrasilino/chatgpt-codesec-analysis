```c
#include <stdio.h>
#include <limits.h>

int findMinSumOfFactors(int n) {
    if (n <= 0) {
        return 0;
    }

    int minSum = n + 1;
    
    for (int i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            int factor1 = i;
            int factor2 = n / i;
            int currentSum = factor1 + factor2;
            
            if (currentSum < minSum) {
                minSum = currentSum;
            }
        }
    }
    
    return minSum;
}

int main() {
    int num;
    
    if (scanf("%d", &num) != 1) {
        return 1;
    }
    
    if (num <= 0) {
        return 1;
    }
    
    int result = findMinSumOfFactors(num);
    printf("%d\n", result);
    
    return 0;
}
```
