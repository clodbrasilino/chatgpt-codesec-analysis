#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_STRING_LENGTH 1024

int replace_whitespace_underscore(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t input_len;
    size_t output_pos = 0;
    size_t i;
    const char *pattern = "[ \t\n\r\f\v]|_";
    
    if (input == NULL || output == NULL || output_size == 0)
    {
        return -1;
    }
    
    input_len = strnlen(input, MAX_STRING_LENGTH);
    if (input_len == 0)
    {
        output[0] = '\0';
        return 0;
    }
    
    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0)
    {
        return -1;
    }
    
    i = 0;
    while (i < input_len)
    {
        ret = regexec(&regex, input + i, 1, &match, 0);
        
        if (ret == REG_NOMATCH)
        {
            size_t remaining = input_len - i;
            if (output_pos + remaining + 1 > output_size)
            {
                regfree(&regex);
                return -1;
            }
            memmove(output + output_pos, input + i, remaining);
            output_pos += remaining;
            break;
        }
        else if (ret != 0)
        {
            regfree(&regex);
            return -1;
        }
        
        if (match.rm_so < 0 || match.rm_eo < 0 || match.rm_so > match.rm_eo)
        {
            regfree(&regex);
            return -1;
        }
        
        size_t prefix_len = (size_t)match.rm_so;
        
        if (output_pos + prefix_len + 2 > output_size)
        {
            regfree(&regex);
            return -1;
        }
        
        if (prefix_len > 0)
        {
            memmove(output + output_pos, input + i, prefix_len);
            output_pos += prefix_len;
        }
        
        if (i + match.rm_so >= input_len)
        {
            regfree(&regex);
            return -1;
        }
        
        if (input[i + match.rm_so] == '_')
        {
            output[output_pos] = ' ';
        }
        else
        {
            output[output_pos] = '_';
        }
        output_pos++;
        
        i += (size_t)match.rm_eo;
    }
    
    if (output_pos >= output_size)
    {
        regfree(&regex);
        return -1;
    }
    
    output[output_pos] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    char *input = NULL;
    char *output = NULL;
    int result;
    size_t input_size = MAX_STRING_LENGTH + 1;
    size_t output_size = MAX_STRING_LENGTH * 2 + 1;
    
    input = (char *)calloc(input_size, sizeof(char));
    if (input == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    
    output = (char *)calloc(output_size, sizeof(char));
    if (output == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        free(input);
        return 1;
    }
    
    printf("Enter a string: ");
    if (fgets(input, input_size, stdin) == NULL)
    {
        fprintf(stderr, "Error reading input\n");
        free(input);
        free(output);
        return 1;
    }
    
    size_t len = strnlen(input, input_size);
    if (len > 0 && input[len - 1] == '\n')
    {
        input[len - 1] = '\0';
    }
    
    result = replace_whitespace_underscore(input, output, output_size);
    if (result != 0)
    {
        fprintf(stderr, "Error processing string\n");
        free(input);
        free(output);
        return 1;
    }
    
    printf("Original: %s\n", input);
    printf("Modified: %s\n", output);
    
    free(input);
    free(output);
    
    return 0;
}
