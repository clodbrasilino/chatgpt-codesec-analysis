```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 32

typedef struct {
    int key;
    char value[MAX_NAME_LEN];
} Tuple;

static void tuple_set(Tuple *t, int key, const char *name)
{
    t->key = key;
    (void)snprintf(t->value, sizeof(t->value), "%s", name);
}

static int sort_on_occurrence(const Tuple *input, size_t count, Tuple *output)
{
    size_t i;
    size_t j;
    size_t out_index;
    int *seen_keys;
    size_t seen_count;

    if (input == NULL || output == NULL || count == 0U) {
        return -1;
    }

    if (count > (size_t)-1 / sizeof(int)) {
        return -1;
    }

    seen_keys = (int *)malloc(count * sizeof(int));
    if (seen_keys == NULL) {
        return -1;
    }

    seen_count = 0U;
    out_index = 0U;

    for (i = 0U; i < count; i++) {
        int already_seen = 0;
        for (j = 0U; j < seen_count; j++) {
            if (seen_keys[j] == input[i].key) {
                already_seen = 1;
                break;
            }
        }
        if (already_seen == 0) {
            seen_keys[seen_count] = input[i].key;
            seen_count++;
            for (j = 0U; j < count; j++) {
                if (input[j].key == input[i].key) {
                    output[out_index].key = input[j].key;
                    (void)snprintf(output[out_index].value,
                                   sizeof(output[out_index].value),
                                   "%s", input[j].value);
                    out_index++;
                }
            }
        }
    }

    free(seen_keys);
    return 0;
}

int main(void)
{
    Tuple data[6];
    size_t count = sizeof(data) / sizeof(data[0]);
    Tuple *sorted;
    size_t i;
    int result;

    tuple_set(&data[0], 1, "Jake");
    tuple_set(&data[1], 2, "Bob");
    tuple_set(&data[2], 1, "Cara");
    tuple_set(&data[3], 3, "Ann");
    tuple_set(&data[4], 2, "Dan");
    tuple_set(&data[5], 1, "Eli");

    sorted = (Tuple *)malloc(count * sizeof(Tuple));
    if (sorted == NULL) {
        (void)fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    result = sort_on_occurrence(data, count, sorted);
    if (result != 0) {
        (void)fprintf(stderr, "Sorting failed\n");
        free(sorted);
        return EXIT_FAILURE;
    }

    for (i = 0U; i < count; i++) {
        if (printf("(%d, %s)\n", sorted[i].key, sorted[i].value) < 0) {
            free(sorted);
            return EXIT_FAILURE;
        }
    }

    free(sorted);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-119/CWE-120 (buffer overflow risk on `char value[MAX_NAME_LEN]`)**: All writes into the fixed-size buffer now go through `snprintf` with `sizeof()` as the bound, guaranteeing truncation instead of overflow and always NUL-terminating. The struct copies in `sort_on_occurrence` were replaced with explicit bounded field copies, and the initializer list in `main` was replaced with a `tuple_set` helper that performs bounds-checked copies.

2. **Variable scope (`already_seen`)**: The variable is now declared inside the outer `for` loop body, reducing its scope to the smallest region where it is used, as flagged by cppcheck.

3. **Defensive addition**: An overflow guard on `count * sizeof(int)` before `malloc` prevents integer-overflow-driven undersized allocations if the function is ever called with an attacker-controlled count.
