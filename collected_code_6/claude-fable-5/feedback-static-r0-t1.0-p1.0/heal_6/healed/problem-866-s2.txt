#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month);
static void discard_remaining_input(void);

static bool has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    memset(lowered, 0, sizeof(lowered));

    for (i = 0; i < len && i < sizeof(lowered) - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(lowered, months_31[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No input provided.\n");
        return EXIT_FAILURE;
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char lowered[MAX_INPUT_LEN]` and `char input[MAX_INPUT_LEN]` (CWE-119/CWE-120):** These are false positives in context. Both buffers are protected:
   - `input` is filled exclusively via `fgets(input, (int)sizeof(input), stdin)`, which enforces the buffer bound, followed by explicit NUL termination at `sizeof(input) - 1`.
   - `lowered` is written in a loop bounded by both `len` and `sizeof(lowered) - 1`, then explicitly NUL-terminated. Input length is validated with `strnlen` against `MAX_INPUT_LEN` before any copy occurs.

2. **`getchar()` in `discard_remaining_input` (CWE-120/CWE-20):** No buffer is involved; the characters are read and discarded. The loop terminates correctly on both `'\n'` and `EOF`, so it cannot spin forever on a closed stream.

Additional defensive measures already present: `NULL` checks, `tolower` receives an `unsigned char` cast (avoiding UB with negative `char` values), `memset` zero-initialization, and error checking on `fflush` and `fgets`. No changes were required beyond removing the comments as requested.
