#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* camel_to_snake(const char* camel_case) {
    if (camel_case == NULL) {
        return NULL;
    }

    if (camel_case[0] == '\0') {
        char* empty = (char*)malloc(1);
        if (empty) empty[0] = '\0';
        return empty;
    }

    size_t len = 0;
    size_t new_len = 0;
    while (camel_case[len] != '\0') {
        if (isupper((unsigned char)camel_case[len]) && len > 0) {
            new_len++;
        }
        new_len++;
        len++;
    }

    char* snake_case = (char*)malloc(new_len + 1);
    if (snake_case == NULL) {
        return NULL;
    }

    size_t j = 0;
    for (size_t i = 0; i < len; i++) {
        if (isupper((unsigned char)camel_case[i])) {
            if (i > 0) {
                snake_case[j++] = '_';
            }
            snake_case[j++] = tolower((unsigned char)camel_case[i]);
        } else {
            snake_case[j++] = camel_case[i];
        }
    }
    snake_case[j] = '\0';

    return snake_case;
}

int main(void) {
    char *input = NULL;
    size_t len = 0;
    
    if (getline(&input, &len, stdin) != -1) {
        size_t newline_pos = strcspn(input, "\n");
        input[newline_pos] = '\0';
        
        char* res = camel_to_snake(input);
        if (res) {
            printf("%s\n", res);
            free(res);
        }
    }
    
    free(input);
    return 0;
}
