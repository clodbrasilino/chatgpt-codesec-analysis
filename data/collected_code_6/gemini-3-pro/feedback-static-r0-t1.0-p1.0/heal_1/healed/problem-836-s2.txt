```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int findMaxSumSubarrayLength(const int arr[], int n) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    long long max_sum = INT_MIN;
    long long current_sum = 0;
    int max_len = 0;
    int current_len = 0;

    for (int i = 0; i < n; i++) {
        current_sum += arr[i];
        current_len++;

        if (current_sum > max_sum) {
            max_sum = current_sum;
            max_len = current_len;
        } else if (current_sum == max_sum) {
            if (current_len > max_len) {
                max_len = current_len;
            }
        }

        if (current_sum < 0) {
            current_sum = 0;
            current_len = 0;
        }
    }

    return max_len;
}

int main(void) {
    int arr[] = { -2, 1, -3, 4, -1, 2, 1, -5, 4 };
    int n = sizeof(arr) / sizeof(arr[0]);

    int length = findMaxSumSubarrayLength(arr, n);
    printf("Length of subarray with maximum sum is: %d\n", length);

    return 0;
}
```
