```c
#include <stdio.h>
#include <stdlib.h>

double calculate_rhombus_perimeter(double side) {
    if (side <= 0.0) {
        return -1.0;
    }
    return 4.0 * side;
}

int main(void) {
    double side = 0.0;
    double perimeter = 0.0;

    printf("Enter the side length of the rhombus: ");
    if (scanf("%lf", &side) != 1) {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    perimeter = calculate_rhombus_perimeter(side);

    if (perimeter < 0.0) {
        fprintf(stderr, "Side length must be positive.\n");
        return EXIT_FAILURE;
    }

    printf("The perimeter of the rhombus is: %.2lf\n", perimeter);

    return EXIT_SUCCESS;
}
```
