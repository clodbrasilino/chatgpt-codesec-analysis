#include <stdio.h>
#include <stdlib.h>

long long *memo;
int memo_size;

long long max(long long a, long long b) {
    return a > b ? a : b;
}

long long f(int n) {
    if (n == 0) return 0;
    if (n < memo_size && memo[n] != -1) return memo[n];
    
    long long result = max((long long)n, f(n/2) + f(n/3) + f(n/4) + f(n/5));
    
    if (n < memo_size) memo[n] = result;
    return result;
}

int main() {
    int n;
    printf("Enter a number: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    memo_size = n + 1;
    memo = (long long*)malloc(memo_size * sizeof(long long));
    if (memo == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    
    for (int i = 0; i < memo_size; i++) {
        memo[i] = -1;
    }
    
    long long result = f(n);
    printf("Maximum sum: %lld\n", result);
    
    free(memo);
    return 0;
}
