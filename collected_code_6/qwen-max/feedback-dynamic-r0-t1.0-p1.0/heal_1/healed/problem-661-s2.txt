#include <stdio.h>
#include <stdlib.h>

int max_sum_no_three_consecutive(int *arr, int n) {
    if (n == 0) return 0;
    if (n == 1) return arr[0];
    if (n == 2) return arr[0] + arr[1];
    if (n == 3) return (arr[0] + arr[1] > arr[1] + arr[2]) ? arr[0] + arr[1] : arr[1] + arr[2];

    int dp[n];
    dp[0] = arr[0];
    dp[1] = arr[0] + arr[1];
    dp[2] = (arr[0] + arr[1] > arr[0] + arr[2]) ? arr[0] + arr[1] : arr[0] + arr[2];

    for (int i = 3; i < n; i++) {
        dp[i] = ((dp[i-2] + arr[i]) > (dp[i-3] + arr[i-1] + arr[i])) ? (dp[i-2] + arr[i]) : (dp[i-3] + arr[i-1] + arr[i]);
        dp[i] = (dp[i-1] > dp[i]) ? dp[i-1] : dp[i];
    }

    return dp[n-1];
}

int main() {
    int arr[] = {12, 34, 15, 20, 30, 5, 1};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = max_sum_no_three_consecutive(arr, n);
    printf("%d\n", result);
    return 0;
}
