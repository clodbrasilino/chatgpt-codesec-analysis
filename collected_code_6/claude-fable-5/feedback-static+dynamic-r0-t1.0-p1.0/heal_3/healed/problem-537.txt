```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 1024
#define MAX_WORD_LEN 256
#define MAX_INPUT_LEN 65536

static char *duplicate_string(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, max_len);
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static int extract_word(const char *str, size_t str_len, size_t *pos,
                        char *word, size_t word_size)
{
    size_t i;
    size_t w = 0;

    if (str == NULL || pos == NULL || word == NULL || word_size == 0) {
        return 0;
    }

    i = *pos;

    while (i < str_len && str[i] != '\0' && !isalnum((unsigned char)str[i])) {
        i++;
    }

    if (i >= str_len || str[i] == '\0') {
        *pos = i;
        return 0;
    }

    while (i < str_len && str[i] != '\0' && isalnum((unsigned char)str[i])) {
        if (w < word_size - 1) {
            word[w] = str[i];
            w++;
        }
        i++;
    }

    word[w] = '\0';
    *pos = i;
    return 1;
}

char *first_repeated_word(const char *str)
{
    char (*words)[MAX_WORD_LEN];
    char current[MAX_WORD_LEN];
    size_t pos = 0;
    size_t count = 0;
    size_t i;
    size_t len;
    size_t str_len;
    char *result;

    if (str == NULL) {
        return duplicate_string("None", MAX_WORD_LEN);
    }

    str_len = strnlen(str, MAX_INPUT_LEN);

    words = malloc((size_t)MAX_WORDS * sizeof(*words));
    if (words == NULL) {
        return NULL;
    }

    while (extract_word(str, str_len, &pos, current, sizeof(current)) == 1) {
        for (i = 0; i < count; i++) {
            if (strncmp(words[i], current, MAX_WORD_LEN) == 0) {
                result = duplicate_string(current, MAX_WORD_LEN);
                free(words);
                return result;
            }
        }
        if (count < MAX_WORDS) {
            len = strnlen(current, sizeof(current));
            if (len >= MAX_WORD_LEN) {
                len = MAX_WORD_LEN - 1;
            }
            memcpy(words[count], current, len);
            words[count][len] = '\0';
            count++;
        } else {
            break;
        }
    }

    free(words);
    return duplicate_string("None", MAX_WORD_LEN);
}

static int run_test(const char *input)
{
    char *repeated;

    repeated = first_repeated_word(input);
    if (repeated == NULL) {
        if (printf("Memory allocation failure\n") < 0) {
            return -1;
        }
        return -1;
    }

    if (strncmp(repeated, "None", MAX_WORD_LEN) == 0) {
        if (printf("No repeated word found\n") < 0) {
            free(repeated);
            return -1;
        }
    } else {
        if (printf("First repeated word: %s\n", repeated) < 0) {
            free(repeated);
            return -1;
        }
    }

    free(repeated);
    return 0;
}

int main(void)
{
    const char *test1 = "the quick brown fox jumps over the lazy dog";
    const char *test2 = "hello world foo bar";
    const char *test3 = "one two two three";

    if (run_test(test1) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test2) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test3) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes addressing the flagged weaknesses:

- **CWE-126 (`strlen` over-read)**: Replaced all `strlen` calls with `strnlen` bounded by an explicit maximum (`MAX_WORD_LEN` for words, `MAX_INPUT_LEN` for input strings), so a non-terminated buffer can no longer cause an unbounded read.
- **CWE-120 (`memcpy` overflow)**: Every `memcpy` destination size is now provably sufficient — `duplicate_string` allocates `len + 1` based on the bounded length, and the copy into `words[count]` clamps `len` to `MAX_WORD_LEN - 1` before copying.
- **CWE-119/CWE-120 (statically-sized arrays)**: `extract_word` now takes an explicit `str_len` bound and checks `i < str_len` on every iteration, and it already truncates words to `word_size - 1` before writing. All indexing into `current` and `words` is bounds-checked against `MAX_WORD_LEN` and `MAX_WORDS`.
- **Defense in depth**: `strcmp` was replaced with `strncmp` bounded by `MAX_WORD_LEN`, so comparison can never run past the fixed-size word buffers even if a terminator were somehow missing.

The program's observable behavior is unchanged: it still reports the first repeated word for each test string ("the", "None", "two").
