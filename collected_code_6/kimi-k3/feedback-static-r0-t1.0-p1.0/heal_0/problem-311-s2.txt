#include <stdio.h>
#include <stdint.h>
#include <limits.h>

unsigned int set_leftmost_unset_bit(unsigned int n) {
    if (n == UINT_MAX) {
        return n;
    }
    
    unsigned int temp = n;
    unsigned int mask = 0;
    
    while (temp != 0) {
        temp >>= 1;
        mask = (mask << 1) | 1;
    }
    
    if (mask == 0) {
        return n | 1;
    }
    
    unsigned int leftmost_unset = (mask + 1) >> 1;
    
    if (leftmost_unset == 0) {
        return n | 1;
    }
    
    return n | leftmost_unset;
}

int main(void) {
    unsigned int test_cases[] = {0, 1, 5, 10, 15, 16, 31, 32, 63, 64, 127, 128, UINT_MAX};
    size_t num_tests = sizeof(test_cases) / sizeof(test_cases[0]);
    
    for (size_t i = 0; i < num_tests; i++) {
        unsigned int original = test_cases[i];
        unsigned int result = set_leftmost_unset_bit(original);
        printf("Input: %u (0x%X), Output: %u (0x%X)\n", 
               original, original, result, result);
    }
    
    return 0;
}
