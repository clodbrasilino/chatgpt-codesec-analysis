```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

static char *duplicate_word(const char *word, size_t len)
{
    size_t alloc_size;
    char *copy;

    if (word == NULL || len >= MAX_WORD_LEN) {
        return NULL;
    }

    alloc_size = len + 1;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        if (len >= alloc_size) {
            free(copy);
            return NULL;
        }
        memcpy(copy, word, len);
    }
    copy[len] = '\0';

    return copy;
}

char **remove_duplicates(const char **words, size_t count, size_t *out_count)
{
    char **result = NULL;
    size_t unique = 0;
    size_t i;
    size_t j;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strcmp(result[j], words[i]) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            result[unique] = duplicate_word(words[i], len);
            if (result[unique] == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Unique words:\n");
    for (i = 0; i < result_count; i++) {
        printf("%s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weakness:

1. **Isolated the copy logic** into `duplicate_word()`, which validates that the source length is strictly less than the allocated destination size (`len >= alloc_size` guard) before calling `memcpy`, directly addressing the CWE-120 flawfinder warning that the destination must always hold the source data.
2. **Bounds validation before allocation** — the function rejects `len >= MAX_WORD_LEN` inputs and only copies `len` bytes into a buffer of `len + 1`, guaranteeing room for the null terminator, which is written explicitly.
3. **Overflow guard on `calloc`** — added a `count > SIZE_MAX / sizeof(char *)` check to prevent any theoretical size calculation overflow on the pointer array allocation.
4. All original behavior (NULL handling, duplicate detection, cleanup on allocation failure) is preserved, and the original code's `memcpy` was in fact already safe — the fix makes the safety invariant explicit and statically checkable.
