```c
#include <stdio.h>
#include <stdlib.h>

long long smallest_prime_divisor(long long n)
{
    long long i;

    if (n < 2) {
        return -1;
    }
    if (n % 2 == 0) {
        return 2;
    }
    for (i = 3; i <= n / i; i += 2) {
        if (n % i == 0) {
            return i;
        }
    }
    return n;
}

int main(void)
{
    long long n;
    long long result;
    int status;

    printf("Enter a number: ");
    status = scanf("%lld", &n);
    if (status != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    if (n < 2) {
        fprintf(stderr, "Number must be greater than or equal to 2\n");
        return EXIT_FAILURE;
    }

    result = smallest_prime_divisor(n);
    if (result < 0) {
        fprintf(stderr, "No prime divisor found\n");
        return EXIT_FAILURE;
    }

    printf("Smallest prime divisor of %lld is %lld\n", n, result);
    return EXIT_SUCCESS;
}
```

Key design decisions to address the security concerns:

- **CWE-703**: The return value of `scanf` is checked before using `n`, and error paths return `EXIT_FAILURE` with a diagnostic on `stderr`. The function also returns a sentinel for invalid domain input, which the caller checks.
- **CWE-664**: No dynamic resources (memory, file handles) are acquired, so there is nothing to leak or misuse across a lifetime; only `stdin`/`stdout`/`stderr` managed by the runtime are used.
- **CWE-710**: The code uses `int main(void)`, standard exit macros, `long long` with matching `%lld` format specifiers, and avoids overflow in the loop condition by using `i <= n / i` instead of `i * i <= n`.
