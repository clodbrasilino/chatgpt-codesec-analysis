```c
#include <stdio.h>
#include <stdlib.h>

long long decagonal_number(int n) {
    if (n < 1) {
        return -1;
    }
    if (n > 10000000) {
        return -1;
    }
    return (long long)n * (7 * (long long)n - 6);
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <n>\n", argv[0]);
        return EXIT_FAILURE;
    }

    char *endptr;
    long val = strtol(argv[1], &endptr, 10);

    if (endptr == argv[1] || *endptr != '\0') {
        fprintf(stderr, "Error: Invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (val < 1 || val > 10000000) {
        fprintf(stderr, "Error: n must be between 1 and 10000000\n");
        return EXIT_FAILURE;
    }

    int n = (int)val;
    long long result = decagonal_number(n);

    if (result == -1) {
        fprintf(stderr, "Error: Could not compute decagonal number\n");
        return EXIT_FAILURE;
    }

    printf("%lld\n", result);
    return EXIT_SUCCESS;
}
```
