#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <limits.h>
#include <errno.h>

int extract_max_numeric(const char *str, size_t str_len, double *max_value) {
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[0-9]+(\\.[0-9]+)?";
    int ret;
    int found = 0;
    double current_max = 0.0;
    
    if (str == NULL || max_value == NULL) {
        return -1;
    }
    
    if (str_len == 0) {
        return 1;
    }
    
    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }
    
    const char *cursor = str;
    size_t remaining_len = str_len;
    
    while (remaining_len > 0) {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret != 0) {
            break;
        }
        
        if (match.rm_so < 0 || match.rm_eo < 0 || match.rm_eo <= match.rm_so) {
            break;
        }
        
        size_t start = (size_t)match.rm_so;
        size_t end = (size_t)match.rm_eo;
        
        if (end > remaining_len || start >= remaining_len) {
            break;
        }
        
        size_t len = end - start;
        
        if (len > remaining_len - start) {
            break;
        }
        
        char *num_str = (char *)malloc(len + 1);
        if (num_str == NULL) {
            regfree(&regex);
            return -1;
        }
        
        if (len > 0 && len <= remaining_len - start) {
            memcpy(num_str, cursor + start, len);
            num_str[len] = '\0';
        } else {
            free(num_str);
            break;
        }
        
        char *endptr;
        errno = 0;
        double value = strtod(num_str, &endptr);
        
        if (endptr != num_str && errno != ERANGE) {
            if (!found || value > current_max) {
                current_max = value;
                found = 1;
            }
        }
        
        free(num_str);
        
        if (end > remaining_len) {
            break;
        }
        
        cursor += end;
        remaining_len -= end;
    }
    
    regfree(&regex);
    
    if (found) {
        *max_value = current_max;
        return 0;
    }
    
    return 1;
}

int main(void) {
    const char *test_strings[] = {
        "The values are 42, 3.14, and 100",
        "No numbers here",
        "Max is 999.99 and min is 0.001",
        "12345",
        ""
    };
    
    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);
    
    for (size_t i = 0; i < num_tests; i++) {
        double max_val;
        
        if (test_strings[i] == NULL) {
            printf("String: (null) -> Error occurred\n");
            continue;
        }
        
        size_t str_len = strlen(test_strings[i]);
        int result = extract_max_numeric(test_strings[i], str_len, &max_val);
        
        if (result == 0) {
            printf("String: \"%s\" -> Max: %f\n", test_strings[i], max_val);
        } else if (result == 1) {
            printf("String: \"%s\" -> No numeric value found\n", test_strings[i]);
        } else {
            printf("String: \"%s\" -> Error occurred\n", test_strings[i]);
        }
    }
    
    return 0;
}
