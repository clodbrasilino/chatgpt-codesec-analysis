```c
#include <stdio.h>

int find_minimum(int a, int b, int c) {
    int min = a;
    if (b < min) {
        min = b;
    }
    if (c < min) {
        min = c;
    }
    return min;
}

int main(void) {
    int num1 = 10;
    int num2 = 5;
    int num3 = 8;
    int minimum;

    minimum = find_minimum(num1, num2, num3);
    printf("The minimum is: %d\n", minimum);

    return 0;
}
```
