```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT 1024

char *snake_to_camel(const char *snake, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    int capitalize_next;
    char *camel;

    if (snake == NULL || max_len == 0) {
        return NULL;
    }

    len = 0;
    while (len < max_len && snake[len] != '\0') {
        len++;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    j = 0;
    i = 0;
    capitalize_next = 1;
    while (i < len) {
        if (snake[i] == '_') {
            capitalize_next = 1;
            i++;
        } else {
            if (capitalize_next) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
            i++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT];
    char *result;
    size_t len;

    memset(input, 0, sizeof(input));

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = 0;
    while (len < sizeof(input) && input[len] != '\0') {
        len++;
    }

    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input, sizeof(input));
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-119/CWE-120 (statically-sized buffer `input`):** The original code was already largely safe because `fgets(input, sizeof(input), stdin)` bounds the read. The rewrite hardens it further by zero-initializing the buffer with `memset` and explicitly forcing a NUL terminator at `input[sizeof(input) - 1]` as defense-in-depth against any non-conforming `fgets` behavior.

2. **CWE-126 (`strlen` over-read on non-terminated strings):** The unbounded `strlen` call was replaced with a manual bounded scan (equivalent to `strnlen(input, sizeof(input))`, but portable to strict C standards where `strnlen` may be unavailable). Combined with the guaranteed NUL termination, an over-read past the buffer is now impossible.

3. **`snake_to_camel` hardening:** Added a `max_len == 0` guard so the function never allocates or writes based on a degenerate length, and the existing bounded length computation already prevents over-reads of the source string.

4. **Existing good practices preserved:** every `malloc` result is checked, `result` is freed on all exit paths, `printf`'s return value is checked, and `toupper` receives an `unsigned char` cast to avoid undefined behavior on negative `char` values.
