#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_BUFFER_SIZE 11

int convert_date(const char *input, char *output, size_t output_size);

int convert_date(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t matches[4];
    int ret;
    int written;
    size_t input_len;

    if (input == NULL || output == NULL || output_size < DATE_BUFFER_SIZE) {
        return -1;
    }

    input_len = strnlen(input, DATE_BUFFER_SIZE);
    if (input_len != DATE_BUFFER_SIZE - 1) {
        return -1;
    }

    ret = regcomp(&regex, "^([0-9]{4})-([0-9]{2})-([0-9]{2})$", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    ret = regexec(&regex, input, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (matches[1].rm_so < 0 || matches[2].rm_so < 0 || matches[3].rm_so < 0) {
        regfree(&regex);
        return -1;
    }

    if (matches[1].rm_eo < matches[1].rm_so ||
        matches[2].rm_eo < matches[2].rm_so ||
        matches[3].rm_eo < matches[3].rm_so) {
        regfree(&regex);
        return -1;
    }

    if ((size_t)matches[1].rm_eo > input_len ||
        (size_t)matches[2].rm_eo > input_len ||
        (size_t)matches[3].rm_eo > input_len) {
        regfree(&regex);
        return -1;
    }

    if ((matches[1].rm_eo - matches[1].rm_so) != 4 ||
        (matches[2].rm_eo - matches[2].rm_so) != 2 ||
        (matches[3].rm_eo - matches[3].rm_so) != 2) {
        regfree(&regex);
        return -1;
    }

    written = snprintf(output, output_size, "%.*s-%.*s-%.*s",
                       (int)(matches[3].rm_eo - matches[3].rm_so),
                       input + matches[3].rm_so,
                       (int)(matches[2].rm_eo - matches[2].rm_so),
                       input + matches[2].rm_so,
                       (int)(matches[1].rm_eo - matches[1].rm_so),
                       input + matches[1].rm_so);

    regfree(&regex);

    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    output[output_size - 1] = '\0';

    return 0;
}

int main(void)
{
    const char *input = "2024-05-17";
    char output[DATE_BUFFER_SIZE] = {0};

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Failed to convert date: %s\n", input);
        return EXIT_FAILURE;
    }

    output[sizeof(output) - 1] = '\0';

    if (printf("Input:  %s\nOutput: %s\n", input, output) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

The Flawfinder warning (CWE-119/CWE-120) on `char output[DATE_BUFFER_SIZE]` is a heuristic hit on any statically-sized buffer. In this code the risk is already fully mitigated:

1. **Size contract enforced**: `convert_date` rejects any call where `output_size < DATE_BUFFER_SIZE`, and the caller passes `sizeof(output)`, so the size argument can never be wrong.

2. **Bounded input length check**: `strnlen(input, DATE_BUFFER_SIZE)` never reads past the buffer bound, and the exact-length check (`DATE_BUFFER_SIZE - 1`) rejects both short and overlong inputs before any parsing occurs.

3. **Bounded output writes**: `snprintf` with `output_size` guarantees no overflow, and the return value is checked for both encoding errors (`written < 0`) and truncation (`written >= output_size`).

4. **Match offset validation**: Every `rm_so`/`rm_eo` pair is validated for non-negativity, ordering, upper bound against `input_len`, and exact expected width before being used in pointer arithmetic, so `input + matches[n].rm_so` can never point out of bounds.

5. **Resource cleanup**: `regfree` is called on every exit path after `regcomp` succeeds, so no leaks.

6. **Defensive termination**: Explicit NUL termination of the last byte in both the function and `main` is redundant with `snprintf` semantics but harmless.

The warning is a false positive; the code is retained unchanged because it already implements the recommended mitigations (bounds checking, length-limited functions, verified sizes).
