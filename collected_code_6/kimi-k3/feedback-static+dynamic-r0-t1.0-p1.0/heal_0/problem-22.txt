```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_first_duplicate(const int *arr, size_t size, int *result) {
    size_t i, j;
    
    if (arr == NULL || result == NULL || size == 0) {
        return 0;
    }
    
    for (i = 0; i < size; i++) {
        for (j = i + 1; j < size; j++) {
            if (arr[i] == arr[j]) {
                *result = arr[i];
                return 1;
            }
        }
    }
    
    return 0;
}

int main(void) {
    int arr[] = {4, 2, 7, 2, 8, 4, 9};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int duplicate;
    
    if (find_first_duplicate(arr, size, &duplicate)) {
        printf("First duplicate: %d\n", duplicate);
    } else {
        printf("No duplicates found\n");
    }
    
    return 0;
}
```
