#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

bool is_palindrome(const char *str) {
    if (str == NULL) {
        return false;
    }
    size_t len = strlen(str);
    if (len == 0) {
        return true;
    }
    size_t left = 0;
    size_t right = len - 1;
    while (left < right) {
        if (str[left] != str[right]) {
            return false;
        }
        left++;
        right--;
    }
    return true;
}

int find_palindromes(char **strings, int count, char ***result) {
    if (strings == NULL || result == NULL || count < 0) {
        return -1;
    }
    if (count == 0) {
        *result = NULL;
        return 0;
    }
    char **palindromes = malloc((size_t)count * sizeof(char *));
    if (palindromes == NULL) {
        return -1;
    }
    int pal_count = 0;
    for (int i = 0; i < count; i++) {
        if (strings[i] != NULL && is_palindrome(strings[i])) {
            palindromes[pal_count] = malloc(strlen(strings[i]) + 1);
            if (palindromes[pal_count] == NULL) {
                for (int j = 0; j < pal_count; j++) {
                    free(palindromes[j]);
                }
                free(palindromes);
                return -1;
            }
            strcpy(palindromes[pal_count], strings[i]);
            pal_count++;
        }
    }
    if (pal_count == 0) {
        free(palindromes);
        *result = NULL;
        return 0;
    }
    char **temp = realloc(palindromes, (size_t)pal_count * sizeof(char *));
    if (temp == NULL) {
        for (int i = 0; i < pal_count; i++) {
            free(palindromes[i]);
        }
        free(palindromes);
        return -1;
    }
    *result = temp;
    return pal_count;
}

void free_palindromes(char **palindromes, int count) {
    if (palindromes == NULL) {
        return;
    }
    for (int i = 0; i < count; i++) {
        free(palindromes[i]);
    }
    free(palindromes);
}

int main(void) {
    char *test_strings[] = {
        "racecar",
        "hello",
        "level",
        "world",
        "madam",
        "noon",
        "example"
    };
    int num_strings = 7;
    char **palindromes = NULL;
    int result = find_palindromes(test_strings, num_strings, &palindromes);
    if (result < 0) {
        fprintf(stderr, "Error: Failed to allocate memory\n");
        return EXIT_FAILURE;
    }
    printf("Found %d palindrome(s):\n", result);
    for (int i = 0; i < result; i++) {
        printf("%s\n", palindromes[i]);
    }
    free_palindromes(palindromes, result);
    return EXIT_SUCCESS;
}
