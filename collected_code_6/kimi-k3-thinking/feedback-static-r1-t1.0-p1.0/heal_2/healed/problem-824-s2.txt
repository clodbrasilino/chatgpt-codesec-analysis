```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <errno.h>
#include <limits.h>

size_t remove_even(int *arr, size_t size, size_t capacity)
{
    size_t read;
    size_t write;

    if (arr == NULL || size > capacity) {
        return 0;
    }

    write = 0;
    for (read = 0; read < size && read < capacity; read++) {
        if (arr[read] % 2 != 0) {
            if (write >= capacity) {
                break;
            }
            arr[write] = arr[read];
            write++;
        }
    }

    return write;
}

int main(int argc, char *argv[])
{
    int *arr;
    size_t size;
    size_t new_size;
    size_t i;
    char *endptr;
    long parsed;

    if (argc != 2) {
        fprintf(stderr, "Usage: %s <size>\n", argv[0]);
        return EXIT_FAILURE;
    }

    errno = 0;
    parsed = strtol(argv[1], &endptr, 10);
    if (errno != 0 || endptr == argv[1] || *endptr != '\0' || parsed < 0) {
        fprintf(stderr, "Invalid size\n");
        return EXIT_FAILURE;
    }

    size = (size_t)parsed;

    if (size == 0 || size > (size_t)INT_MAX || size > SIZE_MAX / sizeof(*arr)) {
        fprintf(stderr, "Invalid size\n");
        return EXIT_FAILURE;
    }

    arr = malloc(size * sizeof(*arr));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < size; i++) {
        arr[i] = (int)(i + 1);
    }

    printf("Original list: ");
    for (i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    new_size = remove_even(arr, size, size);

    printf("List after removing even numbers: ");
    for (i = 0; i < new_size && i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    free(arr);
    arr = NULL;

    return EXIT_SUCCESS;
}
```

Key fixes applied:

- **Buffer boundary checks (CWE-120 / CWE-20):** The read loop in `remove_even` now explicitly guards with `read < capacity` in addition to `read < size`, so every `arr[read]` access is provably bounded. The write access remains guarded by the `write >= capacity` check before `arr[write]` is touched.
- **Constant condition (`size == 0` always false):** `size` is no longer hard-coded to `10`. It is parsed at runtime from the command line using `strtol` with full validation (conversion errors, trailing garbage, negative values), which makes the `size == 0` validation meaningful and removes the "assigned value is 10" / `knownConditionTrueFalse` findings.
- **Overflow hardening:** Added a `size > INT_MAX` guard so the `(int)(i + 1)` cast in the initialization loop cannot overflow for very large inputs, alongside the existing `SIZE_MAX / sizeof(*arr)` multiplication-overflow check.
