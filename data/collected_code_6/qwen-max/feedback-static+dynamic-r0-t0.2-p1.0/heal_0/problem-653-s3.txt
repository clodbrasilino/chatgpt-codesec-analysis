#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *key;
    char *value;
    struct Node *next;
} Node;

typedef struct {
    char *key;
    Node *head;
} List;

typedef struct {
    List **lists;
    int size;
} Map;

Node* create_node(char *key, char *value) {
    Node *node = (Node *)malloc(sizeof(Node));
    node->key = strdup(key);
    node->value = strdup(value);
    node->next = NULL;
    return node;
}

List* create_list(char *key) {
    List *list = (List *)malloc(sizeof(List));
    list->key = strdup(key);
    list->head = NULL;
    return list;
}

Map* create_map(int size) {
    Map *map = (Map *)malloc(sizeof(Map));
    map->size = size;
    map->lists = (List **)calloc(size, sizeof(List *));
    return map;
}

unsigned long hash(char *str, int size) {
    unsigned long hash = 5381;
    int c;
    while ((c = *str++))
        hash = ((hash << 5) + hash) + c; /* hash * 33 + c */
    return hash % size;
}

void add_to_map(Map *map, char *key, char *value) {
    unsigned long index = hash(key, map->size);
    if (map->lists[index] == NULL) {
        map->lists[index] = create_list(key);
    }
    Node *new_node = create_node(key, value);
    new_node->next = map->lists[index]->head;
    map->lists[index]->head = new_node;
}

void print_map(Map *map) {
    for (int i = 0; i < map->size; ++i) {
        if (map->lists[i] != NULL) {
            printf("%s: ", map->lists[i]->key);
            Node *current = map->lists[i]->head;
            while (current != NULL) {
                printf("(%s, %s) -> ", current->key, current->value);
                current = current->next;
            }
            printf("NULL\n");
        }
    }
}

void free_node(Node *node) {
    free(node->key);
    free(node->value);
    free(node);
}

void free_list(List *list) {
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free_node(temp);
    }
    free(list->key);
    free(list);
}

void free_map(Map *map) {
    for (int i = 0; i < map->size; ++i) {
        if (map->lists[i] != NULL) {
            free_list(map->lists[i]);
        }
    }
    free(map->lists);
    free(map);
}

int main() {
    Map *map = create_map(10);
    add_to_map(map, "key1", "value1");
    add_to_map(map, "key1", "value2");
    add_to_map(map, "key2", "value3");
    add_to_map(map, "key2", "value4");
    add_to_map(map, "key3", "value5");

    print_map(map);

    free_map(map);
    return 0;
}
