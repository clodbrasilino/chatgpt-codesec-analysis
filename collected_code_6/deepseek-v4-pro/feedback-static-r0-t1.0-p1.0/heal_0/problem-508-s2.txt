#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    int size;
} List;

List *create_list(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->size = 0;
    return list;
}

void destroy_list(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

int append_list(List *list, int value) {
    if (list == NULL) {
        return 0;
    }
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return 0;
    }
    new_node->data = value;
    new_node->next = NULL;
    
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->size++;
    return 1;
}

int contains_value(List *list, int value) {
    if (list == NULL) {
        return 0;
    }
    Node *current = list->head;
    while (current != NULL) {
        if (current->data == value) {
            return 1;
        }
        current = current->next;
    }
    return 0;
}

int common_elements_same_order(List *list1, List *list2) {
    if (list1 == NULL || list2 == NULL) {
        return 0;
    }
    if (list1->head == NULL || list2->head == NULL) {
        return 0;
    }
    
    int *common_values = (int *)malloc(sizeof(int) * list2->size);
    if (common_values == NULL) {
        return 0;
    }
    
    int common_count = 0;
    Node *current2 = list2->head;
    while (current2 != NULL) {
        if (contains_value(list1, current2->data)) {
            common_values[common_count] = current2->data;
            common_count++;
        }
        current2 = current2->next;
    }
    
    if (common_count == 0) {
        free(common_values);
        return 0;
    }
    
    int last_position = -1;
    int result = 1;
    Node *current1 = list1->head;
    int position = 0;
    
    while (current1 != NULL && result) {
        for (int i = 0; i < common_count; i++) {
            if (current1->data == common_values[i]) {
                if (position < last_position) {
                    result = 0;
                    break;
                }
                last_position = position;
                break;
            }
        }
        current1 = current1->next;
        position++;
    }
    
    free(common_values);
    return result;
}

int main(void) {
    List *list1 = create_list();
    List *list2 = create_list();
    
    if (list1 == NULL || list2 == NULL) {
        destroy_list(list1);
        destroy_list(list2);
        return 1;
    }
    
    append_list(list1, 1);
    append_list(list1, 2);
    append_list(list1, 3);
    append_list(list1, 4);
    append_list(list1, 5);
    
    append_list(list2, 2);
    append_list(list2, 1);
    append_list(list2, 4);
    append_list(list2, 3);
    append_list(list2, 5);
    
    int result = common_elements_same_order(list1, list2);
    printf("Common elements in same order: %s\n", result ? "Yes" : "No");
    
    destroy_list(list1);
    destroy_list(list2);
    
    return 0;
}
