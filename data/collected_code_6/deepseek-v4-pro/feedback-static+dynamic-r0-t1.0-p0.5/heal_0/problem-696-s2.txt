#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    void *data;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    Node *tail;
    int size;
} List;

typedef struct ZipResult {
    List **lists;
    int size;
} ZipResult;

List *list_create(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    return list;
}

int list_append(List *list, void *data) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        return -1;
    }
    node->data = data;
    node->next = NULL;
    if (list->tail == NULL) {
        list->head = node;
        list->tail = node;
    } else {
        list->tail->next = node;
        list->tail = node;
    }
    list->size++;
    return 0;
}

void *list_get(List *list, int index) {
    if (list == NULL || index < 0 || index >= list->size) {
        return NULL;
    }
    Node *current = list->head;
    for (int i = 0; i < index; i++) {
        current = current->next;
    }
    return current->data;
}

void list_destroy(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    free(list);
}

void list_destroy_all(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        List *inner_list = (List *)current->data;
        list_destroy(inner_list);
        free(current);
        current = next;
    }
    free(list);
}

ZipResult *zip_lists(List *list1, List *list2) {
    if (list1 == NULL || list2 == NULL) {
        return NULL;
    }
    
    int min_size = list1->size < list2->size ? list1->size : list2->size;
    
    ZipResult *result = (ZipResult *)malloc(sizeof(ZipResult));
    if (result == NULL) {
        return NULL;
    }
    
    result->lists = (List **)malloc(sizeof(List *) * min_size);
    if (result->lists == NULL) {
        free(result);
        return NULL;
    }
    result->size = min_size;
    
    for (int i = 0; i < min_size; i++) {
        List *inner_list1 = (List *)list_get(list1, i);
        List *inner_list2 = (List *)list_get(list2, i);
        
        if (inner_list1 == NULL || inner_list2 == NULL) {
            for (int j = 0; j < i; j++) {
                list_destroy(result->lists[j]);
            }
            free(result->lists);
            free(result);
            return NULL;
        }
        
        List *zipped = list_create();
        if (zipped == NULL) {
            for (int j = 0; j < i; j++) {
                list_destroy(result->lists[j]);
            }
            free(result->lists);
            free(result);
            return NULL;
        }
        
        int inner_min = inner_list1->size < inner_list2->size ? 
                        inner_list1->size : inner_list2->size;
        
        for (int j = 0; j < inner_min; j++) {
            void *data1 = list_get(inner_list1, j);
            void *data2 = list_get(inner_list2, j);
            
            if (list_append(zipped, data1) != 0) {
                list_destroy(zipped);
                for (int k = 0; k < i; k++) {
                    list_destroy(result->lists[k]);
                }
                free(result->lists);
                free(result);
                return NULL;
            }
            
            if (list_append(zipped, data2) != 0) {
                list_destroy(zipped);
                for (int k = 0; k < i; k++) {
                    list_destroy(result->lists[k]);
                }
                free(result->lists);
                free(result);
                return NULL;
            }
        }
        
        result->lists[i] = zipped;
    }
    
    return result;
}

void zip_result_destroy(ZipResult *result) {
    if (result == NULL) {
        return;
    }
    for (int i = 0; i < result->size; i++) {
        list_destroy(result->lists[i]);
    }
    free(result->lists);
    free(result);
}

int main(void) {
    List *outer1 = list_create();
    List *outer2 = list_create();
    
    if (outer1 == NULL || outer2 == NULL) {
        fprintf(stderr, "Failed to allocate memory\n");
        return 1;
    }
    
    List *inner1_1 = list_create();
    List *inner1_2 = list_create();
    List *inner2_1 = list_create();
    List *inner2_2 = list_create();
    
    if (inner1_1 == NULL || inner1_2 == NULL || 
        inner2_1 == NULL || inner2_2 == NULL) {
        fprintf(stderr, "Failed to allocate memory\n");
        list_destroy(outer1);
        list_destroy(outer2);
        return 1;
    }
    
    int data1 = 1, data2 = 2, data3 = 3, data4 = 4;
    int data5 = 5, data6 = 6, data7 = 7, data8 = 8;
    
    list_append(inner1_1, &data1);
    list_append(inner1_1, &data2);
    list_append(inner1_2, &data3);
    list_append(inner1_2, &data4);
    
    list_append(inner2_1, &data5);
    list_append(inner2_1, &data6);
    list_append(inner2_2, &data7);
    list_append(inner2_2, &data8);
    
    list_append(outer1, inner1_1);
    list_append(outer1, inner1_2);
    list_append(outer2, inner2_1);
    list_append(outer2, inner2_2);
    
    ZipResult *result = zip_lists(outer1, outer2);
    
    if (result != NULL) {
        for (int i = 0; i < result->size; i++) {
            printf("Zipped list %d: ", i);
            List *zipped = result->lists[i];
            Node *current = zipped->head;
            while (current != NULL) {
                printf("%d ", *(int *)current->data);
                current = current->next;
            }
            printf("\n");
        }
        zip_result_destroy(result);
    }
    
    list_destroy_all(outer1);
    list_destroy_all(outer2);
    
    return 0;
}
