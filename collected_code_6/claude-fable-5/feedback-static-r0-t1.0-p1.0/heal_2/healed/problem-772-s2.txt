```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096U

char *remove_words_of_length_k(const char *input, size_t k)
{
    char *result;
    size_t len;
    size_t capacity;
    size_t i;
    size_t out;
    size_t start;
    size_t word_len;
    size_t j;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LENGTH);
    if (len >= MAX_INPUT_LENGTH) {
        return NULL;
    }

    capacity = len + 1U;
    result = (char *)malloc(capacity);
    if (result == NULL) {
        return NULL;
    }

    i = 0U;
    out = 0U;

    while (i <= len) {
        if ((input[i] == ' ') || (input[i] == '\0')) {
            i++;
            continue;
        }

        start = i;
        while ((i < len) && (input[i] != ' ') && (input[i] != '\0')) {
            i++;
        }
        word_len = i - start;

        if (word_len != k) {
            if (out > 0U) {
                if ((out + 1U) >= capacity) {
                    free(result);
                    return NULL;
                }
                result[out] = ' ';
                out++;
            }
            if (word_len > (capacity - 1U - out)) {
                free(result);
                return NULL;
            }
            for (j = 0U; j < word_len; j++) {
                result[out + j] = input[start + j];
            }
            out += word_len;
        }
    }

    if (out >= capacity) {
        free(result);
        return NULL;
    }
    result[out] = '\0';
    return result;
}

int main(void)
{
    const char *text = "the quick brown fox jumps over a lazy dog";
    size_t k = 3U;
    char *filtered;

    filtered = remove_words_of_length_k(text, k);
    if (filtered == NULL) {
        (void)fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", text) < 0) {
        free(filtered);
        return EXIT_FAILURE;
    }

    if (printf("Filtered: %s\n", filtered) < 0) {
        free(filtered);
        return EXIT_FAILURE;
    }

    free(filtered);
    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder CWE-120 warning: the unbounded `memcpy` was replaced with an explicit bounded copy loop guarded by a capacity check (`word_len > capacity - 1U - out`), which accounts for the null terminator and cannot underflow since `out < capacity` is guaranteed at that point. The separator write is now checked against `capacity` rather than `len`, and a final defensive bounds check protects the null-terminator write. All buffer writes are now provably within the allocated `len + 1` bytes.
