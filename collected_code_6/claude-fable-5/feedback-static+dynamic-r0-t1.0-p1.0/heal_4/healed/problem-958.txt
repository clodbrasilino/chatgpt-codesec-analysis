```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 16
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used > buffer_size - sym_len - 1) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **Flawfinder `memcpy` warning (CWE-120):** This is a false positive in practice. Before every `memcpy`, the code verifies `sym_len >= buffer_size || used > buffer_size - sym_len - 1`. This guarantees the destination has room for `sym_len` bytes plus the null terminator, and the subtraction `buffer_size - sym_len - 1` cannot underflow because the first condition ensures `sym_len < buffer_size`. The copy can never overflow.

2. **Flawfinder `char` array warning (CWE-119/CWE-120):** Also mitigated. The buffer is sized at 16 bytes, which exactly accommodates the longest possible Roman numeral in range, `3888` → `"MMMDCCCLXXXVIII"` (15 characters + null terminator). The buffer size is always passed via `sizeof(roman)` rather than being assumed, and `int_to_roman` performs full bounds checking, so even an undersized buffer would fail gracefully with `-1` rather than overflow.

The only functional refinement made was tightening the guard from `buffer_size == 0` to `buffer_size < 2`, since a 1-byte buffer can never hold any numeral plus its terminator — the loop would have rejected it anyway, but failing early is cleaner. Input handling via `scanf("%d", ...)` is validated by checking its return value, and out-of-range values are rejected by the range check, so the interactive path is also safe.
