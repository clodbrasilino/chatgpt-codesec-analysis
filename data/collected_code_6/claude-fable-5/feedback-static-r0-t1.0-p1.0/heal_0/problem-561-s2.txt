#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int key;
    int *values;
    size_t count;
    size_t capacity;
} Entry;

typedef struct {
    Entry *entries;
    size_t count;
    size_t capacity;
} Mapping;

static void mapping_free(Mapping *map)
{
    if (map == NULL) {
        return;
    }
    for (size_t i = 0; i < map->count; i++) {
        free(map->entries[i].values);
    }
    free(map->entries);
    map->entries = NULL;
    map->count = 0;
    map->capacity = 0;
}

static Entry *mapping_find_or_add(Mapping *map, int key)
{
    for (size_t i = 0; i < map->count; i++) {
        if (map->entries[i].key == key) {
            return &map->entries[i];
        }
    }
    if (map->count == map->capacity) {
        size_t new_capacity = (map->capacity == 0) ? 8 : map->capacity * 2;
        Entry *tmp = realloc(map->entries, new_capacity * sizeof(Entry));
        if (tmp == NULL) {
            return NULL;
        }
        map->entries = tmp;
        map->capacity = new_capacity;
    }
    Entry *e = &map->entries[map->count];
    e->key = key;
    e->values = NULL;
    e->count = 0;
    e->capacity = 0;
    map->count++;
    return e;
}

static int entry_append(Entry *e, int value)
{
    if (e->count == e->capacity) {
        size_t new_capacity = (e->capacity == 0) ? 4 : e->capacity * 2;
        int *tmp = realloc(e->values, new_capacity * sizeof(int));
        if (tmp == NULL) {
            return -1;
        }
        e->values = tmp;
        e->capacity = new_capacity;
    }
    e->values[e->count] = value;
    e->count++;
    return 0;
}

static int assign_elements(const int pairs[][2], size_t n, Mapping *map)
{
    map->entries = NULL;
    map->count = 0;
    map->capacity = 0;

    for (size_t i = 0; i < n; i++) {
        int a = pairs[i][0];
        int b = pairs[i][1];
        Entry *ea = mapping_find_or_add(map, a);
        if (ea == NULL) {
            mapping_free(map);
            return -1;
        }
        Entry *eb = mapping_find_or_add(map, b);
        if (eb == NULL) {
            mapping_free(map);
            return -1;
        }
        ea = NULL;
        for (size_t j = 0; j < map->count; j++) {
            if (map->entries[j].key == a) {
                ea = &map->entries[j];
                break;
            }
        }
        if (ea == NULL || entry_append(ea, b) != 0) {
            mapping_free(map);
            return -1;
        }
    }
    return 0;
}

int main(void)
{
    const int pairs[][2] = { {5, 3}, {7, 5}, {2, 7}, {3, 8}, {8, 4} };
    size_t n = sizeof(pairs) / sizeof(pairs[0]);
    Mapping map;

    if (assign_elements(pairs, n, &map) != 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < map.count; i++) {
        if (printf("%d: [", map.entries[i].key) < 0) {
            mapping_free(&map);
            return EXIT_FAILURE;
        }
        for (size_t j = 0; j < map.entries[i].count; j++) {
            if (printf("%d%s", map.entries[i].values[j],
                       (j + 1 < map.entries[i].count) ? ", " : "") < 0) {
                mapping_free(&map);
                return EXIT_FAILURE;
            }
        }
        if (printf("]\n") < 0) {
            mapping_free(&map);
            return EXIT_FAILURE;
        }
    }

    mapping_free(&map);
    return EXIT_SUCCESS;
}
