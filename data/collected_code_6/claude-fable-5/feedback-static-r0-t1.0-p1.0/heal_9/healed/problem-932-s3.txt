#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

static char *duplicate_word(const char *word, size_t len);
char **remove_duplicates(const char **words, size_t count, size_t *out_count);
void free_word_list(char **words, size_t count);

static char *duplicate_word(const char *word, size_t len)
{
    char *copy;
    size_t alloc_size;

    if (word == NULL || len >= MAX_WORD_LEN) {
        return NULL;
    }

    if (len > SIZE_MAX - 1) {
        return NULL;
    }

    alloc_size = len + 1;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        if (len > alloc_size - 1) {
            free(copy);
            return NULL;
        }
        memcpy(copy, word, len);
    }
    copy[len] = '\0';

    return copy;
}

char **remove_duplicates(const char **words, size_t count, size_t *out_count)
{
    char **result = NULL;
    size_t unique = 0;
    size_t i;
    size_t j;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            result[unique] = duplicate_word(words[i], len);
            if (result[unique] == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Unique words:\n");
    for (i = 0; i < result_count; i++) {
        printf("%s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe:

The Flawfinder warning on `memcpy` (CWE-120) is a false positive in this case, and the rewritten code retains all the defensive measures that make it so:

1. **Destination size is guaranteed**: `alloc_size = len + 1` and the copy writes exactly `len` bytes, so the destination always holds the source data plus the null terminator.

2. **Length validation before allocation**: `len >= MAX_WORD_LEN` rejects oversized inputs, and `len > SIZE_MAX - 1` prevents integer overflow when computing `len + 1`.

3. **Redundant guard before the copy**: `len > alloc_size - 1` re-verifies the invariant immediately before `memcpy`, defending against future refactoring mistakes.

4. **Source length is bounded**: The caller uses `strnlen(words[i], MAX_WORD_LEN)` rather than `strlen`, so unterminated input strings cannot cause an over-read, and `strncmp` bounds all comparisons.

5. **Allocation overflow check**: `count > SIZE_MAX / sizeof(char *)` prevents multiplication overflow in `calloc`.

6. **Complete error-path cleanup**: On allocation failure, all previously duplicated words are freed before returning, so there are no leaks.
