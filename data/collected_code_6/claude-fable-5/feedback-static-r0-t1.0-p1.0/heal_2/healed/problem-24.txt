#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * 8U)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}

**Analysis summary:**

The original code was already well-hardened against the Flawfinder finding (CWE-119/CWE-120). The statically-sized `char input[INPUT_SIZE]` buffer is safely handled because:

1. **Bounded read:** `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size, preventing overflow.
2. **Guaranteed termination:** `input[sizeof(input) - 1U] = '\0'` forces NUL-termination defensively.
3. **Bounded length calculation:** `strnlen()` is used instead of `strlen()`, so it never scans past the buffer.
4. **Validated conversion:** `binary_to_decimal()` checks for NULL pointers, empty input, non-binary characters, and multiplication/shift overflow via `ULONG_MAX >> 1` before shifting, plus the `len > sizeof(unsigned long) * 8U` bit-width guard.

The rewrite adds two minor robustness improvements: a `fflush(stdout)` so the prompt is guaranteed to appear before blocking on input, and explicit detection of over-long input (when the buffer fills without a newline), draining the remainder of the line from `stdin` so truncated input is rejected rather than silently processed.
