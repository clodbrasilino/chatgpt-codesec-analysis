#include <stdio.h>
#include <stdlib.h>

size_t count_negative_numbers(const int *list, size_t size) {
    if (list == NULL) {
        return 0;
    }

    size_t count = 0;
    for (size_t i = 0; i < size; ++i) {
        if (list[i] < 0) {
            count++;
        }
    }

    return count;
}

int main(void) {
    int numbers[] = {1, -2, 3, -4, 5, -6, 0, -8};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);

    size_t negative_count = count_negative_numbers(numbers, size);

    printf("Number of negative numbers: %zu\n", negative_count);

    return EXIT_SUCCESS;
}
