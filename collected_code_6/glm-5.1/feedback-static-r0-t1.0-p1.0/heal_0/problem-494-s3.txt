```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

typedef struct {
    int8_t *bits;
    size_t size;
} BinaryTuple;

int binary_tuple_to_int(const BinaryTuple *tuple) {
    if (tuple == NULL || tuple->bits == NULL || tuple->size == 0) {
        return 0;
    }

    int result = 0;
    for (size_t i = 0; i < tuple->size; i++) {
        if (tuple->bits[i] < 0 || tuple->bits[i] > 1) {
            return 0;
        }
        result = (result << 1) | tuple->bits[i];
    }

    return result;
}

int main(void) {
    int8_t bits[] = {1, 0, 1, 1};
    BinaryTuple tuple = {bits, sizeof(bits) / sizeof(bits[0])};

    int value = binary_tuple_to_int(&tuple);
    printf("%d\n", value);

    return 0;
}
```
