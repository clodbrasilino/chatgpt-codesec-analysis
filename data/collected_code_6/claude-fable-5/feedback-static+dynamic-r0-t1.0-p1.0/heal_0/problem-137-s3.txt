```c
#include <stdio.h>
#include <stddef.h>

double zero_ratio(const int *arr, size_t len)
{
    size_t i;
    size_t zero_count = 0;

    if (arr == NULL || len == 0) {
        return 0.0;
    }

    for (i = 0; i < len; i++) {
        if (arr[i] == 0) {
            zero_count++;
        }
    }

    return (double)zero_count / (double)len;
}

int main(void)
{
    int data[] = {0, 3, 0, 7, 0, 9, 1, 0};
    size_t len = sizeof(data) / sizeof(data[0]);
    double ratio;

    ratio = zero_ratio(data, len);

    if (printf("Ratio of zeroes: %f\n", ratio) < 0) {
        return 1;
    }

    return 0;
}
```
