#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NUM_COLOURS 3
#define MAX_COUNT 64

static const char *const colours[NUM_COLOURS] = { "Red", "Green", "Blue" };

static int validate_indices(const int *indices, size_t count)
{
    size_t i;

    if (indices == NULL || count == 0 || count > MAX_COUNT) {
        return 0;
    }

    for (i = 0; i < count; i++) {
        if (indices[i] < 0 || (size_t)indices[i] >= NUM_COLOURS) {
            return 0;
        }
    }

    return 1;
}

static void print_combination(const int *indices, size_t count)
{
    size_t i;

    if (!validate_indices(indices, count)) {
        return;
    }

    for (i = 0; i < count; i++) {
        fputs(colours[indices[i]], stdout);
        if (i + 1 < count) {
            fputc(' ', stdout);
        }
    }
    fputc('\n', stdout);
}

static void generate_combinations(int *indices, size_t depth, size_t count, int start)
{
    int i;

    if (indices == NULL || count == 0 || count > MAX_COUNT || depth > count) {
        return;
    }

    if (start < 0 || start > NUM_COLOURS) {
        return;
    }

    if (depth == count) {
        print_combination(indices, count);
        return;
    }

    for (i = start; i < NUM_COLOURS; i++) {
        if (depth < count) {
            indices[depth] = i;
            generate_combinations(indices, depth + 1, count, i);
        }
    }
}

static int choose_colours(size_t count)
{
    int *indices;

    if (count == 0 || count > MAX_COUNT) {
        return -1;
    }

    if (count > SIZE_MAX / sizeof(*indices)) {
        return -1;
    }

    indices = calloc(count, sizeof(*indices));
    if (indices == NULL) {
        return -1;
    }

    generate_combinations(indices, 0, count, 0);

    free(indices);
    indices = NULL;
    return 0;
}

int main(void)
{
    size_t count = 2;

    printf("Combinations with repetition choosing %zu colours from %d:\n",
           count, NUM_COLOURS);

    if (choose_colours(count) != 0) {
        fprintf(stderr, "Failed to generate combinations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**Fixes applied for the buffer weakness:**

1. **Bounds checking on every array read** — `validate_indices()` verifies each index is within `[0, NUM_COLOURS)` before `colours[indices[i]]` is ever dereferenced, preventing out-of-bounds reads on the static `colours` array.

2. **Bounds checking on every array write** — in `generate_combinations()`, an explicit `depth < count` guard was added immediately before `indices[depth] = i`, guaranteeing no write past the end of the heap buffer even if the earlier `depth > count` guard were bypassed by future refactoring.

3. **Size-capped allocation** — `count` is restricted to `MAX_COUNT` and checked against `SIZE_MAX / sizeof(*indices)` before `calloc()`, eliminating integer-overflow-driven undersized allocations.

4. **Zero-initialized buffer** — `calloc()` ensures the `indices` buffer never contains uninitialized values that could act as rogue array indices.

5. **NULL and range validation at every entry point** — all functions defensively validate their pointer and size parameters before touching any buffer, and the pointer is set to `NULL` after `free()` to prevent use-after-free/double-free.
