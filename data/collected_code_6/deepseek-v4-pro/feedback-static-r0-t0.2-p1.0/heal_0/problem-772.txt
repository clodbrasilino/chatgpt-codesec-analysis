```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_words_with_length(char *str, int k) {
    if (str == NULL || k <= 0) {
        return str;
    }

    char *result = (char *)malloc(strlen(str) + 1);
    if (result == NULL) {
        return NULL;
    }

    int i = 0, j = 0;
    int len = strlen(str);

    while (i < len) {
        while (i < len && isspace((unsigned char)str[i])) {
            result[j++] = str[i++];
        }

        int start = i;
        while (i < len && !isspace((unsigned char)str[i])) {
            i++;
        }

        int word_len = i - start;
        if (word_len != k) {
            for (int m = start; m < i; m++) {
                result[j++] = str[m];
            }
        }
    }

    result[j] = '\0';
    return result;
}

int main(void) {
    char input[] = "This is a test string with words of different lengths";
    int k = 2;

    char *result = remove_words_with_length(input, k);
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    printf("Original: %s\n", input);
    printf("After removing words of length %d: %s\n", k, result);

    free(result);
    return 0;
}
```
