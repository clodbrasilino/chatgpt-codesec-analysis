```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

The identified weaknesses were already properly mitigated in the original code, so the rewrite preserves those defenses:

- **CWE-119/CWE-120 (statically-sized buffers)**: Both `normalized` and `input` are bounds-checked. `strnlen` with an explicit limit prevents reads past the buffer, `fgets` is given `sizeof(input)` so it can never overflow, the copy loop in `has_28_days` is capped at `sizeof(normalized) - 1`, and explicit NUL termination is guaranteed in both functions.
- **CWE-120/CWE-20 (`getchar` in a loop)**: `flush_stdin` never stores the characters it reads, so no buffer can overflow; it terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed input streams.
- **Input validation (CWE-20)**: Non-alphabetic characters, empty input, `NULL` pointers, and over-length input are all rejected before comparison, and leftover stdin data is drained when the input line exceeds the buffer.
