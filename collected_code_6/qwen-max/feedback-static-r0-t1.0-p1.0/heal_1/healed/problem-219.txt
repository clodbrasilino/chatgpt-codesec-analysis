#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int *data;
    size_t size;
} Tuple;

void tuple_init(Tuple *t, const int *arr, size_t size) {
    t->data = (int *)malloc(size * sizeof(int));
    if (t->data == NULL) exit(EXIT_FAILURE);
    for (size_t i = 0; i < size; ++i) {
        t->data[i] = arr[i];
    }
    t->size = size;
}

void tuple_destroy(Tuple *t) {
    free(t->data);
    t->data = NULL;
    t->size = 0;
}

void extract_k_elements(Tuple *t, int k, int **max_elements, int **min_elements) {
    if (k > t->size || k <= 0) return;
    *max_elements = (int *)malloc(k * sizeof(int));
    *min_elements = (int *)malloc(k * sizeof(int));
    if (*max_elements == NULL || *min_elements == NULL) exit(EXIT_FAILURE);
    
    int *temp = (int *)malloc(t->size * sizeof(int));
    if (temp == NULL) exit(EXIT_FAILURE);
    for (size_t i = 0; i < t->size; ++i) {
        temp[i] = t->data[i];
    }
    
    for (int i = 0; i < k; ++i) {
        int max_idx = i, min_idx = i;
        for (size_t j = i + 1; j < t->size; ++j) {
            if (temp[j] > temp[max_idx]) max_idx = j;
            if (temp[j] < temp[min_idx]) min_idx = j;
        }
        (*max_elements)[i] = temp[max_idx];
        (*min_elements)[i] = temp[min_idx];
        temp[max_idx] = temp[min_idx] = INT_MIN;
    }
    
    free(temp);
}

int main() {
    int arr[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    Tuple t;
    tuple_init(&t, arr, size);
    
    int *max_elements, *min_elements;
    extract_k_elements(&t, 3, &max_elements, &min_elements);
    
    printf("Max elements: ");
    for (int i = 0; i < 3; ++i) {
        printf("%d ", max_elements[i]);
    }
    printf("\nMin elements: ");
    for (int i = 0; i < 3; ++i) {
        printf("%d ", min_elements[i]);
    }
    printf("\n");
    
    free(max_elements);
    free(min_elements);
    tuple_destroy(&t);
    return 0;
}
