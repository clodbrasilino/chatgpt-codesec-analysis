```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

int left_rotate(char *str, size_t buf_size, size_t d)
{
    size_t len;
    size_t temp_size;
    size_t first_part;
    size_t second_part;
    char *temp;

    if (str == NULL || buf_size == 0 || buf_size > MAX_STR_LEN) {
        return -1;
    }

    len = strnlen(str, buf_size);
    if (len == buf_size) {
        return -1;
    }

    if (len == 0) {
        return 0;
    }

    d = d % len;
    if (d == 0) {
        return 0;
    }

    if (len + 1 > buf_size) {
        return -1;
    }

    temp_size = len + 1;
    temp = malloc(temp_size);
    if (temp == NULL) {
        return -1;
    }

    first_part = len - d;
    second_part = d;

    if (first_part > temp_size - 1) {
        free(temp);
        return -1;
    }

    if (second_part > (temp_size - 1) - first_part) {
        free(temp);
        return -1;
    }

    if (d > len || first_part + second_part != len) {
        free(temp);
        return -1;
    }

    memcpy(temp, str + d, first_part);
    memcpy(temp + first_part, str, second_part);
    temp[len] = '\0';

    if (temp_size > buf_size) {
        free(temp);
        return -1;
    }

    memcpy(str, temp, temp_size);
    free(temp);

    return 0;
}

int main(void)
{
    char str[] = "HelloWorld";
    size_t d = 3;

    printf("Original string: %s\n", str);

    if (left_rotate(str, sizeof(str), d) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    printf("Rotated string:  %s\n", str);

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the flagged weaknesses:**

The three Flawfinder warnings on the `memcpy` calls are essentially false positives in the original code, because the surrounding logic already guarantees safety. Here's why each copy is safe:

1. **`memcpy(temp, str + d, first_part)`** — `temp` has `temp_size = len + 1` bytes. The guard `first_part > temp_size - 1` ensures `first_part <= len`, so the destination cannot overflow. The source read of `str + d` for `first_part = len - d` bytes stays within the measured string length.

2. **`memcpy(temp + first_part, str, second_part)`** — The guard `second_part > (temp_size - 1) - first_part` ensures `first_part + second_part <= len`, so writing at offset `first_part` cannot exceed the allocation.

3. **`memcpy(str, temp, temp_size)`** — The guard `temp_size > buf_size` immediately before it ensures the destination buffer can hold the copy.

Other defensive measures already present and preserved:

- **`strnlen` instead of `strlen`** bounds the length scan to `buf_size`, and rejecting `len == buf_size` catches unterminated buffers.
- **NULL/zero-size/oversize checks** at entry prevent invalid pointer arithmetic.
- **`d % len`** prevents out-of-range rotation offsets.
- **`malloc` return checked** before use; `free` on every error path prevents leaks.

In the rewrite I added one extra invariant check (`d > len || first_part + second_part != len`) as a belt-and-braces assertion that the two parts exactly reconstruct the string, which makes the safety of both `memcpy` calls into `temp` locally provable to static analyzers without relying on arithmetic reasoning across multiple statements. The code remains functionally identical: rotating `"HelloWorld"` left by 3 yields `"loWorldHel"`.
