#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void find_adverbs(const char *sentence) {
    if (sentence == NULL) {
        return;
    }

    regex_t regex;
    int reti;
    regmatch_t pmatch[1];
    const char *pattern = "\\b[a-zA-Z]+ly\\b";
    
    reti = regcomp(&regex, pattern, REG_EXTENDED | REG_ICASE);
    if (reti != 0) {
        fprintf(stderr, "Could not compile regex\n");
        return;
    }

    const char *cursor = sentence;
    while (regexec(&regex, cursor, 1, pmatch, 0) == 0) {
        int start = pmatch[0].rm_so + (cursor - sentence);
        int end = pmatch[0].rm_eo + (cursor - sentence);
        
        printf("Adverb: '%.*s', Position: [%d, %d]\n", 
               end - start, cursor + pmatch[0].rm_so, start, end - 1);
        
        cursor += pmatch[0].rm_eo;
    }

    regfree(&regex);
}

int main(void) {
    const char *sentence = "He quickly ran quietly and swiftly away.";
    find_adverbs(sentence);
    return 0;
}
