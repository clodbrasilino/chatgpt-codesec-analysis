#include <stdio.h>
#include <ctype.h>
#include <stddef.h>

int remove_leading_zeroes(const char *ip, char *out, size_t out_size);

int remove_leading_zeroes(const char *ip, char *out, size_t out_size) {
    if (ip == NULL || out == NULL || out_size == 0) {
        return -1;
    }

    unsigned int octets[4];
    int count = 0;
    const char *p = ip;

    while (*p != '\0') {
        if (count >= 4) {
            return -1;
        }

        if (!isdigit((unsigned char)*p)) {
            return -1;
        }

        unsigned int value = 0;
        int digits = 0;

        while (isdigit((unsigned char)*p)) {
            if (digits >= 3) {
                return -1;
            }

            value = value * 10u + (unsigned int)(*p - '0');

            if (value > 255u) {
                return -1;
            }

            digits++;
            p++;
        }

        octets[count] = value;
        count++;

        if (*p == '.') {
            p++;
            if (*p == '\0') {
                return -1;
            }
        } else if (*p != '\0') {
            return -1;
        }
    }

    if (count != 4) {
        return -1;
    }

    int needed = snprintf(out, out_size, "%u.%u.%u.%u",
                          octets[0], octets[1], octets[2], octets[3]);

    if (needed < 0 || (size_t)needed >= out_size) {
        return -1;
    }

    return 0;
}

int main(void) {
    const char *test_ips[] = {
        "192.168.001.004",
        "010.000.000.001",
        "255.255.255.255",
        "0.0.0.0",
        "1.2.3.4",
        "001.002.003.256",
        "1.2.3",
        "1.2.3.4.",
        NULL
    };

    char result[16];

    for (int i = 0; test_ips[i] != NULL; i++) {
        if (remove_leading_zeroes(test_ips[i], result, sizeof(result)) == 0) {
            printf("%s -> %s\n", test_ips[i], result);
        } else {
            printf("%s -> invalid\n", test_ips[i]);
        }
    }

    return 0;
}
