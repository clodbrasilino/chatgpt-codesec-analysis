#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_max_sum(int **lists, const int *lengths, int num_lists, int *result) {
    if (lists == NULL || lengths == NULL || result == NULL || num_lists <= 0) {
        return -1;
    }

    int max_sum = INT_MIN;
    int found = 0;

    for (int i = 0; i < num_lists; i++) {
        if (lengths[i] < 0) {
            return -1;
        }
        
        if (lengths[i] == 0) {
            continue;
        }

        if (lists[i] == NULL) {
            return -1;
        }

        int current_sum = 0;
        for (int j = 0; j < lengths[i]; j++) {
            current_sum += lists[i][j];
        }

        if (!found || current_sum > max_sum) {
            max_sum = current_sum;
            found = 1;
        }
    }

    *result = found ? max_sum : 0;
    return 0;
}

int main(void) {
    int num_lists = 3;
    
    int *lengths = malloc(num_lists * sizeof(int));
    if (lengths == NULL) {
        return EXIT_FAILURE;
    }

    int **lists = malloc(num_lists * sizeof(int *));
    if (lists == NULL) {
        free(lengths);
        return EXIT_FAILURE;
    }

    lengths[0] = 3;
    lengths[1] = 4;
    lengths[2] = 2;

    lists[0] = malloc(lengths[0] * sizeof(int));
    lists[1] = malloc(lengths[1] * sizeof(int));
    lists[2] = malloc(lengths[2] * sizeof(int));

    if (lists[0] == NULL || lists[1] == NULL || lists[2] == NULL) {
        free(lists[0]);
        free(lists[1]);
        free(lists[2]);
        free(lists);
        free(lengths);
        return EXIT_FAILURE;
    }

    lists[0][0] = 1;
    lists[0][1] = 2;
    lists[0][2] = 3;

    lists[1][0] = -1;
    lists[1][1] = -2;
    lists[1][2] = -3;
    lists[1][3] = -4;

    lists[2][0] = 10;
    lists[2][1] = 5;

    int max_sum;
    if (find_max_sum(lists, lengths, num_lists, &max_sum) == 0) {
        printf("%d\n", max_sum);
    }

    for (int i = 0; i < num_lists; i++) {
        free(lists[i]);
    }
    free(lists);
    free(lengths);

    return EXIT_SUCCESS;
}
