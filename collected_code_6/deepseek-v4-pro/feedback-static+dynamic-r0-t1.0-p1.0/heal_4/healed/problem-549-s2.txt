#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <limits.h>

long long sum_fifth_power_odd(int n) {
    unsigned long long sum = 0;
    unsigned long long max_term = (n > 0) ? (2ULL * (n - 1) + 1) : 0;
    
    if (n > 0) {
        unsigned long long max_odd = 2ULL * (n - 1) + 1;
        unsigned long long max_odd2 = max_odd * max_odd;
        unsigned long long max_odd4 = max_odd2 * max_odd2;
        unsigned long long max_term_calc = max_odd4 * max_odd;
        
        if (max_term_calc > (unsigned long long)LLONG_MAX) {
            return -1;
        }
    }
    
    for (int i = 0; i < n; i++) {
        unsigned long long odd = 2ULL * i + 1;
        unsigned long long odd2 = odd * odd;
        unsigned long long odd4 = odd2 * odd2;
        unsigned long long term = odd4 * odd;
        
        if (sum > (unsigned long long)LLONG_MAX - term) {
            return -1;
        }
        sum += term;
    }
    
    return (long long)sum;
}

int main(void) {
    int n;
    printf("Enter n: ");
    if (scanf("%d", &n) != 1 || n < 0) {
        printf("Invalid input\n");
        return 1;
    }
    if (n > 100000) {
        printf("Input too large\n");
        return 1;
    }
    long long result = sum_fifth_power_odd(n);
    if (result == -1 && n > 1000) {
        printf("Result too large to store in long long\n");
        return 1;
    }
    printf("Sum = %lld\n", result);
    return 0;
}
