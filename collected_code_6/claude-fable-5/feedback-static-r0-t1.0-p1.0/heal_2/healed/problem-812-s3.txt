```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char *abbreviate_road(const char *input)
{
    static const char pattern[] = "road";
    static const char replacement[] = "rd.";
    const size_t pattern_len = sizeof(pattern) - 1;
    const size_t replacement_len = sizeof(replacement) - 1;
    size_t input_len;
    size_t count;
    size_t result_len;
    size_t remaining;
    const char *p;
    char *result;
    char *dest;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    count = 0;
    p = input;
    while ((p = strstr(p, pattern)) != NULL) {
        count++;
        p += pattern_len;
    }

    result_len = input_len - count * (pattern_len - replacement_len);

    result = malloc(result_len + 1);
    if (result == NULL) {
        return NULL;
    }

    dest = result;
    remaining = result_len;
    p = input;
    while (*p != '\0') {
        if (strncmp(p, pattern, pattern_len) == 0) {
            if (remaining < replacement_len) {
                free(result);
                return NULL;
            }
            memcpy(dest, replacement, replacement_len);
            dest += replacement_len;
            remaining -= replacement_len;
            p += pattern_len;
        } else {
            if (remaining < 1) {
                free(result);
                return NULL;
            }
            *dest = *p;
            dest++;
            remaining--;
            p++;
        }
    }
    *dest = '\0';

    return result;
}

int main(void)
{
    const char *text = "Take the main road until the road forks near Elm road.";
    char *abbreviated;

    abbreviated = abbreviate_road(text);
    if (abbreviated == NULL) {
        fprintf(stderr, "Failed to abbreviate string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original:    %s\n", text) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    if (printf("Abbreviated: %s\n", abbreviated) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    free(abbreviated);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Out-of-bounds read fixed**: The original code checked `input[MAX_INPUT_LEN]` after `strnlen` returned `MAX_INPUT_LEN`, which reads one byte past the guaranteed-valid region. Now inputs of length `>= MAX_INPUT_LEN` are simply rejected without any further read.

2. **Dead code removed (`knownConditionTrueFalse`)**: The `pattern_len == 0` check and the `replacement_len > pattern_len` growth/overflow branch were provably unreachable since both lengths are compile-time constants (4 and 3). Removing them eliminates the static-analysis noise, and the result length computation is now expressed as a guaranteed-safe subtraction (`pattern_len > replacement_len`, so no `size_t` underflow is possible).

3. **`SIZE_MAX` include issue resolved**: The original referenced `SIZE_MAX` without including `<stdint.h>`. Since the overflow branch is gone, the dependency was removed, but `<stdint.h>` is included defensively for portability.

4. **Flawfinder `memcpy` warning (CWE-120)**: The `remaining < replacement_len` bounds check before `memcpy` is retained, so the destination capacity is always verified prior to the copy. Combined with the exact pre-computed `result_len`, the buffer cannot overflow.

5. **Constants made `const`**: `pattern_len` and `replacement_len` are now `const` locals, making the compile-time nature of these values explicit to both readers and analyzers.
