static bool is_vowel(char c)
{
    switch (c) {
    case 'a': case 'e': case 'i': case 'o': case 'u':
    case 'A': case 'E': case 'I': case 'O': case 'U':
        return true;
    default:
        return false;
    }

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t i = 0U;

    while ((i < max_len) && (s[i] != '\0')) {
        i++;
    }

    return i;
}

static int reverse_vowels(char *s, size_t max_len)
{

    size_t right;

}
