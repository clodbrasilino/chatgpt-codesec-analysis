```C
#include<stdio.h>
#include<math.h>

int isPerfectSquare(int num) {
    int sqrRoot = sqrt(num);
    return (sqrRoot * sqrRoot) == num;
}

int main() {
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);

    if (isPerfectSquare(num)) {
        printf("The number is a perfect square.\n");
    }
    else {
        printf("The number is not a perfect square.\n");
    }

    return 0;
}
```
