```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>
#include <stdint.h>

bool find_max_occurring_char(const char *str, char *result)
{
    int freq[256] = {0};
    int max_count = 0;
    size_t i;

    if (str == NULL || result == NULL || str[0] == '\0')
    {
        return false;
    }

    for (i = 0; str[i] != '\0'; i++)
    {
        freq[(unsigned char)str[i]]++;
    }

    for (i = 0; i < 256; i++)
    {
        if (freq[i] > max_count)
        {
            max_count = freq[i];
            *result = (char)i;
        }
    }

    return true;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *buffer;
    int c;

    if (stream == NULL)
    {
        return NULL;
    }

    buffer = malloc(capacity);

    if (buffer == NULL)
    {
        return NULL;
    }

    while ((c = fgetc(stream)) != EOF && c != '\n')
    {
        if (length + 1 >= capacity)
        {
            char *new_buffer;
            size_t new_capacity;

            if (capacity > SIZE_MAX / 2)
            {
                free(buffer);
                return NULL;
            }

            new_capacity = capacity * 2;

            if (new_capacity <= capacity)
            {
                free(buffer);
                return NULL;
            }

            new_buffer = realloc(buffer, new_capacity);

            if (new_buffer == NULL)
            {
                free(buffer);
                return NULL;
            }

            buffer = new_buffer;
            capacity = new_capacity;
        }

        buffer[length++] = (char)c;
    }

    if (ferror(stream))
    {
        free(buffer);
        return NULL;
    }

    if (c == EOF && length == 0)
    {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';
    return buffer;
}

int main(void)
{
    char *input;
    char result = '\0';

    printf("Enter a string: ");
    fflush(stdout);

    input = read_line(stdin);

    if (input == NULL)
    {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    if (!find_max_occurring_char(input, &result))
    {
        fprintf(stderr, "Invalid or empty input string.\n");
        free(input);
        return EXIT_FAILURE;
    }

    printf("The maximum occurring character is '%c'.\n", result);

    free(input);
    return EXIT_SUCCESS;
}
```

Key fixes applied to address the flagged `fgetc` weakness (CWE-120, CWE-20):

1. **Bounds verification retained and hardened** — the `length + 1 >= capacity` check guarantees one byte always remains for the null terminator before any write into the buffer.
2. **Overflow protection on growth** — capacity doubling is guarded with both `capacity > SIZE_MAX / 2` and a sanity check that `new_capacity > capacity`, preventing integer wraparound.
3. **Stream validation** — `read_line` now rejects a `NULL` stream pointer before allocating.
4. **Read-error detection** — `ferror(stream)` is checked after the loop so an I/O error is not silently treated as a legitimate EOF.
5. **Prompt flushing** — `fflush(stdout)` after the prompt ensures the user sees it before blocking on input.
