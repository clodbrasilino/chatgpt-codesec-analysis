#include <stdio.h>
#include <stdlib.h>

unsigned long long jacobsthal_lucas(int n) {
    if (n < 0) {
        return 0;
    }
    if (n == 0) {
        return 2;
    }
    if (n == 1) {
        return 1;
    }

    unsigned long long prev2 = 2;
    unsigned long long prev1 = 1;
    unsigned long long current = 0;

    for (int i = 2; i <= n; i++) {
        current = prev1 + 2 * prev2;
        prev2 = prev1;
        prev1 = current;
    }

    return current;
}

int main(void) {
    int n;
    
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    if (n < 0 || n > 60) {
        fprintf(stderr, "Input out of supported range (0-60).\n");
        return EXIT_FAILURE;
    }

    unsigned long long result = jacobsthal_lucas(n);
    printf("%llu\n", result);

    return EXIT_SUCCESS;
}
