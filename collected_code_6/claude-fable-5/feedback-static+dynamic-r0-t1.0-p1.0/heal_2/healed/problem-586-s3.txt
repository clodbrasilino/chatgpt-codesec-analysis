```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely safe — the flawfinder warnings on `memcpy` are heuristic (CWE-120) and flag every use of the function regardless of context. In the original:

- `temp` is allocated with exactly `head_bytes`, and exactly `head_bytes` are copied into it — no overflow.
- `arr + (n - k)` leaves exactly `k` elements (`head_bytes`) of space at the tail of the array, matching the copy size — no overflow.
- The `n > SIZE_MAX / sizeof(int)` guard already prevents integer overflow when computing byte sizes.
- `memmove` is correctly used for the overlapping shift.

The rewritten version adds explicit defensive checks (`head_bytes` and `tail_bytes` each bounded by `total_bytes`, and their sum verified against it) to make the safety invariants explicit and satisfy static analyzers, without changing the algorithm's behavior. Rotation of `{12, 10, 5, 6, 52, 36}` by `k = 2` still produces `5 6 52 36 12 10`.
