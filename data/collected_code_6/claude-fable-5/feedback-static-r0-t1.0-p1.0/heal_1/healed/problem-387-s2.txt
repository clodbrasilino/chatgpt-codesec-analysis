```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 128

int is_valid_hex(const char *str)
{
    size_t len;
    size_t i;
    size_t start = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return 0;
    }

    if (len > 2 && str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        start = 2;
    }

    if (start >= len) {
        return 0;
    }

    for (i = start; i < len; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

int hex_is_even(const char *hex_str)
{
    size_t len;
    char last;
    int digit_value;

    if (hex_str == NULL) {
        return 0;
    }

    len = strnlen(hex_str, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return 0;
    }

    last = hex_str[len - 1];

    if (last >= '0' && last <= '9') {
        digit_value = last - '0';
    } else if (last >= 'a' && last <= 'f') {
        digit_value = last - 'a' + 10;
    } else if (last >= 'A' && last <= 'F') {
        digit_value = last - 'A' + 10;
    } else {
        return 0;
    }

    return (digit_value % 2) == 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a hexadecimal number: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    } else if (len == sizeof(buffer) - 1) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (!is_valid_hex(buffer)) {
        fprintf(stderr, "Invalid hexadecimal number\n");
        return EXIT_FAILURE;
    }

    if (hex_is_even(buffer)) {
        printf("%s is even\n", buffer);
    } else {
        printf("%s is odd\n", buffer);
    }

    return EXIT_SUCCESS;
}
```

**Summary of the changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** All calls to `strlen` were replaced with `strnlen` bounded to `MAX_INPUT_LEN`, guaranteeing no over-read even if a non-terminated string were ever passed. Additionally, `buffer[sizeof(buffer) - 1] = '\0'` explicitly enforces null-termination after `fgets`.

2. **CWE-119/CWE-120 (statically-sized buffer):** The buffer size is centralized in the `MAX_INPUT_LEN` macro, `fgets` is bounded by `sizeof(buffer)`, and input truncation is now detected — if the line fills the buffer without a newline, the remaining input is drained and the program rejects it rather than silently processing a truncated value.

3. **Latent bug hardening:** `hex_is_even` previously called `strlen` and indexed `hex_str[len - 1]` without checking for `NULL` or an empty string (undefined behavior via `hex_str[-1]` if `len == 0`). It now validates both, and the final `else` branch validates that the character is actually `A`–`F` instead of assuming it, making the function safe even if called with unvalidated input.
