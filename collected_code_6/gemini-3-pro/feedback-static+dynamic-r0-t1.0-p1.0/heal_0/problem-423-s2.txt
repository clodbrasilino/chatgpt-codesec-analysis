#include <stdio.h>
#include <stdlib.h>

#define MAX(a, b) ((a) > (b) ? (a) : (b))

int get_max_gold(int** mine, int rows, int cols) {
    if (mine == NULL || rows <= 0 || cols <= 0) {
        return 0;
    }

    int** dp = (int**)malloc(rows * sizeof(int*));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < rows; i++) {
        dp[i] = (int*)malloc(cols * sizeof(int));
        if (dp[i] == NULL) {
            for (int k = 0; k < i; k++) {
                free(dp[k]);
            }
            free(dp);
            return 0;
        }
    }

    for (int i = 0; i < rows; i++) {
        dp[i][0] = mine[i][0];
    }

    for (int j = 1; j < cols; j++) {
        for (int i = 0; i < rows; i++) {
            int right = (j - 1 >= 0) ? dp[i][j - 1] : 0;
            int right_up = (i - 1 >= 0 && j - 1 >= 0) ? dp[i - 1][j - 1] : 0;
            int right_down = (i + 1 < rows && j - 1 >= 0) ? dp[i + 1][j - 1] : 0;

            dp[i][j] = mine[i][j] + MAX(right, MAX(right_up, right_down));
        }
    }

    int max_gold = dp[0][cols - 1];
    for (int i = 1; i < rows; i++) {
        if (dp[i][cols - 1] > max_gold) {
            max_gold = dp[i][cols - 1];
        }
    }

    for (int i = 0; i < rows; i++) {
        free(dp[i]);
    }
    free(dp);

    return max_gold;
}

int main(void) {
    int rows = 4;
    int cols = 4;

    int** mine = (int**)malloc(rows * sizeof(int*));
    if (mine == NULL) {
        return 1;
    }

    for (int i = 0; i < rows; i++) {
        mine[i] = (int*)malloc(cols * sizeof(int));
        if (mine[i] == NULL) {
            for (int k = 0; k < i; k++) {
                free(mine[k]);
            }
            free(mine);
            return 1;
        }
    }

    int initial_mine[4][4] = {
        {1, 3, 1, 5},
        {2, 2, 4, 1},
        {5, 0, 2, 3},
        {0, 6, 1, 2}
    };

    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            mine[i][j] = initial_mine[i][j];
        }
    }

    int max_gold = get_max_gold(mine, rows, cols);
    printf("Max gold: %d\n", max_gold);

    for (int i = 0; i < rows; i++) {
        free(mine[i]);
    }
    free(mine);

    return 0;
}
