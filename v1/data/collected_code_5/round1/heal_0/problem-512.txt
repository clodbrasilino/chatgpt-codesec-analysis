```C
#include <stdio.h>

#define MAX_ELEMENTS 100

int count_element(int a[], int n, int b[]) {
    int count[MAX_ELEMENTS] = {0};
    int i;
    for (i = 0; i < n; i++)
        count[a[i]]++;
    for (i = 0; i < MAX_ELEMENTS; i++)
        if (count[i])
            b[i] = count[i];
    return 0;
}

int main() {
    int a[] = {1, 2, 3, 4, 1, 2, 3, 1, 2, 1};
    int n = sizeof(a) / sizeof(a[0]);
    int b[MAX_ELEMENTS] = {0};
    count_element(a, n, b);
    for (int i = 0; i < MAX_ELEMENTS; i++)
        if (b[i])
            printf("Element %d appears %d times\n", i, b[i]);
    return 0;
}
```
