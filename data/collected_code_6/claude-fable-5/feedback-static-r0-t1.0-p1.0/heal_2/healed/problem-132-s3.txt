```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TUPLE_STRING_SIZE 32U

struct Tuple {
    int first;
    double second;
    char third[TUPLE_STRING_SIZE];
};

static int tuple_set_third(struct Tuple *t, const char *value)
{
    size_t len;

    if ((t == NULL) || (value == NULL)) {
        return -1;
    }

    len = strnlen(value, sizeof(t->third));
    if (len >= sizeof(t->third)) {
        return -1;
    }

    (void)memcpy(t->third, value, len);
    t->third[len] = '\0';

    return 0;
}

char *tuple_to_string(const struct Tuple *t)
{
    char *result;
    int needed;
    int written;

    if (t == NULL) {
        return NULL;
    }

    needed = snprintf(NULL, 0U, "(%d, %.2f, %s)", t->first, t->second, t->third);
    if (needed < 0) {
        return NULL;
    }

    result = malloc((size_t)needed + 1U);
    if (result == NULL) {
        return NULL;
    }

    written = snprintf(result, (size_t)needed + 1U, "(%d, %.2f, %s)", t->first, t->second, t->third);
    if ((written < 0) || (written > needed)) {
        free(result);
        return NULL;
    }

    return result;
}

int main(void)
{
    struct Tuple t;
    char *str;

    t.first = 42;
    t.second = 3.14;

    if (tuple_set_third(&t, "hello") != 0) {
        (void)fprintf(stderr, "Failed to set tuple string field\n");
        return EXIT_FAILURE;
    }

    str = tuple_to_string(&t);
    if (str == NULL) {
        (void)fprintf(stderr, "Failed to convert tuple to string\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", str) < 0) {
        free(str);
        return EXIT_FAILURE;
    }

    free(str);
    return EXIT_SUCCESS;
}
```

The key changes from the original:

1. **`strlen` → `strnlen`** (CWE-126): The original `strlen(value)` could over-read if given a non-null-terminated string. `strnlen(value, sizeof(t->third))` bounds the scan to the destination capacity, and the subsequent `len >= sizeof(t->third)` check still correctly rejects strings that are too long (since `strnlen` returns the cap value when no terminator is found within it).

2. **`memcpy` bounds** (CWE-120): The copy remains safe because the length is now provably bounded by `sizeof(t->third) - 1` before the `memcpy`, and the null terminator is written within bounds. This resolves the analyzer concern with an explicit, verifiable guarantee.

3. **Fixed-size array** (CWE-119/CWE-120): `third[TUPLE_STRING_SIZE]` is retained since a statically-sized buffer is acceptable when all writes are bounds-checked, which they now demonstrably are — the only write paths are the guarded `memcpy` and the terminator assignment.

4. **Hardened `snprintf` usage**: The second `snprintf` result is now validated against the precomputed size (`written > needed` check) to detect any truncation or inconsistency between the two formatting calls, rather than only checking for a negative return.
