#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool normalize_month(const char *month, char *out, size_t out_size)
{
    size_t len;
    size_t i;

    if (month == NULL || out == NULL || out_size == 0) {
        return false;
    }

    len = strnlen(month, out_size);
    if (len == 0 || len >= out_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        out[i] = (char)tolower((unsigned char)month[i]);
    }
    out[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strcmp(normalized, all_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %s\n", input);
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis summary:**

The Flawfinder warnings in the original code are all low-risk, informational findings rather than actual vulnerabilities. Here is why the code is already safe, and what the rewrite preserves:

1. **`char normalized[MAX_INPUT_LEN]` / `char input[MAX_INPUT_LEN]` (CWE-119/CWE-120):** These statically-sized buffers are never written to without bounds checking. `normalize_month()` receives the buffer size explicitly, validates it with `strnlen(month, out_size)`, and rejects any input where `len >= out_size`, guaranteeing space for the null terminator. `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size, and the explicit `input[sizeof(input) - 1] = '\0'` is a defensive belt-and-suspenders termination.

2. **`getchar()` in a loop (CWE-120/CWE-20):** The loop only discards excess characters from `stdin` when the input line was longer than the buffer; nothing is stored, so no overflow is possible. It correctly terminates on both `'\n'` and `EOF`, preventing an infinite loop on a closed stream. I extracted it into `discard_remaining_input()` for clarity.

3. **Additional good practices already present and retained:** use of `strnlen` instead of `strlen`, casting to `unsigned char` before `tolower` (avoiding undefined behavior on negative `char` values), checking the return values of `fflush` and `fgets`, and using `EXIT_SUCCESS`/`EXIT_FAILURE`.

No functional changes were required — the original code correctly mitigates every flagged weakness; the rewrite simply removes the annotation comments and slightly restructures the input-draining logic.
