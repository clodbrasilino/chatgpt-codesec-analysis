```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            if (match_len >= sizeof(matches[found].text)) {
                match_len = sizeof(matches[found].text) - 1;
            }

            if (match_len > 0 && start < len && match_len <= len - start) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this version is safe:**

1. **`char text[MAX_INPUT]` / `char input[MAX_INPUT]` (CWE-119/CWE-120, statically-sized arrays):** These warnings are informational — static buffers are only dangerous when writes aren't bounded. Every write into these buffers is bounds-checked:
   - `input` is filled exclusively via `fgets(input, sizeof(input), stdin)`, which cannot overflow, and is explicitly NUL-terminated afterward.
   - `matches[found].text` is only written after `match_len` is clamped to `sizeof(matches[found].text) - 1`, leaving room for the terminator.

2. **`memcpy` (CWE-120):** The copy is safe because:
   - `match_len` is clamped against the destination size before the copy.
   - The source range is validated with `start < len && match_len <= len - start`, which avoids the theoretical unsigned-overflow issue in the original `start + match_len <= len` check.
   - The terminating `'\0'` is written at index `match_len`, which is guaranteed `< sizeof(text)`.

3. **Additional hardening already present:** `strnlen` instead of `strlen` bounds all length computations, all `ctype` calls cast to `unsigned char` to avoid UB on negative `char` values, and all pointer parameters are NULL-checked before use.

The Flawfinder hits are false positives in this code; the rewrite retains the defensive checks (and slightly strengthens the range validation) so the reported CWEs are not exploitable.
