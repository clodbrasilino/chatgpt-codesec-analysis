#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <string.h>

int count_hexadecimal_numbers(unsigned long long start, unsigned long long end, unsigned long long *count) {
    if (count == NULL) {
        return -1;
    }
    
    if (start > end) {
        return -1;
    }
    
    if (start == 0 && end == ULLONG_MAX) {
        return -1;
    }
    
    *count = end - start + 1;
    return 0;
}

int read_ull_input(const char *prompt, unsigned long long *value) {
    char buffer[32];
    char *endptr;
    size_t len;
    
    if (prompt != NULL) {
        printf("%s", prompt);
    }
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return -1;
    }
    
    len = strnlen(buffer, sizeof(buffer));
    if (len == 0) {
        fprintf(stderr, "Empty input\n");
        return -1;
    }
    
    if (len == sizeof(buffer) - 1 && buffer[len - 1] != '\n') {
        fprintf(stderr, "Input too long\n");
        return -1;
    }
    
    if (buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }
    
    errno = 0;
    *value = strtoull(buffer, &endptr, 10);
    if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
        fprintf(stderr, "Invalid value\n");
        return -1;
    }
    
    return 0;
}

int main(void) {
    unsigned long long start;
    unsigned long long end;
    unsigned long long count;
    
    if (read_ull_input("Enter start of range (decimal): ", &start) != 0) {
        return EXIT_FAILURE;
    }
    
    if (read_ull_input("Enter end of range (decimal): ", &end) != 0) {
        return EXIT_FAILURE;
    }
    
    if (count_hexadecimal_numbers(start, end, &count) != 0) {
        fprintf(stderr, "Invalid range\n");
        return EXIT_FAILURE;
    }
    
    printf("Count of hexadecimal numbers in range [%llu, %llu]: %llu\n", start, end, count);
    printf("Hexadecimal range: [0x%llX, 0x%llX]\n", start, end);
    
    return EXIT_SUCCESS;
}
