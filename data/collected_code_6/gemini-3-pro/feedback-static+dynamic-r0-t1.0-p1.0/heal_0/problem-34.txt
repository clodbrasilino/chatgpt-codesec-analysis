#include <stdio.h>
#include <stdlib.h>

int find_missing_number(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return -1;
    }

    size_t left = 0;
    size_t right = size - 1;

    while (left <= right) {
        size_t mid = left + (right - left) / 2;

        if (arr[mid] != (int)(mid + 1)) {
            if (mid == 0 || arr[mid - 1] == (int)mid) {
                return (int)(mid + 1);
            }
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }

    return (int)(size + 1);
}

int main(void) {
    int arr1[] = {1, 2, 3, 5, 6};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    int missing1 = find_missing_number(arr1, size1);
    printf("%d\n", missing1);

    int arr2[] = {1, 2, 3, 4, 5};
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);
    int missing2 = find_missing_number(arr2, size2);
    printf("%d\n", missing2);

    int arr3[] = {2, 3, 4, 5};
    size_t size3 = sizeof(arr3) / sizeof(arr3[0]);
    int missing3 = find_missing_number(arr3, size3);
    printf("%d\n", missing3);

    return 0;
}
