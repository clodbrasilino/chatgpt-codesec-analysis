long count_substrings(const char *str, size_t max_len)
{
    long count;
    size_t len;
    size_t i;
    size_t j;

    if (str == NULL) {
        return -1L;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        return -1L;
    }

    count = 0L;

    for (i = 0U; i < len; i++) {
        for (j = i; j < len; j++) {
            if (str[i] == str[j]) {
                count++;
            }
        }
    }

    return count;
}

    char buffer[BUFFER_SIZE];

        return EXIT_FAILURE;

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    }

    result = count_substrings(buffer, sizeof(buffer));

        return EXIT_FAILURE;
