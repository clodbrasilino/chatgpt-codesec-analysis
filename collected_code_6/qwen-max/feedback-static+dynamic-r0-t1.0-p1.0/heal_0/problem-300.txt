#include <stdio.h>
#include <stdlib.h>

long long countWays(int n) {
    long long C[2 * n + 1][2 * n + 1];
    for (int i = 0; i <= 2 * n; ++i) {
        C[i][0] = 1;
        C[i][i] = 1;
        for (int j = 1; j < i; ++j) {
            C[i][j] = C[i - 1][j - 1] + C[i - 1][j];
        }
    }
    return C[2 * n][n];
}

int main() {
    int n;
    printf("Enter value of n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    printf("Count of ways: %lld\n", countWays(n));
    return 0;
}
