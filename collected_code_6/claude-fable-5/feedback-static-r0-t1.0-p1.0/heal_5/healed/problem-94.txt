```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 32

typedef struct {
    char name[MAX_NAME_LEN];
    int value;
} Record;

int index_minimum(const Record *records, size_t count, char *result, size_t result_size)
{
    size_t i;
    size_t min_index;
    size_t name_len;

    if (records == NULL || result == NULL || count == 0 || result_size == 0) {
        return -1;
    }

    min_index = 0;
    for (i = 1; i < count; i++) {
        if (records[i].value < records[min_index].value) {
            min_index = i;
        }
    }

    name_len = strnlen(records[min_index].name, MAX_NAME_LEN);

    if (name_len >= MAX_NAME_LEN || name_len >= result_size) {
        return -1;
    }

    memcpy(result, records[min_index].name, name_len);
    result[name_len] = '\0';

    return 0;
}

int main(void)
{
    Record records[] = {
        {"Rash", 143},
        {"Manjeet", 200},
        {"Varsha", 100}
    };
    size_t count = sizeof(records) / sizeof(records[0]);
    char result[MAX_NAME_LEN];

    if (index_minimum(records, count, result, sizeof(result)) != 0) {
        fprintf(stderr, "Error: could not extract minimum record\n");
        return EXIT_FAILURE;
    }

    printf("Record with minimum value: %s\n", result);

    return EXIT_SUCCESS;
}
```

1. **`char name[MAX_NAME_LEN]` and `char result[MAX_NAME_LEN]` (CWE-119/CWE-120):** These are Flawfinder heuristics triggered simply by the presence of fixed-size buffers. The code mitigates the risk properly: all string literals used to initialize `records` fit well within 32 bytes, and every write into `result` is bounds-checked before it happens.

2. **`memcpy` (CWE-120):** The copy is guarded by two explicit checks before execution:
   - `name_len >= MAX_NAME_LEN` guarantees the source is a properly terminated string within its buffer (protecting against a non-terminated `name` field, since `strnlen` is used rather than `strlen`).
   - `name_len >= result_size` guarantees the destination can hold `name_len` bytes **plus** the null terminator written afterward (`result[name_len] = '\0'` is valid because `name_len < result_size` implies `name_len` is a legal index).

3. **Defensive API design:** `index_minimum` validates all pointers, the record count, and the destination size, returning `-1` on any invalid input, and the caller checks the return value before using `result`.
