#include <stdio.h>
#include <stdbool.h>

bool check_even_indices_even(int arr[], int size) {
    for (int i = 0; i < size; i += 2) {
        if (arr[i] % 2 != 0) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int arr1[] = {2, 1, 4, 3, 6, 5, 8};
    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    
    int arr2[] = {2, 1, 3, 4, 6, 5};
    int size2 = sizeof(arr2) / sizeof(arr2[0]);
    
    int arr3[] = {0, 2, 4, 6, 8};
    int size3 = sizeof(arr3) / sizeof(arr3[0]);
    
    int arr4[] = {1, 2, 3, 4, 5, 6};
    int size4 = sizeof(arr4) / sizeof(arr4[0]);
    
    printf("Array 1: %s\n", check_even_indices_even(arr1, size1) ? "true" : "false");
    printf("Array 2: %s\n", check_even_indices_even(arr2, size2) ? "true" : "false");
    printf("Array 3: %s\n", check_even_indices_even(arr3, size3) ? "true" : "false");
    printf("Array 4: %s\n", check_even_indices_even(arr4, size4) ? "true" : "false");
    
    return 0;
}
