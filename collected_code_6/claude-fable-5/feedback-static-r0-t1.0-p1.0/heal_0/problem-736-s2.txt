#include <stdio.h>
#include <stdlib.h>

size_t left_insertion_point(const int *arr, size_t n, int value)
{
    size_t low = 0;
    size_t high = n;

    while (low < high) {
        size_t mid = low + (high - low) / 2;
        if (arr[mid] < value) {
            low = mid + 1;
        } else {
            high = mid;
        }
    }
    return low;
}

int main(void)
{
    int data[] = {1, 3, 3, 5, 7, 9, 11};
    size_t n = sizeof(data) / sizeof(data[0]);
    int targets[] = {3, 0, 12, 6, 7};
    size_t t = sizeof(targets) / sizeof(targets[0]);
    size_t i;

    for (i = 0; i < t; i++) {
        size_t pos = left_insertion_point(data, n, targets[i]);
        if (printf("Value %d -> left insertion point: %zu\n",
                   targets[i], pos) < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
