```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

double geometric_sum(int n) {
    if (n < 0) {
        return 0.0;
    }
    double sum = 1.0;
    for (int i = 1; i <= n; i++) {
        sum += 1.0 / pow(2, i);
    }
    return sum;
}

int main() {
    int n;
    printf("Enter an integer: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }
    double result = geometric_sum(n);
    printf("Geometric sum of %d is: %.10f\n", n, result);
    return EXIT_SUCCESS;
}
```
