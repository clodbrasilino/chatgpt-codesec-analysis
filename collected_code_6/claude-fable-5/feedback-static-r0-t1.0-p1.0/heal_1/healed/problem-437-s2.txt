```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096U

char *remove_odd_characters(const char *input)
{
    size_t length;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    length = strnlen(input, MAX_INPUT_LENGTH);
    if (length >= MAX_INPUT_LENGTH) {
        return NULL;
    }

    result = (char *)malloc((length / 2U) + 1U);
    if (result == NULL) {
        return NULL;
    }

    j = 0U;
    for (i = 1U; i < length; i += 2U) {
        result[j] = input[i];
        j++;
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test_strings[] = { "python", "program", "language" };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        char *output = remove_odd_characters(test_strings[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (printf("Input: %s -> Output: %s\n", test_strings[i], output) < 0) {
            free(output);
            return EXIT_FAILURE;
        }
        free(output);
        output = NULL;
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder CWE-126 warning: `strlen` was replaced with `strnlen` bounded by `MAX_INPUT_LENGTH`, so a string that is not properly `\0`-terminated can no longer cause an unbounded over-read. If the bound is reached (meaning no terminator was found within the allowed range), the function safely returns `NULL` instead of processing potentially invalid memory. All other defensive measures from the original — `NULL` input checks, `malloc` failure handling, `printf` return-value checking, and freeing/nulling the pointer after use — are preserved.
