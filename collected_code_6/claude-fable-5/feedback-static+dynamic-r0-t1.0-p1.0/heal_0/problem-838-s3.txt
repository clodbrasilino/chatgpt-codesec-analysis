#include <stdio.h>
#include <string.h>

int min_swaps_to_equal(const char *s1, const char *s2, int *result)
{
    size_t len1;
    size_t len2;
    size_t i;
    long count01 = 0;
    long count10 = 0;

    if (s1 == NULL || s2 == NULL || result == NULL) {
        return -1;
    }

    len1 = strlen(s1);
    len2 = strlen(s2);

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] == '1' && s2[i] == '0') {
            count10++;
        } else if (s1[i] == '0' && s2[i] == '1') {
            count01++;
        }
    }

    if ((count10 % 2L) != (count01 % 2L)) {
        return -1;
    }

    *result = (int)(count10 / 2L + count01 / 2L + (count10 % 2L) * 2L);
    return 0;
}

int main(void)
{
    const char *a = "1101";
    const char *b = "0011";
    int swaps = 0;

    if (min_swaps_to_equal(a, b, &swaps) == 0) {
        if (printf("Minimum swaps required: %d\n", swaps) < 0) {
            return 1;
        }
    } else {
        if (printf("Strings cannot be made equal\n") < 0) {
            return 1;
        }
    }

    return 0;
}
