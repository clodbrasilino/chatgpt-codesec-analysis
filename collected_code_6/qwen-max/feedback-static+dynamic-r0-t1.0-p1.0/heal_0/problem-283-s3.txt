#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool check_digit_frequency(long num) {
    int count[10] = {0};
    while (num > 0) {
        int digit = num % 10;
        if (count[digit] >= digit) return false;
        count[digit]++;
        num /= 10;
    }
    for (int i = 0; i < 10; i++) {
        if (count[i] > i) return false;
    }
    return true;
}

int main() {
    long n;
    printf("Enter a number: ");
    if (scanf("%ld", &n) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    if (check_digit_frequency(n)) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    return 0;
}
