#include <stdio.h>
#include <stdlib.h>

double harmonic_sum_n_minus_one(int n) {
    double sum = 0.0;

    if (n <= 1) {
        return 0.0;
    }

    for (int i = 1; i < n; i++) {
        sum += 1.0 / (double)i;
    }

    return sum;
}

int main(void) {
    int n = 5;
    double result = 0.0;

    if (n > 1) {
        result = harmonic_sum_n_minus_one(n);
        printf("%.6f\n", result);
    } else {
        printf("0.000000\n");
    }

    return EXIT_SUCCESS;
}
