#include <stdio.h>
#include <string.h>

void reverseString(char* begin, char* end) {
    while (begin < end) {
        char temp = *begin;
        *begin++ = *end;
        *end-- = temp;
    }
}

void reverseWords(char* s) {
    if (s == NULL) {
        return;
    }

    char* word_begin = s;
    char* temp = s;

    while (*temp) {
        temp++;
        if (*temp == '\0') {
            reverseString(word_begin, temp - 1);
        } else if (*temp == ' ') {
            reverseString(word_begin, temp - 1);
            word_begin = temp + 1;
        }
    }
    
    reverseString(s, temp - 1);
}

int main(void) {
    char str[] = "Hello World C Programming";
    
    reverseWords(str);
    printf("%s\n", str);
    
    return 0;
}
