#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct List {
    int *data;
    size_t length;
} List;

int compare_lists(const void *a, const void *b) {
    List *list1 = (List *)a;
    List *list2 = (List *)b;
    if (list1->length != list2->length) return 0;
    for (size_t i = 0; i < list1->length; ++i) {
        if (list1->data[i] != list2->data[i]) return 0;
    }
    return 1;
}

int count_unique_lists(List *lists, size_t num_lists) {
    int unique_count = 0;
    for (size_t i = 0; i < num_lists; ++i) {
        int is_unique = 1;
        for (size_t j = 0; j < i; ++j) {
            if (compare_lists(&lists[j], &lists[i])) {
                is_unique = 0;
                break;
            }
        }
        if (is_unique) unique_count++;
    }
    return unique_count;
}

int main() {
    List lists[] = {
        { .data = (int[]){1, 2, 3}, .length = 3 },
        { .data = (int[]){4, 5, 6}, .length = 3 },
        { .data = (int[]){1, 2, 3}, .length = 3 },
        { .data = (int[]){7, 8, 9}, .length = 3 }
    };
    size_t num_lists = sizeof(lists) / sizeof(lists[0]);
    int unique_count = count_unique_lists(lists, num_lists);
    printf("Number of unique lists: %d\n", unique_count);
    return 0;
}
