```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <errno.h>

struct rectangular {
    double x;
    double y;
};

static int polar_to_rectangular(double radius, double angle_rad, struct rectangular *result)
{
    if (result == NULL) {
        return -1;
    }

    if (isnan(radius) || isnan(angle_rad) || isinf(radius) || isinf(angle_rad)) {
        return -1;
    }

    if (radius < 0.0) {
        return -1;
    }

    errno = 0;
    result->x = radius * cos(angle_rad);
    result->y = radius * sin(angle_rad);

    if (errno != 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    double radius = 0.0;
    double angle_deg = 0.0;
    double angle_rad = 0.0;
    struct rectangular rect = { 0.0, 0.0 };

    printf("Enter radius: ");
    if (scanf("%lf", &radius) != 1) {
        fprintf(stderr, "Error: invalid input for radius.\n");
        return EXIT_FAILURE;
    }

    printf("Enter angle in degrees: ");
    if (scanf("%lf", &angle_deg) != 1) {
        fprintf(stderr, "Error: invalid input for angle.\n");
        return EXIT_FAILURE;
    }

    angle_rad = angle_deg * (M_PI / 180.0);

    if (polar_to_rectangular(radius, angle_rad, &rect) != 0) {
        fprintf(stderr, "Error: conversion failed due to invalid values.\n");
        return EXIT_FAILURE;
    }

    printf("Rectangular coordinates: x = %.6f, y = %.6f\n", rect.x, rect.y);

    return EXIT_SUCCESS;
}
```

Note: If your compiler does not define `M_PI` (it is not required by strict ISO C), compile with `-D_USE_MATH_DEFINES` or add this before the includes take effect:

```c
#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif
```

- **CWE-703**: All `scanf` return values are checked, `errno` is checked after math operations, and NaN/infinity inputs are rejected before use.
- **CWE-664**: No dynamic resources (heap memory, file handles) are allocated, so nothing can leak or be misused across its lifetime; the output struct is passed by pointer and validated for NULL.
