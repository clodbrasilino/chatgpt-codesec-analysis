```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

typedef void (*PairCallback)(int first, int second, void *context);

static Node *create_node(int value)
{
    Node *node = malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->value = value;
    node->next = NULL;
    return node;
}

static void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

static Node *build_list(const int *values, size_t count)
{
    Node *head = NULL;
    Node *tail = NULL;
    size_t i;

    if (values == NULL || count == 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        Node *node = create_node(values[i]);
        if (node == NULL) {
            free_list(head);
            return NULL;
        }
        if (head == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }
    return head;
}

static int iterate_pairs(const Node *head, PairCallback callback, void *context)
{
    const Node *current;

    if (head == NULL || callback == NULL) {
        return -1;
    }

    current = head;
    while (current->next != NULL) {
        callback(current->value, current->next->value, context);
        current = current->next;
    }
    return 0;
}

static void print_pair(int first, int second, void *context)
{
    size_t *counter = context;

    if (counter != NULL) {
        (*counter)++;
    }
    printf("(%d, %d)\n", first, second);
}

int main(void)
{
    int values[] = {1, 2, 3, 4, 5};
    size_t count = sizeof(values) / sizeof(values[0]);
    size_t pair_count = 0;
    Node *list;

    list = build_list(values, count);
    if (list == NULL) {
        fprintf(stderr, "Failed to build list\n");
        return EXIT_FAILURE;
    }

    if (iterate_pairs(list, print_pair, &pair_count) != 0) {
        fprintf(stderr, "Failed to iterate pairs\n");
        free_list(list);
        return EXIT_FAILURE;
    }

    printf("Total pairs: %zu\n", pair_count);

    free_list(list);
    return EXIT_SUCCESS;
}
```
