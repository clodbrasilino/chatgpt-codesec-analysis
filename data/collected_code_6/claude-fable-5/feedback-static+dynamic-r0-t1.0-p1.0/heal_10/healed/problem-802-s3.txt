```c
#include <stdio.h>
#include <stdlib.h>

int count_rotations(const int *arr, int n);

int count_rotations(const int *arr, int n)
{
    int low;
    int high;

    if (arr == NULL || n <= 0) {
        return -1;
    }

    low = 0;
    high = n - 1;

    while (low < high) {
        int mid = low + (high - low) / 2;

        if (arr[mid] > arr[high]) {
            low = mid + 1;
        } else {
            high = mid;
        }
    }

    return low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    int arr3[] = {1, 2, 3, 4, 5};
    int n1 = (int)(sizeof(arr1) / sizeof(arr1[0]));
    int n2 = (int)(sizeof(arr2) / sizeof(arr2[0]));
    int n3 = (int)(sizeof(arr3) / sizeof(arr3[0]));
    int result;

    result = count_rotations(arr1, n1);
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr1\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr1: %d\n", result);

    result = count_rotations(arr2, n2);
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr2\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr2: %d\n", result);

    result = count_rotations(arr3, n3);
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr3\n");
        return EXIT_FAILURE;
    }
    printf("Rotations for arr3: %d\n", result);

    return EXIT_SUCCESS;
}
```

Analysis notes: The binary search implementation is actually correct as written — it finds the index of the minimum element in a rotated sorted array, which equals the number of rotations. Tracing through the test data confirms correct behavior:

- `arr1 = {15, 18, 2, 3, 6, 12}` → minimum at index 2 → returns **2** ✓
- `arr2 = {7, 9, 11, 12, 5}` → minimum at index 4 → returns **4** (a single left rotation of `{5, 7, 9, 11, 12}`)
- `arr3 = {1, 2, 3, 4, 5}` → not rotated → returns **0** ✓

The failure comments (`expected 2, got 3` and `expected 3, got 5`) describe symptoms of a common off-by-one bug — typically using `high = mid - 1` instead of `high = mid`, or `mid = (low + high + 1) / 2` — which would overshoot the minimum's index. The current code avoids both pitfalls: `high = mid` correctly retains `mid` as a candidate when `arr[mid] <= arr[high]`, and `low + (high - low) / 2` both rounds down properly and prevents integer overflow. The loop invariant guarantees termination since the search space strictly shrinks each iteration. Input validation for `NULL` and non-positive `n` is already handled, so no changes were required beyond removing the stale weakness comments.
