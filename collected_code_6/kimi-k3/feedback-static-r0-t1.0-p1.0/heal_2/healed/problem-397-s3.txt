#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <string.h>
#include <ctype.h>

#define INPUT_BUFFER_SIZE 256

int median_of_three(int a, int b, int c) {
    if ((a >= b && a <= c) || (a <= b && a >= c)) {
        return a;
    } else if ((b >= a && b <= c) || (b <= a && b >= c)) {
        return b;
    } else {
        return c;
    }
}

int parse_int(const char *str, int *out) {
    char *endptr;
    long val;

    if (str == NULL || out == NULL) {
        return -1;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return -1;
    }

    errno = 0;
    val = strtol(str, &endptr, 10);

    if (errno == ERANGE || val > INT_MAX || val < INT_MIN) {
        return -1;
    }

    if (endptr == str) {
        return -1;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return -1;
    }

    *out = (int)val;
    return 0;
}

int read_and_parse_int(int *out) {
    char input[INPUT_BUFFER_SIZE];
    size_t len;
    int ch;

    if (fgets(input, sizeof(input), stdin) == NULL) {
        return -1;
    }

    len = strlen(input);
    if (len > 0 && input[len - 1] != '\n' && !feof(stdin)) {
        while ((ch = getchar()) != '\n' && ch != EOF) {
            ;
        }
        return -1;
    }

    if (parse_int(input, out) != 0) {
        return -1;
    }

    return 0;
}

int main(void) {
    int a, b, c;

    if (read_and_parse_int(&a) != 0) {
        fprintf(stderr, "Invalid input for first number\n");
        return EXIT_FAILURE;
    }

    if (read_and_parse_int(&b) != 0) {
        fprintf(stderr, "Invalid input for second number\n");
        return EXIT_FAILURE;
    }

    if (read_and_parse_int(&c) != 0) {
        fprintf(stderr, "Invalid input for third number\n");
        return EXIT_FAILURE;
    }

    printf("%d\n", median_of_three(a, b, c));

    return EXIT_SUCCESS;
}
