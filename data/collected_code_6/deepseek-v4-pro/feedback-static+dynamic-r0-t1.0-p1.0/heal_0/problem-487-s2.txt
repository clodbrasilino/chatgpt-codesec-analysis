#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
} Tuple;

int compare_tuples(const void *a, const void *b) {
    const Tuple *tuple_a = (const Tuple *)a;
    const Tuple *tuple_b = (const Tuple *)b;
    if (tuple_a->data[tuple_a->size - 1] < tuple_b->data[tuple_b->size - 1]) return -1;
    if (tuple_a->data[tuple_a->size - 1] > tuple_b->data[tuple_b->size - 1]) return 1;
    return 0;
}

void sort_tuples(Tuple *tuples, int count) {
    if (tuples == NULL || count <= 0) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

int main(void) {
    int data1[] = {1, 3};
    int data2[] = {4, 1};
    int data3[] = {2, 2};
    int data4[] = {5, 0};
    int data5[] = {7, 4};

    Tuple tuples[] = {
        {data1, 2},
        {data2, 2},
        {data3, 2},
        {data4, 2},
        {data5, 2}
    };

    int count = sizeof(tuples) / sizeof(tuples[0]);

    sort_tuples(tuples, count);

    for (int i = 0; i < count; i++) {
        printf("(");
        for (int j = 0; j < tuples[i].size; j++) {
            printf("%d", tuples[i].data[j]);
            if (j < tuples[i].size - 1) {
                printf(", ");
            }
        }
        printf(")\n");
    }

    return 0;
}
