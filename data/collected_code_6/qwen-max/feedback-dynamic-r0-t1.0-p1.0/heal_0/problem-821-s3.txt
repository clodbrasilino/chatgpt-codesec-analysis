#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    int value;
} DictionaryEntry;

typedef struct {
    DictionaryEntry *entries;
    size_t count;
} Dictionary;

Dictionary* createDictionary() {
    Dictionary *dict = (Dictionary *)malloc(sizeof(Dictionary));
    dict->count = 0;
    dict->entries = NULL;
    return dict;
}

void addEntry(Dictionary *dict, const char *key, int value) {
    dict->count++;
    dict->entries = (DictionaryEntry *)realloc(dict->entries, dict->count * sizeof(DictionaryEntry));
    dict->entries[dict->count - 1].key = strdup(key);
    dict->entries[dict->count - 1].value = value;
}

Dictionary* mergeDictionaries(Dictionary *dict1, Dictionary *dict2) {
    Dictionary *merged = createDictionary();
    for (size_t i = 0; i < dict1->count; i++) {
        addEntry(merged, dict1->entries[i].key, dict1->entries[i].value);
    }
    for (size_t i = 0; i < dict2->count; i++) {
        int found = 0;
        for (size_t j = 0; j < merged->count; j++) {
            if (strcmp(merged->entries[j].key, dict2->entries[i].key) == 0) {
                found = 1;
                break;
            }
        }
        if (!found) {
            addEntry(merged, dict2->entries[i].key, dict2->entries[i].value);
        }
    }
    return merged;
}

void freeDictionary(Dictionary *dict) {
    for (size_t i = 0; i < dict->count; i++) {
        free(dict->entries[i].key);
    }
    free(dict->entries);
    free(dict);
}

int main() {
    Dictionary *dict1 = createDictionary();
    addEntry(dict1, "one", 1);
    addEntry(dict1, "two", 2);

    Dictionary *dict2 = createDictionary();
    addEntry(dict2, "two", 20);
    addEntry(dict2, "three", 3);

    Dictionary *merged = mergeDictionaries(dict1, dict2);

    for (size_t i = 0; i < merged->count; i++) {
        printf("%s: %d\n", merged->entries[i].key, merged->entries[i].value);
    }

    freeDictionary(dict1);
    freeDictionary(dict2);
    freeDictionary(merged);

    return 0;
}
