#include <stdio.h>
#include <stdlib.h>

int max_path_sum(int **triangle, int rows) {
    if (triangle == NULL || rows <= 0) {
        return 0;
    }

    int **dp = (int **)malloc(rows * sizeof(int *));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < rows; i++) {
        dp[i] = (int *)malloc((i + 1) * sizeof(int));
        if (dp[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return 0;
        }
    }

    dp[0][0] = triangle[0][0];

    for (int i = 1; i < rows; i++) {
        for (int j = 0; j <= i; j++) {
            if (j == 0) {
                dp[i][j] = dp[i - 1][j] + triangle[i][j];
            } else if (j == i) {
                dp[i][j] = dp[i - 1][j - 1] + triangle[i][j];
            } else {
                int left = dp[i - 1][j - 1];
                int right = dp[i - 1][j];
                dp[i][j] = (left > right ? left : right) + triangle[i][j];
            }
        }
    }

    int max_sum = dp[rows - 1][0];
    for (int j = 1; j < rows; j++) {
        if (dp[rows - 1][j] > max_sum) {
            max_sum = dp[rows - 1][j];
        }
    }

    for (int i = 0; i < rows; i++) {
        free(dp[i]);
    }
    free(dp);

    return max_sum;
}

int main(void) {
    int rows = 4;
    int **triangle = (int **)malloc(rows * sizeof(int *));
    if (triangle == NULL) {
        return 1;
    }

    int row0[] = {3};
    int row1[] = {7, 4};
    int row2[] = {2, 4, 6};
    int row3[] = {8, 5, 9, 3};

    triangle[0] = row0;
    triangle[1] = row1;
    triangle[2] = row2;
    triangle[3] = row3;

    int result = max_path_sum(triangle, rows);
    printf("%d\n", result);

    free(triangle);

    return 0;
}
