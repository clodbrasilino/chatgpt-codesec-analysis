#include <stdio.h>
#include <stdbool.h>

bool areArraysEqual(int* array1, int arr1Size, int* array2, int arr2Size) {
    int i;
    
    if (arr1Size != arr2Size) {
        return false;
    }
    
    for(i=0; i<arr1Size; i++) {
        if(array1[i] != array2[i]) {
            return false;
        }
    }
    return true;
}

int main() {
    int array1[] = {1, 2, 3, 4, 5};
    int array2[] = {1, 2, 3, 4, 5};
    int array3[] = {1, 2, 3, 4, 6};

    int arr1Size = sizeof(array1) / sizeof(array1[0]);
    int arr2Size = sizeof(array2) / sizeof(array2[0]);
    int arr3Size = sizeof(array3) / sizeof(array3[0]);

    if(areArraysEqual(array1, arr1Size, array2, arr2Size)) {
        printf("Array1 and Array2 are equal.\n");
    } else {
        printf("Array1 and Array2 are not equal.\n");
    }

    if(areArraysEqual(array1, arr1Size, array3, arr3Size)) {
        printf("Array1 and Array3 are equal.\n");
    } else {
        printf("Array1 and Array3 are not equal.\n");
    }

    return 0;
}
