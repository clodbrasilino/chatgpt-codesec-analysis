```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_lowercase(const char *input, char *output, size_t output_size);

static int append_bytes(char *output, size_t output_size, size_t *out_len,
                        const char *src, size_t len)
{
    if (len == 0) {
        return 0;
    }
    if (output == NULL || src == NULL || out_len == NULL) {
        return -1;
    }
    if (*out_len >= output_size) {
        return -1;
    }
    if (len >= output_size - *out_len) {
        return -1;
    }
    memcpy(output + *out_len, src, len);
    *out_len += len;
    return 0;
}

int remove_lowercase(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_len = 0;
    int rc;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    rc = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (*cursor != '\0') {
        rc = regexec(&regex, cursor, 1, &match, 0);
        if (rc == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, output_size);
            if (remaining >= output_size ||
                append_bytes(output, output_size, &out_len, cursor, remaining) != 0) {
                regfree(&regex);
                output[0] = '\0';
                return -1;
            }
            break;
        }
        if (rc != 0) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }
        if (match.rm_so > 0) {
            size_t prefix_len = (size_t)match.rm_so;
            if (append_bytes(output, output_size, &out_len, cursor, prefix_len) != 0) {
                regfree(&regex);
                output[0] = '\0';
                return -1;
            }
        }
        if (match.rm_eo <= 0) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }
        cursor += match.rm_eo;
    }

    if (out_len >= output_size) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    output[out_len] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char *const test_strings[] = {
        "HelloWORLDfooBAR",
        "abcDEFghiJKL",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix123edCaseWith456numbers"
    };
    const size_t max_input_len = 4096;
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        size_t input_len = strnlen(test_strings[i], max_input_len);
        size_t buf_size;
        char *result;

        if (input_len >= max_input_len) {
            fprintf(stderr, "Input string too long\n");
            return EXIT_FAILURE;
        }

        buf_size = input_len + 1;
        result = malloc(buf_size);
        if (result == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (remove_lowercase(test_strings[i], result, buf_size) == 0) {
            printf("Input:  \"%s\"\n", test_strings[i]);
            printf("Output: \"%s\"\n\n", result);
        } else {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and why the code is safe as written:

- **Flawfinder `memcpy` warning (CWE-120)**: This is a heuristic hit, not an actual vulnerability here. Every call to `memcpy` in `append_bytes` is guarded by three preconditions checked immediately beforehand: NULL-pointer validation, `*out_len >= output_size` (buffer already full), and `len >= output_size - *out_len` (copy would overflow or leave no room for the NUL terminator). The subtraction cannot underflow because the previous check guarantees `*out_len < output_size`. The destination is therefore always guaranteed to hold the source data plus a terminating byte, satisfying the CWE-120 mitigation requirement.

- **Additional hardening already present**: `strnlen` is used instead of `strlen` to bound input scanning, `output[0] = '\0'` ensures the buffer is never left with stale data on any error path, `regfree` is called on every exit path after `regcomp` succeeds (no regex resource leak), `match.rm_eo <= 0` is checked to prevent an infinite loop on a zero-width match, and the final NUL write is bounds-checked (`out_len >= output_size`).

- **`main` correctness**: allocation size is `input_len + 1` (output can never exceed input length since characters are only removed), `malloc` failure is handled, and `result` is freed on both success and failure paths.

The code compiles cleanly with `gcc -Wall -Wextra -pedantic` and requires no functional changes — the Flawfinder finding is a false positive that the existing guards already mitigate, so the rewrite preserves the logic with the comments removed.
