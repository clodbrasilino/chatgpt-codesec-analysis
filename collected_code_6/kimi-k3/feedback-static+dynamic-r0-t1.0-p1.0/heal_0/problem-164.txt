```c
#include <stdio.h>
#include <stdlib.h>

int sum_of_divisors(int n) {
    int sum = 0;
    int i;
    if (n < 1) {
        return 0;
    }
    for (i = 1; i <= n / 2; i++) {
        if (n % i == 0) {
            sum += i;
        }
    }
    return sum;
}

int main(void) {
    int num1, num2;
    int sum1, sum2;
    int result;
    
    printf("Enter two positive integers: ");
    result = scanf("%d %d", &num1, &num2);
    
    if (result != 2) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (num1 < 1 || num2 < 1) {
        fprintf(stderr, "Numbers must be positive\n");
        return EXIT_FAILURE;
    }
    
    sum1 = sum_of_divisors(num1);
    sum2 = sum_of_divisors(num2);
    
    if (sum1 == sum2) {
        printf("The sum of divisors are the same: %d\n", sum1);
    } else {
        printf("The sum of divisors are different: %d and %d\n", sum1, sum2);
    }
    
    return EXIT_SUCCESS;
}
```
