#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>
#include <errno.h>

char *remove_uppercase_substrings(const char *str)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len = 0;
    size_t result_capacity = 0;
    size_t i = 0;
    size_t str_len;
    int ret;

    if (str == NULL) {
        return NULL;
    }

    str_len = strnlen(str, SIZE_MAX);
    if (str_len == SIZE_MAX) {
        return NULL;
    }

    result_capacity = str_len + 1;
    result = malloc(result_capacity);
    if (result == NULL) {
        return NULL;
    }
    result[0] = '\0';

    ret = regcomp(&regex, "[A-Z]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    while (i < str_len) {
        ret = regexec(&regex, str + i, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = str_len - i;
            if (remaining > 0) {
                if (result_len + remaining + 1 > result_capacity) {
                    size_t new_capacity = result_len + remaining + 1;
                    char *new_result = realloc(result, new_capacity);
                    if (new_result == NULL) {
                        free(result);
                        regfree(&regex);
                        return NULL;
                    }
                    result = new_result;
                    result_capacity = new_capacity;
                }
                if (result_len + remaining >= result_capacity) {
                    free(result);
                    regfree(&regex);
                    return NULL;
                }
                if (result_len + remaining + 1 <= result_capacity) {
                    memcpy(result + result_len, str + i, remaining);
                    result_len += remaining;
                    result[result_len] = '\0';
                } else {
                    free(result);
                    regfree(&regex);
                    return NULL;
                }
            }
            break;
        } else if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        size_t match_start = i + match.rm_so;
        size_t match_end = i + match.rm_eo;
        size_t prefix_len = match_start - i;

        if (prefix_len > 0) {
            if (result_len + prefix_len + 1 > result_capacity) {
                size_t new_capacity = result_len + prefix_len + 1;
                char *new_result = realloc(result, new_capacity);
                if (new_result == NULL) {
                    free(result);
                    regfree(&regex);
                    return NULL;
                }
                result = new_result;
                result_capacity = new_capacity;
            }
            if (result_len + prefix_len >= result_capacity) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            if (result_len + prefix_len + 1 <= result_capacity) {
                memcpy(result + result_len, str + i, prefix_len);
                result_len += prefix_len;
                result[result_len] = '\0';
            } else {
                free(result);
                regfree(&regex);
                return NULL;
            }
        }

        i = match_end;
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_string = "HelloWORLDThisISaTESTString";
    char *result = remove_uppercase_substrings(test_string);

    if (result == NULL) {
        fprintf(stderr, "Error processing string\n");
        return EXIT_FAILURE;
    }

    printf("Original: %s\n", test_string);
    printf("Modified: %s\n", result);

    free(result);
    return EXIT_SUCCESS;
}
