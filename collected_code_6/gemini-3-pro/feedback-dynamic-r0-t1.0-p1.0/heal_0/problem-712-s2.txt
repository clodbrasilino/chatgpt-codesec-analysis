#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct Node {
    int value;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
    struct List* next;
} List;

Node* create_node(int value) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) {
        exit(EXIT_FAILURE);
    }
    new_node->value = value;
    new_node->next = NULL;
    return new_node;
}

List* create_list(Node* head) {
    List* new_list = (List*)malloc(sizeof(List));
    if (!new_list) {
        exit(EXIT_FAILURE);
    }
    new_list->head = head;
    new_list->next = NULL;
    return new_list;
}

bool are_lists_equal(Node* head1, Node* head2) {
    Node* curr1 = head1;
    Node* curr2 = head2;
    while (curr1 != NULL && curr2 != NULL) {
        if (curr1->value != curr2->value) {
            return false;
        }
        curr1 = curr1->next;
        curr2 = curr2->next;
    }
    return curr1 == NULL && curr2 == NULL;
}

void free_node_list(Node* head) {
    while (head != NULL) {
        Node* temp = head;
        head = head->next;
        free(temp);
    }
}

void remove_duplicate_lists(List** head_ref) {
    if (head_ref == NULL || *head_ref == NULL || (*head_ref)->next == NULL) {
        return;
    }

    List* current = *head_ref;
    while (current != NULL) {
        List* runner = current;
        while (runner->next != NULL) {
            if (are_lists_equal(current->head, runner->next->head)) {
                List* duplicate = runner->next;
                runner->next = duplicate->next;
                free_node_list(duplicate->head);
                free(duplicate);
            } else {
                runner = runner->next;
            }
        }
        current = current->next;
    }
}

void free_list_of_lists(List* head) {
    while (head != NULL) {
        List* temp = head;
        head = head->next;
        free_node_list(temp->head);
        free(temp);
    }
}

void print_list_of_lists(List* head) {
    while (head != NULL) {
        Node* curr = head->head;
        printf("[");
        while (curr != NULL) {
            printf("%d", curr->value);
            if (curr->next != NULL) {
                printf(", ");
            }
            curr = curr->next;
        }
        printf("]\n");
        head = head->next;
    }
}

int main(void) {
    Node* list1 = create_node(1);
    list1->next = create_node(2);
    list1->next->next = create_node(3);

    Node* list2 = create_node(4);
    list2->next = create_node(5);

    Node* list3 = create_node(1);
    list3->next = create_node(2);
    list3->next->next = create_node(3);

    Node* list4 = create_node(6);

    List* head = create_list(list1);
    head->next = create_list(list2);
    head->next->next = create_list(list3);
    head->next->next->next = create_list(list4);

    remove_duplicate_lists(&head);

    print_list_of_lists(head);

    free_list_of_lists(head);

    return EXIT_SUCCESS;
}
