#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 26
#define MAX_WORD_LEN 64

static int extract_words(const char *str, char words[MAX_WORDS][MAX_WORD_LEN], size_t *count);
static int safe_copy_word(char *dst, size_t dst_size, const char *src);
int follows_pattern(const char *pattern, const char *str);

static int extract_words(const char *str, char words[MAX_WORDS][MAX_WORD_LEN], size_t *count)
{
    size_t i = 0;
    size_t w = 0;

    if (str == NULL || words == NULL || count == NULL) {
        return -1;
    }

    *count = 0;

    while (str[i] != '\0') {
        while (str[i] == ' ') {
            i++;
        }
        if (str[i] == '\0') {
            break;
        }
        if (w >= MAX_WORDS) {
            return -1;
        }
        size_t len = 0;
        while (str[i] != '\0' && str[i] != ' ') {
            if (len >= MAX_WORD_LEN - 1) {
                return -1;
            }
            words[w][len] = str[i];
            len++;
            i++;
        }
        words[w][len] = '\0';
        w++;
    }

    *count = w;
    return 0;
}

static int safe_copy_word(char *dst, size_t dst_size, const char *src)
{
    size_t src_len;

    if (dst == NULL || src == NULL || dst_size == 0) {
        return -1;
    }

    src_len = strnlen(src, dst_size);
    if (src_len >= dst_size) {
        return -1;
    }

    memcpy(dst, src, src_len);
    dst[src_len] = '\0';
    return 0;
}

int follows_pattern(const char *pattern, const char *str)
{
    char words[MAX_WORDS][MAX_WORD_LEN];
    char mapping[26][MAX_WORD_LEN];
    int mapped[26] = {0};
    size_t word_count = 0;
    size_t pattern_len;
    size_t i;

    if (pattern == NULL || str == NULL) {
        return 0;
    }

    memset(words, 0, sizeof(words));
    memset(mapping, 0, sizeof(mapping));

    if (extract_words(str, words, &word_count) != 0) {
        return 0;
    }

    pattern_len = strnlen(pattern, MAX_WORDS + 1);

    if (pattern_len > MAX_WORDS || pattern_len != word_count) {
        return 0;
    }

    for (i = 0; i < pattern_len; i++) {
        char c = pattern[i];
        if (c < 'a' || c > 'z') {
            return 0;
        }
        size_t idx = (size_t)(c - 'a');
        if (mapped[idx]) {
            if (strncmp(mapping[idx], words[i], MAX_WORD_LEN) != 0) {
                return 0;
            }
        } else {
            size_t j;
            for (j = 0; j < 26; j++) {
                if (mapped[j] && strncmp(mapping[j], words[i], MAX_WORD_LEN) == 0) {
                    return 0;
                }
            }
            if (safe_copy_word(mapping[idx], MAX_WORD_LEN, words[i]) != 0) {
                return 0;
            }
            mapped[idx] = 1;
        }
    }

    return 1;
}

int main(void)
{
    const char *patterns[] = {"abba", "abba", "aaaa", "abba"};
    const char *strings[] = {
        "dog cat cat dog",
        "dog cat cat fish",
        "dog cat cat dog",
        "dog dog dog dog"
    };
    size_t n = sizeof(patterns) / sizeof(patterns[0]);
    size_t i;

    for (i = 0; i < n; i++) {
        int result = follows_pattern(patterns[i], strings[i]);
        if (printf("Pattern \"%s\" with \"%s\": %s\n",
                   patterns[i], strings[i],
                   result ? "true" : "false") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why this code is safe:

1. **`char words[MAX_WORDS][MAX_WORD_LEN]` / `char mapping[26][MAX_WORD_LEN]` (CWE-119/CWE-120)**: These are false positives in practice. Flawfinder flags all statically-sized arrays, but every write into these buffers is bounds-checked:
   - `extract_words` rejects input once `w >= MAX_WORDS` and once `len >= MAX_WORD_LEN - 1`, guaranteeing room for the NUL terminator.
   - Both arrays are zero-initialized with `memset` before use, so reads never touch uninitialized memory.

2. **`memcpy` in `safe_copy_word` (CWE-120)**: Also mitigated. The function validates all pointers, rejects a zero-sized destination, measures the source with `strnlen` bounded by `dst_size`, and refuses to copy if `src_len >= dst_size`. The copy can therefore never overflow, and the terminator write at `dst[src_len]` is always in bounds.

3. **Additional hardening already present**:
   - `strnlen(pattern, MAX_WORDS + 1)` bounds the pattern scan, so an unterminated or oversized pattern cannot cause an over-read or a length mismatch bypass.
   - `strncmp(..., MAX_WORD_LEN)` bounds all string comparisons.
   - Pattern characters are validated to `'a'..'z'` before being used as an index, preventing out-of-bounds access into `mapping`/`mapped`.
   - The `printf` return value is checked, satisfying strict error-handling requirements.

The code as written already contains the fixes the Flawfinder comments call for; the rewrite simply removes the annotations while preserving all bounds checks and input validation.
