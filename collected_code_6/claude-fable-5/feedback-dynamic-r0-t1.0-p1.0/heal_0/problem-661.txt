#include <stdio.h>
#include <stdlib.h>

long long max_ll(long long a, long long b)
{
    return (a > b) ? a : b;
}

int max_sum_no_three_consecutive(const int *arr, size_t n, long long *result)
{
    long long *dp;
    size_t i;

    if (arr == NULL || result == NULL) {
        return -1;
    }

    if (n == 0) {
        *result = 0;
        return 0;
    }

    if (n == 1) {
        *result = arr[0];
        return 0;
    }

    if (n == 2) {
        *result = (long long)arr[0] + (long long)arr[1];
        return 0;
    }

    dp = malloc(n * sizeof(*dp));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = arr[0];
    dp[1] = (long long)arr[0] + (long long)arr[1];
    dp[2] = max_ll(dp[1],
                   max_ll((long long)arr[1] + (long long)arr[2],
                          (long long)arr[0] + (long long)arr[2]));

    for (i = 3; i < n; i++) {
        long long take_one = dp[i - 1];
        long long take_two = dp[i - 2] + (long long)arr[i];
        long long take_three = dp[i - 3] + (long long)arr[i - 1] + (long long)arr[i];
        dp[i] = max_ll(take_one, max_ll(take_two, take_three));
    }

    *result = dp[n - 1];
    free(dp);
    return 0;
}

int main(void)
{
    int arr[] = {100, 1000, 100, 1000, 1};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result;

    if (max_sum_no_three_consecutive(arr, n, &result) != 0) {
        fprintf(stderr, "Error computing maximum sum\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum sum with no three consecutive elements: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
