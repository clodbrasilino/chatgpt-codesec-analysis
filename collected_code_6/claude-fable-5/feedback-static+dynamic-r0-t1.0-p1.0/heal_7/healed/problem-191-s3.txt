```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32U

static int has_thirty_days(const char *month);
static void to_lower_string(char *dest, const char *src, size_t dest_size);
static size_t bounded_strlen(const char *str, size_t max_len);
static void flush_stdin(void);

static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < 4096U);
}

static int has_thirty_days(const char *month)
{
    static const char *const thirty_day_months[] = {
        "april",
        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN] = {0};
    size_t i;
    size_t count;

    if (month == NULL) {
        return -1;
    }

    if (bounded_strlen(month, MAX_INPUT_LEN) >= MAX_INPUT_LEN) {
        return -1;
    }

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    for (i = 0U; i < count; i++) {
        if (strncmp(lowered, thirty_day_months[i], sizeof(lowered)) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;
    int result;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = bounded_strlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        flush_stdin();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "Empty input provided.\n");
        return EXIT_FAILURE;
    }

    result = has_thirty_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name provided.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and applied fixes:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — The original `flush_stdin()` was already safe since `getchar()` returns an `int` and terminates on `\n` or `EOF`. As a defensive hardening measure, I added a guard counter to bound the loop iterations, eliminating any possibility of an unbounded loop on pathological streams.

2. **Statically-sized `char` arrays (CWE-119/CWE-120)** — Both `lowered[MAX_INPUT_LEN]` and `input[MAX_INPUT_LEN]` were already bounds-checked correctly: `fgets` limits input to `sizeof(input)`, `bounded_strlen` never reads past `max_len`, and `to_lower_string` respects `dest_size`. I additionally zero-initialized both buffers to guarantee null termination in all code paths, and replaced `strcmp` with `strncmp` bounded by the buffer size to ensure the comparison can never read beyond the local buffer even if an invariant were broken elsewhere.

The original code was fundamentally sound — the Flawfinder hits were risk-level 1–2 informational warnings rather than actual vulnerabilities — so the rewrite focuses on defense-in-depth rather than correcting real defects.
