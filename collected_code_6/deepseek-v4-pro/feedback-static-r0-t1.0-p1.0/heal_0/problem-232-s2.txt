#include <stdio.h>
#include <stdlib.h>

int cmp_desc(const void *a, const void *b) {
    double diff = *(const double *)b - *(const double *)a;
    if (diff > 0) return 1;
    if (diff < 0) return -1;
    return 0;
}

int get_n_largest(const double *dataset, size_t dataset_size, size_t n, double *result) {
    if (dataset == NULL || result == NULL || n > dataset_size) {
        return 0;
    }

    double *copy = malloc(dataset_size * sizeof(*copy));
    if (copy == NULL) {
        return 0;
    }

    for (size_t i = 0; i < dataset_size; i++) {
        copy[i] = dataset[i];
    }

    qsort(copy, dataset_size, sizeof(*copy), cmp_desc);

    for (size_t i = 0; i < n; i++) {
        result[i] = copy[i];
    }

    free(copy);
    return 1;
}

int main(void) {
    double data[] = {12.5, 3.2, 45.1, 7.8, 23.9, 1.1, 56.4, 34.0};
    size_t data_size = sizeof(data) / sizeof(data[0]);
    size_t n = 4;
    double largest[4];

    if (get_n_largest(data, data_size, n, largest)) {
        for (size_t i = 0; i < n; i++) {
            printf("%.1f ", largest[i]);
        }
        printf("\n");
    } else {
        printf("Error processing data\n");
    }

    return 0;
}
