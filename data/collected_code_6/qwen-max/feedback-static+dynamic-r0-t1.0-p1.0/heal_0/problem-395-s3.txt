```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

char findFirstNonRepeatedChar(const char *str) {
    int count[ UCHAR_MAX + 1 ] = { 0 };
    for (int i = 0; str[i]; ++i) {
        count[(unsigned char)str[i]]++;
    }
    for (int i = 0; str[i]; ++i) {
        if (count[(unsigned char)str[i]] == 1) {
            return str[i];
        }
    }
    return '\0';
}

int main() {
    const char *input = "programming";
    char result = findFirstNonRepeatedChar(input);
    if (result != '\0') {
        printf("The first non-repeated character is: %c\n", result);
    } else {
        printf("There is no non-repeated character in the string.\n");
    }
    return 0;
}
```
