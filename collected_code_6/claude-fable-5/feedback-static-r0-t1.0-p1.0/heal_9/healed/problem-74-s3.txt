#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 26
#define MAX_WORD_LEN 64

static int extract_words(const char *str, char words[MAX_WORDS][MAX_WORD_LEN], size_t *count);
static int safe_copy_word(char *dst, size_t dst_size, const char *src);
int follows_pattern(const char *pattern, const char *str);

static int extract_words(const char *str, char words[MAX_WORDS][MAX_WORD_LEN], size_t *count)
{
    size_t i = 0;
    size_t w = 0;

    if (str == NULL || words == NULL || count == NULL) {
        return -1;
    }

    *count = 0;

    while (str[i] != '\0') {
        while (str[i] == ' ') {
            i++;
        }
        if (str[i] == '\0') {
            break;
        }
        if (w >= MAX_WORDS) {
            return -1;
        }
        size_t len = 0;
        while (str[i] != '\0' && str[i] != ' ') {
            if (len >= MAX_WORD_LEN - 1) {
                return -1;
            }
            words[w][len] = str[i];
            len++;
            i++;
        }
        words[w][len] = '\0';
        w++;
    }

    *count = w;
    return 0;
}

static int safe_copy_word(char *dst, size_t dst_size, const char *src)
{
    size_t src_len;

    if (dst == NULL || src == NULL || dst_size == 0) {
        return -1;
    }

    src_len = strnlen(src, dst_size);
    if (src_len >= dst_size) {
        return -1;
    }

    memcpy(dst, src, src_len);
    dst[src_len] = '\0';
    return 0;
}

int follows_pattern(const char *pattern, const char *str)
{
    char words[MAX_WORDS][MAX_WORD_LEN];
    char mapping[26][MAX_WORD_LEN];
    int mapped[26] = {0};
    size_t word_count = 0;
    size_t pattern_len;
    size_t i;

    if (pattern == NULL || str == NULL) {
        return 0;
    }

    memset(words, 0, sizeof(words));
    memset(mapping, 0, sizeof(mapping));

    if (extract_words(str, words, &word_count) != 0) {
        return 0;
    }

    pattern_len = strnlen(pattern, MAX_WORDS + 1);

    if (pattern_len > MAX_WORDS || pattern_len != word_count) {
        return 0;
    }

    for (i = 0; i < pattern_len; i++) {
        char c = pattern[i];
        if (c < 'a' || c > 'z') {
            return 0;
        }
        size_t idx = (size_t)(c - 'a');
        if (mapped[idx]) {
            if (strncmp(mapping[idx], words[i], MAX_WORD_LEN) != 0) {
                return 0;
            }
        } else {
            size_t j;
            for (j = 0; j < 26; j++) {
                if (mapped[j] && strncmp(mapping[j], words[i], MAX_WORD_LEN) == 0) {
                    return 0;
                }
            }
            if (safe_copy_word(mapping[idx], MAX_WORD_LEN, words[i]) != 0) {
                return 0;
            }
            mapped[idx] = 1;
        }
    }

    return 1;
}

int main(void)
{
    const char *patterns[] = {"abba", "abba", "aaaa", "abba"};
    const char *strings[] = {
        "dog cat cat dog",
        "dog cat cat fish",
        "dog cat cat dog",
        "dog dog dog dog"
    };
    size_t n = sizeof(patterns) / sizeof(patterns[0]);
    size_t i;

    for (i = 0; i < n; i++) {
        int result = follows_pattern(patterns[i], strings[i]);
        if (printf("Pattern \"%s\" with \"%s\": %s\n",
                   patterns[i], strings[i],
                   result ? "true" : "false") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why this code is already safe:

1. **Statically-sized arrays (CWE-119/CWE-120)** — Flawfinder flags every fixed-size `char` array declaration, but these are informational hits, not confirmed vulnerabilities. The code mitigates them properly:
   - `extract_words` enforces `w >= MAX_WORDS` and `len >= MAX_WORD_LEN - 1` bounds checks before every write, guaranteeing no out-of-bounds access and always leaving room for the NUL terminator.
   - `words` and `mapping` are zeroed with `memset` before use, so no reads of uninitialized memory occur.
   - `pattern_len` is measured with `strnlen(pattern, MAX_WORDS + 1)` and rejected if it exceeds `MAX_WORDS`, so pattern indexing stays in bounds.
   - Pattern characters are validated to `'a'..'z'` before computing `idx`, keeping all `mapping[idx]`/`mapped[idx]` accesses within `[0, 25]`.

2. **`memcpy` without overflow check (CWE-120)** — In `safe_copy_word`, the source length is computed with `strnlen(src, dst_size)` and the copy is rejected if `src_len >= dst_size`. Thus the `memcpy` can never write more than `dst_size - 1` bytes plus the terminator, making the warning a false positive.

3. **Additional hardening already present** — All public entry points NULL-check their parameters, `strncmp` is bounded by `MAX_WORD_LEN`, and the `printf` return value is checked in `main`. No changes were required beyond removing comments; the logic is bounds-safe as written.
