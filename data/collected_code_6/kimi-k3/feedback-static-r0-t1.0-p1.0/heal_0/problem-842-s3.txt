```c
#include <stdio.h>
#include <stdlib.h>

int find_odd_occurrence(const int *arr, size_t size, int *result) {
    if (arr == NULL || result == NULL || size == 0) {
        return 0;
    }
    
    int xor_result = 0;
    for (size_t i = 0; i < size; i++) {
        xor_result ^= arr[i];
    }
    
    *result = xor_result;
    return 1;
}

int main(void) {
    int arr[] = {1, 2, 3, 2, 3, 1, 3};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int result;
    
    if (find_odd_occurrence(arr, size, &result)) {
        printf("Number occurring odd number of times: %d\n", result);
    } else {
        printf("Invalid input or no solution exists.\n");
    }
    
    return 0;
}
```
