```c
#include <stdio.h>
#include <stdlib.h>

int max_segments(int n, int a, int b, int c) {
    if (n < 0 || a <= 0 || b <= 0 || c <= 0) {
        return -1;
    }

    int *dp = malloc(((size_t)n + 1) * sizeof(*dp));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = 0;
    for (int i = 1; i <= n; ++i) {
        dp[i] = -1;
    }

    const int lengths[3] = {a, b, c};

    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j < 3; ++j) {
            int len = lengths[j];
            if (i >= len && dp[i - len] != -1) {
                int cand = dp[i - len] + 1;
                if (cand > dp[i]) {
                    dp[i] = cand;
                }
            }
        }
    }

    int result = dp[n];
    free(dp);
    return result;
}

int main(void) {
    int n, a, b, c;
    if (scanf("%d %d %d %d", &n, &a, &b, &c) != 4) {
        return 1;
    }

    printf("%d\n", max_segments(n, a, b, c));
    return 0;
}
```
