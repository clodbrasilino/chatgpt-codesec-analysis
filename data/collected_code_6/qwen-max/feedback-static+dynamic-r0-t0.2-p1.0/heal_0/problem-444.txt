```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *data;
    size_t length;
} Tuple;

void trim_tuple(Tuple *t, int k) {
    if (k < 0 || k >= t->length) return;
    for (int i = 0; i < t->length - k; ++i) {
        t->data[i] = t->data[i + k];
    }
    t->length -= k;
}

void free_tuple(Tuple *t) {
    free(t->data);
    t->data = NULL;
    t->length = 0;
}

int main() {
    Tuple t1 = { .data = (int[]){1, 2, 3, 4, 5}, .length = 5 };
    Tuple t2 = { .data = (int[]){6, 7, 8, 9, 10, 11, 12}, .length = 7 };

    int k1 = 2;
    int k2 = 3;

    trim_tuple(&t1, k1);
    trim_tuple(&t2, k2);

    for (size_t i = 0; i < t1.length; ++i) {
        printf("%d ", t1.data[i]);
    }
    printf("\n");

    for (size_t i = 0; i < t2.length; ++i) {
        printf("%d ", t2.data[i]);
    }
    printf("\n");

    free_tuple(&t1);
    free_tuple(&t2);

    return 0;
}
```
