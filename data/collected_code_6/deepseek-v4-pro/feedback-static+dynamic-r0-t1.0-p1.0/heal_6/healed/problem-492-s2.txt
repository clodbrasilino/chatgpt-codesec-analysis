#include <stdio.h>
#include <stdbool.h>

bool binary_search(const int arr[], int size, int target) {
    if (arr == NULL || size <= 0) {
        return false;
    }
    
    int left = 0;
    int right = size - 1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target) {
            return true;
        }
        
        if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return false;
}

int main(void) {
    int arr[] = {2, 5, 8, 12, 16, 23, 38, 45, 56, 72, 91};
    int size = sizeof(arr) / sizeof(arr[0]);
    int target = 23;
    
    bool found = binary_search(arr, size, target);
    
    if (found) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    
    return 0;
}
