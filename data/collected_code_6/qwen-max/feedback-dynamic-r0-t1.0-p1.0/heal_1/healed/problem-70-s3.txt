#include <stdio.h>
#include <stdlib.h>

int check_tuples_equal_length(int **tuples, int n, int *lengths) {
    for (int i = 0; i < n - 1; i++) {
        if (lengths[i] != lengths[i + 1]) {
            return 0;
        }
    }
    return 1;
}

int main() {
    int n = 3;
    int lengths[] = {4, 4, 4};
    int **tuples = (int **)malloc(n * sizeof(int *));
    
    tuples[0] = (int *)malloc(4 * sizeof(int));
    tuples[1] = (int *)malloc(4 * sizeof(int));
    tuples[2] = (int *)malloc(4 * sizeof(int));

    tuples[0][0] = 1; tuples[0][1] = 2; tuples[0][2] = 3; tuples[0][3] = 4;
    tuples[1][0] = 5; tuples[1][1] = 6; tuples[1][2] = 7; tuples[1][3] = 8;
    tuples[2][0] = 9; tuples[2][1] = 10; tuples[2][2] = 11; tuples[2][3] = 12;

    int all_equal = check_tuples_equal_length(tuples, n, lengths);
    
    if (all_equal) {
        printf("All tuples have equal length.\n");
    } else {
        printf("Not all tuples have equal length.\n");
    }

    for (int i = 0; i < n; i++) {
        free(tuples[i]);
    }
    free(tuples);
    return 0;
}
