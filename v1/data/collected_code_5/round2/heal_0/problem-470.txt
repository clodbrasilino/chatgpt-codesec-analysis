```c
#include <stdio.h>

#define SIZE 4

typedef struct {
   int values[SIZE];
} Tuple;

Tuple add_tuples(Tuple t1, Tuple t2) {
    Tuple result;
    for (int i = 0; i < SIZE; i++) {
        result.values[i] = t1.values[i] + t2.values[i];
    }
    return result;
}

void print_tuple(Tuple t) {
    printf("(");
    for (int i = 0; i < SIZE; i++) {
        if (i == SIZE - 1) {
            printf("%d", t.values[i]);
        } else {
            printf("%d, ", t.values[i]);
        }
    }
    printf(")\n");
}

int main() {
    Tuple t1 = {{1, 2, 3, 4}};
    Tuple t2 = {{5, 6, 7, 8}};
    
    Tuple result = add_tuples(t1, t2);

    print_tuple(result);

    return 0;
}
```
