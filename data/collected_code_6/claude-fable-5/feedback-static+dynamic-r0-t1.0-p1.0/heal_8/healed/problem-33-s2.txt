#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define BINARY_BUFFER_SIZE (sizeof(unsigned int) * CHAR_BIT + 1)

static int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[index++] = '0';
        buffer[index] = '\0';
        return 0;
    }

    while (decimal > 0) {
        if (index >= buffer_size - 1) {
            buffer[0] = '\0';
            return -1;
        }
        buffer[index++] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
    }
    buffer[index] = '\0';

    for (size_t i = 0, j = index - 1; i < j; i++, j--) {
        char temp = buffer[i];
        buffer[i] = buffer[j];
        buffer[j] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[BINARY_BUFFER_SIZE] = {0};
    int result;

    printf("Enter a non-negative decimal number: ");

    result = scanf("%u", &number);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary representation: %s\n", binary);

    return EXIT_SUCCESS;
}

**Analysis of the identified weakness (CWE-119/CWE-120):**

The Flawfinder warning on the `char binary[...]` declaration is a heuristic hit on statically-sized arrays. In this code the risk is already fully mitigated, so no functional changes were required:

1. **Buffer size is derived, not guessed** — `BINARY_BUFFER_SIZE` is computed as `sizeof(unsigned int) * CHAR_BIT + 1`, which is exactly the maximum number of binary digits an `unsigned int` can produce plus one byte for the null terminator. It cannot be too small on any conforming platform.

2. **Size is passed explicitly** — the caller passes `sizeof(binary)` rather than relying on a hardcoded constant, so the function and the buffer cannot silently drift out of sync.

3. **Bounds checking inside the loop** — the check `index >= buffer_size - 1` guarantees the write of each digit plus the terminator never exceeds the buffer, failing safely with `-1` and an empty string instead of overflowing.

4. **Defensive parameter validation** — `NULL` and undersized buffers (`buffer_size < 2`) are rejected before any write occurs.

5. **Safe input handling** — `scanf("%u", ...)` writes to a fixed-size integer (no string format specifier), and its return value is checked, so no buffer overflow is possible on input.

The warning is therefore a false positive given the existing mitigations; the code stands as-is.
