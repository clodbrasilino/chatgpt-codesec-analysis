#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void leftRotateString(char* str, size_t n) {
    if (str == NULL) {
        return;
    }

    size_t len = strlen(str);
    if (len == 0 || n == 0 || n % len == 0) {
        return;
    }

    n = n % len;
    
    char* temp = (char*)malloc((n + 1) * sizeof(char));
    if (temp == NULL) {
        return;
    }

    strncpy(temp, str, n);
    temp[n] = '\0';

    memmove(str, str + n, len - n);
    strncpy(str + len - n, temp, n);

    free(temp);
}

int main(void) {
    char str[] = "HelloWorld";
    size_t rotations = 3;

    leftRotateString(str, rotations);
    printf("%s\n", str);

    return 0;
}
