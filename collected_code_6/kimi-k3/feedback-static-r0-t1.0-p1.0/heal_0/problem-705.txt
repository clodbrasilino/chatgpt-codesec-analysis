#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
} IntList;

static int compare_lists(const void *a, const void *b) {
    const IntList *list_a = (const IntList *)a;
    const IntList *list_b = (const IntList *)b;
    
    if (list_a->length < list_b->length) {
        return -1;
    }
    if (list_a->length > list_b->length) {
        return 1;
    }
    
    for (size_t i = 0; i < list_a->length; i++) {
        if (list_a->data[i] < list_b->data[i]) {
            return -1;
        }
        if (list_a->data[i] > list_b->data[i]) {
            return 1;
        }
    }
    
    return 0;
}

void sort_list_of_lists(IntList *lists, size_t count) {
    if (lists == NULL || count == 0) {
        return;
    }
    qsort(lists, count, sizeof(IntList), compare_lists);
}

static void free_lists(IntList *lists, size_t count) {
    if (lists == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(lists[i].data);
        lists[i].data = NULL;
    }
    free(lists);
}

static void print_lists(const IntList *lists, size_t count) {
    if (lists == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        printf("[");
        for (size_t j = 0; j < lists[i].length; j++) {
            printf("%d", lists[i].data[j]);
            if (j < lists[i].length - 1) {
                printf(", ");
            }
        }
        printf("]\n");
    }
}

int main(void) {
    const size_t num_lists = 5;
    IntList *lists = malloc(num_lists * sizeof(IntList));
    if (lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    int values[][3] = {
        {3, 1, 2},
        {1, 2, 0},
        {1, 2, 3},
        {5, 0, 0},
        {1, 1, 0}
    };
    size_t lengths[] = {3, 2, 3, 1, 2};
    
    for (size_t i = 0; i < num_lists; i++) {
        lists[i].length = lengths[i];
        lists[i].data = malloc(lengths[i] * sizeof(int));
        if (lists[i].data == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_lists(lists, i);
            return EXIT_FAILURE;
        }
        for (size_t j = 0; j < lengths[i]; j++) {
            lists[i].data[j] = values[i][j];
        }
    }
    
    printf("Before sorting:\n");
    print_lists(lists, num_lists);
    
    sort_list_of_lists(lists, num_lists);
    
    printf("\nAfter sorting:\n");
    print_lists(lists, num_lists);
    
    free_lists(lists, num_lists);
    return EXIT_SUCCESS;
}
