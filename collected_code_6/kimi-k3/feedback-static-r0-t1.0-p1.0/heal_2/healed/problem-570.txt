#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int remove_words_containing(char **list, int size, const char *substring) {
    int i, j;
    
    if (list == NULL || substring == NULL || size < 0) {
        return -1;
    }
    
    for (i = 0, j = 0; i < size; i++) {
        if (list[i] == NULL) {
            continue;
        }
        if (strstr(list[i], substring) != NULL) {
            free(list[i]);
            list[i] = NULL;
        } else {
            list[j] = list[i];
            if (i != j) {
                list[i] = NULL;
            }
            j++;
        }
    }
    
    return j;
}

        return 1;
    }
    
    for (i = 0; i < size; i++) {
        words[i] = malloc(strlen(word_list[i]) + 1);
        if (words[i] == NULL) {
            int k;
            for (k = 0; k < i; k++) {
                free(words[k]);
            }
            free(words);
            return 1;
        }
        snprintf(words[i], strlen(word_list[i]) + 1, "%s", word_list[i]);
    }
    
    printf("Original list:\n");
    for (i = 0; i < size; i++) {
        printf("%s\n", words[i]);

    return 0;
}
