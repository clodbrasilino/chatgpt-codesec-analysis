#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

void rearrange_negatives_first(int *arr, size_t n)
{
    size_t j = 0;
    size_t i;
    int temp;

    if (arr == NULL || n == 0) {
        return;
    }

    for (i = 0; i < n; i++) {
        if (arr[i] < 0) {
            temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            j++;
        }
    }
}

void print_array(const int *arr, size_t n)
{
    size_t i;

    if (arr == NULL) {
        return;
    }

    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main(void)
{
    int data[] = { 12, -7, 5, -3, 0, -9, 8, -1, 4 };
    size_t n = sizeof(data) / sizeof(data[0]);

    printf("Original array:\n");
    print_array(data, n);

    rearrange_negatives_first(data, n);

    printf("Rearranged array:\n");
    print_array(data, n);

    return EXIT_SUCCESS;
}
