```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SENTENCE_LEN 4096

int find_first_max_even_word(const char *sentence, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len = 0;
    size_t max_len = 0;
    size_t best_start = 0;
    int found = 0;

    if (sentence == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len == MAX_SENTENCE_LEN) {
        return -1;
    }

    while (i < len) {
        while (i < len && sentence[i] == ' ') {
            i++;
        }
        if (i < len) {
            size_t start = i;
            size_t word_len = 0;
            while (i < len && sentence[i] != ' ') {
                i++;
            }
            word_len = i - start;
            if ((word_len % 2 == 0) && (word_len > max_len)) {
                max_len = word_len;
                best_start = start;
                found = 1;
            }
        }
    }

    if (!found) {
        return -1;
    }

    if (max_len >= result_size) {
        return -1;
    }

    if (best_start > len || max_len > len - best_start) {
        return -1;
    }

    memcpy(result, sentence + best_start, max_len);
    result[max_len] = '\0';
    return 0;
}

int main(void)
{
    const char *sentence = "It is a pleasant day today";
    char result[256];

    memset(result, 0, sizeof(result));

    if (find_first_max_even_word(sentence, result, sizeof(result)) == 0) {
        result[sizeof(result) - 1] = '\0';
        if (printf("First maximum length even word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No even length word found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

1. **`memcpy` (CWE-120):** This warning is a false positive in context. The copy is already fully guarded before execution:
   - `max_len >= result_size` returns early, guaranteeing the destination can hold `max_len` bytes **plus** the null terminator.
   - `best_start > len || max_len > len - best_start` guarantees the source read stays within the bounds of `sentence` (written to avoid integer overflow in the addition).
   
   No change was required — the existing checks make the `memcpy` provably safe.

2. **Statically-sized `char result[256]` (CWE-119/CWE-120):** Also mitigated. The buffer size is always passed via `sizeof(result)` rather than a hard-coded constant, the callee validates `result_size` against the data length before writing, the buffer is zero-initialized with `memset`, and a defensive `result[sizeof(result) - 1] = '\0'` guarantees termination before printing.

Additional defensive measures already present and preserved: `strnlen` with a hard cap (`MAX_SENTENCE_LEN`) instead of unbounded `strlen`, NULL-pointer validation on all inputs, use of `size_t` throughout to avoid signed/unsigned issues, and checked `printf` return values.
