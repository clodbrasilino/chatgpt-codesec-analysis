#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} KeyValuePair;

typedef struct {
    KeyValuePair *pairs;
    size_t count;
} Dictionary;

static char *clone_string(const char *src) {
    if (src == NULL) {
        return NULL;
    }
    size_t len = strlen(src) + 1;
    char *dest = (char *)malloc(len);
    if (dest == NULL) {
        return NULL;
    }
    memcpy(dest, src, len);
    return dest;
}

Dictionary *tuple_to_dict(const char **tuple, size_t size) {
    if (tuple == NULL || size == 0 || size % 2 != 0) {
        return NULL;
    }

    Dictionary *dict = (Dictionary *)malloc(sizeof(Dictionary));
    if (dict == NULL) {
        return NULL;
    }

    dict->count = size / 2;
    dict->pairs = (KeyValuePair *)malloc(dict->count * sizeof(KeyValuePair));
    if (dict->pairs == NULL) {
        free(dict);
        return NULL;
    }

    for (size_t i = 0; i < dict->count; i++) {
        if (tuple[i * 2] == NULL || tuple[i * 2 + 1] == NULL) {
            for (size_t j = 0; j < i; j++) {
                free(dict->pairs[j].key);
                free(dict->pairs[j].value);
            }
            free(dict->pairs);
            free(dict);
            return NULL;
        }

        dict->pairs[i].key = clone_string(tuple[i * 2]);
        if (dict->pairs[i].key == NULL) {
            for (size_t j = 0; j < i; j++) {
                free(dict->pairs[j].key);
                free(dict->pairs[j].value);
            }
            free(dict->pairs);
            free(dict);
            return NULL;
        }

        dict->pairs[i].value = clone_string(tuple[i * 2 + 1]);
        if (dict->pairs[i].value == NULL) {
            free(dict->pairs[i].key);
            for (size_t j = 0; j < i; j++) {
                free(dict->pairs[j].key);
                free(dict->pairs[j].value);
            }
            free(dict->pairs);
            free(dict);
            return NULL;
        }
    }

    return dict;
}

void free_dictionary(Dictionary *dict) {
    if (dict != NULL) {
        if (dict->pairs != NULL) {
            for (size_t i = 0; i < dict->count; i++) {
                free(dict->pairs[i].key);
                free(dict->pairs[i].value);
            }
            free(dict->pairs);
        }
        free(dict);
    }
}

int main(void) {
    const char *tuple[] = {"name", "Alice", "age", "30", "city", "Wonderland"};
    size_t tuple_size = sizeof(tuple) / sizeof(tuple[0]);

    Dictionary *dict = tuple_to_dict(tuple, tuple_size);
    if (dict == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < dict->count; i++) {
        printf("%s: %s\n", dict->pairs[i].key, dict->pairs[i].value);
    }

    free_dictionary(dict);

    return EXIT_SUCCESS;
}
