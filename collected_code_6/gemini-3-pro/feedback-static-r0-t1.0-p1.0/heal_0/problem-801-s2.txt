#include <stdio.h>

int count_equal(int a, int b, int c) {
    if (a == b && b == c) {
        return 3;
    } else if (a == b || a == c || b == c) {
        return 2;
    } else {
        return 0;
    }
}

int main(void) {
    int num1, num2, num3;
    int items_read;

    items_read = scanf("%d %d %d", &num1, &num2, &num3);

    if (items_read != 3) {
        return 1;
    }

    int result = count_equal(num1, num2, num3);
    printf("%d\n", result);

    return 0;
}
