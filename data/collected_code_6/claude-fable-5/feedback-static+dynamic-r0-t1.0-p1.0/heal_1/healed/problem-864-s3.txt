```c
#include <stdio.h>
#include <stddef.h>
#include <string.h>
#include <stdbool.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096U

static bool is_palindrome(const char *str)
{
    size_t left;
    size_t right;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_STRING_LENGTH);
    if ((len == 0U) || (len >= MAX_STRING_LENGTH)) {
        return false;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        if (tolower((unsigned char)str[left]) != tolower((unsigned char)str[right])) {
            return false;
        }
        left++;
        right--;
    }

    return true;
}

static size_t find_palindromes(const char *const *strings, size_t count,
                               const char **results, size_t results_capacity)
{
    size_t found;

    if ((strings == NULL) || (results == NULL) || (results_capacity == 0U)) {
        return 0U;
    }

    found = 0U;
    for (size_t i = 0U; i < count; i++) {
        if (is_palindrome(strings[i])) {
            if (found < results_capacity) {
                results[found] = strings[i];
                found++;
            } else {
                break;
            }
        }
    }

    return found;
}

int main(void)
{
    const char *const words[] = {
        "racecar",
        "hello",
        "Level",
        "world",
        "noon",
        "example",
        "madam"
    };
    const size_t word_count = sizeof(words) / sizeof(words[0]);
    const char *palindromes[sizeof(words) / sizeof(words[0])] = { NULL };
    size_t found;

    found = find_palindromes(words, word_count, palindromes, word_count);

    if (found == 0U) {
        if (puts("No palindromes found.") == EOF) {
            return 1;
        }
    } else {
        if (printf("Found %zu palindrome(s):\n", found) < 0) {
            return 1;
        }
        for (size_t i = 0U; i < found; i++) {
            if (palindromes[i] == NULL) {
                return 1;
            }
            if (printf("%s\n", palindromes[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced `strlen` with `strnlen` bounded by `MAX_STRING_LENGTH`, rejecting strings that reach the limit since they may be unterminated.

2. **CWE-119/CWE-120 (statically-sized array)**: The array size is derived directly from `sizeof(words) / sizeof(words[0])`, guaranteeing capacity matches the input exactly. Added zero-initialization and a `NULL` check before dereferencing entries as defense-in-depth, and `find_palindromes` already enforces `results_capacity` bounds checking.

3. **variableScope**: Moved the loop counter `i` declarations into the `for` statement scope in both `find_palindromes` and `main`, minimizing variable lifetime per C99 rules.
