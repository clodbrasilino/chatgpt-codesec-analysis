```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        sum += (unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long result;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (strchr(buffer, '\n') == NULL && !feof(stdin)) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and changes made:**

The Flawfinder warning (CWE-119/CWE-120) about the statically-sized `char buffer[MAX_INPUT_LEN]` was largely a false positive in the original code, since it already applied the correct mitigations:

1. **Bounded reads** — `fgets(buffer, sizeof(buffer), stdin)` never writes past the buffer, unlike `gets()` or unbounded `scanf("%s", ...)`.
2. **Bounded length calculation** — `strnlen(str, max_len)` guarantees the scan never runs past the buffer even if the string were not NUL-terminated, and rejecting the `len == max_len` case guards against unterminated input.
3. **Safe newline stripping** — `strcspn(buffer, "\n")` always returns an index within the buffer (at worst the position of the terminating NUL), so the write is in bounds.
4. **Signedness safety** — casting to `unsigned char` before summing avoids sign-extension issues on platforms where `char` is signed.

The rewrite hardens the remaining edge case: if the user enters more than 255 characters, `fgets` truncates silently and the excess remains in `stdin`. The new version detects truncation (no `\n` present and not at EOF), drains the leftover input, and reports an error instead of silently computing a sum over a truncated string. An `fflush(stdout)` was also added to guarantee the prompt appears before blocking on input on line-buffered or fully-buffered streams.
