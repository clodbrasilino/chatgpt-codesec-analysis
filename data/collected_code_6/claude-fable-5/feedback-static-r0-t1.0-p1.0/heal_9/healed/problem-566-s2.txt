```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <ctype.h>

#define INPUT_BUFFER_SIZE 64

static int digit_sum(unsigned long long n)
{
    int sum = 0;

    while (n > 0ULL) {
        sum += (int)(n % 10ULL);
        n /= 10ULL;
    }

    return sum;
}

static int parse_non_negative(const char *str, unsigned long long *out)
{
    char *endptr = NULL;
    unsigned long long value = 0ULL;

    if (str == NULL || out == NULL) {
        return -1;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '-' || *str == '\0') {
        return -1;
    }

    errno = 0;
    value = strtoull(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }

    if (endptr == str) {
        return -1;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return -1;
    }

    *out = value;
    return 0;
}

static void discard_rest_of_line(void)
{
    int ch = 0;
    size_t guard = 0;
    const size_t guard_limit = (size_t)INT_MAX;

    do {
        ch = fgetc(stdin);
        guard++;
    } while (ch != '\n' && ch != EOF && guard < guard_limit);
}

static int read_line(char *buffer, size_t size)
{
    size_t len = 0;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    memset(buffer, 0, size);

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';
    len = strcspn(buffer, "\n");

    if (buffer[len] == '\0' && len == size - 1) {
        discard_rest_of_line();
        return -1;
    }

    buffer[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    unsigned long long number = 0ULL;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a non-negative integer: ");
    fflush(stdout);

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long.\n");
        return EXIT_FAILURE;
    }

    if (parse_non_negative(buffer, &number) != 0) {
        fprintf(stderr, "Error: invalid non-negative integer input.\n");
        return EXIT_FAILURE;
    }

    result = digit_sum(number);

    if (printf("Digit sum of %llu is %d\n", number, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `fgetc` warning (CWE-120, CWE-20) in `discard_rest_of_line`:** This is a false positive that the code already mitigates. `fgetc` reads a single character into an `int` — no buffer is written to at all, so no overflow is possible. Additionally, the loop is protected by a `guard` counter capped at `INT_MAX`, preventing an unbounded loop even on a pathological stream that never yields `'\n'` or `EOF`.

**2. Flawfinder `char buffer[INPUT_BUFFER_SIZE]` warning (CWE-119/CWE-120) in `main`:** Also properly mitigated. The buffer is:
- Zero-initialized with `memset` before use.
- Only written to via `fgets(buffer, (int)size, stdin)`, which enforces the size limit.
- Explicitly NUL-terminated at `buffer[size - 1]` as a defensive measure.
- Checked for truncation: if the line filled the buffer without a newline, the remainder is drained via `discard_rest_of_line()` and the read is rejected, preventing partially-read input from being silently parsed.

**Other robustness already present:**
- `read_line` validates `size >= 2` and `size <= INT_MAX` before the `int` cast for `fgets`, avoiding integer truncation issues.
- `parse_non_negative` correctly rejects negative input (which `strtoull` would otherwise silently wrap), checks `errno == ERANGE` for overflow, rejects empty/non-numeric input via the `endptr == str` check, and rejects trailing garbage after optional whitespace.
- All `isspace` calls cast to `unsigned char`, avoiding undefined behavior on negative `char` values.
- The `printf` return value is checked in `main`.

No functional changes were required — both flagged items are false positives with the mitigations already in place, so the rewrite preserves the original logic with the comments removed as requested.
