```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *str;
    size_t len;
} StringTuple;

StringTuple* string_to_tuple(const char *input) {
    if (input == NULL) return NULL;
    
    StringTuple *tuple = (StringTuple *)malloc(sizeof(StringTuple));
    if (tuple == NULL) return NULL;

    tuple->len = strnlen(input, 100);
    if (tuple->len >= 100) {
        free(tuple);
        return NULL;
    }
    tuple->str = (char *)malloc((tuple->len + 1) * sizeof(char));
    if (tuple->str == NULL) {
        free(tuple);
        return NULL;
    }

    strncpy(tuple->str, input, tuple->len);
    tuple->str[tuple->len] = '\0';
    return tuple;
}

void free_tuple(StringTuple *tuple) {
    if (tuple != NULL) {
        free(tuple->str);
        free(tuple);
    }
}

int main() {
    const char *inputs[] = {"python3.0", "item1", "15.10"};
    for (size_t i = 0; i < 3; ++i) {
        StringTuple *tuple = string_to_tuple(inputs[i]);

        if (tuple != NULL) {
            printf("Tuple: %s (Length: %zu)\n", tuple->str, tuple->len);
            free_tuple(tuple);
        } else {
            printf("Failed to create tuple.\n");
        }
    }

    return 0;
}
```
