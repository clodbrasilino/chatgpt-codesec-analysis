#include <stdio.h>
#include <stdbool.h>

size_t count_true_booleans(const bool *bool_list, size_t size) {
    if (bool_list == NULL) {
        return 0;
    }

    size_t count = 0;
    for (size_t i = 0; i < size; ++i) {
        if (bool_list[i]) {
            count++;
        }
    }
    return count;
}

int main(void) {
    bool my_booleans[] = {true, false, true, true, false, false, true};
    size_t size = sizeof(my_booleans) / sizeof(my_booleans[0]);

    size_t true_count = count_true_booleans(my_booleans, size);

    printf("Number of true booleans: %zu\n", true_count);

    return 0;
}
