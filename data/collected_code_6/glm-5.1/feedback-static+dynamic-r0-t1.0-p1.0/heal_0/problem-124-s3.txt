```c
#include <stdio.h>
#include <math.h>

double get_complex_angle(double real, double imag) {
    return atan2(imag, real);
}

int main(void) {
    double real = -1.0;
    double imag = 1.0;
    double angle = get_complex_angle(real, imag);
    
    if (fabs(angle - M_PI_4) < 1e-9) {
        return 0;
    }
    
    return 1;
}
```
