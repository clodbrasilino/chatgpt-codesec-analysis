#include <stdio.h>
#include <stdbool.h>
#include <math.h>

bool isValidTriangle(double a, double b, double c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return false;
    }
    
    double epsilon = 1e-9;
    if (a + b <= c + epsilon || b + c <= a + epsilon || a + c <= b + epsilon) {
        return false;
    }
    
    return true;
}

int main(void) {
    double side1, side2, side3;
    
    if (scanf("%lf %lf %lf", &side1, &side2, &side3) != 3) {
        printf("False\n");
        return 0;
    }
    
    if (isValidTriangle(side1, side2, side3)) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    
    return 0;
}
