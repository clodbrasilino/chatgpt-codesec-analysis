#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

bool all_tuples_equal_length(const int *const *tuples, const size_t *lengths, size_t count) {
    if (tuples == NULL || lengths == NULL || count == 0) {
        return false;
    }
    
    size_t first_length = lengths[0];
    for (size_t i = 1; i < count; i++) {
        if (lengths[i] != first_length) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int tuple1[] = {1, 2, 3};
    int tuple2[] = {4, 5, 6};
    int tuple3[] = {7, 8, 9};
    
    const int *tuples[] = {tuple1, tuple2, tuple3};
    size_t lengths[] = {3, 3, 3};
    
    if (all_tuples_equal_length(tuples, lengths, 3)) {
        printf("All tuples have equal length\n");
    } else {
        printf("Tuples have different lengths\n");
    }
    
    int tuple4[] = {10, 11};
    const int *tuples2[] = {tuple1, tuple4};
    size_t lengths2[] = {3, 2};
    
    if (all_tuples_equal_length(tuples2, lengths2, 2)) {
        printf("All tuples have equal length\n");
    } else {
        printf("Tuples have different lengths\n");
    }
    
    return 0;
}
