```C
#include <stdio.h>

float find_area(float a, float b, float h) {
    float area = 0.5 * (a + b) * h;
    return area;
}

int main() {
    float a, b, h, area;

    printf("Enter the length of the first parallel side of the trapezium: ");
    scanf("%f", &a);
    if (a <= 0) {
        printf("Length must be positive.\n");
        return 1;
    }

    printf("Enter the length of the second parallel side of the trapezium: ");
    scanf("%f", &b);
    if (b <= 0) {
        printf("Length must be positive.\n");
        return 1;
    }

    printf("Enter the height of the trapezium: ");
    scanf("%f", &h);
    if (h <= 0) {
        printf("Height must be positive.\n");
        return 1;
    }

    area = find_area(a, b, h);
    printf("The area of the trapezium is: %.2f\n", area);

    return 0;
}
```
