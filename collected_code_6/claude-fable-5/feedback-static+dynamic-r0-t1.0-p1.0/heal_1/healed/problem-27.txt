```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 4096

char *remove_digits_from_string(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_STRING_LENGTH);
    if (len == MAX_STRING_LENGTH && input[len] != '\0') {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!isdigit((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

char **remove_digits_from_list(const char **list, size_t count)
{
    char **output;
    size_t i;
    size_t k;

    if (list == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    output = calloc(count, sizeof(char *));
    if (output == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        output[i] = remove_digits_from_string(list[i]);
        if (output[i] == NULL) {
            for (k = 0; k < i; k++) {
                free(output[k]);
            }
            free(output);
            return NULL;
        }
    }

    return output;
}

void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *strings[] = {
        "hello123world",
        "abc456def",
        "2024report",
        "no_digits_here",
        "999"
    };
    size_t count = sizeof(strings) / sizeof(strings[0]);
    char **cleaned;
    size_t i;

    cleaned = remove_digits_from_list(strings, count);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to process string list\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("\"%s\" -> \"%s\"\n", strings[i], cleaned[i]) < 0) {
            free_string_list(cleaned, count);
            return EXIT_FAILURE;
        }
    }

    free_string_list(cleaned, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weakness:

1. **CWE-126 (buffer over-read via `strlen`)**: Replaced `strlen` with `strnlen` bounded by `MAX_STRING_LENGTH`. If the input reaches the maximum length without a terminator being confirmed, the function safely rejects the input rather than reading past the buffer.

2. **Integer overflow hardening**: Added a `count > SIZE_MAX / sizeof(char *)` guard before allocation and switched `malloc(count * sizeof(char *))` to `calloc(count, sizeof(char *))`, which performs its own overflow-checked multiplication and zero-initializes the pointer array (making cleanup paths safer).

3. **Note**: `strnlen` requires `_POSIX_C_SOURCE >= 200809L` or C23; if targeting strict C89/C99, a small manual bounded-scan loop would be the portable equivalent.
