```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <ctype.h>

#define INPUT_BUFFER_SIZE 64
#define MAX_DISCARD_ITERATIONS 4096

static int digit_sum(unsigned long long n)
{
    int sum = 0;
    while (n > 0ULL) {
        sum += (int)(n % 10ULL);
        n /= 10ULL;
    }
    return sum;
}

static int parse_non_negative(const char *str, unsigned long long *out)
{
    char *endptr = NULL;
    unsigned long long value = 0ULL;

    if (str == NULL || out == NULL) {
        return -1;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '-' || *str == '\0') {
        return -1;
    }

    errno = 0;
    value = strtoull(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }

    if (endptr == str) {
        return -1;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return -1;
    }

    *out = value;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch = 0;
    size_t iterations = 0;

    do {
        ch = getchar();
        iterations++;
    } while (ch != '\n' && ch != EOF && iterations < MAX_DISCARD_ITERATIONS);
}

static int read_line(char *buf, size_t size)
{
    size_t len = 0;

    if (buf == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buf, (int)size, stdin) == NULL) {
        return -1;
    }

    buf[size - 1] = '\0';
    len = strcspn(buf, "\n");

    if (buf[len] == '\0' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    unsigned long long number = 0ULL;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a non-negative integer: ");
    fflush(stdout);

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    if (parse_non_negative(buffer, &number) != 0) {
        fprintf(stderr, "Error: invalid non-negative integer input.\n");
        return EXIT_FAILURE;
    }

    result = digit_sum(number);

    if (printf("Digit sum of %llu is %d\n", number, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the changes made to address the flagged weaknesses:

**1. `getchar()` in a loop (CWE-120, CWE-20):** The original `discard_remaining_input()` loop was already safe from buffer overflows since it never writes to a buffer, but it could theoretically loop unbounded on a pathological input stream. I added an explicit iteration cap (`MAX_DISCARD_ITERATIONS`) as a defensive bound, which satisfies the "check boundaries in a loop" concern.

**2. Statically-sized `char` array (CWE-119/CWE-120):** The original code was already handling this correctly — `read_line()` passes `sizeof(buffer)` to `fgets()`, validates the size parameter, and detects/rejects truncated over-long lines. I strengthened it further by forcibly NUL-terminating `buf[size - 1]` before calling `strcspn()`, guaranteeing termination even if the buffer were passed in an unexpected state. Combined with the existing `size < 2` and `size > INT_MAX` checks, all bounds are verified before any read or write.

The remaining logic (`strtoull` with `errno`/`endptr` validation, explicit rejection of negative input, checked `printf` return value) was already sound and is preserved unchanged.
