#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_WORD_LEN 256U

bool is_word_length_even(const char *word, size_t max_len)
{
    size_t len;

    if (word == NULL || max_len == 0U) {
        return false;
    }

    len = strnlen(word, max_len);
    return (len % 2U) == 0U;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = fgetc(stdin);
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[MAX_WORD_LEN] = {0};
    char *newline = NULL;

    printf("Enter a word: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    newline = memchr(buffer, '\n', sizeof(buffer));
    if (newline != NULL) {
        *newline = '\0';
    } else {
        flush_stdin();
    }

    if (buffer[0] == '\0') {
        fprintf(stderr, "No word entered.\n");
        return EXIT_FAILURE;
    }

    if (is_word_length_even(buffer, sizeof(buffer))) {
        printf("The word \"%s\" has an even length.\n", buffer);
    } else {
        printf("The word \"%s\" has an odd length.\n", buffer);
    }

    return EXIT_SUCCESS;
}

**Analysis notes:**

The two Flawfinder findings in the original source were already effectively mitigated, and the rewrite preserves and hardens those mitigations:

1. **`fgetc` in a loop (CWE-120, CWE-20)** — This is a false positive in practice. `flush_stdin()` reads and discards characters into a single `int`; there is no buffer being written, so no overflow is possible. The loop terminates correctly on both `'\n'` and `EOF`, avoiding an infinite loop on closed input streams.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — The buffer is fully bounds-safe:
   - `fgets` is called with `sizeof(buffer)`, guaranteeing no overflow.
   - Explicit null termination at `buffer[sizeof(buffer) - 1U]` is defensive (technically redundant since `fgets` always terminates, but harmless).
   - `memchr` is bounded by `sizeof(buffer)` rather than using unbounded `strchr`.
   - `strnlen` with an explicit `max_len` is used instead of unbounded `strlen`, with a `NULL`/zero-length guard.

Additional improvements in the rewrite: the buffer is zero-initialized, `stdout` is flushed after the unterminated prompt to guarantee it appears before blocking on input, and `_POSIX_C_SOURCE` is defined to ensure `strnlen` is declared portably.
