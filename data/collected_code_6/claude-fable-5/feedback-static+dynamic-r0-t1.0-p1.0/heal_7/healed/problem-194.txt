```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64
#define FLUSH_LIMIT 4096

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t count = 0;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses:**

1. **Flawfinder `getchar` (CWE-120, CWE-20)** — This is a false positive that is already mitigated. The `flush_stdin()` loop never stores characters into a buffer, so no overflow is possible. Additionally, the loop is bounded by `FLUSH_LIMIT` and terminates on `\n` or `EOF`, preventing unbounded iteration or infinite loops on a closed stream.

2. **Flawfinder `char input[INPUT_SIZE]` (CWE-119/CWE-120)** — Also mitigated. All reads into the buffer use `fgets(input, sizeof(input), stdin)`, which enforces the buffer bound. The code additionally:
   - Forces NUL-termination with `input[sizeof(input) - 1] = '\0'`.
   - Uses `strnlen` with an explicit maximum rather than `strlen`.
   - Detects and rejects over-long input (draining the remainder of the line so it can't leak into subsequent reads).

3. **Integer overflow in conversion** — Correctly guarded by the pre-multiplication check `value > (LONG_MAX - (c - '0')) / 8`, which avoids signed overflow (undefined behavior) before it occurs.

4. **Input validation** — Empty input, non-octal characters, and NULL pointers are all rejected with explicit error paths.

The code as written already addresses the Flawfinder findings; the rewrite simply removes the annotation comments while preserving all defensive measures.
