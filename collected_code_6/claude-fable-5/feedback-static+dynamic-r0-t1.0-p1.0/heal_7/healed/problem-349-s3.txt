#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 256U
#define FLUSH_GUARD_LIMIT 1000000U

static bool is_binary_string(const char *str)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0U; str[i] != '\0'; i++) {
        if ((str[i] != '0') && (str[i] != '1')) {
            return false;
        }
    }

    return true;
}

static size_t safe_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = fgetc(stdin);
        guard++;
        if (guard > (size_t)FLUSH_GUARD_LIMIT) {
            break;
        }
    } while ((ch != '\n') && (ch != EOF));
}

int main(void)
{
    char input[MAX_INPUT_LENGTH];
    size_t len;

    (void)memset(input, 0, sizeof(input));

    printf("Enter a string: ");
    (void)fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        (void)fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = safe_strlen(input, sizeof(input));

    if ((len > 0U) && (input[len - 1U] == '\n')) {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        flush_stdin();
    } else {
    }

    if (is_binary_string(input)) {
        printf("The string is a binary string.\n");
    } else {
        printf("The string is not a binary string.\n");
    }

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already well-hardened, and the two Flawfinder findings are effectively false positives that were already mitigated:

1. **`fgetc` in a loop (CWE-120, CWE-20)** — The `flush_stdin()` function reads characters into an `int` and discards them; nothing is written to a buffer, so no overflow is possible. The `guard` counter with `FLUSH_GUARD_LIMIT` additionally protects against unbounded looping on a pathological stream (e.g., a pipe that never delivers `\n` or `EOF`).

2. **Statically-sized `char` array (CWE-119/CWE-120)** — The `input` buffer is used safely:
   - `fgets(input, (int)sizeof(input), stdin)` bounds the read to the buffer size.
   - The buffer is zeroed with `memset` before use and explicitly NUL-terminated at `sizeof(input) - 1U` as a defensive measure.
   - Length is computed with the bounded `safe_strlen` rather than `strlen`.
   - When input exceeds the buffer (no trailing `\n` and length equals `sizeof(input) - 1U`), leftover characters are drained via `flush_stdin()` so they can't leak into subsequent reads.

Other good practices already present and retained: `NULL` checks in helper functions, checking the `fgets` return value, casting `sizeof` to `int` for the `fgets` parameter, flushing `stdout` before reading, and explicit `(void)` casts for ignored return values (MISRA-style compliance). No functional changes were required — only the comments were removed as requested.
