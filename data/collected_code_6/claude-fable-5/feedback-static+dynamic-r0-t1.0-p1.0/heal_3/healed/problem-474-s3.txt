```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 256

size_t replace_char(char *str, size_t max_len, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    static const char original[] = "hello world, hello everyone";
    char *buffer;
    size_t len;
    size_t buffer_size;
    size_t replaced;

    len = strnlen(original, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not terminated\n");
        return EXIT_FAILURE;
    }

    if (len > SIZE_MAX - 1) {
        fprintf(stderr, "Length overflow\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;

    buffer = malloc(buffer_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    memcpy(buffer, original, len);
    buffer[len] = '\0';

    printf("Before: %s\n", buffer);

    replaced = replace_char(buffer, buffer_size, 'l', 'L');

    printf("After:  %s\n", buffer);
    printf("Replaced %zu occurrence(s)\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Removed the always-false condition** (`len >= buffer_size`): Since `buffer_size = len + 1`, the check `len >= buffer_size` can never be true. Dead code flagged by `knownConditionTrueFalse` was eliminated.

2. **Added an explicit overflow guard for `len + 1`**: Although `len < MAX_INPUT_LEN` already guarantees no wraparound, the check `len > SIZE_MAX - 1` makes the arithmetic provably safe and silences integer-overflow concerns around the `buffer_size` assignment. Added `<stdint.h>` for `SIZE_MAX`.

3. **`memcpy` (CWE-120) is safe by construction**: The destination is allocated with exactly `len + 1` bytes, exactly `len` bytes are copied, and the terminator is written at index `len` — the last valid byte of the allocation. No runtime check is needed because the bound is enforced at allocation time.

4. **The `malloc` NULL check is retained**, as the analyzer's "assuming condition is false" note is merely a path assumption, not a defect — the check is correct and necessary.

The remaining structure (bounded `strnlen`, bounded traversal in `replace_char` with a NULL guard, freeing and nulling the pointer) was already sound and preserved as-is.
