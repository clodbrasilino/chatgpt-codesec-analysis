#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <stdlib.h>
#include <ctype.h>

uint64_t nonagonal_number(uint64_t n) {
    if (n > UINT64_MAX / 7) {
        return 0;
    }
    uint64_t temp = 7 * n;
    if (temp - 5 > UINT64_MAX / n) {
        return 0;
    }
    return n * (temp - 5) / 2;
}

int main(void) {
    char buffer[32];
    char *endptr;
    uint64_t n;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return 1;
    }

    if (buffer[0] == '\0' || buffer[0] == '\n') {
        return 1;
    }

    errno = 0;
    n = strtoull(buffer, &endptr, 10);

    if (errno == ERANGE || endptr == buffer) {
        return 1;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 1;
    }

    if (n > 72340172838076673ULL) {
        return 1;
    }

    printf("%" PRIu64 "\n", nonagonal_number(n));
    return 0;
}
