```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

static int binary_to_decimal(const char *binary, size_t max_len, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_line(int *was_truncated)
{
    int ch;
    unsigned long guard = 0UL;

    if (was_truncated == NULL) {
        return -1;
    }

    *was_truncated = 0;

    while ((ch = getchar()) != '\n' && ch != EOF) {
        *was_truncated = 1;
        guard++;
        if (guard == ULONG_MAX) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error: failed to flush output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int truncated = 0;

        if (flush_stdin_line(&truncated) != 0) {
            fprintf(stderr, "Error: internal failure.\n");
            return EXIT_FAILURE;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** — In the original, `getchar()` in a loop was already safe since no buffer is written. In the rewrite I added a `guard` counter as a defensive upper bound so the drain loop cannot spin unbounded on a pathological stream, making the fix explicit rather than implicit.

2. **Flawfinder `char` static array warning (CWE-119/CWE-120)** — The fixed-size `input[INPUT_SIZE]` buffer is protected by:
   - `fgets` with `sizeof(input)` limiting the read length,
   - forced NUL termination at `input[sizeof(input) - 1U]`,
   - `strnlen` (never `strlen`) bounded by `sizeof(input)` everywhere,
   - explicit truncation detection with a clear error path,
   - `binary_to_decimal` re-validating length bounds (`len >= max_len` rejected) and rejecting a zero `max_len`.

3. **Integer overflow hardening** — the `value > (ULLONG_MAX >> 1)` check before each shift prevents silent overflow for inputs longer than 64 significant bits, which is preserved.

4. **Minor robustness addition** — `fflush(stdout)` after the prompt guarantees the prompt is visible before blocking on `fgets`, with its return value checked per strict error-handling style.

The original code was already well-hardened; the rewrite keeps all validation and adds the loop guard and output flush as belt-and-suspenders measures.
