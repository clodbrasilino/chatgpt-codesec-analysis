#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>
#include <ctype.h>

long long sum_even_factors(long long n) {
    long long sum = 0;
    long long abs_n;
    long long i;
    
    if (n == 0) {
        return 0;
    }
    
    abs_n = (n < 0) ? -n : n;
    
    for (i = 2; i <= abs_n / 2; i += 2) {
        if (abs_n % i == 0) {
            sum += i;
        }
    }
    
    if (abs_n % 2 == 0) {
        sum += abs_n;
    }
    
    return sum;
}

int main(void) {
    char input[128];
    char *endptr;
    long long num;
    long long result;
    size_t len;
    int truncated = 0;
    
    if (fgets(input, sizeof(input), stdin) == NULL) {
        return EXIT_FAILURE;
    }
    
    len = strnlen(input, sizeof(input));
    
    if (len == sizeof(input) && input[len - 1] != '\n') {
        truncated = 1;
    }
    
    if (truncated) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
        return EXIT_FAILURE;
    }
    
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }
    
    errno = 0;
    num = strtoll(input, &endptr, 10);
    
    if (errno == ERANGE || endptr == input) {
        return EXIT_FAILURE;
    }
    
    while (*endptr != '\0') {
        if (!isspace((unsigned char)*endptr)) {
            return EXIT_FAILURE;
        }
        endptr++;
    }
    
    if (num == LLONG_MIN) {
        return EXIT_FAILURE;
    }
    
    result = sum_even_factors(num);
    printf("%lld\n", result);
    
    return EXIT_SUCCESS;
}
