#include <stdio.h>
#include <ctype.h>

int is_vowel(char c) {
    c = tolower(c);
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

int count_chars_with_vowel_neighbors(const char *str, int max_len) {
    if (str == NULL || max_len < 3) {
        return 0;
    }
    
    int count = 0;
    int i = 1;
    
    while (i < max_len - 1 && str[i] != '\0' && str[i + 1] != '\0') {
        if (is_vowel(str[i - 1]) && is_vowel(str[i + 1])) {
            count++;
        }
        i++;
    }
    
    return count;
}

int main(void) {
    const char *test_strings[] = {
        "hello world",
        "a test string with vowels",
        "abcdefg",
        "aeiou",
        "",
        NULL
    };
    
    for (int i = 0; test_strings[i] != NULL; i++) {
        int len = 0;
        const char *ptr = test_strings[i];
        while (*ptr != '\0') {
            len++;
            ptr++;
        }
        printf("String: \"%s\" -> Count: %d\n", 
               test_strings[i], 
               count_chars_with_vowel_neighbors(test_strings[i], len));
    }
    
    return 0;
}
