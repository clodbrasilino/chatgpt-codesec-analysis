#include <stdio.h>
#include <math.h>

typedef struct {
    double x, y;
} Point;

double ellipse_area(double a, double b) {
    return M_PI * a * b;
}

int is_valid_triangle(Point p1, Point p2, Point p3) {
    return (p1.x != p2.x || p1.y != p2.y) && (p1.x != p3.x || p1.y != p3.y) && (p2.x != p3.x || p2.y != p3.y);
}

double triangle_area(Point p1, Point p2, Point p3) {
    return fabs(0.5 * (p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y)));
}

void inscribed_triangle_in_ellipse(double a, double b, Point *p1, Point *p2, Point *p3) {
    *p1 = (Point){-a, 0.0};
    *p2 = (Point){a, 0.0};
    *p3 = (Point){0.0, b};
}

int main() {
    double a = 5.0, b = 3.0;
    Point p1, p2, p3;

    inscribed_triangle_in_ellipse(a, b, &p1, &p2, &p3);

    if (is_valid_triangle(p1, p2, p3)) {
        double area = triangle_area(p1, p2, p3);
        printf("Largest inscribed triangle area: %.2f\n", area);
    } else {
        printf("Invalid triangle points.\n");
    }

    return 0;
}
