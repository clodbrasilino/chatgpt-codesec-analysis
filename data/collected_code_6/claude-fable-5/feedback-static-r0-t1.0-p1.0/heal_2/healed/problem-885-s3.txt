```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096

bool are_isomorphic(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    int map1[256];
    int map2[256];
    size_t i;

    if (s1 == NULL || s2 == NULL) {
        return false;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN);
    len2 = strnlen(s2, MAX_INPUT_LEN);

    if (len1 >= MAX_INPUT_LEN || len2 >= MAX_INPUT_LEN) {
        return false;
    }

    if (len1 != len2) {
        return false;
    }

    for (i = 0; i < 256; i++) {
        map1[i] = -1;
        map2[i] = -1;
    }

    for (i = 0; i < len1; i++) {
        const unsigned char c1 = (unsigned char)s1[i];
        const unsigned char c2 = (unsigned char)s2[i];

        if (map1[c1] == -1 && map2[c2] == -1) {
            map1[c1] = (int)c2;
            map2[c2] = (int)c1;
        } else if (map1[c1] != (int)c2 || map2[c2] != (int)c1) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *test1a = "egg";
    const char *test1b = "add";
    const char *test2a = "foo";
    const char *test2b = "bar";
    const char *test3a = "paper";
    const char *test3b = "title";

    if (are_isomorphic(test1a, test1b)) {
        printf("\"%s\" and \"%s\" are isomorphic\n", test1a, test1b);
    } else {
        printf("\"%s\" and \"%s\" are not isomorphic\n", test1a, test1b);
    }

    if (are_isomorphic(test2a, test2b)) {
        printf("\"%s\" and \"%s\" are isomorphic\n", test2a, test2b);
    } else {
        printf("\"%s\" and \"%s\" are not isomorphic\n", test2a, test2b);
    }

    if (are_isomorphic(test3a, test3b)) {
        printf("\"%s\" and \"%s\" are isomorphic\n", test3a, test3b);
    } else {
        printf("\"%s\" and \"%s\" are not isomorphic\n", test3a, test3b);
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

- Replaced both `strlen` calls with `strnlen` bounded by `MAX_INPUT_LEN`, which addresses the CWE-126 over-read risk flagged by Flawfinder: if a string is not properly NUL-terminated, the scan stops at the bound instead of reading past the buffer.
- Added a guard that rejects inputs reaching the bound (`len >= MAX_INPUT_LEN`), since such strings are either truncated or unterminated, and treating them as valid could mask the original defect.
- The existing defensive measures were already sound and preserved: NULL-pointer checks before any dereference, `unsigned char` casts before indexing the 256-entry maps (preventing negative-index UB on platforms where `char` is signed), and the bidirectional mapping ensuring a true one-to-one isomorphism check.

Note that `strnlen` is POSIX (and C23's `strnlen_s` via Annex K is an alternative); if strict C89/C99 portability is required, a small local bounded-length helper can replace it.
