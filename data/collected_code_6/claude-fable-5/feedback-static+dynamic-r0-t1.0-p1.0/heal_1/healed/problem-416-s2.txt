```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>
#include <ctype.h>

static long long max2(long long a, long long b)
{
    return (a > b) ? a : b;
}

static long long maxSum(long long n, long long *memo, long long limit)
{
    long long divided;

    if (n <= 0) {
        return 0;
    }

    if (n < limit && memo[n] != -1) {
        return memo[n];
    }

    divided = maxSum(n / 2, memo, limit) +
              maxSum(n / 3, memo, limit) +
              maxSum(n / 4, memo, limit);

    divided = max2(n, divided);

    if (n < limit) {
        memo[n] = divided;
    }

    return divided;
}

static int findMaxSum(long long n, long long *result)
{
    long long *memo;
    long long limit;
    long long i;

    if (n < 0 || result == NULL) {
        return -1;
    }

    if (n >= 999999) {
        limit = 1000000;
    } else {
        limit = n + 1;
    }

    if (limit <= 0) {
        return -1;
    }

    if ((unsigned long long)limit > SIZE_MAX / sizeof(long long)) {
        return -1;
    }

    memo = malloc((size_t)limit * sizeof(long long));
    if (memo == NULL) {
        return -1;
    }

    for (i = 0; i < limit; i++) {
        memo[i] = -1;
    }

    *result = maxSum(n, memo, limit);

    free(memo);
    return 0;
}

static int readNonNegative(long long *out)
{
    char buffer[128];
    char *endptr;
    long long value;

    if (out == NULL) {
        return -1;
    }

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return -1;
    }

    if (strchr(buffer, '\n') == NULL && !feof(stdin)) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
        return -1;
    }

    errno = 0;
    endptr = NULL;
    value = strtoll(buffer, &endptr, 10);

    if (endptr == buffer) {
        return -1;
    }

    if (errno == ERANGE) {
        return -1;
    }

    while (*endptr != '\0') {
        if (!isspace((unsigned char)*endptr)) {
            return -1;
        }
        endptr++;
    }

    if (value < 0) {
        return -1;
    }

    *out = value;
    return 0;
}

int main(void)
{
    long long n;
    long long result;

    printf("Enter a non-negative number: ");
    fflush(stdout);

    if (readNonNegative(&n) != 0) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (findMaxSum(n, &result) != 0) {
        fprintf(stderr, "Computation failed\n");
        return EXIT_FAILURE;
    }

    printf("Maximum sum for %lld is %lld\n", n, result);

    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weakness:

1. **Signed integer overflow eliminated**: The original `limit = n + 1` invokes undefined behavior when `n == LLONG_MAX`. The overflowed (negative) `limit` was then cast to `size_t`, producing an enormous allocation request — the likely source of the SIGABRT (signal 6) found by the fuzzer. The rewrite clamps `limit` *before* the increment (`n >= 999999`), so the addition can never overflow.

2. **Allocation-size overflow guard**: An explicit check against `SIZE_MAX / sizeof(long long)` prevents multiplication wraparound in the `malloc` call, which could otherwise yield an undersized buffer followed by out-of-bounds writes in the initialization loop.

3. **Safe input parsing**: `scanf("%lld", ...)` has undefined behavior when the input value overflows `long long` — another plausible trigger for the fuzzer crash. It has been replaced with `fgets` + `strtoll`, with explicit `ERANGE` detection, trailing-garbage rejection, and oversized-line rejection.

4. **Defensive `limit <= 0` check** in `findMaxSum` as a belt-and-suspenders guard, ensuring the memoization bounds check `n < limit` can never index with a negative or unallocated region.

The recursion itself is bounded (depth ≈ log₂(n) ≤ 63 for 64-bit inputs), so no stack-depth mitigation is required.
