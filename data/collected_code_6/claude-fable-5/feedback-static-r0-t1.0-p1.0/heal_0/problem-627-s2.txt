```c
#include <stdio.h>
#include <stdlib.h>

int find_smallest_missing(const int *arr, size_t n, int *result)
{
    unsigned char *seen;
    size_t i;
    int candidate;

    if (arr == NULL || result == NULL || n == 0) {
        return -1;
    }

    seen = calloc(n + 2, sizeof(unsigned char));
    if (seen == NULL) {
        return -1;
    }

    for (i = 0; i < n; i++) {
        if (arr[i] >= 0 && (size_t)arr[i] <= n) {
            seen[arr[i]] = 1;
        }
    }

    candidate = 0;
    for (i = 0; i <= n; i++) {
        if (seen[i] == 0) {
            candidate = (int)i;
            break;
        }
    }

    free(seen);
    *result = candidate;
    return 0;
}

int main(void)
{
    int arr[] = {0, 1, 2, 6, 9, 11, 15};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int missing;

    if (find_smallest_missing(arr, n, &missing) != 0) {
        fprintf(stderr, "Error: could not compute smallest missing number\n");
        return EXIT_FAILURE;
    }

    if (printf("Smallest missing number: %d\n", missing) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

This program finds the smallest missing non-negative integer from an array. Key safety measures taken:

- **CWE-664 avoidance**: The dynamically allocated `seen` buffer is freed exactly once before the function returns, on every path where it was successfully allocated.
- **CWE-703 avoidance**: The return value of `calloc` is checked before use, `printf` failure is checked, `NULL` pointers and zero-length arrays are rejected, and the function reports errors through a status code rather than a sentinel value.
- **CWE-710 avoidance**: The code uses `size_t` for sizes and indices, casts explicitly between signed and unsigned where required, uses `main(void)` with standard exit codes, and avoids out-of-bounds writes by bounds-checking array values before indexing into `seen`.
