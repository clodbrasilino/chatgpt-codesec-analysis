```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_KEY_LEN 1024U

typedef struct InnerDict {
    char *key;
    int value;
} InnerDict;

typedef struct NestedDict {
    char *key;
    InnerDict inner;
} NestedDict;

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }
    len = strnlen(src, MAX_KEY_LEN);
    if (len >= MAX_KEY_LEN) {
        return NULL;
    }
    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static void free_nested_dict(NestedDict *dict, size_t count)
{
    size_t i;

    if (dict == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(dict[i].key);
        free(dict[i].inner.key);
    }
    free(dict);
}

static NestedDict *convert_lists_to_nested_dict(const char *const *outer_keys,
                                                const char *const *inner_keys,
                                                const int *values,
                                                size_t count)
{
    NestedDict *dict;
    size_t i;

    if (outer_keys == NULL || inner_keys == NULL || values == NULL || count == 0U) {
        return NULL;
    }
    if (count > SIZE_MAX / sizeof(NestedDict)) {
        return NULL;
    }
    dict = calloc(count, sizeof(NestedDict));
    if (dict == NULL) {
        return NULL;
    }
    for (i = 0; i < count; i++) {
        dict[i].key = duplicate_string(outer_keys[i]);
        dict[i].inner.key = duplicate_string(inner_keys[i]);
        dict[i].inner.value = values[i];
        if (dict[i].key == NULL || dict[i].inner.key == NULL) {
            free_nested_dict(dict, i + 1U);
            return NULL;
        }
    }
    return dict;
}

static void print_nested_dict(const NestedDict *dict, size_t count)
{
    size_t i;

    if (dict == NULL) {
        return;
    }
    printf("{");
    for (i = 0; i < count; i++) {
        printf("'%s': {'%s': %d}", dict[i].key, dict[i].inner.key, dict[i].inner.value);
        if (i + 1U < count) {
            printf(", ");
        }
    }
    printf("}\n");
}

int main(void)
{
    const char *names[] = { "S001", "S002", "S003" };
    const char *students[] = { "Adina Park", "Leyton Marsh", "Duncan Boyle" };
    int marks[] = { 85, 98, 89 };
    size_t count = sizeof(marks) / sizeof(marks[0]);
    NestedDict *dict;

    dict = convert_lists_to_nested_dict(names, students, marks, count);
    if (dict == NULL) {
        fprintf(stderr, "Failed to build nested dictionary\n");
        return EXIT_FAILURE;
    }
    print_nested_dict(dict, count);
    free_nested_dict(dict, count);
    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Added `<stdint.h>`** — the original code used `SIZE_MAX` without including its defining header, which would fail to compile on conforming implementations.

2. **Replaced `strlen` with `strnlen`** (addressing the CWE-126 over-read concern) — a `MAX_KEY_LEN` bound guarantees the scan terminates even if the input isn't NUL-terminated, and inputs at or beyond the bound are rejected rather than truncated silently into an unterminated copy.

3. **Hardened the `memcpy` pattern** (addressing the CWE-120 concern) — the destination is allocated as `len + 1U`, exactly `len` bytes are copied, and the terminator is written explicitly. The copy size is now provably bounded by `MAX_KEY_LEN`, so the destination can always hold the source data.

The rest of the logic (overflow guard before `calloc`, partial-cleanup on allocation failure via `free_nested_dict(dict, i + 1U)`, NULL-safe frees) was already sound and is preserved as-is.
