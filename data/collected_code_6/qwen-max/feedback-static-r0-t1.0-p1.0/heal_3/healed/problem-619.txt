#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* move_numbers(char *str) {
    size_t len = strlen(str);
    char *numStr = (char *)malloc(len + 1);
    if (numStr == NULL) {
        return NULL;
    }
    int numIndex = 0;
    
    for (int i = 0, j = 0; i < (int)len; ++i) {
        if (isdigit((unsigned char)str[i])) {
            numStr[numIndex++] = str[i];
        } else {
            str[j++] = str[i];
        }
    }
    numIndex--;
    for (int i = (int)len - numIndex - 1; numIndex >= 0; --i) {
        str[i] = numStr[numIndex--];
    }
    str[len] = '\0';
    free(numStr);
    return str;
}

int main() {
    char str[] = "Hello123World456";
    char *result = move_numbers(str);
    if (result != NULL) {
        printf("%s\n", result);
    } else {
        fprintf(stderr, "Memory allocation failed\n");
    }
    return 0;
}
