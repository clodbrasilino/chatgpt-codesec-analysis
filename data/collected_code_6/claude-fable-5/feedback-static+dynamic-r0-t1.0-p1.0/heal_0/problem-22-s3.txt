```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool find_first_duplicate(const int *arr, size_t len, int *result)
{
    size_t i;
    size_t j;

    if ((arr == NULL) || (result == NULL) || (len < 2U)) {
        return false;
    }

    for (i = 1U; i < len; i++) {
        for (j = 0U; j < i; j++) {
            if (arr[i] == arr[j]) {
                *result = arr[i];
                return true;
            }
        }
    }

    return false;
}

int main(void)
{
    int numbers[] = {2, 5, 1, 5, 3, 2};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int duplicate = 0;

    if (find_first_duplicate(numbers, count, &duplicate)) {
        if (printf("First duplicate: %d\n", duplicate) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No duplicate found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```
