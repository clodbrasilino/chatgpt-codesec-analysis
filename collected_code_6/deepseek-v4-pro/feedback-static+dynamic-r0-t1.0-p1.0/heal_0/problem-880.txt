#include <stdio.h>
#include <math.h>

int quadratic_solutions(double a, double b, double c) {
    if (fabs(a) < 1e-12) {
        if (fabs(b) < 1e-12) {
            return fabs(c) < 1e-12 ? -1 : 0;
        }
        return 1;
    }

    double discriminant = b * b - 4.0 * a * c;

    if (discriminant > 1e-12) {
        return 2;
    } else if (fabs(discriminant) <= 1e-12) {
        return 1;
    } else {
        return 0;
    }
}

int main(void) {
    double a, b, c;

    printf("Enter coefficients a, b, c: ");
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        printf("Invalid input\n");
        return 1;
    }

    int solutions = quadratic_solutions(a, b, c);

    if (solutions == -1) {
        printf("Infinite solutions\n");
    } else {
        printf("Number of real solutions: %d\n", solutions);
    }

    return 0;
}
