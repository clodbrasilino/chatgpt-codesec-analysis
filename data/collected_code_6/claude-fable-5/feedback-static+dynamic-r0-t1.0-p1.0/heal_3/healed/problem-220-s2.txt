```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

size_t replace_chars(char *str, size_t n)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0' && count < n; i++) {
        if (str[i] == ' ' || str[i] == ',' || str[i] == '.') {
            str[i] = ':';
            count++;
        }
    }

    return count;
}

int main(void)
{
    static const char source[] = "Hello world, this is a test. Replace them all.";
    char *buffer;
    size_t buf_size;
    size_t len;
    size_t replaced;

    buf_size = sizeof(source);
    len = strnlen(source, buf_size - 1U);

    buffer = malloc(buf_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (len >= buf_size) {
        fprintf(stderr, "Source length exceeds buffer capacity\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (len + 1U > buf_size) {
        fprintf(stderr, "Copy would exceed destination buffer\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    memcpy(buffer, source, len);
    buffer[len] = '\0';

    replaced = replace_chars(buffer, 5U);

    if (printf("Original: %s\n", source) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }
    if (printf("Modified: %s\n", buffer) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }
    if (printf("Replacements made: %zu\n", replaced) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive, but the flawfinder warning on `memcpy` (CWE-120) flagged that the copy operation itself had no explicit guarantee that the destination could hold `len` bytes plus the null terminator at the point of the call.

Key points about the rewrite:

1. **memcpy bounds guarantee** — An explicit check `len + 1U > buf_size` is added immediately before `memcpy`, guaranteeing the destination buffer can hold both the copied bytes and the null terminator written at `buffer[len]`. This directly addresses the CWE-120 concern rather than relying on implicit reasoning about `strnlen`'s cap.

2. **Existing safe practices preserved** — `strnlen` bounded by `buf_size - 1U` prevents over-reads, `malloc` return is checked, all error paths free the buffer, `printf` return values are validated, and the pointer is nulled after `free` to avoid dangling-pointer reuse.

3. **No overflow risk in the check** — Since `len` is capped at `buf_size - 1U` by `strnlen`, `len + 1U` cannot wrap around, making the guard arithmetically safe.

The program remains functionally identical: it copies the string, replaces up to 5 occurrences of space, comma, or period with `:`, and reports the results.
