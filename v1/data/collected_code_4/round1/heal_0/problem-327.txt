#include <stdio.h>

int isTriangleValid(int a, int b, int c) {
    if (a + b <= c || a + c <= b || b + c <= a) {
        return 0;
    }
    return 1;
}

int isIsosceles(int a, int b, int c) {
    if (!isTriangleValid(a, b, c)) {
        return -1;
    }

    if (a == b || b == c || c == a) {
        return 1;
    }

    return 0;
}

int main() {
    int a, b, c;
    printf("Enter the sides of the triangle:\n");
    scanf("%d%d%d", &a, &b, &c);

    int result = isIsosceles(a, b, c);

    if(result == -1) {
        printf("The inputs do not form a valid triangle.\n");
    } else if (result == 1) {
        printf("The triangle is isosceles.\n");
    } else {
        printf("The triangle is not isosceles.\n");
    }

    return 0;
}
