```c
#include <stdio.h>
#include <stdlib.h>

double parallelogram_area(double base, double height);

double parallelogram_area(double base, double height)
{
    return base * height;
}

int main(void)
{
    double base;
    double height;
    double area;

    printf("Enter the base of the parallelogram: ");
    if (scanf("%lf", &base) != 1) {
        fprintf(stderr, "Error: invalid input for base.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the height of the parallelogram: ");
    if (scanf("%lf", &height) != 1) {
        fprintf(stderr, "Error: invalid input for height.\n");
        return EXIT_FAILURE;
    }

    if (base < 0.0 || height < 0.0) {
        fprintf(stderr, "Error: base and height must be non-negative.\n");
        return EXIT_FAILURE;
    }

    area = parallelogram_area(base, height);
    printf("Area of the parallelogram: %f\n", area);

    return EXIT_SUCCESS;
}
```
