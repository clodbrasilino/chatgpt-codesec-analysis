#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *snake_to_camel(const char *snake) {
    if (snake == NULL) {
        return NULL;
    }

    size_t len = 0;
    while (snake[len] != '\0') {
        len++;
        if (len > 65536) {
            return NULL;
        }
    }

    if (len == 0) {
        char *empty = malloc(1);
        if (empty != NULL) {
            empty[0] = '\0';
        }
        return empty;
    }

    char *result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t dst_pos = 0;
    int capitalize = 0;
    int first_char = 1;

    for (size_t i = 0; snake[i] != '\0'; i++) {
        if (snake[i] == '_') {
            capitalize = 1;
        } else {
            if (capitalize) {
                result[dst_pos++] = toupper((unsigned char)snake[i]);
                capitalize = 0;
                first_char = 0;
            } else {
                if (first_char) {
                    result[dst_pos++] = tolower((unsigned char)snake[i]);
                    first_char = 0;
                } else {
                    result[dst_pos++] = snake[i];
                }
            }
        }
    }

    result[dst_pos] = '\0';

    return result;
}

int main(void) {
    const char *test_cases[] = {
        "android_tv",
        "google_pixel",
        "apple_watch",
        "hello_world",
        "snake_case_string",
        "convert_this_to_camel",
        "already_camel",
        "multiple__underscores",
        "trailing_underscore_",
        "_leading_underscore",
        "single",
        "",
        NULL
    };

    for (int i = 0; test_cases[i] != NULL; i++) {
        char *camel = snake_to_camel(test_cases[i]);
        if (camel != NULL) {
            printf("%s -> %s\n", test_cases[i], camel);
            free(camel);
        } else {
            printf("%s -> (conversion failed)\n", test_cases[i]);
        }
    }

    return 0;
}
