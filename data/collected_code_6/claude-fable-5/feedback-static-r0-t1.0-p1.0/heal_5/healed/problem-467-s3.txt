```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define OCTAL_TEMP_SIZE (((sizeof(unsigned long) * CHAR_BIT + 2) / 3) + 1)
#define OCTAL_BUFFER_SIZE (((sizeof(unsigned long) * CHAR_BIT + 2) / 3) + 3)

int decimal_to_octal(long decimal, char *buffer, size_t buffer_size);

int decimal_to_octal(long decimal, char *buffer, size_t buffer_size)
{
    char temp[OCTAL_TEMP_SIZE];
    size_t index = 0;
    size_t i;
    int negative = 0;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal < 0) {
        negative = 1;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    if (value == 0UL) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0UL) {
        if (index >= sizeof(temp)) {
            return -1;
        }
        temp[index] = (char)('0' + (value % 8UL));
        value /= 8UL;
        index++;
    }

    if (index + (size_t)negative + 1 > buffer_size) {
        return -1;
    }

    i = 0;
    if (negative == 1) {
        buffer[i] = '-';
        i++;
    }

    while (index > 0) {
        index--;
        if (i >= buffer_size - 1) {
            return -1;
        }
        buffer[i] = temp[index];
        i++;
    }
    buffer[i] = '\0';

    return 0;
}

int main(void)
{
    long decimal;
    char octal[OCTAL_BUFFER_SIZE];
    int scan_result;

    printf("Enter a decimal number: ");
    scan_result = scanf("%ld", &decimal);

    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_octal(decimal, octal, sizeof(octal)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Octal representation: %s\n", octal);

    return EXIT_SUCCESS;
}
```

Analysis of the original code and the fixes applied:

**Weaknesses identified by Flawfinder (CWE-119/CWE-120):**

1. **`temp[OCTAL_TEMP_SIZE]` in `decimal_to_octal`** — The original code already mitigated this correctly: the buffer is sized to hold the maximum number of octal digits for an `unsigned long` (computed portably via `sizeof(unsigned long) * CHAR_BIT`), and the write loop performs an explicit bounds check (`index >= sizeof(temp)`) before every store. This mitigation is preserved.

2. **`octal[OCTAL_BUFFER_SIZE]` in `main`** — Also correctly mitigated in the original: the caller passes `sizeof(octal)` so the callee knows the real capacity, and the function validates `index + negative + 1 > buffer_size` before writing. The buffer size accounts for maximum digits, a sign character, and the NUL terminator.

**Additional hardening added in the rewrite:**

- A defensive bounds check (`i >= buffer_size - 1`) inside the copy loop, guaranteeing that no write to `buffer` can exceed its capacity even if the earlier arithmetic check were ever modified incorrectly. This makes the write loop independently safe rather than relying solely on the pre-computed size check.

**Existing correct practices worth noting (retained):**

- The `LONG_MIN` negation hazard is handled properly: `-(decimal + 1) + 1UL` avoids the undefined behavior of negating `LONG_MIN` directly.
- `scanf` uses `%ld` with a validated return value — safe for numeric conversion (no unbounded string read).
- `buffer == NULL` and minimum-size checks guard the API against misuse.
- All writes are NUL-terminated, so `printf("%s", ...)` never reads past the buffer.

The code compiles cleanly with `-Wall -Wextra -pedantic` and is portable across platforms with differing `long` widths (32-bit and 64-bit).
