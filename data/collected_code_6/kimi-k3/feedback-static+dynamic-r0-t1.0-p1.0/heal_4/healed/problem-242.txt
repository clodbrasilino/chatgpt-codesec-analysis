```c
#include <stdio.h>
#include <string.h>
#include <stddef.h>
#include <stdlib.h>

size_t count_characters(const char *str, size_t max_len)
{
    if (str == NULL)
    {
        return 0;
    }
    return strnlen(str, max_len);
}

int main(void)
{
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t characters_read;

    printf("Enter a string: ");

    characters_read = getline(&buffer, &buffer_size, stdin);

    if (characters_read == -1)
    {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return 1;
    }

    if (characters_read > 0 && buffer[characters_read - 1] == '\n')
    {
        buffer[characters_read - 1] = '\0';
        characters_read--;
    }

    printf("Total characters: %zu\n", count_characters(buffer, (size_t)characters_read + 1));

    free(buffer);
    return 0;
}
```
