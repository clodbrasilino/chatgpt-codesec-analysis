#include <stdio.h>
#include <stdlib.h>
#include <regex.h>
#include <string.h>

void add_spaces_between_capital_words(char *str, size_t str_len) {
    regex_t regex;
    regmatch_t matches[2];
    char pattern[] = "([A-Z][^A-Z]*)";
    char replacement[] = " \\1";
    char result[512] = "";
    int start = 0;

    if (regcomp(&regex, pattern, REG_EXTENDED | REG_NEWLINE) != 0) {
        return;
    }

    while (regexec(&regex, str + start, 2, matches, 0) == 0) {
        int end = matches[0].rm_so + start;
        size_t len1 = (end - start < sizeof(result) - 1) ? end - start : sizeof(result) - 1;
        strncat(result, str + start, len1);
        size_t len2 = (sizeof(replacement) - 2 < sizeof(result) - strlen(result) - 1) ? sizeof(replacement) - 2 : sizeof(result) - strlen(result) - 1;
        strncat(result, replacement, len2);
        start = matches[0].rm_eo + start;
    }

    size_t remaining_len = (str_len - start - 1 < sizeof(result) - strlen(result) - 1) ? str_len - start - 1 : sizeof(result) - strlen(result) - 1;
    strncat(result, str + start, remaining_len);
    snprintf(str, str_len, "%s", result);

    regfree(&regex);
}

int main() {
    char str[] = "HelloWorldThisIsATest";
    add_spaces_between_capital_words(str, sizeof(str));
    printf("%s\n", str);
    return 0;
}
