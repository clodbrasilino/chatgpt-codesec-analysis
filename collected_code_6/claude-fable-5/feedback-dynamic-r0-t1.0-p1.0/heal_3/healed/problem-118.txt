#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node {
    char *word;
    struct node *next;
};

static struct node *create_node(const char *start, size_t len)
{
    struct node *n = malloc(sizeof(struct node));
    if (n == NULL) {
        return NULL;
    }
    n->word = malloc(len + 1);
    if (n->word == NULL) {
        free(n);
        return NULL;
    }
    memcpy(n->word, start, len);
    n->word[len] = '\0';
    n->next = NULL;
    return n;
}

static void free_list(struct node *head)
{
    while (head != NULL) {
        struct node *tmp = head;
        head = head->next;
        free(tmp->word);
        free(tmp);
    }
}

static struct node *string_to_word_list(const char *str)
{
    struct node *head = NULL;
    struct node *tail = NULL;
    size_t i = 0;

    if (str == NULL) {
        return NULL;
    }

    while (str[i] != '\0') {
        while (str[i] == ' ' || str[i] == '\t' || str[i] == '\n' || str[i] == '\r') {
            i++;
        }
        if (str[i] == '\0') {
            break;
        }
        {
            size_t start = i;
            while (str[i] != '\0' && str[i] != ' ' && str[i] != '\t' && str[i] != '\n' && str[i] != '\r') {
                i++;
            }
            {
                struct node *n = create_node(&str[start], i - start);
                if (n == NULL) {
                    free_list(head);
                    return NULL;
                }
                if (head == NULL) {
                    head = n;
                    tail = n;
                } else {
                    tail->next = n;
                    tail = n;
                }
            }
        }
    }

    return head;
}

static void print_list(const struct node *head)
{
    const struct node *cur = head;

    (void)printf("[");
    while (cur != NULL) {
        (void)printf("'%s'", cur->word);
        if (cur->next != NULL) {
            (void)printf(", ");
        }
        cur = cur->next;
    }
    (void)printf("]\n");
    (void)fflush(stdout);
}

static char *read_all_input(void)
{
    size_t cap = 4096;
    size_t len = 0;
    char *buf = malloc(cap);

    if (buf == NULL) {
        return NULL;
    }

    for (;;) {
        int c = fgetc(stdin);
        if (c == EOF) {
            break;
        }
        if (len + 1 >= cap) {
            size_t new_cap = cap * 2;
            char *tmp = realloc(buf, new_cap);
            if (tmp == NULL) {
                free(buf);
                return NULL;
            }
            buf = tmp;
            cap = new_cap;
        }
        buf[len++] = (char)c;
    }

    buf[len] = '\0';
    return buf;
}

int main(void)
{
    char *input;
    struct node *list;

    input = read_all_input();
    if (input == NULL) {
        (void)printf("[]\n");
        (void)fflush(stdout);
        return EXIT_SUCCESS;
    }

    list = string_to_word_list(input);

    print_list(list);

    free_list(list);
    free(input);

    return EXIT_SUCCESS;
}

**Analysis of the original code and the fixes applied:**

The core parsing, list construction, and printing logic in the original code was correct. The most likely cause of the `<no output>` failures was the input-reading strategy in `main`:

1. **Single `fgets` call limitation** — The original code read only one line via `fgets`. If the test harness delivered input without a trailing newline in a way that interacted badly with buffering, or piped input that arrived differently than expected, the program could bail out early and print nothing before the harness captured output. The rewrite reads **all of stdin until EOF** character by character, which is robust regardless of how the harness delivers the input (with or without newlines, in multiple chunks, etc.).

2. **Fixed-size buffer** — The original `char buffer[4096]` silently truncated longer inputs. The rewrite uses a dynamically grown buffer (`realloc` doubling strategy), so arbitrarily long input is handled correctly with proper cleanup on allocation failure.

3. **Memory hygiene preserved** — All existing strengths were kept: `create_node` checks both allocations and cleans up partially allocated nodes, `string_to_word_list` frees the partial list on failure, and `main` frees both the list and the input buffer before exit.

4. **Output behavior unchanged** — The output format (`['word1', 'word2']`) and the explicit `fflush(stdout)` calls are retained, so the expected test output format is matched exactly, and output is guaranteed to be flushed even if the harness terminates the process abruptly.
