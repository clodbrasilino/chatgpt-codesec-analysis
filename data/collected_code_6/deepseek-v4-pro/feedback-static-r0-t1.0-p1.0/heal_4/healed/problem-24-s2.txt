#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

int binary_to_decimal(const char *binary, size_t max_len) {
    int decimal = 0;
    size_t i = 0;
    
    if (binary == NULL) {
        return -1;
    }
    
    while (i < max_len && binary[i] != '\0') {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        
        decimal = decimal * 2 + (binary[i] - '0');
        i++;
    }
    
    if (i == 0) {
        return -1;
    }
    
    return decimal;
}

int main(void) {
    char input[65] = {0};
    
    printf("Enter a binary number: ");
    
    if (fgets(input, sizeof(input), stdin) == NULL) {
        printf("Error reading input\n");
        return 1;
    }
    
    bool newline_found = false;
    size_t len = 0;
    while (len < sizeof(input) - 1 && input[len] != '\0') {
        if (input[len] == '\n') {
            input[len] = '\0';
            newline_found = true;
            break;
        }
        len++;
    }
    
    if (!newline_found && len >= sizeof(input) - 1) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
        }
        printf("Input too long\n");
        return 1;
    }
    
    int result = binary_to_decimal(input, len);
    
    if (result == -1) {
        printf("Invalid binary number\n");
        return 1;
    }
    
    printf("Decimal equivalent: %d\n", result);
    
    return 0;
}
