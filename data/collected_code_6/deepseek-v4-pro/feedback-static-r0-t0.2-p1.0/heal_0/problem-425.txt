#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    int size;
} List;

List *create_list(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->size = 0;
    return list;
}

int append(List *list, int value) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return -1;
    }
    new_node->data = value;
    new_node->next = NULL;

    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->size++;
    return 0;
}

void free_list(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

int count_sublists_with_element(List *list, int element) {
    if (list == NULL || list->head == NULL) {
        return 0;
    }

    int total_count = 0;
    int n = list->size;
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        return -1;
    }

    Node *current = list->head;
    for (int i = 0; i < n; i++) {
        arr[i] = current->data;
        current = current->next;
    }

    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            int found = 0;
            for (int k = i; k <= j; k++) {
                if (arr[k] == element) {
                    found = 1;
                    break;
                }
            }
            if (found) {
                total_count++;
            }
        }
    }

    free(arr);
    return total_count;
}

int main(void) {
    List *list = create_list();
    if (list == NULL) {
        fprintf(stderr, "Failed to create list\n");
        return 1;
    }

    int values[] = {1, 2, 3, 2, 4, 2, 5};
    int num_values = sizeof(values) / sizeof(values[0]);

    for (int i = 0; i < num_values; i++) {
        if (append(list, values[i]) != 0) {
            fprintf(stderr, "Failed to append value\n");
            free_list(list);
            return 1;
        }
    }

    int element = 2;
    int result = count_sublists_with_element(list, element);

    if (result < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        free_list(list);
        return 1;
    }

    printf("Number of sublists containing %d: %d\n", element, result);

    free_list(list);
    return 0;
}
