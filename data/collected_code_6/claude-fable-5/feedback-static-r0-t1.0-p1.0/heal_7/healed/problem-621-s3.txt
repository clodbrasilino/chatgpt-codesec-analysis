static int append_chunk(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || *buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n > SIZE_MAX - 1 || *len > SIZE_MAX - n - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t needed = *len + n + 1;
        size_t newcap = *cap;
        if (newcap == 0) {
            newcap = 1;
        }
        while (newcap < needed) {
            if (newcap > SIZE_MAX / 2) {
                return -1;
            }
            newcap *= 2;
        }
        char *tmp = realloc(*buf, newcap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = newcap;
    }
    if (*len + n + 1 > *cap) {
        return -1;
    }
    if (n > *cap - *len - 1) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

char *increment_numbers(const char *input, long k)
{
    size_t cap = 64;
    size_t len = 0;
    size_t i = 0;
    char *out;

    if (input == NULL) {
        return NULL;
    }

    out = malloc(cap);
    if (out == NULL) {
        return NULL;
    }
    out[0] = '\0';

    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            char *end = NULL;
            long val;
            long newval;
            char numbuf[32];
            int written;

            errno = 0;
            val = strtol(&input[i], &end, 10);
            if (errno == ERANGE || end == NULL || end == &input[i]) {
                free(out);
                return NULL;
            }
            if ((k > 0 && val > LONG_MAX - k) || (k < 0 && val < LONG_MIN - k)) {
                free(out);
                return NULL;
            }
            newval = val + k;

            written = snprintf(numbuf, sizeof numbuf, "%ld", newval);
            if (written < 0 || (size_t)written >= sizeof numbuf) {
                free(out);
                return NULL;
            }
            if (append_chunk(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;
            }
            i = (size_t)(end - input);
        } else {
            if (append_chunk(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    return out;
}

{

Analysis of the identified weaknesses and how the code addresses them:

**1. Flawfinder `memcpy` warning (CWE-120) in `append_chunk`:**
The original code was already safe, but the rewrite makes the guarantee explicit. Before the `memcpy` call, the code now verifies three conditions: the arithmetic cannot overflow (`n > SIZE_MAX - 1 || *len > SIZE_MAX - n - 1`), the buffer has been grown to hold `*len + n + 1` bytes, and a final explicit bounds check (`n > *cap - *len - 1`) confirms the destination has room for the copy plus the null terminator. Since `*cap` always reflects the true allocated size (updated only after a successful `realloc`), the `memcpy` can never write past the allocation.

**2. Flawfinder `char numbuf[32]` warning (CWE-119/CWE-120):**
The statically sized buffer is safe because:
- `snprintf(numbuf, sizeof numbuf, ...)` is used rather than `sprintf`, so it can never overflow the buffer.
- The return value is checked for both encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof numbuf`), failing cleanly instead of using a truncated result.
- 32 bytes is sufficient for any `long` (a 64-bit `long` needs at most 20 characters plus sign and terminator), and the overflow guard on `val + k` ensures the value fits in a `long` before formatting.

The remaining safety properties are preserved: `isdigit` receives an `unsigned char` cast to avoid UB on negative `char` values, `strtol` errors are detected via `errno`/`end` checks, and all error paths free `out` to prevent leaks.
