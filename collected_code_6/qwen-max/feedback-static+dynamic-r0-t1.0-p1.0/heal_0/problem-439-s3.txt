```c
#include <stdio.h>
#include <stdlib.h>

long long combineIntegers(int *nums, int count) {
    long long result = 0;
    for (int i = 0; i < count; ++i) {
        result = result * 10 + nums[i];
    }
    return result;
}

int main() {
    int numbers[] = {1, 2, 3, 4, 5};
    int n = sizeof(numbers) / sizeof(numbers[0]);
    long long combined = combineIntegers(numbers, n);
    printf("%lld\n", combined);
    return 0;
}
```
