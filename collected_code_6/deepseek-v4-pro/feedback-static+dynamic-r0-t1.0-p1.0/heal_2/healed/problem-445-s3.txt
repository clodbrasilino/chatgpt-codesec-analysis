#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    size_t length;
} Tuple;

int tuple_multiply(const Tuple *t1, const Tuple *t2, Tuple *result) {
    if (t1 == NULL || t2 == NULL || result == NULL) {
        return -1;
    }
    
    if (t1->length != t2->length) {
        return -1;
    }
    
    if (t1->elements == NULL || t2->elements == NULL) {
        return -1;
    }
    
    result->elements = (int *)malloc(t1->length * sizeof(int));
    if (result->elements == NULL) {
        return -1;
    }
    
    result->length = t1->length;
    
    for (size_t i = 0; i < t1->length; i++) {
        result->elements[i] = t1->elements[i] * t2->elements[i];
    }
    
    return 0;
}

void tuple_free(Tuple *t) {
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->length = 0;
    }
}

int tuple_init(Tuple *t, const int *data, size_t length) {
    if (t == NULL) {
        return -1;
    }
    
    if (length > 0 && data == NULL) {
        return -1;
    }
    
    t->elements = (int *)malloc(length * sizeof(int));
    if (t->elements == NULL) {
        return -1;
    }
    
    t->length = length;
    
    for (size_t i = 0; i < length; i++) {
        t->elements[i] = data[i];
    }
    
    return 0;
}

int main(void) {
    const int data1[] = {1, 2, 3, 4};
    const int data2[] = {5, 6, 7, 8};
    
    Tuple t1, t2, result;
    
    if (tuple_init(&t1, data1, 4) != 0) {
        return 1;
    }
    
    if (tuple_init(&t2, data2, 4) != 0) {
        tuple_free(&t1);
        return 1;
    }
    
    if (tuple_multiply(&t1, &t2, &result) != 0) {
        tuple_free(&t1);
        tuple_free(&t2);
        return 1;
    }
    
    printf("Result: ");
    for (size_t i = 0; i < result.length; i++) {
        printf("%d ", result.elements[i]);
    }
    printf("\n");
    
    tuple_free(&t1);
    tuple_free(&t2);
    tuple_free(&result);
    
    return 0;
}
