```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_ROMAN_LEN 64

static int roman_char_value(char c)
{
    switch (toupper((unsigned char)c)) {
    case 'I': return 1;
    case 'V': return 5;
    case 'X': return 10;
    case 'L': return 50;
    case 'C': return 100;
    case 'D': return 500;
    case 'M': return 1000;
    default:  return -1;
    }
}

static int roman_to_int(const char *s, size_t max_len, int *result)
{
    size_t len;
    size_t i;
    int total = 0;

    if (s == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(s, max_len);
    if (len == 0 || len >= max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        int current = roman_char_value(s[i]);
        if (current < 0) {
            return -1;
        }

        if (i + 1 < len) {
            int next = roman_char_value(s[i + 1]);
            if (next < 0) {
                return -1;
            }
            if (current < next) {
                total += next - current;
                i++;
                continue;
            }
        }
        total += current;
    }

    *result = total;
    return 0;
}

int main(void)
{
    char buffer[MAX_ROMAN_LEN];
    int value = 0;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a Roman numeral: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    if (strchr(buffer, '\n') == NULL && strlen(buffer) == sizeof(buffer) - 1) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\r\n")] = '\0';

    if (roman_to_int(buffer, sizeof(buffer), &value) != 0) {
        fprintf(stderr, "Invalid Roman numeral: %s\n", buffer);
        return EXIT_FAILURE;
    }

    printf("%s = %d\n", buffer, value);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive, but the rewrite hardens it against the flagged weakness (CWE-119/CWE-120 — statically-sized buffer) and a few adjacent issues:

1. **Bounded input** — `fgets(buffer, sizeof(buffer), stdin)` already prevented overflow; this is retained, plus explicit null termination at `buffer[sizeof(buffer) - 1]`.

2. **Truncation detection** — The original silently truncated oversized input, which could cause a valid-looking but wrong parse. The rewrite detects when the buffer filled without a newline, drains the remaining `stdin` characters (preventing them from leaking into subsequent reads), and rejects the input outright.

3. **Bounded string measurement** — `strnlen(s, max_len)` with a `len >= max_len` rejection guards against unterminated strings inside `roman_to_int`, and a `max_len == 0` check was added to avoid an edge case where `strnlen` semantics could mislead.

4. **Safe character classification** — `toupper((unsigned char)c)` correctly avoids undefined behavior with negative `char` values on platforms where `char` is signed.

5. **Output flushing** — `fflush(stdout)` ensures the prompt appears before the blocking read on line-buffered or piped streams.
