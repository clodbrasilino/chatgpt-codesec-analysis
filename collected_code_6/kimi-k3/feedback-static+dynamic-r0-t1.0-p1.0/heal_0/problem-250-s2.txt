```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    void *data;
    size_t size;
    size_t elem_size;
    int (*cmp)(const void *, const void *);
} Tuple;

int int_cmp(const void *a, const void *b)
{
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    return (ia > ib) - (ia < ib);
}

size_t count_occurrences(const Tuple *tuple, const void *element)
{
    size_t count = 0;
    size_t i;
    const char *base;

    if (tuple == NULL || element == NULL || tuple->data == NULL || 
        tuple->cmp == NULL || tuple->elem_size == 0) {
        return 0;
    }

    base = (const char *)tuple->data;
    for (i = 0; i < tuple->size; ++i) {
        if (tuple->cmp(base + i * tuple->elem_size, element) == 0) {
            ++count;
        }
    }
    return count;
}

int main(void)
{
    int arr[] = {1, 2, 3, 2, 4, 2, 5};
    size_t arr_size = sizeof(arr) / sizeof(arr[0]);
    int target = 2;
    Tuple tuple;
    size_t result;

    tuple.data = arr;
    tuple.size = arr_size;
    tuple.elem_size = sizeof(int);
    tuple.cmp = int_cmp;

    result = count_occurrences(&tuple, &target);
    printf("Occurrences of %d: %zu\n", target, result);

    return 0;
}
```
