```c
#include <stdio.h>
#include <stdlib.h>

void swap_numbers(int *a, int *b) {
    if (a == NULL || b == NULL) {
        fprintf(stderr, "Error: NULL pointer passed to swap_numbers\n");
        exit(EXIT_FAILURE);
    }
    if (a == b) {
        return;
    }
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int num1 = 10;
    int num2 = 20;
    
    printf("Before swap: num1 = %d, num2 = %d\n", num1, num2);
    swap_numbers(&num1, &num2);
    printf("After swap: num1 = %d, num2 = %d\n", num1, num2);
    
    return 0;
}
```
