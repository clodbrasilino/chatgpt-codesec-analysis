#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

int compare(const void *a, const void *b) {
    return (*(int *)a - *(int *)b);
}

bool find3Numbers(int A[], int arr_size, int sum) {
    if (A == NULL || arr_size < 3) {
        return false;
    }

    qsort(A, arr_size, sizeof(int), compare);

    for (int i = 0; i < arr_size - 2; i++) {
        int l = i + 1;
        int r = arr_size - 1;
        while (l < r) {
            if (A[i] + A[l] + A[r] == sum) {
                return true;
            } else if (A[i] + A[l] + A[r] < sum) {
                l++;
            } else {
                r--;
            }
        }
    }
    return false;
}

int main(void) {
    int A[] = {1, 4, 45, 6, 10, 8};
    int sum = 22;
    int arr_size = sizeof(A) / sizeof(A[0]);

    if (find3Numbers(A, arr_size, sum)) {
        printf("Triplet found.\n");
    } else {
        printf("Triplet not found.\n");
    }

    return 0;
}
