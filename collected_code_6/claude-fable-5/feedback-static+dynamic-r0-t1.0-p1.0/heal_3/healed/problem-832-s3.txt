```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <errno.h>
#include <limits.h>

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;
        long value;
        const char *start;
        char *endptr;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            regfree(&regex);
            return -1;
        }

        len = (size_t)(match.rm_eo - match.rm_so);
        if (len == 0) {
            regfree(&regex);
            return -1;
        }

        start = cursor + match.rm_so;

        errno = 0;
        value = strtol(start, &endptr, 10);
        if (errno == ERANGE || endptr == start ||
            (size_t)(endptr - start) != len) {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (match.rm_eo == match.rm_so) {
            if (*cursor == '\0') {
                break;
            }
            cursor++;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test = "abc 42 def -17 ghi 1000 jkl 999";
    long max_value;

    if (extract_max_number(test, &max_value) != 0) {
        fprintf(stderr, "No numeric value found or error occurred\n");
        return EXIT_FAILURE;
    }

    printf("Maximum numeric value: %ld\n", max_value);
    return EXIT_SUCCESS;
}
```

The key change eliminates both flagged weaknesses at their root: instead of copying each regex match into a statically-sized `buffer` via `memcpy` (CWE-119/CWE-120), the rewritten version parses the number in place with `strtol` directly on the matched region of the input string. Correctness is preserved by verifying that `strtol` consumed exactly the matched span (`endptr - start == len`), which replaces the old `*endptr != '\0'` check. Since no intermediate buffer exists, there is no overflow risk, no arbitrary `NUM_BUFFER_SIZE` limit on numeric token length, and matches longer than 63 characters no longer cause a spurious failure — they are instead correctly rejected only if they actually overflow `long` (via the `ERANGE` check). All other behavior, including the zero-length match guard and resource cleanup with `regfree`, is retained.
