#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_vowel(char c) {
    c = (char)tolower((unsigned char)c);
    return (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u');
}

size_t count_chars_with_vowel_neighbors(const char *str) {
    size_t count = 0;
    size_t len;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    len = strlen(str);
    if (len < 3) {
        return 0;
    }

    for (i = 1; i < len - 1; i++) {
        if (is_vowel(str[i - 1]) && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void) {
    char buffer[1024];
    size_t result;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer);
    printf("Count: %zu\n", result);

    return 0;
}
