#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **remove_words(char **words, size_t count, char **targets, size_t target_count, size_t *new_count);
void free_word_list(char **list, size_t count);
static char **split_line(char *line, size_t *count);

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }
    len = strlen(src);
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, src, len + 1);
    return copy;
}

char **remove_words(char **words, size_t count, char **targets, size_t target_count, size_t *new_count)
{
    char **result;
    size_t kept;
    size_t i;

    if (new_count == NULL) {
        return NULL;
    }

    *new_count = 0;

    if (words == NULL && count > 0) {
        return NULL;
    }

    if (count > ((size_t)-1) / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count > 0 ? count : 1, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    kept = 0;
    for (i = 0; i < count; i++) {
        int match;
        size_t j;

        if (words[i] == NULL) {
            continue;
        }

        match = 0;
        if (targets != NULL) {
            for (j = 0; j < target_count; j++) {
                if (targets[j] != NULL && strcmp(words[i], targets[j]) == 0) {
                    match = 1;
                    break;
                }
            }
        }

        if (match == 0) {
            result[kept] = duplicate_string(words[i]);
            if (result[kept] == NULL) {
                free_word_list(result, kept);
                return NULL;
            }
            kept++;
        }
    }

    *new_count = kept;
    return result;
}

void free_word_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char *read_line(FILE *fp)
{
    char *buf;
    size_t capacity;
    size_t length;
    int c;

    if (fp == NULL) {
        return NULL;
    }

    capacity = 128;
    length = 0;
    buf = malloc(capacity);
    if (buf == NULL) {
        return NULL;
    }

    c = fgetc(fp);
    if (c == EOF) {
        free(buf);
        return NULL;
    }

    while (c != EOF && c != '\n') {
        if (length + 2 >= capacity) {
            char *tmp;

            if (capacity > ((size_t)-1) / 2) {
                free(buf);
                return NULL;
            }
            capacity *= 2;
            tmp = realloc(buf, capacity);
            if (tmp == NULL) {
                free(buf);
                return NULL;
            }
            buf = tmp;
        }
        buf[length] = (char)c;
        length++;
        c = fgetc(fp);
    }

    buf[length] = '\0';
    return buf;
}

static int is_delim(char c)
{
    return c == ' ' || c == '\t' || c == '\r' || c == '\n' ||
           c == ',' || c == '[' || c == ']' || c == '\'' || c == '"';
}

static char **split_line(char *line, size_t *count)
{
    char **tokens;
    size_t capacity;
    size_t n;
    size_t pos;
    size_t len;

    if (line == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;
    capacity = 8;
    n = 0;
    tokens = malloc(sizeof(char *) * capacity);
    if (tokens == NULL) {
        return NULL;
    }

    len = strlen(line);
    pos = 0;
    while (pos < len) {
        size_t start;

        while (pos < len && is_delim(line[pos])) {
            pos++;
        }
        if (pos >= len) {
            break;
        }
        start = pos;
        while (pos < len && !is_delim(line[pos])) {
            pos++;
        }
        line[pos] = '\0';
        if (pos < len) {
            pos++;
        }

        if (n >= capacity) {
            char **tmp;

            if (capacity > ((size_t)-1) / (2 * sizeof(char *))) {
                free(tokens);
                return NULL;
            }
            capacity *= 2;
            tmp = realloc(tokens, sizeof(char *) * capacity);
            if (tmp == NULL) {
                free(tokens);
                return NULL;
            }
            tokens = tmp;
        }
        tokens[n] = line + start;
        n++;
    }

    *count = n;
    return tokens;
}

static void print_list(char **list, size_t count)
{
    printf("[");
    if (list != NULL) {
        size_t i;

        for (i = 0; i < count; i++) {
            if (list[i] != NULL) {
                printf("'%s'", list[i]);
            } else {
                printf("''");
            }
            if (i + 1 < count) {
                printf(", ");
            }
        }
    }
    printf("]\n");
}

int main(void)
{
    char *word_line;
    char *target_line;
    char **words;
    char **targets;
    char **filtered;
    size_t word_count;
    size_t target_count;
    size_t new_count;

    word_count = 0;
    target_count = 0;
    new_count = 0;

    word_line = read_line(stdin);
    if (word_line == NULL) {
        word_line = malloc(1);
        if (word_line == NULL) {
            fprintf(stderr, "Out of memory\n");
            return EXIT_FAILURE;
        }
        word_line[0] = '\0';
    }

    target_line = read_line(stdin);
    if (target_line == NULL) {
        target_line = malloc(1);
        if (target_line == NULL) {
            free(word_line);
            fprintf(stderr, "Out of memory\n");
            return EXIT_FAILURE;
        }
        target_line[0] = '\0';
    }

    words = split_line(word_line, &word_count);
    if (words == NULL) {
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to parse words\n");
        return EXIT_FAILURE;
    }

    targets = split_line(target_line, &target_count);
    if (targets == NULL) {
        free(words);
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to parse targets\n");
        return EXIT_FAILURE;
    }

    filtered = remove_words(words, word_count, targets, target_count, &new_count);
    if (filtered == NULL) {
        free(words);
        free(targets);
        free(word_line);
        free(target_line);
        fprintf(stderr, "Failed to remove words\n");
        return EXIT_FAILURE;
    }

    print_list(filtered, new_count);
    fflush(stdout);

    free_word_list(filtered, new_count);
    free(words);
    free(targets);
    free(word_line);
    free(target_line);
    return EXIT_SUCCESS;
}
