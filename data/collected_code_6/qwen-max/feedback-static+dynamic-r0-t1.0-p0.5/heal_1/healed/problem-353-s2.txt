#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    struct List *down;
} List;

List* create_list() {
    return (List*)malloc(sizeof(List));
}

Node* create_node(int value) {
    Node *new_node = (Node*)malloc(sizeof(Node));
    new_node->value = value;
    new_node->next = NULL;
    return new_node;
}

void append_node(List *list, int value) {
    Node *new_node = create_node(value);
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
}

void append_down(List *list, List *new_list) {
    if (list->down == NULL) {
        list->down = new_list;
    } else {
        List *current = list;
        while (current->down != NULL) {
            current = current->down;
        }
        current->down = new_list;
    }
}

void free_list(List *list) {
    if (list == NULL) return;
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    List *down_current = list->down;
    while (down_current != NULL) {
        List *temp = down_current;
        down_current = down_current->down;
        free_list(temp);
    }
    free(list);
}

List* remove_column(List *list, int column) {
    if (list == NULL) return NULL;
    List *result = create_list();
    List *current = list;
    List *result_current = result;
    int col = 0;
    while (current != NULL) {
        if (col != column) {
            result_current->head = current->head;
            result_current->down = NULL;
            if (current->down != NULL) {
                result_current->down = create_list();
                result_current = result_current->down;
            }
        }
        current = current->down;
        col++;
    }
    return result;
}

int main() {
    List *list = create_list();
    List *row1 = create_list();
    append_node(row1, 1);
    append_node(row1, 2);
    append_node(row1, 3);
    append_down(list, row1);

    List *row2 = create_list();
    append_node(row2, 4);
    append_node(row2, 5);
    append_node(row2, 6);
    append_down(list, row2);

    List *row3 = create_list();
    append_node(row3, 7);
    append_node(row3, 8);
    append_node(row3, 9);
    append_down(list, row3);

    List *new_list = remove_column(list, 1);

    List *current = new_list;
    while (current != NULL) {
        Node *node = current->head;
        while (node != NULL) {
            printf("%d ", node->value);
            node = node->next;
        }
        printf("\n");
        current = current->down;
    }

    free_list(list);
    free_list(new_list);
    return 0;
}
