```c
#include <stdio.h>
#include <string.h>
#include <regex.h>
#include <stdlib.h>

int match_string(const char *str) {
    if (str == NULL) {
        return 0;
    }

    regex_t regex;
    int reti;

    reti = regcomp(&regex, "^ab*$", REG_EXTENDED);
    if (reti) {
        return 0;
    }

    reti = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);

    if (!reti) {
        return 1;
    } else if (reti == REG_NOMATCH) {
        return 0;
    } else {
        return 0;
    }
}

int main(void) {
    const char *test_strings[] = {
        "a",
        "ab",
        "abb",
        "b",
        "ba",
        "abc",
        NULL
    };

    for (int i = 0; test_strings[i] != NULL; ++i) {
        int result = match_string(test_strings[i]);
        printf("String: '%s', Match: %d\n", test_strings[i], result);
    }

    return 0;
}
```
