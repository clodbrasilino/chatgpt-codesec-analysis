```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_EMAIL_LEN 254

int is_valid_email(const char *email)
{
    regex_t regex;
    int result;
    size_t len;
    const char *pattern = "^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$";

    if (email == NULL) {
        return 0;
    }

    len = strnlen(email, MAX_EMAIL_LEN + 1);
    if (len == 0 || len > MAX_EMAIL_LEN) {
        return 0;
    }

    result = regcomp(&regex, pattern, REG_EXTENDED | REG_NOSUB);
    if (result != 0) {
        return -1;
    }

    result = regexec(&regex, email, 0, NULL, 0);
    regfree(&regex);

    if (result == 0) {
        return 1;
    }

    return 0;
}

int main(void)
{
    const char *test_emails[] = {
        "user@example.com",
        "john.doe@company.org",
        "invalid-email",
        "missing@domain",
        "@nodomain.com",
        "user@sub.domain.co.uk",
        "spaces in@email.com",
        ""
    };
    size_t count = sizeof(test_emails) / sizeof(test_emails[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int status = is_valid_email(test_emails[i]);
        if (status == -1) {
            fprintf(stderr, "Error: failed to compile regex\n");
            return EXIT_FAILURE;
        }
        if (status == 1) {
            printf("\"%s\" is a valid email address\n", test_emails[i]);
        } else {
            printf("\"%s\" is NOT a valid email address\n", test_emails[i]);
        }
    }

    return EXIT_SUCCESS;
}
```

The key changes address both identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced the two `strlen()` calls with a single `strnlen()` call bounded to `MAX_EMAIL_LEN + 1`. This guarantees the read never exceeds 255 bytes even if the input string is not null-terminated, and also avoids scanning the string twice.

2. **Variable scope (cppcheck)**: Moved the `status` declaration inside the `for` loop body, reducing its scope to where it's actually used.

Additionally, the magic number 254 was extracted into a named constant (`MAX_EMAIL_LEN`), which improves maintainability and makes the bounded-length check self-documenting.
