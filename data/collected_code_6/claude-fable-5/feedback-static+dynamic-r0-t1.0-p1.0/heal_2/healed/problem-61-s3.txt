```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U

long long count_substrings(const char *s, size_t max_len);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL || max_len == 0U) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n >= max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;

    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        long long idx;

        prefix_sum += (long long)(s[i] - '0') - 1;
        idx = prefix_sum + offset;

        if (idx < 0 || (size_t)idx >= table_size) {
            free(freq);
            return -1;
        }

        count += freq[idx];
        freq[idx]++;
    }

    free(freq);
    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    if (fgets(buffer, (int)sizeof buffer, stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof buffer - 1U] = '\0';
    buffer[strcspn(buffer, "\r\n")] = '\0';

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("%lld\n", result);
    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Removed the interactive prompt** (`"Enter a digit string: "`) and the `"Number of substrings: "` prefix from the output. The failed test cases (`expected 6, got <no output>`) strongly suggest an automated test harness reads stdout expecting only the numeric result. The extra prompt/label text meant the harness could not match the expected value, and the removal of `fflush(stdout)` error path also eliminates a spurious failure exit before any output.

2. **Simplified and hardened the truncation check** in `count_substrings`. The original condition `n >= max_len && s[max_len - 1U] != '\0'` was redundant: `strnlen` returning `max_len` already proves no terminator exists within the buffer, so `n >= max_len` alone correctly rejects unterminated/truncated input. A `max_len == 0U` guard was also added to prevent any out-of-bounds reasoning on a zero-length limit.

3. **Flawfinder CWE-119/CWE-120 concern on `buffer`**: the statically sized array remains, but every access is bounds-safe — `fgets` is limited to `sizeof buffer`, the buffer is explicitly null-terminated at `sizeof buffer - 1U`, `strcspn` operates only on the terminated string, and `count_substrings` receives `sizeof buffer` and independently verifies termination via `strnlen`. The overflow-checked `calloc` sizing (`10U * n + 1U` with `SIZE_MAX` guards) and the runtime index bounds check on `idx` are retained, so no unchecked buffer writes remain.
