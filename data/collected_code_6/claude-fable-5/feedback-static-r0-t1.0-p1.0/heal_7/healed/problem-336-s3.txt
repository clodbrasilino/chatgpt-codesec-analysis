```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`char normalized[MAX_INPUT_LEN]` in `has_28_days` (CWE-119/CWE-120):** This is a low-risk, informational finding. The buffer is never overflowed because the copy loop is bounded by both `len` (which is guaranteed `< MAX_INPUT_LEN` via `strnlen`) and `i < sizeof(normalized) - 1`, and the null terminator is written at index `i`, which is always within bounds. No fix required.

2. **`getchar()` in `flush_stdin` (CWE-120/CWE-20):** No buffer is written to; the return value is only compared against `'\n'` and `EOF`. Correctly using `int` (not `char`) for the return value means `EOF` detection is reliable and the loop terminates. No fix required.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, sizeof(input), stdin)`, which limits the read to the buffer size. Defensive null termination at `sizeof(input) - 1`, `strnlen` bounded length calculation, and flushing residual stdin when input exceeds the buffer all mitigate overflow and input-truncation issues. No fix required.

The Flawfinder hits are pattern-based warnings on the mere presence of fixed-size arrays and `getchar`; each is already mitigated by bounds-checked operations (`fgets`, `strnlen`, explicit loop limits, and guaranteed null termination), so the code stands as-is with the comments removed.
