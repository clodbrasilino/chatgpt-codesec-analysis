```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_non_alphanumeric(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_pos = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t match_len;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret != 0) {
            break;
        }
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            break;
        }

        match_len = (size_t)(match.rm_eo - match.rm_so);

        if (match_len > input_len ||
            result_pos > input_len - match_len ||
            result_pos + match_len > input_len) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        memcpy(result + result_pos, cursor + match.rm_so, match_len);
        result_pos += match_len;
        cursor += match.rm_eo;
    }

    result[result_pos] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a@b#c$d%e^1&2*3",
        "!!!???...",
        "AlreadyClean123"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alphanumeric(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n",
                    test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: %s\nCleaned:  %s\n\n",
               test_strings[i], cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning on `memcpy` (CWE-120) was a low-risk, largely informational finding. The original code already mitigated it correctly, and the rewrite retains and slightly hardens those protections:

1. **Bounded allocation:** `result` is allocated as `input_len + 1` bytes, and since the output can never exceed the input length (we only ever copy subsets of the input), the destination is guaranteed large enough by construction.

2. **Explicit pre-copy bounds checks:** Before each `memcpy`, the code verifies `match_len > input_len`, `result_pos > input_len - match_len` (written to avoid unsigned overflow), and redundantly `result_pos + match_len > input_len`. Any violation aborts safely with cleanup.

3. **Input length capping:** `strnlen` with `MAX_INPUT_LEN` prevents unbounded reads on non-terminated or hostile inputs.

4. **Defensive regex offset validation:** Added `match.rm_so < 0` check, since `regoff_t` is signed and a paranoid check costs nothing.

5. **Resource hygiene:** All exit paths correctly pair `malloc`/`free` and `regcomp`/`regfree`, with no leaks on error paths.

The code compiles cleanly with `-Wall -Wextra -pedantic` and is safe against buffer overflow in the flagged `memcpy` call.
