```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "/([0-9]{4})/([0-9]{1,2})/([0-9]{1,2})(/|$)"
#define MAX_MATCHES 5
#define BUF_SIZE 16
#define MAX_URL_LEN 2048

typedef struct {
    int year;
    int month;
    int day;
} url_date_t;

static int copy_match(const char *source, const regmatch_t *match,
                      char *dest, size_t dest_size)
{
    size_t length;
    size_t source_len;

    if (source == NULL || match == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }
    if (match->rm_so < 0 || match->rm_eo < match->rm_so) {
        return -1;
    }

    source_len = strnlen(source, MAX_URL_LEN);
    if (source_len >= MAX_URL_LEN) {
        return -1;
    }
    if ((size_t)match->rm_so > source_len || (size_t)match->rm_eo > source_len) {
        return -1;
    }

    length = (size_t)(match->rm_eo - match->rm_so);
    if (length >= dest_size) {
        return -1;
    }

    memcpy(dest, source + match->rm_so, length);
    dest[length] = '\0';
    return 0;
}

static int extract_date_from_url(const char *url, url_date_t *out_date)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCHES];
    char year_buf[BUF_SIZE];
    char month_buf[BUF_SIZE];
    char day_buf[BUF_SIZE];
    int rc;
    long year;
    long month;
    long day;
    char *endptr;

    if (url == NULL || out_date == NULL) {
        return -1;
    }

    if (strnlen(url, MAX_URL_LEN) >= MAX_URL_LEN) {
        return -1;
    }

    rc = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    rc = regexec(&regex, url, MAX_MATCHES, matches, 0);
    if (rc != 0) {
        regfree(&regex);
        return -1;
    }

    if (copy_match(url, &matches[1], year_buf, sizeof(year_buf)) != 0 ||
        copy_match(url, &matches[2], month_buf, sizeof(month_buf)) != 0 ||
        copy_match(url, &matches[3], day_buf, sizeof(day_buf)) != 0) {
        regfree(&regex);
        return -1;
    }

    regfree(&regex);

    year = strtol(year_buf, &endptr, 10);
    if (endptr == year_buf || *endptr != '\0' || year < 1 || year > 9999) {
        return -1;
    }

    month = strtol(month_buf, &endptr, 10);
    if (endptr == month_buf || *endptr != '\0' || month < 1 || month > 12) {
        return -1;
    }

    day = strtol(day_buf, &endptr, 10);
    if (endptr == day_buf || *endptr != '\0' || day < 1 || day > 31) {
        return -1;
    }

    out_date->year = (int)year;
    out_date->month = (int)month;
    out_date->day = (int)day;
    return 0;
}

int main(void)
{
    const char *urls[] = {
        "https://example.com/blog/2023/11/25/some-article",
        "https://example.com/news/2019/7/4/",
        "https://example.com/no-date-here",
        "https://example.com/2020/13/40/invalid"
    };
    size_t count;
    size_t i;
    url_date_t date;

    count = sizeof(urls) / sizeof(urls[0]);

    for (i = 0; i < count; i++) {
        if (extract_date_from_url(urls[i], &date) == 0) {
            if (printf("URL: %s -> Year: %d, Month: %d, Day: %d\n",
                       urls[i], date.year, date.month, date.day) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("URL: %s -> No valid date found\n", urls[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`memcpy` (CWE-120) in `copy_match`**: This is a false positive. The code validates that `length >= dest_size` fails before the copy, guaranteeing the destination can hold `length` bytes plus the null terminator. Additionally, the source offsets (`rm_so`, `rm_eo`) are bounds-checked against the actual source length via `strnlen` before any copy occurs.

2. **Statically-sized `char` arrays (CWE-119/CWE-120)**: These are informational warnings triggered by any fixed-size buffer declaration. All three buffers (`year_buf`, `month_buf`, `day_buf`) are only written through `copy_match`, which enforces bounds via the `dest_size` parameter passed with `sizeof()`. `BUF_SIZE` (16) comfortably exceeds the maximum match lengths dictated by the regex (`{4}` and `{1,2}` digit groups).

Additional defensive measures already present that I preserved:

- `strnlen` with `MAX_URL_LEN` cap prevents unbounded reads on non-terminated input.
- `regfree` is called on every exit path after `regcomp` succeeds, avoiding resource leaks.
- `strtol` results are validated with `endptr` checks and semantic range checks (year 1–9999, month 1–12, day 1–31), so no integer conversion issues propagate.
- All pointer parameters are NULL-checked before use.

No functional changes were required — the original code correctly mitigates all flagged issues; the warnings are static-analysis noise inherent to `memcpy` and fixed-size arrays.
