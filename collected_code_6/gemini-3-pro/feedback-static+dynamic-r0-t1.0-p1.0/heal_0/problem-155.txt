#include <stdio.h>
#include <stdint.h>

uint32_t toggle_even_bits(uint32_t num);

uint32_t toggle_even_bits(uint32_t num) {
    return num ^ 0x55555555U;
}

int main(void) {
    uint32_t num = 42;
    uint32_t result;

    result = toggle_even_bits(num);

    if (printf("%u\n", result) < 0) {
        return 1;
    }

    return 0;
}
