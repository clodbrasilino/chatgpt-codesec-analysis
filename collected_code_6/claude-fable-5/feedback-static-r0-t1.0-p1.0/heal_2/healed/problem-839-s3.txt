```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static void sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

static int print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return -1;
    }

    for (i = 0U; i < count; i++) {
        if (printf("(%.*s, %d)\n", MAX_NAME_LEN - 1, tuples[i].first, tuples[i].second) < 0) {
            return -1;
        }
    }
    return 0;
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Initialization error\n");
        return EXIT_FAILURE;
    }

    sort_tuples(tuples, count);

    if (print_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Output error\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-119/CWE-120 (statically-sized array `char first[MAX_NAME_LEN]`):** The buffer itself remains fixed-size, but every write path is bounds-checked. `set_tuple` is the only writer, and it rejects any input whose length is `>= MAX_NAME_LEN`, guaranteeing room for the NUL terminator.

2. **CWE-126 (`strlen` over-read on non-terminated strings):** Replaced `strlen(name)` with `strnlen(name, MAX_NAME_LEN)`. This bounds the read to `MAX_NAME_LEN` bytes; a string that isn't terminated within that window yields `len == MAX_NAME_LEN`, which the subsequent check rejects. No over-read past the relevant limit can occur.

3. **CWE-120 (`memcpy` without destination check):** The `memcpy` is now provably safe: `len < MAX_NAME_LEN` is enforced immediately before the copy, and the destination is exactly `MAX_NAME_LEN` bytes, so `len` bytes plus the explicit terminator always fit.

4. **Defensive reads elsewhere:** `strncmp` with `MAX_NAME_LEN` and the `%.*s` precision in `printf` bound all read operations, so even a hypothetically unterminated buffer can't cause an over-read in comparison or output.

The program compiles cleanly with `-Wall -Wextra -Wpedantic`, checks all return values (including `printf`), and null-checks all pointer parameters. Expected output is the five tuples sorted lexicographically by name.
