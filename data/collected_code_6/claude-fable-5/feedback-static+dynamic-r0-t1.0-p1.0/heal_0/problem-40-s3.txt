#include <stdio.h>
#include <stdlib.h>

struct Frequency {
    int value;
    size_t count;
};

static int find_value(const struct Frequency *freqs, size_t n, int value, size_t *index)
{
    size_t i;

    if (freqs == NULL || index == NULL) {
        return 0;
    }

    for (i = 0; i < n; i++) {
        if (freqs[i].value == value) {
            *index = i;
            return 1;
        }
    }

    return 0;
}

static struct Frequency *element_frequencies(const int *const *lists,
                                             const size_t *list_sizes,
                                             size_t num_lists,
                                             size_t *num_frequencies)
{
    struct Frequency *freqs;
    struct Frequency *tmp;
    size_t capacity;
    size_t count;
    size_t i;
    size_t j;
    size_t idx;

    if (lists == NULL || list_sizes == NULL || num_frequencies == NULL) {
        return NULL;
    }

    *num_frequencies = 0;
    capacity = 16;
    count = 0;

    freqs = malloc(capacity * sizeof(*freqs));
    if (freqs == NULL) {
        return NULL;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] == NULL && list_sizes[i] > 0) {
            free(freqs);
            return NULL;
        }
        for (j = 0; j < list_sizes[i]; j++) {
            if (find_value(freqs, count, lists[i][j], &idx)) {
                freqs[idx].count++;
            } else {
                if (count == capacity) {
                    if (capacity > (size_t)-1 / (2 * sizeof(*freqs))) {
                        free(freqs);
                        return NULL;
                    }
                    capacity *= 2;
                    tmp = realloc(freqs, capacity * sizeof(*freqs));
                    if (tmp == NULL) {
                        free(freqs);
                        return NULL;
                    }
                    freqs = tmp;
                }
                freqs[count].value = lists[i][j];
                freqs[count].count = 1;
                count++;
            }
        }
    }

    *num_frequencies = count;
    return freqs;
}

int main(void)
{
    int list1[] = {1, 2, 3, 2};
    int list2[] = {4, 5, 6, 2, 7};
    int list3[] = {1, 3, 5, 2, 8};
    const int *lists[] = {list1, list2, list3};
    size_t sizes[] = {
        sizeof(list1) / sizeof(list1[0]),
        sizeof(list2) / sizeof(list2[0]),
        sizeof(list3) / sizeof(list3[0])
    };
    size_t num_lists = sizeof(lists) / sizeof(lists[0]);
    size_t num_freqs = 0;
    size_t i;
    struct Frequency *freqs;

    freqs = element_frequencies(lists, sizes, num_lists, &num_freqs);
    if (freqs == NULL) {
        fprintf(stderr, "Failed to compute frequencies\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_freqs; i++) {
        if (printf("%d: %zu\n", freqs[i].value, freqs[i].count) < 0) {
            free(freqs);
            return EXIT_FAILURE;
        }
    }

    free(freqs);
    return EXIT_SUCCESS;
}
