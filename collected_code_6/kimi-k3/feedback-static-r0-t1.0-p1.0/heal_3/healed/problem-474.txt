#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void replace_chars(char *str, size_t str_size, const char *old_chars, size_t old_size, const char *new_chars, size_t new_size) {
    if (str == NULL || old_chars == NULL || new_chars == NULL) {
        return;
    }
    
    size_t old_len = strnlen(old_chars, old_size);
    size_t new_len = strnlen(new_chars, new_size);
    
    if (old_len != new_len) {
        return;
    }
    
    size_t str_len = strnlen(str, str_size);
    for (size_t i = 0; i < str_len; i++) {
        for (size_t j = 0; j < old_len; j++) {
            if (str[i] == old_chars[j]) {
                str[i] = new_chars[j];
                break;
            }
        }
    }
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read_len;
    
    printf("Enter a string: ");
    read_len = getline(&buffer, &buffer_size, stdin);
    if (read_len == -1) {
        free(buffer);
        return EXIT_FAILURE;
    }
    
    if (read_len > 0 && buffer[read_len - 1] == '\n') {
        buffer[read_len - 1] = '\0';
        read_len--;
    }
    
    char *old_chars = NULL;
    size_t old_size = 0;
    ssize_t old_len;
    
    printf("Enter characters to replace: ");
    old_len = getline(&old_chars, &old_size, stdin);
    if (old_len == -1) {
        free(buffer);
        free(old_chars);
        return EXIT_FAILURE;
    }
    
    if (old_len > 0 && old_chars[old_len - 1] == '\n') {
        old_chars[old_len - 1] = '\0';
        old_len--;
    }
    
    char *new_chars = NULL;
    size_t new_size = 0;
    ssize_t new_len;
    
    printf("Enter replacement characters: ");
    new_len = getline(&new_chars, &new_size, stdin);
    if (new_len == -1) {
        free(buffer);
        free(old_chars);
        free(new_chars);
        return EXIT_FAILURE;
    }
    
    if (new_len > 0 && new_chars[new_len - 1] == '\n') {
        new_chars[new_len - 1] = '\0';
        new_len--;
    }
    
    replace_chars(buffer, read_len + 1, old_chars, old_len + 1, new_chars, new_len + 1);
    
    printf("Result: %s\n", buffer);
    
    free(buffer);
    free(old_chars);
    free(new_chars);
    
    return EXIT_SUCCESS;
}
