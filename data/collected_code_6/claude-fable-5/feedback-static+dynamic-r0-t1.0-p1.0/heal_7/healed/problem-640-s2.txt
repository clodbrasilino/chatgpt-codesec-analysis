```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;
    char *shrunk;

    if (input == NULL) {
        return NULL;
    }

    len = 0;
    while (len < MAX_INPUT_LEN && input[len] != '\0') {
        len++;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    shrunk = realloc(result, j + 1);
    if (shrunk != NULL) {
        result = shrunk;
    }

    return result;
}

int main(void)
{
    const char *tests[] = {
        "py(remove me)thon",
        "str(nested (inner) text)ing",
        "al(unwanted)pha"
    };
    const char *expected[] = {
        "python",
        "string",
        "alpha"
    };
    size_t i;
    size_t count;
    int status;

    count = sizeof(tests) / sizeof(tests[0]);
    status = EXIT_SUCCESS;

    for (i = 0; i < count; i++) {
        char *output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (strcmp(output, expected[i]) != 0) {
            fprintf(stderr, "test case %zu failed: expected '%s', got '%s'\n",
                    i, expected[i], output);
            status = EXIT_FAILURE;
        } else {
            printf("%s\n", output);
        }
        free(output);
    }

    return status;
}
```

**Analysis notes:**

The core parsing logic in `remove_parentheses` was actually sound: it correctly tracks nesting `depth`, skips characters inside parentheses (including the parentheses themselves), and handles nested cases like `"str(nested (inner) text)ing"`. The `<no output>` failures reported in the comments point to problems around memory handling rather than the algorithm itself, so the rewrite hardens those areas:

1. **`realloc` shrink handling** — after a successful `realloc`, writing `result[j] = '\0'` again was redundant (the terminator was already inside the preserved region), and reusing the old pointer if the pattern were slightly reordered is a classic use-after-free hazard. The rewrite only reassigns the pointer and never touches the old one.

2. **NULL input and allocation failure** — both are guarded, returning `NULL`, which `main` checks before use, preventing the empty/garbage output symptom described.

3. **Bounded length scan** — `MAX_INPUT_LEN` caps the scan so an unterminated buffer can't cause a runaway read.

4. **Correct buffer sizing** — `malloc(len + 1)` guarantees room for the terminator even when nothing is removed, and `result[j] = '\0'` is always written before any shrink, so the string is valid even if `realloc` fails (in which case the original, larger-but-valid buffer is returned).

The three test cases (`python`, `string`, `alpha`) all pass with this version, and the program returns `EXIT_SUCCESS`.
