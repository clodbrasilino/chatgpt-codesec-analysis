#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

static void flip(int *arr, size_t k)
{
    size_t i = 0;
    size_t j = k;

    while (i < j) {
        int tmp = arr[i];
        arr[i] = arr[j];
        arr[j] = tmp;
        i++;
        j--;
    }
}

static size_t find_max_index(const int *arr, size_t n)
{
    size_t max_idx = 0;
    size_t i;

    for (i = 1; i < n; i++) {
        if (arr[i] > arr[max_idx]) {
            max_idx = i;
        }
    }
    return max_idx;
}

static void pancake_sort(int *arr, size_t n)
{
    size_t curr_size;

    if (arr == NULL || n < 2) {
        return;
    }

    for (curr_size = n; curr_size > 1; curr_size--) {
        size_t max_idx = find_max_index(arr, curr_size);

        if (max_idx != curr_size - 1) {
            if (max_idx != 0) {
                flip(arr, max_idx);
            }
            flip(arr, curr_size - 1);
        }
    }
}

static void print_array(const int *arr, size_t n)
{
    size_t i;

    for (i = 0; i < n; i++) {
        if (printf("%d%s", arr[i], (i + 1 < n) ? " " : "\n") < 0) {
            exit(EXIT_FAILURE);
        }
    }
}

int main(void)
{
    int data[] = { 23, -4, 10, 1, 0, 99, -75, 42, 8, 15 };
    size_t n = sizeof(data) / sizeof(data[0]);

    if (printf("Before: ") < 0) {
        return EXIT_FAILURE;
    }
    print_array(data, n);

    pancake_sort(data, n);

    if (printf("After:  ") < 0) {
        return EXIT_FAILURE;
    }
    print_array(data, n);

    return EXIT_SUCCESS;
}
