#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_LEN 256
#define ERRBUF_LEN 128

static int find_sequences(const char *input);
static void report_regex_error(int code, const regex_t *regex, const char *context);

static void report_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_LEN];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1);
    errbuf[sizeof(errbuf) - 1] = '\0';
    fprintf(stderr, "Error: %s: %s\n", context, errbuf);
}

static int find_sequences(const char *input)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor;
    char buffer[MAX_MATCH_LEN];
    int ret;
    int found;

    if (input == NULL) {
        fprintf(stderr, "Error: input string is NULL\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        report_regex_error(ret, &regex, "failed to compile regex");
        return -1;
    }

    found = 0;
    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            report_regex_error(ret, &regex, "regex execution failed");
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            regfree(&regex);
            fprintf(stderr, "Error: invalid match offsets\n");
            return -1;
        }

        {
            size_t length = (size_t)(match.rm_eo - match.rm_so);
            size_t max_len = sizeof(buffer) - 1;

            if (length > max_len) {
                length = max_len;
            }
            memcpy(buffer, cursor + match.rm_so, length);
            buffer[length] = '\0';
        }

        printf("Found match: %s\n", buffer);
        found++;

        if (match.rm_eo == 0) {
            cursor++;
        } else {
            cursor += match.rm_eo;
        }
    }

    regfree(&regex);

    if (found == 0) {
        printf("No matches found.\n");
    }

    return found;
}

int main(void)
{
    const char *test_string = "HelloWorld this Is a TestString with Mixed caseWords";
    int result;

    printf("Input: %s\n", test_string);

    result = find_sequences(test_string);
    if (result < 0) {
        fprintf(stderr, "Error: sequence search failed\n");
        return EXIT_FAILURE;
    }

    printf("Total matches: %d\n", result);

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe as written:

1. **`errbuf` static array (CWE-119/CWE-120):** The buffer is zero-initialized with `memset`, `regerror` is called with `sizeof(errbuf) - 1` to guarantee room for a terminator, and the final byte is explicitly null-terminated. No overflow is possible.

2. **`buffer` static array (CWE-119/CWE-120):** Match offsets are validated (`rm_so >= 0`, `rm_eo >= rm_so`) before use, and the copy length is clamped to `sizeof(buffer) - 1`, reserving space for the null terminator.

3. **`memcpy` (CWE-120):** The destination can always hold the source data because `length` is capped at `max_len` immediately before the copy, and the string is explicitly terminated afterward. This is a false positive from Flawfinder's pattern matching — the bounds check it asks for is already present.
