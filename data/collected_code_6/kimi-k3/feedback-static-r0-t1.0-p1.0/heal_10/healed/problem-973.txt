#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

void left_rotate_string(char *str, size_t n, size_t max_len) {
    size_t len;
    char *temp;
    size_t i;
    
    if (str == NULL || max_len == 0) {
        return;
    }
    
    len = strnlen(str, max_len);
    
    if (len == 0 || n == 0) {
        return;
    }
    
    n = n % len;
    
    if (n == 0) {
        return;
    }
    
    temp = (char *)malloc(n);
    if (temp == NULL) {
        return;
    }
    
    memcpy(temp, str, n);
    memmove(str, str + n, len - n);
    memcpy(str + (len - n), temp, n);
    
    free(temp);
}

    }
