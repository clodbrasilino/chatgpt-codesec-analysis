#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void remove_words_of_length_k(char *str, size_t max_size, int k) {
    if (str == NULL || k <= 0 || max_size == 0) {
        return;
    }

    size_t len = strnlen(str, max_size);
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return;
    }

    size_t i = 0;
    size_t j = 0;
    size_t word_start = 0;
    size_t word_len = 0;

    while (i <= len) {
        if (str[i] != ' ' && str[i] != '\0') {
            if (word_len == 0) {
                word_start = i;
            }
            word_len++;
        } else {
            if (word_len > 0) {
                if (word_len != (size_t)k) {
                    if (j > 0 && result[j - 1] != ' ') {
                        result[j++] = ' ';
                    }
                    for (size_t w = word_start; w < word_start + word_len; w++) {
                        result[j++] = str[w];
                    }
                }
                word_len = 0;
            }
        }
        i++;
    }

    result[j] = '\0';
    snprintf(str, max_size, "%s", result);
    free(result);
}

int main(void) {
    size_t buf_size = 256;
    char *str = (char *)malloc(buf_size);
    if (str == NULL) {
        return 1;
    }
    
    int k;

    if (fgets(str, (int)buf_size, stdin) != NULL) {
        size_t len = strnlen(str, buf_size);
        if (len > 0 && str[len - 1] == '\n') {
            str[len - 1] = '\0';
        }

        if (scanf("%d", &k) == 1) {
            remove_words_of_length_k(str, buf_size, k);
            printf("%s\n", str);
        }
    }

    free(str);
    return 0;
}
