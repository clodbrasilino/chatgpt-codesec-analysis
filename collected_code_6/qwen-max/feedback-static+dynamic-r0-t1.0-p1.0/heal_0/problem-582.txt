```c
#include <stdio.h>
#include <stdbool.h>

typedef struct {
    int count;
} Dictionary;

bool isDictionaryEmpty(const Dictionary *dict) {
    if (dict == NULL) {
        return true;
    }
    return dict->count == 0;
}

int main() {
    Dictionary myDict = {0};
    if (isDictionaryEmpty(&myDict)) {
        printf("Dictionary is empty\n");
    } else {
        printf("Dictionary is not empty\n");
    }
    return 0;
}
```
