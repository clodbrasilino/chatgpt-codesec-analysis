#include <stdio.h>
#include <stdlib.h>

int find_product_first_even_odd(const int *arr, size_t size, int *product) {
    if (arr == NULL || product == NULL || size == 0) {
        return -1;
    }

    int first_even = 0;
    int first_odd = 0;
    int found_even = 0;
    int found_odd = 0;

    for (size_t i = 0; i < size; ++i) {
        if (!found_even && arr[i] % 2 == 0) {
            first_even = arr[i];
            found_even = 1;
        } else if (!found_odd && arr[i] % 2 != 0) {
            first_odd = arr[i];
            found_odd = 1;
        }

        if (found_even && found_odd) {
            *product = first_even * first_odd;
            return 0;
        }
    }

    return -1;
}

int main(void) {
    int arr[] = {3, 7, 4, 9, 2, 8};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int product = 0;

    if (find_product_first_even_odd(arr, size, &product) == 0) {
        printf("Product: %d\n", product);
    } else {
        printf("Could not find both an even and an odd number.\n");
    }

    return EXIT_SUCCESS;
}
