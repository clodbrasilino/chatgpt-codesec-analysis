#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void insert_spaces(char *str) {
    regex_t regex;
    regmatch_t match[1];
    char pattern[] = "([A-Z][^A-Z]*)";

    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        fprintf(stderr, "Could not compile regex\n");
        exit(1);
    }

    size_t str_len = strlen(str);

    while (regexec(&regex, str, 1, match, 0) == 0) {
        size_t match_end = match[0].rm_eo;
        if (match_end + 1 < str_len) {
            memmove(str + match_end + 2, str + match_end + 1, str_len - match_end);
            str[match_end] = ' ';
            str[match_end + 1] = '\0';
            str += match_end + 1;
            str_len = strlen(str);
        } else {
            break;
        }
    }

    regfree(&regex);
}

int main() {
    char input[] = "HelloWorldThisIsATest";
    insert_spaces(input);
    printf("%s\n", input);
    return 0;
}
