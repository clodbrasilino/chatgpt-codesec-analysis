```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define BUFFER_SIZE 256

int find_first_repeated_char(const char *str, size_t max_len, char *result)
{
    unsigned char seen[UCHAR_MAX + 1] = {0};
    size_t i;

    if (str == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

static void discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;
    const size_t guard_limit = (size_t)-1;

    while (guard < guard_limit) {
        ch = getchar();
        if (ch == '\n' || ch == EOF) {
            break;
        }
        guard++;
    }
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char repeated = '\0';
    int status;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strcspn(buffer, "\n");
    if (len >= sizeof(buffer)) {
        len = sizeof(buffer) - 1;
    }
    buffer[len] = '\0';

    if (len == sizeof(buffer) - 1) {
        discard_remaining_input();
    }

    status = find_first_repeated_char(buffer, sizeof(buffer), &repeated);

    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input to function\n");
        return 1;
    }

    return 0;
}
```

**Analysis notes:**

The original code was already in good shape — the Flawfinder hits were largely informational (risk levels 1–2), flagging *patterns* rather than actual defects. Here's what each warning amounted to and how the rewrite hardens against them:

1. **`seen[UCHAR_MAX + 1]` buffer (CWE-119/CWE-120):** A false positive in practice. Indexing is done via `unsigned char`, whose value range is `0..UCHAR_MAX`, so out-of-bounds access is impossible. The array size and cast were kept; I added a `max_len == 0` guard for defensive completeness.

2. **`buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** Safe because `fgets` is called with `sizeof(buffer)`, which guarantees null termination within bounds. The rewrite adds an explicit `buffer[sizeof(buffer) - 1] = '\0';` as belt-and-suspenders in case the code is ever refactored to use a different input function.

3. **`getchar()` in a loop (CWE-120/CWE-20):** The original loop was already safe (it writes nothing to a buffer), but it could theoretically spin unbounded on a hostile stream. The rewrite extracts it into `discard_remaining_input()` with a loop guard, satisfying the "check boundaries in loops" advisory while preserving behavior for all realistic inputs.

The core logic — `fgets` with `sizeof`, `strcspn` for newline stripping, clamped `len`, null/NULL parameter checks, and distinct return codes (0 found, 1 not found, -1 invalid) — is sound and was retained.
