#include <stdio.h>
#include <stdlib.h>
#include <stdarg.h>

typedef enum {
    TYPE_INT,
    TYPE_DOUBLE,
    TYPE_FLOAT
} DataType;

typedef struct Node {
    DataType type;
    union {
        int i_val;
        double d_val;
        float f_val;
    } data;
    struct Node* next;
} Node;

Node* create_int_node(int value) {
    Node* node = (Node*)malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->type = TYPE_INT;
    node->data.i_val = value;
    node->next = NULL;
    return node;
}

Node* create_double_node(double value) {
    Node* node = (Node*)malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->type = TYPE_DOUBLE;
    node->data.d_val = value;
    node->next = NULL;
    return node;
}

Node* create_float_node(float value) {
    Node* node = (Node*)malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->type = TYPE_FLOAT;
    node->data.f_val = value;
    node->next = NULL;
    return node;
}

void free_list(Node* head) {
    Node* current = head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
}

double get_node_value(const Node* node) {
    if (node == NULL) {
        return 0.0;
    }
    switch (node->type) {
        case TYPE_INT:
            return (double)node->data.i_val;
        case TYPE_DOUBLE:
            return node->data.d_val;
        case TYPE_FLOAT:
            return (double)node->data.f_val;
        default:
            return 0.0;
    }
}

double find_minimum(Node* head) {
    if (head == NULL) {
        return 0.0;
    }
    
    double min_value = get_node_value(head);
    Node* current = head->next;
    
    while (current != NULL) {
        double current_value = get_node_value(current);
        if (current_value < min_value) {
            min_value = current_value;
        }
        current = current->next;
    }
    
    return min_value;
}

int main(void) {
    Node* head = create_int_node(42);
    if (head == NULL) {
        return 1;
    }
    
    Node* node2 = create_double_node(3.14);
    if (node2 == NULL) {
        free_list(head);
        return 1;
    }
    head->next = node2;
    
    Node* node3 = create_float_node(1.5f);
    if (node3 == NULL) {
        free_list(head);
        return 1;
    }
    node2->next = node3;
    
    Node* node4 = create_int_node(-10);
    if (node4 == NULL) {
        free_list(head);
        return 1;
    }
    node3->next = node4;
    
    Node* node5 = create_double_node(99.99);
    if (node5 == NULL) {
        free_list(head);
        return 1;
    }
    node4->next = node5;
    
    double min_val = find_minimum(head);
    printf("Minimum value: %.2f\n", min_val);
    
    free_list(head);
    
    return 0;
}
