#include <stdio.h>
#include <stdbool.h>

int findSingleElement(int arr[], int n) {
    int start = 0;
    int end = n - 1;

    while (start < end) {
        int mid = start + (end - start) / 2;
        if (mid % 2 == 0) {
            if (arr[mid] == arr[mid + 1]) {
                start = mid + 2;
            } else {
                end = mid;
            }
        } else {
            if (arr[mid] == arr[mid - 1]) {
                start = mid + 1;
            } else {
                end = mid;
            }
        }
    }

    return arr[start];
}

int main() {
    int arr[] = {1, 1, 2, 3, 3, 4, 4, 8, 8};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = findSingleElement(arr, n);
    printf("The element that appears only once is: %d\n", result);
    return 0;
}
