```c
#include <stdio.h>

int countSetBits(unsigned int num) {
    int count = 0;
    while (num > 0) {
        count += num & 1;
        num >>= 1;
    }
    return count;
}

int main() {
    unsigned int number = 29; // Example number
    printf("Number of set bits: %d\n", countSetBits(number));
    return 0;
}
```
