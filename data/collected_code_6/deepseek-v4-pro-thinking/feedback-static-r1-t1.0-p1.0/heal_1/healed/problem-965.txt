#include <ctype.h>
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

char *camel_to_snake(const char *str)
{
    size_t len = 0;
    size_t upper_count = 0;
    const char *p;
    char *result;
    size_t out_index = 0;
    size_t i;

    if (str == NULL) {
        return NULL;
    }

    p = str;
    while (*p) {
        if (isupper((unsigned char)*p)) {
            upper_count++;
        }
        len++;
        p++;
    }

    if (len > SIZE_MAX - upper_count - 1) {
        return NULL;
    }

    result = malloc(len + upper_count + 1);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];

        if (isupper(c)) {
            if (out_index > 0) {
                result[out_index++] = '_';
            }
            result[out_index++] = (char)tolower(c);
        } else {
            result[out_index++] = (char)c;
        }
    }

    result[out_index] = '\0';
    return result;
}

int main(void)
{
    const char *input = "camelCaseString";
    char *output = camel_to_snake(input);

    if (output == NULL) {
        return EXIT_FAILURE;
    }

    printf("%s\n", output);
    free(output);

    return EXIT_SUCCESS;
}
