#include <stdio.h>
#include <stdlib.h>

#define TUPLE_SIZE 3

int tupleContainsAllValidValues(const int *tuple, int elementsCount) {
    for (int i = 0; i < elementsCount; ++i) {
        if (tuple[i] < 1 || tuple[i] > 100) {
            return 0;
        }
    }
    return 1;
}

int main() {
    const int tuple[TUPLE_SIZE] = {50, 75, 3};
    if (tupleContainsAllValidValues(tuple, TUPLE_SIZE)) {
        printf("All values are valid.\n");
    } else {
        printf("Some values are not valid.\n");
    }

    const int invalidTuple[TUPLE_SIZE] = {0, -2, 101};
    if (tupleContainsAllValidValues(invalidTuple, TUPLE_SIZE)) {
        printf("All values are valid.\n");
    } else {
        printf("Some values are not valid.\n");
    }

    return 0;
}
