#include <stdint.h>
#include <stdbool.h>
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t length;
    size_t pos;
} Input;

typedef struct {
    Input *inputs;
    size_t count;
    size_t *heap;
    size_t heap_size;
} MergeIterator;

static int merge_input_compare(const MergeIterator *iter, size_t left, size_t right) {
    const Input *left_input = &iter->inputs[left];
    const Input *right_input = &iter->inputs[right];
    int left_value = left_input->data[left_input->pos];
    int right_value = right_input->data[right_input->pos];

    if (left_value != right_value) {
        return left_value < right_value ? -1 : 1;
    }
    if (left != right) {
        return left < right ? -1 : 1;
    }
    return 0;
}

static void merge_heap_sift_down(MergeIterator *iter, size_t index) {
    while (1) {
        size_t left = index * 2 + 1;
        size_t right = index * 2 + 2;
        size_t smallest = index;

        if (left < iter->heap_size &&
            merge_input_compare(iter, iter->heap[left], iter->heap[smallest]) < 0) {
            smallest = left;
        }
        if (right < iter->heap_size &&
            merge_input_compare(iter, iter->heap[right], iter->heap[smallest]) < 0) {
            smallest = right;
        }
        if (smallest == index) {
            break;
        }

        size_t temp = iter->heap[index];
        iter->heap[index] = iter->heap[smallest];
        iter->heap[smallest] = temp;
        index = smallest;
    }
}

static void merge_heap_build(MergeIterator *iter) {
    if (iter->heap_size > 1) {
        size_t i = iter->heap_size / 2;
        while (i > 0) {
            --i;
            merge_heap_sift_down(iter, i);
        }
    }
}

MergeIterator *merge_iterator_create(int **arrays, const size_t *lengths, size_t input_count) {
    if (arrays == NULL || lengths == NULL || input_count == 0) {
        return NULL;
    }
    if (input_count > SIZE_MAX / sizeof(Input) ||
        input_count > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    MergeIterator *iter = malloc(sizeof(*iter));
    if (iter == NULL) {
        return NULL;
    }

    iter->inputs = malloc(input_count * sizeof(*iter->inputs));
    if (iter->inputs == NULL) {
        free(iter);
        return NULL;
    }

    iter->heap = malloc(input_count * sizeof(*iter->heap));
    if (iter->heap == NULL) {
        free(iter->inputs);
        free(iter);
        return NULL;
    }

    iter->count = input_count;
    iter->heap_size = 0;

    for (size_t i = 0; i < input_count; ++i) {
        iter->inputs[i].data = arrays[i];
        iter->inputs[i].length = lengths[i];
        iter->inputs[i].pos = 0;

        if (lengths[i] > 0) {
            if (arrays[i] == NULL) {
                free(iter->heap);
                free(iter->inputs);
                free(iter);
                return NULL;
            }
            iter->heap[iter->heap_size++] = i;
        }
    }

    merge_heap_build(iter);
    return iter;
}

bool merge_iterator_next(MergeIterator *iter, int *value) {
    if (iter == NULL || value == NULL || iter->heap_size == 0) {
        return false;
    }

    size_t input_index = iter->heap[0];
    Input *input = &iter->inputs[input_index];

    *value = input->data[input->pos];
    ++input->pos;

    if (input->pos < input->length) {
        merge_heap_sift_down(iter, 0);
    } else {
        --iter->heap_size;
        if (iter->heap_size > 0) {
            iter->heap[0] = iter->heap[iter->heap_size];
            merge_heap_sift_down(iter, 0);
        }
    }

    return true;
}

void merge_iterator_free(MergeIterator *iter) {
    if (iter != NULL) {
        free(iter->heap);
        free(iter->inputs);
        free(iter);
    }
}

int main(void) {
    int first[] = {1, 3, 5, 7};
    int second[] = {2, 4, 6};
    int third[] = {0, 8, 9};

    int *arrays[] = {first, second, third};
    size_t lengths[] = {
        sizeof(first) / sizeof(first[0]),
        sizeof(second) / sizeof(second[0]),
        sizeof(third) / sizeof(third[0])
    };

    MergeIterator *iter = merge_iterator_create(arrays, lengths, 3);
    if (iter == NULL) {
        return EXIT_FAILURE;
    }

    int value;
    while (merge_iterator_next(iter, &value)) {
        printf("%d\n", value);
    }

    merge_iterator_free(iter);
    return EXIT_SUCCESS;
}
