```c
#include <stdio.h>
#include <stdlib.h>

int digit_distance(int num1, int num2) {
    int distance = 0;
    int digit1, digit2;
    
    num1 = abs(num1);
    num2 = abs(num2);
    
    while (num1 > 0 || num2 > 0) {
        digit1 = num1 % 10;
        digit2 = num2 % 10;
        
        distance += abs(digit1 - digit2);
        
        num1 /= 10;
        num2 /= 10;
    }
    
    return distance;
}

int main(void) {
    int num1, num2;
    int result;
    
    if (scanf("%d %d", &num1, &num2) != 2) {
        return EXIT_FAILURE;
    }
    
    result = digit_distance(num1, num2);
    printf("%d\n", result);
    
    return EXIT_SUCCESS;
}
```
