```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_INPUT_LEN 10U
#define OUTPUT_MIN_SIZE 11U
#define INPUT_BUF_SIZE 32U
#define OUTPUT_BUF_SIZE 16U

int convert_date(const char *input, char *output, size_t output_size);

int convert_date(const char *input, char *output, size_t output_size)
{
    unsigned int year;
    unsigned int month;
    unsigned int day;
    size_t i;
    size_t len;
    int written;
    static const unsigned int days_in_month[12] = {
        31U, 29U, 31U, 30U, 31U, 30U, 31U, 31U, 30U, 31U, 30U, 31U
    };

    if (input == NULL || output == NULL) {
        return -1;
    }

    if (output_size < OUTPUT_MIN_SIZE) {
        return -1;
    }

    len = strnlen(input, DATE_INPUT_LEN + 1U);
    if (len != DATE_INPUT_LEN) {
        return -1;
    }

    for (i = 0U; i < DATE_INPUT_LEN; i++) {
        if (i == 4U || i == 7U) {
            if (input[i] != '-') {
                return -1;
            }
        } else {
            if (!isdigit((unsigned char)input[i])) {
                return -1;
            }
        }
    }

    if (sscanf(input, "%4u-%2u-%2u", &year, &month, &day) != 3) {
        return -1;
    }

    if (year < 1U || year > 9999U) {
        return -1;
    }

    if (month < 1U || month > 12U) {
        return -1;
    }

    if (day < 1U || day > days_in_month[month - 1U]) {
        return -1;
    }

    if (month == 2U && day == 29U) {
        if (!((year % 4U == 0U && year % 100U != 0U) || (year % 400U == 0U))) {
            return -1;
        }
    }

    written = snprintf(output, output_size, "%02u-%02u-%04u", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_SIZE];
    char output[OUTPUT_BUF_SIZE];

    memset(input, 0, sizeof(input));
    memset(output, 0, sizeof(output));

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date format\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", output);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**CWE-119/CWE-120 (statically-sized `char` arrays):** Flawfinder flags `input[INPUT_BUF_SIZE]` and `output[OUTPUT_BUF_SIZE]` as potential overflow risks simply because fixed-size buffers *can* be misused. In this code, however, all accesses are properly bounded:

1. **`input` buffer:** Reads are done via `fgets(input, (int)sizeof(input), stdin)`, which never writes more than `sizeof(input) - 1` characters plus the null terminator. The explicit `input[sizeof(input) - 1U] = '\0'` is defensive redundancy, and `strcspn` safely strips the newline without exceeding bounds.

2. **`output` buffer:** Writes go through `snprintf(output, output_size, ...)` with the size passed explicitly (`sizeof(output)` from the caller), and the return value is checked for both errors (`written < 0`) and truncation (`(size_t)written >= output_size`). Additionally, `output_size < OUTPUT_MIN_SIZE` is validated upfront, guaranteeing room for the full `dd-mm-yyyy` result (10 chars + NUL = 11).

3. **Input validation:** `strnlen` with an explicit cap prevents reading past unterminated data, the character-by-character format check uses a bounded loop, `isdigit` receives an `unsigned char` cast (avoiding UB on negative `char` values), and `sscanf` uses width-limited specifiers (`%4u`, `%2u`) so it cannot over-read.

Since the warnings are informational (risk level 2) rather than actual defects, the correct "fix" is to retain the existing bounds-checking discipline — the rewritten code preserves all of these mitigations unchanged.
