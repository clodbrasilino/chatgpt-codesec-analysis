```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int loss;
} Transaction;

char *check_loss(Transaction t) {
    if (t.loss > 0) {
        char *result = malloc(12 * sizeof(char));
        if (result == NULL) {
            return "Memory Allocation Failed";
        }
        snprintf(result, 12, "Loss: %d", t.loss);
        return result;
    }
    return "None";
}

int main() {
    Transaction t = {100};
    char *result = check_loss(t);
    printf("%s\n", result);
    free(result);
    return 0;
}
```
