#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void find_max_product_pair(int *arr, int size, int *res) {
    if (arr == NULL || res == NULL || size < 2) {
        return;
    }

    int max1 = INT_MIN;
    int max2 = INT_MIN;
    int min1 = INT_MAX;
    int min2 = INT_MAX;

    for (int i = 0; i < size; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    long long prod1 = (long long)max1 * max2;
    long long prod2 = (long long)min1 * min2;

    if (prod1 > prod2) {
        res[0] = max1;
        res[1] = max2;
    } else {
        res[0] = min1;
        res[1] = min2;
    }
}

int main() {
    int arr[] = {1, -4, 3, -6, 7, 0};
    int size = sizeof(arr) / sizeof(arr[0]);
    int res[2];

    find_max_product_pair(arr, size, res);

    if (size >= 2) {
        printf("%d %d\n", res[0], res[1]);
    }

    return 0;
}
