#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void split_at_lowercase(const char *input) {
    size_t len = 0;
    while (input[len] != '\0' && isprint((unsigned char)input[len])) {
        len++;
    }
    const char *start = input;
    for (size_t i = 0; i <= len; ++i) {
        if (i == len || !islower((unsigned char)input[i])) {
            if (i > 0 && islower((unsigned char)input[i - 1])) {
                size_t token_len = i - (start - input);
                char *token = (char *)malloc((token_len + 1) * sizeof(char));
                if (token != NULL) {
                    memcpy(token, start, token_len);
                    token[token_len] = '\0';
                    printf("%s\n", token);
                    free(token);
                }
            }
            start = input + i;
        }
    }
}

int main(int argc, const char *argv[]) {
    if (argc < 2) {
        return EXIT_FAILURE;
    }
    split_at_lowercase(argv[1]);
    return EXIT_SUCCESS;
}
