```c
#include <stdio.h>
#include <string.h>

int match_words(const char *word1, const char *word2) {
    return strcmp(word1, word2) == 0;
}

int main() {
    const char *words[] = {"python", "programming", "project", "practice", "passion"};
    int found = 0;

    for (int i = 0; i < 5; i++) {
        if (words[i][0] == 'p') {
            for (int j = i + 1; j < 5; j++) {
                if (words[j][0] == 'p' && match_words(words[i], words[j])) {
                    printf("Matched: %s and %s\n", words[i], words[j]);
                    found = 1;
                }
            }
        }
    }

    if (!found) {
        printf("No matches found.\n");
    }

    return 0;
}
```
