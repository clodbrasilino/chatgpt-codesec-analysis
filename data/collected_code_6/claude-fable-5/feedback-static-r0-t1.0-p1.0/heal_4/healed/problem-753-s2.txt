```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_LEN 32

typedef struct {
    char name[NAME_LEN];
    int value;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static Tuple *find_min_k_records(const Tuple *records, size_t n, size_t k)
{
    Tuple *sorted;
    Tuple *result;
    size_t i;
    size_t sorted_bytes;
    size_t result_bytes;

    if (records == NULL || n == 0U || k == 0U || k > n) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(*sorted)) {
        return NULL;
    }

    sorted_bytes = n * sizeof(*sorted);
    sorted = malloc(sorted_bytes);
    if (sorted == NULL) {
        return NULL;
    }

    memcpy(sorted, records, sorted_bytes);

    for (i = 0U; i < n; i++) {
        sorted[i].name[NAME_LEN - 1] = '\0';
    }

    qsort(sorted, n, sizeof(*sorted), compare_tuples);

    if (k > SIZE_MAX / sizeof(*result)) {
        free(sorted);
        return NULL;
    }

    result_bytes = k * sizeof(*result);
    result = malloc(result_bytes);
    if (result == NULL) {
        free(sorted);
        return NULL;
    }

    memcpy(result, sorted, result_bytes);
    free(sorted);

    return result;
}

int main(void)
{
    Tuple records[] = {
        { "Alice", 45 },
        { "Bob", 12 },
        { "Carol", 78 },
        { "Dave", 3 },
        { "Eve", 56 },
        { "Frank", 23 }
    };
    size_t n = sizeof(records) / sizeof(records[0]);
    size_t k = 3U;
    Tuple *min_records;
    size_t i;

    if (k > n) {
        fprintf(stderr, "Invalid k value\n");
        return EXIT_FAILURE;
    }

    min_records = find_min_k_records(records, n, k);
    if (min_records == NULL) {
        fprintf(stderr, "Failed to find minimum records\n");
        return EXIT_FAILURE;
    }

    printf("Minimum %zu records:\n", k);
    for (i = 0U; i < k; i++) {
        printf("%.*s: %d\n", NAME_LEN - 1, min_records[i].name, min_records[i].value);
    }

    free(min_records);

    return EXIT_SUCCESS;
}
```

Analysis of the fixes applied to the identified weaknesses:

**1. Statically-sized array `char name[NAME_LEN]` (CWE-119/CWE-120):**
- Every `Tuple` copied into the working buffer is forcibly null-terminated via `sorted[i].name[NAME_LEN - 1] = '\0'`, guaranteeing that no string operation can read past the array boundary, even if the caller passed non-terminated data.
- Output uses `%.*s` with an explicit maximum field width of `NAME_LEN - 1`, so `printf` can never read beyond the buffer regardless of termination state.

**2. First `memcpy` into `sorted` (CWE-120):**
- The destination size is computed once (`sorted_bytes = n * sizeof(*sorted)`) and the exact same value is used for both the `malloc` and the `memcpy`, making a destination/source size mismatch impossible.
- The multiplication is guarded against `size_t` overflow (`n > SIZE_MAX / sizeof(*sorted)`) before allocation, preventing an undersized allocation followed by a heap overflow.

**3. Second `memcpy` into `result` (CWE-120):**
- Identical pattern: overflow check on `k * sizeof(*result)`, a single `result_bytes` variable shared between allocation and copy, and the invariant `k <= n` (validated at function entry) guarantees the source buffer `sorted` contains at least `k` elements to read.

Additional robustness already present and preserved: `NULL`/zero-size parameter validation, `malloc` failure handling with proper cleanup (no leaks on error paths), and a `qsort` comparator that avoids integer-overflow pitfalls of the naive `return a - b` idiom.
