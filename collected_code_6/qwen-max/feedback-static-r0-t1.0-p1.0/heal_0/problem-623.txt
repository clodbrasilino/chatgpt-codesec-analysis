#include <stdio.h>
#include <stdlib.h>

double power(double base, int exp) {
    double result = 1.0;
    for (int i = 0; i < exp; ++i) {
        result *= base;
    }
    return result;
}

void nth_power_list(double *list, size_t length, int n) {
    for (size_t i = 0; i < length; ++i) {
        list[i] = power(list[i], n);
    }
}

int main() {
    double list[] = {2.0, 3.0, 4.0};
    size_t length = sizeof(list) / sizeof(list[0]);
    int n = 3;

    nth_power_list(list, length, n);

    for (size_t i = 0; i < length; ++i) {
        printf("%.2f ", list[i]);
    }
    printf("\n");

    return 0;
}
