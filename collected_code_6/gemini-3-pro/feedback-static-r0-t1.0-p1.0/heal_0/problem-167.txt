#include <stdio.h>
#include <stdint.h>

uint32_t next_power_of_2(uint32_t n) {
    if (n == 0) {
        return 1;
    }
    
    n--;
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    n++;
    
    return n;
}

int main(void) {
    uint32_t test_values[] = {0, 1, 2, 3, 4, 5, 15, 16, 17, 1000};
    size_t num_values = sizeof(test_values) / sizeof(test_values[0]);

    for (size_t i = 0; i < num_values; i++) {
        uint32_t result = next_power_of_2(test_values[i]);
        printf("%u\n", result);
    }

    return 0;
}
