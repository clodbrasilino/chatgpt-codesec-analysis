#include <stdio.h>
#include <stdlib.h>

int maxSubsequenceLength(int arr[], int n, int diff) {
    if (n <= 0) {
        return 0;
    }

    int *dp = (int *)calloc(n, sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    int max_len = 1;

    for (int i = 0; i < n; i++) {
        dp[i] = 1;
        for (int j = 0; j < i; j++) {
            if (arr[i] - arr[j] == diff || arr[j] - arr[i] == diff) {
                if (dp[j] + 1 > dp[i]) {
                    dp[i] = dp[j] + 1;
                }
            }
        }
        if (dp[i] > max_len) {
            max_len = dp[i];
        }
    }

    free(dp);
    return max_len;
}

int main() {
    int arr[] = {1, 5, 3, 4, 2};
    int diff = 1;
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = maxSubsequenceLength(arr, n, diff);
    printf("%d\n", result);

    return 0;
}
