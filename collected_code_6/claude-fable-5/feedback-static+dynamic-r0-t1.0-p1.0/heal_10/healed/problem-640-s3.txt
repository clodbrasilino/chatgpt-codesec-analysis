#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *tests[] = {
        "python(java)",
        "string(remove)",
        "alpha(beta(gamma)delta)",
        "Hello (remove me) World",
        "abc(def(ghi)jkl)mno",
        "no parentheses here"
    };
    const char *expected[] = {
        "python",
        "string",
        "alpha",
        "Hello  World",
        "abcmno",
        "no parentheses here"
    };
    size_t num_tests;
    size_t i;
    int status;

    status = EXIT_SUCCESS;
    num_tests = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < num_tests; i++) {
        char *output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (strcmp(output, expected[i]) != 0) {
            fprintf(stderr, "test case %zu failed: expected '%s', got '%s'\n",
                    i, expected[i], output);
            status = EXIT_FAILURE;
        }
        if (printf("%s\n", output) < 0) {
            free(output);
            return EXIT_FAILURE;
        }
        free(output);
        output = NULL;
    }

    return status;
}

1. **Missing null terminator** — writing characters into the buffer but never terminating with `'\0'`, so `strcmp`/`printf` read past the valid data into uninitialized heap bytes.
2. **Undersized allocation** — e.g., `malloc(len)` instead of `len + 1`, leaving no room for the terminator (heap overflow when writing it).

The version above is defensively correct against all of those:

- `calloc(len + 1, sizeof(char))` guarantees a zero-initialized buffer with room for the terminator, so even a missed `result[j] = '\0'` couldn't produce garbage.
- The explicit `result[j] = '\0';` terminates the string regardless.
- `j` can never exceed `len`, so no out-of-bounds write is possible.
- `NULL` input and allocation failure are both handled, and every allocation is freed on all exit paths (no leaks, no double-free, pointer set to `NULL` after free).
