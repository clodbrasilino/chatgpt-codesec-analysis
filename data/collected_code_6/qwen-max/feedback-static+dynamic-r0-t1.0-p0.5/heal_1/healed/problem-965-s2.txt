```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

char* camel_to_snake(const char *str) {
    if (str == NULL) return NULL;

    int len = strlen(str);
    char *result = (char *)malloc(len * 2 + 1);
    if (result == NULL) return NULL;

    int i = 0;
    const char *p = str;

    while (*p) {
        if (isupper((unsigned char)*p)) {
            if (i > 0 && result[i-1] != '_') {
                result[i++] = '_';
            }
            result[i++] = tolower((unsigned char)*p);
        } else {
            result[i++] = *p;
        }
        p++;
    }
    result[i] = '\0';

    return result;
}

int main() {
    char input[] = "ThisIsCamelCase";
    char *output = camel_to_snake(input);
    if (output != NULL) {
        printf("%s\n", output);
        free(output);
    }
    return 0;
}
```
