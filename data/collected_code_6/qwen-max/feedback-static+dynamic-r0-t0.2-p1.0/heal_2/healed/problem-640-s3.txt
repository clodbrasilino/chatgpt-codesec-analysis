#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* remove_parenthesis_area(const char *input) {
    if (input == NULL) {
        return NULL;
    }
    int len = strlen(input);
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }
    int result_index = 0;
    int depth = 0;

    for (int i = 0; i <= len; ++i) { // Include the null terminator in the loop
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            depth--;
        } else if (depth == 0) {
            result[result_index++] = input[i];
        }
    }

    result[result_index] = '\0';
    return result;
}

int main() {
    const char *test_cases[] = {
        "python (language)",
        "(alpha) test",
        "example (with (nested) parentheses) string"
    };
    const char *expected_results[] = {
        "python ",
        " test",
        "example  string"
    };
    int num_tests = sizeof(test_cases) / sizeof(test_cases[0]);

    for (int i = 0; i < num_tests; ++i) {
        char *output = remove_parenthesis_area(test_cases[i]);
        if (output != NULL) {
            if (strcmp(output, expected_results[i]) == 0) {
                printf("Test case %d passed: %s\n", i, output);
            } else {
                printf("Test case %d failed: expected '%s', got '%s'\n", i, expected_results[i], output);
            }
            free(output);
        } else {
            fprintf(stderr, "Memory allocation failed for test case %d\n", i);
        }
    }
    return 0;
}
