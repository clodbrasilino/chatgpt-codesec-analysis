```c
#include <stdio.h>
#include <stdlib.h>

int calculate_square_perimeter(double side, double *perimeter) {
    if (side < 0.0 || perimeter == NULL) {
        return -1;
    }
    
    *perimeter = 4.0 * side;
    
    return 0;
}

int main(void) {
    double side_length = 5.5;
    double square_perimeter = 0.0;
    int status_code = 0;

    status_code = calculate_square_perimeter(side_length, &square_perimeter);

    if (status_code != 0) {
        fprintf(stderr, "Invalid input for calculating the perimeter of a square.\n");
        return EXIT_FAILURE;
    }

    printf("The perimeter of a square with side %f is %f\n", side_length, square_perimeter);

    return EXIT_SUCCESS;
}
```
