```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_KEY_LEN 64

typedef struct {
    char key[MAX_KEY_LEN];
    int value;
} DictEntry;

static int dict_entry_init(DictEntry *entry, const char *key, int value)
{
    size_t key_len;

    if (entry == NULL || key == NULL) {
        return -1;
    }

    key_len = strnlen(key, MAX_KEY_LEN);
    if (key_len >= MAX_KEY_LEN) {
        return -1;
    }

    memcpy(entry->key, key, key_len);
    entry->key[key_len] = '\0';
    entry->value = value;
    return 0;
}

static int contains(const int *arr, size_t count, int value)
{
    size_t i;

    if (arr == NULL) {
        return 0;
    }

    for (i = 0; i < count; i++) {
        if (arr[i] == value) {
            return 1;
        }
    }
    return 0;
}

static int *extract_unique_values(const DictEntry *dict, size_t dict_size, size_t *unique_count)
{
    int *unique;
    size_t i;
    size_t count;

    if (dict == NULL || unique_count == NULL || dict_size == 0U) {
        if (unique_count != NULL) {
            *unique_count = 0U;
        }
        return NULL;
    }

    if (dict_size > SIZE_MAX / sizeof(*unique)) {
        *unique_count = 0U;
        return NULL;
    }

    unique = malloc(dict_size * sizeof(*unique));
    if (unique == NULL) {
        *unique_count = 0U;
        return NULL;
    }

    count = 0U;
    for (i = 0; i < dict_size; i++) {
        if (!contains(unique, count, dict[i].value)) {
            unique[count] = dict[i].value;
            count++;
        }
    }

    *unique_count = count;
    return unique;
}

int main(void)
{
    static const struct {
        const char *key;
        int value;
    } input[] = {
        {"apple", 10},
        {"banana", 20},
        {"cherry", 10},
        {"date", 30},
        {"elderberry", 20},
        {"fig", 40}
    };
    size_t dict_size = sizeof(input) / sizeof(input[0]);
    DictEntry dict[sizeof(input) / sizeof(input[0])];
    size_t unique_count = 0U;
    size_t i;
    int *unique_values;

    for (i = 0; i < dict_size; i++) {
        if (dict_entry_init(&dict[i], input[i].key, input[i].value) != 0) {
            fprintf(stderr, "Invalid dictionary entry at index %zu\n", i);
            return EXIT_FAILURE;
        }
    }

    printf("Dictionary contents:\n");
    for (i = 0; i < dict_size; i++) {
        printf("  %s = %d\n", dict[i].key, dict[i].value);
    }

    unique_values = extract_unique_values(dict, dict_size, &unique_count);
    if (unique_values == NULL) {
        fprintf(stderr, "Failed to extract unique values\n");
        return EXIT_FAILURE;
    }

    printf("Unique values:");
    for (i = 0; i < unique_count; i++) {
        printf(" %d", unique_values[i]);
    }
    printf("\n");

    free(unique_values);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two Flawfinder findings flagged in the original source were already properly mitigated, so the rewrite preserves those defenses rather than restructuring them:

1. **CWE-119/CWE-120 (statically-sized `key` buffer):** Mitigated in `dict_entry_init` by using `strnlen(key, MAX_KEY_LEN)` and rejecting any key where `key_len >= MAX_KEY_LEN`, guaranteeing room for the NUL terminator. `strnlen` (rather than `strlen`) also protects against unterminated input strings.

2. **CWE-120 (`memcpy` without bounds check):** Safe because the copy length is validated before the `memcpy` executes — the destination is `MAX_KEY_LEN` bytes and at most `MAX_KEY_LEN - 1` bytes plus an explicit terminator are ever written.

Additional hardening retained/added:

- **Integer overflow guard** on the `malloc` size computation (`dict_size > SIZE_MAX / sizeof(*unique)`), preventing an undersized allocation (CWE-190 → CWE-131).
- **NULL checks** on all pointer parameters, including a defensive `arr == NULL` check added in `contains`.
- **Allocation failure handling** for `malloc`, with `*unique_count` zeroed on every error path so the caller never iterates over an invalid buffer.
- **Single ownership of the heap buffer** with a matching `free`, avoiding leaks or double-frees.

The code is C99-conformant (`strnlen` requires POSIX or C11 Annex K availability; on strict C99 toolchains compile with `-D_POSIX_C_SOURCE=200809L`).
