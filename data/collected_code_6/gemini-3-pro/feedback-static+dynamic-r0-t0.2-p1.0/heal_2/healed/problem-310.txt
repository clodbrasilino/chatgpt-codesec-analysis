#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char **elements;
    size_t count;
} Tuple;

Tuple* string_to_tuple(const char *str) {
    if (!str) {
        return NULL;
    }

    Tuple *tuple = malloc(sizeof(Tuple));
    if (!tuple) {
        return NULL;
    }

    size_t len = 0;
    while (str[len] != '\0') {
        len++;
    }

    tuple->count = len;
    if (len == 0) {
        tuple->elements = NULL;
        return tuple;
    }

    tuple->elements = malloc(len * sizeof(char *));
    if (!tuple->elements) {
        free(tuple);
        return NULL;
    }

    for (size_t i = 0; i < len; i++) {
        tuple->elements[i] = malloc(2);
        if (tuple->elements[i]) {
            tuple->elements[i][0] = str[i];
            tuple->elements[i][1] = '\0';
        }
    }

    return tuple;
}

void free_tuple(Tuple *tuple) {
    if (tuple) {
        for (size_t i = 0; i < tuple->count; i++) {
            free(tuple->elements[i]);
        }
        free(tuple->elements);
        free(tuple);
    }
}

int main(void) {
    Tuple *t1 = string_to_tuple("item1");
    if (t1) {
        for (size_t i = 0; i < t1->count; i++) {
            printf("Element %zu: %s\n", i, t1->elements[i]);
        }
        free_tuple(t1);
    }

    Tuple *t2 = string_to_tuple("15.10");
    if (t2) {
        for (size_t i = 0; i < t2->count; i++) {
            printf("Element %zu: %s\n", i, t2->elements[i]);
        }
        free_tuple(t2);
    }

    Tuple *t3 = string_to_tuple("python3.0");
    if (t3) {
        for (size_t i = 0; i < t3->count; i++) {
            printf("Element %zu: %s\n", i, t3->elements[i]);
        }
        free_tuple(t3);
    }

    return 0;
}
