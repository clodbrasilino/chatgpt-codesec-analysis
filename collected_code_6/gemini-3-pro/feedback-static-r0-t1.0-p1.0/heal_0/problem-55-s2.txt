```c
#include <stdio.h>
#include <math.h>
#include <stdlib.h>

double find_geometric_term(double a, double r, int n) {
    if (n < 1) {
        return 0.0;
    }
    return a * pow(r, (double)(n - 1));
}

int main() {
    double a;
    double r;
    int n;
    double result;

    if (scanf("%lf", &a) != 1) {
        return EXIT_FAILURE;
    }
    if (scanf("%lf", &r) != 1) {
        return EXIT_FAILURE;
    }
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }

    if (n < 1) {
        return EXIT_FAILURE;
    }

    result = find_geometric_term(a, r, n);
    printf("%f\n", result);

    return EXIT_SUCCESS;
}
```
