```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long decagonal_number(long long n)
{
    return 4LL * n * n - 3LL * n;
}

int main(void)
{
    long long n = 0;
    int result = 0;

    printf("Enter n: ");
    result = scanf("%lld", &n);

    if (result != 1) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 1) {
        fprintf(stderr, "Error: n must be a positive integer\n");
        return EXIT_FAILURE;
    }

    if (n > 1518500249LL) {
        fprintf(stderr, "Error: n too large, result would overflow\n");
        return EXIT_FAILURE;
    }

    printf("The %lld-th decagonal number is %lld\n", n, decagonal_number(n));

    return EXIT_SUCCESS;
}
```
