#include <stdio.h>
#include <stdlib.h>

unsigned long long calculate_pell(int n) {
    if (n < 0) {
        return 0; 
    }
    if (n == 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }

    unsigned long long a = 0;
    unsigned long long b = 1;
    unsigned long long c = 0;

    for (int i = 2; i <= n; i++) {
        c = 2 * b + a;
        a = b;
        b = c;
    }

    return c;
}

int main(void) {
    int n = 0;
    
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }

    if (n < 0 || n > 93) { 
        return EXIT_FAILURE;
    }

    unsigned long long result = calculate_pell(n);
    printf("%llu\n", result);

    return EXIT_SUCCESS;
}
