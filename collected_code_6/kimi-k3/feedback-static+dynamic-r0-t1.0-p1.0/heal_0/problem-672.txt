#include <stdio.h>

int max_of_three(int a, int b, int c)
{
    int max = a;
    
    if (b > max)
    {
        max = b;
    }
    
    if (c > max)
    {
        max = c;
    }
    
    return max;
}

int main(void)
{
    int num1;
    int num2;
    int num3;
    int result;
    int scan_result;
    
    printf("Enter three integers: ");
    scan_result = scanf("%d %d %d", &num1, &num2, &num3);
    
    if (scan_result != 3)
    {
        fprintf(stderr, "Error: Invalid input\n");
        return 1;
    }
    
    result = max_of_three(num1, num2, num3);
    printf("Maximum: %d\n", result);
    
    return 0;
}
