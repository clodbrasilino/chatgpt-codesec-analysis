#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    void *data;
    size_t element_size;
    size_t length;
} Tuple;

size_t count_occurrences(const Tuple *tuple, const void *element, int (*compare)(const void *, const void *)) {
    size_t count = 0;
    size_t i;
    
    if (tuple == NULL || element == NULL || compare == NULL || tuple->data == NULL) {
        return 0;
    }
    
    for (i = 0; i < tuple->length; i++) {
        void *current = (char *)tuple->data + (i * tuple->element_size);
        if (compare(current, element) == 0) {
            count++;
        }
    }
    
    return count;
}

int compare_int(const void *a, const void *b) {
    if (a == NULL || b == NULL) {
        return -1;
    }
    
    int int_a = *(const int *)a;
    int int_b = *(const int *)b;
    
    if (int_a < int_b) {
        return -1;
    } else if (int_a > int_b) {
        return 1;
    }
    return 0;
}

int main(void) {
    int values[] = {1, 2, 3, 2, 4, 2, 5};
    Tuple tuple;
    int target = 2;
    size_t result;
    
    tuple.data = values;
    tuple.element_size = sizeof(int);
    tuple.length = sizeof(values) / sizeof(values[0]);
    
    result = count_occurrences(&tuple, &target, compare_int);
    
    printf("Element %d occurs %zu times in the tuple.\n", target, result);
    
    return 0;
}
