```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count);
void free_shortlist(char **list, size_t count);

static void cleanup_partial(char **result, size_t filled)
{
    size_t j;

    for (j = 0; j < filled; j++) {
        free(result[j]);
    }
    free(result);
}

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count)
{
    char **result = NULL;
    size_t matched = 0;
    size_t i;

    if (words == NULL || result_count == NULL) {
        if (result_count != NULL) {
            *result_count = 0;
        }
        return NULL;
    }

    *result_count = 0;

    for (i = 0; i < count; i++) {
        if (words[i] != NULL) {
            size_t len = strnlen(words[i], MAX_WORD_LEN);
            if (len > n && len < MAX_WORD_LEN) {
                matched++;
            }
        }
    }

    if (matched == 0) {
        return NULL;
    }

    if (matched > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(matched * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count && *result_count < matched; i++) {
        if (words[i] != NULL) {
            size_t len = strnlen(words[i], MAX_WORD_LEN);
            if (len > n && len < MAX_WORD_LEN) {
                size_t copy_size = len + 1;
                char *copy = malloc(copy_size);

                if (copy == NULL) {
                    cleanup_partial(result, *result_count);
                    *result_count = 0;
                    return NULL;
                }

                memcpy(copy, words[i], len);
                copy[len] = '\0';

                result[*result_count] = copy;
                (*result_count)++;
            }
        }
    }

    if (*result_count == 0) {
        free(result);
        return NULL;
    }

    return result;
}

void free_shortlist(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *const words[] = {
        "apple", "cat", "banana", "dog", "elephant", "sun", "computer"
    };
    size_t word_count = sizeof(words) / sizeof(words[0]);
    size_t n = 4;
    size_t result_count = 0;
    char **shortlisted;
    size_t i;

    shortlisted = shortlist_words(words, word_count, n, &result_count);

    if (shortlisted == NULL) {
        if (printf("No words longer than %zu found.\n", n) < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("Words longer than %zu characters:\n", n) < 0) {
        free_shortlist(shortlisted, result_count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_count; i++) {
        if (printf("%s\n", shortlisted[i]) < 0) {
            free_shortlist(shortlisted, result_count);
            return EXIT_FAILURE;
        }
    }

    free_shortlist(shortlisted, result_count);

    return EXIT_SUCCESS;
}
```

**Summary of changes made:**

1. **Removed the dead `len > SIZE_MAX - 1` check** — since `len < MAX_WORD_LEN` is already guaranteed by the filter condition, this overflow check was unreachable and `copy_size = len + 1` can never overflow.

2. **Removed the always-true `copy_size > len` condition** — flagged by `knownConditionTrueFalse`. Since `copy_size` is assigned `len + 1` and overflow is impossible (len is bounded by `MAX_WORD_LEN`), the condition and its unreachable `else` branch were eliminated.

3. **Addressed the Flawfinder `memcpy` warning (CWE-120)** — the copy is provably safe: the destination buffer is allocated with exactly `len + 1` bytes, `memcpy` writes exactly `len` bytes, and the null terminator is written at index `len`. The destination always holds the source data, so the copy remains but with the invariant now structurally guaranteed.

4. **Factored the duplicated cleanup loops** into a static `cleanup_partial()` helper, reducing the chance of future maintenance errors in the error-handling paths.

All original safety properties are preserved: null-pointer guards, `strnlen` bounding against unterminated strings, the `SIZE_MAX / sizeof(char *)` allocation-overflow check, full cleanup on allocation failure, and checked `printf` return values in `main`.
