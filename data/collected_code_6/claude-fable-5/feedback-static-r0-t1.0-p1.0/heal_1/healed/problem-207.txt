#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 1024U

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

static int longest_repeating_subsequence(const char *str, size_t n)
{
    size_t i;
    size_t j;
    int result;
    int **dp;

    if (str == NULL || n == 0U) {
        return 0;
    }

    dp = (int **)malloc((n + 1U) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0U; i <= n; i++) {
        dp[i] = (int *)calloc(n + 1U, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0U; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1U; i <= n; i++) {
        for (j = 1U; j <= n; j++) {
            if ((str[i - 1U] == str[j - 1U]) && (i != j)) {
                dp[i][j] = dp[i - 1U][j - 1U] + 1;
            } else {
                dp[i][j] = max_int(dp[i - 1U][j], dp[i][j - 1U]);
            }
        }
    }

    result = dp[n][n];

    for (i = 0U; i <= n; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;
    int result;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
        len--;
    }

    result = longest_repeating_subsequence(buffer, len);
    if (result < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Length of the longest repeating subsequence: %d\n", result);

    return EXIT_SUCCESS;
}

**Analysis of the original code and applied fixes:**

1. **CWE-119/CWE-120 (statically-sized `buffer[1024]`)** — The original code was already safe here because `fgets(buffer, sizeof(buffer), stdin)` bounds the read to the buffer size. The rewrite keeps this pattern and adds a defensive explicit null terminator at `buffer[sizeof(buffer) - 1U]` to guarantee termination regardless of libc behavior.

2. **CWE-126 (`strlen` over-read on non-terminated strings)** — `fgets` always null-terminates on success, so the original was technically safe, but the rewrite replaces `strlen` with `strnlen(buffer, sizeof(buffer))`, which caps the scan at the buffer boundary and eliminates any possibility of an over-read even if the termination guarantee were violated.

3. **Existing strengths preserved** — The DP allocation already had correct cleanup on partial `calloc` failure (freeing `dp[0..i-1]` then `dp`), a `-1` error sentinel distinguished from valid results, and `NULL`/zero-length guards; all of that logic is retained unchanged.

4. **Minor addition** — `fflush(stdout)` after the prompt ensures the prompt appears before blocking on `fgets` when stdout is fully buffered (e.g., piped output).
