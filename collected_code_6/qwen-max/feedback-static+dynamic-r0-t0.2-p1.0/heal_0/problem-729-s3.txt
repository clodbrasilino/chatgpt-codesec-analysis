#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node* create_node(int data) {
    Node *new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        exit(EXIT_FAILURE);
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

void append(Node **head, int data) {
    Node *new_node = create_node(data);
    if (*head == NULL) {
        *head = new_node;
    } else {
        Node *current = *head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
}

Node* add_lists(Node *list1, Node *list2) {
    Node *result = NULL;
    Node *current1 = list1;
    Node *current2 = list2;
    while (current1 != NULL || current2 != NULL) {
        int sum = 0;
        if (current1 != NULL) {
            sum += current1->data;
            current1 = current1->next;
        }
        if (current2 != NULL) {
            sum += current2->data;
            current2 = current2->next;
        }
        append(&result, sum);
    }
    return result;
}

void print_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        printf("%d -> ", current->data);
        current = current->next;
    }
    printf("NULL\n");
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int main() {
    Node *list1 = NULL;
    Node *list2 = NULL;

    append(&list1, 1);
    append(&list1, 2);
    append(&list1, 3);

    append(&list2, 4);
    append(&list2, 5);
    append(&list2, 6);

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);

    Node *result = add_lists(list1, list2);
    printf("Result: ");
    print_list(result);

    free_list(list1);
    free_list(list2);
    free_list(result);

    return 0;
}
