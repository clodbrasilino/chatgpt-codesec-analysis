#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *items;
    int *counts;
    int size;
    int capacity;
} ElementCollection;

int init_collection(ElementCollection *collection, int capacity) {
    if (collection == NULL || capacity <= 0) {
        return -1;
    }
    
    collection->items = (int *)malloc(sizeof(int) * capacity);
    if (collection->items == NULL) {
        return -1;
    }
    
    collection->counts = (int *)malloc(sizeof(int) * capacity);
    if (collection->counts == NULL) {
        free(collection->items);
        collection->items = NULL;
        return -1;
    }
    
    collection->size = 0;
    collection->capacity = capacity;
    return 0;
}

void free_collection(ElementCollection *collection) {
    if (collection != NULL) {
        free(collection->items);
        free(collection->counts);
        collection->items = NULL;
        collection->counts = NULL;
        collection->size = 0;
        collection->capacity = 0;
    }
}

int add_element(ElementCollection *collection, int item, int count) {
    if (collection == NULL || count <= 0) {
        return -1;
    }
    
    if (collection->size >= collection->capacity) {
        return -1;
    }
    
    collection->items[collection->size] = item;
    collection->counts[collection->size] = count;
    collection->size++;
    return 0;
}

int iterate_elements(ElementCollection *collection, int **output, int *output_size) {
    if (collection == NULL || output == NULL || output_size == NULL) {
        return -1;
    }
    
    int total_elements = 0;
    for (int i = 0; i < collection->size; i++) {
        if (collection->counts[i] > 0) {
            total_elements += collection->counts[i];
        }
    }
    
    if (total_elements == 0) {
        *output = NULL;
        *output_size = 0;
        return 0;
    }
    
    int *result = (int *)malloc(sizeof(int) * total_elements);
    if (result == NULL) {
        return -1;
    }
    
    int index = 0;
    for (int i = 0; i < collection->size; i++) {
        for (int j = 0; j < collection->counts[i]; j++) {
            result[index++] = collection->items[i];
        }
    }
    
    *output = result;
    *output_size = total_elements;
    return 0;
}

int main(void) {
    ElementCollection collection;
    int *output = NULL;
    int output_size = 0;
    
    if (init_collection(&collection, 5) != 0) {
        fprintf(stderr, "Failed to initialize collection\n");
        return EXIT_FAILURE;
    }
    
    add_element(&collection, 1, 3);
    add_element(&collection, 2, 2);
    add_element(&collection, 3, 1);
    add_element(&collection, 4, 4);
    add_element(&collection, 5, 2);
    
    if (iterate_elements(&collection, &output, &output_size) != 0) {
        fprintf(stderr, "Failed to iterate elements\n");
        free_collection(&collection);
        return EXIT_FAILURE;
    }
    
    for (int i = 0; i < output_size; i++) {
        printf("%d ", output[i]);
    }
    printf("\n");
    
    free(output);
    free_collection(&collection);
    
    return EXIT_SUCCESS;
}
