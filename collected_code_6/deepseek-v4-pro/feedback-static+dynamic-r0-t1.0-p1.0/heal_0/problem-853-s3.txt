```c
#include <stdio.h>

int sum_of_odd_factors(int n) {
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        if (n % i == 0 && i % 2 != 0) {
            sum += i;
        }
    }
    return sum;
}

int main() {
    int num;
    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    if (num <= 0) {
        printf("Please enter a positive integer.\n");
        return 1;
    }
    printf("Sum of odd factors of %d is %d\n", num, sum_of_odd_factors(num));
    return 0;
}
```
