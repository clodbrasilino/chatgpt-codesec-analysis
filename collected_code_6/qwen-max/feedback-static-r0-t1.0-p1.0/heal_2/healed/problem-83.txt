```c
#include <stdio.h>
#include <string.h>

char addChars(const char *str, size_t len) {
    int sum = 0;
    for (size_t i = 0; i < len; ++i) {
        sum += str[i];
    }
    return (char)(sum % 256);
}

int main() {
    const char *input = "Hello, World!";
    size_t length = strlen(input);
    char result = addChars(input, length);
    printf("Resulting character: %c\n", result);
    return 0;
}
```
