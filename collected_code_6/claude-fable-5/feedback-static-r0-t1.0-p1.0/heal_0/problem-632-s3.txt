```c
#include <stdio.h>
#include <stdlib.h>

void move_zeroes(int *arr, size_t n)
{
    size_t insert_pos = 0;
    size_t i;

    if (arr == NULL || n == 0) {
        return;
    }

    for (i = 0; i < n; i++) {
        if (arr[i] != 0) {
            arr[insert_pos] = arr[i];
            insert_pos++;
        }
    }

    while (insert_pos < n) {
        arr[insert_pos] = 0;
        insert_pos++;
    }
}

int main(void)
{
    int arr[] = {0, 1, 0, 3, 12, 0, 5};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t i;

    move_zeroes(arr, n);

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
