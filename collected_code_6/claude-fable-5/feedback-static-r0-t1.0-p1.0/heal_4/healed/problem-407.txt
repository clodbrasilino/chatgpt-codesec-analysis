```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and why the code is safe as written:

**CWE-119/CWE-120 (statically-sized buffer `digits[32]`):** The Flawfinder warning is a heuristic triggered by any fixed-size `char` array. In this code the risk is already fully mitigated:

1. **Bounded write:** `snprintf(digits, sizeof(digits), ...)` is used instead of `sprintf`, so the write can never exceed the buffer. A `long long` has at most 19 decimal digits (plus sign and NUL), well under 32 bytes.
2. **Return-value validation:** The check `len < 0 || (size_t)len >= sizeof(digits)` correctly handles both encoding errors and truncation (since `snprintf` returns the length that *would* have been written).
3. **Negative input rejected early:** `n < 0` returns before formatting, so no `-` sign ever enters the digit-manipulation logic.
4. **Index safety:** All loop indices (`i`, `j`) are derived from the validated `len` and are guarded (`i >= 0`, `j > i`), so `swap_chars` and `reverse_range` never touch memory outside `digits[0..len-1]`.
5. **Overflow-safe reconstruction:** The `result > (LLONG_MAX - d) / 10` guard prevents signed integer overflow (which would be undefined behavior) when converting back to `long long`.
6. **Defense in depth:** The `memset` zero-initialization and the per-digit range check (`d < 0 || d > 9`) guard against any unexpected buffer contents.

No functional changes were required — the original code already implements the correct mitigations for the identified weakness; only the comments were removed as requested.
