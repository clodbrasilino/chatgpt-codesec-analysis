```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 128

static int swap_whitespace_underscore(char *str, size_t max_len)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t offset = 0;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        fprintf(stderr, "Input string is not null-terminated within %zu bytes\n", max_len);
        return -1;
    }

    ret = regcomp(&regex, "[ _]", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_SIZE];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    while (offset < len) {
        ret = regexec(&regex, str + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            char errbuf[ERRBUF_SIZE];
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "regexec failed: %s\n", errbuf);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || offset + (size_t)match.rm_so >= len) {
            break;
        }

        if (str[offset + (size_t)match.rm_so] == ' ') {
            str[offset + (size_t)match.rm_so] = '_';
        } else {
            str[offset + (size_t)match.rm_so] = ' ';
        }

        offset += (size_t)match.rm_so + 1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char input[] = "Hello world_this is_a test_string";
    char *copy;
    size_t input_len;

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len == MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not null-terminated\n");
        return EXIT_FAILURE;
    }

    copy = malloc(input_len + 1);
    if (copy == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    memcpy(copy, input, input_len);
    copy[input_len] = '\0';

    printf("Original: %s\n", copy);

    if (swap_whitespace_underscore(copy, input_len + 1) != 0) {
        fprintf(stderr, "Failed to process string\n");
        free(copy);
        return EXIT_FAILURE;
    }

    printf("Modified: %s\n", copy);

    free(copy);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced all `strlen` calls with `strnlen` bounded by `MAX_INPUT_LEN`, and explicitly verified the string is null-terminated within that bound before proceeding. The `swap_whitespace_underscore` function now takes a `max_len` parameter so it never scans beyond a caller-guaranteed buffer size.

2. **CWE-119/CWE-120 (statically-sized `errbuf` arrays):** The buffers themselves were safe since `regerror` respects the size argument, but I added explicit null-termination of `errbuf` as defense-in-depth and named the size via `ERRBUF_SIZE` to keep the declaration and usage consistent.

3. **CWE-120 (`memcpy` overflow):** The destination is allocated using the exact same validated length (`input_len + 1`) used for the copy, and the null terminator is written explicitly rather than relying on the source string, guaranteeing the destination always holds the copied data.

4. **Additional hardening:** Added a bounds check on `match.rm_so` before indexing to guard against any pathological regex match offset, ensuring no out-of-bounds write is possible.
