#include <stdio.h>
#include <stdlib.h>

int find_occur_count(int *arr, int len, int num) {
    int i, occur_count = 0;
    for(i = 0; i < len; i++) {
        if(arr[i] == num) {
            break;
        }
        else {
            occur_count++;
        }
    }
    // if element is not found in array
    if(i == len){
        return -1;
    }
    return occur_count;
}

int main() {
    int len, i, count, num;
    printf("Enter array length: ");
    scanf("%d", &len);
    if(len <= 0) {
        printf("Invalid array length.\n");
        return 1;
    }
    int *arr = (int*) malloc(len*sizeof(int));
    if(arr == NULL) {
        printf("Memory allocation failed.\n");
        return 1;
    }
    for(i = 0; i < len; i++) {
        printf("Enter array element %d: ", i+1);
        scanf("%d", &arr[i]);
    }
    printf("Enter number to search: ");
    scanf("%d", &num);
    count = find_occur_count(arr, len, num);
    if(count == -1){
        printf("Element not found in array.\n");
    }
    else{
        printf("Element count that occurs before %d is: %d\n", num, count);
    }
    free(arr);
    return 0;
}
