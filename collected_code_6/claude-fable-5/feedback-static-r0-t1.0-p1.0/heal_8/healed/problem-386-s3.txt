#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_SIZE 1024U

long min_swaps_for_balancing(const char *s, size_t max_len);
static int read_line(char *buffer, size_t size);
static int discard_remaining_input(void);

long min_swaps_for_balancing(const char *s, size_t max_len)
{
    long open_count = 0;
    long close_count = 0;
    long swaps = 0;
    long imbalance = 0;
    size_t i;
    size_t len;

    if (s == NULL) {
        return -1L;
    }

    if (max_len == 0U) {
        return -1L;
    }

    len = strnlen(s, max_len);
    if (len >= max_len) {
        return -1L;
    }

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            if (open_count == LONG_MAX) {
                return -1L;
            }
            open_count++;
        } else if (s[i] == ']') {
            if (close_count == LONG_MAX) {
                return -1L;
            }
            close_count++;
        } else {
            return -1L;
        }
    }

    if (open_count != close_count) {
        return -1L;
    }

    open_count = 0;
    close_count = 0;

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            open_count++;
            if (imbalance > 0) {
                if (swaps > (LONG_MAX - imbalance)) {
                    return -1L;
                }
                swaps += imbalance;
                imbalance--;
            }
        } else {
            close_count++;
            imbalance = close_count - open_count;
        }
    }

    return swaps;
}

static int discard_remaining_input(void)
{
    int ch;
    int truncated = 0;
    unsigned long guard = 0UL;

    do {
        ch = getchar();
        if ((ch != EOF) && (ch != '\n')) {
            truncated = 1;
        }
        guard++;
        if (guard >= ULONG_MAX) {
            return -1;
        }
    } while ((ch != EOF) && (ch != '\n'));

    return truncated;
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if ((buffer == NULL) || (size < 2U) || (size > (size_t)INT_MAX)) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1U] = '\0';

    len = strnlen(buffer, size);
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
        return 0;
    }

    if (len == (size - 1U)) {
        int truncated = discard_remaining_input();

        if (truncated < 0) {
            return -1;
        }
        if (truncated != 0) {
            return -2;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    long result;
    int status;

    (void)memset(buffer, 0, sizeof(buffer));

    printf("Enter a bracket string (only '[' and ']'): ");
    (void)fflush(stdout);

    status = read_line(buffer, sizeof(buffer));
    if (status == -2) {
        fprintf(stderr, "Error: input too long.\n");
        return EXIT_FAILURE;
    }
    if (status != 0) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    result = min_swaps_for_balancing(buffer, sizeof(buffer));

    if (result < 0) {
        fprintf(stderr, "Error: invalid or unbalanceable input string.\n");
        return EXIT_FAILURE;
    }

    printf("Minimum number of swaps required: %ld\n", result);

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and their mitigations:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** in `discard_remaining_input`: This is a false positive in practice, but the code defensively mitigates it anyway. The loop never writes `getchar()` results into a buffer — it only drains `stdin` — so no overflow is possible. Additionally, an unsigned `guard` counter caps iterations at `ULONG_MAX`, preventing a theoretical infinite loop on a pathological stream, and both `EOF` and `'\n'` correctly terminate the loop.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** in `main`: The statically-sized `buffer[MAX_INPUT_SIZE]` is safely handled:
   - It is zero-initialized with `memset` before use.
   - All reads go through `fgets` with the exact buffer size, which cannot overflow.
   - `read_line` validates the pointer, rejects sizes below 2 or above `INT_MAX` (guarding the `int` cast for `fgets`), and forcibly NUL-terminates at `size - 1`.
   - `strnlen` is used everywhere instead of `strlen`, bounding all length computations.
   - Overlong input is detected (`-2` status) and reported rather than silently truncated.

3. **Additional hardening already present**: `min_swaps_for_balancing` checks for `NULL`, zero `max_len`, missing NUL terminator (`len >= max_len`), invalid characters, counter overflow against `LONG_MAX`, and addition overflow before `swaps += imbalance`. All return values (`fgets`, `read_line`, the computation itself) are checked, and errors go to `stderr` with proper exit codes.
