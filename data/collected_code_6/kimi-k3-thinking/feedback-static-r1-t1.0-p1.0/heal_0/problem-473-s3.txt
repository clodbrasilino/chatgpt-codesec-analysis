#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

#define TUPLE_SIZE 2

typedef struct {
    int data[TUPLE_SIZE];
} Tuple;

typedef struct {
    int *elements;
    size_t size;
} IntArray;

static int tuple_contains(const Tuple *tuple, int value)
{
    size_t i;

    if (tuple == NULL) {
        return 0;
    }

    for (i = 0; i < TUPLE_SIZE; i++) {
        if (tuple->data[i] == value) {
            return 1;
        }
    }

    return 0;
}

static int array_contains(const int *elements, size_t size, int value)
{
    size_t i;

    if (elements == NULL) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (elements[i] == value) {
            return 1;
        }
    }

    return 0;
}

static size_t tuple_presence_count(const Tuple *tuples, size_t count, int value)
{
    size_t i;
    size_t presence;

    presence = 0;

    for (i = 0; i < count; i++) {
        if (tuple_contains(&tuples[i], value)) {
            presence++;
        }
    }

    return presence;
}

IntArray tuple_intersection(const Tuple *tuples, size_t count)
{
    IntArray result;
    size_t i;
    size_t j;
    size_t capacity;
    int value;

    result.elements = NULL;
    result.size = 0;

    if (tuples == NULL || count < 2) {
        return result;
    }

    if (count > SIZE_MAX / TUPLE_SIZE / sizeof(int)) {
        return result;
    }

    capacity = count * TUPLE_SIZE;

    result.elements = malloc(capacity * sizeof(int));
    if (result.elements == NULL) {
        return result;
    }

    for (i = 0; i < count; i++) {
        for (j = 0; j < TUPLE_SIZE; j++) {
            value = tuples[i].data[j];

            if (array_contains(result.elements, result.size, value)) {
                continue;
            }

            if (tuple_presence_count(tuples, count, value) >= 2) {
                result.elements[result.size] = value;
                result.size++;
            }
        }
    }

    if (result.size == 0) {
        free(result.elements);
        result.elements = NULL;
    }

    return result;
}

static void print_intersection(const char *label, const IntArray *result)
{
    size_t i;

    if (label == NULL || result == NULL) {
        return;
    }

    if (result->elements == NULL) {
        printf("%s: { }\n", label);
        return;
    }

    printf("%s: { ", label);
    for (i = 0; i < result->size; i++) {
        printf("%d ", result->elements[i]);
    }
    printf("}\n");
}

int main(void)
{
    const Tuple tuples1[] = { {3, 4}, {5, 6}, {9, 8}, {4, 3} };
    const Tuple tuples2[] = { {1, 2}, {3, 4}, {5, 6} };
    const Tuple tuples3[] = { {1, 2}, {2, 3}, {3, 4} };

    IntArray result1 = tuple_intersection(tuples1, sizeof(tuples1) / sizeof(tuples1[0]));
    IntArray result2 = tuple_intersection(tuples2, sizeof(tuples2) / sizeof(tuples2[0]));
    IntArray result3 = tuple_intersection(tuples3, sizeof(tuples3) / sizeof(tuples3[0]));

    print_intersection("Intersection 1", &result1);
    print_intersection("Intersection 2", &result2);
    print_intersection("Intersection 3", &result3);

    free(result1.elements);
    result1.elements = NULL;

    free(result2.elements);
    result2.elements = NULL;

    free(result3.elements);
    result3.elements = NULL;

    return 0;
}
