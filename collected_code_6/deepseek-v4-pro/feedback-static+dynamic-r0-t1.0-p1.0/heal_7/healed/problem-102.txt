#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *snake_to_camel(const char *snake) {
    if (snake == NULL) {
        return NULL;
    }

    size_t len = strlen(snake);
    char *camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    size_t j = 0;
    int capitalize_next = 0;
    int has_chars = 0;

    for (size_t i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next && has_chars) {
                camel[j++] = (char)toupper((unsigned char)snake[i]);
            } else {
                camel[j++] = (char)tolower((unsigned char)snake[i]);
            }
            capitalize_next = 0;
            has_chars = 1;
        }
    }

    camel[j] = '\0';

    if (j == 0) {
        free(camel);
        return NULL;
    }

    return camel;
}
