```c
#include <stdio.h>
#include <stddef.h>

int sum_array(const int *arr, size_t size) {
    int sum = 0;
    if (arr == NULL) {
        return 0;
    }
    for (size_t i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int result = sum_array(arr, size);
    printf("Sum: %d\n", result);
    return 0;
}
```
