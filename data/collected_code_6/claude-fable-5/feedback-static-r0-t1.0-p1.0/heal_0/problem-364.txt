```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int min_flips_to_alternate(const char *s)
{
    size_t len;
    size_t i;
    int flips_start_zero;
    int flips_start_one;
    char expected_zero;
    char expected_one;

    if (s == NULL) {
        return -1;
    }

    len = strlen(s);
    if (len == 0) {
        return 0;
    }

    flips_start_zero = 0;
    flips_start_one = 0;

    for (i = 0; i < len; i++) {
        if (s[i] != '0' && s[i] != '1') {
            return -1;
        }

        if ((i % 2) == 0) {
            expected_zero = '0';
            expected_one = '1';
        } else {
            expected_zero = '1';
            expected_one = '0';
        }

        if (s[i] != expected_zero) {
            flips_start_zero++;
        }
        if (s[i] != expected_one) {
            flips_start_one++;
        }
    }

    return (flips_start_zero < flips_start_one) ? flips_start_zero : flips_start_one;
}

int main(void)
{
    char buffer[1024];
    int result;
    size_t len;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    len = strlen(buffer);
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = min_flips_to_alternate(buffer);
    if (result < 0) {
        fprintf(stderr, "Invalid binary string\n");
        return EXIT_FAILURE;
    }

    printf("%d\n", result);
    return EXIT_SUCCESS;
}
```

**How it works:**

- `min_flips_to_alternate` computes the number of mismatches against both possible alternating patterns (`0101...` and `1010...`) in a single pass and returns the smaller count.
- The function validates its input: it rejects `NULL` pointers and any character other than `'0'` or `'1'`, returning `-1` as an error indicator, which the caller checks explicitly.
- `main` reads input safely with `fgets` (bounded, no overflow), checks the return value for I/O failure, strips the trailing newline, and reports errors to `stderr` with proper exit codes (`EXIT_SUCCESS` / `EXIT_FAILURE`).
- No dynamic resources are allocated, so there is nothing to leak or misuse across its lifetime; all buffers are stack-allocated with fixed, checked bounds.
