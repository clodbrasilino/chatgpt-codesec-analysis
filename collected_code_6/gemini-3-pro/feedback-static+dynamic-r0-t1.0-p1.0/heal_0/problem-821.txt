#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} KeyValuePair;

typedef struct {
    KeyValuePair *items;
    size_t count;
    size_t capacity;
} Dictionary;

Dictionary* create_dictionary(size_t initial_capacity) {
    if (initial_capacity == 0) {
        initial_capacity = 4;
    }
    Dictionary *dict = (Dictionary*)malloc(sizeof(Dictionary));
    if (!dict) {
        return NULL;
    }
    dict->items = (KeyValuePair*)malloc(initial_capacity * sizeof(KeyValuePair));
    if (!dict->items) {
        free(dict);
        return NULL;
    }
    dict->count = 0;
    dict->capacity = initial_capacity;
    return dict;
}

void free_dictionary(Dictionary *dict) {
    if (!dict) {
        return;
    }
    for (size_t i = 0; i < dict->count; i++) {
        free(dict->items[i].key);
        free(dict->items[i].value);
    }
    free(dict->items);
    free(dict);
}

int add_to_dictionary(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) {
        return -1;
    }
    
    for (size_t i = 0; i < dict->count; i++) {
        if (strcmp(dict->items[i].key, key) == 0) {
            char *new_value = strdup(value);
            if (!new_value) {
                return -1;
            }
            free(dict->items[i].value);
            dict->items[i].value = new_value;
            return 0;
        }
    }

    if (dict->count >= dict->capacity) {
        size_t new_capacity = dict->capacity * 2;
        KeyValuePair *new_items = (KeyValuePair*)realloc(dict->items, new_capacity * sizeof(KeyValuePair));
        if (!new_items) {
            return -1;
        }
        dict->items = new_items;
        dict->capacity = new_capacity;
    }

    char *new_key = strdup(key);
    if (!new_key) {
        return -1;
    }
    char *new_value = strdup(value);
    if (!new_value) {
        free(new_key);
        return -1;
    }

    dict->items[dict->count].key = new_key;
    dict->items[dict->count].value = new_value;
    dict->count++;
    return 0;
}

Dictionary* merge_dictionaries(const Dictionary *dict1, const Dictionary *dict2) {
    if (!dict1 || !dict2) {
        return NULL;
    }

    Dictionary *merged = create_dictionary(dict1->count + dict2->count);
    if (!merged) {
        return NULL;
    }

    for (size_t i = 0; i < dict1->count; i++) {
        if (add_to_dictionary(merged, dict1->items[i].key, dict1->items[i].value) != 0) {
            free_dictionary(merged);
            return NULL;
        }
    }

    for (size_t i = 0; i < dict2->count; i++) {
        if (add_to_dictionary(merged, dict2->items[i].key, dict2->items[i].value) != 0) {
            free_dictionary(merged);
            return NULL;
        }
    }

    return merged;
}

int main(void) {
    Dictionary *dict1 = create_dictionary(4);
    if (!dict1) return 1;

    Dictionary *dict2 = create_dictionary(4);
    if (!dict2) {
        free_dictionary(dict1);
        return 1;
    }

    if (add_to_dictionary(dict1, "key1", "value1") != 0 ||
        add_to_dictionary(dict1, "key2", "value2") != 0) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return 1;
    }

    if (add_to_dictionary(dict2, "key2", "new_value2") != 0 ||
        add_to_dictionary(dict2, "key3", "value3") != 0) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return 1;
    }

    Dictionary *merged = merge_dictionaries(dict1, dict2);
    if (!merged) {
        free_dictionary(dict1);
        free_dictionary(dict2);
        return 1;
    }

    for (size_t i = 0; i < merged->count; i++) {
        printf("%s: %s\n", merged->items[i].key, merged->items[i].value);
    }

    free_dictionary(dict1);
    free_dictionary(dict2);
    free_dictionary(merged);

    return 0;
}
