#include <stdio.h>

long long remArrayMul(int arr[], int len, int n) {
    long long mul = 1;
    for(int i=0; i<len; i++) {
        if(arr[i] == 0) {
            return 0;
        }

        mul = (mul * (arr[i] % n)) % n;
    }
    return mul;
}

int main(){
    int arr[] = {1,2,3,4,5};
    int n = 3;
    int len = sizeof(arr)/sizeof(arr[0]);
    printf("%lld", remArrayMul(arr,len,n));
    return 0;
}
