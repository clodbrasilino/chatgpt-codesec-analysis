```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int* data;
    size_t size;
} Tuple;

Tuple* repeat_tuple(const Tuple* tuple, size_t n) {
    if (!tuple || !tuple->data || tuple->size == 0 || n == 0) {
        return NULL;
    }

    if (n > ((size_t)-1) / tuple->size) {
        return NULL;
    }

    Tuple* result = (Tuple*)malloc(sizeof(Tuple));
    if (!result) {
        return NULL;
    }

    result->size = tuple->size * n;
    result->data = (int*)malloc(result->size * sizeof(int));
    if (!result->data) {
        free(result);
        return NULL;
    }

    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < tuple->size; ++j) {
            result->data[i * tuple->size + j] = tuple->data[j];
        }
    }

    return result;
}

void print_tuple(const Tuple* tuple) {
    if (!tuple || !tuple->data) {
        printf("()\n");
        return;
    }
    printf("(");
    for (size_t i = 0; i < tuple->size; ++i) {
        printf("%d", tuple->data[i]);
        if (i < tuple->size - 1) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void) {
    int initial_data[] = {1, 2, 3};
    Tuple tuple = {initial_data, 3};

    Tuple* repeated = repeat_tuple(&tuple, 4);
    if (repeated) {
        print_tuple(repeated);
        free(repeated->data);
        free(repeated);
    } else {
        printf("Failed to repeat tuple.\n");
    }

    return 0;
}
```
