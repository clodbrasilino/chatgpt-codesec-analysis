#include <stdio.h>
#include <stdlib.h>

#define PI 3.14159265358979323846

double calculate_cylinder_perimeter(double radius, double height) {
    if (radius < 0.0 || height < 0.0) {
        return -1.0;
    }
    return (4.0 * radius) + (2.0 * height);
}

int main() {
    double radius = 0.0;
    double height = 0.0;
    double perimeter = 0.0;
    int result = 0;

    result = scanf("%lf %lf", &radius, &height);
    if (result != 2) {
        return EXIT_FAILURE;
    }

    perimeter = calculate_cylinder_perimeter(radius, height);
    if (perimeter < 0.0) {
        return EXIT_FAILURE;
    }

    printf("%.2f\n", perimeter);

    return EXIT_SUCCESS;
}
