#include <stdio.h>
#include <string.h>

int areIsomorphic(const char *s, size_t len_s, const char *t, size_t len_t) {
    if (s == NULL || t == NULL) {
        return 0;
    }
    if (len_s != len_t) {
        return 0;
    }
    unsigned char map_st[256] = {0};
    unsigned char map_ts[256] = {0};
    for (size_t i = 0; i < len_s; i++) {
        unsigned char c_s = (unsigned char)s[i];
        unsigned char c_t = (unsigned char)t[i];
        if (map_st[c_s] == 0 && map_ts[c_t] == 0) {
            map_st[c_s] = c_t;
            map_ts[c_t] = c_s;
        } else {
            if (map_st[c_s] != c_t || map_ts[c_t] != c_s) {
                return 0;
            }
        }
    }
    return 1;
}

int main(int argc, char *argv[]) {
    const char *str1 = "egg";
    const char *str2 = "add";
    const char *str3 = "foo";
    const char *str4 = "bar";
    const char *str5 = "paper";
    const char *str6 = "title";
    const char *str7 = argc > 1 ? argv[1] : NULL;
    const char *str8 = "abc";
    size_t max_len = 256;

    printf("%d\n", areIsomorphic(str1, strnlen(str1, max_len), str2, strnlen(str2, max_len)));
    printf("%d\n", areIsomorphic(str3, strnlen(str3, max_len), str4, strnlen(str4, max_len)));
    printf("%d\n", areIsomorphic(str5, strnlen(str5, max_len), str6, strnlen(str6, max_len)));
    printf("%d\n", areIsomorphic(str7, str7 ? strnlen(str7, max_len) : 0, str8, strnlen(str8, max_len)));

    return 0;
}
