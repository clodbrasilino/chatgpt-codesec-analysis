```c
#include <stdio.h>
#include <stdbool.h>

bool is_greater_than_all(int number, const int *array, size_t size) {
    if (array == NULL) {
        if (size == 0) {
            return true;
        }
        return false;
    }
    for (size_t i = 0; i < size; i++) {
        if (number <= array[i]) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int arr[] = {1, 5, 8, 12, 3};
    size_t arr_size = sizeof(arr) / sizeof(arr[0]);
    int num;

    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    if (is_greater_than_all(num, arr, arr_size)) {
        printf("%d is greater than all elements in the array.\n", num);
    } else {
        printf("%d is not greater than all elements in the array.\n", num);
    }

    return 0;
}
```
