```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *output, size_t output_size);

int int_to_roman(int number, char *output, size_t output_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t symbol_len = symbol_lengths[i];
            size_t j;

            if (used >= output_size || symbol_len >= output_size - used) {
                output[0] = '\0';
                return -1;
            }

            for (j = 0; j < symbol_len; j++) {
                output[used + j] = symbols[i][j];
            }
            used += symbol_len;

            if (used >= output_size) {
                output[0] = '\0';
                return -1;
            }

            output[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char buffer[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], buffer, sizeof(buffer)) == 0) {
            if (printf("%d -> %s\n", test_values[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 0\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 4000\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning (CWE-119/CWE-120) flags the statically-sized `char buffer[ROMAN_BUFFER_SIZE]` as a *potential* overflow risk, but this is a heuristic hit rather than a real vulnerability in this code. The rewritten code is safe as-is because:

1. **Size is always passed explicitly** — every call to `int_to_roman` passes `sizeof(buffer)`, so the callee knows the true capacity and never relies on assumptions.

2. **Strict bounds checking before every write** — the check `used >= output_size || symbol_len >= output_size - used` is done *before* copying each symbol. Note the subtraction form (`output_size - used`) deliberately avoids integer overflow that a naive `used + symbol_len >= output_size` could theoretically introduce.

3. **NUL terminator space is guaranteed** — the `symbol_len >= output_size - used` comparison (using `>=`, not `>`) reserves room for the terminating `'\0'`, and `output[used] = '\0'` is guarded by a second `used >= output_size` check.

4. **Fail-safe output state** — on any error path the buffer is reset to an empty string (`output[0] = '\0'`), so callers never read stale or partially-written data.

5. **Input range validation** — values outside `[1, 3999]` are rejected up front. The worst case, 3888 (`MMMDCCCLXXXVIII`), needs 15 characters plus the terminator, so `ROMAN_BUFFER_SIZE` of 32 provides comfortable headroom, and even an undersized buffer would fail gracefully rather than overflow.

6. **No unsafe library calls** — the code avoids `strcpy`/`strcat`/`sprintf`, using manual bounded copies instead, so there is no path where an unbounded write can occur.
