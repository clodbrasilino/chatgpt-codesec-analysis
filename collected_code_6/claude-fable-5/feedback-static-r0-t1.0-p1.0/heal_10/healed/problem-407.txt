#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i || j >= len || i >= len) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

**The Flawfinder warning (CWE-119/CWE-120)** flags the statically-sized array `char digits[DIGITS_BUF_SIZE]` as a potential buffer overflow risk. However, examining the code shows the risk is already fully mitigated:

1. **Buffer sizing is provably sufficient**: A `long long` is at most 64 bits, so its decimal representation needs at most 19 digits (plus sign and NUL). `DIGITS_BUF_SIZE` of 32 provides comfortable headroom, and negative values are rejected before formatting.

2. **Bounded write with result validation**: `snprintf(digits, sizeof(digits), ...)` limits the write to the buffer size, and the return value is checked both for encoding errors (`len < 0`) and truncation (`(size_t)len >= sizeof(digits)`), so no truncated or malformed string is ever processed.

3. **Defensive initialization**: `memset` zeroes the buffer before use, guaranteeing NUL termination even in edge cases.

4. **Index bounds are guarded**: The loops over `i` and `j` are bounded by `len`, and explicit checks (`j <= i || j >= len || i >= len`) reject any inconsistent state before `swap_chars` or `reverse_range` touch the buffer.

5. **Integer overflow protection**: The reconstruction loop checks `result > (LLONG_MAX - d) / 10` before each multiply-accumulate, preventing signed overflow (undefined behavior in C) when converting back to `long long`.

The warning is therefore a false positive triggered by the mere presence of a fixed-size array; the surrounding checks satisfy all the mitigations Flawfinder recommends (bounds checking, length-limiting functions, adequate sizing), so the code stands as-is with no functional changes required.
