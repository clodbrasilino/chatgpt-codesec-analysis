```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <ctype.h>

#define MIN_LENGTH 8
#define MAX_LENGTH 64
#define BUFFER_SIZE 128
#define MAX_DISCARD 4096

static void secure_clear(void *ptr, size_t size)
{
    volatile unsigned char *p;

    if (ptr == NULL || size == 0) {
        return;
    }

    p = (volatile unsigned char *)ptr;

    while (size--) {
        *p++ = 0;
    }
}

static bool flush_stdin(void)
{
    int ch;
    bool discarded = false;
    size_t count = 0;

    do {
        ch = getchar();
        if (ch != '\n' && ch != EOF) {
            discarded = true;
        }
        count++;
    } while (ch != '\n' && ch != EOF && count < MAX_DISCARD);

    return discarded;
}

static bool is_valid_password(const char *password, size_t max_size)
{
    size_t length;
    size_t i;
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    bool has_special = false;

    if (password == NULL || max_size == 0) {
        return false;
    }

    length = strnlen(password, max_size);

    if (length == max_size) {
        return false;
    }

    if (length < MIN_LENGTH || length > MAX_LENGTH) {
        return false;
    }

    for (i = 0; i < length; i++) {
        unsigned char c = (unsigned char)password[i];

        if (!isprint(c)) {
            return false;
        }

        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        } else if (ispunct(c)) {
            has_special = true;
        } else {
            return false;
        }
    }

    return has_upper && has_lower && has_digit && has_special;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t length;
    bool truncated = false;

    printf("Enter password: ");

    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        secure_clear(buffer, sizeof(buffer));
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    length = strnlen(buffer, sizeof(buffer));

    if (length > 0 && buffer[length - 1] == '\n') {
        buffer[length - 1] = '\0';
    } else {
        truncated = flush_stdin();
    }

    if (!truncated && is_valid_password(buffer, sizeof(buffer))) {
        printf("Password is valid\n");
    } else {
        printf("Password is invalid\n");
    }

    secure_clear(buffer, sizeof(buffer));

    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The Flawfinder warning about `getchar()` in `flush_stdin()` is a false positive in practice here. The loop never writes the read characters into any buffer — it only discards them. Additionally, the loop is bounded by `MAX_DISCARD` (4096 iterations), preventing an unbounded loop on a malicious stream that never delivers `\n` or `EOF`. No fix required.

2. **Statically-sized `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** This is mitigated correctly:
   - Input is read with `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces the buffer bound and always null-terminates within the given size.
   - A defensive `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination even in pathological cases.
   - Length computation uses `strnlen()` with an explicit bound rather than `strlen()`, so it can never read past the buffer.
   - If the input did not fit (no trailing newline), the remainder is drained via `flush_stdin()` and the input is rejected as truncated, avoiding silent validation of a partial password.

3. **Sensitive data hygiene:** `secure_clear()` uses a `volatile` pointer to prevent the compiler from optimizing away the memset-like wipe (avoiding CWE-14, dead-store elimination of sensitive data), and it is invoked on every exit path after the buffer holds a password.

The code as given already contains the correct fixes for both flagged issues, so the rewrite preserves the logic verbatim with the audit comments removed.
