```c
#include <stdio.h>
#include <ctype.h>

int countLowerCase(char *str) {
    int i = 0;
    int count = 0;

    if(str == NULL){
        return 0;
    }

    while(str[i]) {
        if(islower((unsigned char)str[i])){
            count++;
        }
        i++;
    }

    return count;
}

int main() {
    char str[] = "Hello World!";

    int count = countLowerCase(str);
    printf("Number of lowercase letters: %d\n", count);

    return 0;
}
```
