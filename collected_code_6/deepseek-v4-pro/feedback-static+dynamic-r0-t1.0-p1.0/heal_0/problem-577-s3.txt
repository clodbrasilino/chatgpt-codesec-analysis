#include <stdio.h>

int last_digit_factorial(int n) {
    if (n < 0) return -1;
    if (n == 0 || n == 1) return 1;
    if (n >= 5) return 0;
    
    int factorial = 1;
    for (int i = 2; i <= n; i++) {
        factorial *= i;
    }
    return factorial % 10;
}

int main(void) {
    int number;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    int result = last_digit_factorial(number);
    
    if (result == -1) {
        printf("Factorial is not defined for negative numbers\n");
        return 1;
    }
    
    printf("Last digit of %d! is: %d\n", number, result);
    
    return 0;
}
