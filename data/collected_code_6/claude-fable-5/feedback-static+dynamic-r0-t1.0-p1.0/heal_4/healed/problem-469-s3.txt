```c
#include <stdio.h>
#include <stdlib.h>

int maxProfit(int k, const int *prices, int n)
{
    int *prev;
    int *curr;
    int *tmp;
    int t;
    int d;
    int result;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (d = 1; d < n; d++) {
            if (prices[d] > prices[d - 1]) {
                result += prices[d] - prices[d - 1];
            }
        }
        return result;
    }

    prev = calloc((size_t)n, sizeof(int));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc((size_t)n, sizeof(int));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    result = 0;

    for (t = 1; t <= k; t++) {
        int maxDiff = prev[0] - prices[0];
        curr[0] = 0;
        for (d = 1; d < n; d++) {
            if (prev[d] - prices[d] > maxDiff) {
                maxDiff = prev[d] - prices[d];
            }
            curr[d] = curr[d - 1];
            if (prices[d] + maxDiff > curr[d]) {
                curr[d] = prices[d] + maxDiff;
            }
        }
        if (curr[n - 1] > result) {
            result = curr[n - 1];
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    free(prev);
    free(curr);

    return result;
}

int main(void)
{
    int prices0[] = {10, 22, 5, 75, 65, 80};
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int prices2[] = {2, 4, 1};

    struct {
        int k;
        const int *prices;
        int n;
        int expected;
    } tests[] = {
        {2, prices0, (int)(sizeof(prices0) / sizeof(prices0[0])), 87},
        {2, prices1, (int)(sizeof(prices1) / sizeof(prices1[0])), 7},
        {2, prices2, (int)(sizeof(prices2) / sizeof(prices2[0])), 2}
    };

    int numTests = (int)(sizeof(tests) / sizeof(tests[0]));
    int i;
    int failures = 0;

    for (i = 0; i < numTests; i++) {
        int actual = maxProfit(tests[i].k, tests[i].prices, tests[i].n);
        if (actual == tests[i].expected) {
            printf("Test %d passed: expected %d, got %d\n",
                   i, tests[i].expected, actual);
        } else {
            printf("Test %d FAILED: expected %d, got %d\n",
                   i, tests[i].expected, actual);
            failures++;
        }
    }

    if (failures > 0) {
        printf("%d of %d tests failed\n", failures, numTests);
        return EXIT_FAILURE;
    }

    printf("All %d tests passed\n", numTests);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses listed in the original comments (`got 0`, `got 5`) are symptomatic of three classic bugs in earlier iterations of this k-transactions DP:

1. **Returning 0 spuriously** — typically caused by an overly aggressive guard clause (e.g., rejecting `k < n` instead of `k <= 0`), or by reading `curr[n-1]` after the pointer swap, which reads the stale buffer.

2. **Getting 5 instead of the full profit** — the signature of computing only a *single* transaction, usually because `prev` and `curr` were never swapped between iterations of the `t` loop, or `maxDiff` was reset from `-prices[0]` rather than `prev[0] - prices[0]`, discarding profit from previous transactions.

3. **Missing the `k >= n/2` fast path** — without it, large `k` wastes memory and iterations; with it, the greedy sum of all positive deltas is both correct and O(n).

The rewritten version addresses all of these: `maxDiff` is seeded from `prev[0] - prices[0]` and updated with `prev[d] - prices[d]` so each transaction level builds on the previous one; the buffer swap via `tmp` occurs at the end of each transaction iteration and `result` is captured *before* the swap; and both buffers are correctly freed regardless of which pointer holds which allocation after an odd/even number of swaps. Verified by hand-tracing: test 0 yields 87 (12+75... via 22−10 and 80−5 → actually 12 + 75 = 87), test 1 yields 7 (4 + 3), and test 2 hits the greedy path yielding 2.
