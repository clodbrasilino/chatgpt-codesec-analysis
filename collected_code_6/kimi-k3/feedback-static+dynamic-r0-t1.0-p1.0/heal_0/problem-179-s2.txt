#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int isKeithNumber(int num) {
    if (num < 10) {
        return 0;
    }
    
    int digits[20];
    int count = 0;
    int temp = num;
    
    while (temp > 0) {
        digits[count] = temp % 10;
        temp /= 10;
        count++;
    }
    
    int *sequence = (int *)malloc(count * sizeof(int));
    if (sequence == NULL) {
        return -1;
    }
    
    for (int i = 0; i < count; i++) {
        sequence[i] = digits[count - 1 - i];
    }
    
    int sum = 0;
    int index = count;
    
    while (sum < num) {
        sum = 0;
        for (int i = index - count; i < index; i++) {
            sum += sequence[i];
        }
        
        if (sum == num) {
            free(sequence);
            sequence = NULL;
            return 1;
        }
        
        if (sum > num) {
            break;
        }
        
        int *temp_seq = (int *)realloc(sequence, (index + 1) * sizeof(int));
        if (temp_seq == NULL) {
            free(sequence);
            sequence = NULL;
            return -1;
        }
        sequence = temp_seq;
        sequence[index] = sum;
        index++;
    }
    
    free(sequence);
    sequence = NULL;
    return 0;
}

int main(void) {
    int number;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    if (number < 0) {
        printf("Please enter a non-negative number\n");
        return 1;
    }
    
    int result = isKeithNumber(number);
    
    if (result == 1) {
        printf("%d is a Keith number\n", number);
    } else if (result == 0) {
        printf("%d is not a Keith number\n", number);
    } else {
        fprintf(stderr, "Memory allocation error\n");
        return 1;
    }
    
    return 0;
}
