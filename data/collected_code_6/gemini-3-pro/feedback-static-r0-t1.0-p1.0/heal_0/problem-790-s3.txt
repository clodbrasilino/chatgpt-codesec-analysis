#include <stdio.h>
#include <stdbool.h>

bool check_even_indices(const int *array, size_t size) {
    if (array == NULL) {
        return false;
    }
    
    for (size_t i = 0; i < size; i += 2) {
        if (array[i] % 2 != 0) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int valid_array[] = {2, 5, 4, 7, 8, 9};
    size_t valid_size = sizeof(valid_array) / sizeof(valid_array[0]);
    
    int invalid_array[] = {2, 5, 3, 7, 8, 9};
    size_t invalid_size = sizeof(invalid_array) / sizeof(invalid_array[0]);
    
    if (check_even_indices(valid_array, valid_size)) {
        printf("Valid array passes.\n");
    } else {
        printf("Valid array fails.\n");
    }
    
    if (check_even_indices(invalid_array, invalid_size)) {
        printf("Invalid array passes.\n");
    } else {
        printf("Invalid array fails.\n");
    }
    
    return 0;
}
