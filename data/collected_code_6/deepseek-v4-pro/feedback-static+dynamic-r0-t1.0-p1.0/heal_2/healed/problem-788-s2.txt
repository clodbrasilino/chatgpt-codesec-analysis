#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *str;
    int *list;
    int list_size;
} Tuple;

Tuple create_tuple(const char *str, const int *list, int list_size) {
    Tuple t;
    size_t str_len = strnlen(str, 4096);
    
    if (str_len == 4096 && str[4096] != '\0') {
        fprintf(stderr, "String too long or not null-terminated\n");
        exit(EXIT_FAILURE);
    }
    
    t.str = malloc(str_len + 1);
    if (t.str == NULL) {
        fprintf(stderr, "Memory allocation failed for string\n");
        exit(EXIT_FAILURE);
    }
    
    if (str_len > 0) {
        size_t dest_size = str_len + 1;
        if (str_len >= dest_size) {
            free(t.str);
            fprintf(stderr, "Buffer overflow detected in string copy\n");
            exit(EXIT_FAILURE);
        }
        memcpy(t.str, str, str_len);
    }
    t.str[str_len] = '\0';
    
    t.list_size = list_size;
    if (list_size > 0) {
        if (list_size > (SIZE_MAX / sizeof(int))) {
            free(t.str);
            fprintf(stderr, "Integer overflow detected in list allocation\n");
            exit(EXIT_FAILURE);
        }
        size_t alloc_size = sizeof(int) * list_size;
        t.list = malloc(alloc_size);
        if (t.list == NULL) {
            free(t.str);
            fprintf(stderr, "Memory allocation failed for list\n");
            exit(EXIT_FAILURE);
        }
        size_t copy_size = sizeof(int) * list_size;
        if (copy_size > alloc_size) {
            free(t.str);
            free(t.list);
            fprintf(stderr, "Buffer overflow detected in list copy\n");
            exit(EXIT_FAILURE);
        }
        memcpy(t.list, list, copy_size);
    } else {
        t.list = NULL;
    }
    
    return t;
}

void free_tuple(Tuple *t) {
    if (t != NULL) {
        free(t->str);
        free(t->list);
        t->str = NULL;
        t->list = NULL;
        t->list_size = 0;
    }
}

int main(void) {
    const char *str = "hello";
    int list_data[] = {1, 2, 3, 4, 5};
    int list_size = sizeof(list_data) / sizeof(list_data[0]);
    
    Tuple t = create_tuple(str, list_data, list_size);
    
    printf("String: %s\n", t.str);
    printf("List: ");
    for (int i = 0; i < t.list_size; i++) {
        printf("%d ", t.list[i]);
    }
    printf("\n");
    
    free_tuple(&t);
    
    return 0;
}
