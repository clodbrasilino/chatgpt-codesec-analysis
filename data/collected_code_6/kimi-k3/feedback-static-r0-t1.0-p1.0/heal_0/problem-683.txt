```c
#include <stdio.h>
#include <math.h>
#include <stdbool.h>

bool isSumOfTwoSquares(int n) {
    if (n < 0) {
        return false;
    }
    
    int limit = (int)sqrt(n);
    
    for (int i = 0; i <= limit; i++) {
        int remainder = n - (i * i);
        int root = (int)sqrt(remainder);
        
        if (root * root == remainder) {
            return true;
        }
    }
    
    return false;
}

int main(void) {
    int test_cases[] = {5, 4, 13, 7, 25, 0, 1, 2, 3, 8, 10, 11, 18, 20};
    int num_tests = sizeof(test_cases) / sizeof(test_cases[0]);
    
    for (int i = 0; i < num_tests; i++) {
        int num = test_cases[i];
        if (isSumOfTwoSquares(num)) {
            printf("%d can be represented as sum of two squares\n", num);
        } else {
            printf("%d cannot be represented as sum of two squares\n", num);
        }
    }
    
    return 0;
}
```
