```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char name[MAX_NAME_LEN];
    double price;
} Item;

typedef struct {
    Item *data;
    size_t size;
    size_t capacity;
} MinHeap;

static int item_set_name(Item *item, const char *name)
{
    size_t len;

    if (item == NULL || name == NULL) {
        return -1;
    }

    len = strlen(name);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    memcpy(item->name, name, len);
    item->name[len] = '\0';
    return 0;
}

static int item_init(Item *item, const char *name, double price)
{
    if (item == NULL) {
        return -1;
    }

    if (item_set_name(item, name) != 0) {
        return -1;
    }

    item->price = price;
    return 0;
}

static void swap_items(Item *a, Item *b)
{
    Item tmp = *a;
    *a = *b;
    *b = tmp;
}

static MinHeap *heap_create(size_t capacity)
{
    MinHeap *heap;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(Item)) {
        return NULL;
    }

    heap = malloc(sizeof(MinHeap));
    if (heap == NULL) {
        return NULL;
    }

    heap->data = malloc(capacity * sizeof(Item));
    if (heap->data == NULL) {
        free(heap);
        return NULL;
    }

    heap->size = 0;
    heap->capacity = capacity;
    return heap;
}

static void heap_destroy(MinHeap *heap)
{
    if (heap != NULL) {
        free(heap->data);
        free(heap);
    }
}

static void sift_up(MinHeap *heap, size_t index)
{
    while (index > 0) {
        size_t parent = (index - 1) / 2;
        if (heap->data[index].price < heap->data[parent].price) {
            swap_items(&heap->data[index], &heap->data[parent]);
            index = parent;
        } else {
            break;
        }
    }
}

static void sift_down(MinHeap *heap, size_t index)
{
    for (;;) {
        size_t left = 2 * index + 1;
        size_t right = 2 * index + 2;
        size_t smallest = index;

        if (left < heap->size &&
            heap->data[left].price < heap->data[smallest].price) {
            smallest = left;
        }
        if (right < heap->size &&
            heap->data[right].price < heap->data[smallest].price) {
            smallest = right;
        }
        if (smallest == index) {
            break;
        }
        swap_items(&heap->data[index], &heap->data[smallest]);
        index = smallest;
    }
}

static int heap_push(MinHeap *heap, const Item *item)
{
    if (heap == NULL || item == NULL || heap->size >= heap->capacity) {
        return -1;
    }

    heap->data[heap->size] = *item;
    sift_up(heap, heap->size);
    heap->size++;
    return 0;
}

static int heap_pop(MinHeap *heap, Item *out)
{
    if (heap == NULL || out == NULL || heap->size == 0) {
        return -1;
    }

    *out = heap->data[0];
    heap->size--;
    heap->data[0] = heap->data[heap->size];
    sift_down(heap, 0);
    return 0;
}

int find_n_cheapest(const Item *items, size_t count, size_t n, Item *result)
{
    MinHeap *heap;
    size_t i;

    if (items == NULL || result == NULL || count == 0 || n == 0 || n > count) {
        return -1;
    }

    heap = heap_create(count);
    if (heap == NULL) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        if (heap_push(heap, &items[i]) != 0) {
            heap_destroy(heap);
            return -1;
        }
    }

    for (i = 0; i < n; i++) {
        if (heap_pop(heap, &result[i]) != 0) {
            heap_destroy(heap);
            return -1;
        }
    }

    heap_destroy(heap);
    return 0;
}

int main(void)
{
    static const struct {
        const char *name;
        double price;
    } raw_data[] = {
        {"Laptop", 999.99},
        {"Mouse", 19.99},
        {"Keyboard", 49.99},
        {"Monitor", 249.99},
        {"USB Cable", 5.99},
        {"Headphones", 89.99},
        {"Webcam", 39.99},
        {"Speaker", 129.99}
    };
    size_t count = sizeof(raw_data) / sizeof(raw_data[0]);
    size_t n = 3;
    Item dataset[sizeof(raw_data) / sizeof(raw_data[0])];
    Item *result;
    size_t i;

    for (i = 0; i < count; i++) {
        if (item_init(&dataset[i], raw_data[i].name, raw_data[i].price) != 0) {
            fprintf(stderr, "Invalid item data at index %zu\n", i);
            return EXIT_FAILURE;
        }
    }

    result = malloc(n * sizeof(Item));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (find_n_cheapest(dataset, count, n, result) != 0) {
        fprintf(stderr, "Failed to find cheapest items\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("The %zu cheapest items:\n", n);
    for (i = 0; i < n; i++) {
        printf("%.*s: %.2f\n", MAX_NAME_LEN - 1, result[i].name, result[i].price);
    }

    free(result);
    return EXIT_SUCCESS;
}
```

Key changes addressing the flagged weakness (CWE-119/CWE-120):

1. **Bounded name initialization**: Added `item_set_name`/`item_init`, which validate the input length against `MAX_NAME_LEN` before copying with `memcpy` and explicitly null-terminating, so the fixed-size `name` buffer can never overflow or end up unterminated.
2. **Integer-overflow guard in allocation**: `heap_create` now rejects capacities where `capacity * sizeof(Item)` would overflow, preventing an undersized allocation followed by out-of-bounds writes.
3. **Bounded output**: `printf` uses `%.*s` with a width limit, so printing is safe even if a name were somehow not null-terminated.
