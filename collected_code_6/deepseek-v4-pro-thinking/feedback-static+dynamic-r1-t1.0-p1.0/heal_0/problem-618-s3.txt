#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node *createNode(int data) {
    Node *node = malloc(sizeof(*node));
    if (!node) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

int appendNode(Node **head, int data) {
    Node *node = createNode(data);
    if (!node) {
        return 0;
    }
    if (*head == NULL) {
        *head = node;
    } else {
        Node *current = *head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = node;
    }
    return 1;
}

void freeList(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
}

Node *divideLists(const Node *list1, const Node *list2) {
    Node *result = NULL;
    Node **tail = &result;

    while (list1 != NULL && list2 != NULL) {
        if (list2->data == 0) {
            freeList(result);
            return NULL;
        }

        Node *node = createNode(list1->data / list2->data);
        if (!node) {
            freeList(result);
            return NULL;
        }

        *tail = node;
        tail = &node->next;

        list1 = list1->next;
        list2 = list2->next;
    }

    if (list1 != NULL || list2 != NULL) {
        freeList(result);
        return NULL;
    }

    return result;
}

int main(void) {
    Node *list1 = NULL;
    Node *list2 = NULL;
    int values1[] = {10, 20, 30};
    int values2[] = {2, 4, 5};
    size_t i;
    size_t count1 = sizeof(values1) / sizeof(values1[0]);
    size_t count2 = sizeof(values2) / sizeof(values2[0]);

    for (i = 0; i < count1; ++i) {
        if (!appendNode(&list1, values1[i])) {
            freeList(list1);
            freeList(list2);
            return EXIT_FAILURE;
        }
    }

    for (i = 0; i < count2; ++i) {
        if (!appendNode(&list2, values2[i])) {
            freeList(list1);
            freeList(list2);
            return EXIT_FAILURE;
        }
    }

    Node *result = divideLists(list1, list2);

    if (result == NULL) {
        printf("Division failed\n");
    } else {
        Node *current = result;
        while (current != NULL) {
            printf("%d ", current->data);
            current = current->next;
        }
        printf("\n");
    }

    freeList(list1);
    freeList(list2);
    freeList(result);

    return EXIT_SUCCESS;
}
