```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *remove_non_alnum(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len;
    size_t input_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t chunk_len;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        chunk_len = (size_t)(match.rm_eo - match.rm_so);
        if (result_len > input_len || chunk_len > input_len - result_len) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        memcpy(result + result_len, cursor + match.rm_so, chunk_len);
        result_len += chunk_len;
        cursor += match.rm_eo;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a-b_c=d+e",
        "!!!@@@###",
        "NoSpecialChars2024"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alnum(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: %s\nCleaned:  %s\n\n",
                   test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) was a false positive in the original code, but the rewrite hardens the guard further:

1. **Buffer sizing:** `result` is allocated with `input_len + 1` bytes. Since every copied chunk is a substring of `input`, the total copied bytes can never exceed `input_len`, leaving room for the NUL terminator.

2. **Strengthened bounds check:** The condition now includes `result_len > input_len` before computing `input_len - result_len`, eliminating any possibility of unsigned underflow in the subtraction — a defensive improvement over relying solely on the loop invariant.

3. **Input length enforcement:** `strnlen` with `MAX_INPUT_LEN` guarantees a bounded, NUL-terminated input before any allocation or copying occurs.

4. **Match offset validation:** `match.rm_so < 0 || match.rm_eo < match.rm_so` is checked before the signed-to-unsigned cast, so `chunk_len` is always well-defined.

5. **Resource hygiene:** Every error path frees `result` and calls `regfree`, so there are no memory or regex-object leaks.

The `memcpy` call is therefore provably safe: `result_len + chunk_len <= input_len < input_len + 1` (the allocation size) on every iteration.
