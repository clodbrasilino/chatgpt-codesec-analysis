#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>
#include <limits.h>

#define MAX_STRING_LENGTH 4096

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0;

    if (s == NULL) {
        return 0;
    }

    while (len < max_len && s[len] != '\0') {
        len++;
    }

    return len;
}

bool are_isomorphic(const char *str1, const char *str2)
{
    const size_t table_size = (size_t)UCHAR_MAX + 1;
    unsigned char *map_forward = NULL;
    bool *mapped_forward = NULL;
    bool *mapped_backward = NULL;
    bool result = true;
    size_t len1;
    size_t len2;
    size_t i;

    if (str1 == NULL || str2 == NULL) {
        return false;
    }

    len1 = bounded_strlen(str1, MAX_STRING_LENGTH);
    len2 = bounded_strlen(str2, MAX_STRING_LENGTH);

    if (len1 != len2) {
        return false;
    }

    map_forward = calloc(table_size, sizeof(*map_forward));
    mapped_forward = calloc(table_size, sizeof(*mapped_forward));
    mapped_backward = calloc(table_size, sizeof(*mapped_backward));

    if (map_forward == NULL || mapped_forward == NULL || mapped_backward == NULL) {
        free(map_forward);
        free(mapped_forward);
        free(mapped_backward);
        return false;
    }

    for (i = 0; i < len1; i++) {
        unsigned char c1 = (unsigned char)str1[i];
        unsigned char c2 = (unsigned char)str2[i];

        if ((size_t)c1 >= table_size || (size_t)c2 >= table_size) {
            result = false;
            break;
        }

        if (mapped_forward[c1]) {
            if (map_forward[c1] != c2) {
                result = false;
                break;
            }
        } else {
            if (mapped_backward[c2]) {
                result = false;
                break;
            }
            map_forward[c1] = c2;
            mapped_forward[c1] = true;
            mapped_backward[c2] = true;
        }
    }

    free(map_forward);
    free(mapped_forward);
    free(mapped_backward);

    return result;
}

int main(int argc, char *argv[])
{
    if (argc != 3) {
        const char *program_name = (argc > 0 && argv[0] != NULL) ? argv[0] : "program";
        fprintf(stderr, "Usage: %s <string1> <string2>\n", program_name);
        return EXIT_FAILURE;
    }

    if (are_isomorphic(argv[1], argv[2])) {
        printf("\"%s\" and \"%s\" are isomorphic.\n", argv[1], argv[2]);
    } else {
        printf("\"%s\" and \"%s\" are not isomorphic.\n", argv[1], argv[2]);
    }

    return EXIT_SUCCESS;
}

Key changes addressing the flagged weakness (CWE-119/CWE-120):

1. **Replaced statically-sized stack arrays** with dynamically allocated buffers (`calloc`) sized explicitly to `UCHAR_MAX + 1`, eliminating the large fixed stack allocations and the static-array warning.
2. **Zero-initialization via `calloc`** guarantees the mapping tables start in a known state (equivalent to the previous `= {0}` / `= {false}` initializers).
3. **Allocation failure handling** — if any allocation fails, all previously allocated memory is freed and the function returns safely instead of dereferencing `NULL`.
4. **Explicit bounds validation** before every array index (`(size_t)c1 >= table_size || (size_t)c2 >= table_size`) as a defensive guard, with cleanup on the failure path.
5. **Proper cleanup on all exit paths** — memory is freed both on early termination (via `result`/`break` pattern) and on success, preventing leaks.
