#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 65536U

char *remove_duplicate_words(const char *input)
{
    size_t len;
    size_t result_len;
    char *copy;
    char *result;
    char **seen;
    size_t seen_count;
    size_t seen_cap;
    char *token;
    char *saveptr;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, input, len);
    copy[len] = '\0';

    result = malloc(len + 1U);
    if (result == NULL) {
        free(copy);
        return NULL;
    }
    result[0] = '\0';
    result_len = 0U;

    seen = NULL;
    seen_count = 0U;
    seen_cap = 0U;
    saveptr = NULL;

    token = strtok_r(copy, " \t\n", &saveptr);
    while (token != NULL) {
        int is_duplicate = 0;
        size_t i;

        for (i = 0U; i < seen_count; i++) {
            if (strcmp(seen[i], token) == 0) {
                is_duplicate = 1;
                break;
            }
        }

        if (is_duplicate == 0) {
            size_t token_len;

            if (seen_count == seen_cap) {
                size_t new_cap = (seen_cap == 0U) ? 8U : (seen_cap * 2U);
                char **tmp;

                if (new_cap > (SIZE_MAX / sizeof(char *))) {
                    free(seen);
                    free(copy);
                    free(result);
                    return NULL;
                }
                tmp = realloc(seen, new_cap * sizeof(char *));
                if (tmp == NULL) {
                    free(seen);
                    free(copy);
                    free(result);
                    return NULL;
                }
                seen = tmp;
                seen_cap = new_cap;
            }
            seen[seen_count] = token;
            seen_count++;

            token_len = strnlen(token, len);

            if (result_len > 0U) {
                if (result_len + 1U > len) {
                    free(seen);
                    free(copy);
                    free(result);
                    return NULL;
                }
                result[result_len] = ' ';
                result_len++;
            }

            if (token_len > len || result_len > len - token_len) {
                free(seen);
                free(copy);
                free(result);
                return NULL;
            }
            memcpy(result + result_len, token, token_len);
            result_len += token_len;
            result[result_len] = '\0';
        }

        token = strtok_r(NULL, " \t\n", &saveptr);
    }

    free(seen);
    free(copy);
    return result;
}

{

    if (len > 0U && buffer[len - 1U] == '\n') {

    if (deduplicated == NULL) {
        fprintf(stderr, "Error: failed to process the string\n");

**Analysis of the original code and the changes made:**

The original code was already quite defensive, and the Flawfinder hits were largely false positives — but here is how each flagged item was addressed:

1. **First `memcpy` (CWE-120) into `copy`:** This was already safe — `copy` is allocated with `len + 1` bytes and exactly `len` bytes are copied, with `len` bounded by `MAX_INPUT_LEN`. Kept as-is since the invariant is provably correct.

2. **Second `memcpy` (CWE-120) into `result`:** The original check `result_len + token_len > len` could theoretically wrap on addition if the values were ever corrupted. Rewritten as `token_len > len || result_len > len - token_len`, which is mathematically equivalent under valid invariants but provably free of unsigned overflow, making the bound check robust by construction.

3. **Statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120):** A 64 KiB array on the stack risks stack exhaustion on platforms with small default stack sizes. Replaced with a heap allocation via `malloc`, with the allocation checked and properly freed on all exit paths. `fgets` still limits reads to the buffer size, so no overflow is possible.

Other notes: `strtok_r` (the reentrant variant) is correctly used instead of `strtok`, `strnlen` is used everywhere instead of `strlen` to enforce upper bounds, all allocation failures are handled with full cleanup, and the `realloc` growth path guards against `size_t` multiplication overflow. The code remains C99/POSIX-2008 compliant.
