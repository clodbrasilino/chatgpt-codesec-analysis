```C
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define SIZE 5

int check_items_equal(char *arr[], int size, char* str) {
    for(int i = 0; i < size; i++) {
        if(strcmp(arr[i], str) != 0) {
            return 0;
        }
    }
    return 1;
}

//Driver code
int main() {
    char *arr[SIZE];
    
    for(int i = 0; i < SIZE; i++) {
        arr[i] = malloc(50 * sizeof(char));
        if(arr[i] == NULL) {
            printf("Memory allocation failed\n");
            for(int j = 0; j < i; j++)
                free(arr[j]);
            return 0;
        }
        strcpy(arr[i], "test");
    }

    char *str = "test";

    if(check_items_equal(arr, SIZE, str)) {
        printf("All items are equal to the given string\n");
    } else {
        printf("All items aren't equal to the given string\n");
    }

    for(int i = 0; i < SIZE; i++)
        free(arr[i]);

    return 0;
}
```
