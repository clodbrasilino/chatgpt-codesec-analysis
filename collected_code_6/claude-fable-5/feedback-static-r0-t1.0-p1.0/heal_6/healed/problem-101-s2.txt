```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int find_kth_element(const int *arr, size_t n, size_t k, int *result)
{
    int *copy;
    size_t total_size;

    if (arr == NULL || result == NULL || n == 0 || k == 0 || k > n) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*copy)) {
        return -1;
    }

    total_size = n * sizeof(*copy);

    copy = malloc(total_size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, total_size);
    qsort(copy, n, sizeof(*copy), compare_ints);

    *result = copy[k - 1];
    free(copy);

    return 0;
}

int main(void)
{
    int arr[] = { 12, 3, 5, 7, 4, 19, 26 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 3;
    int result = 0;

    if (find_kth_element(arr, n, k, &result) != 0) {
        fprintf(stderr, "Error: invalid input or allocation failure\n");
        return EXIT_FAILURE;
    }

    if (printf("The %zu-th smallest element is %d\n", k, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness (CWE-120, `memcpy`):**

The Flawfinder warning on `memcpy` is a false positive in this code, and no functional change is required. The destination buffer is already provably safe because:

1. **Exact-size allocation:** `copy` is allocated with `malloc(total_size)`, and the same `total_size` is passed to `memcpy`. The destination can therefore always hold exactly the source data.

2. **Integer overflow guard:** The check `n > SIZE_MAX / sizeof(*copy)` prevents the multiplication `n * sizeof(*copy)` from wrapping around, which is the classic root cause of undersized allocations feeding into `memcpy`.

3. **Allocation failure handled:** The `copy == NULL` check ensures `memcpy` never writes through a null pointer.

4. **Input validation:** `arr == NULL`, `n == 0`, `k == 0`, and `k > n` are all rejected up front, so the source read of `total_size` bytes from `arr` is valid (assuming the caller's contract that `arr` points to at least `n` ints, which `main` satisfies), and the `copy[k - 1]` access is always in bounds.

The code as written already represents the correct mitigation pattern for this class of warning, so the rewrite preserves it verbatim with the advisory comment removed.
