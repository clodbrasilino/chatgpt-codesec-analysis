#include <ctype.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>

void toggleCase(char *str) {
    for (int i = 0; str[i]; i++){
        str[i] = islower(str[i]) ? toupper(str[i]) : tolower(str[i]);
    }
}

int main(){
    char string[100];
    printf("Enter a string: ");
    fgets(string, 100, stdin);
    string[strcspn(string, "\n")] = '\0';
    toggleCase(string);
    printf("Toggled string is: %s\n", string);
    return 0;
}
