```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SENTENCE_LEN 4096

int find_first_max_even_word(const char *sentence, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len = 0;
    size_t max_len = 0;
    size_t best_start = 0;
    int found = 0;

    if (sentence == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len == MAX_SENTENCE_LEN) {
        return -1;
    }

    while (i < len) {
        while (i < len && sentence[i] == ' ') {
            i++;
        }
        if (i < len) {
            size_t start = i;
            size_t word_len = 0;
            while (i < len && sentence[i] != ' ') {
                i++;
            }
            word_len = i - start;
            if ((word_len % 2 == 0) && (word_len > max_len)) {
                max_len = word_len;
                best_start = start;
                found = 1;
            }
        }
    }

    if (!found) {
        return -1;
    }

    if (max_len >= result_size) {
        return -1;
    }

    if (best_start > len || max_len > len - best_start) {
        return -1;
    }

    memcpy(result, sentence + best_start, max_len);
    result[max_len] = '\0';
    return 0;
}

int main(void)
{
    const char *sentence = "It is a pleasant day today";
    char result[256];

    memset(result, 0, sizeof(result));

    if (find_first_max_even_word(sentence, result, sizeof(result)) == 0) {
        result[sizeof(result) - 1] = '\0';
        if (printf("First maximum length even word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No even length word found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

1. **`memcpy` (CWE-120)** — Already mitigated. Before the copy, the code verifies `max_len >= result_size` (rejecting copies that would not leave room for the NUL terminator) and validates that `best_start` and `max_len` lie within the source string bounds. The destination write of `result[max_len]` is safe because `max_len < result_size` is guaranteed at that point.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — Already mitigated. The buffer size is passed explicitly via `sizeof(result)` rather than being assumed inside the callee, and the function performs bounds checking against `result_size`. The defensive `result[sizeof(result) - 1] = '\0'` in `main` is redundant but harmless.

Additional robustness already present that's worth noting: `strnlen` with `MAX_SENTENCE_LEN` prevents unbounded reads of a non-terminated input, NULL-pointer and zero-size guards handle invalid arguments, and all length arithmetic uses `size_t` with overflow-safe comparisons (`max_len > len - best_start` instead of `best_start + max_len > len`).
