```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by an explicit maximum length parameter passed to `octal_to_decimal`, guaranteeing no over-read even on a non-terminated buffer. If the string fills the entire buffer without a terminator being found, the function rejects it.

2. **CWE-120 (`scanf` with `%s`):** Replaced `scanf("%63s", ...)` with `fgets(input, sizeof(input), stdin)`, which is always bounded by the buffer size and guarantees null-termination. The trailing newline is stripped safely, and overly long input is detected and rejected (with the remainder of the line drained from `stdin`).

3. **CWE-119/CWE-120 (statically-sized array):** The buffer size is defined once via `INPUT_SIZE` and consistently referenced with `sizeof(input)`, eliminating the risk of the size and the limit drifting apart during maintenance.

4. The existing overflow guard (`value > (LONG_MAX - digit) / 8`) was already correct and is retained, so the conversion can never overflow `long`.
