```c
#include <stdio.h>
#include <regex.h>
#include <string.h>

int match_pattern(const char *str) {
    regex_t regex;
    int reti;
    char msgbuf[100];

    /* Compile regular expression */
    reti = regcomp(&regex, "a.*b$", 0);
    if (reti) {
        return -1;
    }

    /* Execute regular expression */
    reti = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);
    if (!reti) {
        return 1;
    }
    else if (reti == REG_NOMATCH) {
        return 0;
    }
    else {
        regerror(reti, &regex, msgbuf, sizeof(msgbuf));
        printf("Regex match failed: %s\n", msgbuf);
        return -1;
    }
}

int main(void) {
    char str1[] = "abc";
    char str2[] = "ab";
    char str3[] = "bc";

    if (match_pattern(str1) == 1) {
        printf("Match found in %s\n", str1);
    }

    if (match_pattern(str2) == 1) {
        printf("Match found in %s\n", str2);
    }

    if (match_pattern(str3) == 1) {
        printf("Match found in %s\n", str3);
    }

    return 0;
}
```
