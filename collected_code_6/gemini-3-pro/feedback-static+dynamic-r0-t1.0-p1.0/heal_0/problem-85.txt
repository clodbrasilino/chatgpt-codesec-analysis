```c
#include <stdio.h>
#include <stdlib.h>

#define PI 3.14159265358979323846

double calculate_sphere_surface_area(double radius) {
    if (radius < 0.0) {
        return -1.0;
    }
    return 4.0 * PI * radius * radius;
}

int main(void) {
    double radius;
    double surface_area;

    if (scanf("%lf", &radius) != 1) {
        return EXIT_FAILURE;
    }

    surface_area = calculate_sphere_surface_area(radius);

    if (surface_area < 0.0) {
        return EXIT_FAILURE;
    }

    printf("%.6f\n", surface_area);

    return EXIT_SUCCESS;
}
```
