```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

size_t count_std_occurrences(const char *str)
{
    size_t count = 0;
    const char *pos = NULL;

    if (str == NULL)
    {
        return 0;
    }

    pos = str;
    while ((pos = strstr(pos, "std")) != NULL)
    {
        count++;
        pos += 3;
    }

    return count;
}

int main(void)
{
    const char *test_string = "std is used in C++ like std::cout and std::vector";
    size_t occurrences = 0;

    occurrences = count_std_occurrences(test_string);

    if (printf("The string \"%s\" contains 'std' %zu time(s).\n", test_string, occurrences) < 0)
    {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
