#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool is_divisible_by_digits(int num) {
    if (num == 0) {
        return false;
    }
    
    int temp = num;
    if (temp < 0) {
        temp = -temp;
    }
    
    while (temp > 0) {
        int digit = temp % 10;
        if (digit == 0 || num % digit != 0) {
            return false;
        }
        temp /= 10;
    }
    
    return true;
}

void find_numbers(int start, int end, int** result, int* count) {
    if (start > end || result == NULL || count == NULL) {
        if (count != NULL) *count = 0;
        if (result != NULL) *result = NULL;
        return;
    }

    int capacity = 10;
    *result = (int*)malloc(capacity * sizeof(int));
    if (*result == NULL) {
        *count = 0;
        return;
    }

    *count = 0;
    
    for (int i = start; i <= end; ++i) {
        if (is_divisible_by_digits(i)) {
            if (*count >= capacity) {
                capacity *= 2;
                int* temp = (int*)realloc(*result, capacity * sizeof(int));
                if (temp == NULL) {
                    free(*result);
                    *result = NULL;
                    *count = 0;
                    return;
                }
                *result = temp;
            }
            (*result)[*count] = i;
            (*count)++;
        }
    }
}

int main(void) {
    int start = 1;
    int end = 100;
    int* result = NULL;
    int count = 0;

    find_numbers(start, end, &result, &count);

    if (result != NULL) {
        for (int i = 0; i < count; ++i) {
            printf("%d ", result[i]);
        }
        printf("\n");
        free(result);
    }

    return 0;
}
