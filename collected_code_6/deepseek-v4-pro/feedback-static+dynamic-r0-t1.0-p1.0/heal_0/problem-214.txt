```c
#include <stdio.h>

#define PI 3.14159265358979323846

double radians_to_degrees(double radians) {
    return radians * (180.0 / PI);
}

int main(void) {
    double radians;

    if (scanf("%lf", &radians) != 1) {
        return 1;
    }

    double degrees = radians_to_degrees(radians);
    printf("%.6f\n", degrees);

    return 0;
}
```
