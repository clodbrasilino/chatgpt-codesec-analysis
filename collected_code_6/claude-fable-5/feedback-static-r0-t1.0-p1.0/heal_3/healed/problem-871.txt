```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096U

int are_rotations(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t temp_size;
    char *temp;
    int result;
    int written;

    if (s1 == NULL || s2 == NULL) {
        return 0;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN);
    len2 = strnlen(s2, MAX_INPUT_LEN);

    if (len1 >= MAX_INPUT_LEN || len2 >= MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2) {
        return 0;
    }

    if (len1 == 0) {
        return 1;
    }

    if (len1 > (SIZE_MAX - 1U) / 2U) {
        return -1;
    }

    temp_size = 2U * len1 + 1U;

    if (temp_size > (size_t)INT32_MAX) {
        return -1;
    }

    temp = (char *)malloc(temp_size);
    if (temp == NULL) {
        return -1;
    }

    written = snprintf(temp, temp_size, "%s%s", s1, s1);
    if (written < 0 || (size_t)written >= temp_size) {
        free(temp);
        return -1;
    }

    result = (strstr(temp, s2) != NULL) ? 1 : 0;

    free(temp);
    return result;
}

int main(void)
{
    const char *str1 = "ABCD";
    const char *str2 = "CDAB";
    const char *str3 = "ACBD";
    int res;

    res = are_rotations(str1, str2);
    if (res < 0) {
        fprintf(stderr, "Error: allocation failure or input too long\n");
        return EXIT_FAILURE;
    }
    printf("\"%s\" and \"%s\": %s\n", str1, str2,
           (res == 1) ? "rotations" : "not rotations");

    res = are_rotations(str1, str3);
    if (res < 0) {
        fprintf(stderr, "Error: allocation failure or input too long\n");
        return EXIT_FAILURE;
    }
    printf("\"%s\" and \"%s\": %s\n", str1, str3,
           (res == 1) ? "rotations" : "not rotations");

    return EXIT_SUCCESS;
}
```

The key change addresses the flagged `memcpy` weaknesses (CWE-120): both unchecked copies were replaced with a single bounded `snprintf` call, which guarantees the destination buffer `temp` can never be overflowed since the write is limited to `temp_size` and the return value is validated for truncation. All existing defensive measures are preserved — the `NULL` checks, `strnlen` with an upper bound to guard against unterminated inputs, the overflow check on `2 * len1 + 1`, the length-equality short-circuit, and the allocation failure handling — so the function still returns `1` for rotations, `0` for non-rotations, and `-1` on error conditions.
