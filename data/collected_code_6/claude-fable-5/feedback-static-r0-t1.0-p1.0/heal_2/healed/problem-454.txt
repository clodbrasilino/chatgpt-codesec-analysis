```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

int match_word_with_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return 0;
    }

    result[0] = '\0';

    while (text[i] != '\0') {
        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }
        if (text[i] == '\0') {
            break;
        }
        {
            size_t start = i;
            int has_z = 0;

            while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
                if (tolower((unsigned char)text[i]) == 'z') {
                    has_z = 1;
                }
                i++;
            }
            if (has_z) {
                size_t len = i - start;

                if (len >= result_size) {
                    return 0;
                }
                memcpy(result, &text[start], len);
                result[len] = '\0';
                return 1;
            }
        }
    }
    return 0;
}

int main(void)
{
    const char *text = "The lazy dog jumped over the fence";
    char result[256];

    if (match_word_with_z(text, result, sizeof(result))) {
        if (printf("Found word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No word containing 'z' found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-defended against the flagged weaknesses; the Flawfinder hits are largely false positives here, but the rewrite keeps and slightly strengthens the mitigations:

1. **`memcpy` (CWE-120):** The copy is guarded by `if (len >= result_size) return 0;`, which guarantees room for both the `len` bytes and the NUL terminator. The destination size is passed explicitly via `result_size` and validated against `0` and `NULL` on entry, so no overflow is possible.

2. **Statically-sized array (CWE-119/CWE-120):** `result[256]` is passed together with `sizeof(result)`, so the callee never assumes a size. Bounds checking happens before any write.

3. **Additional hardening added:** `result[0] = '\0';` at function entry ensures the output buffer is always a valid string even on the failure path, preventing accidental reads of uninitialized memory by careless callers.

4. **Correct `ctype` usage:** All `isalpha`/`tolower` calls cast to `unsigned char`, avoiding undefined behavior with negative `char` values — already correct in the original and preserved.

5. **Return-value checking on `printf`** is retained, satisfying strict error-handling requirements.
