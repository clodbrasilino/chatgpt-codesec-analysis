```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096

int remove_whitespace(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t out_pos = 0;
    size_t input_len;
    const char *cursor;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    input_len = strnlen(input, MAX_INPUT_LENGTH);
    if (input_len >= MAX_INPUT_LENGTH) {
        return -1;
    }

    ret = regcomp(&regex, "[[:space:]]+", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;

    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_len - (size_t)(cursor - input));
            if (remaining >= output_size || out_pos >= output_size - remaining) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_pos, cursor, remaining);
            out_pos += remaining;
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            return -1;
        }

        if (match.rm_so > 0) {
            size_t chunk = (size_t)match.rm_so;
            if (chunk >= output_size || out_pos >= output_size - chunk) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_pos, cursor, chunk);
            out_pos += chunk;
        }

        if (match.rm_eo <= 0) {
            regfree(&regex);
            return -1;
        }

        cursor += match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        return -1;
    }

    output[out_pos] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "  Hello,   World!\tThis is\na test string.  ";
    char result[256];

    memset(result, 0, sizeof(result));

    if (remove_whitespace(test, result, sizeof(result)) != 0) {
        fprintf(stderr, "Failed to remove whitespace\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: \"%s\"\n", test) < 0) {
        return EXIT_FAILURE;
    }
    if (printf("Result:   \"%s\"\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-126 (over-read via `strlen`)** — Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LENGTH`. The input length is validated up front, and the remaining-length calculation inside the loop is bounded by the already-validated total length, so an unterminated buffer can never cause an unbounded read.

2. **CWE-120 (unchecked `memcpy`)** — The original bounds checks used `out_pos + remaining >= output_size`, which is vulnerable to integer overflow if `out_pos + remaining` wraps around `SIZE_MAX`. The rewritten checks use the subtraction form (`out_pos >= output_size - chunk` after verifying `chunk < output_size`), which cannot overflow and guarantees every `memcpy` destination has sufficient space, including room for the terminating NUL.

3. **CWE-119/CWE-120 (statically-sized array `result`)** — The buffer size is always passed via `sizeof(result)` and enforced inside `remove_whitespace`, and the buffer is zero-initialized with `memset` so it is never read uninitialized even on a partial failure path.

4. **Additional hardening** — Added a guard against `match.rm_eo <= 0`, which would otherwise cause an infinite loop (a zero-length match would never advance the cursor). All `regfree` calls remain on every exit path, so no regex resources leak.
