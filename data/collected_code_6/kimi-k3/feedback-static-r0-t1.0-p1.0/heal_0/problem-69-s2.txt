#include <stdio.h>
#include <stdlib.h>

int contains_sublist(const int *list, size_t list_size, const int *sublist, size_t sublist_size) {
    size_t i;
    size_t j;

    if (sublist_size == 0) {
        return 1;
    }

    if (list == NULL || sublist == NULL) {
        return 0;
    }

    if (sublist_size > list_size) {
        return 0;
    }

    for (i = 0; i <= list_size - sublist_size; i++) {
        for (j = 0; j < sublist_size; j++) {
            if (list[i + j] != sublist[j]) {
                break;
            }
        }
        if (j == sublist_size) {
            return 1;
        }
    }

    return 0;
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5, 6};
    int sublist[] = {3, 4, 5};
    int missing_sublist[] = {3, 5};
    size_t list_size = sizeof(list) / sizeof(list[0]);
    size_t sublist_size = sizeof(sublist) / sizeof(sublist[0]);
    size_t missing_sublist_size = sizeof(missing_sublist) / sizeof(missing_sublist[0]);

    if (contains_sublist(list, list_size, sublist, sublist_size)) {
        printf("Sublist found\n");
    } else {
        printf("Sublist not found\n");
    }

    if (contains_sublist(list, list_size, missing_sublist, missing_sublist_size)) {
        printf("Sublist found\n");
    } else {
        printf("Sublist not found\n");
    }

    return 0;
}
