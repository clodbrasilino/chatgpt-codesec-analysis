#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <ctype.h>

bool is_undulating(const char *num) {
    size_t len = strnlen(num, 256);
    
    if (len == 0 || len >= 256 || len < 3) return false;
    
    for (size_t i = 0; i < len; i++) {
        if (!isdigit((unsigned char)num[i])) return false;
    }
    
    if (num[0] == num[1]) return false;
    
    for (size_t i = 1; i < len - 1; i++) {
        if (num[i] == num[i-1] || num[i] == num[i+1]) {
            return false;
        }
        if (num[i-1] != num[i+1]) {
            return false;
        }
    }
    
    return true;
}

    char input[260];

    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;

    }
    
    if (len == 0 || len >= 256) {
        printf("Invalid input\n");
        return 1;
    }

    }

}
