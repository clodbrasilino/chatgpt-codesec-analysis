char *remove_uppercase(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_len;
    size_t result_cap;
    const char *cursor;
    int ret;

    if (input == NULL) {

    }

    input_len = strnlen(input, MAX_INPUT_LEN);

        return NULL;
    }

    result = malloc(result_cap);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[A-Z]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t consumed = (size_t)(cursor - input);

        if (consumed > input_len) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if (ret == REG_NOMATCH) {

                regfree(&regex);
                free(result);
                return NULL;
            }
            memcpy(result + result_len, cursor, remaining);
            result_len += remaining;
            break;
        }

            regfree(&regex);
            free(result);
            return NULL;
        }

            free(result);
            return NULL;

            (size_t)match.rm_eo > input_len - consumed) {

            free(result);
            return NULL;

            free(result);
            return NULL;

        result_len += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    if (result_len >= result_cap) {
        regfree(&regex);
        free(result);
        return NULL;
    }

    result[result_len] = '\0';

}
