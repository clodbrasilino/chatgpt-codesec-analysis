#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int findLongestConseqSubseq(int arr[], int n) {
    if (n <= 0) {
        return 0;
    }

    qsort(arr, n, sizeof(int), compare);

    int max_len = 1;
    int current_len = 1;

    for (int i = 1; i < n; i++) {
        if (arr[i] == arr[i - 1]) {
            continue;
        } else if (arr[i] == arr[i - 1] + 1) {
            current_len++;
        } else {
            if (current_len > max_len) {
                max_len = current_len;
            }
            current_len = 1;
        }
    }

    if (current_len > max_len) {
        max_len = current_len;
    }

    return max_len;
}

int main() {
    int arr[] = {1, 9, 3, 10, 4, 20, 2};
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = findLongestConseqSubseq(arr, n);
    printf("%d\n", result);

    return 0;
}
