```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

long min_swaps_for_balance(const char *str);
static int report_result(const char *str, long result);

long min_swaps_for_balance(const char *str)
{
    long open_count = 0;
    long close_count = 0;
    long swaps = 0;
    long imbalance = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (str[i] == '[') {
            open_count++;
        } else if (str[i] == ']') {
            close_count++;
        } else {
            return -1;
        }
    }

    if (open_count != close_count) {
        return -1;
    }

    open_count = 0;
    close_count = 0;

    for (i = 0; i < len; i++) {
        if (str[i] == '[') {
            open_count++;
            if (imbalance > 0) {
                swaps += imbalance;
                imbalance--;
            }
        } else {
            close_count++;
            imbalance = close_count - open_count;
        }
    }

    return swaps;
}

static int report_result(const char *str, long result)
{
    if (result < 0) {
        if (fprintf(stderr, "Invalid input: %s\n", str) < 0) {
            return -1;
        }
    } else {
        if (printf("String: %s -> Minimum swaps: %ld\n", str, result) < 0) {
            return -1;
        }
    }
    return 0;
}

int main(void)
{
    const char *test1 = "[]][][";
    const char *test2 = "[[][]]";
    const char *test3 = "]]][[[";
    long result;

    result = min_swaps_for_balance(test1);
    if (report_result(test1, result) < 0) {
        return EXIT_FAILURE;
    }

    result = min_swaps_for_balance(test2);
    if (report_result(test2, result) < 0) {
        return EXIT_FAILURE;
    }

    result = min_swaps_for_balance(test3);
    if (report_result(test3, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **CWE-126 (strlen over-read) fix:** Replaced `strlen()` with `strnlen()` bounded by `MAX_INPUT_LEN`. If the string is not null-terminated within the bound, the function safely rejects the input with `-1` instead of reading past the buffer.

2. **Bounded length validation:** Inputs at or beyond `MAX_INPUT_LEN` are treated as invalid, guaranteeing all subsequent loop iterations stay within a known-safe range.

3. **Reduced duplication:** Extracted the repeated result-reporting logic in `main` into a `static` helper (`report_result`), which also centralizes the I/O error checking on `printf`/`fprintf` return values.

4. **Preserved semantics:** The `NULL` check, character validation (only `[` and `]` allowed), balance verification (`open_count == close_count`), and the greedy minimum-swap algorithm remain unchanged, as they were already correct and defensive.
