#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

bool string_starts_with_number(const char *str, int number)
{
    char prefix[32];
    int written;
    size_t prefix_len;

    if (str == NULL) {
        return false;
    }

    written = snprintf(prefix, sizeof(prefix), "%d", number);
    if (written < 0 || (size_t)written >= sizeof(prefix)) {
        return false;
    }

    prefix_len = (size_t)written;

    if (strncmp(str, prefix, prefix_len) != 0) {
        return false;
    }

    return true;
}

int main(void)
{
    const char *test1 = "42 is the answer";
    const char *test2 = "hello world";
    const char *test3 = "100 percent";
    int result_status = EXIT_SUCCESS;

    if (string_starts_with_number(test1, 42)) {
        if (printf("\"%s\" starts with 42\n", test1) < 0) {
            result_status = EXIT_FAILURE;
        }
    } else {
        if (printf("\"%s\" does not start with 42\n", test1) < 0) {
            result_status = EXIT_FAILURE;
        }
    }

    if (string_starts_with_number(test2, 7)) {
        if (printf("\"%s\" starts with 7\n", test2) < 0) {
            result_status = EXIT_FAILURE;
        }
    } else {
        if (printf("\"%s\" does not start with 7\n", test2) < 0) {
            result_status = EXIT_FAILURE;
        }
    }

    if (string_starts_with_number(test3, 100)) {
        if (printf("\"%s\" starts with 100\n", test3) < 0) {

        }
    } else {

}

The original code was already well-defended against the flagged weakness (CWE-119/CWE-120). The Flawfinder warning on `char prefix[32]` is a heuristic hit on any statically-sized array, but the code mitigates the risk properly:

1. **Bounded write:** `snprintf(prefix, sizeof(prefix), ...)` guarantees no write beyond the buffer, unlike `sprintf`. A 32-byte buffer is more than sufficient for any `int` representation (max ~12 characters including sign and NUL, even on 64-bit `int` platforms it would fit).

2. **Truncation/error detection:** The return value of `snprintf` is checked both for encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof(prefix)`), so a truncated prefix can never produce a false-positive match.

3. **NULL guard:** `str` is validated before use, preventing a NULL dereference in `strncmp`.

4. **Bounded comparison:** `strncmp` with an exact, verified length avoids reading past the prefix buffer.

5. **I/O error handling:** `printf` return values are checked in `main`, propagating failures through the exit status.
