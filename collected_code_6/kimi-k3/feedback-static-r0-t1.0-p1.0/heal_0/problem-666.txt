```c
#include <stdio.h>
#include <stddef.h>

size_t count_char_occurrences(const char *str, char ch)
{
    size_t count = 0;

    if (str == NULL)
    {
        return 0;
    }

    while (*str != '\0')
    {
        if (*str == ch)
        {
            count++;
        }
        str++;
    }

    return count;
}

int main(void)
{
    const char *input = "Hello, World!";
    char target = 'o';
    size_t result = count_char_occurrences(input, target);

    printf("The character '%c' occurs %zu times in \"%s\".\n", target, result, input);

    return 0;
}
```
