#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long count_factorial_digits(long n);

long count_factorial_digits(long n)
{
    double sum = 0.0;
    long i;

    if (n < 0) {
        return -1;
    }

    if (n <= 1) {
        return 1;
    }

    for (i = 2; i <= n; i++) {
        sum += log10((double)i);
    }

    return (long)floor(sum) + 1;
}

int main(void)
{
    long n = 0;
    long result = 0;
    int status = 0;

    printf("Enter a non-negative integer: ");

    status = scanf("%ld", &n);
    if (status != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    result = count_factorial_digits(n);
    if (result < 0) {
        fprintf(stderr, "Error: factorial is not defined for negative numbers.\n");
        return EXIT_FAILURE;
    }

    printf("Number of digits in %ld! is %ld\n", n, result);

    return EXIT_SUCCESS;
}
