#include <stdio.h>
#include <stdlib.h>

int find_solution(int a, int b, int n, int *x, int *y) {
    if (a == 0 && b == 0) return 0;
    if (a == 0) {
        if (n % b == 0) {
            *x = 0;
            *y = n / b;
            return 1;
        } else {
            return 0;
        }
    }
    if (b == 0) {
        if (n % a == 0) {
            *x = n / a;
            *y = 0;
            return 1;
        } else {
            return 0;
        }
    }

    int gcd, x1, y1;

    // Extended Euclidean Algorithm
    for (int i = 0; ; i++) {
        if (a % b == 0) {
            gcd = b;
            break;
        }
        a = a % b;
        if (a % b == 0) {
            gcd = a;
            break;
        }
        b = b % a;
    }

    if (n % gcd != 0) return 0;

    a = a / gcd;
    n = n / gcd;

    x1 = 1;
    y1 = -(a - 1);

    if (n < 0) {
        n = -n;
        x1 = -x1;
        y1 = -y1;
    }

    while (y1 < 0) {
        x1 += a;
        y1 += b / gcd;
    }

    *x = x1 * n;
    *y = y1 * n / (b / gcd);

    return 1;
}

int main() {
    int a, b, n, x, y;
    printf("Enter values of a, b, and n: ");
    scanf("%d %d %d", &a, &b, &n);

    if (find_solution(a, b, n, &x, &y)) {
        printf("Solution found: x = %d, y = %d\n", x, y);
    } else {
        printf("No solution found.\n");
    }

    return 0;
}
