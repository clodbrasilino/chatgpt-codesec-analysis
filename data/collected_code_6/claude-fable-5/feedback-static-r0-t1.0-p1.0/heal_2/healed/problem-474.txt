```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

size_t replace_chars(char *str, size_t max_len, char old_char, char new_char);

size_t replace_chars(char *str, size_t max_len, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    if (old_char == '\0' || new_char == '\0') {
        return 0;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    const char source[] = "hello world, hello everyone";
    size_t len;
    size_t buffer_size;
    char *buffer;
    size_t replaced;

    len = strnlen(source, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not terminated\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    buffer = malloc(buffer_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (len >= buffer_size) {
        fprintf(stderr, "Buffer too small for copy\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    memcpy(buffer, source, len);
    buffer[len] = '\0';

    printf("Original: %s\n", buffer);

    replaced = replace_chars(buffer, buffer_size, 'l', 'L');

    printf("Modified: %s\n", buffer);
    printf("Replacements made: %zu\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fix applied:**

The original code was already quite defensive, but the flagged weakness concerned the `memcpy` call (CWE-120, classic buffer overflow risk). While the original logic was actually safe — `buffer_size` is computed as `len + 1` and only `len` bytes are copied, leaving room for the null terminator — static analyzers like Flawfinder cannot verify this invariant and flag `memcpy` unconditionally.

The fix adds an explicit runtime guard (`if (len >= buffer_size)`) immediately before the `memcpy`, guaranteeing the destination can always hold the source data plus the terminator. This makes the safety invariant explicit and verifiable rather than implicit, satisfies the CWE-120 concern, and costs virtually nothing at runtime.

Other strengths already present in the code that were preserved:

- **`strnlen` with a bound** instead of `strlen`, protecting against unterminated input (CWE-126).
- **NULL checks** on both the `malloc` result and the `str` parameter in `replace_chars`.
- **Bounded iteration** in `replace_chars` (`i < max_len && str[i] != '\0'`), preventing out-of-bounds reads even if the string were unterminated.
- **Explicit null termination** of the buffer after the copy.
- **Pointer nulling after `free`**, mitigating use-after-free/double-free mistakes (CWE-416/CWE-415).
- **Correct format specifier** (`%zu`) for `size_t`.
