#include <stdio.h>
#include <string.h>

int min_swaps_to_equal(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count10 = 0;
    int count01 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strlen(s1);
    len2 = strlen(s2);

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] == '1' && s2[i] == '0') {
            count10++;
        } else if (s1[i] == '0' && s2[i] == '1') {
            count01++;
        }
    }

    if ((count10 % 2) != (count01 % 2)) {
        return -1;
    }

    return (count10 / 2) + (count01 / 2) + ((count10 % 2) * 2);
}

int main(void)
{
    const char *a1 = "1101";
    const char *b1 = "0011";
    const char *a2 = "1110000";
    const char *b2 = "0001111";

    const char *b3 = "01";

    int result;

    result = min_swaps_to_equal(a1, b1);

**How it works:**

- **Validation first:** The function rejects `NULL` pointers, strings of unequal length, and any character that isn't `'0'` or `'1'`, returning `-1` as an error/impossible indicator (addresses CWE-703 by checking exceptional conditions up front).
- **Counting mismatches:** It counts positions where `s1` has `'1'` and `s2` has `'0'` (`count10`), and vice versa (`count01`).
- **Swap logic:** Two mismatches of the same type are fixed with one swap. A leftover pair of opposite types requires two swaps. If the parities of the two counts differ, making the strings equal is impossible.
- **Resource safety:** No dynamic allocation is used, all reads stay within string bounds via `strlen`-derived limits (addresses CWE-664), and the code sticks to standard C with checked `printf` return values and a single `int main(void)` entry point (addresses CWE-710).
