#include <stdio.h>
#include <stdlib.h>

int isLinear(const int *sequence, size_t length) {
    if (sequence == NULL || length < 2) {
        return 0;
    }
    
    int difference = sequence[1] - sequence[0];
    
    for (size_t i = 2; i < length; i++) {
        if (sequence[i] - sequence[i - 1] != difference) {
            return 0;
        }
    }
    
    return 1;
}

int main(void) {
    size_t n;
    
    if (scanf("%zu", &n) != 1 || n == 0) {
        return EXIT_FAILURE;
    }
    
    int *sequence = malloc(n * sizeof(int));
    if (sequence == NULL) {
        return EXIT_FAILURE;
    }
    
    for (size_t i = 0; i < n; i++) {
        if (scanf("%d", &sequence[i]) != 1) {
            free(sequence);
            return EXIT_FAILURE;
        }
    }
    
    if (isLinear(sequence, n)) {
        printf("Linear\n");
    } else {
        printf("Not linear\n");
    }
    
    free(sequence);
    return EXIT_SUCCESS;
}
