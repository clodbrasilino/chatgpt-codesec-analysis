#include <stdio.h>
#include <stdlib.h>

int find_highest_product_pair(const int *arr, size_t size, int *pair1, int *pair2) {
    if (arr == NULL || pair1 == NULL || pair2 == NULL || size < 2) {
        return -1;
    }

    int max1, max2, min1, min2;

    if (arr[0] > arr[1]) {
        max1 = arr[0];
        max2 = arr[1];
        min1 = arr[1];
        min2 = arr[0];
    } else {
        max1 = arr[1];
        max2 = arr[0];
        min1 = arr[0];
        min2 = arr[1];
    }

    for (size_t i = 2; i < size; ++i) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    if ((long long)max1 * max2 > (long long)min1 * min2) {
        *pair1 = max1;
        *pair2 = max2;
    } else {
        *pair1 = min1;
        *pair2 = min2;
    }

    return 0;
}

int main(void) {
    int arr[] = {1, 4, 3, 6, 7, 0, -10, -11};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int pair1 = 0;
    int pair2 = 0;

    if (find_highest_product_pair(arr, size, &pair1, &pair2) == 0) {
        printf("%d %d\n", pair1, pair2);
    } else {
        fprintf(stderr, "Error: Invalid input or insufficient array elements.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
