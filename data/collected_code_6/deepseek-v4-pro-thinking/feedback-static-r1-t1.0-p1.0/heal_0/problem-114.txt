#include <stdio.h>

typedef struct {
    int x;
    int y;
    unsigned int frequency;
} Tuple;

void assign_frequencies(Tuple *tuples, size_t count) {
    if (tuples == NULL || count == 0) {
        return;
    }

    for (size_t i = 0; i < count; ++i) {
        tuples[i].frequency = 0;
    }

    for (size_t i = 0; i < count; ++i) {
        if (tuples[i].frequency != 0) {
            continue;
        }

        unsigned int freq = 1;
        for (size_t j = i + 1; j < count; ++j) {
            if (tuples[j].x == tuples[i].x && tuples[j].y == tuples[i].y) {
                ++freq;
            }
        }

        for (size_t j = i; j < count; ++j) {
            if (tuples[j].x == tuples[i].x && tuples[j].y == tuples[i].y) {
                tuples[j].frequency = freq;
            }
        }
    }
}

int main(void) {
    Tuple tuples[] = {
        {1, 2, 0},
        {3, 4, 0},
        {1, 2, 0},
        {5, 6, 0},
        {3, 4, 0},
        {1, 2, 0}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    assign_frequencies(tuples, count);

    for (size_t i = 0; i < count; ++i) {
        printf("(%d, %d): %u\n", tuples[i].x, tuples[i].y, tuples[i].frequency);
    }

    return 0;
}
