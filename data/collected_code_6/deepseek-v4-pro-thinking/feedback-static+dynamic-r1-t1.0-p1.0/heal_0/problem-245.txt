#include <stdio.h>
#include <stdlib.h>

int maxBitonicSum(int arr[], int n) {
    if (n <= 0) {
        return 0;
    }

    int *inc = (int *)malloc((size_t)n * sizeof(int));
    int *dec = (int *)malloc((size_t)n * sizeof(int));

    if (inc == NULL || dec == NULL) {
        free(inc);
        free(dec);
        return 0;
    }

    for (int i = 0; i < n; i++) {
        inc[i] = arr[i];
        dec[i] = arr[i];
    }

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[j] < arr[i] && inc[i] < inc[j] + arr[i]) {
                inc[i] = inc[j] + arr[i];
            }
        }
    }

    for (int i = n - 1; i >= 0; i--) {
        for (int j = n - 1; j > i; j--) {
            if (arr[j] < arr[i] && dec[i] < dec[j] + arr[i]) {
                dec[i] = dec[j] + arr[i];
            }
        }
    }

    int maxSum = inc[0] + dec[0] - arr[0];

    for (int i = 1; i < n; i++) {
        int sum = inc[i] + dec[i] - arr[i];
        if (sum > maxSum) {
            maxSum = sum;
        }
    }

    free(inc);
    free(dec);

    return maxSum;
}

int main(void) {
    int arr[] = {1, 15, 51, 45, 33, 100, 12, 18, 9};
    int n = (int)(sizeof(arr) / sizeof(arr[0]));

    printf("%d\n", maxBitonicSum(arr, n));

    return 0;
}
