```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_NAME_LEN 64
#define MAX_ENTRIES 128

typedef struct {
    char name[MAX_NAME_LEN];
    int score;
} Tuple;

typedef struct {
    char name[MAX_NAME_LEN];
    int total;
} Aggregate;

static int find_aggregate(const Aggregate *aggs, size_t count, const char *name)
{
    size_t i;

    if (aggs == NULL || name == NULL) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        if (strcmp(aggs[i].name, name) == 0) {
            return (int)i;
        }
    }
    return -1;
}

static int copy_name(char *dest, size_t dest_size, const char *src)
{
    int written;

    if (dest == NULL || src == NULL || dest_size == 0) {
        return -1;
    }

    written = snprintf(dest, dest_size, "%s", src);
    if (written < 0 || (size_t)written >= dest_size) {
        return -1;
    }
    return 0;
}

static int max_aggregate(const Tuple *tuples, size_t count, Aggregate *result)
{
    Aggregate aggs[MAX_ENTRIES];
    size_t agg_count = 0;
    size_t i;
    size_t max_index = 0;

    if (tuples == NULL || result == NULL || count == 0 || count > MAX_ENTRIES) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        int idx = find_aggregate(aggs, agg_count, tuples[i].name);
        if (idx >= 0) {
            aggs[idx].total += tuples[i].score;
        } else {
            if (agg_count >= MAX_ENTRIES) {
                return -1;
            }
            if (copy_name(aggs[agg_count].name, sizeof(aggs[agg_count].name),
                          tuples[i].name) != 0) {
                return -1;
            }
            aggs[agg_count].total = tuples[i].score;
            agg_count++;
        }
    }

    for (i = 1; i < agg_count; i++) {
        if (aggs[i].total > aggs[max_index].total) {
            max_index = i;
        }
    }

    if (copy_name(result->name, sizeof(result->name),
                  aggs[max_index].name) != 0) {
        return -1;
    }
    result->total = aggs[max_index].total;

    return 0;
}

int main(void)
{
    Tuple tuples[] = {
        {"Juan Whelan", 90},
        {"Sabah Colley", 88},
        {"Peter Nichols", 7},
        {"Juan Whelan", 122},
        {"Sabah Colley", 84}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    Aggregate result;

    if (max_aggregate(tuples, count, &result) != 0) {
        fprintf(stderr, "Failed to compute maximum aggregate\n");
        return EXIT_FAILURE;
    }

    if (printf("(%s, %d)\n", result.name, result.total) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Reduced scope of `idx`** — the variable is now declared inside the loop where it is used, addressing the cppcheck `variableScope` warning.

2. **Truncation detection added** — the original code only checked `snprintf` for a negative return (encoding error), but silently accepted truncated names. Truncated names could cause distinct long names to collide in `find_aggregate`, producing incorrect aggregates. The new `copy_name` helper rejects both errors and truncation (`written >= dest_size`), which properly mitigates the CWE-119/CWE-120 concern flagged on the fixed-size `name` buffers.

3. **Centralized bounded copy** — all string copies into fixed-size buffers now go through one audited helper using `sizeof(dest)` rather than a repeated macro, reducing the chance of a future size mismatch.

The remaining structure (NULL checks, `count` bounds validation against `MAX_ENTRIES`, checked `printf` return) was already sound and is preserved. The statically-sized arrays themselves are acceptable since every write path is now bounds-checked with explicit truncation handling.
