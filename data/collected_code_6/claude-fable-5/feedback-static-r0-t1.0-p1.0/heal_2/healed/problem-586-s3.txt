```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes ||
        head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fixes applied:**

The original code was already largely safe. The two Flawfinder warnings on `memcpy` are heuristic pattern matches — Flawfinder flags every use of `memcpy` regardless of context, since it cannot prove the destination buffer is large enough. Reviewing the logic:

1. **First `memcpy(temp, arr, head_bytes)`** — `temp` is allocated with exactly `head_bytes`, and `k < n` guarantees `head_bytes <= n * sizeof(int)`, so reading from `arr` stays in bounds. Safe.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`** — writes `k` elements starting at offset `n - k`, ending exactly at `arr + n`. Safe.

3. **`memmove(arr, arr + k, tail_bytes)`** — correctly uses `memmove` for the overlapping regions. Safe.

The rewritten version adds an explicit defensive invariant check (`head_bytes + tail_bytes == total_bytes` and individual bounds against `total_bytes`) to satisfy the CWE-120 concern by documenting, in code, that the destination regions can always hold the source data. Combined with the pre-existing `n > SIZE_MAX / sizeof(int)` overflow guard, all size arithmetic is provably free of unsigned wraparound before any copy occurs.

Other good practices already present and retained: NULL/zero-length input rejection, `k % n` normalization, checked `malloc`, checked `printf` return values, and no memory leaks on any path.
