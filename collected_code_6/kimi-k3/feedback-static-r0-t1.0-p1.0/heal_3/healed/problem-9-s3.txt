#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

int minRotations(const char *str, size_t n) {
    if (n == 0) return 0;
    
    if (n > SIZE_MAX / 2) return -1;
    
    int *concat = (int *)malloc(2 * n * sizeof(int));
    if (concat == NULL) return -1;
    
    for (size_t i = 0; i < 2 * n; i++) {
        concat[i] = (unsigned char)str[i % n];
    }
    
    for (size_t i = 1; i <= n; i++) {
        size_t j;
        for (j = 0; j < n; j++) {
            if (concat[i + j] != (unsigned char)str[j]) break;
        }
        if (j == n) {
            free(concat);
            return (int)i;
        }
    }
    
    free(concat);
    return (int)n;
}

int main(void) {
    char *str = NULL;
    size_t size = 0;
    ssize_t len;
    
    printf("Enter a string: ");
    
    len = getline(&str, &size, stdin);
    if (len == -1) {
        free(str);
        return 1;
    }
    
    if (len > 0 && str[len - 1] == '\n') {
        str[len - 1] = '\0';
        len--;
    }
    
    int result = minRotations(str, (size_t)len);
    if (result >= 0) {
        printf("Minimum rotations: %d\n", result);
    } else {
        printf("Error occurred\n");
        free(str);
        return 1;
    }
    
    free(str);
    return 0;
}
