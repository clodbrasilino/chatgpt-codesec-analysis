```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    total_bytes = n * sizeof(int);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);

    if (head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was already defensively written, but contained several **redundant and logically unreachable checks** that added noise without improving safety. Regarding the two Flawfinder warnings (CWE-120):

1. **First `memcpy(temp, arr, head_bytes)`** — This is safe because `temp` was allocated with exactly `head_bytes`, and `head_bytes` was already validated to not exceed `total_bytes` (the size of `arr`). The duplicate `head_bytes > total_bytes` check immediately after `malloc` was dead code (already verified earlier) and was removed.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`** — Safe because the destination offset `(n - k)` leaves exactly `k` elements (`head_bytes` bytes) of valid space in `arr`, and the source `temp` holds exactly `head_bytes` bytes. The guard `head_bytes > total_bytes - tail_bytes` is retained as an explicit invariant check satisfying the CWE-120 concern (destination capacity verified against copy length).

3. **Removed impossible checks**: `k > n` can never be true after `k = k % n`; `(n - k) > n` can never be true for unsigned values when `k <= n`; and the duplicate `tail_bytes > total_bytes` check before `memmove` was already performed.

4. **Retained essential protections**: the `n > SIZE_MAX / sizeof(int)` multiplication-overflow guard, the NULL/zero-length input validation, the `malloc` failure check, and the `head_bytes + tail_bytes != total_bytes` consistency invariant.

The result is functionally identical (output: `5 6 52 36 12 10`), keeps every bounds check that actually protects the two flagged `memcpy` calls, and eliminates the unreachable branches that obscured the real safety logic.
