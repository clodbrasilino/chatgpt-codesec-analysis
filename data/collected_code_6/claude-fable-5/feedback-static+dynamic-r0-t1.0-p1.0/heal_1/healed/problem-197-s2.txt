#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define TUPLE_SIZE 4

int safe_power(long long base, long long exp, long long *result)
{
    long long acc = 1;
    long long i;

    if (result == NULL) {
        return -1;
    }

    if (exp < 0) {
        return -1;
    }

    for (i = 0; i < exp; i++) {
        if (base != 0 && (acc > LLONG_MAX / base || acc < LLONG_MIN / base)) {
            return -1;
        }
        acc *= base;
    }

    *result = acc;
    return 0;
}

int tuple_exponentiation(const long long *tup1, const long long *tup2,
                         long long *out, size_t size)
{
    size_t i;

    if (tup1 == NULL || tup2 == NULL || out == NULL || size == 0) {
        return -1;
    }

    for (i = 0; i < size; i++) {
        if (safe_power(tup1[i], tup2[i], &out[i]) != 0) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    const long long tuple1[TUPLE_SIZE] = {2, 3, 4, 5};
    const long long tuple2[TUPLE_SIZE] = {3, 2, 2, 3};
    long long result[TUPLE_SIZE] = {0};
    size_t i;

    if (tuple_exponentiation(tuple1, tuple2, result, TUPLE_SIZE) != 0) {
        fprintf(stderr, "Error: exponentiation failed\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (i = 0; i < TUPLE_SIZE; i++) {
        printf("%lld", result[i]);
        if (i < TUPLE_SIZE - 1) {
            printf(", ");
        }
    }
    printf(")\n");

    return EXIT_SUCCESS;
}
