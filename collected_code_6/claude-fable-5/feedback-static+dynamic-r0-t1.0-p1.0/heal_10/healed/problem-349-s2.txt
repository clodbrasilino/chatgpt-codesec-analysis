#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U
#define FLUSH_LIMIT 1000000U

static bool is_binary_string(const char *str, size_t max_len);
static void flush_stdin(void);

static bool is_binary_string(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (max_len == 0U) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0U; (i < max_len) && (str[i] != '\0'); i++) {
        if ((str[i] != '0') && (str[i] != '1')) {
            return false;
        }
    }

    if (i >= max_len) {
        return false;
    }

    return true;
}

static void flush_stdin(void)
{
    int c;
    size_t count = 0U;

    do {
        c = fgetc(stdin);
        count++;
    } while ((c != '\n') && (c != EOF) && (count < FLUSH_LIMIT));
}

int main(void)
{
    char buffer[BUFFER_SIZE] = {0};
    size_t len;
    char *newline;

    printf("Enter a string: ");
    (void)fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    newline = memchr(buffer, '\n', sizeof(buffer));
    if (newline != NULL) {
        *newline = '\0';
    } else {
        len = strnlen(buffer, sizeof(buffer));
        if (len == (sizeof(buffer) - 1U)) {
            flush_stdin();
            fprintf(stderr, "Input too long\n");
            return EXIT_FAILURE;
        }
    }

    len = strnlen(buffer, sizeof(buffer));
    if (len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (is_binary_string(buffer, sizeof(buffer))) {
        printf("The string is a binary string\n");
    } else {
        printf("The string is not a binary string\n");
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why this code is safe:

**1. Flawfinder warning on `fgetc` in `flush_stdin` (CWE-120, CWE-20):** This is a false positive in practice. The loop never stores the read characters into any buffer — it only discards them. There is no buffer to overflow. Additionally, the `FLUSH_LIMIT` counter guards against an unbounded loop (e.g., a pathological stream that never delivers `\n` or `EOF`), so no denial-of-service via infinite loop is possible.

**2. Flawfinder warning on `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** The statically-sized array is properly protected:

- `fgets(buffer, (int)sizeof(buffer), stdin)` limits the read to at most `BUFFER_SIZE - 1` characters plus the null terminator — no overflow is possible.
- `buffer[sizeof(buffer) - 1U] = '\0';` guarantees null termination defensively.
- `memchr` and `strnlen` are both bounded by `sizeof(buffer)`, never scanning past the array.
- Truncated input (line longer than the buffer) is explicitly detected: if no newline was found and the string fills the buffer, the remainder of stdin is drained via `flush_stdin()` and the program exits with an error, preventing the leftover input from being misinterpreted by a subsequent read.
- `is_binary_string` takes an explicit `max_len` bound and rejects strings that are not null-terminated within that bound.
