```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t capacity = 8;
    size_t n = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && input[MAX_INPUT_LEN - 1] != '\0') {
        return NULL;
    }

    if (capacity > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;
                size_t alloc_len;
                char *segment;

                if (start > len || seg_len > len - start || seg_len >= SIZE_MAX) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                alloc_len = seg_len + 1;
                segment = malloc(alloc_len);
                if (segment == NULL) {
                    free_tokens(tokens, n);
                    return NULL;
                }

                memcpy(segment, input + start, seg_len);
                segment[seg_len] = '\0';

                if (n == capacity) {
                    if (capacity > SIZE_MAX / (2 * sizeof(char *))) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    capacity *= 2;
                    temp = realloc(tokens, capacity * sizeof(char *));
                    if (temp == NULL) {
                        free(segment);
                        free_tokens(tokens, n);
                        return NULL;
                    }
                    tokens = temp;
                }
                tokens[n] = segment;
                n++;
            }
            start = i + 1;
        }
    }

    *count = n;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *text = "HELLOworldFOObarBAZ123";
    char **parts = NULL;
    size_t count = 0;
    size_t i;

    parts = split_at_lowercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);
    printf("Found %zu segment(s):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, parts[i]);
    }

    free_tokens(parts, count);
    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes:**

1. **`alloc_len = seg_len + 1` overflow concern** — The original guard `seg_len > SIZE_MAX - 1` was tightened to `seg_len >= SIZE_MAX`, which is equivalent but clearer in intent: it guarantees `seg_len + 1` cannot wrap around before the `malloc(alloc_len)` call.

2. **`seg_len >= alloc_len` always-false condition (cppcheck `knownConditionTrueFalse`)** — This dead check was removed entirely. Since `alloc_len` is provably `seg_len + 1` with overflow already excluded, the condition could never trigger and was pure noise that hurt readability.

3. **Flawfinder `memcpy` warning (CWE-120)** — This is a false positive in context: the destination buffer is allocated with exactly `seg_len + 1` bytes immediately before the copy of `seg_len` bytes, and the NUL terminator is written within bounds at `segment[seg_len]`. The bounds relationship is enforced by construction, so `memcpy` is retained safely.

4. **`malloc` failure assumption** — The `segment == NULL` check was already correct and is preserved; all error paths properly release previously allocated tokens via `free_tokens`, avoiding leaks.

The remaining defensive checks (`strnlen` truncation detection, `SIZE_MAX / sizeof(char *)` multiplication guards before `malloc`/`realloc`, `realloc` via a temporary pointer) were all sound and kept intact.
