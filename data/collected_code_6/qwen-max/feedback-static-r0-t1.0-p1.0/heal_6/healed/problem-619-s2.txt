#include <stdio.h>
#include <string.h>
#include <ctype.h>

void move_numbers(char *str, size_t str_len) {
    char result[200] = "";
    char temp[200] = "";
    int j = 0, k = 0;

    for (int i = 0; i < str_len && str[i] != '\0'; i++) {
        if (isdigit((unsigned char)str[i])) {
            if (j < sizeof(result) - 1) {
                result[j++] = str[i];
            }
        } else {
            if (k < sizeof(temp) - 1) {
                temp[k++] = str[i];
            }
        }
    }

    result[j] = '\0';
    temp[k] = '\0';

    size_t remaining_space = sizeof(result) - strlen(result) - 1;
    if (remaining_space > 0) {
        strncat(result, temp, remaining_space);
    }

    size_t result_len = strlen(result);
    if (result_len <= str_len) {
        memcpy(str, result, result_len);
        str[result_len] = '\0';
    } else {
        memcpy(str, result, str_len - 1);
        str[str_len - 1] = '\0';
    }
}

int main() {
    char str[] = "a2b3c4d5e6";
    move_numbers(str, sizeof(str) - 1);
    printf("%s\n", str);
    return 0;
}
