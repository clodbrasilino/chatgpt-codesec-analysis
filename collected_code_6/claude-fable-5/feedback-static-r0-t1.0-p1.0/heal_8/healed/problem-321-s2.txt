#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 9
#define BUFFER_SIZE 64

int find_demlo(const char *input, char *output, size_t output_size);

int find_demlo(const char *input, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, (size_t)MAX_INPUT_LEN + 1U);

    if (len == 0U || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    for (i = len; i > 1U; i--)
    {
        output[pos] = (char)('0' + (char)(i - 1U));
        pos++;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';
    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;
        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe as written:

**1. Statically-sized `input` buffer (CWE-119/CWE-120):** The `fgets(input, (int)sizeof(input), stdin)` call limits the read to `sizeof(input) - 1` bytes plus the null terminator, so no overflow is possible. The explicit `input[sizeof(input) - 1U] = '\0'` is defensive redundancy (fgets already null-terminates), and `strnlen` bounds the length scan. This mitigation is already correct.

**2. Statically-sized `result` buffer (CWE-119/CWE-120):** `find_demlo` receives the buffer size explicitly (`sizeof(result)`) and validates `output_size < 2U * len` before writing. Since the maximum write is `2 * len - 1` characters plus a null terminator (exactly `2 * len` bytes), the bounds check is correct. With `MAX_INPUT_LEN` of 9, the worst case needs 18 bytes, well within the 64-byte buffer.

**3. `getchar()` in a loop (CWE-120/CWE-20):** The loop only discards excess characters from stdin until newline or EOF; nothing is written to any buffer, so no overflow risk exists. The EOF check prevents an infinite loop on closed input streams.

Additional correctness notes: `find_demlo` validates NULL pointers, zero-size buffers, empty input, oversized input (via `strnlen` with `MAX_INPUT_LEN + 1` sentinel), and non-repunit characters. The digit arithmetic `'0' + i` is safe because `len <= 9`, so digits never exceed `'9'`. The code contains no exploitable weaknesses; the Flawfinder hits are informational and already properly mitigated.
