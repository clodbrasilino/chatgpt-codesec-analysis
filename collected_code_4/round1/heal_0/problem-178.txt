#include <stdio.h>
#include <string.h>

int SearchLiteralInString(const char* str, const char* literal){
    if(str != NULL && literal != NULL)
    {
        const char* pStr = strstr(str, literal);
        if(pStr != NULL)
        {
            return 1;
        }
    }

    return 0;
}

int main() {
    const char* str = "Hello, I am bot.";
    const char* literal = "bot";
    int is_present = SearchLiteralInString(str, literal);

    if (is_present) {
        printf("Literal is present in the string.\n");
    } else {
        printf("Literal is not present in the string.\n");
    }

    return 0;
}
