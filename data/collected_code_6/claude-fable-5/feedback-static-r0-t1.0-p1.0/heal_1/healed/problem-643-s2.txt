```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_TEXT_LEN 4096

int find_word_with_inner_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len;

    if (text == NULL || result == NULL || result_size == 0) {
        return 0;
    }

    len = strnlen(text, MAX_TEXT_LEN);

    while (i < len) {
        while (i < len && !isalpha((unsigned char)text[i])) {
            i++;
        }

        if (i >= len) {
            break;
        }

        size_t start = i;

        while (i < len && isalpha((unsigned char)text[i])) {
            i++;
        }

        size_t word_len = i - start;

        if (word_len >= 3) {
            size_t j;
            int found = 0;

            for (j = start + 1; j < i - 1; j++) {
                if (tolower((unsigned char)text[j]) == 'z') {
                    found = 1;
                    break;
                }
            }

            if (found) {
                if (word_len >= result_size) {
                    return 0;
                }
                memcpy(result, &text[start], word_len);
                result[word_len] = '\0';
                return 1;
            }
        }
    }

    return 0;
}

int main(void)
{
    const char *tests[] = {
        "the amazing puzzle was solved",
        "zebra runs fast",
        "jazz music plays",
        "no match here",
        "a hazy day"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    char buffer[256];

    for (i = 0; i < count; i++) {
        memset(buffer, 0, sizeof(buffer));
        if (find_word_with_inner_z(tests[i], buffer, sizeof(buffer))) {
            if (printf("Input: \"%s\" -> Match: \"%s\"\n", tests[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("Input: \"%s\" -> No match\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Summary of the applied fixes:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_TEXT_LEN`, guaranteeing the scan never reads past a fixed limit even if the input string is not properly `\0`-terminated.

2. **CWE-120 (`memcpy` overflow):** The existing guard `if (word_len >= result_size) return 0;` already ensures the destination can hold `word_len` bytes plus the terminating `\0`, so the `memcpy` is provably safe. This check is retained.

3. **CWE-119/CWE-120 (statically-sized `char buffer[256]`):** The buffer is always passed together with its exact size via `sizeof(buffer)`, and every write into it is bounds-checked inside the function. Added a `memset` before each use to guarantee the buffer is always in a defined state.

The original code was already largely defensive (NULL checks, size checks before copying, `unsigned char` casts for the `ctype.h` functions); the main real improvement is the bounded string-length computation.
