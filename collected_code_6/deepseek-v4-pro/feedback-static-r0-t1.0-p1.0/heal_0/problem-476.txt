#include <stdio.h>
#include <limits.h>
#include <stddef.h>

int sum_of_extremes(const int arr[], size_t n, int *result) {
    if (arr == NULL || n == 0 || result == NULL) {
        return -1;
    }

    int min_val = INT_MAX;
    int max_val = INT_MIN;

    for (size_t i = 0; i < n; i++) {
        if (arr[i] < min_val) {
            min_val = arr[i];
        }
        if (arr[i] > max_val) {
            max_val = arr[i];
        }
    }

    *result = min_val + max_val;
    return 0;
}

int main(void) {
    int arr[] = {5, 12, 3, 19, 8, 1, 25};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result;

    if (sum_of_extremes(arr, n, &result) == 0) {
        printf("Sum of smallest and largest: %d\n", result);
    } else {
        printf("Invalid input\n");
    }

    return 0;
}
