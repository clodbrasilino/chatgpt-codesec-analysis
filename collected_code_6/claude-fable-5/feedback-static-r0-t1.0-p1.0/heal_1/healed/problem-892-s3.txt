```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256

void remove_multiple_spaces(char *str)
{
    size_t read_idx;
    size_t write_idx;
    int prev_space;

    if (str == NULL) {
        return;
    }

    read_idx = 0;
    write_idx = 0;
    prev_space = 0;

    while (str[read_idx] != '\0') {
        if (str[read_idx] == ' ') {
            if (!prev_space) {
                str[write_idx] = str[read_idx];
                write_idx++;
                prev_space = 1;
            }
        } else {
            str[write_idx] = str[read_idx];
            write_idx++;
            prev_space = 0;
        }
        read_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    const char *test = "This   is  a    string   with  multiple    spaces";
    size_t test_len;
    int written;

    test_len = strnlen(test, sizeof(buffer));
    if (test_len >= sizeof(buffer)) {
        fprintf(stderr, "Input string too long\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, sizeof(buffer), "%s", test);
    if (written < 0 || (size_t)written >= sizeof(buffer)) {
        fprintf(stderr, "Failed to copy input string\n");
        return EXIT_FAILURE;
    }

    printf("Before: \"%s\"\n", buffer);

    remove_multiple_spaces(buffer);

    printf("After:  \"%s\"\n", buffer);

    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **`strlen` → `strnlen`** (CWE-126): Bounds the read to `sizeof(buffer)`, so even a non-terminated string cannot cause an over-read beyond the limit we care about.

2. **`strncpy` → `snprintf`** (CWE-120): `snprintf` always null-terminates the destination (when size > 0) and its return value is checked to detect truncation or encoding errors, eliminating the classic `strncpy` pitfalls of missing termination and silent truncation.

3. **Statically-sized array** (CWE-119/CWE-120): The buffer size is centralized in a `BUFFER_SIZE` macro, all writes are bounds-checked before and after the copy, and `remove_multiple_spaces` only ever shrinks the string in place (`write_idx <= read_idx`), so it cannot overflow a properly terminated buffer.
