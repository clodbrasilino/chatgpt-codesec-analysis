```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <regex.h>

#define NUM_BUFFER_SIZE 64

static int copy_match(char *dest, size_t dest_size, const char *src, size_t len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (len >= dest_size) {
        return -1;
    }
    memcpy(dest, src, len);
    dest[len] = '\0';
    return 0;
}

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[NUM_BUFFER_SIZE];
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;
        long value;
        char *endptr;

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            regfree(&regex);
            return -1;
        }

        len = (size_t)(match.rm_eo - match.rm_so);

        if (copy_match(buffer, sizeof(buffer), cursor + match.rm_so, len) != 0) {
            regfree(&regex);
            return -1;
        }

        errno = 0;
        value = strtol(buffer, &endptr, 10);
        if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test1 = "abc 42 def -17 ghi 305 jkl 7";
    const char *test2 = "no numbers here";
    long max;

    if (extract_max_number(test1, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test1, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test1);
    }

    if (extract_max_number(test2, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test2, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test2);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive, but I made the following adjustments to address the Flawfinder warnings:

1. **CWE-119/CWE-120 (statically-sized `buffer[64]`)** — The buffer size is now defined via a named constant (`NUM_BUFFER_SIZE`), and all bounds logic references `sizeof(buffer)` rather than magic numbers, so the size check and the array declaration cannot silently drift apart.

2. **CWE-120 (`memcpy` without overflow check)** — The copy is now encapsulated in `copy_match()`, which performs the bounds check (`len >= dest_size`, reserving room for the NUL terminator) *atomically* with the `memcpy` and termination. This guarantees the length validation can never be separated from the copy by future refactoring, which was the residual risk in the original code where the check and the `memcpy` were several lines apart.

Other safety properties preserved from the original: `strtol` with `errno`/`endptr` validation guards against overflow and malformed input (CWE-190), `regfree` is called on every exit path (no resource leak), and the loop cannot spin on empty matches because the pattern `-?[0-9]+` requires at least one digit and `match.rm_eo <= match.rm_so` is rejected explicitly.
