#include <stdio.h>

#define MAX(a, b) ((a) > (b) ? (a) : (b))

int maxProfit(int k, const int* prices, int n) {
    if (n == 0 || k == 0) return 0;
    if (2 * k > n) {
        int profit = 0;
        for (int i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                profit += prices[i] - prices[i - 1];
            }
        }
        return profit;
    }

    int dp[k + 1][n];
    for (int i = 0; i <= k; i++) {
        for (int j = 0; j < n; j++) {
            if (i == 0 || j == 0) {
                dp[i][j] = 0;
            } else {
                int maxDiff = -prices[0];
                for (int m = 1; m <= j; m++) {
                    maxDiff = MAX(maxDiff, dp[i - 1][m - 1] - prices[m - 1]);
                }
                dp[i][j] = MAX(dp[i][j - 1], prices[j] + maxDiff);
            }
        }
    }
    return dp[k][n - 1];
}

int main() {
    int prices[] = {7, 1, 5, 3, 6, 4};
    int n = sizeof(prices) / sizeof(prices[0]);
    int k = 3;
    printf("Maximum profit: %d\n", maxProfit(k, prices, n));

    int prices2[] = {1, 2, 3, 4, 5};
    n = sizeof(prices2) / sizeof(prices2[0]);
    k = 2;
    printf("Maximum profit: %d\n", maxProfit(k, prices2, n));

    int prices3[] = {7, 6, 4, 3, 1};
    n = sizeof(prices3) / sizeof(prices3[0]);
    k = 1;
    printf("Maximum profit: %d\n", maxProfit(k, prices3, n));

    return 0;
}
