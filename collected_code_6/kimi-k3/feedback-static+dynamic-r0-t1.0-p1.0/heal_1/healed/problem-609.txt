#include <stdio.h>
#include <math.h>
#include <float.h>

double periodic_function(double x) {
    return sin(x) + cos(2.0 * x) + 0.5 * sin(3.0 * x);
}

double find_minimum(double start, double end, double step) {
    double min_value = DBL_MAX;
    double x;
    
    if (start >= end || step <= 0.0) {
        return DBL_MAX;
    }
    
    for (x = start; x <= end; x += step) {
        double current_value = periodic_function(x);
        if (current_value < min_value) {
            min_value = current_value;
        }
    }
    
    return min_value;
}

int main(void) {
    double period = 2.0 * M_PI;
    double start = 0.0;
    double end = period;
    double step = 0.0001;
    double result;
    
    result = find_minimum(start, end, step);
    
    if (result != DBL_MAX) {
        printf("Minimum value: %.10f\n", result);
    } else {
        printf("Error: Invalid parameters\n");
        return 1;
    }
    
    return 0;
}
