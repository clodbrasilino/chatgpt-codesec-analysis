```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_LEN 4096

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] != '(' && input[i] != ')') {
            if (j < len) {
                result[j] = input[i];
                j++;
            }
        }
    }
    result[j] = '\0';

    return result;
}

static int process_line(const char *input)
{
    char *output;

    if (input == NULL) {
        return -1;
    }

    output = remove_parentheses(input);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return -1;
    }
    if (printf("%s\n", output) < 0) {
        free(output);
        return -1;
    }
    free(output);
    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];

    memset(buffer, 0, sizeof(buffer));

    while (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        buffer[sizeof(buffer) - 1] = '\0';
        size_t len = strnlen(buffer, sizeof(buffer));
        if (len > 0 && buffer[len - 1] == '\n') {
            buffer[len - 1] = '\0';
        }
        if (process_line(buffer) != 0) {
            return EXIT_FAILURE;
        }
        memset(buffer, 0, sizeof(buffer));
    }

    if (ferror(stdin)) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fix:**

The failing test cases were the key diagnostic. Expected outputs like `'python'`, `'string'`, and `'alpha'` produced `<no output>` (an empty string), which indicates the test inputs were fully wrapped in parentheses, e.g. `(python)`. The original implementation removed the parentheses **and everything inside them** (tracking nesting via `depth` and only copying characters at `depth == 0`). The intended behavior, per the tests, is to strip only the `(` and `)` characters themselves while preserving the enclosed content.

Changes made:

1. **Removed the `depth` tracking logic** — characters are now copied unless they are `(` or `)`, which makes `(python)` correctly yield `python`.
2. **Dropped the now-unneeded `INT_MAX` check and `<limits.h>` dependency logic** (header retained harmlessly, `depth` variable removed).
3. **Retained the existing hardening** already present against the Flawfinder CWE-119/CWE-120 findings: `fgets` with `sizeof(buffer)` bounds the read, `strnlen` bounds all length computations to `MAX_INPUT_LEN`, the buffer is explicitly NUL-terminated after every read, `calloc` failure is checked, the `j < len` guard prevents any write past the allocated result, and the buffer is zeroed between iterations.
