#include <stdio.h>
#include <stdlib.h>

int find_max_triangle_sum(int **triangle, int n) {
    if (triangle == NULL || n <= 0) {
        return 0;
    }
    for (int i = 0; i < n; i++) {
        if (triangle[i] == NULL) {
            return 0;
        }
    }
    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }
    dp[0] = triangle[0][0];
    for (int i = 1; i < n; i++) {
        dp[i] = dp[i - 1] + triangle[i][i];
        for (int j = i - 1; j > 0; j--) {
            dp[j] = (dp[j - 1] > dp[j] ? dp[j - 1] : dp[j]) + triangle[i][j];
        }
        dp[0] = dp[0] + triangle[i][0];
    }
    int max_sum = dp[0];
    for (int i = 1; i < n; i++) {
        if (dp[i] > max_sum) {
            max_sum = dp[i];
        }
    }
    free(dp);
    return max_sum;
}

int main(void) {
    int n = 4;
    int r0[] = {3};
    int r1[] = {4, 5};
    int r2[] = {6, 7, 8};
    int r3[] = {9, 10, 11, 12};
    int *triangle[] = {r0, r1, r2, r3};
    int result = find_max_triangle_sum(triangle, n);
    printf("%d\n", result);
    return 0;
}
