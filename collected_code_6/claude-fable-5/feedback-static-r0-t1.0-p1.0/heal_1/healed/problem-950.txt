```c
#include <stdio.h>
#include <stdlib.h>

#define ZODIAC_COUNT 12
#define MIN_YEAR 1
#define MAX_YEAR 9999

static const char *get_chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return NULL;
    }
    return signs[index];
}

static int display_zodiac(int year)
{
    const char *sign;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        fprintf(stderr, "Error: year must be between %d and %d.\n",
                MIN_YEAR, MAX_YEAR);
        return -1;
    }
    sign = get_chinese_zodiac(year);
    if (sign == NULL) {
        fprintf(stderr, "Error: could not determine zodiac sign.\n");
        return -1;
    }
    if (printf("The Chinese zodiac sign for year %d is: %s\n", year, sign) < 0) {
        return -1;
    }
    return 0;
}

int main(void)
{
    int year;
    int result;

    printf("Enter a year: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }
    result = scanf("%d", &year);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input, expected an integer year.\n");
        return EXIT_FAILURE;
    }
    if (display_zodiac(year) != 0) {
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized array `signs[12]` as a potential buffer overflow risk. While the original code already normalized negative modulo results, the rewritten version hardens it further:

1. **Explicit bounds check** — after computing `index`, it is verified to be within `[0, ZODIAC_COUNT)` before indexing the array, guaranteeing no out-of-bounds read even if the logic above were later modified incorrectly.

2. **Named constants** — `ZODIAC_COUNT`, `MIN_YEAR`, and `MAX_YEAR` replace magic numbers, so the array size and its index arithmetic can never silently drift apart.

3. **`const`-correctness** — the array is declared `static const char *const signs[]`, preventing accidental modification of both the pointers and the strings.

4. **NULL propagation** — `get_chinese_zodiac` can now signal failure, and `display_zodiac` checks for it rather than passing a potentially invalid pointer to `printf` (`%s` with NULL is undefined behavior).

The rest of the program was already sound: `scanf` return value is checked, `fflush` and `printf` failures are handled, and the year range is validated before use.
