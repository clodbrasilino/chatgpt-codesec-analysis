```c
#include <stdio.h>
#include <stdlib.h>

int maxProfit(int k, const int *prices, int n)
{
    int *prev;
    int *curr;
    int *tmp;
    int t;
    int d;
    int maxDiff;
    int result;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (d = 1; d < n; d++) {
            if (prices[d] > prices[d - 1]) {
                result += prices[d] - prices[d - 1];
            }
        }
        return result;
    }

    prev = calloc((size_t)n, sizeof(int));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc((size_t)n, sizeof(int));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (t = 1; t <= k; t++) {
        maxDiff = prev[0] - prices[0];
        curr[0] = 0;
        for (d = 1; d < n; d++) {
            curr[d] = curr[d - 1];
            if (prices[d] + maxDiff > curr[d]) {
                curr[d] = prices[d] + maxDiff;
            }
            if (prev[d] - prices[d] > maxDiff) {
                maxDiff = prev[d] - prices[d];
            }
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    result = prev[n - 1];

    free(prev);
    free(curr);

    return result;
}

int main(void)
{
    int prices1[] = {10, 22, 5, 75, 65, 80};
    int prices2[] = {3, 2, 6, 5, 0, 3};
    int prices3[] = {2, 4, 1};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses hinted at in the comments (returning `0` or partial results) typically arise in this algorithm from three classic mistakes, all of which are addressed in the rewritten version:

1. **Reading the result from the wrong buffer** — because `prev` and `curr` are swapped at the end of every transaction iteration, the final answer lives in `prev[n - 1]`, not `curr[n - 1]`. Reading `curr` after an even number of swaps yields stale data (often `0`), which explains failures like *"expected 7, got 0"*.

2. **Incorrect `maxDiff` initialization** — `maxDiff` must be seeded as `prev[0] - prices[0]` at the start of each transaction pass (equivalent to `-prices[0]` since `prev[0]` is always `0`, but written explicitly for correctness and clarity). Initializing it to `0` or resetting it incorrectly truncates profits, producing undervalued results like *"expected 10, got 5"*.

3. **Missing the unlimited-transaction shortcut** — when `k >= n / 2`, the transaction cap is effectively unlimited, and the greedy sum of all positive deltas is both correct and avoids unnecessary allocation. Without it, small arrays like `{2, 4, 1}` can fall through DP edge cases (*"expected 2, got 0"*).

The rewrite also preserves the defensive qualities of the original: `NULL`/bounds guards, checked `calloc` calls with cleanup of the first buffer if the second allocation fails, and no memory leaks on any path.
