#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

void get_n_largest(int *data, int size, int n, int *result) {
    if (n > size) n = size;
    qsort(data, size, sizeof(int), compare);
    for (int i = 0; i < n; ++i) {
        result[i] = data[size - 1 - i];
    }
}

int main() {
    int data[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5};
    int size = sizeof(data) / sizeof(data[0]);
    int n = 5;
    int *result = (int *)malloc(n * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    get_n_largest(data, size, n, result);
    for (int i = 0; i < n; ++i) {
        printf("%d ", result[i]);
    }
    free(result);
    return 0;
}
