#include <stdio.h>
#include <stdlib.h>

int *get_first_elements(int **sublists, const int *sizes, int count, int *out_count)
{
    int *result;
    int valid_count;
    int i;

    if (sublists == NULL || sizes == NULL || out_count == NULL || count <= 0)
    {
        return NULL;
    }

    result = (int *)malloc((size_t)count * sizeof(int));
    if (result == NULL)
    {
        return NULL;
    }

    valid_count = 0;
    for (i = 0; i < count; i++)
    {
        if (sublists[i] != NULL && sizes[i] > 0)
        {
            result[valid_count] = sublists[i][0];
            valid_count++;
        }
    }

    *out_count = valid_count;
    return result;
}

int main(void)
{
    int a[] = {1, 2, 3};
    int b[] = {4, 5};
    int c[] = {6, 7, 8, 9};
    int *sublists[] = {a, b, c};
    int sizes[] = {3, 2, 4};
    int count = 3;
    int out_count = 0;
    int *firsts;
    int i;

    firsts = get_first_elements(sublists, sizes, count, &out_count);
    if (firsts == NULL)
    {
        fprintf(stderr, "Error: failed to retrieve first elements\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < out_count; i++)
    {
        printf("%d\n", firsts[i]);
    }

    free(firsts);
    firsts = NULL;

    return EXIT_SUCCESS;
}
