```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096U

char *abbreviate_road(const char *input);

char *abbreviate_road(const char *input)
{
    size_t len = 0;
    size_t i = 0;
    size_t j = 0;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    while (len < MAX_INPUT_LEN && input[len] != '\0') {
        len++;
    }

    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1U);
    if (result == NULL) {
        return NULL;
    }

    while (i < len) {
        if ((len - i) >= 4U && strncmp(&input[i], "road", 4U) == 0) {
            result[j] = 'r';
            j++;
            result[j] = 'd';
            j++;
            result[j] = '.';
            j++;
            i += 4U;
        } else {
            result[j] = input[i];
            j++;
            i++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *text = "Take the main road until the road ends.";
    char *abbreviated;

    abbreviated = abbreviate_road(text);
    if (abbreviated == NULL) {
        (void)fprintf(stderr, "Error: failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original:    %s\n", text) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    if (printf("Abbreviated: %s\n", abbreviated) < 0) {
        free(abbreviated);
        return EXIT_FAILURE;
    }

    free(abbreviated);
    abbreviated = NULL;

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weakness:

1. **Bounded length scan instead of `strlen`** — The flagged `strlen` call (CWE-126) was replaced with a manual, bounded scan capped at `MAX_INPUT_LEN`. If the input is not NUL-terminated, the scan cannot over-read past the defined limit; it fails gracefully by returning `NULL` instead of crashing.

2. **Guarded `strncmp` comparison** — Added a `(len - i) >= 4U` check before calling `strncmp`, guaranteeing the comparison never inspects bytes beyond the validated length, even in theory. While `strncmp` stops at a NUL byte, this makes the bounds safety explicit and independent of termination.

3. **Preserved correct allocation sizing** — Since `"road"` (4 chars) is replaced by `"rd."` (3 chars), the output can never exceed the input length, so `malloc(len + 1U)` remains provably sufficient and no overflow of `result` is possible.

The rest of the original program's defensive structure (NULL checks, `printf` return-value checks, freeing and nulling the pointer) was already sound and has been retained.
