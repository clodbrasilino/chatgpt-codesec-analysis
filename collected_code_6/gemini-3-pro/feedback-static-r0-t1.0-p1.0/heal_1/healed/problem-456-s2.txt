#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void reverse_string(char *str, size_t max_len) {
    if (str == NULL) {
        return;
    }
    
    size_t len = strnlen(str, max_len);
    if (len <= 1) {
        return;
    }
    
    size_t i = 0;
    size_t j = len - 1;
    
    while (i < j) {
        char temp = str[i];
        str[i] = str[j];
        str[j] = temp;
        i++;
        j--;
    }
}

void reverse_strings_in_list(char **string_list, size_t list_size, size_t max_len) {
    if (string_list == NULL) {
        return;
    }
    
    for (size_t i = 0; i < list_size; i++) {
        reverse_string(string_list[i], max_len);
    }
}

int main(void) {
    size_t num_strings = 3;

    }
    
    string_list[0] = strdup("Hello");

    string_list[2] = strdup("C Programming");

}
