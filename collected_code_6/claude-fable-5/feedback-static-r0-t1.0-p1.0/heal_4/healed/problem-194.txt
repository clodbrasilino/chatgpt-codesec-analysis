```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** in `discard_remaining_input()`: This is a false positive. The loop only reads and discards characters — nothing is stored into a buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, preventing an infinite loop if the stream closes.

2. **Flawfinder `char input[INPUT_SIZE]` warning (CWE-119/CWE-120)**: The statically-sized buffer is safely handled:
   - `fgets(input, sizeof(input), stdin)` limits reads to the buffer size, never overflowing.
   - The buffer is explicitly NUL-terminated defensively (`input[sizeof(input) - 1] = '\0'`).
   - `strnlen` is used instead of `strlen`, bounding the length scan.
   - Overlong input is detected (no trailing newline with a full buffer), the remainder of the line is drained via `discard_remaining_input()`, and the input is rejected — preventing truncated data from being silently accepted.

3. **Integer overflow protection**: `octal_to_decimal()` checks `value > (LONG_MAX - digit) / 8` before each multiply-accumulate, correctly rejecting values that would overflow `long` without invoking undefined behavior.

4. **Input validation**: Every character is validated to be in `'0'`–`'7'`, empty strings are rejected, and NULL pointers / zero lengths are guarded.

The code as given already mitigates both Flawfinder findings; the rewrite above is identical in behavior with the annotation comments removed.
