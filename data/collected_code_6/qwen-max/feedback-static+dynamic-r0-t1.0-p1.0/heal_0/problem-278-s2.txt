```c
#include <stdio.h>
#include <stdlib.h>

int count_before(const int *tuple, size_t tuple_size, int record) {
    if (tuple == NULL || tuple_size == 0) {
        return -1;
    }
    for (size_t i = 0; i < tuple_size; ++i) {
        if (tuple[i] == record) {
            return (int)i;
        }
    }
    return -1;
}

int main() {
    int tuple[] = {2, 3, 4, 5, 2, 1};
    size_t tuple_size = sizeof(tuple) / sizeof(tuple[0]);
    int record = 2;

    int result = count_before(tuple, tuple_size, record);
    if (result != -1) {
        printf("Element count before the record: %d\n", result);
    } else {
        printf("Record not found in the tuple.\n");
    }

    return 0;
}
```
