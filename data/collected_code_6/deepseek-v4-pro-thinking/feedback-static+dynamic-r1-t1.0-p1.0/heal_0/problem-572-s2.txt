#include <stdio.h>
#include <stdlib.h>

int remove_duplicate_numbers(int *arr, int n);
int remove_duplicate_numbers_from_lists(int **lists, int *sizes, int list_count);

int remove_duplicate_numbers(int *arr, int n) {
    if (n < 0) {
        return -1;
    }
    if (n == 0) {
        return 0;
    }
    if (arr == NULL) {
        return -1;
    }

    if ((size_t)n > (size_t)-1 / sizeof(int)) {
        return -1;
    }

    int *temp = malloc((size_t)n * sizeof(int));
    if (temp == NULL) {
        return -1;
    }

    int new_size = 0;
    for (int i = 0; i < n; ++i) {
        int count = 0;
        for (int j = 0; j < n; ++j) {
            if (arr[j] == arr[i]) {
                ++count;
            }
        }
        if (count == 1) {
            temp[new_size] = arr[i];
            ++new_size;
        }
    }

    for (int i = 0; i < new_size; ++i) {
        arr[i] = temp[i];
    }

    free(temp);
    return new_size;
}

int remove_duplicate_numbers_from_lists(int **lists, int *sizes, int list_count) {
    if (lists == NULL || sizes == NULL || list_count <= 0) {
        return -1;
    }

    for (int i = 0; i < list_count; ++i) {
        if (sizes[i] < 0) {
            return -1;
        }
        if (sizes[i] > 0 && lists[i] == NULL) {
            return -1;
        }
        if (sizes[i] == 0) {
            continue;
        }

        int new_size = remove_duplicate_numbers(lists[i], sizes[i]);
        if (new_size < 0) {
            return -1;
        }
        sizes[i] = new_size;
    }

    return 0;
}

int main(void) {
    int list1[] = {1, 2, 2, 3, 3, 3, 4};
    int list2[] = {5, 5, 6, 7, 7, 8, 8};
    int list3[] = {9, 10, 10, 11};

    int sizes[] = {7, 7, 4};
    int *lists[] = {list1, list2, list3};
    int list_count = 3;

    if (remove_duplicate_numbers_from_lists(lists, sizes, list_count) != 0) {
        fprintf(stderr, "Error removing duplicate numbers\n");
        return EXIT_FAILURE;
    }

    for (int i = 0; i < list_count; ++i) {
        printf("List %d: ", i + 1);
        for (int j = 0; j < sizes[i]; ++j) {
            printf("%d ", lists[i][j]);
        }
        printf("\n");
    }

    return EXIT_SUCCESS;
}
