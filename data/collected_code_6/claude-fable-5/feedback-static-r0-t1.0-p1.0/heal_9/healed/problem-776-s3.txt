```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
```

**Analysis notes:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120), and the rewrite preserves those mitigations:

1. **Bounded input reading** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits the read to the buffer size, preventing overflow. The return value is checked for `NULL` to handle EOF/read errors.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure (technically redundant since `fgets` always null-terminates, but harmless and explicit).

3. **Newline stripping via `strcspn`** — a safe idiom that works correctly even if no newline is present (it writes over the existing terminator).

4. **Bounded string length** — `strnlen(str, max_len)` instead of `strlen` ensures the counting function never scans past the caller-declared buffer size, even if passed a non-terminated buffer.

5. **Safe `ctype` usage** — casting to `(unsigned char)` before calling `tolower` avoids undefined behavior for negative `char` values.

6. **NULL and empty-string guards** in `count_chars_with_vowel_neighbors` make the function safe for defensive reuse.

The Flawfinder hit on `char buffer[MAX_INPUT_LEN]` is a low-risk (risk 2) informational flag on any statically sized array; all accesses to it are properly bounds-checked, so no functional changes were required.
