#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STR_LEN 4096

typedef struct {
    char **elements;
    size_t size;
} Tuple;

void free_tuple(Tuple *t) {
    if (t) {
        if (t->elements) {
            for (size_t i = 0; i < t->size; i++) {
                free(t->elements[i]);
            }
            free(t->elements);
        }
        free(t);
    }
}

Tuple* concatenate_adjacent(const Tuple *input) {
    if (!input || input->size == 0) {
        return NULL;
    }

    Tuple *result = malloc(sizeof(Tuple));
    if (!result) {
        return NULL;
    }

    if (input->size == 1) {
        result->size = 1;
        result->elements = malloc(sizeof(char*));
        if (!result->elements) {
            free(result);
            return NULL;
        }
        
        size_t len = strnlen(input->elements[0], MAX_STR_LEN);
        result->elements[0] = malloc(len + 1);
        if (!result->elements[0]) {
            free(result->elements);
            free(result);
            return NULL;
        }
        
        memcpy(result->elements[0], input->elements[0], len);
        result->elements[0][len] = '\0';
        return result;
    }

    result->size = input->size - 1;
    result->elements = malloc(result->size * sizeof(char*));
    if (!result->elements) {
        free(result);
        return NULL;
    }

    for (size_t i = 0; i < result->size; i++) {
        size_t len1 = strnlen(input->elements[i], MAX_STR_LEN);
        size_t len2 = strnlen(input->elements[i + 1], MAX_STR_LEN);
        size_t total_len = len1 + len2 + 1;
        
        result->elements[i] = malloc(total_len);
        if (!result->elements[i]) {
            for (size_t j = 0; j < i; j++) {
                free(result->elements[j]);
            }
            free(result->elements);
            free(result);
            return NULL;
        }
        
        snprintf(result->elements[i], total_len, "%s%s", input->elements[i], input->elements[i + 1]);
    }

    return result;
}

int main(void) {
    Tuple *input = malloc(sizeof(Tuple));
    if (!input) {
        return 1;
    }

    input->size = 4;
    input->elements = malloc(input->size * sizeof(char*));
    if (!input->elements) {
        free(input);
        return 1;
    }

    input->elements[0] = strdup("A");
    input->elements[1] = strdup("B");
    input->elements[2] = strdup("C");
    input->elements[3] = strdup("D");

    for (size_t i = 0; i < input->size; i++) {
        if (!input->elements[i]) {
            free_tuple(input);
            return 1;
        }
    }

    Tuple *result = concatenate_adjacent(input);
    if (result) {
        for (size_t i = 0; i < result->size; i++) {
            printf("%s\n", result->elements[i]);
        }
        free_tuple(result);
    }

    free_tuple(input);
    return 0;
}
