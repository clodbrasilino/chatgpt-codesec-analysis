#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>
#include <limits.h>

int *modulo_division(const int *list1, size_t len1, const int *list2, size_t len2, size_t *result_len)
{
    size_t i;
    int *result;

    if (list1 == NULL || list2 == NULL || result_len == NULL) {
        if (result_len != NULL) {
            *result_len = 0;
        }
        return NULL;
    }

    if (len1 != len2 || len1 == 0) {
        *result_len = 0;
        return NULL;
    }

    for (i = 0; i < len2; i++) {
        if (list2[i] == 0 || (list2[i] == -1 && list1[i] == INT_MIN)) {
            *result_len = 0;
            return NULL;
        }
    }

    if (len1 > (size_t)-1 / sizeof(int)) {
        *result_len = 0;
        return NULL;
    }

    result = malloc(len1 * sizeof(int));
    if (result == NULL) {
        *result_len = 0;
        return NULL;
    }

    for (i = 0; i < len1; i++) {
        result[i] = list1[i] % list2[i];
    }

    *result_len = len1;
    return result;
}

int main(void)
{
    int list1[] = {10, 20, 30, 40, 50};
    int list2[] = {3, 7, 4, 9, 6};
    size_t len1 = sizeof(list1) / sizeof(list1[0]);
    size_t len2 = sizeof(list2) / sizeof(list2[0]);
    size_t result_len = 0;
    size_t i;
    int *result;

    result = modulo_division(list1, len1, list2, len2, &result_len);

    if (result == NULL) {
        return 1;
    }

    for (i = 0; i < result_len; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);

    return 0;
}
