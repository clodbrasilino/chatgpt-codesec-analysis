```c
#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

double negative_ratio(const int *arr, size_t len) {
    if (arr == NULL || len == 0) {
        return 0.0;
    }

    size_t neg_count = 0;
    for (size_t i = 0; i < len; ++i) {
        if (arr[i] < 0) {
            ++neg_count;
        }
    }

    return (double)neg_count / (double)len;
}

int main(void) {
    int arr1[] = {-1, 2, -3, 4, -5};
    size_t len1 = sizeof(arr1) / sizeof(arr1[0]);

    int arr2[] = {1, 2, 3};
    size_t len2 = sizeof(arr2) / sizeof(arr2[0]);

    int arr3[] = {-1, -2, -3, -4};
    size_t len3 = sizeof(arr3) / sizeof(arr3[0]);

    printf("Ratio 1: %.2f\n", negative_ratio(arr1, len1));
    printf("Ratio 2: %.2f\n", negative_ratio(arr2, len2));
    printf("Ratio 3: %.2f\n", negative_ratio(arr3, len3));
    printf("Ratio 4: %.2f\n", negative_ratio(NULL, 0));

    return 0;
}
```
