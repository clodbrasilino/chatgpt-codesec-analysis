#include <stdio.h>

int count_rotations(const int arr[], int n) {
    if (arr == NULL || n <= 0) {
        return -1;
    }
    int low = 0;
    int high = n - 1;
    while (low < high) {
        int mid = low + (high - low) / 2;
        if (arr[mid] < arr[high]) {
            high = mid;
        } else if (arr[mid] > arr[high]) {
            low = mid + 1;
        } else {
            if (high > 0 && arr[high] < arr[high - 1]) {
                return high;
            }
            high--;
        }
    }
    return low;
}

int main() {
    int arr1[] = {5, 6, 1, 2, 3, 4};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int res1 = count_rotations(arr1, n1);
    printf("%d\n", res1);

    int arr2[] = {1, 2, 3, 4, 5};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int res2 = count_rotations(arr2, n2);
    printf("%d\n", res2);

    int arr3[] = {3, 4, 5, 1, 2};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    int res3 = count_rotations(arr3, n3);
    printf("%d\n", res3);

    int res4 = count_rotations(NULL, 0);
    printf("%d\n", res4);

    return 0;
}
