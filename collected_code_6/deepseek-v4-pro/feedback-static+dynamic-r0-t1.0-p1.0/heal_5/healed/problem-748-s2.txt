char *add_spaces_before_capitals(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t input_len = 0;
    while (input[input_len] != '\0') {
        if (input_len >= SIZE_MAX - 1) {
            return NULL;
        }
        input_len++;
    }

    if (input_len == 0) {
        char *result = (char *)malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    size_t capital_count = 0;
    for (size_t i = 0; i < input_len; i++) {
        if (input[i] >= 'A' && input[i] <= 'Z') {
            if (i > 0) {
                capital_count++;
            }
        }
    }

    size_t result_len = input_len + capital_count + 1;
    if (result_len <= input_len) {
        return NULL;
    }

    char *result = (char *)malloc(result_len);
    if (result == NULL) {
        return NULL;
    }

    size_t dst_pos = 0;
    size_t src_pos = 0;

    while (src_pos < input_len && dst_pos < result_len) {
        if (input[src_pos] >= 'A' && input[src_pos] <= 'Z' && src_pos > 0) {
            if (dst_pos + 1 >= result_len) {
                free(result);
                return NULL;
            }
            result[dst_pos++] = ' ';
        }

        if (dst_pos >= result_len) {
            free(result);
            return NULL;
        }
        result[dst_pos++] = input[src_pos++];
    }

    if (dst_pos >= result_len) {
        free(result);
        return NULL;
    }
    result[dst_pos] = '\0';

    return result;
}
