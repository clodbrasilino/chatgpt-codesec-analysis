```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 64
#define MAX_ROMAN_VALUE 3999
#define DISCARD_GUARD_LIMIT 65536UL

static int roman_char_value(char c)
{
    switch (toupper((unsigned char)c)) {
    case 'I':
        return 1;
    case 'V':
        return 5;
    case 'X':
        return 10;
    case 'L':
        return 50;
    case 'C':
        return 100;
    case 'D':
        return 500;
    case 'M':
        return 1000;
    default:
        return -1;
    }
}

static int roman_to_int(const char *roman, size_t max_len, int *result)
{
    size_t len;
    size_t i;
    int total;

    if (roman == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(roman, max_len);
    if (len == 0U || len >= max_len) {
        return -1;
    }

    total = 0;
    for (i = 0; i < len; i++) {
        int current = roman_char_value(roman[i]);
        int next = 0;

        if (current < 0) {
            return -1;
        }

        if ((i + 1U) < len) {
            next = roman_char_value(roman[i + 1U]);
            if (next < 0) {
                return -1;
            }
        }

        if (current < next) {
            total += next - current;
            i++;
        } else {
            total += current;
        }

        if (total < 0 || total > MAX_ROMAN_VALUE) {
            return -1;
        }
    }

    *result = total;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;
    unsigned long guard = 0UL;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < DISCARD_GUARD_LIMIT);
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int value;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a Roman numeral: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
        len--;
    } else if (len == (sizeof(buffer) - 1U)) {
        discard_remaining_input();
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (len == 0U) {
        fprintf(stderr, "Empty input\n");
        return EXIT_FAILURE;
    }

    if (roman_to_int(buffer, sizeof(buffer), &value) != 0) {
        fprintf(stderr, "Invalid Roman numeral: %s\n", buffer);
        return EXIT_FAILURE;
    }

    printf("%s = %d\n", buffer, value);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — The `discard_remaining_input()` function does not write into any buffer; it merely discards characters. The loop is bounded by `DISCARD_GUARD_LIMIT` and terminates on `\n` or `EOF`, so there is no unbounded read or overflow risk. No fix required.

2. **Statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — The buffer is used safely:
   - It is zero-initialized with `memset`.
   - Input is read with `fgets(buffer, sizeof(buffer), stdin)`, which enforces the bound.
   - Explicit NUL termination is applied at `buffer[sizeof(buffer) - 1]`.
   - Length checks use `strnlen` with `sizeof(buffer)` as the cap, never `strlen`.
   - Overlong input is detected (no trailing newline with a full buffer), the remainder is drained, and the program exits with an error rather than silently truncating.

3. **Additional robustness already present** — `roman_to_int` validates all pointers, rejects empty/overlong strings, checks every character, and guards against integer overflow by rejecting totals outside `[0, 3999]`. The `toupper` call correctly casts to `unsigned char` to avoid undefined behavior with negative `char` values.

The Flawfinder hits are informational (risk levels 1–2) and both are false positives given the existing bounds checking, so the rewritten code preserves the original logic with the comments removed.
