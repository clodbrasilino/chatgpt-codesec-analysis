```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *extract_quoted(const char *input)
{
    const char *start;
    const char *end;
    char *result;
    size_t length;

    if (input == NULL) {
        return NULL;
    }

    start = strchr(input, '"');
    if (start == NULL) {
        return NULL;
    }
    start++;

    end = strchr(start, '"');
    if (end == NULL) {
        return NULL;
    }

    length = (size_t)(end - start);
    result = malloc(length + 1U);
    if (result == NULL) {
        return NULL;
    }

    memcpy(result, start, length);
    result[length] = '\0';

    return result;
}

int main(void)
{
    const char *test = "Hello \"world of C\" example";
    char *extracted;

    extracted = extract_quoted(test);
    if (extracted == NULL) {
        if (fprintf(stderr, "No quoted value found or allocation failed\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Extracted: %s\n", extracted) < 0) {
        free(extracted);
        return EXIT_FAILURE;
    }

    free(extracted);
    return EXIT_SUCCESS;
}
```

**Key safety measures in this implementation:**

- **CWE-664 avoidance:** The heap-allocated buffer returned by `extract_quoted` is freed exactly once on every exit path in `main`, including the error path after a failed `printf`, so the resource lifetime is fully controlled with no leaks or double-frees.

- **CWE-703 avoidance:** Every fallible operation is checked — `NULL` input, missing opening quote, missing closing quote, `malloc` failure, and even the return values of `printf`/`fprintf` are validated.

- **CWE-710 avoidance:** The code adheres to standard conventions: `main` returns `int` with `EXIT_SUCCESS`/`EXIT_FAILURE`, uses `size_t` for lengths, explicit casts where narrowing occurs, `const` correctness on read-only parameters, and `memcpy` with an explicit null terminator instead of unsafe string functions like `strcpy`.
