```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

typedef struct {
    char *first;
    char *second;
} StringTuple;

static char *concat_strings(const char *a, const char *b)
{
    size_t len_a;
    size_t len_b;
    size_t total_len;
    char *result;

    if (a == NULL || b == NULL) {
        return NULL;
    }

    len_a = strnlen(a, MAX_STRING_LEN);
    len_b = strnlen(b, MAX_STRING_LEN);

    if (len_a >= MAX_STRING_LEN || len_b >= MAX_STRING_LEN) {
        return NULL;
    }

    if (len_b > SIZE_MAX - 1U || len_a > SIZE_MAX - len_b - 1U) {
        return NULL;
    }

    total_len = len_a + len_b + 1U;

    result = (char *)malloc(total_len);
    if (result == NULL) {
        return NULL;
    }

    if (len_a > 0U && len_a < total_len) {
        memcpy(result, a, len_a);
    }

    if (len_b > 0U && len_b <= total_len - len_a - 1U) {
        memcpy(result + len_a, b, len_b);
    }

    result[total_len - 1U] = '\0';

    return result;
}

static int concat_tuples(const StringTuple *t1, const StringTuple *t2, StringTuple *out)
{
    if (t1 == NULL || t2 == NULL || out == NULL) {
        return -1;
    }

    out->first = concat_strings(t1->first, t2->first);
    if (out->first == NULL) {
        return -1;
    }

    out->second = concat_strings(t1->second, t2->second);
    if (out->second == NULL) {
        free(out->first);
        out->first = NULL;
        return -1;
    }

    return 0;
}

static void free_tuple(StringTuple *t)
{
    if (t != NULL) {
        free(t->first);
        free(t->second);
        t->first = NULL;
        t->second = NULL;
    }
}

int main(void)
{
    StringTuple t1;
    StringTuple t2;
    StringTuple result;

    t1.first = "Hello, ";
    t1.second = "Good ";
    t2.first = "World";
    t2.second = "Morning";

    result.first = NULL;
    result.second = NULL;

    if (concat_tuples(&t1, &t2, &result) != 0) {
        fprintf(stderr, "Tuple concatenation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("(%s, %s)\n", result.first, result.second) < 0) {
        free_tuple(&result);
        return EXIT_FAILURE;
    }

    free_tuple(&result);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

Both Flawfinder findings (CWE-120, `memcpy` buffer overflow risk) are **false positives** in this code — the original already mitigates them properly, so no functional changes were required:

1. **First `memcpy(result, a, len_a)`** — The destination buffer `result` is allocated with exactly `total_len = len_a + len_b + 1` bytes, and the guard `len_a < total_len` guarantees the copy of `len_a` bytes fits.

2. **Second `memcpy(result + len_a, b, len_b)`** — The guard `len_b <= total_len - len_a - 1` ensures the copy ends before the final byte reserved for the NUL terminator.

**Additional defensive measures already present:**

- `strnlen` with `MAX_STRING_LEN` bounds prevents unbounded reads of non-terminated inputs.
- The `SIZE_MAX` checks prevent integer overflow in `len_a + len_b + 1` (technically redundant given the `MAX_STRING_LEN` cap, but harmless defense-in-depth).
- `malloc` return is checked; NUL termination is explicit.
- `concat_tuples` cleans up `out->first` on partial failure, avoiding a memory leak.
- `free_tuple` nulls pointers after freeing, preventing double-free/use-after-free.

The code is memory-safe as written; the rewrite simply removes the advisory comments per your request.
