```c
#include <stdio.h>
#include <stdbool.h>

bool is_prime(int num) {
    if (num <= 1) {
        return false;
    }
    for (int i = 2; i * i <= num; i++) {
        if (num % i == 0) {
            return false;
        }
    }
    return true;
}

bool is_non_prime(int num) {
    return !is_prime(num);
}

int main(void) {
    int num;
    printf("Enter an integer: ");
    if (scanf("%d", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    if (is_non_prime(num)) {
        printf("%d is a non-prime number.\n", num);
    } else {
        printf("%d is a prime number.\n", num);
    }

    return 0;
}
```
