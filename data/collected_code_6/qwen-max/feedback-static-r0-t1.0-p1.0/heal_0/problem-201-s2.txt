#include <stdio.h>
#include <stdbool.h>

bool areAllElementsEqual(int *list, int length) {
    if (length <= 1) {
        return true;
    }
    for (int i = 1; i < length; ++i) {
        if (list[i] != list[0]) {
            return false;
        }
    }
    return true;
}

int main() {
    int list1[] = {1, 1, 1, 1};
    int list2[] = {1, 2, 1, 1};
    int length1 = sizeof(list1) / sizeof(list1[0]);
    int length2 = sizeof(list2) / sizeof(list2[0]);

    bool result1 = areAllElementsEqual(list1, length1);
    bool result2 = areAllElementsEqual(list2, length2);

    printf("List 1 all elements equal: %s\n", result1 ? "true" : "false");
    printf("List 2 all elements equal: %s\n", result2 ? "true" : "false");

    return 0;
}
