#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct KeyValuePair {
    char *key;
    char *value;
    struct KeyValuePair *next;
} KeyValuePair;

typedef struct Dictionary {
    KeyValuePair *head;
} Dictionary;

Dictionary* create_dictionary() {
    Dictionary *dict = (Dictionary*)malloc(sizeof(Dictionary));
    if (dict) {
        dict->head = NULL;
    }
    return dict;
}

void add_to_dictionary(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) return;

    KeyValuePair *current = dict->head;
    while (current) {
        if (strcmp(current->key, key) == 0) {
            char *new_value = strdup(value);
            if (new_value) {
                free(current->value);
                current->value = new_value;
            }
            return;
        }
        current = current->next;
    }

    KeyValuePair *new_pair = (KeyValuePair*)malloc(sizeof(KeyValuePair));
    if (!new_pair) return;

    new_pair->key = strdup(key);
    new_pair->value = strdup(value);
    
    if (!new_pair->key || !new_pair->value) {
        free(new_pair->key);
        free(new_pair->value);
        free(new_pair);
        return;
    }

    new_pair->next = dict->head;
    dict->head = new_pair;
}

Dictionary* merge_dictionaries(const Dictionary *dict1, const Dictionary *dict2) {
    if (!dict1 && !dict2) return NULL;

    Dictionary *merged = create_dictionary();
    if (!merged) return NULL;

    if (dict1) {
        KeyValuePair *current = dict1->head;
        while (current) {
            add_to_dictionary(merged, current->key, current->value);
            current = current->next;
        }
    }

    if (dict2) {
        KeyValuePair *current = dict2->head;
        while (current) {
            add_to_dictionary(merged, current->key, current->value);
            current = current->next;
        }
    }

    return merged;
}

void free_dictionary(Dictionary *dict) {
    if (!dict) return;

    KeyValuePair *current = dict->head;
    while (current) {
        KeyValuePair *next = current->next;
        free(current->key);
        free(current->value);
        free(current);
        current = next;
    }
    free(dict);
}

void print_dictionary(const Dictionary *dict) {
    if (!dict) return;

    KeyValuePair *current = dict->head;
    while (current) {
        printf("%s: %s\n", current->key, current->value);
        current = current->next;
    }
}

int main() {
    Dictionary *dict1 = create_dictionary();
    if (!dict1) return 1;

    Dictionary *dict2 = create_dictionary();
    if (!dict2) {
        free_dictionary(dict1);
        return 1;
    }

    add_to_dictionary(dict1, "key1", "value1");
    add_to_dictionary(dict1, "key2", "value2");

    add_to_dictionary(dict2, "key2", "new_value2");
    add_to_dictionary(dict2, "key3", "value3");

    Dictionary *merged = merge_dictionaries(dict1, dict2);

    if (merged) {
        print_dictionary(merged);
    }

    free_dictionary(dict1);
    free_dictionary(dict2);
    free_dictionary(merged);

    return 0;
}
