```c
#include <stdio.h>
#include <string.h>

const char *const romanNumerals[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};

char *intToRoman(int num) {
    static char roman[20];
    roman[0] = '\0';

    for (int i = 0; i < 13; i++) {
        while (num >= values[i]) {
            size_t len = strlen(roman);
            if (len + strlen(romanNumerals[i]) + 1 <= sizeof(roman)) {
                strncat(roman, romanNumerals[i], sizeof(roman) - len - 1);
                num -= values[i];
            } else {
                break;
            }
        }
    }

    return roman;
}

int main() {
    int number = 1994;
    printf("%s\n", intToRoman(number));
    return 0;
}
```
