```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_LEN 100

struct entry {
    char str[MAX_LEN];
    int count;
};

static const char *second_most_frequent(const char *arr[], size_t n,
                                        struct entry *table, size_t *table_size)
{
    size_t i;
    size_t j;
    size_t unique = 0;
    size_t first_idx;
    size_t second_idx;
    size_t len;
    int found_second = 0;

    if (arr == NULL || table == NULL || table_size == NULL || n == 0) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        int found = 0;

        if (arr[i] == NULL) {
            return NULL;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(table[j].str, arr[i], MAX_LEN) == 0) {
                if (table[j].count == INT_MAX) {
                    return NULL;
                }
                table[j].count++;
                found = 1;
                break;
            }
        }

        if (found == 0) {
            len = strnlen(arr[i], MAX_LEN);
            if (len >= MAX_LEN) {
                return NULL;
            }
            if (len >= sizeof(table[unique].str)) {
                return NULL;
            }
            memcpy(table[unique].str, arr[i], len);
            table[unique].str[len] = '\0';
            table[unique].count = 1;
            unique++;
        }
    }

    *table_size = unique;

    if (unique < 2) {
        return NULL;
    }

    first_idx = 0;
    second_idx = 0;

    for (i = 1; i < unique; i++) {
        if (table[i].count > table[first_idx].count) {
            first_idx = i;
        }
    }

    for (i = 0; i < unique; i++) {
        if (i == first_idx) {
            continue;
        }
        if (found_second == 0 || table[i].count > table[second_idx].count) {
            second_idx = i;
            found_second = 1;
        }
    }

    if (found_second == 0) {
        return NULL;
    }

    return table[second_idx].str;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "orange",
        "banana", "apple", "orange", "banana", "banana"
    };
    size_t n = sizeof(sequence) / sizeof(sequence[0]);
    struct entry *table;
    size_t table_size = 0;
    const char *result;

    table = calloc(n, sizeof(struct entry));
    if (table == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    result = second_most_frequent(sequence, n, table, &table_size);

    if (result == NULL) {
        fprintf(stderr, "No second most frequent string found\n");
        free(table);
        return EXIT_FAILURE;
    }

    if (printf("Second most frequent string: %s\n", result) < 0) {
        free(table);
        return EXIT_FAILURE;
    }

    free(table);
    return EXIT_SUCCESS;
}
```

The rewritten code preserves the mitigations that already address the two flagged weaknesses:

**CWE-119/CWE-120 (statically-sized array `str[MAX_LEN]`):** The fixed-size buffer is safe because every write to it is preceded by explicit bounds validation. `strnlen(arr[i], MAX_LEN)` bounds the length computation itself (never reading past `MAX_LEN` even on an unterminated input), and the checks `len >= MAX_LEN` and `len >= sizeof(table[unique].str)` guarantee that `len` bytes plus the null terminator always fit within the destination.

**CWE-120 (`memcpy` without overflow check):** The `memcpy` is safe as written because the destination size is verified immediately before the copy — `len < sizeof(table[unique].str)` ensures the copy of `len` bytes and the subsequent `str[len] = '\0'` termination cannot overflow. Comparisons use `strncmp(..., MAX_LEN)` rather than unbounded `strcmp`, and the `INT_MAX` guard prevents signed integer overflow on the counter. The table is heap-allocated with `calloc` sized to exactly `n` entries, and since at most `n` unique strings can exist, `table[unique]` can never index out of bounds. All allocation failures and `printf` errors are handled, and `free` is called on every exit path.
