```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and their mitigations:**

1. **Statically-sized array `first[MAX_NAME_LEN]` (CWE-119/CWE-120):** The risk of overflow is mitigated because every write into the buffer goes through `set_tuple`, which validates the input length with `strnlen(name, MAX_NAME_LEN)` and rejects any string that would not fit including its null terminator (`len >= MAX_NAME_LEN`). An additional explicit guard against `sizeof(t->first)` makes the bounds check robust even if the struct definition changes independently of the macro.

2. **`memcpy` without bounds check (CWE-120):** The copy is safe because `len` is strictly less than `MAX_NAME_LEN` at the point of the call, guaranteeing the destination can hold `len` bytes plus the manually appended `'\0'`. The copy length is derived from the validated `strnlen` result, never from an unbounded `strlen` or user-controlled size.

3. **Defensive practices already present and preserved:** null-pointer checks on all public entry points, bounded string comparison via `strncmp`, bounded output via the `%.*s` precision specifier in `printf`, and checked return values for all fallible calls.

The program compiles cleanly with `-Wall -Wextra -Wpedantic` and outputs the tuples sorted lexicographically by name.
