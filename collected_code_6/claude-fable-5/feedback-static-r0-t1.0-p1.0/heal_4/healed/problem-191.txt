#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 32

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN] = {0};
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    for (i = 0; i < len && i < MAX_INPUT_LEN - 1; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No month name provided.\n");
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and fixes applied:**

The original code was already well-written from a security standpoint — the Flawfinder hits were largely informational rather than actual vulnerabilities. Here is my assessment as an experienced C developer:

1. **`char normalized[MAX_INPUT_LEN]` (CWE-119/CWE-120):** The original already guarded against overflow via `strnlen` and the `len >= MAX_INPUT_LEN` check. In the rewrite I hardened it further by zero-initializing the buffer and adding an explicit `i < MAX_INPUT_LEN - 1` bound in the copy loop, so the null terminator write can never go out of bounds even if the length logic were ever modified incorrectly.

2. **`char input[MAX_INPUT_LEN]` (CWE-119/CWE-120):** Safe because `fgets` is called with `sizeof(input)`, which guarantees null termination and no overflow. I added zero-initialization and a defensive `input[sizeof(input) - 1] = '\0'` as belt-and-suspenders.

3. **`getchar` in a loop (CWE-120/CWE-20):** This is a false positive — the loop discards characters without storing them into any buffer, so no overflow is possible. The `EOF` check correctly prevents an infinite loop on closed stdin. No change was needed.

4. **General correctness retained:** the `(unsigned char)` cast before `tolower` avoids undefined behavior on negative `char` values, `strnlen` bounds all length computations, and leftover stdin data is drained when input is truncated.
