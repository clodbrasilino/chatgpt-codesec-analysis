The key fix is replacing the unbounded `strlen` call (CWE-126 — potential over-read on non-null-terminated input) with a bounded `memchr` scan limited to `MAX_STR_LEN` bytes. If no null terminator is found within that bound, the function returns an error instead of reading past the buffer. Bounding `n` also guarantees the `(n + 1) * (n + 1)` allocation and the `int` result cannot overflow.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int longestRepeatingSubsequence(const char *str)
{
    size_t n;
    size_t i;
    size_t j;
    int *dp;
    int result;
    const char *nul;

    if (str == NULL)
    {
        return -1;
    }

    nul = memchr(str, '\0', MAX_STR_LEN);
    if (nul == NULL)
    {
        return -1;
    }
    n = (size_t)(nul - str);

    if (n == 0)
    {
        return 0;
    }

    if ((n + 1) > SIZE_MAX / (n + 1))
    {
        return -1;
    }

    dp = calloc((n + 1) * (n + 1), sizeof(*dp));
    if (dp == NULL)
    {
        return -1;
    }

    for (i = 1; i <= n; i++)
    {
        for (j = 1; j <= n; j++)
        {
            if (str[i - 1] == str[j - 1] && i != j)
            {
                dp[i * (n + 1) + j] = dp[(i - 1) * (n + 1) + (j - 1)] + 1;
            }
            else
            {
                dp[i * (n + 1) + j] = max_int(dp[(i - 1) * (n + 1) + j],
                                              dp[i * (n + 1) + (j - 1)]);
            }
        }
    }

    result = dp[n * (n + 1) + n];

    free(dp);
    dp = NULL;

    return result;
}

int main(void)
{
    const char *str = "AABEBCDD";
    int result;

    result = longestRepeatingSubsequence(str);

    if (result < 0)
    {
        fprintf(stderr, "Error: failed to compute longest repeating subsequence\n");
        return EXIT_FAILURE;
    }

    printf("Longest repeating subsequence length: %d\n", result);

    return EXIT_SUCCESS;
}
```

Summary of changes:

1. **CWE-126 fix**: `strlen(str)` replaced with `memchr(str, '\0', MAX_STR_LEN)`, so the scan for the terminator is strictly bounded — a non-null-terminated buffer can no longer cause an over-read; the function returns `-1` instead.
2. The pre-existing integer-overflow guard on `(n + 1) * (n + 1)` is retained as defense-in-depth, though the length bound now makes overflow impossible.
3. Allocation failure, `NULL` input, and empty-string cases remain handled, and the allocated buffer is always freed before returning.
