#include <stdlib.h>

typedef struct {
    char **keys;
    char **values;
    size_t size;
    size_t capacity;
} Dict;

typedef struct {
    Dict **items;
    size_t size;
    size_t capacity;
} DictList;

Dict *create_empty_dict(void) {
    Dict *d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }
    d->capacity = 8;
    d->size = 0;
    d->keys = malloc(d->capacity * sizeof(char *));
    if (d->keys == NULL) {
        free(d);
        return NULL;
    }
    d->values = malloc(d->capacity * sizeof(char *));
    if (d->values == NULL) {
        free(d->keys);
        free(d);
        return NULL;
    }
    return d;
}

void free_dict(Dict *d) {
    if (d == NULL) {
        return;
    }
    for (size_t i = 0; i < d->size; i++) {
        free(d->keys[i]);
        free(d->values[i]);
    }
    free(d->keys);
    free(d->values);
    free(d);
}

DictList *create_dict_list(size_t count) {
    DictList *list = malloc(sizeof(DictList));
    if (list == NULL) {
        return NULL;
    }
    list->size = count;
    list->capacity = count < 2 ? 4 : count * 2;
    list->items = malloc(list->capacity * sizeof(Dict *));
    if (list->items == NULL) {
        free(list);
        return NULL;
    }
    for (size_t i = 0; i < count; i++) {
        list->items[i] = create_empty_dict();
        if (list->items[i] == NULL) {
            for (size_t j = 0; j < i; j++) {
                free_dict(list->items[j]);
            }
            free(list->items);
            free(list);
            return NULL;
        }
    }
    return list;
}

void free_dict_list(DictList *list) {
    if (list == NULL) {
        return;
    }
    for (size_t i = 0; i < list->size; i++) {
        free_dict(list->items[i]);
    }
    free(list->items);
    free(list);
}

int main(void) {
    size_t count = 10;
    DictList *list = create_dict_list(count);
    if (list == NULL) {
        return EXIT_FAILURE;
    }
    free_dict_list(list);
    return EXIT_SUCCESS;
}
