```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t capacity;
    size_t num_tokens;
    size_t start;
    size_t i;
    char **tokens;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return NULL;
    }

    capacity = 8;
    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    num_tokens = 0;
    start = 0;

    for (i = 0; i <= len; i++) {
        if (i == len || (isupper((unsigned char)input[i]) && i > start)) {
            size_t token_len = i - start;

            if (token_len > 0) {
                char *token;
                size_t alloc_size;

                if (num_tokens == capacity) {
                    size_t new_capacity;
                    char **resized;

                    if (capacity > SIZE_MAX / 2 ||
                        (capacity * 2) > SIZE_MAX / sizeof(char *)) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    new_capacity = capacity * 2;
                    resized = realloc(tokens, new_capacity * sizeof(char *));
                    if (resized == NULL) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    tokens = resized;
                    capacity = new_capacity;
                }

                if (start >= len || token_len > len - start ||
                    token_len >= SIZE_MAX) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                alloc_size = token_len + 1;

                token = malloc(alloc_size);
                if (token == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                memcpy(token, &input[start], token_len);
                token[token_len] = '\0';
                tokens[num_tokens] = token;
                num_tokens++;
            }
            start = i;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "helloWorldThisIsATest";
    size_t count = 0;
    size_t i;
    char **tokens;

    tokens = split_at_uppercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens (%zu):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Key changes from the original:

1. **Removed the redundant `token_len >= alloc_size` check** — the static analyzer correctly flagged this as always false, since `alloc_size = token_len + 1` was assigned immediately before and the overflow case was already excluded. Dead code like this adds noise and confuses future maintainers.

2. **Consolidated the overflow guard** — the check `token_len > SIZE_MAX - 1` was rewritten as `token_len >= SIZE_MAX` and merged into the bounds validation block. This guarantees `token_len + 1` cannot wrap around, which makes the subsequent `malloc(alloc_size)` provably large enough.

3. **The `memcpy` (CWE-120 / Flawfinder risk 2) is safe by construction** — because `token` is allocated with exactly `token_len + 1` bytes, overflow-checked beforehand, and `token_len` is verified to fit within `len - start` of the source buffer, the copy of `token_len` bytes plus the NUL terminator can never overflow the destination. No code change was needed beyond keeping those guards intact; the Flawfinder hit is a false positive once bounds are enforced.

4. **Retained the genuinely useful defenses** — the `strnlen` cap against unterminated input, the capacity-doubling overflow checks before `realloc`, and the `resized` temporary that avoids leaking `tokens` on `realloc` failure are all correct patterns and were preserved as-is.
