#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 1000
#define MAX_WORD_LEN 100
#define MAX_OUTPUT_LEN (MAX_WORDS * MAX_WORD_LEN)

typedef struct {
    char words[MAX_WORDS][MAX_WORD_LEN];
    int count;
} WordList;

void init_word_list(WordList *list) {
    list->count = 0;
}

int contains_word(WordList *list, const char *word) {
    for (int i = 0; i < list->count; i++) {
        if (strcmp(list->words[i], word) == 0) {
            return 1;
        }
    }
    return 0;
}

void add_word(WordList *list, const char *word) {
    if (list->count < MAX_WORDS) {
        size_t len = strlen(word);
        size_t copy_len = len < MAX_WORD_LEN - 1 ? len : MAX_WORD_LEN - 1;
        memcpy(list->words[list->count], word, copy_len);
        list->words[list->count][copy_len] = '\0';
        list->count++;
    }
}

void remove_duplicate_words(const char *input, char *output) {
    WordList seen;
    init_word_list(&seen);

    size_t input_len = strlen(input);
    char *copy = malloc(input_len + 1);
    if (copy == NULL) {
        output[0] = '\0';
        return;
    }
    memcpy(copy, input, input_len + 1);

    char *result = malloc(MAX_OUTPUT_LEN);
    if (result == NULL) {
        free(copy);
        output[0] = '\0';
        return;
    }
    result[0] = '\0';

    char *token = strtok(copy, " \t\n\r");
    int first = 1;
    size_t result_len = 0;

    while (token != NULL) {
        if (!contains_word(&seen, token)) {
            add_word(&seen, token);
            size_t token_len = strlen(token);
            size_t space_needed = first ? 0 : 1;
            
            if (result_len + space_needed + token_len < MAX_OUTPUT_LEN) {
                if (!first) {
                    result[result_len] = ' ';
                    result_len++;
                }
                memcpy(result + result_len, token, token_len);
                result_len += token_len;
                result[result_len] = '\0';
                first = 0;
            } else {
                break;
            }
        }
        token = strtok(NULL, " \t\n\r");
    }

    size_t output_len = result_len < 2047 ? result_len : 2047;
    memcpy(output, result, output_len);
    output[output_len] = '\0';

    free(result);
    free(copy);
}

int main(void) {
    char input[2048];
    char output[2048];

    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    input[strcspn(input, "\n")] = '\0';

    remove_duplicate_words(input, output);

    printf("Original: %s\n", input);
    printf("After removing duplicates: %s\n", output);

    return 0;
}
