```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static long long merge(int *arr, int *temp, size_t left, size_t mid, size_t right, size_t n)
{
    size_t i = left;
    size_t j = mid;
    size_t k = left;
    long long inv_count = 0;

    if (arr == NULL || temp == NULL || right >= n || left > mid || mid > right || mid == 0) {
        return -1;
    }

    while (i <= mid - 1 && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k] = arr[i];
            k++;
            i++;
        } else {
            temp[k] = arr[j];
            k++;
            j++;
            inv_count += (long long)(mid - i);
        }
    }

    while (i <= mid - 1) {
        temp[k] = arr[i];
        k++;
        i++;
    }

    while (j <= right) {
        temp[k] = arr[j];
        k++;
        j++;
    }

    if (right - left + 1 > n - left) {
        return -1;
    }

    memcpy(&arr[left], &temp[left], (right - left + 1) * sizeof(int));

    return inv_count;
}

static long long merge_sort(int *arr, int *temp, size_t left, size_t right, size_t n)
{
    long long inv_count = 0;

    if (right > left) {
        size_t mid = left + (right - left) / 2;
        long long sub_count;

        sub_count = merge_sort(arr, temp, left, mid, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;

        sub_count = merge_sort(arr, temp, mid + 1, right, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;

        sub_count = merge(arr, temp, left, mid + 1, right, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;
    }

    return inv_count;
}

long long count_inversions(const int *arr, size_t n)
{
    int *copy = NULL;
    int *temp = NULL;
    long long result = -1;

    if (arr == NULL || n == 0) {
        return 0;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy = malloc(n * sizeof(int));
    if (copy == NULL) {
        return -1;
    }

    temp = malloc(n * sizeof(int));
    if (temp == NULL) {
        free(copy);
        return -1;
    }

    memcpy(copy, arr, n * sizeof(int));

    result = merge_sort(copy, temp, 0, n - 1, n);

    free(copy);
    free(temp);

    return result;
}

int main(void)
{
    int arr[] = { 8, 4, 2, 1 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long inversions;

    inversions = count_inversions(arr, n);
    if (inversions < 0) {
        if (fprintf(stderr, "Error: memory allocation failed\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Number of inversions: %lld\n", inversions) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-120 (`memcpy` in `merge`)**: The copy is guarded on multiple levels. The guard `right >= n` ensures indices stay within the allocated buffer, and the explicit check `right - left + 1 > n - left` verifies the destination region `&arr[left]` can hold `(right - left + 1) * sizeof(int)` bytes before the copy executes. Since `temp` and `arr` are both allocated with `n` elements, the source region is equally bounded. This is a safe, checked use of `memcpy`.

2. **CWE-120 (`memcpy` in `count_inversions`)**: The destination `copy` is allocated with exactly `n * sizeof(int)` bytes, matching the copy size precisely. The overflow check `n > SIZE_MAX / sizeof(int)` prevents integer overflow in the allocation size calculation, so the buffer is guaranteed to be large enough.

3. **`variableScope` (cppcheck)**: `sub_count` has been moved from function scope into the `if (right > left)` block in `merge_sort`, the narrowest scope in which it is used, resolving the cppcheck style finding.

4. **Additional hardening**: A `mid == 0` guard was added in `merge` to make the `mid - 1` expression provably safe against `size_t` underflow (in practice `merge` is always called with `mid + 1 >= 1`, but the check makes the invariant explicit and defensive).

The remaining Flawfinder hits are risk-2 informational warnings inherent to any `memcpy` usage; both call sites now have verifiable bounds guarantees, so they can be safely suppressed or annotated in a real audit.
