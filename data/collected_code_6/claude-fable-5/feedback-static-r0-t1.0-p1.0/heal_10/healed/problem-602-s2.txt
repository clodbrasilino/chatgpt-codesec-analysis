#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>

#define BUFFER_SIZE 256

int find_first_repeated(const char *str, size_t max_len, char *result)
{
    int seen[256] = {0};
    size_t i;
    size_t len;

    if (str == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c]) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char repeated;
    int status;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    status = find_first_repeated(buffer, sizeof(buffer), &repeated);
    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    return 0;
}

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning about `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120) is a static-analysis heuristic that flags *any* fixed-size character array, since they are a common source of buffer overflows. In this case, the code is already properly hardened, so no functional changes were required:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits the read to `BUFFER_SIZE - 1` characters plus the null terminator, preventing overflow at the input stage. The return value is also checked for `NULL`, handling EOF and stream errors.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0';` is defensive redundancy (fgets already null-terminates), ensuring the buffer is always a valid C string before any string operations.

3. **Length-bounded string functions** — `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so even a non-terminated buffer could never cause an out-of-bounds read.

4. **Defensive API design** — `find_first_repeated` validates all pointers, rejects a zero `max_len`, and treats `len == max_len` (a possibly unterminated string) as an error. It also casts to `unsigned char` before indexing `seen[]`, avoiding negative-index UB on platforms where `char` is signed.

The warning is therefore a false positive; the rewrite preserves the existing mitigations, which represent correct handling of a fixed-size buffer.
