#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void move_numbers_to_string(const char *input, char *output, size_t output_size) {
    size_t i, j = 0;
    
    if (input == NULL || output == NULL || output_size == 0) {
        return;
    }
    
    for (i = 0; input[i] != '\0' && j < output_size - 1; i++) {
        if (isdigit((unsigned char)input[i])) {
            output[j++] = input[i];
        }
    }
    
    output[j] = '\0';
}

int main(void) {
    const char test_string[] = "abc123def456ghi789";
    size_t result_size;
    char *result = NULL;
    
    result_size = sizeof(test_string);
    result = (char *)malloc(result_size);
    
    if (result == NULL) {
        return 1;
    }
    
    move_numbers_to_string(test_string, result, result_size);
    printf("Original string: %s\n", test_string);
    printf("Numbers extracted: %s\n", result);
    
    free(result);
    
    return 0;
}
