#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int replace_whitespace_underscore(const char *input, size_t input_len, char *output, size_t output_size) {
    regex_t regex;
    regmatch_t match;
    size_t cursor = 0;
    size_t output_len = 0;
    int ret;
    char *temp_output;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    if (input_len >= output_size) {
        return -1;
    }

    temp_output = malloc(input_len + 1);
    if (temp_output == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[ \t\n\r]|_", REG_EXTENDED);
    if (ret != 0) {
        free(temp_output);
        return -1;
    }

    while (cursor < input_len) {
        ret = regexec(&regex, input + cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = input_len - cursor;
            if (output_len + remaining >= output_size) {
                regfree(&regex);
                free(temp_output);
                return -1;
            }
            if (remaining > 0) {
                memcpy(temp_output + output_len, input + cursor, remaining);
                output_len += remaining;
            }
            break;
        } else if (ret != 0) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        if (output_len + (size_t)match.rm_so + 1 >= output_size) {
            regfree(&regex);
            free(temp_output);
            return -1;
        }

        if (match.rm_so > 0) {
            memcpy(temp_output + output_len, input + cursor, (size_t)match.rm_so);
            output_len += (size_t)match.rm_so;
        }

        if (input[cursor + (size_t)match.rm_so] == '_') {
            temp_output[output_len] = ' ';
        } else {
            temp_output[output_len] = '_';
        }
        output_len++;

        cursor += (size_t)match.rm_eo;
    }

    if (output_len + 1 > output_size) {
        regfree(&regex);
        free(temp_output);
        return -1;
    }

    temp_output[output_len] = '\0';
    memcpy(output, temp_output, output_len + 1);

    regfree(&regex);
    free(temp_output);
    return 0;
}

int main(void) {
    char input[256];
    char output[256];
    int result;
    size_t len;

    if (fgets(input, sizeof(input), stdin) == NULL) {
        return 1;
    }

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }

    result = replace_whitespace_underscore(input, len, output, sizeof(output));
    if (result != 0) {
        fprintf(stderr, "Error: replacement failed\n");
        return 1;
    }

    printf("%s\n", output);
    return 0;
}
