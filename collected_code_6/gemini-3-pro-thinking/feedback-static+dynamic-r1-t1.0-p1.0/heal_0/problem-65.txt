#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* create_node(int data);
void free_list(Node* head);
long long recursive_sum(const Node* head);

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

void free_list(Node* head) {
    Node* current = head;
    Node* next_node = NULL;
    
    while (current != NULL) {
        next_node = current->next;
        free(current);
        current = next_node;
    }
}

long long recursive_sum(const Node* head) {
    if (head == NULL) {
        return 0;
    }
    return (long long)head->data + recursive_sum(head->next);
}

int main(void) {
    Node* head = create_node(15);
    if (head == NULL) {
        return EXIT_FAILURE;
    }

    head->next = create_node(25);
    if (head->next == NULL) {
        free_list(head);
        return EXIT_FAILURE;
    }

    head->next->next = create_node(35);
    if (head->next->next == NULL) {
        free_list(head);
        return EXIT_FAILURE;
    }

    long long sum = recursive_sum(head);
    printf("%lld\n", sum);

    free_list(head);
    
    return EXIT_SUCCESS;
}
