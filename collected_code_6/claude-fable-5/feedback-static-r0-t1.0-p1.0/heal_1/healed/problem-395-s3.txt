```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_LEN 4096

int first_non_repeated_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t i;
    size_t len;

    if (str == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);

    if (len == MAX_INPUT_LEN && str[MAX_INPUT_LEN - 1] != '\0') {
        return -1;
    }

    for (i = 0; i < len; i++) {
        counts[(unsigned char)str[i]]++;
    }

    for (i = 0; i < len; i++) {
        if (counts[(unsigned char)str[i]] == 1) {
            *result = str[i];
            return 0;
        }
    }

    return -1;
}

int main(void)
{
    const char *test_strings[] = {
        "swiss",
        "aabbcc",
        "programming",
        "",
        "x"
    };
    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;
    char result;

    for (i = 0; i < num_tests; i++) {
        if (first_non_repeated_char(test_strings[i], &result) == 0) {
            if (printf("First non-repeated character in \"%s\": %c\n",
                       test_strings[i], result) < 0) {
                return 1;
            }
        } else {
            if (printf("No non-repeated character found in \"%s\"\n",
                       test_strings[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

**Analysis of the fix:**

The original code's only flagged weakness was the unbounded `strlen` call (CWE-126). Since `first_non_repeated_char` accepts an arbitrary `const char *`, it cannot guarantee the caller passed a properly NUL-terminated string. The rewrite addresses this by:

1. **Replacing `strlen` with `strnlen`**, bounded by `MAX_INPUT_LEN`, which guarantees the read never exceeds a fixed limit even if the input lacks a terminator.
2. **Rejecting inputs at the boundary** — if `strnlen` returns `MAX_INPUT_LEN` and no terminator was found within that window, the function returns an error rather than processing a potentially truncated or unterminated buffer.

The rest of the original code was already defensively written: NULL checks on both parameters, casting to `unsigned char` before indexing (avoiding UB with negative `char` values), a correctly sized `UCHAR_MAX + 1` count table, and checked `printf` return values in `main`. Those properties are preserved unchanged.

Note that `strnlen` is POSIX / C23; if strict C89/C99 portability is required, an equivalent bounded loop can be substituted.
