#include <stdio.h>

int findMinimum(int arr[], int size) {
    int low = 0;
    int high = size - 1;

    while (low < high) {
        if (arr[low] < arr[high]) {
            break;
        }
        
        int mid = low + (high - low) / 2;
        
        if (arr[mid] >= arr[low]) {
            low = mid + 1;
        } else {
            high = mid;
        }
    }

    return arr[low];
}

int main(void) {
    int arr[] = {4, 5, 6, 7, 1, 2, 3};
    int size = sizeof(arr) / sizeof(arr[0]);
    
    int min_val = findMinimum(arr, size);
    
    printf("%d\n", min_val);
    
    return 0;
}
