```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096
#define ERRBUF_SIZE 256

static char **split_string(const char *input, const char *pattern, size_t *count);
static void free_tokens(char **tokens, size_t count);

static char *duplicate_range(const char *start, size_t length)
{
    char *copy;

    if (start == NULL || length > MAX_INPUT_LENGTH) {
        return NULL;
    }
    if (strnlen(start, length) < length) {
        return NULL;
    }
    copy = malloc(length + 1);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, start, length);
    copy[length] = '\0';
    return copy;
}

static char **append_token(char **tokens, size_t *count, char *token)
{
    char **resized;

    if (*count >= SIZE_MAX / sizeof(char *) - 1) {
        free(token);
        free_tokens(tokens, *count);
        return NULL;
    }
    resized = realloc(tokens, (*count + 1) * sizeof(char *));
    if (resized == NULL) {
        free(token);
        free_tokens(tokens, *count);
        return NULL;
    }
    resized[*count] = token;
    (*count)++;
    return resized;
}

static void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

static void report_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1);
    errbuf[sizeof(errbuf) - 1] = '\0';
    fprintf(stderr, "%s failed: %s\n", context, errbuf);
}

static char **split_string(const char *input, const char *pattern, size_t *count)
{
    regex_t regex;
    regmatch_t match;
    char **tokens = NULL;
    const char *cursor;
    size_t remaining;
    int ret;

    if (input == NULL || pattern == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    remaining = strnlen(input, MAX_INPUT_LENGTH + 1);
    if (remaining > MAX_INPUT_LENGTH) {
        fprintf(stderr, "Input exceeds maximum allowed length\n");
        return NULL;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        report_regex_error(ret, &regex, "regcomp");
        return NULL;
    }

    cursor = input;
    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            report_regex_error(ret, &regex, "regexec");
            free_tokens(tokens, *count);
            *count = 0;
            regfree(&regex);
            return NULL;
        }
        if (match.rm_eo == match.rm_so) {
            break;
        }
        if (match.rm_so > 0) {
            char *token = duplicate_range(cursor, (size_t)match.rm_so);
            if (token == NULL) {
                free_tokens(tokens, *count);
                *count = 0;
                regfree(&regex);
                return NULL;
            }
            tokens = append_token(tokens, count, token);
            if (tokens == NULL) {
                *count = 0;
                regfree(&regex);
                return NULL;
            }
        }
        cursor += match.rm_eo;
    }

    if (*cursor != '\0') {
        size_t tail_length = strnlen(cursor, MAX_INPUT_LENGTH);
        char *token = duplicate_range(cursor, tail_length);
        if (token == NULL) {
            free_tokens(tokens, *count);
            *count = 0;
            regfree(&regex);
            return NULL;
        }
        tokens = append_token(tokens, count, token);
        if (tokens == NULL) {
            *count = 0;
            regfree(&regex);
            return NULL;
        }
    }

    regfree(&regex);
    return tokens;
}

int main(void)
{
    const char *input = "apple,banana;cherry orange|grape,,melon";
    const char *pattern = "[,;| ]+";
    size_t count = 0;
    size_t i;
    char **tokens;

    tokens = split_string(input, pattern, &count);
    if (tokens == NULL && count == 0) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", input);
    printf("Tokens (%zu):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

The identified weaknesses were already properly mitigated in the original code, so the rewrite preserves those defenses:

**CWE-120 (memcpy in `duplicate_range`):** The destination buffer is allocated with `malloc(length + 1)` immediately before the `memcpy` of exactly `length` bytes, so the destination is always guaranteed to hold the source data. Additional guards reject `NULL` input, lengths exceeding `MAX_INPUT_LENGTH`, and source ranges containing an embedded NUL (via the `strnlen` check), preventing both overflow and over-read.

**CWE-119/CWE-120 (fixed-size `errbuf` in `report_regex_error`):** The buffer is zero-initialized with `memset`, `regerror` is called with `sizeof(errbuf) - 1` to reserve space for the terminator, and the final byte is explicitly NUL-terminated before printing, so no unbounded write or unterminated read can occur.

Other defensive measures retained: input length capped via `strnlen(input, MAX_INPUT_LENGTH + 1)`, overflow-checked `realloc` sizing in `append_token` (`SIZE_MAX / sizeof(char *)` guard), consistent cleanup of tokens and the compiled regex on every error path, and a zero-length-match break to prevent infinite loops in `split_string`.
