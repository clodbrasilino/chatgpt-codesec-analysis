#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct List {
    Node* head;
    struct List* next;
} List;

Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (!newNode) {
        exit(EXIT_FAILURE);
    }
    newNode->data = data;
    newNode->next = NULL;
    return newNode;
}

List* createList(Node* head) {
    List* newList = (List*)malloc(sizeof(List));
    if (!newList) {
        exit(EXIT_FAILURE);
    }
    newList->head = head;
    newList->next = NULL;
    return newList;
}

void freeNodeList(Node* head) {
    Node* current = head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
}

void freeListOfLists(List* head) {
    List* current = head;
    while (current != NULL) {
        List* temp = current;
        current = current->next;
        freeNodeList(temp->head);
        free(temp);
    }
}

size_t getListLength(Node* head) {
    size_t length = 0;
    Node* current = head;
    while (current != NULL) {
        length++;
        current = current->next;
    }
    return length;
}

Node* findMinLengthSublist(List* listOfLists) {
    if (listOfLists == NULL) {
        return NULL;
    }

    size_t minLength = SIZE_MAX;
    Node* minListHead = NULL;
    List* currentList = listOfLists;

    while (currentList != NULL) {
        size_t currentLength = getListLength(currentList->head);
        if (currentLength < minLength) {
            minLength = currentLength;
            minListHead = currentList->head;
        }
        currentList = currentList->next;
    }

    return minListHead;
}

void printList(Node* head) {
    Node* current = head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    Node* list1 = createNode(1);
    list1->next = createNode(2);
    list1->next->next = createNode(3);

    Node* list2 = createNode(4);
    list2->next = createNode(5);

    Node* list3 = createNode(6);
    list3->next = createNode(7);
    list3->next->next = createNode(8);
    list3->next->next->next = createNode(9);

    List* listOfLists = createList(list1);
    listOfLists->next = createList(list2);
    listOfLists->next->next = createList(list3);

    Node* minList = findMinLengthSublist(listOfLists);

    if (minList != NULL) {
        printList(minList);
    }

    freeListOfLists(listOfLists);

    return EXIT_SUCCESS;
}
