#include <stdio.h>
#include <stdlib.h>

static unsigned long long binomial_coefficient(unsigned int n, unsigned int k)
{
    unsigned long long result = 1ULL;
    unsigned int i;

    if (k > n) {
        return 0ULL;
    }
    if (k > n - k) {
        k = n - k;
    }
    for (i = 0U; i < k; i++) {
        result = result * (unsigned long long)(n - i);
        result = result / (unsigned long long)(i + 1U);
    }
    return result;
}

static unsigned long long derangement(unsigned int n)
{
    unsigned long long prev2;
    unsigned long long prev1;
    unsigned long long current;
    unsigned int i;

    if (n == 0U) {
        return 1ULL;
    }
    if (n == 1U) {
        return 0ULL;
    }
    prev2 = 1ULL;
    prev1 = 0ULL;
    current = 0ULL;
    for (i = 2U; i <= n; i++) {
        current = (unsigned long long)(i - 1U) * (prev1 + prev2);
        prev2 = prev1;
        prev1 = current;
    }
    return current;
}

static unsigned long long rencontres_number(unsigned int n, unsigned int m)
{
    if (m > n) {
        return 0ULL;
    }
    return binomial_coefficient(n, m) * derangement(n - m);
}

int main(void)
{
    unsigned int n;
    unsigned int m;
    int scan_result;

    printf("Enter n: ");
    scan_result = scanf("%u", &n);
    if (scan_result != 1) {
        fprintf(stderr, "Invalid input for n\n");
        return EXIT_FAILURE;
    }

    printf("Enter m: ");
    scan_result = scanf("%u", &m);
    if (scan_result != 1) {
        fprintf(stderr, "Invalid input for m\n");
        return EXIT_FAILURE;
    }

    if (n > 20U) {
        fprintf(stderr, "Input too large, result would overflow\n");
        return EXIT_FAILURE;
    }

    printf("Rencontres number D(%u, %u) = %llu\n", n, m, rencontres_number(n, m));

    return EXIT_SUCCESS;
}

This program computes the rencontres number D(n, m), which counts permutations of n elements with exactly m fixed points, using the formula D(n, m) = C(n, m) × d(n − m), where d is the derangement count. Input is validated, the derangement is computed iteratively to avoid stack issues, and n is capped at 20 to prevent unsigned long long overflow.
