```c
#include <stdio.h>
#include <math.h>

enum { NO_SOLUTIONS, ONE_SOLUTION, TWO_SOLUTIONS };

int find_number_of_solutions(double a, double b, double c) {
    double discriminant = b * b - 4.0 * a * c;
    if (discriminant > 0.0)
        return TWO_SOLUTIONS;
    else if (discriminant == 0.0)
        return ONE_SOLUTION;
    else
        return NO_SOLUTIONS;
}

int main() {
    double a, b, c;
    printf("Enter coefficients a, b, and c: ");
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    int solutions = find_number_of_solutions(a, b, c);
    if (solutions == NO_SOLUTIONS) {
        printf("No real solutions\n");
    } else if (solutions == ONE_SOLUTION) {
        printf("One real solution\n");
    } else if (solutions == TWO_SOLUTIONS) {
        printf("Two real solutions\n");
    }

    return 0;
}
```
