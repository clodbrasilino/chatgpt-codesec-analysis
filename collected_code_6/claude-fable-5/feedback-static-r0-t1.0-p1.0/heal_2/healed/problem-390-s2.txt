```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096

static void cleanup_partial(char **result, size_t filled)
{
    while (filled > 0) {
        filled--;
        free(result[filled]);
    }
    free(result);
}

char **prepend_to_all(const char *prefix, const char **items, size_t count)
{
    char **result;
    size_t i;
    size_t prefix_len;

    if (prefix == NULL || items == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    prefix_len = strnlen(prefix, MAX_STR_LEN);
    if (prefix_len == MAX_STR_LEN && prefix[MAX_STR_LEN - 1] != '\0') {
        return NULL;
    }

    result = malloc(count * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t item_len;
        size_t total_len;
        int written;

        if (items[i] == NULL) {
            cleanup_partial(result, i);
            return NULL;
        }

        item_len = strnlen(items[i], MAX_STR_LEN);
        if (item_len == MAX_STR_LEN && items[i][MAX_STR_LEN - 1] != '\0') {
            cleanup_partial(result, i);
            return NULL;
        }

        if (item_len > SIZE_MAX - prefix_len - 1) {
            cleanup_partial(result, i);
            return NULL;
        }

        total_len = prefix_len + item_len + 1;

        result[i] = malloc(total_len);
        if (result[i] == NULL) {
            cleanup_partial(result, i);
            return NULL;
        }

        written = snprintf(result[i], total_len, "%s%s", prefix, items[i]);
        if (written < 0 || (size_t)written >= total_len) {
            free(result[i]);
            cleanup_partial(result, i);
            return NULL;
        }
    }

    return result;
}

void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *items[] = { "apple", "banana", "cherry" };
    size_t count = sizeof(items) / sizeof(items[0]);
    char **result;
    size_t i;

    result = prepend_to_all("fruit_", items, count);
    if (result == NULL) {
        fprintf(stderr, "Failed to build list\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", result[i]) < 0) {
            free_list(result, count);
            return EXIT_FAILURE;
        }
    }

    free_list(result, count);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already largely defensive, but the rewrite addresses the flagged concerns and improves maintainability:

1. **CWE-120 (memcpy warnings):** The two `memcpy` calls were technically safe because `total_len = prefix_len + item_len + 1` was validated before allocation. However, the safety depended on maintaining an invariant across multiple lines of code. Replacing them with a single bounded `snprintf(dest, total_len, "%s%s", ...)` makes the bound explicit at the call site, guarantees null-termination, and adds a verifiable post-condition check (`written >= total_len` detects any mismatch).

2. **Code duplication:** The cleanup loop (`while (i > 0) { i--; free(result[i]); } free(result);`) was repeated five times in the original. This was extracted into `cleanup_partial()`, reducing the chance of a copy-paste error in error-handling paths — a common source of leaks and double-frees.

3. **Retained protections:** overflow guard on `count * sizeof(char *)`, `strnlen` capping against unterminated strings via `MAX_STR_LEN`, per-item NULL checks, `SIZE_MAX` addition-overflow check, and full rollback on any mid-loop failure.
