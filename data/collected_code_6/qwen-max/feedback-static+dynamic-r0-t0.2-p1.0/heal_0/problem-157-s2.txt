#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    int count;
    struct Node *next;
} Node;

Node* createNode(int value, int count) {
    Node *newNode = (Node *)malloc(sizeof(Node));
    if (newNode == NULL) {
        exit(EXIT_FAILURE);
    }
    newNode->value = value;
    newNode->count = count;
    newNode->next = NULL;
    return newNode;
}

void freeList(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

Node* runLengthEncoding(int *arr, int n) {
    if (n == 0) return NULL;

    Node *head = NULL;
    Node *tail = NULL;

    for (int i = 0; i < n; ) {
        int count = 1;
        while (i + 1 < n && arr[i] == arr[i + 1]) {
            i++;
            count++;
        }

        Node *newNode = createNode(arr[i], count);
        if (head == NULL) {
            head = tail = newNode;
        } else {
            tail->next = newNode;
            tail = newNode;
        }
        i++;
    }

    return head;
}

void printList(Node *head) {
    Node *current = head;
    while (current != NULL) {
        printf("%d %d\n", current->value, current->count);
        current = current->next;
    }
}

int main() {
    int arr[] = {1, 1, 2, 2, 2, 3, 4, 4, 4, 4};
    int n = sizeof(arr) / sizeof(arr[0]);

    Node *encodedList = runLengthEncoding(arr, n);
    printList(encodedList);

    freeList(encodedList);
    return 0;
}
