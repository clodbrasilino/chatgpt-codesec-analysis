#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool is_valid_parentheses(const char *str) {
    int balance = 0;
    while (*str) {
        if (*str == '(') {
            balance++;
        } else if (*str == ')') {
            balance--;
            if (balance < 0) {
                return false;
            }
        }
        str++;
    }
    return balance == 0;
}

int main() {
    const char *test1 = "()";
    const char *test2 = "(()())";
    const char *test3 = ")(";
    const char *test4 = "(()";

    printf("%s: %s\n", test1, is_valid_parentheses(test1) ? "True" : "False");
    printf("%s: %s\n", test2, is_valid_parentheses(test2) ? "True" : "False");
    printf("%s: %s\n", test3, is_valid_parentheses(test3) ? "True" : "False");
    printf("%s: %s\n", test4, is_valid_parentheses(test4) ? "True" : "False");

    return 0;
}
