#include <stdio.h>
#include <stdlib.h>

int are_tuples_equal_length(int **tuples, int num_tuples) {
    if (num_tuples == 0) return 1;

    int length = 0;
    for (int i = 0; i < num_tuples; ++i) {
        if (i == 0) {
            length = tuples[i][0];
        } else if (tuples[i][0] != length) {
            return 0;
        }
    }
    return 1;
}

int main() {
    int num_tuples = 3;
    int lengths[] = {2, 2, 2};
    int *tuples[3] = {lengths, lengths + 1, lengths + 2};

    if (are_tuples_equal_length(tuples, num_tuples)) {
        printf("All tuples have equal length.\n");
    } else {
        printf("Not all tuples have equal length.\n");
    }

    return 0;
}
