```C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *concatenateElements(char **tuple, int size, char *delimiter) {
    int i;
    unsigned long total_length = 0;

    for (i = 0; i < size; i++) {
        total_length += strlen(tuple[i]);
    }

    total_length += (size-1)*strlen(delimiter)+1;

    char *result = malloc(total_length * sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    strcpy(result, tuple[0]);

    for (i = 1; i < size; i++) {
        strcat(result, delimiter);
        strcat(result, tuple[i]);
    }

    return result;
}

int main() {
    int size = 3;
    char *tuple[] = {"Hello", "World", "!"};
    char *delimiter = ", ";

    char *result = concatenateElements(tuple, size, delimiter);
    if (result == NULL) {
        fprintf(stderr, "Failed to allocate memory for the resultant string.\n");
        return 1;
    }

    printf("%s\n", result);
    free(result);

    return 0;
}
```
