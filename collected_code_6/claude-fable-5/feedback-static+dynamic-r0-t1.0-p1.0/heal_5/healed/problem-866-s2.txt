#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN] = {0};
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    for (i = 0; i < len && i < MAX_INPUT_LEN - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(lowered, months_31[i]) == 0) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;

    printf("Enter a month name: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis notes:**

The two Flawfinder hits (CWE-119/CWE-120) are informational warnings triggered simply by the presence of statically-sized `char` arrays — they are not actual vulnerabilities in this code. Both were already properly mitigated, so the rewrite preserves the same defensive structure:

1. **`lowered[MAX_INPUT_LEN]` in `has_31_days`** — protected by:
   - `NULL` pointer check before use.
   - `strnlen()` with an explicit bound instead of `strlen()`, preventing reads past unterminated buffers.
   - Rejection of inputs where `len >= MAX_INPUT_LEN`.
   - The copy loop is double-bounded (`i < len && i < MAX_INPUT_LEN - 1`), guaranteeing room for the null terminator, which is explicitly written afterward.
   - `tolower()` is correctly called with an `unsigned char` cast, avoiding undefined behavior on negative `char` values.

2. **`input[MAX_INPUT_LEN]` in `main`** — protected by:
   - `fgets(input, sizeof(input), stdin)` which never writes more than `sizeof(input) - 1` characters plus the terminator, with its return value checked for I/O failure.
   - A defensive forced null termination at the last byte.
   - Bounded `strnlen()` before the newline-stripping logic, with a `len > 0` guard preventing an out-of-bounds access at `input[len - 1]`.

No unbounded functions (`gets`, `strcpy`, `sprintf`, `strlen` on untrusted data) are used, so the code is safe against the buffer-overflow classes flagged. The warnings can be safely suppressed or documented as reviewed false positives.
