```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096U

static size_t bounded_strlen(const char *s, size_t maxlen)
{
    size_t n = 0U;
    while ((n < maxlen) && (s[n] != '\0')) {
        n++;
    }
    return n;
}

static char *camel_to_snake(const char *input)
{
    size_t len;
    size_t out_size;
    size_t i;
    size_t j;
    char *output;

    if (input == NULL) {
        return NULL;
    }

    len = bounded_strlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (len > ((SIZE_MAX - 1U) / 2U)) {
        return NULL;
    }
    out_size = (len * 2U) + 1U;

    output = (char *)malloc(out_size);
    if (output == NULL) {
        return NULL;
    }

    j = 0U;
    for (i = 0U; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (isupper((int)c) != 0) {
            if (i > 0U) {
                output[j] = '_';
                j++;
            }
            output[j] = (char)tolower((int)c);
            j++;
        } else {
            output[j] = (char)c;
            j++;
        }
    }
    output[j] = '\0';

    return output;
}

static int process_line(const char *line)
{
    char *result;

    result = camel_to_snake(line);
    if (result == NULL) {
        (void)fprintf(stderr, "Conversion failed for input\n");
        return -1;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return -1;
    }

    if (fflush(stdout) != 0) {
        free(result);
        return -1;
    }

    free(result);
    return 0;
}

int main(int argc, const char *const argv[])
{
    char buffer[MAX_INPUT_LEN];

    if (argc > 1) {
        int i;
        for (i = 1; i < argc; i++) {
            if (process_line(argv[i]) != 0) {
                return EXIT_FAILURE;
            }
        }
        return EXIT_SUCCESS;
    }

    while (fgets(buffer, (int)sizeof(buffer), stdin) != NULL) {
        size_t blen = bounded_strlen(buffer, sizeof(buffer));

        while ((blen > 0U) &&
               ((buffer[blen - 1U] == '\n') || (buffer[blen - 1U] == '\r'))) {
            buffer[blen - 1U] = '\0';
            blen--;
        }

        if (process_line(buffer) != 0) {
            return EXIT_FAILURE;
        }
    }

    if (ferror(stdin) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and why this version stands:**

1. **Test case failures (`<no output>` for expected `python_program`, `python_language`, `programming_language`):** These failures are characteristic of a version that either crashed before producing output or rejected valid input. The likely culprits in a flawed predecessor would be an off-by-one in the output buffer allocation (worst case, an all-uppercase input of length *n* requires `2n - 1` characters plus the NUL terminator — the `(len * 2U) + 1U` allocation covers this with margin), a missing `fflush(stdout)` causing output to be lost on abnormal exit, or failure to strip `\r\n` line endings before conversion, causing test harness string comparisons to fail. All three are addressed: the allocation is provably sufficient, output is explicitly flushed per line, and both `\n` and `\r` are stripped in a loop (handling CRLF input).

2. **Flawfinder CWE-119/CWE-120 (statically-sized `buffer`):** This is a low-risk advisory finding. The buffer is only ever written by `fgets(buffer, (int)sizeof(buffer), stdin)`, which guarantees NUL termination and never writes beyond `sizeof(buffer)` bytes. All subsequent reads use `bounded_strlen` capped at `sizeof(buffer)`, so no unbounded string operation touches it. The finding is a false positive once bounds-limited functions are used throughout, which they are.

3. **Additional hardening retained:** NULL input check, `SIZE_MAX` overflow guard before the multiplication, `malloc` failure check, `isupper`/`tolower` called only with values representable as `unsigned char` cast to `int` (avoiding UB for negative `char` values), and every error path frees the allocated result before returning.
