#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *tuple_join(const char **tuple, size_t count, const char *delimiter) {
    if (tuple == NULL || delimiter == NULL || count == 0) {
        return NULL;
    }

    size_t delim_len = strlen(delimiter);
    size_t total_len = 0;

    for (size_t i = 0; i < count; i++) {
        if (tuple[i] == NULL) {
            return NULL;
        }
        total_len += strlen(tuple[i]);
        if (i < count - 1) {
            total_len += delim_len;
        }
    }

    char *result = (char *)malloc(total_len + 1);
    if (result == NULL) {
        return NULL;
    }

    char *ptr = result;

    for (size_t i = 0; i < count; i++) {
        size_t len = strlen(tuple[i]);
        memcpy(ptr, tuple[i], len);
        ptr += len;
        if (i < count - 1) {
            memcpy(ptr, delimiter, delim_len);
            ptr += delim_len;
        }
    }

    *ptr = '\0';

    return result;
}

int main(void) {
    const char *tuple[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    const char *delimiter = ", ";

    char *result = tuple_join(tuple, count, delimiter);
    if (result == NULL) {
        fprintf(stderr, "Failed to join tuple\n");
        return 1;
    }

    printf("%s\n", result);

    free(result);

    return 0;
}
