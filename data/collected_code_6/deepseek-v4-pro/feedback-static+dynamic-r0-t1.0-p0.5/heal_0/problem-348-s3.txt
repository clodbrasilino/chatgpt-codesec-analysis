#include <stdio.h>
#include <stdlib.h>

int count_sequences(int *values, int num_values, int length) {
    if (values == NULL || num_values <= 0 || length <= 0) {
        return 0;
    }

    int *dp = (int *)calloc((size_t)length + 1, sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    dp[0] = 1;

    for (int i = 1; i <= length; i++) {
        dp[i] = 0;
        for (int j = 0; j < num_values; j++) {
            if (i - values[j] >= 0) {
                dp[i] += dp[i - values[j]];
            }
        }
    }

    int result = dp[length];
    free(dp);
    return result;
}

int main(void) {
    int values[] = {1, 2, 3};
    int num_values = sizeof(values) / sizeof(values[0]);
    int length = 5;

    int count = count_sequences(values, num_values, length);
    printf("%d\n", count);

    return 0;
}
