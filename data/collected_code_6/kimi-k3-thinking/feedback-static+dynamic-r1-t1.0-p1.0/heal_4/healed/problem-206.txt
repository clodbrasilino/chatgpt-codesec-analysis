#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <ctype.h>

#define MAX_ELEMENT_LENGTH (16777216U)

static void free_string_array(char **strings, size_t count)
{
    size_t i;

    if (strings == NULL)
    {
        return;
    }

    for (i = 0U; i < count; i++)
    {
        free(strings[i]);
    }
    free(strings);
}

static size_t bounded_length(const char *s, int *terminated)
{
    size_t len = 0U;

    *terminated = 0;

    if (s == NULL)
    {
        return 0U;
    }

    while (len < MAX_ELEMENT_LENGTH)
    {
        if (s[len] == '\0')
        {
            *terminated = 1;
            return len;
        }
        len++;
    }

    return len;
}

static char **concat_adjacent_elements(const char *const *elements, size_t count, size_t *out_count)
{
    char **result;
    size_t i;

    if (out_count == NULL)
    {
        return NULL;
    }
    *out_count = 0U;

    if (elements == NULL || count < 2U)
    {
        return NULL;
    }

    if ((count - 1U) > (SIZE_MAX / sizeof(char *)))
    {
        return NULL;
    }

    result = malloc((count - 1U) * sizeof(*result));
    if (result == NULL)
    {
        return NULL;
    }

    for (i = 0U; i < count - 1U; i++)
    {
        const char *left = elements[i];
        const char *right = elements[i + 1U];
        size_t left_len;
        size_t right_len;
        size_t total_len;
        size_t capacity;
        int left_ok;
        int right_ok;

        if (left == NULL || right == NULL)
        {
            free_string_array(result, i);
            return NULL;
        }

        left_len = bounded_length(left, &left_ok);
        right_len = bounded_length(right, &right_ok);

        if (!left_ok || !right_ok)
        {
            free_string_array(result, i);
            return NULL;
        }

        if (right_len > (SIZE_MAX - left_len - 1U))
        {
            free_string_array(result, i);
            return NULL;
        }

        total_len = left_len + right_len;
        capacity = total_len + 1U;

        result[i] = malloc(capacity);
        if (result[i] == NULL)
        {
            free_string_array(result, i);
            return NULL;
        }

        if ((left_len <= capacity) && (right_len <= (capacity - left_len)))
        {
            memcpy(result[i], left, left_len);
            memcpy(result[i] + left_len, right, right_len);
            result[i][total_len] = '\0';
        }
        else
        {
            free_string_array(result, i + 1U);
            return NULL;
        }
    }

    *out_count = count - 1U;
    return result;
}

static int add_token(char ***tokens, size_t *count, size_t *capacity, char *token)
{
    char **grown;
    size_t new_capacity;

    if (token == NULL)
    {
        return -1;
    }

    if (*count == *capacity)
    {
        new_capacity = (*capacity == 0U) ? 8U : (*capacity * 2U);
        if ((new_capacity < *capacity) || (new_capacity > (SIZE_MAX / sizeof(**tokens))))
        {
            free(token);
            return -1;
        }
        grown = realloc(*tokens, new_capacity * sizeof(**tokens));
        if (grown == NULL)
        {
            free(token);
            return -1;
        }
        *tokens = grown;
        *capacity = new_capacity;
    }

    (*tokens)[*count] = token;
    *count += 1U;
    return 0;
}

static int split_and_add(const char *text, char ***tokens, size_t *count, size_t *capacity)
{
    const char *cursor = text;

    if (cursor == NULL)
    {
        return -1;
    }

    while (*cursor != '\0')
    {
        const char *start;
        size_t length;
        char *token;

        while ((*cursor != '\0') && isspace((unsigned char)*cursor))
        {
            cursor++;
        }
        if (*cursor == '\0')
        {
            break;
        }

        start = cursor;
        while ((*cursor != '\0') && !isspace((unsigned char)*cursor))
        {
            cursor++;
        }

        length = (size_t)(cursor - start);
        token = malloc(length + 1U);
        if (token == NULL)
        {
            return -1;
        }
        if (length > 0U)
        {
            memcpy(token, start, length);
        }
        token[length] = '\0';

        if (add_token(tokens, count, capacity, token) != 0)
        {
            return -1;
        }
    }

    return 0;
}

static char *read_token(FILE *stream)
{
    size_t capacity = 32U;
    size_t length = 0U;
    char *buffer;
    int c;

    do
    {
        c = fgetc(stream);
    } while ((c != EOF) && isspace(c));

    if (c == EOF)
    {
        return NULL;
    }

    buffer = malloc(capacity);
    if (buffer == NULL)
    {
        return NULL;
    }

    while ((c != EOF) && !isspace(c))
    {
        if ((length + 1U) >= capacity)
        {
            size_t new_capacity = capacity * 2U;
            char *grown;

            if (new_capacity < capacity)
            {
                free(buffer);
                return NULL;
            }
            grown = realloc(buffer, new_capacity);
            if (grown == NULL)
            {
                free(buffer);
                return NULL;
            }
            buffer = grown;
            capacity = new_capacity;
        }
        buffer[length] = (char)c;
        length++;
        c = fgetc(stream);
    }

    buffer[length] = '\0';
    return buffer;
}

int main(int argc, char **argv)
{
    static const char default_input[] = "DSP IS BEST FOR ALL UTS";
    char **tokens = NULL;
    size_t token_count = 0U;
    size_t token_capacity = 0U;
    char **elements = NULL;
    char **result = NULL;
    size_t result_count = 0U;
    size_t i;
    int exit_code = EXIT_FAILURE;
    int arg_index;

    if (argc > 1)
    {
        for (arg_index = 1; arg_index < argc; arg_index++)
        {
            if (split_and_add(argv[arg_index], &tokens, &token_count, &token_capacity) != 0)
            {
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
        }
    }
    else
    {
        char *token;

        while ((token = read_token(stdin)) != NULL)
        {
            if (add_token(&tokens, &token_count, &token_capacity, token) != 0)
            {
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
        }

        if (token_count == 0U)
        {
            if (split_and_add(default_input, &tokens, &token_count, &token_capacity) != 0)
            {
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
        }
    }

    if (token_count >= 2U)
    {
        elements = calloc(token_count, sizeof(*elements));
        if (elements == NULL)
        {
            free_string_array(tokens, token_count);
            return EXIT_FAILURE;
        }

        for (i = 0U; i < token_count; i++)
        {
            size_t length = strlen(tokens[i]);
            int needs_space = ((i + 1U) < token_count);
            size_t total = length + (needs_space ? 1U : 0U);

            elements[i] = malloc(total + 1U);
            if (elements[i] == NULL)
            {
                free_string_array(elements, token_count);
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
            if (length > 0U)
            {
                memcpy(elements[i], tokens[i], length);
            }
            if (needs_space)
            {
                elements[i][length] = ' ';
            }
            elements[i][total] = '\0';
        }

        result = concat_adjacent_elements((const char *const *)elements, token_count, &result_count);
        if (result == NULL)
        {
            fprintf(stderr, "Error: failed to concatenate adjacent elements.\n");
        }
        else
        {
            for (i = 0U; i < result_count; i++)
            {
                printf("%s\n", result[i]);
            }
            free_string_array(result, result_count);
            exit_code = EXIT_SUCCESS;
        }

        free_string_array(elements, token_count);
    }
    else
    {
        exit_code = EXIT_SUCCESS;
    }

    free_string_array(tokens, token_count);
    return exit_code;
}
