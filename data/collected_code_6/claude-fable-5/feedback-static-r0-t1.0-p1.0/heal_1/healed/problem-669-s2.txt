```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define IP_PATTERN "^((25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9]?[0-9])\\.){3}(25[0-5]|2[0-4][0-9]|1[0-9][0-9]|[1-9]?[0-9])$"

static int is_valid_ip(const char *ip)
{
    regex_t regex;
    int ret;

    if (ip == NULL) {
        return 0;
    }

    ret = regcomp(&regex, IP_PATTERN, REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        size_t errlen = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errlen);
        if (errbuf != NULL) {
            regerror(ret, &regex, errbuf, errlen);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Regex compilation failed\n");
        }
        return -1;
    }

    ret = regexec(&regex, ip, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex execution failed\n");
        return -1;
    }
}

int main(void)
{
    const char *test_ips[] = {
        "192.168.1.1",
        "255.255.255.255",
        "0.0.0.0",
        "256.1.1.1",
        "192.168.1",
        "192.168.1.1.1",
        "abc.def.ghi.jkl",
        "10.0.0.01",
        ""
    };
    size_t num_ips = sizeof(test_ips) / sizeof(test_ips[0]);
    size_t i;

    for (i = 0; i < num_ips; i++) {
        int result = is_valid_ip(test_ips[i]);
        if (result < 0) {
            fprintf(stderr, "Error validating IP: %s\n", test_ips[i]);
            return EXIT_FAILURE;
        }
        printf("%-20s : %s\n", test_ips[i], result == 1 ? "Valid" : "Invalid");
    }

    return EXIT_SUCCESS;
}
```

**Summary of changes addressing the identified weaknesses:**

1. **CWE-119/CWE-120 (statically-sized buffer in `regerror` handling):** Replaced the fixed 256-byte stack buffer with a two-step call to `regerror()` — the first call queries the exact required buffer size, and the buffer is then dynamically allocated to that size. This eliminates any possibility of truncation or improper size assumptions, with a graceful fallback if `malloc` fails.

2. **Variable scope (`result`):** Moved the declaration of `result` inside the `for` loop body, reducing its scope to where it is actually used, as flagged by cppcheck.

3. **Minor hardening (bonus):** Added `REG_NOSUB` to the `regcomp` flags since no subexpression match data is used, made `is_valid_ip` `static` as it's not referenced outside the translation unit, and ensured the `errbuf` allocation is always freed on the error path.
