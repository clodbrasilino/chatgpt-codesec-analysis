```C
#include <stdio.h>

long long multiplyList(int *arr, int n) {
    long long product = 1;
    for(int i = 0; i < n; i++) {
        product *= arr[i];
    }
    return product;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);

    long long product = multiplyList(arr, n);
    
    printf("The product of all list elements is: %llu\n", product);
    
    return 0;
}
```
