```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_WORD_LEN 256

bool is_length_odd(const char *word)
{
    size_t len;

    if (word == NULL) {
        return false;
    }

    len = strnlen(word, MAX_WORD_LEN);
    return (len % 2U) != 0U;
}

int main(void)
{
    char buffer[MAX_WORD_LEN];
    size_t len;

    printf("Enter a word: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
    }

    if (buffer[0] == '\0') {
        fprintf(stderr, "No word entered\n");
        return EXIT_FAILURE;
    }

    if (is_length_odd(buffer)) {
        printf("The word \"%s\" has an odd length.\n", buffer);
    } else {
        printf("The word \"%s\" has an even length.\n", buffer);
    }

    return EXIT_SUCCESS;
}
```

- **Bounded length checks**: `strnlen` is used instead of `strlen`, both in `main` and in `is_length_odd`, so length computation can never run past the buffer.
- **NULL guard**: `is_length_odd` checks for a `NULL` pointer before dereferencing.
- **Input validation**: the code handles `fgets` failure, strips the trailing newline safely (checking `len > 0U` before indexing `len - 1U`), and rejects empty input.
