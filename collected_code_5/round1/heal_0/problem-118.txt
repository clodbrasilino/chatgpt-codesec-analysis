```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

typedef struct Node {
    char data;
    struct Node* next;
} Node;

Node* convertToList(char* str) {
    if (str == NULL || *str == '\0') return NULL;
  
    Node *head = malloc(sizeof(Node));
    if (head == NULL) return NULL;
    
    Node *temp = head;
    for (size_t i = 0; i < strlen(str); i++) {
        temp->data = str[i];
        temp->next = i < strlen(str) - 1 ? malloc(sizeof(Node)) : NULL;
        if (i < strlen(str) - 1 && temp->next == NULL) {
            Node *ptr = head;
            while (ptr) {
                Node *del = ptr;
                ptr = ptr->next;
                free(del);
            }
            return NULL;
        }
        temp = temp->next;
    }
    return head;
}

int main() {
    char str[] = "Hello World!";
    Node* list = convertToList(str);
  
    Node* temp = list;
    while(temp != NULL){
        printf("%c ", temp -> data);
        temp = temp -> next;
    }

    // free memory
    Node *ptr = list;
    while (ptr) {
        Node *del = ptr;
        ptr = ptr->next;
        free(del);
    }

    return 0;
}
```
This C program includes a function `convertToList()` that creates a linked list from a given string. Each node in the list contains one character from the string. In this program, memory is dynamically allocated for each new node. If the string is NULL or empty, the function returns NULL. If malloc fails to allocate memory, it also returns NULL after freeing any memory that was allocated. The `main` function demonstrates how to use this, by converting a string and then freeing the memory used.
