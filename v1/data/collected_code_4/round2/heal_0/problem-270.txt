```C
#include <stdio.h>

int sumEvenPositionEvenNumbers(int arr[], int size) {
    int sum = 0;
    for (int i = 0; i < size; i += 2) {
        if (arr[i] % 2 == 0) { 
            sum += arr[i];
        }
    }
    return sum;
}

int main() {
    int arr[] = {4, 3, 2, 9, 6, 7, 4, 8};
    int size = sizeof(arr) / sizeof(arr[0]);

    int sum = sumEvenPositionEvenNumbers(arr, size);

    printf("Sum of even numbers at even positions: %d\n", sum);
    return 0;
}
```
