#include <stdio.h>
#include <stdlib.h>
#include <float.h>

double findMaxAveragePath(int **matrix, int n) {
    if (matrix == NULL || n <= 0) {
        return 0.0;
    }

    double **dp = (double **)malloc(n * sizeof(double *));
    int **count = (int **)malloc(n * sizeof(int *));
    if (dp == NULL || count == NULL) {
        if (dp != NULL) free(dp);
        if (count != NULL) free(count);
        return 0.0;
    }

    for (int i = 0; i < n; i++) {
        dp[i] = (double *)malloc(n * sizeof(double));
        count[i] = (int *)malloc(n * sizeof(int));
        if (dp[i] == NULL || count[i] == NULL) {
            for (int j = 0; j <= i; j++) {
                if (dp[j] != NULL) free(dp[j]);
                if (count[j] != NULL) free(count[j]);
            }
            free(dp);
            free(count);
            return 0.0;
        }
    }

    dp[0][0] = matrix[0][0];
    count[0][0] = 1;

    for (int j = 1; j < n; j++) {
        dp[0][j] = dp[0][j-1] + matrix[0][j];
        count[0][j] = count[0][j-1] + 1;
    }

    for (int i = 1; i < n; i++) {
        dp[i][0] = dp[i-1][0] + matrix[i][0];
        count[i][0] = count[i-1][0] + 1;
    }

    for (int i = 1; i < n; i++) {
        for (int j = 1; j < n; j++) {
            double avg_up = dp[i-1][j] / count[i-1][j];
            double avg_left = dp[i][j-1] / count[i][j-1];

            if (avg_up > avg_left) {
                dp[i][j] = dp[i-1][j] + matrix[i][j];
                count[i][j] = count[i-1][j] + 1;
            } else {
                dp[i][j] = dp[i][j-1] + matrix[i][j];
                count[i][j] = count[i][j-1] + 1;
            }
        }
    }

    double result = dp[n-1][n-1] / count[n-1][n-1];

    for (int i = 0; i < n; i++) {
        free(dp[i]);
        free(count[i]);
    }
    free(dp);
    free(count);

    return result;
}

int main(void) {
    int n;
    if (scanf("%d", &n) != 1 || n <= 0) {
        return 1;
    }

    int **matrix = (int **)malloc(n * sizeof(int *));
    if (matrix == NULL) {
        return 1;
    }

    for (int i = 0; i < n; i++) {
        matrix[i] = (int *)malloc(n * sizeof(int));
        if (matrix[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(matrix[j]);
            }
            free(matrix);
            return 1;
        }
    }

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (scanf("%d", &matrix[i][j]) != 1) {
                for (int k = 0; k < n; k++) {
                    free(matrix[k]);
                }
                free(matrix);
                return 1;
            }
        }
    }

    double max_avg = findMaxAveragePath(matrix, n);
    printf("%.6f\n", max_avg);

    for (int i = 0; i < n; i++) {
        free(matrix[i]);
    }
    free(matrix);

    return 0;
}
