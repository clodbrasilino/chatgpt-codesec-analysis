```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_LEN 32

typedef struct {
    char name[NAME_LEN];
    int value;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static Tuple *find_min_k_records(const Tuple *records, size_t n, size_t k)
{
    Tuple *sorted;
    Tuple *result;
    size_t i;

    if (records == NULL || n == 0U || k == 0U || k > n) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(*sorted)) {
        return NULL;
    }

    sorted = malloc(n * sizeof(*sorted));
    if (sorted == NULL) {
        return NULL;
    }

    memcpy(sorted, records, n * sizeof(*sorted));

    for (i = 0U; i < n; i++) {
        sorted[i].name[NAME_LEN - 1] = '\0';
    }

    qsort(sorted, n, sizeof(*sorted), compare_tuples);

    if (k > SIZE_MAX / sizeof(*result)) {
        free(sorted);
        return NULL;
    }

    result = malloc(k * sizeof(*result));
    if (result == NULL) {
        free(sorted);
        return NULL;
    }

    memcpy(result, sorted, k * sizeof(*result));
    free(sorted);

    return result;
}

int main(void)
{
    Tuple records[] = {
        { "Alice", 45 },
        { "Bob", 12 },
        { "Carol", 78 },
        { "Dave", 3 },
        { "Eve", 56 },
        { "Frank", 23 }
    };
    size_t n = sizeof(records) / sizeof(records[0]);
    size_t k = 3U;
    Tuple *min_records;
    size_t i;

    min_records = find_min_k_records(records, n, k);
    if (min_records == NULL) {
        fprintf(stderr, "Failed to find minimum records\n");
        return EXIT_FAILURE;
    }

    printf("Minimum %zu records:\n", k);
    for (i = 0U; i < k; i++) {
        printf("%.*s: %d\n", NAME_LEN - 1, min_records[i].name, min_records[i].value);
    }

    free(min_records);

    return EXIT_SUCCESS;
}
```

**Summary of fixes applied:**

1. **Integer overflow before allocation (CWE-190 leading to CWE-120):** Both `malloc(n * sizeof(*sorted))` and `malloc(k * sizeof(*result))` could overflow if `n` or `k` were very large, producing an undersized buffer that the subsequent `memcpy` would overrun. Guards against `SIZE_MAX / sizeof(...)` were added before each allocation, guaranteeing the destination can always hold the source data — directly addressing both flagged `memcpy` calls.

2. **Statically-sized `name` buffer (CWE-119/CWE-120):** After copying the input records, every `name` field is forcibly NUL-terminated at `NAME_LEN - 1`. This ensures that even if a caller passes a `Tuple` with an unterminated name, no string operation reads past the array bounds.

3. **Bounded output:** `printf` now uses `%.*s` with an explicit maximum width of `NAME_LEN - 1`, so printing can never read beyond the fixed-size buffer even in the presence of a corrupted record.

4. **Existing safeguards retained:** the parameter validation (`NULL`, `n == 0`, `k == 0`, `k > n`), allocation failure checks, and the `free(sorted)` on the error path were already correct and remain in place, so there are no leaks or NULL dereferences.
