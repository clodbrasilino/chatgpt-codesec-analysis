#include <stdio.h>
#include <string.h>

int is_length_even(const char *word, size_t max_len) {
    if (word == NULL) {
        return 0;
    }
    
    size_t length = 0;
    while (length < max_len && word[length] != '\0') {
        length++;
    }
    
    if (length % 2 == 0) {
        return 1;
    } else {
        return 0;
    }
}

int main(void) {
    const char *test_words[] = {"hello", "word", "C", "programming", NULL};
    size_t max_len = 256;
    
    for (int i = 0; test_words[i] != NULL; i++) {
        if (is_length_even(test_words[i], max_len)) {
            printf("The length of '%s' is even.\n", test_words[i]);
        } else {
            printf("The length of '%s' is odd.\n", test_words[i]);
        }
    }
    
    return 0;
}
