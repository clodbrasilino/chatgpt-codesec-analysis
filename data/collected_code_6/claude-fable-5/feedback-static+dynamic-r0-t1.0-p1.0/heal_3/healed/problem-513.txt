#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INSERT_LEN 4096U
#define NUMBUF_SIZE 32U

static void cleanup_partial(char **list, size_t count)
{
    size_t j;

    if (list == NULL) {
        return;
    }
    for (j = 0U; j < count; j++) {
        free(list[j]);
    }
    free(list);
}

static char *dup_buffer(const char *src, size_t len)
{
    char *dst;

    if (src == NULL) {
        return NULL;
    }
    if (len >= SIZE_MAX) {
        return NULL;
    }
    dst = (char *)malloc(len + 1U);
    if (dst == NULL) {
        return NULL;
    }
    memcpy(dst, src, len);
    dst[len] = '\0';
    return dst;
}

char **tuple_to_list(const int *tuple, size_t tuple_len, const char *insert_str, size_t *out_len)
{
    char **list = NULL;
    size_t total = 0U;
    size_t insert_len = 0U;
    size_t i;

    if ((tuple == NULL) || (insert_str == NULL) || (out_len == NULL) || (tuple_len == 0U)) {
        return NULL;
    }

    if (tuple_len > (SIZE_MAX / 2U)) {
        return NULL;
    }
    total = tuple_len * 2U;

    if (total > (SIZE_MAX / sizeof(char *))) {
        return NULL;
    }

    insert_len = strnlen(insert_str, MAX_INSERT_LEN + 1U);
    if (insert_len > MAX_INSERT_LEN) {
        return NULL;
    }

    list = (char **)calloc(total, sizeof(char *));
    if (list == NULL) {
        return NULL;
    }

    for (i = 0U; i < tuple_len; i++) {
        char numbuf[NUMBUF_SIZE];
        int written;
        size_t num_len;
        size_t idx = i * 2U;

        written = snprintf(numbuf, sizeof(numbuf), "%d", tuple[i]);
        if ((written < 0) || ((size_t)written >= sizeof(numbuf))) {
            cleanup_partial(list, idx);
            return NULL;
        }
        num_len = (size_t)written;

        list[idx] = dup_buffer(numbuf, num_len);
        if (list[idx] == NULL) {
            cleanup_partial(list, idx);
            return NULL;
        }

        list[idx + 1U] = dup_buffer(insert_str, insert_len);
        if (list[idx + 1U] == NULL) {
            cleanup_partial(list, idx + 1U);
            return NULL;
        }
    }

    *out_len = total;
    return list;
}

void free_list(char **list, size_t len)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0U; i < len; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    int tuple[] = {1, 5, 7, 4, 6};
    size_t tuple_len = sizeof(tuple) / sizeof(tuple[0]);
    const char *insert_str = "FDF";
    char **result = NULL;
    size_t result_len = 0U;
    size_t i;

    result = tuple_to_list(tuple, tuple_len, insert_str, &result_len);
    if (result == NULL) {
        (void)fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    (void)printf("[");
    for (i = 0U; i < result_len; i++) {
        (void)printf("'%s'", result[i]);
        if (i < (result_len - 1U)) {
            (void)printf(", ");
        }
    }
    (void)printf("]\n");

    free_list(result, result_len);
    return EXIT_SUCCESS;
}

**Analysis summary of the flagged weaknesses:**

1. **`memcpy` in `dup_buffer` (CWE-120):** This was already safe in the original code — the destination is allocated with `malloc(len + 1U)` immediately before the copy of exactly `len` bytes, so the buffer always holds the source data plus the NUL terminator. In the rewrite I added defensive hardening: a `NULL` check on `src` and a guard against `len + 1U` overflowing `SIZE_MAX`, which formally eliminates any integer-wrap path into an undersized allocation.

2. **Statically-sized `char numbuf[NUMBUF_SIZE]` (CWE-119/CWE-120):** This was also already mitigated — `snprintf` is used with `sizeof(numbuf)` as the limit, and the return value is checked for both encoding errors (`written < 0`) and truncation (`written >= sizeof(numbuf)`), with proper cleanup on failure. A 32-byte buffer comfortably holds any `int` representation (max 11 chars plus NUL for 32-bit, well within bounds even for 64-bit `int`). No change was required beyond keeping those checks intact.

Both flawfinder hits were false positives in the original code; the rewrite preserves all existing bounds checks, overflow guards (`SIZE_MAX / 2U`, `SIZE_MAX / sizeof(char *)`), `strnlen`-based input length capping, and leak-free error paths via `cleanup_partial`.
