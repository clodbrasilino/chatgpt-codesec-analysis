```c
#include <stdio.h>
#include <stdlib.h>

void add_k_to_tuple(int *tuple, int size, int k) {
    for (int i = 0; i < size; i++) {
        tuple[i] += k;
    }
}

int main(void) {
    int tuple[] = {1, 2, 3, 4, 5};
    int size = sizeof(tuple) / sizeof(tuple[0]);
    int k = 10;

    add_k_to_tuple(tuple, size, k);

    for (int i = 0; i < size; i++) {
        printf("%d ", tuple[i]);
    }
    printf("\n");

    return 0;
}
```
