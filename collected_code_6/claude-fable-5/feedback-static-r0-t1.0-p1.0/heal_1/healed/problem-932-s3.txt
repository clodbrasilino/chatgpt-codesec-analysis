```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

char **remove_duplicates(const char **words, size_t count, size_t *out_count)
{
    char **result = NULL;
    size_t unique = 0;
    size_t i;
    size_t j;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    result = calloc(count, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            result[unique] = malloc(len + 1);
            if (result[unique] == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            memcpy(result[unique], words[i], len);
            result[unique][len] = '\0';
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Unique words:\n");
    for (i = 0; i < result_count; i++) {
        printf("%s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}
```

**Key changes addressing the flagged weaknesses:**

1. **`strlen` (CWE-126, over-read on non-terminated strings):** Replaced with `strnlen(words[i], MAX_WORD_LEN)`, which bounds the scan. If no terminator is found within the limit (`len >= MAX_WORD_LEN`), the entry is skipped rather than over-read.

2. **`memcpy` (CWE-120, destination size not verified):** The destination is now allocated as exactly `len + 1` bytes, and `memcpy` copies precisely `len` bytes with the null terminator written explicitly. Since `len` is derived from the bounded `strnlen` and the allocation is sized from the same value, the destination is guaranteed to hold the source data.

3. **Defensive comparison:** `strcmp` was swapped for `strncmp` with the same bound, keeping all string operations length-limited. (The stored strings are guaranteed terminated, so this is belt-and-suspenders, but it satisfies static analyzers consistently.)

The rest of the logic — null checks, cleanup on allocation failure, and ownership semantics via `free_word_list` — was already sound and is preserved.
