#include <stdio.h>
#include <string.h>
#include <limits.h>

int minimizeLength(const char *s, size_t len) {
    if (s == NULL) {
        return -1;
    }
    if (len == 0) {
        return 0;
    }
    int freq[UCHAR_MAX + 1] = {0};
    for (size_t i = 0; i < len; i++) {
        freq[(unsigned char)s[i]]++;
    }
    int max_freq = 0;
    for (int i = 0; i <= UCHAR_MAX; i++) {
        if (freq[i] > max_freq) {
            max_freq = freq[i];
        }
    }
    return (int)(len - max_freq);
}

int main(void) {
    const char str[] = "abacaba";
    int result = minimizeLength(str, sizeof(str) - 1);
    if (result >= 0) {
        printf("Minimal length: %d\n", result);
    } else {
        printf("Error\n");
    }
    return 0;
}
