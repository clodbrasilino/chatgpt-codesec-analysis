#include <stdio.h>
#include <stdlib.h>

unsigned long long eulerian_number(int n, int m) {
    if (n < 0 || m < 0 || m > n) {
        return 0;
    }
    if (m == 0 || m == n) {
        return 1;
    }

    unsigned long long **dp = (unsigned long long **)malloc((size_t)(n + 1) * sizeof(unsigned long long *));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i <= n; i++) {
        dp[i] = (unsigned long long *)malloc((size_t)(m + 1) * sizeof(unsigned long long));
        if (dp[i] == NULL) {
            for (int k = 0; k < i; k++) {
                free(dp[k]);
            }
            free(dp);
            return 0;
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= m; j++) {
            if (j == 0 || j == i) {
                dp[i][j] = 1;
            } else if (j > i) {
                dp[i][j] = 0;
            } else {
                dp[i][j] = (unsigned long long)(i - j) * dp[i - 1][j - 1] + (unsigned long long)(j + 1) * dp[i - 1][j];
            }
        }
    }

    unsigned long long result = dp[n][m];

    for (int i = 0; i <= n; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void) {
    int n = 5;
    int m = 2;

    unsigned long long result = eulerian_number(n, m);

    if (result != 0) {
        printf("%llu\n", result);
    } else {
        printf("Error calculating Eulerian number or invalid input.\n");
    }

    return 0;
}
