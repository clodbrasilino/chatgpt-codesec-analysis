```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    if (k > n) {
        free(temp);
        return -1;
    }

    memcpy(temp, arr, head_bytes);

    if (k >= n || tail_bytes > total_bytes - head_bytes + head_bytes) {
        free(temp);
        return -1;
    }

    memmove(arr, arr + k, tail_bytes);

    if ((n - k) + k != n) {
        free(temp);
        return -1;
    }

    if ((n - k) > n || head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The Flawfinder warnings flagged both `memcpy` calls as potential buffer overflow risks (CWE-120). These are heuristic warnings — Flawfinder flags every use of `memcpy` regardless of context. Reviewing the original code:

1. **First `memcpy(temp, arr, head_bytes)`**: `temp` is allocated with exactly `head_bytes`, and `head_bytes = k * sizeof(int)` where `k < n` after the modulo. The source `arr` holds `total_bytes >= head_bytes`. This was already safe, but the rewrite adds an explicit `k > n` guard before the copy to make the invariant verifiable at the call site.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`**: The destination region starts at offset `n - k` and receives `k` elements, ending exactly at `arr + n`. This was also safe by construction, but the rewrite adds a check that `head_bytes` fits in the remaining space (`total_bytes - tail_bytes`) before copying.

3. **Existing defenses retained**: the `SIZE_MAX / sizeof(int)` overflow check, the `k % n` normalization, the byte-arithmetic consistency checks, the `malloc` failure check, and the `free(temp)` on every exit path (no memory leaks).

The added guards are redundant given the arithmetic invariants, but they document and enforce the bounds assumptions the `memcpy` calls rely on, which satisfies the CWE-120 mitigation requirement ("make sure destination can always hold the source data") with explicit runtime verification rather than implicit reasoning.
