#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_SIZE 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i = 0;

    while (i < max_len && str[i] != '\0') {
        i++;
    }

    return i;
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL || max_len == 0) {
        return 0;
    }

    len = bounded_strlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    int result;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
