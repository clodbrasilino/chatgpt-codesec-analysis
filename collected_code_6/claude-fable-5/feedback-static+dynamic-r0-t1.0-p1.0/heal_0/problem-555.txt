#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long cube_sum_minus_sum(long long n, int *error)
{
    long long sum;
    long long cube_sum;

    if (error == NULL) {
        return 0;
    }

    *error = 0;

    if (n < 0 || n > 3000000LL) {
        *error = 1;
        return 0;
    }

    sum = (n * (n + 1LL)) / 2LL;

    if (sum != 0 && sum > LLONG_MAX / sum) {
        *error = 1;
        return 0;
    }

    cube_sum = sum * sum;

    return cube_sum - sum;
}

int main(void)
{
    long long n;
    long long result;
    int error;

    printf("Enter n: ");

    if (scanf("%lld", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = cube_sum_minus_sum(n, &error);

    if (error != 0) {
        fprintf(stderr, "Value out of supported range\n");
        return EXIT_FAILURE;
    }

    printf("Difference: %lld\n", result);

    return EXIT_SUCCESS;
}
