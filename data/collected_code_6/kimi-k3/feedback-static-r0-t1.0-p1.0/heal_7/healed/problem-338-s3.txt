#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stdint.h>
#include <stdlib.h>

int countSubstrings(const char *str, size_t max_len) {
    int count = 0;
    if (str == NULL) {
        return 0;
    }
    
    size_t n = 0;
    while (n < max_len && str[n] != '\0') {
        n++;
    }
    
    for (size_t i = 0; i < n; i++) {
        for (size_t j = i; j < n; j++) {
            if (str[i] == str[j]) {
                if (count == INT_MAX) {
                    return -1;
                }
                count++;
            }
        }
    }
    
    return count;
}

int main(void) {
    char *str = NULL;
    size_t str_size = 0;
    ssize_t len;
    
    printf("Enter a string: ");
    
    len = getline(&str, &str_size, stdin);
    
    if (len != -1) {
        if (len > 0 && str[len - 1] == '\n') {
            str[len - 1] = '\0';
            len--;
        }
        
        if (len == 0) {
            printf("Number of substrings with same first and last characters: 0\n");
        } else {
            int result = countSubstrings(str, (size_t)len);
            if (result >= 0) {
                printf("Number of substrings with same first and last characters: %d\n", result);
            } else {
                fprintf(stderr, "Error: count overflow\n");
                free(str);
                return 1;
            }
        }
        free(str);
    } else {
        fprintf(stderr, "Error reading input\n");
        free(str);
        return 1;
    }
    
    return 0;
}
