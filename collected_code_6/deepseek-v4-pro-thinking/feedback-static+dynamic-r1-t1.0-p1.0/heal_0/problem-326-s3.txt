#include <stdio.h>
#include <string.h>
#include <stddef.h>

const char *most_frequent_word(const char *words[], size_t count)
{
    const char *result = NULL;
    size_t max_count = 0;
    size_t i;

    if (words == NULL || count == 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t current_count = 0;
        size_t j;

        if (words[i] == NULL) {
            continue;
        }

        for (j = 0; j < count; j++) {
            if (words[j] != NULL && strcmp(words[i], words[j]) == 0) {
                current_count++;
            }
        }

        if (current_count > max_count) {
            max_count = current_count;
            result = words[i];
        }
    }

    return result;
}

int main(void)
{
    const char *words[] = {
        "apple",
        "banana",
        "apple",
        "orange",
        "banana",
        "apple"
    };
    size_t count = sizeof(words) / sizeof(words[0]);
    const char *result = most_frequent_word(words, count);

    if (result != NULL) {
        printf("%s\n", result);
    } else {
        printf("no words\n");
    }

    return 0;
}
