```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_whitespace(const char *input);

char *remove_whitespace(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result = NULL;
    size_t result_len = 0;
    size_t result_cap;
    size_t input_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result_cap = input_len + 1;
    result = malloc(result_cap);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:space:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t offset = (size_t)(cursor - input);
            size_t remaining;

            if (offset > input_len) {
                regfree(&regex);
                free(result);
                return NULL;
            }

            remaining = strnlen(cursor, input_len - offset);
            if (remaining > result_cap - 1 ||
                result_len > result_cap - 1 - remaining) {
                regfree(&regex);
                free(result);
                return NULL;
            }

            memcpy(result + result_len, cursor, remaining);
            result_len += remaining;
            break;
        } else if (ret != 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if (match.rm_so < 0 || match.rm_eo <= 0 ||
            match.rm_so > match.rm_eo ||
            (size_t)(cursor - input) > input_len ||
            (size_t)match.rm_eo > input_len - (size_t)(cursor - input)) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        if ((size_t)match.rm_so > result_cap - 1 ||
            result_len > result_cap - 1 - (size_t)match.rm_so) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(result + result_len, cursor, (size_t)match.rm_so);
        result_len += (size_t)match.rm_so;
        cursor += match.rm_eo;
    }

    if (result_len >= result_cap) {
        regfree(&regex);
        free(result);
        return NULL;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test = "  Hello \t World \n from  C  ";
    char *cleaned;

    cleaned = remove_whitespace(test);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to remove whitespace\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: \"%s\"\n", test) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  \"%s\"\n", cleaned) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    free(cleaned);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**First `memcpy` (CWE-120, risk 2):** The copy of `remaining` bytes to `result + result_len` is guarded immediately before the call. `remaining` is bounded via `strnlen(cursor, input_len - offset)`, and the check `remaining > result_cap - 1 || result_len > result_cap - 1 - remaining` guarantees `result_len + remaining <= result_cap - 1` without any risk of integer overflow, since the subtraction form avoids wrapping. The destination buffer of `result_cap` bytes can therefore always hold the copy plus the terminating NUL.

**Second `memcpy` (CWE-120, risk 2):** The copy of `match.rm_so` bytes is similarly guarded. The match offsets are validated (`rm_so >= 0`, `rm_so <= rm_eo`, `rm_eo` bounded by the remaining input length), and the capacity check `(size_t)match.rm_so > result_cap - 1 || result_len > result_cap - 1 - (size_t)match.rm_so` ensures the destination cannot overflow.

Additional defensive properties retained: `strnlen` with `MAX_INPUT_LEN` caps unbounded input, `NULL` input is rejected, all error paths release both the compiled regex and the allocated buffer (no leaks), the final NUL write is bounds-checked, and `printf` return values are checked in `main`. The Flawfinder hits are informational; both copies are correctly bounded, so no functional changes were required.
