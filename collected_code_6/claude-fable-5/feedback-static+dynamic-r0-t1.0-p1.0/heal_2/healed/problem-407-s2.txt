```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int pivot;
    int swap_idx;
    char tmp;
    int lo;
    int hi;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    pivot = -1;
    for (i = len - 2; i >= 0; i--) {
        if (digits[i] < digits[i + 1]) {
            pivot = i;
            break;
        }
    }

    if (pivot == -1) {
        return -1;
    }

    swap_idx = -1;
    for (i = len - 1; i > pivot; i--) {
        if (digits[i] > digits[pivot]) {
            swap_idx = i;
            break;
        }
    }

    if (swap_idx == -1) {
        return -1;
    }

    tmp = digits[pivot];
    digits[pivot] = digits[swap_idx];
    digits[swap_idx] = tmp;

    lo = pivot + 1;
    hi = len - 1;
    while (lo < hi) {
        tmp = digits[lo];
        digits[lo] = digits[hi];
        digits[hi] = tmp;
        lo++;
        hi--;
    }

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        long long res = next_bigger(inputs[i]);
        if (res == -1) {
            if (printf("%lld -> no bigger number possible\n", inputs[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("%lld -> %lld\n", inputs[i], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes on the Flawfinder finding (CWE-119/CWE-120):**

The original code was already effectively safe against the flagged weakness, and the rewrite preserves and slightly hardens those mitigations:

1. **Bounded write via `snprintf`**: The buffer is only ever written through `snprintf(digits, sizeof(digits), ...)`, which cannot overflow. `DIGITS_BUF_SIZE` (32) comfortably exceeds the maximum decimal length of a `long long` (20 chars for `-9223372036854775808` plus NUL), so truncation cannot occur for valid inputs.

2. **Truncation/error check**: The return value of `snprintf` is validated — negative values (encoding errors) and values `>= sizeof(digits)` (would-be truncation) both cause an early `-1` return. I changed the cast comparison to `(size_t)len >= sizeof(digits)` to compare in the unsigned domain rather than casting `sizeof` down to `int`, which is marginally more robust.

3. **All indexing is derived from `len`**: Every loop index (`pivot`, `swap_idx`, `lo`, `hi`, `i`) is bounded by the validated `len`, so no out-of-bounds read or write is possible.

4. **Defense in depth**: `memset` zero-initializes the buffer, the digit-validation check (`d < 0 || d > 9`) guards against corrupted buffer contents, and the `LLONG_MAX` overflow guard prevents signed integer overflow (UB) when reconstructing the result — relevant for the wraparound case where the permuted digits exceed `LLONG_MAX` (e.g., `next_bigger` of a number near the type limit).

The Flawfinder hit is therefore a low-risk informational finding; the statically-sized array is used correctly with strict bounds enforcement throughout.
