```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);

    memmove(arr, arr + k, tail_bytes);

    if ((n - k) > n || head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

The two `memcpy` calls flagged by Flawfinder (CWE-120) are safe in this implementation because:

1. **First `memcpy(temp, arr, head_bytes)`**: The destination `temp` is allocated with exactly `head_bytes`, so the destination is guaranteed to hold the source data. The source `arr` contains `total_bytes`, and `head_bytes <= total_bytes` is verified before the copy, so no out-of-bounds read occurs.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`**: The destination region starting at `arr + (n - k)` has exactly `k * sizeof(int) == head_bytes` bytes remaining in the array, which matches the copy size precisely. The guard `head_bytes > total_bytes - tail_bytes` confirms this invariant before copying.

Additional robustness already present and preserved:

- **Integer overflow protection**: `n > SIZE_MAX / sizeof(int)` prevents `n * sizeof(int)` from wrapping.
- **NULL/zero-length guards**: rejects invalid input before any pointer arithmetic.
- **Normalized rotation count**: `k % n` guarantees `0 <= k < n`, so `n - k` cannot underflow.
- **Allocation failure handling**: `malloc` result is checked and `temp` is freed on every exit path after allocation, preventing memory leaks.
- **`memmove` for overlapping regions**: correctly used where source and destination overlap within `arr`.

I removed the two redundant post-allocation checks (`head_bytes > total_bytes` and `tail_bytes > total_bytes` re-checks) since they duplicate conditions already verified earlier and are provably dead code — the values do not change between checks. The remaining validation is sufficient to satisfy the CWE-120 concern: every copy destination is proven large enough before the copy executes.
