#include <stdio.h>
#include <stdlib.h>

int count_sequences(const int *values, int values_count, int length) {
    if (values == NULL || values_count <= 0 || length <= 0) {
        return 0;
    }

    int *dp = (int *)calloc((size_t)length + 1, sizeof(int));
    int *prefix = (int *)calloc((size_t)length + 1, sizeof(int));
    if (dp == NULL || prefix == NULL) {
        free(dp);
        free(prefix);
        return 0;
    }

    dp[0] = 1;
    prefix[0] = 1;

    for (int pos = 1; pos <= length; pos++) {
        int total = 0;
        for (int i = 0; i < values_count; i++) {
            int step = values[i];
            if (step < 0) continue;
            if (pos - step >= 0) {
                total += dp[pos - step];
            }
        }
        dp[pos] = total;
        prefix[pos] = prefix[pos - 1] + total;
    }

    int result = 0;
    for (int i = 0; i <= length; i++) {
        int remaining = length - i;
        result += dp[i] * prefix[remaining];
    }

    free(dp);
    free(prefix);
    return result;
}

int main(void) {
    int values[] = {1, 2, 3};
    int values_count = sizeof(values) / sizeof(values[0]);
    int length = 4;

    int result = count_sequences(values, values_count, length);
    printf("%d\n", result);

    return 0;
}
