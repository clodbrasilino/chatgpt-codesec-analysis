#include <stdio.h>
#include <string.h>
#include <stdbool.h>

int minSwaps(char* A, const char* B) {
    int len = strlen(A);
    int count1[2] = {0}, count2[2] = {0};
    for (int i = 0; i < len; ++i) {
        count1[A[i] - '0']++;
        count2[B[i] - '0']++;
    }
    if (count1[0] != count2[0] || count1[1] != count2[1]) return -1;

    int swapCount = 0;
    bool swapped[len];
    memset(swapped, false, sizeof(swapped));

    for (int i = 0; i < len; ++i) {
        if (A[i] != B[i] && !swapped[i]) {
            for (int j = i + 1; j < len; ++j) {
                if (A[j] != B[j] && !swapped[j] && A[i] != A[j]) {
                    char temp = A[i];
                    A[i] = A[j];
                    A[j] = temp;
                    swapCount++;
                    swapped[i] = true;
                    swapped[j] = true;
                    break;
                }
            }
        }
    }

    for (int i = 0; i < len; ++i) {
        if (A[i] != B[i]) return -1;
    }

    return swapCount;
}

int main() {
    char A[] = "100110";
    const char B[] = "010001";
    int result = minSwaps(A, B);
    if (result != -1) {
        printf("Minimum number of swaps: %d\n", result);
    } else {
        printf("Binary strings are not compatible.\n");
    }
    return 0;
}
