#include <stdio.h>
#include <stdlib.h>

int checkProductEquality(int num) {
    int evenProduct = 1;
    int oddProduct = 1;
    int position = 1;
    int digit;
    int absNum = abs(num);
    
    if (absNum == 0) {
        return 1;
    }
    
    while (absNum > 0) {
        digit = absNum % 10;
        if (position % 2 == 0) {
            evenProduct *= digit;
        } else {
            oddProduct *= digit;
        }
        absNum /= 10;
        position++;
    }
    
    return (evenProduct == oddProduct);
}

int main(void) {
    int number;
    int result;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    result = checkProductEquality(number);
    
    if (result) {
        printf("Product of digits at even and odd places are equal\n");
    } else {
        printf("Product of digits at even and odd places are not equal\n");
    }
    
    return 0;
}
