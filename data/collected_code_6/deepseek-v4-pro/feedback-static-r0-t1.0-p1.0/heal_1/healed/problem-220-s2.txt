#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void replace_chars(char *str, int n) {
    int count = 0;
    char *p = str;
    
    while (*p && count < n) {
        if (*p == ' ' || *p == ',' || *p == '.') {
            *p = ':';
            count++;
        }
        p++;
    }
}

int main(void) {
    char *str = NULL;
    size_t len = 0;
    int n;
    
    printf("Enter a string: ");
    if (getline(&str, &len, stdin) == -1) {
        free(str);
        return 1;
    }
    str[strcspn(str, "\n")] = '\0';
    
    printf("Enter maximum replacements: ");
    if (scanf("%d", &n) != 1 || n < 0) {
        free(str);
        return 1;
    }
    
    replace_chars(str, n);
    printf("Result: %s\n", str);
    
    free(str);
    return 0;
}
