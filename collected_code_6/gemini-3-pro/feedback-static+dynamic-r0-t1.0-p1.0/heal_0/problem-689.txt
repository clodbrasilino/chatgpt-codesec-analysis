#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int minJumps(int arr[], int n) {
    if (n <= 1) {
        return 0;
    }
    if (arr[0] == 0) {
        return -1;
    }

    int maxReach = arr[0];
    int step = arr[0];
    int jumps = 1;
    int i;

    for (i = 1; i < n; i++) {
        if (i == n - 1) {
            return jumps;
        }

        if (i + arr[i] > maxReach) {
            maxReach = i + arr[i];
        }

        step--;

        if (step == 0) {
            jumps++;
            if (i >= maxReach) {
                return -1;
            }
            step = maxReach - i;
        }
    }

    return -1;
}

int main() {
    int arr[] = {1, 3, 5, 8, 9, 2, 6, 7, 6, 8, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    int jumps;

    jumps = minJumps(arr, n);

    if (jumps != -1) {
        printf("%d\n", jumps);
    } else {
        printf("-1\n");
    }

    return 0;
}
