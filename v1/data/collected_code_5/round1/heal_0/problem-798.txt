```C
#include <stdio.h>

int array_sum(int *arr, int size) {
    int sum = 0;

    if (!arr || size <= 0) {
        return sum;
    }

    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }

    return sum;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int size = sizeof(arr) / sizeof(arr[0]);

    int sum = array_sum(arr, size);
    printf("Sum of array is %d\n", sum);
    
    return 0;
}
```
