```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = (char)tolower((unsigned char)snake[i]);
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    const char *input = "hello_world_example";
    char *result;

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\n", input) < 0) {
        free(result);
        return EXIT_FAILURE;
    }
    if (printf("Output: %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weakness (CWE-126, buffer over-read):**

1. **Replaced `strlen` with `strnlen`**: The original code's `strlen` call would read indefinitely past the buffer boundary if given a non-null-terminated string. `strnlen(snake, MAX_INPUT_LEN)` bounds the scan to a defined maximum, preventing an unbounded over-read.

2. **Added a length sanity check**: If `strnlen` returns `MAX_INPUT_LEN`, the string is either too long or not properly terminated within the bound, so the function fails safely by returning `NULL` rather than processing potentially truncated or malformed data.

3. **Everything else preserved**: The original code was already careful in other respects — checked `malloc` return, cast to `unsigned char` before `toupper`/`tolower` (avoiding UB on negative `char` values), null-checked the input, checked `printf` return values, and freed memory on all paths — so those defensive patterns remain intact.

Note that `strnlen` is POSIX (and available in glibc, musl, and MSVC as `strnlen`/`strnlen_s`); if strict C89/C99 portability is required, a small manual bounded-scan loop could replace it.
