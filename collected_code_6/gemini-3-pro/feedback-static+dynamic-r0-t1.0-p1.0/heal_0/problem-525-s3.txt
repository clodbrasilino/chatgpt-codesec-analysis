#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define EPSILON 1e-9

typedef struct {
    double x;
    double y;
} Point;

typedef struct {
    Point p1;
    Point p2;
} Line;

int are_lines_parallel(Line l1, Line l2) {
    double dx1 = l1.p2.x - l1.p1.x;
    double dy1 = l1.p2.y - l1.p1.y;
    double dx2 = l2.p2.x - l2.p1.x;
    double dy2 = l2.p2.y - l2.p1.y;

    double cross_product = (dx1 * dy2) - (dy1 * dx2);

    if (fabs(cross_product) < EPSILON) {
        return 1;
    }
    return 0;
}

int main(void) {
    Line line1 = {{0.0, 0.0}, {2.0, 2.0}};
    Line line2 = {{1.0, 0.0}, {3.0, 2.0}};
    Line line3 = {{0.0, 0.0}, {2.0, 1.0}};

    if (are_lines_parallel(line1, line2)) {
        printf("Line 1 and Line 2 are parallel.\n");
    } else {
        printf("Line 1 and Line 2 are not parallel.\n");
    }

    if (are_lines_parallel(line1, line3)) {
        printf("Line 1 and Line 3 are parallel.\n");
    } else {
        printf("Line 1 and Line 3 are not parallel.\n");
    }

    return EXIT_SUCCESS;
}
