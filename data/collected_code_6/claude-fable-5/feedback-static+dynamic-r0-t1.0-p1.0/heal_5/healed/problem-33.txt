```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define BINARY_BUFFER_SIZE (CHAR_BIT * sizeof(long) + 2)

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size);

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size)
{
    size_t index;
    size_t start;
    size_t end;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    index = 0;

    if (decimal < 0) {
        buffer[index] = '-';
        index++;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    start = index;

    if (value == 0UL) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index] = '0';
        index++;
    } else {
        while (value > 0UL) {
            if (index + 1 >= buffer_size) {
                return -1;
            }
            buffer[index] = (char)('0' + (char)(value % 2UL));
            index++;
            value /= 2UL;
        }
    }

    buffer[index] = '\0';

    end = index - 1;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    long decimal = 0;
    char binary[BINARY_BUFFER_SIZE] = {0};
    int scan_result;

    printf("Enter a decimal number: ");

    scan_result = scanf("%ld", &decimal);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(decimal, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary representation: %s\n", binary);

    return EXIT_SUCCESS;
}
```

Notes on the changes made relative to the flagged findings:

- **`unsignedLessThanZero` / redundant `index >= limit` before writing `'-'`**: Since `buffer_size >= 2` is already guaranteed by the guard clause, writing the sign character at index 0 can never overflow, so the misleading check (which triggered the analyzer warning) was removed. The `limit` variable was eliminated entirely; bounds checks now use `index + 1 >= buffer_size`, which directly guarantees room for both the digit being written and the terminating NUL, making the final pre-NUL check unnecessary as well.
- **`Assignment 'index=0'` warning**: `index` is now declared uninitialized and assigned only after the argument validation, so the assignment is clearly meaningful rather than a dead store before an early return.
- **Flawfinder CWE-119/CWE-120 (statically-sized array)**: This is a low-risk informational finding. The buffer is sized as `CHAR_BIT * sizeof(long) + 2` (all bits, sign, NUL — the worst case for `LONG_MIN`), the size is passed explicitly via `sizeof(binary)`, and every write is bounds-checked, so the mitigation Flawfinder asks for is already in place.
- **`LONG_MIN` negation**: The idiom `(unsigned long)(-(decimal + 1)) + 1UL` was kept, as it correctly avoids signed overflow UB when `decimal == LONG_MIN`.
