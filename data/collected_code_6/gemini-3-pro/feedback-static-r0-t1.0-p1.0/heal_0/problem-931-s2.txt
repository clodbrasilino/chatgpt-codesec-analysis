#include <stdio.h>
#include <stdlib.h>

unsigned long long calculate_cube_sum(unsigned int n) {
    unsigned long long sum = 0;
    for (unsigned int i = 1; i <= n; i++) {
        unsigned long long current = (unsigned long long)i;
        sum += current * current * current;
    }
    return sum;
}

int main(void) {
    int input;
    unsigned int n;

    if (scanf("%d", &input) != 1) {
        return EXIT_FAILURE;
    }

    if (input < 0) {
        return EXIT_FAILURE;
    }

    n = (unsigned int)input;

    unsigned long long result = calculate_cube_sum(n);
    printf("%llu\n", result);

    return EXIT_SUCCESS;
}
