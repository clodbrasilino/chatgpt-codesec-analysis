```c
#include <stdio.h>
#include <stdlib.h>

#define PI 3.14159265358979323846

static double cylinder_surface_area(double radius, double height)
{
    return (2.0 * PI * radius * (radius + height));
}

int main(void)
{
    double radius = 0.0;
    double height = 0.0;
    double area = 0.0;

    printf("Enter the radius of the cylinder: ");
    if (scanf("%lf", &radius) != 1) {
        fprintf(stderr, "Error: invalid input for radius.\n");
        return EXIT_FAILURE;
    }

    if (radius < 0.0) {
        fprintf(stderr, "Error: radius must be non-negative.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the height of the cylinder: ");
    if (scanf("%lf", &height) != 1) {
        fprintf(stderr, "Error: invalid input for height.\n");
        return EXIT_FAILURE;
    }

    if (height < 0.0) {
        fprintf(stderr, "Error: height must be non-negative.\n");
        return EXIT_FAILURE;
    }

    area = cylinder_surface_area(radius, height);

    printf("Surface area of the cylinder: %.6f\n", area);

    return EXIT_SUCCESS;
}
```
