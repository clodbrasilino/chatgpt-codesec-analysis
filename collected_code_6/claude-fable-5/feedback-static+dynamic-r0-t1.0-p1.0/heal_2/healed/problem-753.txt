```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_MAX_LEN 64U

typedef struct {
    char name[NAME_MAX_LEN];
    int value;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, NAME_MAX_LEN);
    if (len >= NAME_MAX_LEN) {
        return -1;
    }

    memcpy(t->name, name, len);
    t->name[len] = '\0';
    t->value = value;

    return 0;
}

static Tuple *find_min_k_records(const Tuple *records, size_t n, size_t k)
{
    Tuple *sorted = NULL;
    Tuple *result = NULL;

    if (records == NULL || n == 0U || k == 0U || k > n) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(Tuple)) {
        return NULL;
    }

    sorted = malloc(n * sizeof(Tuple));
    if (sorted == NULL) {
        return NULL;
    }

    memcpy(sorted, records, n * sizeof(Tuple));
    qsort(sorted, n, sizeof(Tuple), compare_tuples);

    result = malloc(k * sizeof(Tuple));
    if (result == NULL) {
        free(sorted);
        return NULL;
    }

    memcpy(result, sorted, k * sizeof(Tuple));
    free(sorted);

    return result;
}

int main(void)
{
    Tuple records[5];
    size_t n = sizeof(records) / sizeof(records[0]);
    size_t k = 2U;
    Tuple *min_records = NULL;
    size_t i;

    if (set_tuple(&records[0], "Manjeet", 10) != 0 ||
        set_tuple(&records[1], "Akshat", 4) != 0 ||
        set_tuple(&records[2], "Akash", 2) != 0 ||
        set_tuple(&records[3], "Nikhil", 8) != 0 ||
        set_tuple(&records[4], "Rohit", 6) != 0) {
        if (fprintf(stderr, "Failed to initialize records\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    min_records = find_min_k_records(records, n, k);
    if (min_records == NULL) {
        if (fprintf(stderr, "Failed to find minimum records\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Minimum %zu records:\n", k) < 0) {
        free(min_records);
        return EXIT_FAILURE;
    }

    for (i = 0U; i < k; i++) {
        if (printf("(%s, %d)\n", min_records[i].name, min_records[i].value) < 0) {
            free(min_records);
            return EXIT_FAILURE;
        }
    }

    free(min_records);
    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **`strlen` over-read (CWE-126)** — Replaced `strlen(name)` with `strnlen(name, NAME_MAX_LEN)`. This bounds the read to at most `NAME_MAX_LEN` bytes, so a non-NUL-terminated input can no longer cause an unbounded over-read. The subsequent `len >= NAME_MAX_LEN` check still rejects any string that is too long (or unterminated within the limit).

2. **`memcpy` into `t->name` (CWE-120)** — Already safe: `len` is verified to be strictly less than `NAME_MAX_LEN` before the copy, and the NUL terminator is written explicitly at `t->name[len]`, which is guaranteed in bounds.

3. **Statically-sized `name` array (CWE-119/CWE-120)** — Mitigated by the explicit bounds validation in `set_tuple`; every write path into `name` is length-checked against `NAME_MAX_LEN`.

4. **`memcpy` in `find_min_k_records` (CWE-120)** — Already safe: the `n > SIZE_MAX / sizeof(Tuple)` guard prevents integer overflow in the allocation size, `k > n` is rejected, and both destination buffers are allocated with exactly the sizes being copied, with `malloc` failures handled and no memory leaks on error paths.

The only substantive code change required was the `strlen` → `strnlen` substitution; the remaining warnings were false positives already mitigated by the existing validation logic.
