```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dict;

static char *safe_strdup(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(src, max_len);
    if (len >= max_len) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static Dict *dict_create(size_t capacity)
{
    Dict *d;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(DictEntry)) {
        return NULL;
    }

    d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }

    d->entries = malloc(capacity * sizeof(DictEntry));
    if (d->entries == NULL) {
        free(d);
        return NULL;
    }

    d->size = 0;
    d->capacity = capacity;
    return d;
}

static int dict_add(Dict *d, const char *key, const char *value)
{
    char *key_copy;
    char *value_copy;

    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }

    if (d->size >= d->capacity) {
        return -1;
    }

    key_copy = safe_strdup(key, 4096);
    if (key_copy == NULL) {
        return -1;
    }

    value_copy = safe_strdup(value, 4096);
    if (value_copy == NULL) {
        free(key_copy);
        return -1;
    }

    d->entries[d->size].key = key_copy;
    d->entries[d->size].value = value_copy;
    d->size++;
    return 0;
}

static const DictEntry *dict_get_by_index(const Dict *d, size_t index)
{
    if (d == NULL || index >= d->size) {
        return NULL;
    }
    return &d->entries[index];
}

static void dict_destroy(Dict *d)
{
    size_t i;

    if (d == NULL) {
        return;
    }

    for (i = 0; i < d->size; i++) {
        free(d->entries[i].key);
        free(d->entries[i].value);
    }
    free(d->entries);
    free(d);
}

int main(void)
{
    Dict *d;
    const DictEntry *entry;
    size_t i;

    d = dict_create(8);
    if (d == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }

    if (dict_add(d, "apple", "fruit") != 0 ||
        dict_add(d, "carrot", "vegetable") != 0 ||
        dict_add(d, "salmon", "fish") != 0) {
        fprintf(stderr, "Failed to add entries\n");
        dict_destroy(d);
        return EXIT_FAILURE;
    }

    for (i = 0; i < d->size; i++) {
        entry = dict_get_by_index(d, i);
        if (entry != NULL) {
            printf("Index %zu: key=%s, value=%s\n", i, entry->key, entry->value);
        }
    }

    entry = dict_get_by_index(d, 100);
    if (entry == NULL) {
        printf("Index 100 is out of range\n");
    }

    dict_destroy(d);
    return EXIT_SUCCESS;
}
```

Analysis notes on the original code and changes made:

- **The `memcpy` warning (CWE-120) was a false positive.** The destination buffer is allocated as `malloc(len + 1)` and exactly `len` bytes are copied, followed by explicit null termination, so the destination always holds the source data. The pattern is preserved because it is already correct — `strnlen` bounds the length before allocation, which is the proper mitigation.

- **Added `max_len == 0` guard in `safe_strdup`.** Previously, `strnlen(src, 0)` would return 0, `0 >= 0` would trigger the rejection path anyway, but the explicit check makes the contract clearer and avoids relying on that subtlety.

- **Added integer overflow check in `dict_create`.** The expression `capacity * sizeof(DictEntry)` could wrap around for very large `capacity` values, resulting in an undersized allocation and subsequent heap overflow in `dict_add` (CWE-190 leading to CWE-787). The check `capacity > SIZE_MAX / sizeof(DictEntry)` prevents this.

- Everything else was already sound: bounds checking in `dict_add` (`size >= capacity`), index validation in `dict_get_by_index`, proper cleanup on partial failure (freeing `key_copy` if `value_copy` allocation fails), and complete resource release in `dict_destroy`.
