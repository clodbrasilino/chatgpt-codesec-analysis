#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long long calculate_rounded_sum(const double *numbers, size_t length) {
    if (numbers == NULL || length == 0) {
        return 0;
    }

    long long total_sum = 0;

    for (size_t i = 0; i < length; i++) {
        total_sum += (long long)round(numbers[i]);
    }

    return total_sum * (long long)length;
}

int main(void) {
    double *numbers = NULL;
    size_t length = 5;

    numbers = (double *)malloc(length * sizeof(double));
    if (numbers == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    numbers[0] = 1.2;
    numbers[1] = 2.5;
    numbers[2] = 3.8;
    numbers[3] = 4.1;
    numbers[4] = 5.9;

    long long result = calculate_rounded_sum(numbers, length);

    printf("Result: %lld\n", result);

    free(numbers);
    numbers = NULL;

    return 0;
}
