#include <stdbool.h>
#include <ctype.h>
#include <stddef.h>

bool isInteger(const char* str) {
    if (str == NULL || *str == '\0') {
        return false;
    }
    
    if (*str == '+' || *str == '-') {
        str++;
    }
    
    while (*str != '\0') {
        if (!isdigit(*str)) {
            return false;
            str++;
        }
    }
    
    return true;
}

int main() {
    const char *str = "-1234";
    if (isInteger(str)) {
        printf("The string is an integer.\n");
    } else {
        printf("The string is not an integer.\n");
    }
    
    return 0;
}
