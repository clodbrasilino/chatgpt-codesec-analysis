#include <stdio.h>
#include <string.h>
#include <limits.h>

#define BUFFER_SIZE 256

int max_occurring_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    unsigned char best = 0;
    int found = 0;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
            best = c;
            found = 1;
        }
    }

    if (!found) {
        return -1;
    }

    *result = (char)best;
    return 0;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char result;
    size_t len;

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strcspn(buffer, "\n");
    if (len < sizeof(buffer)) {
        buffer[len] = '\0';
    }

    if (max_occurring_char(buffer, &result) != 0) {
        fprintf(stderr, "No characters found in input\n");
        return 1;
    }

    printf("Maximum occurring character: %c\n", result);
    return 0;
}
