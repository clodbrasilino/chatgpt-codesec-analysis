```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef struct {
    int sign;
    unsigned long long mantissa;
    int exponent;
} Tuple;

double tuple_to_double(Tuple t) {
    if (t.mantissa == 0) {
        return 0.0;
    }
    
    double result = (double)t.mantissa;
    result *= pow(10.0, (double)t.exponent);
    
    if (t.sign < 0) {
        result = -result;
    }
    
    return result;
}

int main(void) {
    Tuple t1 = {1, 314159, -5};
    Tuple t2 = {-1, 271828, -4};
    Tuple t3 = {1, 0, 10};
    Tuple t4 = {1, 123456789, 2};
    
    printf("%f\n", tuple_to_double(t1));
    printf("%f\n", tuple_to_double(t2));
    printf("%f\n", tuple_to_double(t3));
    printf("%f\n", tuple_to_double(t4));
    
    return 0;
}
```
