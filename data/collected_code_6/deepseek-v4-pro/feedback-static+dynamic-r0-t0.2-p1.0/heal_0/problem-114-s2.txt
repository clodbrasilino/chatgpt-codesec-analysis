```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *elements;
    int size;
    int frequency;
} Tuple;

typedef struct {
    Tuple *tuples;
    int count;
} TupleList;

int tuple_equals(const Tuple *a, const Tuple *b) {
    if (a->size != b->size) return 0;
    for (int i = 0; i < a->size; i++) {
        if (a->elements[i] != b->elements[i]) return 0;
    }
    return 1;
}

void assign_frequencies(TupleList *list) {
    if (list == NULL || list->tuples == NULL || list->count <= 0) return;

    for (int i = 0; i < list->count; i++) {
        list->tuples[i].frequency = 0;
    }

    for (int i = 0; i < list->count; i++) {
        if (list->tuples[i].frequency == 0) {
            int freq = 1;
            for (int j = i + 1; j < list->count; j++) {
                if (tuple_equals(&list->tuples[i], &list->tuples[j])) {
                    freq++;
                }
            }
            for (int j = i; j < list->count; j++) {
                if (tuple_equals(&list->tuples[i], &list->tuples[j])) {
                    list->tuples[j].frequency = freq;
                }
            }
        }
    }
}

Tuple create_tuple(int *elements, int size) {
    Tuple t;
    t.size = size;
    t.elements = (int *)malloc(size * sizeof(int));
    if (t.elements != NULL) {
        memcpy(t.elements, elements, size * sizeof(int));
    }
    t.frequency = 0;
    return t;
}

void free_tuple(Tuple *t) {
    if (t != NULL && t->elements != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->size = 0;
        t->frequency = 0;
    }
}

void free_tuple_list(TupleList *list) {
    if (list == NULL) return;
    if (list->tuples != NULL) {
        for (int i = 0; i < list->count; i++) {
            free_tuple(&list->tuples[i]);
        }
        free(list->tuples);
        list->tuples = NULL;
    }
    list->count = 0;
}

int main(void) {
    int elements1[] = {1, 2, 3};
    int elements2[] = {4, 5};
    int elements3[] = {1, 2, 3};
    int elements4[] = {6};
    int elements5[] = {4, 5};

    TupleList list;
    list.count = 5;
    list.tuples = (Tuple *)malloc(list.count * sizeof(Tuple));
    if (list.tuples == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    list.tuples[0] = create_tuple(elements1, 3);
    list.tuples[1] = create_tuple(elements2, 2);
    list.tuples[2] = create_tuple(elements3, 3);
    list.tuples[3] = create_tuple(elements4, 1);
    list.tuples[4] = create_tuple(elements5, 2);

    for (int i = 0; i < list.count; i++) {
        if (list.tuples[i].elements == NULL) {
            fprintf(stderr, "Memory allocation failed for tuple %d\n", i);
            free_tuple_list(&list);
            return 1;
        }
    }

    assign_frequencies(&list);

    for (int i = 0; i < list.count; i++) {
        printf("Tuple %d: (", i);
        for (int j = 0; j < list.tuples[i].size; j++) {
            printf("%d", list.tuples[i].elements[j]);
            if (j < list.tuples[i].size - 1) printf(", ");
        }
        printf(") - Frequency: %d\n", list.tuples[i].frequency);
    }

    free_tuple_list(&list);

    return 0;
}
```
