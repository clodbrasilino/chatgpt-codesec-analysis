```c
#include <stdio.h>
#include <stdlib.h>

int countRotations(const int arr[], int n)
{
    int low;
    int high;

    if (arr == NULL || n <= 0)
    {
        return -1;
    }

    low = 0;
    high = n - 1;

    while (low < high)
    {
        int mid = low + (high - low) / 2;

        if (arr[mid] > arr[high])
        {
            low = mid + 1;
        }
        else if (arr[mid] < arr[high])
        {
            high = mid;
        }
        else
        {
            high = high - 1;
        }
    }

    return (n - low) % n;
}

int main(void)
{
    int n;
    int i;
    int rotations;
    int *arr;

    if (scanf("%d", &n) != 1 || n <= 0)
    {
        return 1;
    }

    arr = (int *)malloc((size_t)n * sizeof(int));
    if (arr == NULL)
    {
        return 1;
    }

    for (i = 0; i < n; i++)
    {
        if (scanf("%d", &arr[i]) != 1)
        {
            free(arr);
            return 1;
        }
    }

    rotations = countRotations(arr, n);

    if (rotations >= 0)
    {
        printf("%d\n", rotations);
    }

    free(arr);

    return 0;
}
```

**Summary of the fix:** The binary search correctly located the pivot (index of the minimum element), but the function returned that index directly. For an array that was rotated anti-clockwise `k` times from sorted order, the minimum sits at index `n - k`, so the actual rotation count is `k = n - index`, not `index` itself.

- For `[3, 4, 5, 1, 2]` (n=5): old code returned index `3`; the fixed code returns `(5 - 3) % 5 = 2` ✓
- For `[4, 5, 6, 7, 8, 1, 2, 3]` (n=8): old code returned index `5`; the fixed code returns `(8 - 5) % 8 = 3` ✓
- For an already-sorted array, the pivot index is `0`, and `(n - 0) % n = 0`, so the modulo also correctly handles the zero-rotation edge case.
