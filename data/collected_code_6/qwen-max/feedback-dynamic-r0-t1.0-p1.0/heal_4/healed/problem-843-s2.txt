#include <stdio.h>
#include <stdlib.h>

#define MAX_HEAP_SIZE 10000

typedef struct {
    int* array;
    int size;
} MinHeap;

void swap(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void heapify(MinHeap* heap, int i) {
    int smallest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < heap->size && heap->array[left] < heap->array[smallest])
        smallest = left;

    if (right < heap->size && heap->array[right] < heap->array[smallest])
        smallest = right;

    if (smallest != i) {
        swap(&heap->array[i], &heap->array[smallest]);
        heapify(heap, smallest);
    }
}

void buildHeap(MinHeap* heap) {
    for (int i = (heap->size / 2) - 1; i >= 0; i--)
        heapify(heap, i);
}

void insert(MinHeap* heap, int key) {
    if (heap->size == MAX_HEAP_SIZE) return;
    heap->array[heap->size] = key;
    int i = heap->size;
    heap->size++;
    while (i > 0 && heap->array[(i - 1) / 2] > heap->array[i]) {
        swap(&heap->array[(i - 1) / 2], &heap->array[i]);
        i = (i - 1) / 2;
    }
}

int extractMin(MinHeap* heap) {
    if (heap->size <= 0) return -1;
    int root = heap->array[0];
    if (heap->size == 1) {
        heap->size--;
        return root;
    }

    heap->array[0] = heap->array[heap->size - 1];
    heap->size--;
    heapify(heap, 0);

    return root;
}

int nthSuperUglyNumber(int n, int k, int primes[]) {
    int ugly[n];
    int index[k];
    for (int i = 0; i < k; i++) index[i] = 0;

    MinHeap heap;
    heap.array = (int*)malloc(sizeof(int) * MAX_HEAP_SIZE);
    heap.size = 0;

    ugly[0] = 1;
    insert(&heap, ugly[0]);

    for (int i = 1; i < n; i++) {
        ugly[i] = extractMin(&heap);
        for (int j = 0; j < k; j++) {
            while (primes[j] * ugly[index[j]] <= ugly[i]) {
                index[j]++;
            }
            insert(&heap, primes[j] * ugly[index[j]]);
        }
    }

    free(heap.array);
    return ugly[n - 1];
}

int main() {
    int n = 12;
    int k = 4;
    int primes[] = {2, 7, 13, 19};
    printf("The %dth super ugly number is: %d\n", n, nthSuperUglyNumber(n, k, primes));
    return 0;
}
