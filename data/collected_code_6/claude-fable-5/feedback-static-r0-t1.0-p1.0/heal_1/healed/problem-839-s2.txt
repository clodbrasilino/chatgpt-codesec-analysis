```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strlen(name);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;
    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Key changes addressing the Flawfinder finding (CWE-119/CWE-120):**

1. **Bounds-checked initialization** — All writes into the fixed-size `first` buffer now go through `set_tuple()`, which validates the input length against `MAX_NAME_LEN` *before* copying and guarantees null termination explicitly. This eliminates any possibility of overflow when populating the buffer.

2. **Length-limited comparison** — `strcmp` was replaced with `strncmp(..., MAX_NAME_LEN)`, so the comparison can never read past the buffer even if a string were somehow not null-terminated.

3. **Length-limited output** — `printf` now uses the `%.*s` precision specifier bounded by `MAX_NAME_LEN`, preventing an out-of-bounds read during printing under the same failure scenario.

4. **Defensive null checks preserved** — The existing `NULL`/zero-count guards in `sort_tuples` and `print_tuples` are retained, and initialization failures are now detected and reported rather than silently proceeding.

The statically-sized array itself is retained (which is fine per CWE-119/120 guidance), but every read and write against it is now explicitly bounded, satisfying the "perform bounds checking / use functions that limit length" remediation advice.
