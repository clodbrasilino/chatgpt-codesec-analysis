```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_LEN 256U

static bool contains_all_vowels(const char *str, size_t max_len)
{
    bool found[5] = { false, false, false, false, false };
    size_t i;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);

    for (i = 0U; i < len; i++) {
        int c = tolower((unsigned char)str[i]);

        switch (c) {
        case 'a':
            found[0] = true;
            break;
        case 'e':
            found[1] = true;
            break;
        case 'i':
            found[2] = true;
            break;
        case 'o':
            found[3] = true;
            break;
        case 'u':
            found[4] = true;
            break;
        default:
            break;
        }
    }

    return found[0] && found[1] && found[2] && found[3] && found[4];
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while ((ch != '\n') && (ch != EOF));
}

int main(void)
{
    char input[MAX_LEN];
    size_t len;

    (void)memset(input, 0, sizeof(input));

    (void)printf("Enter a string: ");
    (void)fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        (void)fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if ((len > 0U) && (input[len - 1U] == '\n')) {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        discard_remaining_input();
    }

    if (contains_all_vowels(input, sizeof(input))) {
        (void)printf("Accepted: the string contains all vowels\n");
    } else {
        (void)printf("Rejected: the string does not contain all vowels\n");
    }

    return EXIT_SUCCESS;
}
```

The rewritten code preserves the original's mitigations and hardens them slightly:

- **CWE-119/CWE-120 (buffer overflow on `input`)**: The buffer is fixed-size but every access is bounds-safe — `fgets` receives `sizeof(input)`, a forced NUL terminator is written at `sizeof(input) - 1U`, and `strnlen` is used instead of `strlen` so length computation can never run past the buffer even if termination were somehow lost.

- **CWE-20/CWE-120 (`getchar` loop)**: The drain loop is only entered when the line was truncated (`len == sizeof(input) - 1U`), reads into no buffer at all (characters are discarded), and terminates on both `'\n'` and `EOF`, so it cannot overflow or loop infinitely. It has been factored into `discard_remaining_input()` and restructured as a `do/while` to avoid an assignment inside the loop condition.

- **Defense in depth**: `memset` zero-initializes the buffer before use, `str == NULL` is rejected in `contains_all_vowels`, `tolower` is given an `unsigned char` cast to avoid undefined behavior on negative `char` values, and both helper functions are `static` to limit linkage scope.
