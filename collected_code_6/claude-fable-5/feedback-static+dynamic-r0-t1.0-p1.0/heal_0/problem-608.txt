#include <stdio.h>
#include <stdlib.h>

unsigned long long bell_number(unsigned int n, int *error)
{
    unsigned long long **triangle;
    unsigned long long result;
    unsigned int i;
    unsigned int j;

    *error = 0;

    if (n > 25U) {
        *error = 1;
        return 0ULL;
    }

    triangle = malloc((n + 1U) * sizeof(*triangle));
    if (triangle == NULL) {
        *error = 1;
        return 0ULL;
    }

    for (i = 0U; i <= n; i++) {
        triangle[i] = malloc((i + 1U) * sizeof(**triangle));
        if (triangle[i] == NULL) {
            for (j = 0U; j < i; j++) {
                free(triangle[j]);
            }
            free(triangle);
            *error = 1;
            return 0ULL;
        }
    }

    triangle[0][0] = 1ULL;
    for (i = 1U; i <= n; i++) {
        triangle[i][0] = triangle[i - 1U][i - 1U];
        for (j = 1U; j <= i; j++) {
            triangle[i][j] = triangle[i][j - 1U] + triangle[i - 1U][j - 1U];
        }
    }

    result = triangle[n][0];

    for (i = 0U; i <= n; i++) {
        free(triangle[i]);
    }
    free(triangle);

    return result;
}

int main(void)
{
    unsigned int n;
    unsigned long long bell;
    int error;
    int scan_result;

    printf("Enter n: ");
    scan_result = scanf("%u", &n);
    if (scan_result != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    bell = bell_number(n, &error);
    if (error != 0) {
        fprintf(stderr, "Computation failed: n too large or out of memory\n");
        return EXIT_FAILURE;
    }

    printf("Bell number B(%u) = %llu\n", n, bell);
    return EXIT_SUCCESS;
}
