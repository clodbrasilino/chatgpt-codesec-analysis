```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

static int max_sum_no_adjacent(const int grid[2][100], int n)
{
    int incl;
    int excl;
    int i;

    if (n <= 0) {
        return 0;
    }

    incl = max_int(grid[0][0], grid[1][0]);
    excl = 0;

    for (i = 1; i < n; i++) {
        const int current = max_int(grid[0][i], grid[1][i]);
        const int new_excl = max_int(incl, excl);
        incl = excl + current;
        excl = new_excl;
    }

    return max_int(incl, excl);
}

int main(void)
{
    int grid[2][100];
    int n;
    int i;
    int j;
    int result;

    printf("Enter number of columns (1 to 100): ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    if (n < 1 || n > 100) {
        fprintf(stderr, "Number of columns out of range.\n");
        return EXIT_FAILURE;
    }

    printf("Enter grid values row by row:\n");
    for (i = 0; i < 2; i++) {
        for (j = 0; j < n; j++) {
            if (scanf("%d", &grid[i][j]) != 1) {
                fprintf(stderr, "Invalid grid value.\n");
                return EXIT_FAILURE;
            }
        }
    }

    result = max_sum_no_adjacent((const int (*)[100])grid, n);
    printf("Maximum sum with no two adjacent: %d\n", result);

    return EXIT_SUCCESS;
}
```

The changes made address the `[variableScope]` weaknesses flagged by static analysis: the variables `current` and `new_excl` were previously declared at function scope but only used inside the `for` loop body. They are now declared inside the loop with the narrowest possible scope, and marked `const` since they are never modified after initialization. This improves readability, reduces the risk of accidental misuse outside the loop, and allows the compiler to reason more precisely about their lifetimes. The algorithm's logic, input validation, and overall behavior remain unchanged.
