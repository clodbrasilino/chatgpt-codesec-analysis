#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    void *data;
    struct Node *next;
} Node;

size_t count_lists(Node **array_of_lists, size_t num_lists) {
    if (array_of_lists == NULL) {
        return 0;
    }

    size_t count = 0;
    for (size_t i = 0; i < num_lists; i++) {
        if (array_of_lists[i] != NULL) {
            count++;
        }
    }
    return count;
}

void free_lists(Node **array_of_lists, size_t num_lists) {
    if (array_of_lists == NULL) {
        return;
    }

    for (size_t i = 0; i < num_lists; i++) {
        Node *current = array_of_lists[i];
        while (current != NULL) {
            Node *next = current->next;
            free(current);
            current = next;
        }
    }
    free(array_of_lists);
}

int main(void) {
    size_t num_lists = 4;
    Node **lists = (Node **)malloc(num_lists * sizeof(Node *));
    if (lists == NULL) {
        return 1;
    }

    Node *list1 = (Node *)malloc(sizeof(Node));
    Node *list2 = (Node *)malloc(sizeof(Node));
    if (list1 == NULL || list2 == NULL) {
        free(list1);
        free(list2);
        free(lists);
        return 1;
    }

    list1->data = NULL;
    list1->next = NULL;
    list2->data = NULL;
    list2->next = NULL;

    lists[0] = list1;
    lists[1] = NULL;
    lists[2] = list2;
    lists[3] = NULL;

    size_t count = count_lists(lists, num_lists);
    printf("%zu\n", count);

    free_lists(lists, num_lists);

    return 0;
}
