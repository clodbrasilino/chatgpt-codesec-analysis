#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define CHARSET_SIZE 256
#define BUFFER_SIZE 1024

int count_char_frequency(const char *str, size_t max_len, size_t freq[CHARSET_SIZE])
{
    size_t i;
    size_t len;

    if (str == NULL || freq == NULL || max_len == 0) {
        return -1;
    }

    memset(freq, 0, CHARSET_SIZE * sizeof(size_t));

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        freq[(unsigned char)str[i]]++;
    }

    return 0;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t freq[CHARSET_SIZE];
    size_t i;
    size_t len;

    memset(buffer, 0, sizeof(buffer));
    memset(freq, 0, sizeof(freq));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (count_char_frequency(buffer, sizeof(buffer), freq) != 0) {
        fprintf(stderr, "Error counting character frequency\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < CHARSET_SIZE; i++) {
        if (freq[i] > 0) {
            if (i >= 32 && i < 127) {
                printf("'%c' : %zu\n", (int)i, freq[i]);
            } else {
                printf("0x%02zX : %zu\n", i, freq[i]);
            }
        }
    }

    return EXIT_SUCCESS;
}

**Analysis summary:**

The weaknesses flagged by Flawfinder (CWE-119/CWE-120, risk 2) relate to the statically-sized arrays `buffer[BUFFER_SIZE]` and `freq[CHARSET_SIZE]`. Upon review, the original code already mitigates these risks correctly, so the rewritten code preserves those defenses rather than restructuring them:

1. **Bounded input reading** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow on read (mitigates CWE-120).

2. **Guaranteed NUL termination** — `buffer[sizeof(buffer) - 1] = '\0';` is a defensive belt-and-suspenders measure even though `fgets` already terminates.

3. **Bounded string length computation** — `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so length calculation can never run past the buffer even if termination were lost (mitigates CWE-119).

4. **Safe indexing into `freq`** — the cast `(unsigned char)str[i]` guarantees the index is in `[0, 255]`, which fits `CHARSET_SIZE`. Without the cast, a signed `char` with a negative value would cause an out-of-bounds write.

5. **Parameter validation** — `count_char_frequency` rejects `NULL` pointers and a zero `max_len` before touching memory.

6. **Explicit initialization** — both arrays are zeroed with `memset`, and `freq` is re-zeroed inside the counting function so its contract doesn't depend on caller state.
