#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

void find_triplet(int arr[], int n, int target) {
    if (n < 3) {
        printf("No triplet found\n");
        return;
    }
    
    qsort(arr, n, sizeof(int), compare);
    
    for (int i = 0; i < n - 2; i++) {
        int left = i + 1;
        int right = n - 1;
        
        while (left < right) {
            int sum = arr[i] + arr[left] + arr[right];
            
            if (sum == target) {
                printf("Triplet: %d, %d, %d\n", arr[i], arr[left], arr[right]);
                return;
            } else if (sum < target) {
                left++;
            } else {
                right--;
            }
        }
    }
    
    printf("No triplet found\n");
}

int main() {
    int arr[] = {1, 4, 45, 6, 10, 8};
    int n = sizeof(arr) / sizeof(arr[0]);
    int target = 22;
    
    find_triplet(arr, n, target);
    
    return 0;
}
