#include <stdio.h>
#include <stdlib.h>

int* find_differences(const int* list, size_t size, size_t* out_size) {
    if (list == NULL || out_size == NULL) {
        return NULL;
    }

    if (size < 2) {
        *out_size = 0;
        return NULL;
    }

    *out_size = size - 1;
    int* differences = (int*)malloc(*out_size * sizeof(int));
    if (differences == NULL) {
        *out_size = 0;
        return NULL;
    }

    for (size_t i = 0; i < *out_size; i++) {
        differences[i] = list[i + 1] - list[i];
    }

    return differences;
}

int main(void) {
    int list[] = {10, 15, 22, 30, 45};
    size_t size = sizeof(list) / sizeof(list[0]);
    size_t diff_size = 0;

    int* differences = find_differences(list, size, &diff_size);

    if (differences != NULL) {
        for (size_t i = 0; i < diff_size; i++) {
            printf("%d ", differences[i]);
        }
        printf("\n");
        free(differences);
        differences = NULL;
    }

    return 0;
}
