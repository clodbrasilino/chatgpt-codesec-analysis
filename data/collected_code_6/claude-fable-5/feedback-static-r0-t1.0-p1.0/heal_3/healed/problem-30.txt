```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_SIZE 1024

long long countSubstrings(const char *str, size_t max_len)
{
    long long counts[256] = {0};
    long long result = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);

    for (i = 0; i < len; i++) {
        counts[(unsigned char)str[i]]++;
    }

    for (i = 0; i < 256; i++) {
        result += (counts[i] * (counts[i] + 1)) / 2;
    }

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    size_t len;
    long long result;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = countSubstrings(buffer, sizeof(buffer));

    if (printf("Count of substrings: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-hardened against the Flawfinder-flagged weakness (CWE-119/CWE-120, statically-sized buffer). The mitigations retained in the rewrite are:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow, with its return value checked for read failure.
2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` acts as a defensive belt-and-suspenders measure even though `fgets` already null-terminates.
3. **Length-limited string operations** — `strnlen` is used instead of `strlen` (both in `main` and `countSubstrings`), ensuring reads never exceed the buffer bounds even if termination were somehow lost. This is why `_POSIX_C_SOURCE 200809L` is defined.
4. **Buffer initialization** — `memset` zeroes the buffer before use, avoiding any read of uninitialized memory.
5. **Safe indexing** — the cast to `(unsigned char)` before indexing `counts[]` prevents negative array indices when `char` is signed and the input contains bytes ≥ 0x80.
6. **NULL guard** — `countSubstrings` checks for a `NULL` pointer before dereferencing.
7. **Overflow-resistant counting** — `long long` accumulators keep the arithmetic `(counts[i] * (counts[i] + 1)) / 2` safe for any input fitting in the 1024-byte buffer.

No further changes were required; the code as written already addresses the identified weakness.
