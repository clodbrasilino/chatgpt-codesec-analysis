#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096

char *string_list_to_tuple(const char *list[], size_t count)
{
    size_t total_length = 3;
    size_t i;
    char *tuple;
    char *cursor;
    size_t remaining;

    if (list == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t len;

        if (list[i] == NULL) {
            return NULL;
        }

        len = strnlen(list[i], MAX_STRING_LENGTH);

        if (total_length > SIZE_MAX - (len + 4)) {
            return NULL;
        }

        total_length += len + 2;

        if (i > 0) {
            total_length += 2;
        }
    }

    tuple = malloc(total_length);
    if (tuple == NULL) {
        return NULL;
    }

    cursor = tuple;
    remaining = total_length;

    if (remaining < 2) {
        free(tuple);
        return NULL;
    }

    *cursor++ = '(';
    remaining--;

    for (i = 0; i < count; i++) {
        size_t len = strnlen(list[i], MAX_STRING_LENGTH);
        int written;

        if (i > 0) {
            if (remaining < 2) {
                free(tuple);
                return NULL;
            }
            *cursor++ = ',';
            *cursor++ = ' ';
            remaining -= 2;
        }

        if (remaining < len + 4) {
            free(tuple);
            return NULL;
        }

        written = snprintf(cursor, remaining, "'%.*s'", (int)len, list[i]);
        if (written < 0 || (size_t)written >= remaining) {
            free(tuple);
            return NULL;
        }

        cursor += (size_t)written;
        remaining -= (size_t)written;
    }

    if (remaining < 2) {
        free(tuple);
        return NULL;
    }

    *cursor++ = ')';
    *cursor = '\0';

    return tuple;
}

int main(void)
{
    const char *fruits[] = {"apple", "banana", "cherry", "date"};
    size_t count = sizeof(fruits) / sizeof(fruits[0]);
    char *result = string_list_to_tuple(fruits, count);

    if (result == NULL) {
        fprintf(stderr, "Error: failed to convert string list to tuple\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", result);

    free(result);
    result = NULL;

    return EXIT_SUCCESS;
}
