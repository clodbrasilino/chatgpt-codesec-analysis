    int count = 1; 
    int j = 0;

        if(list[i] == list[i + 1]) {
            ++count;
        } else {
            j += sprintf(&result[j], "%d%c", count, list[i]);
            count = 1;
        }
    } 
    result[j] = '\0';
    return result;
}
