```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_STR_LEN 4096

bool ends_with_number(const char *str)
{
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len == 0 || len >= MAX_STR_LEN) {
        return false;
    }

    return isdigit((unsigned char)str[len - 1]) != 0;
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "hello",
        "42",
        "",
        "abc9",
        "number one"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        if (ends_with_number(tests[i])) {
            if (printf("\"%s\" ends with a number\n", tests[i]) < 0) {
                return 1;
            }
        } else {
            if (printf("\"%s\" does not end with a number\n", tests[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

**Key changes made:**

1. **Replaced `strlen` with `strnlen`** — this addresses the Flawfinder CWE-126 warning. `strnlen` bounds the scan to `MAX_STR_LEN` bytes, so a non-NUL-terminated buffer cannot cause an unbounded over-read.

2. **Added a sanity bound check** — if `strnlen` returns `MAX_STR_LEN`, the string is either too long or unterminated within the bound, so the function safely returns `false` rather than trusting the input.

3. **Retained existing defensive measures** — the `NULL` check, the empty-string check, the `(unsigned char)` cast before `isdigit` (required to avoid undefined behavior for negative `char` values), and the `printf` return-value checks were already correct and kept intact.

Note: `strnlen` is POSIX (and C23 via Annex K as `strnlen_s`); if strict C89/C99 portability is required, a small local loop-based equivalent can be substituted.
