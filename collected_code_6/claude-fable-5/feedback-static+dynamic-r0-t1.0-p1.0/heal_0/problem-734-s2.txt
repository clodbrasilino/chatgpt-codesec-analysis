```c
#include <stdio.h>
#include <stdlib.h>

long long sumOfSubarrayProducts(const int *arr, size_t n)
{
    long long total = 0;
    long long endingHere = 0;
    size_t i;

    if (arr == NULL || n == 0) {
        return 0;
    }

    for (i = 0; i < n; i++) {
        endingHere = (endingHere + 1) * (long long)arr[i];
        total += endingHere;
    }

    return total;
}

int main(void)
{
    int arr[] = {1, 2, 3};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result;

    result = sumOfSubarrayProducts(arr, n);

    if (printf("Sum of products of all subarrays: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
