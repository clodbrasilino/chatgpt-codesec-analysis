```c
#include <stdio.h>
#include <stdlib.h>

#define ZODIAC_COUNT 12

static const char *chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return "Unknown";
    }
    return signs[index];
}

int main(void)
{
    int year;
    int result;

    printf("Enter a year: ");
    result = scanf("%d", &year);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input, expected an integer year.\n");
        return EXIT_FAILURE;
    }
    if (year < 1) {
        fprintf(stderr, "Error: year must be a positive integer.\n");
        return EXIT_FAILURE;
    }

    printf("The Chinese zodiac sign for year %d is: %s\n",
           year, chinese_zodiac(year));

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning on the statically-sized `signs` array (CWE-119/CWE-120) is a low-risk, largely false-positive finding here, and the code already contains the appropriate mitigations:

1. **Bounds checking is performed before indexing.** The `index < 0 || index >= ZODIAC_COUNT` guard guarantees `signs[index]` can never be accessed out of bounds, satisfying Flawfinder's recommendation to "perform bounds checking."

2. **Negative modulo is handled.** In C, `%` with a negative operand can yield a negative result; the `if (index < 0) index += ZODIAC_COUNT;` correction handles this defensively, even though `main` already rejects years `< 1`.

3. **No writes occur to the array.** It is declared `static const char *const`, making both the pointers and the array immutable — a buffer overflow (CWE-120) is impossible since nothing is ever copied into it.

4. **Input validation is sound.** The `scanf` return value is checked, and invalid or non-positive input results in a clean `EXIT_FAILURE` rather than undefined behavior.

No functional changes were required; the code as written already addresses the identified weakness.
