#include <stdio.h>
#include <string.h>

void remove_chars(char *str1, const char *str2) {
    char result[100] = "";
    int j;
    for (int i = 0; str1[i] != '\0'; i++) {
        for (j = 0; str2[j] != '\0' && str1[i] != str2[j]; j++);
        if (str2[j] == '\0') {
            strcat(result, &str1[i]);
        }
    }
    strcpy(str1, result);
}

int main() {
    char str1[100] = "hello world";
    const char *str2 = "lo";
    remove_chars(str1, str2);
    printf("%s\n", str1);
    return 0;
}
