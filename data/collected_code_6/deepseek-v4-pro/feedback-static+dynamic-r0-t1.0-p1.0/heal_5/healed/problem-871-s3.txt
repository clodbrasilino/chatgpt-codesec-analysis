#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int areRotations(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return 0;
    }
    
    size_t len1 = strnlen(str1, 4096);
    size_t len2 = strnlen(str2, 4096);
    
    if (len1 == 4096 || len2 == 4096) {
        return 0;
    }
    
    if (len1 == 0 && len2 == 0) {
        return 1;
    }
    
    if (len1 != len2) {
        return 0;
    }
    
    size_t tempSize = 2 * len1 + 1;
    char *temp = (char *)malloc(tempSize);
    if (temp == NULL) {
        return 0;
    }
    
    if (tempSize > len1) {
        memcpy(temp, str1, len1);
    }
    if (tempSize - len1 > len1) {
        memcpy(temp + len1, str1, len1);
    }
    if (tempSize > 0) {
        temp[tempSize - 1] = '\0';
    }
    
    int result = (strstr(temp, str2) != NULL);
    
    free(temp);
    return result;
}

int main(void) {
    const char *str1 = "ABCD";
    const char *str2 = "CDAB";
    
    if (areRotations(str1, str2)) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    
    str1 = "ABCD";
    str2 = "ACBD";
    
    if (areRotations(str1, str2)) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    
    str1 = "";
    str2 = "";
    
    if (areRotations(str1, str2)) {
        printf("True\n");
    } else {
        printf("False\n");
    }
    
    return 0;
}
