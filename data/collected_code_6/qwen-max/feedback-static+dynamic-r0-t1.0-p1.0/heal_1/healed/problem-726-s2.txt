#include <stdio.h>
#include <stdlib.h>

void multiply_adjacent_elements(const int *array, int length, int *result) {
    for (int i = 0; i < length - 1; i++) {
        result[i] = array[i] * array[i + 1];
    }
}

int main() {
    int tuple[] = {2, 3, 4, 5};
    int length = sizeof(tuple) / sizeof(tuple[0]);
    int *result = (int *)malloc((length - 1) * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    multiply_adjacent_elements(tuple, length, result);

    for (int i = 0; i < length - 1; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);
    return 0;
}
