```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    long long *buy;
    long long *sell;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k > n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    buy = malloc((size_t)(k + 1) * sizeof(long long));
    if (buy == NULL) {
        return -1;
    }

    sell = malloc((size_t)(k + 1) * sizeof(long long));
    if (sell == NULL) {
        free(buy);
        return -1;
    }

    for (j = 0; j <= k; j++) {
        buy[j] = LLONG_MIN;
        sell[j] = 0;
    }

    for (i = 0; i < n; i++) {
        for (j = 1; j <= k; j++) {
            if (sell[j - 1] - (long long)prices[i] > buy[j]) {
                buy[j] = sell[j - 1] - (long long)prices[i];
            }
            if (buy[j] != LLONG_MIN && buy[j] + (long long)prices[i] > sell[j]) {
                sell[j] = buy[j] + (long long)prices[i];
            }
        }
    }

    result = (int)sell[k];

    free(buy);
    free(sell);

    return result;
}

int main(void)
{
    int prices0[] = {0, 5, 0, 5};
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int prices2[] = {2, 4, 1};
    const int expected[] = {10, 7, 2};
    const int *tests[3];
    int sizes[3];
    int ks[3];
    int i;
    int failures;

    tests[0] = prices0;
    tests[1] = prices1;
    tests[2] = prices2;

    sizes[0] = (int)(sizeof(prices0) / sizeof(prices0[0]));
    sizes[1] = (int)(sizeof(prices1) / sizeof(prices1[0]));
    sizes[2] = (int)(sizeof(prices2) / sizeof(prices2[0]));

    ks[0] = 2;
    ks[1] = 2;
    ks[2] = 2;

    failures = 0;

    for (i = 0; i < 3; i++) {
        int profit = maxProfit(ks[i], tests[i], sizes[i]);
        if (profit < 0) {
            fprintf(stderr, "Error: memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (profit != expected[i]) {
            fprintf(stderr, "test case %d failed: expected %d, got %d\n",
                    i, expected[i], profit);
            failures++;
        } else {
            printf("test case %d passed: maximum profit with at most %d transactions: %d\n",
                   i, ks[i], profit);
        }
    }

    if (failures > 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The failure messages in the original comment (`expected 10, got 5`, `expected 7, got 0`, `expected 2, got 0`) are the classic symptoms of a broken DP recurrence in the "at most k transactions" stock problem — typically caused by one of these bugs:

1. **Sentinel overflow**: using `INT_MIN` as the "no buy yet" sentinel and computing `sell[j-1] - prices[i]` can produce signed integer overflow (undefined behavior) if the comparison logic ever adds to the sentinel. Getting `0` on cases requiring two transactions is consistent with `buy[j]` never leaving the sentinel state correctly, and `got 5` on the `{0,5,0,5}` case means only a single transaction was ever credited (i.e., `sell[j-1]` was not being consulted when opening the j-th buy).

2. **Wrong update order or wrong index** (`sell[j]` instead of `sell[j-1]` when computing `buy[j]`), which collapses all k transactions into one.

The rewrite hardens against both:

- The DP arrays are `long long` with `LLONG_MIN` as the sentinel, and `prices[i]` is promoted to `long long` before arithmetic, eliminating any overflow/UB risk near the sentinel.
- The recurrence is explicitly `buy[j] = max(buy[j], sell[j-1] - price)` followed by `sell[j] = max(sell[j], buy[j] + price)`, guarded so the sentinel is never used in addition.
- The `k > n/2` fast path (unlimited-transaction greedy), NULL/size guards, and allocation-failure cleanup are preserved.

With these fixes all three test cases produce `10`, `7`, and `2` respectively, and the program exits with `EXIT_SUCCESS`.
