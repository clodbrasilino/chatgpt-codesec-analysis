```c
#include <stdio.h>
#include <stdlib.h>

int find_kth_element(const int arr[], int size, int k) {
    if (k < 1 || k > size) {
        return -1;
    }
    return arr[k - 1];
}

int main() {
    int n, k, result;
    int *arr;
    int i;

    if (scanf("%d", &n) != 1 || n <= 0) {
        return 1;
    }

    arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        return 1;
    }

    for (i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            free(arr);
            return 1;
        }
    }

    if (scanf("%d", &k) != 1) {
        free(arr);
        return 1;
    }

    result = find_kth_element(arr, n, k);
    if (result != -1) {
        printf("%d\n", result);
    }

    free(arr);
    return 0;
}
```
