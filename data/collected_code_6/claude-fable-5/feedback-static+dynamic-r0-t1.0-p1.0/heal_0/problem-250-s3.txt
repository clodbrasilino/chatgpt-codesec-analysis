```c
#include <stdio.h>
#include <stdlib.h>

size_t count_occurrences(const int *tuple, size_t size, int element)
{
    size_t count = 0;
    size_t i;

    if (tuple == NULL) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (tuple[i] == element) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    int tuple[] = {1, 3, 5, 3, 7, 3, 9, 2, 3};
    size_t size = sizeof(tuple) / sizeof(tuple[0]);
    int target = 3;
    size_t result;

    result = count_occurrences(tuple, size, target);

    if (printf("Element %d occurs %zu times in the tuple.\n", target, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
