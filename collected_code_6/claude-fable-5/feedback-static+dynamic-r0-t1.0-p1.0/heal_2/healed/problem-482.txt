```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_LEN 128

int find_sequences(const char *input);

int find_sequences(const char *input)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor;
    int ret;
    int found;

    if (input == NULL) {
        fprintf(stderr, "Error: input string is NULL\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Error: failed to compile regex: %s\n", errbuf);
        return -1;
    }

    found = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t length;
        char *buffer;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            char errbuf[ERRBUF_LEN];
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Error: regex execution failed: %s\n", errbuf);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            fprintf(stderr, "Error: invalid match offsets\n");
            regfree(&regex);
            return -1;
        }

        length = (size_t)(match.rm_eo - match.rm_so);

        buffer = malloc(length + 1);
        if (buffer == NULL) {
            fprintf(stderr, "Error: memory allocation failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(buffer, cursor + match.rm_so, length);
        buffer[length] = '\0';
        printf("Found match: %s\n", buffer);
        free(buffer);
        found++;

        if (match.rm_eo == 0) {
            cursor++;
        } else {
            cursor += match.rm_eo;
        }
    }

    regfree(&regex);

    if (found == 0) {
        printf("No matches found.\n");
    }

    return found;
}

int main(void)
{
    const char *test_string = "HelloWorld this Is a TestString with Mixed caseWords";
    int result;

    printf("Input: %s\n", test_string);

    result = find_sequences(test_string);
    if (result < 0) {
        fprintf(stderr, "Error: sequence search failed\n");
        return EXIT_FAILURE;
    }

    printf("Total matches: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

1. **Fixed-size `buffer[MAX_MATCH_LEN]` (CWE-119/CWE-120):** Replaced the statically-sized match buffer with a heap allocation sized exactly to the match length (`length + 1`). This eliminates both the overflow risk and the silent truncation behavior of the original code, which capped matches at 255 characters. The allocation is checked for failure and freed after each use, so there are no leaks even on error paths.

2. **`memcpy` warning (CWE-120):** The `memcpy` is now provably safe because the destination is allocated with exactly `length + 1` bytes immediately before the copy of `length` bytes, followed by explicit NUL termination.

3. **`errbuf` warnings:** These were effectively false positives — `regerror` receives `sizeof(errbuf)` and guarantees NUL termination, and the code adds a defensive terminator anyway. The pattern is retained as-is since it is already correct and bounded.

4. **Preserved correctness properties:** NULL input check, regex compile/exec error reporting, invalid-offset validation, zero-length match guard against infinite loops (`rm_eo == 0` advances the cursor), and `regfree` on all exit paths all remain intact.
