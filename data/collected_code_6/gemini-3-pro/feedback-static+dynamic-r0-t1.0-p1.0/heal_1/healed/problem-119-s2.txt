#include <stdio.h>
#include <stdlib.h>

int findSingleElement(const int* arr, size_t size) {
    if (arr == NULL || size == 0 || size % 2 == 0) {
        return -1;
    }

    if (size == 1) {
        return arr[0];
    }

    if (arr[0] != arr[1]) {
        return arr[0];
    }

    if (arr[size - 1] != arr[size - 2]) {
        return arr[size - 1];
    }

    size_t low = 1;
    size_t high = size - 2;

    while (low <= high) {
        size_t mid = low + (high - low) / 2;

        if (arr[mid] != arr[mid - 1] && arr[mid] != arr[mid + 1]) {
            return arr[mid];
        }

        if ((mid % 2 == 0 && arr[mid] == arr[mid + 1]) ||
            (mid % 2 == 1 && arr[mid] == arr[mid - 1])) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    return -1; 
}

int main(void) {
    int arr[] = {1, 1, 2, 2, 3, 3, 4, 5, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    int singleElement = findSingleElement(arr, size);

    if (singleElement != -1) {
        printf("The element that appears only once is: %d\n", singleElement);
    } else {
        printf("No single element found or invalid input.\n");
    }

    return EXIT_SUCCESS;
}
