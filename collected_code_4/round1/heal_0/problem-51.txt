```C
#include <stdio.h>
#include <stdbool.h>

bool is_equilateral(int a, int b, int c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        printf("Not a valid triangle\n");
        return false;
    }
    
    if ((a == b) && (b == c)) {
        return true;
    }
    
    return false;
}

int main() {
    int a, b, c;
    
    printf("Enter the sides of the triangle: \n");
    scanf("%d%d%d", &a, &b, &c);
    
    if (is_equilateral(a, b, c)) {
        printf("The triangle is Equilateral\n");
    }
    else {
        printf("The triangle is not Equilateral\n");
    }

    return 0;
}
```
