```c
#include <stdio.h>
#include <stdlib.h>

long long countOddXorPairs(const int *arr, size_t n)
{
    long long oddCount = 0;
    long long evenCount = 0;
    size_t i;

    if (arr == NULL || n == 0U)
    {
        return 0;
    }

    for (i = 0U; i < n; i++)
    {
        if ((arr[i] & 1) != 0)
        {
            oddCount++;
        }
        else
        {
            evenCount++;
        }
    }

    return oddCount * evenCount;
}

int main(void)
{
    int values[] = { 1, 2, 3, 4, 5 };
    size_t n = sizeof(values) / sizeof(values[0]);
    long long result;

    result = countOddXorPairs(values, n);

    if (printf("Number of pairs with odd XOR: %lld\n", result) < 0)
    {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The logic works as follows: the XOR of two numbers is odd only when exactly one of them is odd (differing least significant bits). So instead of a naive O(n²) pair check, the function counts odd and even elements and multiplies the counts, giving an O(n) solution. Null pointer and zero-length inputs are handled defensively, `size_t` is used for sizes, and the return value of `printf` is checked before exiting.
