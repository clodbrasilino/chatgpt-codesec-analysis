#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <limits.h>
#include <errno.h>

#define INPUT_BUFFER_SIZE 64U

static int jacobsthal(unsigned int n, unsigned long long *result)
{
    unsigned long long prev;
    unsigned long long curr;
    unsigned int i;

    if (result == NULL) {
        return -1;
    }

    if (n == 0U) {
        *result = 0ULL;
        return 0;
    }

    if (n == 1U) {
        *result = 1ULL;
        return 0;
    }

    prev = 0ULL;
    curr = 1ULL;

    for (i = 2U; i <= n; i++) {
        unsigned long long next;

        if (prev > (ULLONG_MAX / 2ULL)) {
            return -1;
        }
        if (curr > ULLONG_MAX - (2ULL * prev)) {
            return -1;
        }
        next = curr + (2ULL * prev);
        prev = curr;
        curr = next;
    }

    *result = curr;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;
    unsigned long guard;

    guard = 0UL;
    do {
        ch = getchar();
        guard++;
        if (guard == ULONG_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;
    const char *terminator;

    if (buffer == NULL || size == 0U || size > (size_t)INT_MAX) {
        return -1;
    }

    buffer[0] = '\0';

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1U] = '\0';

    terminator = memchr(buffer, '\0', size);
    if (terminator == NULL) {
        return -1;
    }

    len = (size_t)(terminator - buffer);

    if (len == 0U) {
        return -1;
    }

    if (buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
        return 0;
    }

    if (feof(stdin)) {
        return 0;
    }

    discard_remaining_input();

    return -1;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    char *endptr;
    const char *p;
    unsigned long value;
    unsigned long long answer;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter n: ");

    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error: output failure\n");
        return EXIT_FAILURE;
    }

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    p = buffer;
    while (isspace((unsigned char)*p)) {
        p++;
    }

    if (*p == '\0') {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (*p == '-') {
        fprintf(stderr, "Error: n out of range\n");
        return EXIT_FAILURE;
    }

    if (!isdigit((unsigned char)*p) && *p != '+') {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    endptr = NULL;
    value = strtoul(p, &endptr, 10);

    if (endptr == NULL || endptr == p) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (errno == ERANGE || value > (unsigned long)UINT_MAX) {
        fprintf(stderr, "Error: n out of range\n");
        return EXIT_FAILURE;
    }

    if (jacobsthal((unsigned int)value, &answer) != 0) {
        fprintf(stderr, "Error: result overflow or invalid argument\n");
        return EXIT_FAILURE;
    }

    printf("Jacobsthal(%lu) = %llu\n", value, answer);

    return EXIT_SUCCESS;
}

The two flagged weaknesses were already mitigated in the original code, so the rewrite preserves those defenses rather than altering them:

**1. `getchar()` in a loop (CWE-120, CWE-20)** — The `discard_remaining_input()` function reads into a single `int` (no buffer to overflow), checks for both `'\n'` and `EOF` as termination conditions, and includes a `guard` counter capped at `ULONG_MAX` to guarantee the loop cannot run unbounded even on a stream that never yields `EOF`. This is a safe pattern; no change was needed.

**2. Statically-sized `char` array (CWE-119/CWE-120)** — The `buffer[INPUT_BUFFER_SIZE]` array is protected by multiple layers: it is zeroed with `memset` before use, passed to `read_line()` together with `sizeof(buffer)` so the size is never assumed, filled only via `fgets()` which enforces the length limit, explicitly NUL-terminated at `size - 1`, and verified with `memchr()` to contain a terminator before any string operation is performed. Overlong lines are detected (no trailing `'\n'` and not `EOF`) and the excess input is drained and rejected, preventing truncated-input logic errors.

Additional hardening already present and retained: `fgets` size cast is guarded against `size > INT_MAX`, `strtoul` results are validated via `errno == ERANGE`, `endptr` checks, trailing-whitespace tolerance, an explicit `UINT_MAX` range check before narrowing to `unsigned int`, rejection of negative input before parsing, and overflow-safe Jacobsthal computation using pre-checks against `ULLONG_MAX` instead of detecting wraparound after the fact.
