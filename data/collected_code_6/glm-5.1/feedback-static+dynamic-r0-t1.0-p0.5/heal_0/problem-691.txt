#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int first;
    int second;
} Tuple;

typedef struct {
    int key;
    int *values;
    int count;
    int capacity;
} Group;

typedef struct {
    Group *groups;
    int count;
    int capacity;
} GroupList;

void init_group(Group *g, int key) {
    g->key = key;
    g->capacity = 4;
    g->values = (int *)malloc(g->capacity * sizeof(int));
    g->count = 0;
}

void add_value_to_group(Group *g, int value) {
    if (g->count == g->capacity) {
        g->capacity *= 2;
        g->values = (int *)realloc(g->values, g->capacity * sizeof(int));
    }
    g->values[g->count++] = value;
}

void free_group(Group *g) {
    free(g->values);
    g->values = NULL;
    g->count = 0;
    g->capacity = 0;
}

void init_group_list(GroupList *gl) {
    gl->capacity = 4;
    gl->groups = (Group *)malloc(gl->capacity * sizeof(Group));
    gl->count = 0;
}

Group *find_group(GroupList *gl, int key) {
    for (int i = 0; i < gl->count; i++) {
        if (gl->groups[i].key == key) {
            return &gl->groups[i];
        }
    }
    return NULL;
}

Group *add_group_to_list(GroupList *gl, int key) {
    if (gl->count == gl->capacity) {
        gl->capacity *= 2;
        gl->groups = (Group *)realloc(gl->groups, gl->capacity * sizeof(Group));
    }
    Group *g = &gl->groups[gl->count++];
    init_group(g, key);
    return g;
}

void free_group_list(GroupList *gl) {
    for (int i = 0; i < gl->count; i++) {
        free_group(&gl->groups[i]);
    }
    free(gl->groups);
    gl->groups = NULL;
    gl->count = 0;
    gl->capacity = 0;
}

void group_by_second(Tuple *tuples, int tuple_count, GroupList *gl) {
    init_group_list(gl);
    for (int i = 0; i < tuple_count; i++) {
        Group *g = find_group(gl, tuples[i].second);
        if (g == NULL) {
            g = add_group_to_list(gl, tuples[i].second);
        }
        add_value_to_group(g, tuples[i].first);
    }
}

int main() {
    Tuple tuples[] = {{1, 2}, {3, 1}, {5, 2}, {7, 1}, {9, 3}};
    int tuple_count = sizeof(tuples) / sizeof(tuples[0]);

    GroupList gl;
    group_by_second(tuples, tuple_count, &gl);

    for (int i = 0; i < gl.count; i++) {
        printf("Key %d: ", gl.groups[i].key);
        for (int j = 0; j < gl.groups[i].count; j++) {
            printf("%d ", gl.groups[i].values[j]);
        }
        printf("\n");
    }

    free_group_list(&gl);

    return 0;
}
