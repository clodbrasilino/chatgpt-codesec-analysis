```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * 8U)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = fgetc(stdin);
        guard++;
        if (guard >= (size_t)INT_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);

    return ch;
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        (void)discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `fgetc` warning (CWE-120, CWE-20) in `discard_remaining_input`:**
This is a false positive. The loop reads characters one at a time without storing them into any buffer, so no overflow is possible. The function also includes a `guard` counter capped at `INT_MAX` to prevent an unbounded loop on a pathological stream, and it correctly terminates on both `'\n'` and `EOF`. No fix required.

**2. Flawfinder `char` array warning (CWE-119/CWE-120) in `main`:**
Also mitigated. The buffer is:
- Zero-initialized at declaration.
- Filled only via `fgets(input, (int)sizeof(input), stdin)`, which bounds the read to the buffer size.
- Explicitly NUL-terminated at `input[sizeof(input) - 1U]` as defense-in-depth.
- Measured with `strnlen` (bounded), never `strlen`.
- Checked for truncation: if no newline was consumed and the buffer is full, remaining input is drained and the program rejects the input rather than silently processing a truncated value.

**Additional robustness already present:**
- `binary_to_decimal` validates all pointers, rejects empty/oversized input, restricts characters to `'0'`/`'1'`, and guards against overflow via the `value > (ULONG_MAX >> 1)` check before each shift.
- The `len > (sizeof(unsigned long) * 8U)` check prevents accepting more bits than an `unsigned long` can hold.
- Return values of `fflush`, `fgets`, and the conversion function are all checked.

The code as written already addresses both flagged findings; the rewrite above simply removes the comments as requested.
