```c
#include <stdio.h>

double findMedian(double a, double b, double c) {
    if ((a - b) * (c - a) >= 0) return a;
    else if ((b - a) * (c - b) >= 0) return b;
    else return c;
}

int main() {
    double x, y, z;
    printf("Enter three numbers: ");
    scanf("%lf %lf %lf", &x, &y, &z);
    double median = findMedian(x, y, z);
    printf("The median is: %lf\n", median);
    return 0;
}
```
