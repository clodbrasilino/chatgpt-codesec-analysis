```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <errno.h>
#include <float.h>
#include <math.h>

#define NUM_BUFFER_SIZE 64

int extract_max_number(const char *input, double *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[NUM_BUFFER_SIZE];
    double max_value;
    int found;
    int ret;
    char *endptr;

    if (input == NULL || result == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "-?[0-9]+(\\.[0-9]+)?", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    found = 0;
    max_value = -DBL_MAX;

    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            break;
        }

        len = (size_t)(match.rm_eo - match.rm_so);

        if (len > 0 && len < sizeof(buffer)) {
            double value;

            memcpy(buffer, cursor + match.rm_so, len);
            buffer[len] = '\0';

            errno = 0;
            value = strtod(buffer, &endptr);
            if (errno == 0 && endptr != buffer && isfinite(value)) {
                if (!found || value > max_value) {
                    max_value = value;
                    found = 1;
                }
            }
        }

        if (match.rm_eo == 0) {
            break;
        }
        cursor += match.rm_eo;
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test = "Values: 42, -17, 3.14, 1000.5, and 999";
    double max_value;

    if (extract_max_number(test, &max_value) != 0) {
        fprintf(stderr, "No numeric value found\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum value: %g\n", max_value) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**CWE-119/CWE-120 (statically-sized array `buffer`):** The buffer is a fixed 64-byte stack array, which Flawfinder flags on principle. However, every write into it is guarded. The match length is validated with `len > 0 && len < sizeof(buffer)` before any copy occurs. Because the check uses strict less-than (`<`), there is always room for the null terminator written at `buffer[len]`. Matches longer than 63 characters are simply skipped rather than truncated or overflowed, which is the correct defensive behavior.

**CWE-120 (`memcpy` without bounds check):** Flawfinder flags all `memcpy` calls because it cannot verify bounds statically. Here the bound is explicitly enforced by the preceding `len < sizeof(buffer)` condition, and `len` is derived safely from validated regex offsets (`rm_so >= 0`, `rm_eo >= rm_so`), so the copy cannot exceed the destination.

Additional robustness already present that I preserved: null-pointer argument checks, `regcomp` failure handling, `errno`/`endptr` validation of `strtod`, `isfinite` rejection of overflow values, protection against zero-length matches causing an infinite loop (`rm_eo == 0` break), and guaranteed `regfree` before every return path after compilation. No functional changes were required — the identified weaknesses are false positives given the existing guards.
