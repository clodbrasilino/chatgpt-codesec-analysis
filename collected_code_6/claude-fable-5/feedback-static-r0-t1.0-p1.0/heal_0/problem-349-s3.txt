```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 256

bool is_binary_string(const char *str)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0U; str[i] != '\0'; i++) {
        if ((str[i] != '0') && (str[i] != '1')) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    char input[MAX_INPUT_LENGTH];
    size_t len;

    printf("Enter a string: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    len = strlen(input);
    if ((len > 0U) && (input[len - 1U] == '\n')) {
        input[len - 1U] = '\0';
    }

    if (is_binary_string(input)) {
        printf("The string is a binary string.\n");
    } else {
        printf("The string is not a binary string.\n");
    }

    return EXIT_SUCCESS;
}
```
