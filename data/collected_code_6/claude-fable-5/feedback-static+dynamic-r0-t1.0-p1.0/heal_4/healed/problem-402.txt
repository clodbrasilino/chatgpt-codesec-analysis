```c
#include <stdio.h>
#include <stdlib.h>

static long long mulmod(long long a, long long b, long long mod)
{
    long long result = 0;
    a %= mod;
    if (a < 0) {
        a += mod;
    }
    b %= mod;
    if (b < 0) {
        b += mod;
    }
    while (b > 0) {
        if (b & 1) {
            result = (result + a) % mod;
        }
        a = (a + a) % mod;
        b >>= 1;
    }
    return result;
}

long long power(long long base, long long exp, long long mod)
{
    long long result = 1 % mod;
    base %= mod;
    if (base < 0) {
        base += mod;
    }
    while (exp > 0) {
        if (exp & 1) {
            result = mulmod(result, base, mod);
        }
        exp >>= 1;
        base = mulmod(base, base, mod);
    }
    return result;
}

long long mod_inverse(long long a, long long p)
{
    return power(a, p - 2, p);
}

static long long ncr_small(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }
    if (r == 0 || r == n) {
        return 1 % p;
    }
    if (r > n - r) {
        r = n - r;
    }

    long long result = 1 % p;

    for (long long i = 0; i < r; i++) {
        result = mulmod(result, (n - i) % p, p);
        result = mulmod(result, mod_inverse((i + 1) % p, p), p);
    }

    return result;
}

long long ncr_mod_p(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }

    long long result = 1 % p;

    while (n > 0 || r > 0) {
        long long ni = n % p;
        long long ri = r % p;

        if (ri > ni) {
            return 0;
        }

        result = mulmod(result, ncr_small(ni, ri, p), p);

        n /= p;
        r /= p;
    }

    return result;
}

int main(void)
{
    long long n, r, p;

    if (scanf("%lld %lld %lld", &n, &r, &p) != 3) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0 || r < 0 || p <= 1) {
        fprintf(stderr, "Invalid values: require n >= 0, r >= 0, p > 1\n");
        return EXIT_FAILURE;
    }

    long long answer = ncr_mod_p(n, r, p);
    printf("%lld\n", answer);
    fflush(stdout);

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Portable `mulmod`**: The original relied on `unsigned __int128`, a non-standard GCC/Clang extension. On compilers without `__int128` support (e.g., MSVC, some 32-bit targets), the code fails to compile entirely — which explains the `<no output>` results across all test cases rather than wrong answers. The replacement uses binary (double-and-add) multiplication with only standard `long long` arithmetic, remaining overflow-safe for any `p` that fits in a signed 64-bit range when doubled (i.e., `p < 2^62`).

2. **Normalized inputs in `mulmod`**: Both operands are reduced and adjusted for negativity defensively, so the function is safe regardless of caller behavior.

3. **`fflush(stdout)`**: Added after `printf` to guard against output being lost in buffered environments (another common cause of "expected X, got no output" in automated test harnesses if the runtime terminates abnormally).

The underlying algorithm (Lucas' theorem with Fermat's little theorem for modular inverses) was correct and is preserved unchanged; the failures were environmental/portability issues, not logic errors.
