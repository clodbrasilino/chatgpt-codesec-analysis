#include <stdio.h>
#include <stdlib.h>

#define MAX_INPUT_LENGTH 4096U

typedef struct {
    char *items;
    size_t count;
} Tuple;

static size_t bounded_length(const char *str, size_t max_length)
{
    size_t len = 0U;

    if (str == NULL) {
        return 0U;
    }

    while ((len < max_length) && (str[len] != '\0')) {
        len++;
    }

    return len;
}

Tuple *string_to_tuple(const char *str)
{
    Tuple *tuple = NULL;
    size_t length;
    size_t i;

    if ((str == NULL) || (str[0] == '\0')) {
        return NULL;
    }

    length = bounded_length(str, MAX_INPUT_LENGTH);
    if (length == 0U) {
        return NULL;
    }

    tuple = (Tuple *)malloc(sizeof(Tuple));
    if (tuple == NULL) {
        return NULL;
    }

    tuple->items = (char *)malloc((length + 1U) * sizeof(char));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }

    for (i = 0U; i < length; i++) {
        tuple->items[i] = str[i];
    }
    tuple->items[length] = '\0';
    tuple->count = length;

    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        tuple->items = NULL;
        tuple->count = 0U;
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if ((tuple == NULL) || (tuple->items == NULL) || (tuple->count == 0U)) {
        return;
    }

    (void)fputs("[", stdout);
    for (i = 0U; i < tuple->count; i++) {
        (void)printf("'%c'", tuple->items[i]);
        if (i < (tuple->count - 1U)) {
            (void)fputs(", ", stdout);
        }
    }
    (void)fputs("]\n", stdout);
    (void)fflush(stdout);
}

static void strip_newline(char *str)
{
    size_t len;

    if (str == NULL) {
        return;
    }

    len = bounded_length(str, MAX_INPUT_LENGTH);
    while ((len > 0U) && ((str[len - 1U] == '\n') || (str[len - 1U] == '\r'))) {
        str[len - 1U] = '\0';
        len--;
    }
}

static void safe_copy(char *dest, size_t dest_size, const char *src)
{
    size_t i = 0U;

    if ((dest == NULL) || (dest_size == 0U)) {
        return;
    }

    if (src != NULL) {
        while ((i < (dest_size - 1U)) && (src[i] != '\0')) {
            dest[i] = src[i];
            i++;
        }
    }
    dest[i] = '\0';
}

static int process_string(const char *str)
{
    Tuple *tuple = string_to_tuple(str);

    if (tuple == NULL) {
        (void)fprintf(stderr, "Failed to convert string to tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(tuple);
    free_tuple(tuple);

    return EXIT_SUCCESS;
}

int main(int argc, const char *const argv[])
{
    char buffer[MAX_INPUT_LENGTH];

    (void)setvbuf(stdout, NULL, _IONBF, 0);

    if (argc > 1) {
        int i;

        for (i = 1; i < argc; i++) {
            if (argv[i] == NULL) {
                continue;
            }

            safe_copy(buffer, sizeof(buffer), argv[i]);
            strip_newline(buffer);

            if (buffer[0] == '\0') {
                continue;
            }

            if (process_string(buffer) != EXIT_SUCCESS) {
                return EXIT_FAILURE;
            }
        }
        return EXIT_SUCCESS;
    }

    while (fgets(buffer, (int)sizeof(buffer), stdin) != NULL) {
        buffer[sizeof(buffer) - 1U] = '\0';
        strip_newline(buffer);

        if (buffer[0] == '\0') {
            continue;
        }

        if (process_string(buffer) != EXIT_SUCCESS) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the identified weaknesses and how the code addresses them:

**Test case failures (`<no output>`):** The most likely cause of the original failures was output buffering — when stdout is block-buffered (as it is when piped to a test harness rather than a terminal), output could be lost if the process terminated abnormally. The rewrite addresses this in two redundant ways: `setvbuf(stdout, NULL, _IONBF, 0)` disables buffering entirely at startup, and `fflush(stdout)` is called after each tuple is printed. Additionally, input arriving via stdin often carries trailing `\n` or `\r\n`, which `strip_newline` removes before processing, so inputs like `"item1\n"` correctly produce `['i', 't', 'e', 'm', '1']` rather than including a newline character or producing mismatched output.

**Flawfinder CWE-119/CWE-120 (statically-sized buffer):** The `buffer[MAX_INPUT_LENGTH]` array is protected against overflow on every write path:
- `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to the buffer size, and the explicit `buffer[sizeof(buffer) - 1U] = '\0'` guarantees termination.
- `safe_copy` performs bounds-checked copying of `argv` strings, always leaving room for and writing the null terminator.
- `bounded_length` never scans past `MAX_INPUT_LENGTH`, so even a non-terminated string cannot cause an out-of-bounds read.

**Memory safety:** All `malloc` results are checked; the partial-allocation path in `string_to_tuple` frees the `Tuple` before returning `NULL`, and `free_tuple` nulls pointers after freeing to avoid dangling references.
