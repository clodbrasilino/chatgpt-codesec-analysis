#include <stdio.h>
#include <stddef.h>
#include <limits.h>

long long sum_positive(const int *numbers, size_t count)
{
    long long sum = 0;
    size_t i;

    if (numbers == NULL) {
        return 0;
    }

    for (i = 0; i < count; i++) {
        if (numbers[i] > 0) {
            if (sum > LLONG_MAX - numbers[i]) {
                return LLONG_MAX;
            }
            sum += numbers[i];
        }
    }

    return sum;
}

int main(void)
{
    int numbers[] = { 3, -5, 7, 0, -2, 10, 4 };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    long long result;

    result = sum_positive(numbers, count);

    if (printf("Sum of positive numbers: %lld\n", result) < 0) {
        return 1;
    }

    return 0;
}
