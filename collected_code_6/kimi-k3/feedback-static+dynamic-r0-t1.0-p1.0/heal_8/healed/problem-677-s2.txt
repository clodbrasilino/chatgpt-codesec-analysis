#include <stdio.h>
#include <stdbool.h>
#include <math.h>

bool is_valid_triangle(double a, double b, double c) {
    if (isnan(a) || isnan(b) || isnan(c)) {
        return false;
    }
    if (isinf(a) || isinf(b) || isinf(c)) {
        return false;
    }
    if (a <= 0.0 || b <= 0.0 || c <= 0.0) {
        return false;
    }
    if (a + b <= c || a + c <= b || b + c <= a) {
        return false;
    }
    if (fabs(a - b) >= c || fabs(a - c) >= b || fabs(b - c) >= a) {
        return false;
    }
    return true;
}

int main(void) {
    double a, b, c;

    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        printf("False\n");
        return 0;
    }

    if (is_valid_triangle(a, b, c)) {
        printf("True\n");
    } else {
        printf("False\n");
    }

    return 0;
}
