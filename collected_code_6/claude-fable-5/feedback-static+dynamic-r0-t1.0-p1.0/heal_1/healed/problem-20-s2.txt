```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int is_woodall(unsigned long long x)
{
    unsigned long long n;
    unsigned long long power;

    if (x < 1ULL) {
        return 0;
    }

    power = 2ULL;
    for (n = 1ULL; n < 64ULL; n++) {
        unsigned long long value;

        if (n > ULLONG_MAX / power) {
            break;
        }
        value = n * power;
        if (value - 1ULL == x) {
            return 1;
        }
        if (value - 1ULL > x) {
            break;
        }
        if (power > ULLONG_MAX / 2ULL) {
            break;
        }
        power *= 2ULL;
    }

    return 0;
}

int main(void)
{
    unsigned long long num;
    int result;

    printf("Enter a number: ");
    if (scanf("%llu", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = is_woodall(num);
    if (result == 1) {
        printf("%llu is a Woodall number\n", num);
    } else {
        printf("%llu is not a Woodall number\n", num);
    }

    return EXIT_SUCCESS;
}
```

The change addresses the identified weakness: the variable `value` is now declared inside the `for` loop body, reducing its scope to the block where it is actually used. This improves readability and makes the code less error-prone, with no impact on performance since any reasonable compiler generates identical machine code.
