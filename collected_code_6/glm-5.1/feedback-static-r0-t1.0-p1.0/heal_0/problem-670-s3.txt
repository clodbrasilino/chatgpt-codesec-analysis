#include <stdio.h>
#include <stdbool.h>

bool has_decreasing_trend(const int *seq, size_t len) {
    if (seq == NULL || len <= 1) {
        return false;
    }

    for (size_t i = 1; i < len; ++i) {
        if (seq[i] >= seq[i - 1]) {
            return false;
        }
    }

    return true;
}

int main(void) {
    int arr1[] = {5, 4, 3, 2, 1};
    int arr2[] = {1, 2, 3, 4, 5};
    int arr3[] = {5, 5, 3, 2, 1};
    int arr4[] = {5};
    int arr5[] = {10, 8, 6, 4, 2, 0, -2};

    printf("%d\n", has_decreasing_trend(arr1, sizeof(arr1) / sizeof(arr1[0])));
    printf("%d\n", has_decreasing_trend(arr2, sizeof(arr2) / sizeof(arr2[0])));
    printf("%d\n", has_decreasing_trend(arr3, sizeof(arr3) / sizeof(arr3[0])));
    printf("%d\n", has_decreasing_trend(arr4, sizeof(arr4) / sizeof(arr4[0])));
    printf("%d\n", has_decreasing_trend(arr5, sizeof(arr5) / sizeof(arr5[0])));
    printf("%d\n", has_decreasing_trend(NULL, 5));

    return 0;
}
