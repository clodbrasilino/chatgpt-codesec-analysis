```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SENTENCE_LEN 4096

int find_first_max_even_word(const char *sentence, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len = 0;
    size_t max_len = 0;
    size_t best_start = 0;
    int found = 0;

    if (sentence == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len == MAX_SENTENCE_LEN) {
        return -1;
    }

    while (i < len) {
        while (i < len && sentence[i] == ' ') {
            i++;
        }
        if (i < len) {
            size_t start = i;
            size_t word_len = 0;
            while (i < len && sentence[i] != ' ') {
                i++;
            }
            word_len = i - start;
            if ((word_len % 2 == 0) && (word_len > max_len)) {
                max_len = word_len;
                best_start = start;
                found = 1;
            }
        }
    }

    if (!found) {
        return -1;
    }

    if (max_len >= result_size) {
        return -1;
    }

    memcpy(result, sentence + best_start, max_len);
    result[max_len] = '\0';
    return 0;
}

int main(void)
{
    const char *sentence = "It is a pleasant day today";
    char result[256] = {0};

    if (find_first_max_even_word(sentence, result, sizeof(result)) == 0) {
        if (printf("First maximum length even word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No even length word found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_SENTENCE_LEN`. If the string is not null-terminated within that limit, the function bails out with `-1` instead of reading past the buffer.

2. **CWE-120 (`memcpy` overflow):** The existing guard `max_len >= result_size` already ensured the destination can hold the word plus the null terminator; this check is retained so the `memcpy` is provably safe.

3. **CWE-119/CWE-120 (statically-sized `char` array):** The buffer is now zero-initialized at declaration, and all writes into it go through the size-checked `find_first_max_even_word` which receives `sizeof(result)`, ensuring bounds are respected.
