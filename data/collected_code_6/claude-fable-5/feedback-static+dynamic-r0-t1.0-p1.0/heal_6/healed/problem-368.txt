```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int *elements;
    size_t size;
} Tuple;

Tuple *tuple_create(const int *values, size_t size)
{
    Tuple *t = NULL;
    size_t bytes = 0U;

    if (values == NULL || size == 0U) {
        return NULL;
    }

    if (size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    bytes = size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->elements = malloc(bytes);
    if (t->elements == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->elements, values, bytes);
    t->size = size;

    return t;
}

void tuple_destroy(Tuple *t)
{
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->size = 0U;
        free(t);
    }
}

Tuple *tuple_repeat(const Tuple *t, size_t n)
{
    Tuple *result = NULL;
    size_t total = 0U;
    size_t chunk_bytes = 0U;
    size_t total_bytes = 0U;
    size_t i = 0U;

    if (t == NULL || t->elements == NULL || t->size == 0U || n == 0U) {
        return NULL;
    }

    if (t->size > (SIZE_MAX / sizeof(int)) / n) {
        return NULL;
    }

    total = t->size * n;
    chunk_bytes = t->size * sizeof(int);
    total_bytes = total * sizeof(int);

    if (chunk_bytes > total_bytes) {
        return NULL;
    }

    result = malloc(sizeof(Tuple));
    if (result == NULL) {
        return NULL;
    }

    result->elements = malloc(total_bytes);
    if (result->elements == NULL) {
        free(result);
        return NULL;
    }

    for (i = 0U; i < n; i++) {
        const size_t offset = i * t->size;
        const size_t offset_bytes = offset * sizeof(int);

        if (offset > total - t->size) {
            free(result->elements);
            free(result);
            return NULL;
        }

        if (chunk_bytes > total_bytes - offset_bytes) {
            free(result->elements);
            free(result);
            return NULL;
        }

        memcpy(&result->elements[offset], t->elements, chunk_bytes);
    }

    result->size = total;

    return result;
}

void tuple_print(const Tuple *t)
{
    size_t i = 0U;

    if (t == NULL || t->elements == NULL || t->size == 0U) {
        printf("(empty)\n");
        return;
    }

    printf("(");
    for (i = 0U; i < t->size; i++) {
        printf("%d", t->elements[i]);
        if (i < t->size - 1U) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void)
{
    int values[] = {1, 2, 3};
    Tuple *original = NULL;
    Tuple *repeated = NULL;
    size_t n = 3U;

    original = tuple_create(values, sizeof(values) / sizeof(values[0]));
    if (original == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    repeated = tuple_repeat(original, n);
    if (repeated == NULL) {
        fprintf(stderr, "Failed to repeat tuple\n");
        tuple_destroy(original);
        return EXIT_FAILURE;
    }

    printf("Original tuple: ");
    tuple_print(original);
    printf("Repeated %zu times: ", n);
    tuple_print(repeated);

    tuple_destroy(original);
    tuple_destroy(repeated);

    return EXIT_SUCCESS;
}
```

The original code was already well-defended against the flagged CWE-120 weaknesses. In this rewrite I preserved all existing mitigations and added one further hardening step:

**Existing mitigations retained:**

1. **`tuple_create` memcpy:** The destination buffer is allocated with exactly `bytes = size * sizeof(int)`, and the multiplication is guarded against overflow via the `size > SIZE_MAX / sizeof(int)` check before allocation. The copy length always matches the destination capacity, so the Flawfinder warning is a false positive here.

2. **`tuple_repeat` memcpy:** The total allocation size is overflow-checked with `t->size > (SIZE_MAX / sizeof(int)) / n`, which guards both the `t->size * n` and the subsequent byte multiplication. The per-iteration `offset > total - t->size` check ensures each chunk fits within the allocated region.

**Added hardening:** In `tuple_repeat`, an explicit byte-level bounds check (`chunk_bytes > total_bytes - offset_bytes`) now verifies before every `memcpy` that the destination has sufficient remaining capacity, expressed in the same units (bytes) as the copy itself. This makes the invariant directly auditable at the copy site rather than relying solely on element-count arithmetic, which addresses the spirit of the CWE-120 finding.
