```c
#include <stdio.h>
#include <stdlib.h>

double get_positive_ratio(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return 0.0;
    }

    size_t positive_count = 0;
    for (size_t i = 0; i < size; ++i) {
        if (arr[i] > 0) {
            positive_count++;
        }
    }

    return (double)positive_count / (double)size;
}

int main(void) {
    int arr[] = {-1, 2, 3, -4, 5, 0, -7};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    double ratio = get_positive_ratio(arr, size);
    
    printf("Ratio of positive numbers: %.2f\n", ratio);

    return 0;
}
```
