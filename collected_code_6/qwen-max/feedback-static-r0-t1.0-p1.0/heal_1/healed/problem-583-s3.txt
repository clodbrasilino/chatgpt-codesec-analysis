#include <stdio.h>
#include <stdlib.h>

unsigned long long catalan_number(int n) {
    unsigned long long *c = (unsigned long long *)malloc((n + 1) * sizeof(unsigned long long));
    if (c == NULL) return 0;
    c[0] = c[1] = 1;

    for (int i = 2; i <= n; i++) {
        c[i] = 0;
        for (int j = 0; j < i; j++) {
            c[i] += c[j] * c[i - j - 1];
        }
    }

    unsigned long long result = c[n];
    free(c);
    return result;
}

int main() {
    int n = 0;
    printf("Enter a value for n: ");
    if (scanf("%d", &n) != 1 || n < 0) {
        printf("Invalid input\n");
        return 1;
    }

    unsigned long long cat_num = catalan_number(n);
    if (cat_num == 0) {
        printf("Memory allocation failed\n");
        return 1;
    }

    printf("Catalan number C(%d) is %llu\n", n, cat_num);
    return 0;
}
