```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **remove_words(char **words, size_t count, char **targets, size_t target_count, size_t *new_count)
{
    char **result;
    size_t i;
    size_t j;
    size_t kept;
    int match;

    if (words == NULL || new_count == NULL) {
        return NULL;
    }

    result = malloc(sizeof(char *) * (count > 0 ? count : 1));
    if (result == NULL) {
        return NULL;
    }

    kept = 0;
    for (i = 0; i < count; i++) {
        if (words[i] == NULL) {
            continue;
        }
        match = 0;
        for (j = 0; j < target_count; j++) {
            if (targets != NULL && targets[j] != NULL &&
                strcmp(words[i], targets[j]) == 0) {
                match = 1;
                break;
            }
        }
        if (match == 0) {
            size_t len = strlen(words[i]) + 1;
            result[kept] = malloc(len);
            if (result[kept] == NULL) {
                while (kept > 0) {
                    kept--;
                    free(result[kept]);
                }
                free(result);
                return NULL;
            }
            memcpy(result[kept], words[i], len);
            kept++;
        }
    }

    *new_count = kept;
    return result;
}

void free_word_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static int is_delim(char c)
{
    return c == ' ' || c == '\t' || c == '\r' || c == '\n' ||
           c == ',' || c == '\'' || c == '"' || c == '[' || c == ']';
}

static char *read_line(void)
{
    char *buffer;
    size_t capacity;
    size_t length;
    int c;

    capacity = 64;
    length = 0;
    buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    c = getchar();
    if (c == EOF) {
        free(buffer);
        return NULL;
    }

    while (c != EOF && c != '\n') {
        if (length + 1 >= capacity) {
            char *tmp;
            capacity *= 2;
            tmp = realloc(buffer, capacity);
            if (tmp == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = tmp;
        }
        buffer[length] = (char)c;
        length++;
        c = getchar();
    }

    buffer[length] = '\0';
    return buffer;
}

static char **split_line(char *line, size_t *count_out)
{
    char **tokens;
    size_t capacity;
    size_t count;
    char *p;

    if (line == NULL || count_out == NULL) {
        return NULL;
    }

    capacity = 8;
    count = 0;
    tokens = malloc(sizeof(char *) * capacity);
    if (tokens == NULL) {
        return NULL;
    }

    p = line;
    while (*p != '\0') {
        while (*p != '\0' && is_delim(*p)) {
            p++;
        }
        if (*p == '\0') {
            break;
        }
        if (count >= capacity) {
            char **tmp;
            capacity *= 2;
            tmp = realloc(tokens, sizeof(char *) * capacity);
            if (tmp == NULL) {
                free(tokens);
                return NULL;
            }
            tokens = tmp;
        }
        tokens[count] = p;
        count++;
        while (*p != '\0' && !is_delim(*p)) {
            p++;
        }
        if (*p != '\0') {
            *p = '\0';
            p++;
        }
    }

    *count_out = count;
    return tokens;
}

static void print_list(char **list, size_t count)
{
    size_t i;

    printf("[");
    for (i = 0; i < count; i++) {
        printf("'%s'", list[i]);
        if (i + 1 < count) {
            printf(", ");
        }
    }
    printf("]\n");
    fflush(stdout);
}

int main(void)
{
    char *words_line;
    char *targets_line;
    char **words;
    char **targets;
    size_t word_count;
    size_t target_count;
    size_t new_count;
    char **filtered;

    words_line = read_line();
    if (words_line == NULL) {
        printf("[]\n");
        fflush(stdout);
        return EXIT_SUCCESS;
    }

    targets_line = read_line();
    if (targets_line == NULL) {
        targets_line = malloc(1);
        if (targets_line == NULL) {
            printf("[]\n");
            fflush(stdout);
            free(words_line);
            return EXIT_SUCCESS;
        }
        targets_line[0] = '\0';
    }

    word_count = 0;
    target_count = 0;
    new_count = 0;

    words = split_line(words_line, &word_count);
    if (words == NULL) {
        printf("[]\n");
        fflush(stdout);
        free(words_line);
        free(targets_line);
        return EXIT_SUCCESS;
    }

    targets = split_line(targets_line, &target_count);
    if (targets == NULL) {
        target_count = 0;
    }

    filtered = remove_words(words, word_count, targets, target_count, &new_count);
    if (filtered == NULL) {
        printf("[]\n");
        fflush(stdout);
        free(words);
        free(targets);
        free(words_line);
        free(targets_line);
        return EXIT_SUCCESS;
    }

    print_list(filtered, new_count);

    free_word_list(filtered, new_count);
    free(words);
    free(targets);
    free(words_line);
    free(targets_line);
    return EXIT_SUCCESS;
}
```

The key change addressing the failing test cases: the original code depended on `getline`, a POSIX extension whose availability hinges on `_POSIX_C_SOURCE` and the `ssize_t` type from `<sys/types.h>`. On a strict C or non-POSIX toolchain this fails to compile or link, producing no output at all — consistent with all three test cases reporting `<no output>`. The rewrite replaces `getline` with a portable `read_line` implementation built on `getchar` with dynamic buffer growth, removing the POSIX dependency entirely while preserving the original filtering logic, memory-cleanup paths, and output format.
