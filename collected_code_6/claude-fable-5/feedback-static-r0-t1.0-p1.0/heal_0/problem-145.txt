#include <stdio.h>
#include <stdlib.h>

long long max_difference(const int *arr, size_t size, int *status)
{
    size_t i;
    int min_val;
    int max_val;

    if (arr == NULL || size < 2U || status == NULL) {
        if (status != NULL) {
            *status = -1;
        }
        return 0;
    }

    min_val = arr[0];
    max_val = arr[0];

    for (i = 1U; i < size; i++) {
        if (arr[i] < min_val) {
            min_val = arr[i];
        }
        if (arr[i] > max_val) {
            max_val = arr[i];
        }
    }

    *status = 0;
    return (long long)max_val - (long long)min_val;
}

int main(void)
{
    int arr[] = { 3, -7, 12, 5, 0, 25, -14, 8 };
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int status = 0;
    long long result;

    result = max_difference(arr, size, &status);

    if (status != 0) {
        if (fprintf(stderr, "Error: invalid input to max_difference\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Maximum difference: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
