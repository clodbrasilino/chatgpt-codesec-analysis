#include <stdio.h>
#include <stdlib.h>
#include <string.h>

long minAdjacentSwaps(const int *arr, size_t n)
{
    long swaps = 0;
    size_t zeroCount = 0;
    size_t i;

    if (arr == NULL || n == 0)
    {
        return 0;
    }

    for (i = 0; i < n; i++)
    {
        if (arr[i] == 0)
        {
            swaps += (long)(i - zeroCount);
            zeroCount++;
        }
    }

    return swaps;
}

int main(void)
{
    int arr[] = {1, 0, 1, 0, 1, 0, 0, 1};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long result;

    result = minAdjacentSwaps(arr, n);
    printf("%ld\n", result);

    return 0;
}
