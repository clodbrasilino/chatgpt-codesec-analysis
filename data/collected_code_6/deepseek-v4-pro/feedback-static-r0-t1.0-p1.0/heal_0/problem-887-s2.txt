```c
#include <stdio.h>

int is_odd(int number) {
    return number & 1;
}

int main(void) {
    int num;
    
    printf("Enter a number: ");
    scanf("%d", &num);
    
    if (is_odd(num)) {
        printf("%d is odd.\n", num);
    } else {
        printf("%d is even.\n", num);
    }
    
    return 0;
}
```
