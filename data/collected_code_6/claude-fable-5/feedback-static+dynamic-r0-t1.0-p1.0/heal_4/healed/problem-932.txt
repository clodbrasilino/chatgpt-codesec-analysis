```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

char **remove_duplicates(const char *const *words, size_t count, size_t *out_count);
void free_word_list(char **words, size_t count);

char **remove_duplicates(const char *const *words, size_t count, size_t *out_count)
{
    char **result;
    size_t i;
    size_t j;
    size_t unique;

    if (words == NULL || out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(count * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    unique = 0;
    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;
        char *copy;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            if (len == SIZE_MAX) {
                continue;
            }

            copy = malloc(len + 1);
            if (copy == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }

            memcpy(copy, words[i], len);
            copy[len] = '\0';

            result[unique] = copy;
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "cherry"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Original list:\n");
    for (i = 0; i < input_count; i++) {
        printf("  %s\n", input[i]);
    }

    printf("Deduplicated list:\n");
    for (i = 0; i < result_count; i++) {
        printf("  %s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged `memcpy` was already safe in the original code, since the destination buffer is allocated with `malloc(len + 1)` and exactly `len` bytes are copied, followed by explicit null termination. The rewrite preserves that guarantee while making the fix more auditable:

1. **Bounded copy invariant** — `len` is derived from `strnlen(words[i], MAX_WORD_LEN)` and rejected if it reaches `MAX_WORD_LEN`, so the source read is bounded and the destination (`len + 1` bytes) can always hold the `len` bytes copied plus the terminator. This satisfies the CWE-120 concern.

2. **Overflow guard on `len + 1`** — an explicit `len == SIZE_MAX` check was added before `malloc(len + 1)` to rule out integer wraparound, even though it's unreachable given the `MAX_WORD_LEN` bound. Defense in depth.

3. **Intermediate `copy` pointer** — the allocation is assigned to a local before being placed in `result[unique]`, which keeps the array containing only fully initialized, null-terminated strings at all times, simplifying the cleanup path reasoning.

4. **Existing protections retained** — the `SIZE_MAX / sizeof(char *)` multiplication-overflow check, NULL parameter validation, per-element NULL skips, bounded `strncmp` comparisons, and full rollback on allocation failure are all preserved.
