#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define ARRAY_SIZE 10
#define HASH_SIZE 1000

int find_first_duplicate(const int *arr, int size) {
    if (arr == NULL || size <= 0) {
        return INT_MIN;
    }

    int *hash_table = (int *)calloc(HASH_SIZE, sizeof(int));
    if (hash_table == NULL) {
        return INT_MIN;
    }

    int *first_occurrence = (int *)malloc(HASH_SIZE * sizeof(int));
    if (first_occurrence == NULL) {
        free(hash_table);
        return INT_MIN;
    }

    for (int i = 0; i < HASH_SIZE; i++) {
        first_occurrence[i] = INT_MIN;
    }

    int first_duplicate = INT_MIN;
    int first_duplicate_index = size;

    for (int i = 0; i < size; i++) {
        int index = arr[i] % HASH_SIZE;
        if (index < 0) {
            index += HASH_SIZE;
        }

        while (hash_table[index] != 0 && first_occurrence[index] != arr[i]) {
            index = (index + 1) % HASH_SIZE;
        }

        if (hash_table[index] == 0) {
            hash_table[index] = 1;
            first_occurrence[index] = arr[i];
        } else {
            if (i < first_duplicate_index) {
                first_duplicate_index = i;
                first_duplicate = arr[i];
            }
        }
    }

    free(hash_table);
    free(first_occurrence);
    return first_duplicate;
}

int main(void) {
    const int arr[ARRAY_SIZE] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3};
    int result = find_first_duplicate(arr, ARRAY_SIZE);

    if (result != INT_MIN) {
        printf("%d\n", result);
    } else {
        printf("-1\n");
    }

    return 0;
}
