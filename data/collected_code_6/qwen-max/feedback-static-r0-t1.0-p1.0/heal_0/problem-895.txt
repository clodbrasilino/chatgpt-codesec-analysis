#include <stdio.h>
#include <stdlib.h>

int max_subsequence_sum(int *arr, int n) {
    if (n == 0) return 0;
    if (n == 1) return arr[0];

    int incl = arr[0];
    int excl = 0;
    int excl_new;

    for (int i = 1; i < n; i++) {
        excl_new = (incl > excl) ? incl : excl;
        incl = excl + arr[i];
        excl = excl_new;
    }

    return ((incl > excl) ? incl : excl);
}

int main() {
    int arr[] = {5, 5, 10, 100, 10, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = max_subsequence_sum(arr, n);
    printf("Maximum sum of non-adjacent elements: %d\n", result);
    return 0;
}
