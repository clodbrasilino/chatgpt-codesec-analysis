```c
#include <stdio.h>
#include <stdlib.h>

static int print_combination(const int *arr, const int *indices, size_t k)
{
    size_t i;

    if (arr == NULL || indices == NULL) {
        return -1;
    }

    printf("{ ");
    for (i = 0; i < k; i++) {
        printf("%d ", arr[indices[i]]);
    }
    printf("}\n");

    return 0;
}

static int combinations_of_size(const int *arr, size_t n, size_t k)
{
    int *indices;
    size_t i;

    if (arr == NULL || k > n) {
        return -1;
    }

    if (k == 0) {
        printf("{ }\n");
        return 0;
    }

    indices = malloc(k * sizeof(*indices));
    if (indices == NULL) {
        return -1;
    }

    for (i = 0; i < k; i++) {
        indices[i] = (int)i;
    }

    for (;;) {
        size_t pos;

        if (print_combination(arr, indices, k) != 0) {
            free(indices);
            return -1;
        }

        pos = k;
        while (pos > 0) {
            pos--;
            if ((size_t)indices[pos] < n - k + pos) {
                indices[pos]++;
                for (i = pos + 1; i < k; i++) {
                    indices[i] = indices[i - 1] + 1;
                }
                break;
            }
            if (pos == 0) {
                free(indices);
                return 0;
            }
        }
    }
}

static int all_combinations(const int *arr, size_t n)
{
    size_t k;

    if (arr == NULL && n > 0) {
        return -1;
    }

    for (k = 0; k <= n; k++) {
        if (combinations_of_size(arr, n, k) != 0) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    int list[] = { 1, 2, 3, 4 };
    size_t n = sizeof(list) / sizeof(list[0]);

    if (all_combinations(list, n) != 0) {
        fprintf(stderr, "Failed to generate combinations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The only weakness flagged was `[variableScope]` on `pos`, which was declared at function scope but only used inside the infinite loop. The fix moves its declaration into the `for (;;)` block, minimizing its scope. The rest of the code was already sound: `malloc` results are checked, `indices` is freed on every exit path after allocation, `NULL` and boundary conditions (`k > n`, `k == 0`) are validated, and the index arithmetic uses explicit casts between `int` and `size_t` to avoid signed/unsigned comparison issues.
