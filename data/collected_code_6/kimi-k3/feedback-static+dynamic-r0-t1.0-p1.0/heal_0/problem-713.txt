```c
#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

typedef struct {
    int *data;
    size_t size;
} Tuple;

bool is_valid_tuple(const Tuple *tuple) {
    if (tuple == NULL || tuple->data == NULL || tuple->size == 0) {
        return false;
    }
    
    for (size_t i = 0; i < tuple->size; i++) {
        if (tuple->data[i] < 0) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int values1[] = {1, 2, 3, 4, 5};
    int values2[] = {1, -2, 3, 4, 5};
    int values3[] = {0, 0, 0};
    
    Tuple tuple1 = {values1, 5};
    Tuple tuple2 = {values2, 5};
    Tuple tuple3 = {values3, 3};
    Tuple tuple4 = {NULL, 0};
    
    printf("Tuple 1 valid: %d\n", is_valid_tuple(&tuple1));
    printf("Tuple 2 valid: %d\n", is_valid_tuple(&tuple2));
    printf("Tuple 3 valid: %d\n", is_valid_tuple(&tuple3));
    printf("Tuple 4 valid: %d\n", is_valid_tuple(&tuple4));
    
    return 0;
}
```
