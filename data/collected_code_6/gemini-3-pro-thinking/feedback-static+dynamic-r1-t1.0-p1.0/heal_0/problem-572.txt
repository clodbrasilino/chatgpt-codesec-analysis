#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

int process_single_list(int **list, size_t *size, int val1, int val2) {
    if (list == NULL || *list == NULL || size == NULL) {
        return -1;
    }

    size_t orig_size = *size;
    if (orig_size == 0) {
        return 0;
    }

    int *temp = malloc(orig_size * sizeof(int));
    if (temp == NULL) {
        return -1;
    }

    size_t new_size = 0;
    int seen1 = 0;
    int seen2 = 0;

    for (size_t i = 0; i < orig_size; i++) {
        int current = (*list)[i];
        int add = 1;

        if (current == val1) {
            if (seen1) {
                add = 0;
            }
            seen1 = 1;
        } else if (current == val2) {
            if (seen2) {
                add = 0;
            }
            seen2 = 1;
        }

        if (add) {
            temp[new_size++] = current;
        }
    }

    if (new_size == 0) {
        free(*list);
        free(temp);
        *list = NULL;
        *size = 0;
        return 0;
    }

    int *resized = realloc(temp, new_size * sizeof(int));
    if (resized == NULL) {
        free(temp);
        return -1;
    }

    free(*list);
    *list = resized;
    *size = new_size;

    return 0;
}

int remove_two_duplicates_from_lists(int **lists, size_t *sizes, size_t num_lists, int val1, int val2) {
    if (lists == NULL || sizes == NULL) {
        return -1;
    }

    for (size_t i = 0; i < num_lists; i++) {
        if (process_single_list(&lists[i], &sizes[i], val1, val2) != 0) {
            return -1;
        }
    }

    return 0;
}
