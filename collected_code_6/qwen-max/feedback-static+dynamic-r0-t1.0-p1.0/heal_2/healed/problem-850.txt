#include <stdio.h>
#include <stdbool.h>

bool canFormTriangle(int a, int b, int c) {
    if (a > 0 && b > 0 && c > 0 && a + b + c == 180) {
        return true;
    }
    return false;
}

int main() {
    int angle1, angle2, angle3;
    printf("Enter three angles of a triangle: ");
    scanf("%d %d %d", &angle1, &angle2, &angle3);

    if (canFormTriangle(angle1, angle2, angle3)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }

    return 0;
}
