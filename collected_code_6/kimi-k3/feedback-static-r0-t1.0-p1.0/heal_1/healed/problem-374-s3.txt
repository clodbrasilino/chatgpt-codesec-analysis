#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void swap(char *a, char *b) {
    char temp = *a;
    *a = *b;
    *b = temp;
}

void permute(char *str, size_t left, size_t right) {
    if (left == right) {
        printf("%s\n", str);
    } else {
        for (size_t i = left; i <= right; i++) {
            swap((str + left), (str + i));
            permute(str, left + 1, right);
            swap((str + left), (str + i));
        }
    }
}

int main(void) {
    char *str = NULL;
    size_t size = 0;
    ssize_t len;
    
    printf("Enter a string: ");
    len = getline(&str, &size, stdin);
    
    if (len != -1) {
        if (len > 0 && str[len - 1] == '\n') {
            str[len - 1] = '\0';
            len--;
        }
        
        if (len > 0) {
            printf("All permutations:\n");
            permute(str, 0, (size_t)len - 1);
        } else {
            printf("Empty string provided.\n");
        }
        
        free(str);
    } else {
        fprintf(stderr, "Error reading input.\n");
        free(str);
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
