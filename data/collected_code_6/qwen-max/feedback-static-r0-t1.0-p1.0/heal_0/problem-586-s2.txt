#include <stdio.h>
#include <stdlib.h>

void splitAndAdd(int *arr, int n, int k) {
    int temp[k];
    for (int i = 0; i < k; i++) {
        temp[i] = arr[i];
    }
    for (int i = 0; i < n - k; i++) {
        arr[i] = arr[i + k];
    }
    for (int i = 0; i < k; i++) {
        arr[n - k + i] = temp[i];
    }
}

int* createArray(int n) {
    int *arr = (int*)malloc(n * sizeof(int));
    if (arr == NULL) {
        exit(1);
    }
    for (int i = 0; i < n; i++) {
        arr[i] = i + 1;
    }
    return arr;
}

void printArray(int *arr, int n) {
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int n = 10;
    int k = 3;
    int *arr = createArray(n);
    printArray(arr, n);
    splitAndAdd(arr, n, k);
    printArray(arr, n);
    free(arr);
    return 0;
}
