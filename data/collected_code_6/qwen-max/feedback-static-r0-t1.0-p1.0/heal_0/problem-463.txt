#include <stdio.h>
#include <limits.h>

int maxProductSubarray(int arr[], int n) {
    int maxEndHere = arr[0];
    int minEndHere = arr[0];
    int maxSoFar = arr[0];

    for (int i = 1; i < n; i++) {
        int temp = maxEndHere;
        maxEndHere = (arr[i] > arr[i] * maxEndHere) ? arr[i] : (arr[i] * maxEndHere > arr[i] * minEndHere) ? arr[i] * maxEndHere : arr[i] * minEndHere;
        minEndHere = (arr[i] < arr[i] * temp) ? arr[i] : (arr[i] * temp < arr[i] * minEndHere) ? arr[i] * temp : arr[i] * minEndHere;

        if (maxEndHere > maxSoFar)
            maxSoFar = maxEndHere;
    }
    return maxSoFar;
}

int main() {
    int arr[] = {2, 3, -2, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("Maximum product subarray: %d\n", maxProductSubarray(arr, n));
    return 0;
}
