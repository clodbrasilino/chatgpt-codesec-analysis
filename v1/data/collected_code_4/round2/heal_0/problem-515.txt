#include <stdbool.h>
#include <stdio.h>
#include <stdlib.h>

bool isDivisibleSubset(int arr[], int n, int m)
{
    if(m==0)
        return true;

    if(n>m)
        return true;

    bool DP[m];
    for (int i=0; i<m; i++)
        DP[i] = false;
 
    for (int i=0; i<n; i++)
    {
        if (DP[0])
            return true;
 
        bool temp[m];
        for (int i=0; i<m; i++)
            temp[i] = DP[i];
 
        for (int j=0; j<m; j++)
        {
            if (DP[j] == true)
            {
                if (DP[(j+arr[i]) % m] == false)
                    temp[(j+arr[i]) % m] = true;
            }
        }
 
        for (int j=0; j<m; j++)
            DP[j] = temp[j];
 
        DP[arr[i]%m] = true;
    }
 
    return DP[0];
}

int main()
{
    int arr[] = {3, 1, 7, 5};
    int n = sizeof(arr)/sizeof(arr[0]);
    int m = 6;

    if (isDivisibleSubset(arr, n, m) == true)
       printf("Found a subset with subset sum divisble by %d\n", m);
    else
        printf("No subset with subset sum divisible by %d\n", m);
    return 0;
}
