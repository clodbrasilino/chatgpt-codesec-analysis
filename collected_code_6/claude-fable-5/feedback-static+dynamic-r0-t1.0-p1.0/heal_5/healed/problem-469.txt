```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            best = max_int(best, dp[i - 1][j - 1] - prices[j - 1]);
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

Analysis notes on what was addressed:

1. **DP recurrence correctness** — the reported failures (`expected 7, got 0`, etc.) point to the transaction DP producing zero or stale values. The rewritten inner loop computes `best` from `dp[i - 1][j - 1] - prices[j - 1]` *before* updating `dp[i][j]`, guaranteeing the "buy" state considered at day `j` reflects the best achievable balance from the previous transaction level up to the prior day. This keeps the invariant explicit and avoids any ordering ambiguity between the two updates.

2. **Guard clauses preserved** — `NULL` prices, `n <= 1`, and `k <= 0` still short-circuit to `0`, preventing out-of-bounds reads and pointless allocation.

3. **Unlimited-transaction fast path** — when `k >= n / 2`, the greedy sum of positive deltas is used, which both fixes potential over-allocation (`k` can be huge) and yields the correct answer in O(n).

4. **Allocation failure handling** — partial allocations are rolled back before returning `-1`, and `main` checks for the sentinel, so there are no leaks or dereferences of `NULL` rows.

With these fixes the program prints `7`, `4`, and `2` for the three test arrays, matching the expected results for at most two transactions.
