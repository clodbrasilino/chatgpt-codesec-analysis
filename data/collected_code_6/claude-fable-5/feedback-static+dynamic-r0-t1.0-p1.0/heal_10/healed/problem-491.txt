#include <stdio.h>
#include <math.h>
#include <stdlib.h>

double sum_of_gp(double a, double r, int n)
{
    double sum;
    double term;
    int i;

    if (n <= 0) {
        return 0.0;
    }

    if (fabs(r - 1.0) < 1e-12) {
        return a * (double)n;
    }

    sum = 0.0;
    term = a;
    for (i = 0; i < n; i++) {
        sum += term;
        term *= r;
    }

    return sum;
}

int main(void)
{
    double a;
    double r;
    int n;
    double result;

    printf("Enter the first term: ");
    if (scanf("%lf", &a) != 1) {
        fprintf(stderr, "Invalid input for first term.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the common ratio: ");
    if (scanf("%lf", &r) != 1) {
        fprintf(stderr, "Invalid input for common ratio.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the number of terms: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input for number of terms.\n");
        return EXIT_FAILURE;
    }

    if (n < 0) {
        fprintf(stderr, "Number of terms must be non-negative.\n");
        return EXIT_FAILURE;
    }

    result = sum_of_gp(a, r, n);

    if (isinf(result) || isnan(result)) {
        fprintf(stderr, "Result out of representable range.\n");
        return EXIT_FAILURE;
    }

    if (fabs(result - round(result)) < 1e-9) {
        printf("Sum of the geometric progression: %.0f\n", result);
    } else {
        printf("Sum of the geometric progression: %f\n", result);
    }

    return EXIT_SUCCESS;
}

**Analysis of the weaknesses and the fix:**

The failing test cases reveal a variable-swap bug in the tested build. Working backwards from the reported values:

- **Test 0:** expected 31 (a=1, r=2, n=5 → 1+2+4+8+16), got 6 — which is exactly what you get with r=5, n=2 → (5²−1)/4 = 6.
- **Test 1:** expected 341 (a=1, r=4, n=5), got 156 — which equals 1+5+25+125, i.e., r=5, n=4.
- **Test 2:** expected 728 (a=2, r=3, n=6), got 86 — which equals 2·(6³−1)/5, i.e., r=6, n=3.

In every case, the "got" value results from the **common ratio and number of terms being swapped** when read from input. The rewrite ensures `r` is read into the ratio variable and `n` into the term count, in the order the test harness supplies them (first term, common ratio, number of terms).

Additionally, the closed-form `pow`-based formula was replaced with an iterative accumulation. This avoids floating-point precision drift from `pow(r, n)` for integer-valued inputs, guaranteeing exact results like 31, 341, and 728 rather than near-integer approximations, while retaining the special case for r = 1 and the guards for invalid input, negative n, and non-finite results.
