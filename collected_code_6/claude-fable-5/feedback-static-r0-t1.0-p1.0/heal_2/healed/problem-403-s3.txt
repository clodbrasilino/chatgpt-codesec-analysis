```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define URL_PATTERN "^(https?|ftp)://[A-Za-z0-9.-]+(\\.[A-Za-z]{2,})(:[0-9]{1,5})?(/[A-Za-z0-9._~:/?#@!$&'()*+,;=%-]*)?$"
#define MAX_URL_LEN 2048

static void print_regex_error(int errcode, const regex_t *regex)
{
    size_t needed;
    char *errbuf;

    needed = regerror(errcode, regex, NULL, 0);
    if (needed == 0) {
        fprintf(stderr, "Regex compilation failed: unknown error\n");
        return;
    }

    errbuf = malloc(needed);
    if (errbuf == NULL) {
        fprintf(stderr, "Regex compilation failed: out of memory retrieving error message\n");
        return;
    }

    regerror(errcode, regex, errbuf, needed);
    errbuf[needed - 1] = '\0';
    fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
    free(errbuf);
}

int is_valid_url(const char *url)
{
    regex_t regex;
    int ret;
    size_t len;

    if (url == NULL) {
        return 0;
    }

    len = strnlen(url, MAX_URL_LEN + 1);
    if (len == 0 || len > MAX_URL_LEN) {
        return 0;
    }

    ret = regcomp(&regex, URL_PATTERN, REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        print_regex_error(ret, &regex);
        return -1;
    }

    ret = regexec(&regex, url, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    }

    if (ret == REG_NOMATCH) {
        return 0;
    }

    fprintf(stderr, "Regex execution failed\n");
    return -1;
}

int main(void)
{
    const char *test_urls[] = {
        "https://www.example.com",
        "http://example.com/path/to/page?query=value",
        "ftp://files.example.org:2121/downloads",
        "https://sub.domain.co.uk/index.html",
        "not_a_url",
        "http://",
        "www.example.com",
        "https://example",
        ""
    };
    size_t count = sizeof(test_urls) / sizeof(test_urls[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = is_valid_url(test_urls[i]);
        if (result < 0) {
            fprintf(stderr, "Error while validating URL: %s\n", test_urls[i]);
            return EXIT_FAILURE;
        }
        printf("%-60s : %s\n", test_urls[i], result == 1 ? "VALID" : "INVALID");
    }

    return EXIT_SUCCESS;
}
```

**Summary of the fix applied:**

The Flawfinder warning (CWE-119/CWE-120) flagged the statically-sized `char errbuf[ERRBUF_LEN]` array used with `regerror()`. While the original code was actually safe (it passed `sizeof(errbuf)` and force-terminated the string), the rewrite eliminates the fixed-size buffer entirely:

1. **Two-pass `regerror()` call** — the first call with a `NULL` buffer and size `0` returns the exact number of bytes required for the error message, so no arbitrary size limit can truncate or overflow.
2. **Dynamic allocation** — the buffer is `malloc()`'d to the exact required size, with the allocation failure handled gracefully.
3. **Defensive null-termination** — `errbuf[needed - 1] = '\0'` is retained as a belt-and-braces guard against non-conforming `regerror()` implementations.
4. **Memory hygiene** — the buffer is freed immediately after use, avoiding leaks even on the error path.

All other bounds checks from the original (`strnlen` with `MAX_URL_LEN + 1` to reject over-long URLs without unbounded scanning, NULL pointer check, proper `regfree()` after `regexec()`) were already sound and are preserved unchanged.
