#include <stdio.h>
#include <string.h>
#include <stddef.h>

#define BUFFER_SIZE 256U

static size_t bounded_strlen(const char *str, size_t max_len)
{
    const char *end;

    if (str == NULL) {
        return 0U;
    }

    end = memchr(str, '\0', max_len);
    if (end == NULL) {
        return max_len;
    }

    return (size_t)(end - str);
}

static long count_substrings(const char *str, size_t max_len)
{
    long count;
    size_t len;
    size_t i;
    size_t j;

    if (str == NULL) {
        return -1L;
    }

    len = bounded_strlen(str, max_len);
    if (len == max_len) {
        return -1L;
    }

    count = 0L;

    for (i = 0U; i < len; i++) {
        for (j = i; j < len; j++) {
            if (str[i] == str[j]) {
                count++;
            }
        }
    }

    return count;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    (void)memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        (void)fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = bounded_strlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    }

    result = count_substrings(buffer, sizeof(buffer));
    if (result < 0L) {
        (void)fprintf(stderr, "Invalid input\n");
        return 1;
    }

    if (printf("%ld\n", result) < 0) {
        return 1;
    }

    return 0;
}

Analysis of the original code and the fix status:

**The flagged weakness (CWE-119/CWE-120, statically-sized array `buffer`)** is already fully mitigated in this code, so the rewrite preserves the existing defenses rather than changing logic:

1. **Bounded input reading**: `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to the buffer size, preventing overflow — this is the primary fix for the Flawfinder warning.

2. **Guaranteed null termination**: `buffer[sizeof(buffer) - 1U] = '\0'` is defensive (technically redundant after `fgets`, but harmless and MISRA-friendly).

3. **No unbounded string functions**: `bounded_strlen` uses `memchr` with an explicit maximum instead of `strlen`, so even a non-terminated buffer cannot cause an out-of-bounds read.

4. **Sentinel checking**: `count_substrings` returns `-1L` when `len == max_len`, treating a missing terminator as invalid input rather than proceeding.

5. **Pre-initialization**: `memset(buffer, 0, sizeof(buffer))` ensures no uninitialized reads.

6. **Robust error handling**: NULL checks on pointers, checked return values from `fgets` and `printf`, and distinct exit codes.

One residual (non-security) consideration: `count` could theoretically overflow `long` only if `len` were enormous, but with `BUFFER_SIZE = 256` the maximum count is bounded by 255·256/2 ≈ 32,640, far below `LONG_MAX`, so no overflow is possible here.
