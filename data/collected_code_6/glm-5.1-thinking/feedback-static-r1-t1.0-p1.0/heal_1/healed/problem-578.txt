```c
#include <stdio.h>
#include <stdlib.h>

void interleave(const int *list1, const int *list2, size_t length, int *result) {
    if (list1 == NULL || list2 == NULL || result == NULL) {
        return;
    }
    for (size_t i = 0; i < length; i++) {
        result[i * 2] = list1[i];
        result[i * 2 + 1] = list2[i];
    }
}

int main(void) {
    const int list1[] = {1, 2, 3, 4};
    const int list2[] = {5, 6, 7, 8};
    size_t length = sizeof(list1) / sizeof(list1[0]);
    
    int *result = malloc(length * 2 * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    
    interleave(list1, list2, length, result);
    
    for (size_t i = 0; i < length * 2; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");
    
    free(result);
    return 0;
}
```
