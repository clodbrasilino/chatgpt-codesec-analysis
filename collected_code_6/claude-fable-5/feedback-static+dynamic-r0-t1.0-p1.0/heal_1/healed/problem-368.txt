```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    int *elements;
    size_t size;
} Tuple;

Tuple *tuple_create(const int *values, size_t size)
{
    Tuple *t = NULL;
    size_t bytes = 0U;

    if (values == NULL || size == 0U) {
        return NULL;
    }

    if (size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    bytes = size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->elements = malloc(bytes);
    if (t->elements == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->elements, values, bytes);
    t->size = size;

    return t;
}

void tuple_destroy(Tuple *t)
{
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->size = 0U;
        free(t);
    }
}

Tuple *tuple_repeat(const Tuple *t, size_t n)
{
    Tuple *result = NULL;
    size_t total = 0U;
    size_t chunk_bytes = 0U;
    size_t total_bytes = 0U;
    size_t i = 0U;

    if (t == NULL || t->elements == NULL || t->size == 0U || n == 0U) {
        return NULL;
    }

    if (t->size > (SIZE_MAX / sizeof(int)) / n) {
        return NULL;
    }

    total = t->size * n;
    chunk_bytes = t->size * sizeof(int);
    total_bytes = total * sizeof(int);

    result = malloc(sizeof(Tuple));
    if (result == NULL) {
        return NULL;
    }

    result->elements = malloc(total_bytes);
    if (result->elements == NULL) {
        free(result);
        return NULL;
    }

    for (i = 0U; i < n; i++) {
        if ((i * t->size) + t->size > total) {
            free(result->elements);
            free(result);
            return NULL;
        }
        memcpy(&result->elements[i * t->size], t->elements, chunk_bytes);
    }

    result->size = total;

    return result;
}

void tuple_print(const Tuple *t)
{
    size_t i = 0U;

    if (t == NULL || t->elements == NULL) {
        printf("(empty)\n");
        return;
    }

    printf("(");
    for (i = 0U; i < t->size; i++) {
        printf("%d", t->elements[i]);
        if (i < t->size - 1U) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void)
{
    int values[] = {1, 2, 3};
    Tuple *original = NULL;
    Tuple *repeated = NULL;
    size_t n = 3U;

    original = tuple_create(values, sizeof(values) / sizeof(values[0]));
    if (original == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    repeated = tuple_repeat(original, n);
    if (repeated == NULL) {
        fprintf(stderr, "Failed to repeat tuple\n");
        tuple_destroy(original);
        return EXIT_FAILURE;
    }

    printf("Original tuple: ");
    tuple_print(original);
    printf("Repeated %zu times: ", n);
    tuple_print(repeated);

    tuple_destroy(original);
    tuple_destroy(repeated);

    return EXIT_SUCCESS;
}
```

**Key fixes applied:**

1. **Missing `<stdint.h>` include** — The original code used `SIZE_MAX` in `tuple_repeat` without including `<stdint.h>`, which would fail to compile (or rely on transitive includes). This is now explicitly included.

2. **Integer overflow check in `tuple_create` (CWE-190 → CWE-120)** — The original computed `size * sizeof(int)` unchecked. If `size` were huge, the multiplication could wrap, causing a short allocation followed by a heap buffer overflow in `memcpy`. A guard `size > SIZE_MAX / sizeof(int)` now precedes the allocation, so the destination is guaranteed to hold the source data.

3. **Precomputed byte counts** — `bytes`, `chunk_bytes`, and `total_bytes` are computed once after validation, ensuring the exact same validated value is used for both `malloc` and `memcpy`, eliminating any mismatch between allocation size and copy size.

4. **Defensive bounds check in the repeat loop** — Before each `memcpy`, the write offset plus chunk length is verified against `total`, guaranteeing writes stay within the allocated buffer even if invariants were somehow violated.

5. **Guard against `t->size == 0` in `tuple_repeat`** — Prevents a division-related edge case in the overflow check and avoids zero-size allocations with undefined semantics downstream.

The `memcpy` calls themselves are retained (they are the correct tool here); the CWE-120 findings are resolved by proving, via the overflow guards and explicit bounds checks, that the destination buffers can always hold the copied data.
