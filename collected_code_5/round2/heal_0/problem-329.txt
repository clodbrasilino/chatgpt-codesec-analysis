```C
#include<stdio.h>

int count_negative(int arr[], int n) {
  int count = 0;
  for(int i = 0; i < n; i++) {
    if(arr[i] < 0)
      count++;
  }
  return count;
}

int main() {
  int arr[] = {4, -2, -8, 5, -2};
  int n = sizeof(arr) / sizeof(arr[0]);
  printf("%d", count_negative(arr, n));

  return 0;
}
```
