#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node* next;
};

struct Node* create_node(int data);
void free_list(struct Node* head);
void free_split_lists(struct Node** lists, size_t count);
void print_list(const struct Node* head);
struct Node** split_list(struct Node* head, size_t n, size_t* out_count);

struct Node* create_node(int data) {
    struct Node* node = (struct Node*)malloc(sizeof(struct Node));
    if (node != NULL) {
        node->data = data;
        node->next = NULL;
    }
    return node;
}

void free_list(struct Node* head) {
    struct Node* temp = NULL;
    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

void free_split_lists(struct Node** lists, size_t count) {
    if (lists != NULL) {
        for (size_t i = 0; i < count; i++) {
            free_list(lists[i]);
        }
        free(lists);
    }
}

void print_list(const struct Node* head) {
    const struct Node* temp = head;
    while (temp != NULL) {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

struct Node** split_list(struct Node* head, size_t n, size_t* out_count) {
    if (out_count == NULL) {
        return NULL;
    }
    if (head == NULL || n == 0) {
        *out_count = 0;
        return NULL;
    }

    size_t length = 0;
    struct Node* temp = head;
    while (temp != NULL) {
        length++;
        temp = temp->next;
    }

    size_t num_splits = (length + n - 1) / n;
    struct Node** lists = (struct Node**)malloc(num_splits * sizeof(struct Node*));
    if (lists == NULL) {
        *out_count = 0;
        return NULL;
    }

    temp = head;
    for (size_t i = 0; i < num_splits; i++) {
        lists[i] = temp;
        struct Node* prev = NULL;
        for (size_t j = 0; j < n && temp != NULL; j++) {
            prev = temp;
            temp = temp->next;
        }
        if (prev != NULL) {
            prev->next = NULL;
        }
    }

    *out_count = num_splits;
    return lists;
}

int main(void) {
    struct Node* head = NULL;
    struct Node* tail = NULL;

    for (int i = 1; i <= 10; i++) {
        struct Node* node = create_node(i);
        if (node == NULL) {
            free_list(head);
            return EXIT_FAILURE;
        }
        if (head == NULL) {
            head = node;
            tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
    }

    size_t out_count = 0;
    struct Node** lists = split_list(head, 3, &out_count);

    if (lists == NULL) {
        free_list(head);
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < out_count; i++) {
        print_list(lists[i]);
    }

    free_split_lists(lists, out_count);

    return EXIT_SUCCESS;
}
