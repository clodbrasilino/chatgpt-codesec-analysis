#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

void remove_non_alnum(char *str)
{
    size_t read_idx;
    size_t write_idx;

    if (str == NULL) {
        return;
    }

    read_idx = 0;
    write_idx = 0;

    while (str[read_idx] != '\0') {
        if (isalnum((unsigned char)str[read_idx])) {
            str[write_idx] = str[read_idx];
            write_idx++;
        }
        read_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[256];
    const char *input = "Hello, World! 123 #$% foo_bar";
    size_t input_len;
    int written;

    input_len = strnlen(input, sizeof(buffer));
    if (input_len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, sizeof(buffer), "%s", input);
    if (written < 0 || (size_t)written >= sizeof(buffer)) {
        fprintf(stderr, "Failed to copy input safely\n");
        return EXIT_FAILURE;
    }

    printf("Before: %s\n", buffer);

    remove_non_alnum(buffer);

    printf("After:  %s\n", buffer);

    return EXIT_SUCCESS;
}

2. **Safe copy with `snprintf`** — the copy into `buffer` is bounded by `sizeof(buffer)`, and both the negative return (encoding error) and truncation cases (`written >= sizeof(buffer)`) are explicitly detected and handled, so no overflow or silent truncation can occur.

3. **Safe in-place filtering** — `remove_non_alnum` uses a two-index read/write compaction that can only shrink the string, never grow it, so it cannot overflow. It also guards against `NULL`, and the cast to `unsigned char` before calling `isalnum` avoids undefined behavior for negative `char` values on platforms where `char` is signed.

4. **Null termination guaranteed** — the write index terminates the string after compaction, and `snprintf` always null-terminates within the given size.
