#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_SIZE 1024

int count_matches(const char *pattern, const char *str) {
    regex_t regex;
    regmatch_t match;
    int count = 0;
    const char *p = str;

    if (pattern == NULL || str == NULL) {
        return -1;
    }

    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        return -1;
    }

    while (regexec(&regex, p, 1, &match, 0) == 0) {
        count++;
        p += match.rm_eo;
        if (match.rm_so == match.rm_eo) {
            p++;
        }
    }

    regfree(&regex);
    return count;
}

void categorize_characters(const char *str, int *uppercase, int *lowercase, int *special, int *numeric) {
    if (str == NULL || uppercase == NULL || lowercase == NULL || special == NULL || numeric == NULL) {
        return;
    }

    *uppercase = count_matches("[A-Z]", str);
    *lowercase = count_matches("[a-z]", str);
    *special = count_matches("[^a-zA-Z0-9]", str);
    *numeric = count_matches("[0-9]", str);
}

int main(void) {
    char *input = NULL;
    size_t buffer_size = MAX_INPUT_SIZE;
    int uppercase, lowercase, special, numeric;

    input = (char *)malloc(buffer_size * sizeof(char));
    if (input == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter a string: ");
    if (fgets(input, (int)buffer_size, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(input);
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\n")] = '\0';

    categorize_characters(input, &uppercase, &lowercase, &special, &numeric);

    printf("Uppercase: %d\n", uppercase);
    printf("Lowercase: %d\n", lowercase);
    printf("Special characters: %d\n", special);
    printf("Numeric values: %d\n", numeric);

    free(input);
    return EXIT_SUCCESS;
}
