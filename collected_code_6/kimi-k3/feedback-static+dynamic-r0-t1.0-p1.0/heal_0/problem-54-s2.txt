#include <stdio.h>
#include <stdlib.h>

void countingSort(int arr[], int n) {
    if (arr == NULL || n <= 0) {
        return;
    }

    int max = arr[0];
    int min = arr[0];
    int i;

    for (i = 1; i < n; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
        if (arr[i] < min) {
            min = arr[i];
        }
    }

    int range = max - min + 1;
    if (range <= 0) {
        return;
    }

    int *count = (int *)calloc(range, sizeof(int));
    if (count == NULL) {
        return;
    }

    int *output = (int *)malloc(n * sizeof(int));
    if (output == NULL) {
        free(count);
        return;
    }

    for (i = 0; i < n; i++) {
        count[arr[i] - min]++;
    }

    for (i = 1; i < range; i++) {
        count[i] += count[i - 1];
    }

    for (i = n - 1; i >= 0; i--) {
        output[count[arr[i] - min] - 1] = arr[i];
        count[arr[i] - min]--;
    }

    for (i = 0; i < n; i++) {
        arr[i] = output[i];
    }

    free(count);
    free(output);
}

int main(void) {
    int arr[] = {4, 2, 2, 8, 3, 3, 1};
    int n = sizeof(arr) / sizeof(arr[0]);
    int i;

    printf("Original array: ");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    countingSort(arr, n);

    printf("Sorted array: ");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
