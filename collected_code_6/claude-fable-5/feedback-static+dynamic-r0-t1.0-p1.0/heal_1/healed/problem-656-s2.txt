```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int min_abs_diff_sum(const int *arr1, const int *arr2, size_t n, long long *result)
{
    int *a;
    int *b;
    size_t i;
    size_t bytes;
    long long sum;

    if (arr1 == NULL || arr2 == NULL || result == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*a)) {
        return -1;
    }

    bytes = n * sizeof(*a);

    a = malloc(bytes);
    if (a == NULL) {
        return -1;
    }

    b = malloc(bytes);
    if (b == NULL) {
        free(a);
        return -1;
    }

    memcpy(a, arr1, bytes);
    memcpy(b, arr2, bytes);

    qsort(a, n, sizeof(*a), compare_ints);
    qsort(b, n, sizeof(*b), compare_ints);

    sum = 0;
    for (i = 0; i < n; i++) {
        long long diff = (long long)a[i] - (long long)b[i];
        if (diff < 0) {
            diff = -diff;
        }
        sum += diff;
    }

    free(a);
    free(b);

    *result = sum;
    return 0;
}

int main(void)
{
    static const int arr1[] = {4, 1, 8, 7};
    static const int arr2[] = {2, 3, 6, 5};
    size_t n = sizeof(arr1) / sizeof(arr1[0]);
    long long result;

    if (min_abs_diff_sum(arr1, arr2, n, &result) != 0) {
        fprintf(stderr, "Error computing minimum sum\n");
        return EXIT_FAILURE;
    }

    printf("Minimum sum of absolute differences: %lld\n", result);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-120 (memcpy buffer overflow concern):** Added an explicit overflow guard (`n > SIZE_MAX / sizeof(*a)`) before computing the allocation size, so `n * sizeof(*a)` can never wrap around. The byte count is computed once into `bytes` and used consistently for both `malloc` and `memcpy`, guaranteeing the destination buffers are exactly as large as the amount copied.

2. **constVariable (cppcheck):** Both `arr1` and `arr2` are now declared `const` (and `static`, since they are never modified), which also matches the `const int *` parameters of `min_abs_diff_sum` without any implicit qualification conversion.

The `<stdint.h>` header was added for `SIZE_MAX`. All original behavior—sorting both copies and summing absolute pairwise differences in `long long` to avoid intermediate overflow—remains unchanged.
