#include <stdio.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

double calculate_cone_volume(double radius, double height) {
    if (radius < 0.0 || height < 0.0) {
        return -1.0;
    }
    return (1.0 / 3.0) * M_PI * radius * radius * height;
}

int main(void) {
    double radius = 0.0;
    double height = 0.0;
    double volume = 0.0;
    int result = 0;

    result = scanf("%lf %lf", &radius, &height);
    if (result != 2) {
        return 1;
    }

    volume = calculate_cone_volume(radius, height);
    if (volume < 0.0) {
        return 1;
    }

    printf("%f\n", volume);

    return 0;
}
