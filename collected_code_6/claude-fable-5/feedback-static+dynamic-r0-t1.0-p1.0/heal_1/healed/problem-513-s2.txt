```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }
    len += 1U;

    copy = malloc(len);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len - 1U);
    copy[len - 1U] = '\0';
    return copy;
}

static void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char **tuple_to_list_with_string(const int *tuple, size_t tuple_len,
                                        const char *insert, size_t *out_len)
{
    char **result;
    size_t total;
    size_t i;
    size_t idx;

    if (tuple == NULL || insert == NULL || out_len == NULL || tuple_len == 0U) {
        return NULL;
    }
    if (tuple_len > (SIZE_MAX / sizeof(char *)) / 2U) {
        return NULL;
    }

    total = tuple_len * 2U;
    result = calloc(total, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    idx = 0U;
    for (i = 0U; i < tuple_len; i++) {
        char buffer[32];
        int written;

        written = snprintf(buffer, sizeof(buffer), "%d", tuple[i]);
        if (written < 0 || (size_t)written >= sizeof(buffer)) {
            free_string_list(result, idx);
            return NULL;
        }

        result[idx] = duplicate_string(buffer);
        if (result[idx] == NULL) {
            free_string_list(result, idx);
            return NULL;
        }
        idx++;

        result[idx] = duplicate_string(insert);
        if (result[idx] == NULL) {
            free_string_list(result, idx);
            return NULL;
        }
        idx++;
    }

    *out_len = total;
    return result;
}

int main(void)
{
    const int tuple[] = {5, 6, 7, 4, 9};
    const size_t tuple_len = sizeof(tuple) / sizeof(tuple[0]);
    const char *insert = "FDF";
    char **list;
    size_t list_len;
    size_t i;

    list = tuple_to_list_with_string(tuple, tuple_len, insert, &list_len);
    if (list == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (i = 0U; i < list_len; i++) {
        printf("'%s'", list[i]);
        if (i + 1U < list_len) {
            printf(", ");
        }
    }
    printf("]\n");

    free_string_list(list, list_len);
    return EXIT_SUCCESS;
}
```

Key changes from the original:

1. **Added `<stdint.h>`** — the original used `SIZE_MAX` without including the header that defines it, which would fail to compile on conforming implementations.

2. **Replaced `strlen` with `strnlen`** (addressing the CWE-126 finding) — `duplicate_string` now bounds the read with `MAX_STRING_LEN`, so a non-NUL-terminated input can no longer cause an unbounded over-read. If the string reaches the cap, the function fails safely rather than copying a truncated or over-read buffer.

3. **Hardened the `memcpy`** (addressing the CWE-120 finding) — the copy length is derived from the bounded `strnlen` result and the destination is allocated with exactly that size, then explicitly NUL-terminated, guaranteeing the destination always holds the source data.

4. **`snprintf` bounds check retained** (addressing the CWE-119/CWE-120 finding on the fixed-size `char buffer[32]`) — the return value is validated against the buffer size, so truncation or encoding errors are detected and handled by cleaning up and returning `NULL`.

The overflow guard on `tuple_len * 2U * sizeof(char *)`, the `calloc` zero-initialization, and the careful partial-cleanup on every failure path were already correct and are preserved.
