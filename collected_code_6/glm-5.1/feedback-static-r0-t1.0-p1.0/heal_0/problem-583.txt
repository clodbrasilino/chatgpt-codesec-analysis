#include <stdio.h>
#include <stdlib.h>

unsigned long long nth_catalan_number(unsigned int n) {
    if (n > 34) {
        exit(EXIT_FAILURE);
    }

    unsigned long long *c = (unsigned long long *)malloc((n + 1) * sizeof(unsigned long long));
    if (c == NULL) {
        exit(EXIT_FAILURE);
    }

    c[0] = 1;
    c[1] = 1;

    for (unsigned int i = 2; i <= n; i++) {
        c[i] = 0;
        for (unsigned int j = 0; j < i; j++) {
            c[i] += c[j] * c[i - j - 1];
        }
    }

    unsigned long long result = c[n];
    free(c);

    return result;
}

int main(void) {
    unsigned int n = 10;
    unsigned long long result = nth_catalan_number(n);
    printf("%llu\n", result);
    return 0;
}
