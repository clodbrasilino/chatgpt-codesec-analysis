```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, sizeof(normalized));
    if (len == 0 || len >= sizeof(normalized)) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code already mitigates every weakness flagged by Flawfinder, so the rewrite preserves that logic while stripping the comments:

1. **`char normalized[MAX_INPUT_LEN]` (CWE-119/CWE-120):** Bounds are enforced via `strnlen(month, sizeof(normalized))`, rejection of inputs where `len >= sizeof(normalized)`, and the loop guard `i < sizeof(normalized) - 1` before null-terminating at `normalized[i]`. No overflow is possible.

2. **`getchar()` in a loop (CWE-120/CWE-20):** The `flush_stdin` loop never writes to a buffer — it only discards characters until newline or EOF, so there is no buffer boundary to violate. The `EOF` check prevents an infinite loop on closed input.

3. **`char input[MAX_INPUT_LEN]` (CWE-119/CWE-120):** Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits the read length; the buffer is zero-initialized, explicitly null-terminated at the last index as a defensive measure, and any overlong line is drained with `flush_stdin` to keep `stdin` consistent.
