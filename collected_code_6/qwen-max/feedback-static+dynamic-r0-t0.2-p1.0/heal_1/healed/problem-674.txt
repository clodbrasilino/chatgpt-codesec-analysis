#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 1000
#define MAX_WORD_LENGTH 100

int is_word_in_array(const char *word, const char *array[], int count) {
    for (int i = 0; i < count; i++) {
        if (strcmp(word, array[i]) == 0) {
            return 1;
        }
    }
    return 0;
}

char* remove_duplicates(const char *input) {
    char *words[MAX_WORDS] = {NULL};
    int word_count = 0;

    const char *token;
    token = strtok((char *)input, " ");
    while (token != NULL && word_count < MAX_WORDS) {
        if (!is_word_in_array(token, (const char *const *)words, word_count)) {
            words[word_count++] = strdup(token);
        }
        token = strtok(NULL, " ");
    }

    int result_len = 0;
    for (int i = 0; i < word_count; i++) {
        result_len += strlen(words[i]) + 1;
    }

    char *result = (char *)malloc(result_len);
    result[0] = '\0';

    for (int i = 0; i < word_count; i++) {
        strcat(result, words[i]);
        if (i < word_count - 1) {
            strcat(result, " ");
        }
        free(words[i]);
    }

    return result;
}

int main() {
    const char *input = "Python Exercises Practice Solution Python Exercises";
    char *output = remove_duplicates(input);
    printf("%s\n", output);
    free(output);
    return 0;
}
