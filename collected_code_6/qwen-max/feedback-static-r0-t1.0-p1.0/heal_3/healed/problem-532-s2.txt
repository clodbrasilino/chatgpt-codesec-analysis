#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <limits.h>

bool are_permutations(const char *str1, const char *str2) {
    size_t len1 = strnlen(str1, SIZE_MAX);
    size_t len2 = strnlen(str2, SIZE_MAX);

    if (len1 != len2) {
        return false;
    }

    int count[256] = {0};
    for (size_t i = 0; i < len1; ++i) {
        if ((unsigned char)str1[i] >= 256 || (unsigned char)str2[i] >= 256) {
            return false;
        }
        count[(unsigned char)str1[i]]++;
        count[(unsigned char)str2[i]]--;
    }

    for (int i = 0; i < 256; ++i) {
        if (count[i] != 0) {
            return false;
        }
    }

    return true;
}

int main() {
    const char *str1 = "listen";
    const char *str2 = "silent";
    if (are_permutations(str1, str2)) {
        printf("The strings are permutations of each other.\n");
    } else {
        printf("The strings are not permutations of each other.\n");
    }
    return 0;
}
