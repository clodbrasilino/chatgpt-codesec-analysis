#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} DictionaryEntry;

typedef struct {
    DictionaryEntry *entries;
    size_t size;
    size_t capacity;
} Dictionary;

typedef struct {
    void **items;
    size_t size;
    size_t capacity;
} Tuple;

Dictionary *dictionary_create(void) {
    Dictionary *dict = malloc(sizeof(Dictionary));
    if (!dict) return NULL;
    dict->entries = malloc(4 * sizeof(DictionaryEntry));
    if (!dict->entries) {
        free(dict);
        return NULL;
    }
    dict->size = 0;
    dict->capacity = 4;
    return dict;
}

void dictionary_destroy(Dictionary *dict) {
    if (!dict) return;
    for (size_t i = 0; i < dict->size; i++) {
        free(dict->entries[i].key);
        free(dict->entries[i].value);
    }
    free(dict->entries);
    free(dict);
}

int dictionary_set(Dictionary *dict, const char *key, const char *value) {
    if (!dict || !key || !value) return -1;
    
    for (size_t i = 0; i < dict->size; i++) {
        if (strcmp(dict->entries[i].key, key) == 0) {
            char *new_value = strdup(value);
            if (!new_value) return -1;
            free(dict->entries[i].value);
            dict->entries[i].value = new_value;
            return 0;
        }
    }
    
    if (dict->size == dict->capacity) {
        size_t new_capacity = dict->capacity * 2;
        DictionaryEntry *new_entries = realloc(dict->entries, new_capacity * sizeof(DictionaryEntry));
        if (!new_entries) return -1;
        dict->entries = new_entries;
        dict->capacity = new_capacity;
    }
    
    char *key_copy = strdup(key);
    char *value_copy = strdup(value);
    if (!key_copy || !value_copy) {
        free(key_copy);
        free(value_copy);
        return -1;
    }
    
    dict->entries[dict->size].key = key_copy;
    dict->entries[dict->size].value = value_copy;
    dict->size++;
    return 0;
}

Tuple *tuple_create(void) {
    Tuple *tuple = malloc(sizeof(Tuple));
    if (!tuple) return NULL;
    tuple->items = malloc(4 * sizeof(void *));
    if (!tuple->items) {
        free(tuple);
        return NULL;
    }
    tuple->size = 0;
    tuple->capacity = 4;
    return tuple;
}

void tuple_destroy(Tuple *tuple) {
    if (!tuple) return;
    for (size_t i = 0; i < tuple->size; i++) {
        dictionary_destroy((Dictionary *)tuple->items[i]);
    }
    free(tuple->items);
    free(tuple);
}

Dictionary *dictionary_clone(const Dictionary *dict) {
    if (!dict) return NULL;
    Dictionary *clone = dictionary_create();
    if (!clone) return NULL;
    for (size_t i = 0; i < dict->size; i++) {
        if (dictionary_set(clone, dict->entries[i].key, dict->entries[i].value) != 0) {
            dictionary_destroy(clone);
            return NULL;
        }
    }
    return clone;
}

int tuple_add_dictionary(Tuple *tuple, const Dictionary *dict) {
    if (!tuple || !dict) return -1;
    
    Dictionary *dict_copy = dictionary_clone(dict);
    if (!dict_copy) return -1;
    
    if (tuple->size == tuple->capacity) {
        size_t new_capacity = tuple->capacity * 2;
        void **new_items = realloc(tuple->items, new_capacity * sizeof(void *));
        if (!new_items) {
            dictionary_destroy(dict_copy);
            return -1;
        }
        tuple->items = new_items;
        tuple->capacity = new_capacity;
    }
    
    tuple->items[tuple->size] = dict_copy;
    tuple->size++;
    return 0;
}

void dictionary_print(const Dictionary *dict) {
    if (!dict) return;
    printf("{");
    for (size_t i = 0; i < dict->size; i++) {
        printf("\"%s\": \"%s\"", dict->entries[i].key, dict->entries[i].value);
        if (i < dict->size - 1) printf(", ");
    }
    printf("}");
}

void tuple_print(const Tuple *tuple) {
    if (!tuple) return;
    printf("(");
    for (size_t i = 0; i < tuple->size; i++) {
        dictionary_print((Dictionary *)tuple->items[i]);
        if (i < tuple->size - 1) printf(", ");
    }
    printf(")\n");
}

int main(void) {
    Dictionary *dict1 = dictionary_create();
    Dictionary *dict2 = dictionary_create();
    Tuple *tuple = tuple_create();
    
    if (!dict1 || !dict2 || !tuple) {
        dictionary_destroy(dict1);
        dictionary_destroy(dict2);
        tuple_destroy(tuple);
        return 1;
    }
    
    dictionary_set(dict1, "name", "Alice");
    dictionary_set(dict1, "age", "30");
    
    dictionary_set(dict2, "city", "New York");
    dictionary_set(dict2, "country", "USA");
    
    tuple_add_dictionary(tuple, dict1);
    tuple_add_dictionary(tuple, dict2);
    
    tuple_print(tuple);
    
    dictionary_destroy(dict1);
    dictionary_destroy(dict2);
    tuple_destroy(tuple);
    
    return 0;
}
