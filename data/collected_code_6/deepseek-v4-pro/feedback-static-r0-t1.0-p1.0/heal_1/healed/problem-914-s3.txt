#include <stdio.h>
#include <stdbool.h>

bool is_alternating(const char *str) {
    if (str == NULL) {
        return false;
    }
    
    if (str[0] == '\0' || str[1] == '\0') {
        return true;
    }
    
    if (str[0] == str[1]) {
        return false;
    }
    
    for (size_t i = 2; str[i] != '\0'; i++) {
        if (str[i] != str[i - 2]) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    const char *test_strings[] = {
        "ababab",
        "aaaa",
        "a",
        "ab",
        "ababa",
        "abcabc",
        "abababa",
        ""
    };
    
    size_t num_tests = sizeof(test_strings) / sizeof(test_strings[0]);
    
    for (size_t i = 0; i < num_tests; i++) {
        printf("\"%s\" -> %s\n", 
               test_strings[i], 
               is_alternating(test_strings[i]) ? "true" : "false");
    }
    
    return 0;
}
