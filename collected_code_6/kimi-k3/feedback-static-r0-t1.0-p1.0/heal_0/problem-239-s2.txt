#include <stdio.h>
#include <stdlib.h>
#include <string.h>

long long countSequences(int n, int m) {
    if (n <= 0 || m <= 0) {
        return 0;
    }
    if (n == 1) {
        return m;
    }
    
    long long *prev = (long long *)calloc((size_t)m + 1, sizeof(long long));
    long long *curr = (long long *)calloc((size_t)m + 1, sizeof(long long));
    
    if (prev == NULL || curr == NULL) {
        free(prev);
        free(curr);
        return 0;
    }
    
    for (int i = 1; i <= m; i++) {
        prev[i] = 1;
    }
    
    for (int len = 2; len <= n; len++) {
        memset(curr, 0, ((size_t)m + 1) * sizeof(long long));
        for (int j = 1; j <= m; j++) {
            for (int k = 1; k <= j / 2; k++) {
                curr[j] += prev[k];
            }
        }
        long long *temp = prev;
        prev = curr;
        curr = temp;
    }
    
    long long result = 0;
    for (int i = 1; i <= m; i++) {
        result += prev[i];
    }
    
    free(prev);
    free(curr);
    
    return result;
}

int main(void) {
    int n, m;
    
    if (scanf("%d %d", &n, &m) != 2) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    if (n < 1 || m < 1) {
        printf("0\n");
        return 0;
    }
    
    long long result = countSequences(n, m);
    printf("%lld\n", result);
    
    return 0;
}
