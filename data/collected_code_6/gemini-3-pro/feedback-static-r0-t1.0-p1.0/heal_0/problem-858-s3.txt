#include <stdio.h>
#include <stdlib.h>

struct Node {
    struct Node* nested_list;
    struct Node* next;
};

int count_and_square(struct Node* head) {
    if (head == NULL) {
        return 0;
    }
    
    int count = 0;
    struct Node* current = head;
    
    while (current != NULL) {
        if (current->nested_list != NULL) {
            count++;
        }
        current = current->next;
    }
    
    return count * count;
}

int main() {
    struct Node* head = malloc(sizeof(struct Node));
    if (head == NULL) return 1;

    struct Node* nested1 = malloc(sizeof(struct Node));
    if (nested1 == NULL) {
        free(head);
        return 1;
    }
    nested1->nested_list = NULL;
    nested1->next = NULL;

    head->nested_list = nested1;
    
    struct Node* second = malloc(sizeof(struct Node));
    if (second == NULL) {
        free(nested1);
        free(head);
        return 1;
    }
    second->nested_list = NULL;
    second->next = NULL;
    head->next = second;

    int result = count_and_square(head);
    
    printf("%d\n", result);

    free(second);
    free(nested1);
    free(head);

    return 0;
}
