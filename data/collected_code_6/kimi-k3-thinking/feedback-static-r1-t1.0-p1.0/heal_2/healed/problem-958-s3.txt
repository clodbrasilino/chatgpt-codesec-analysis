#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999
#define MAX_ROMAN_NUMERAL_LENGTH 15
#define ROMAN_BUFFER_SIZE (MAX_ROMAN_NUMERAL_LENGTH + 1)

typedef struct {
    int value;
    const char *symbol;
    size_t symbol_length;
} RomanEntry;

static int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const RomanEntry table[] = {
        {1000, "M", 1U},
        {900, "CM", 2U},
        {500, "D", 1U},
        {400, "CD", 2U},
        {100, "C", 1U},
        {90, "XC", 2U},
        {50, "L", 1U},
        {40, "XL", 2U},
        {10, "X", 1U},
        {9, "IX", 2U},
        {5, "V", 1U},
        {4, "IV", 2U},
        {1, "I", 1U}
    };
    const size_t entry_count = sizeof(table) / sizeof(table[0]);
    size_t used;
    size_t i;
    int remaining;

    if (buffer == NULL || buffer_size == 0U)
    {
        return -1;
    }

    buffer[0] = '\0';

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE)
    {
        return -1;
    }

    if (buffer_size < ROMAN_BUFFER_SIZE)
    {
        return -1;
    }

    used = 0U;
    remaining = number;

    for (i = 0U; i < entry_count; i++)
    {
        while (remaining >= table[i].value)
        {
            size_t j;
            const size_t symbol_length = table[i].symbol_length;

            if (symbol_length >= (buffer_size - used))
            {
                buffer[0] = '\0';
                return -1;
            }

            for (j = 0U; j < symbol_length; j++)
            {
                buffer[used + j] = table[i].symbol[j];
            }

            used += symbol_length;
            remaining -= table[i].value;
        }
    }

    buffer[used] = '\0';
    return 0;
}

int main(void)
{
    char *buffer;
    int number;
    int result;

    buffer = malloc(ROMAN_BUFFER_SIZE);
    if (buffer == NULL)
    {
        fprintf(stderr, "Error: memory allocation failed\n");
        return 1;
    }

    printf("Enter an integer (1-3999): ");
    result = fflush(stdout);
    if (result != 0)
    {
        fprintf(stderr, "Error: failed to write prompt\n");
        free(buffer);
        return 1;
    }

    result = scanf("%d", &number);
    if (result != 1)
    {
        fprintf(stderr, "Error: invalid input\n");
        free(buffer);
        return 1;
    }

    result = int_to_roman(number, buffer, ROMAN_BUFFER_SIZE);
    if (result != 0)
    {
        fprintf(stderr, "Error: value must be between 1 and 3999\n");
        free(buffer);
        return 1;
    }

    printf("%d -> %s\n", number, buffer);
    free(buffer);
    return 0;
}
