#include <stdio.h>
#include <stdlib.h>

int* find_greater_than(const int* list, size_t size, int threshold, size_t* result_size) {
    if (list == NULL || result_size == NULL) {
        return NULL;
    }
    
    size_t count = 0;
    size_t i;
    
    for (i = 0; i < size; i++) {
        if (list[i] > threshold) {
            count++;
        }
    }
    
    int* result = NULL;
    if (count > 0) {
        result = (int*)malloc(count * sizeof(int));
        if (result == NULL) {
            return NULL;
        }
        
        size_t index = 0;
        for (i = 0; i < size; i++) {
            if (list[i] > threshold) {
                result[index] = list[i];
                index++;
            }
        }
    }
    
    *result_size = count;
    return result;
}

int main(void) {
    int list[] = {5, 12, 3, 18, 7, 25, 1, 10};
    size_t size = sizeof(list) / sizeof(list[0]);
    int threshold = 8;
    size_t result_size = 0;
    
    int* result = find_greater_than(list, size, threshold, &result_size);
    
    if (result == NULL && result_size > 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    printf("Values greater than %d: ", threshold);
    for (size_t i = 0; i < result_size; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");
    
    free(result);
    result = NULL;
    
    return EXIT_SUCCESS;
}
