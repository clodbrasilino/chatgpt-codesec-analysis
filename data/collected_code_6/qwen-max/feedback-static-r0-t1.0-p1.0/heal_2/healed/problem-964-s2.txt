```c
#include <stdio.h>
#include <stdbool.h>

bool is_length_even(const char *word, size_t length) {
    if (word == NULL || length == 0) {
        return false;
    }
    return (length % 2) == 0;
}

int main() {
    const char test_word1[] = "even";
    const char test_word2[] = "odd";

    printf("%s: %s\n", test_word1, is_length_even(test_word1, sizeof(test_word1) - 1) ? "Even" : "Odd");
    printf("%s: %s\n", test_word2, is_length_even(test_word2, sizeof(test_word2) - 1) ? "Even" : "Odd");

    return 0;
}
```
