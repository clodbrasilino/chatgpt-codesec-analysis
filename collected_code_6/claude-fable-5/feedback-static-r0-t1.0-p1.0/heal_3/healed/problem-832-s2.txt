```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <errno.h>
#include <float.h>
#include <math.h>

#define NUMBER_BUFFER_SIZE 64

static int extract_max_number(const char *input, double *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    double max_value;
    int found;
    int ret;

    if (input == NULL || result == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "-?[0-9]+(\\.[0-9]+)?", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    found = 0;
    max_value = -DBL_MAX;

    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        if (match.rm_so >= 0 && match.rm_eo >= match.rm_so) {
            size_t len = (size_t)(match.rm_eo - match.rm_so);

            if (len > 0 && len < NUMBER_BUFFER_SIZE) {
                char buffer[NUMBER_BUFFER_SIZE];
                char *endptr = NULL;
                double value;

                if (len <= sizeof(buffer) - 1) {
                    memcpy(buffer, cursor + match.rm_so, len);
                    buffer[len] = '\0';

                    errno = 0;
                    value = strtod(buffer, &endptr);
                    if (errno == 0 && endptr != buffer && isfinite(value)) {
                        if (!found || value > max_value) {
                            max_value = value;
                            found = 1;
                        }
                    }
                }
            }
        }

        if (match.rm_eo == 0) {
            break;
        }
        cursor += match.rm_eo;
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test = "Values: 42, -17, 3.14, 1000.5, and 999";
    double max_value = 0.0;

    if (extract_max_number(test, &max_value) != 0) {
        if (fprintf(stderr, "No numeric value found\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Maximum value: %g\n", max_value) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two Flawfinder findings in the original code were both low-risk (risk 2) heuristic warnings, and the original code already mitigated them correctly:

1. **CWE-119/CWE-120 (`char buffer[NUMBER_BUFFER_SIZE]`)** — Flawfinder flags any statically-sized array. The mitigation is bounds checking before use, which the code performs via `len <= NUMBER_BUFFER_SIZE - 1`. In the rewrite I strengthened this with an explicit guard tied directly to `sizeof(buffer)` so the bound check cannot silently drift if the buffer declaration changes.

2. **CWE-120 (`memcpy`)** — Flawfinder warns that `memcpy` doesn't self-check destination size. Since `len` is validated against the destination capacity immediately before the copy, and `buffer[len] = '\0'` stays within bounds, no overflow is possible. The `len <= sizeof(buffer) - 1` check makes this invariant explicit at the call site.

Other defensive properties preserved from the original:

- NULL-pointer checks on `input` and `result`
- `regcomp` return value checked; `regfree` always called after successful compile
- `errno`/`endptr` validation on `strtod` plus `isfinite()` to reject overflow-to-infinity
- Zero-length match guard (`match.rm_eo == 0`) preventing an infinite loop
- Return values of `printf`/`fprintf` checked in `main`

The code is functionally identical: it correctly prints `Maximum value: 1000.5` for the test input.
