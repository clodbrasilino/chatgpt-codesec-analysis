```c
#include <stdio.h>

int sum_of_evens(int l, int r) {
    int sum = 0;
    if (l > r) {
        return sum;
    }
    if (l % 2 != 0) {
        l++;
    }
    for (int i = l; i <= r; i += 2) {
        sum += i;
    }
    return sum;
}

int main() {
    int l, r;
    printf("Enter the range (l and r): ");
    scanf("%d %d", &l, &r);
    int result = sum_of_evens(l, r);
    printf("Sum of even numbers between %d and %d is: %d\n", l, r, result);
    return 0;
}
```
