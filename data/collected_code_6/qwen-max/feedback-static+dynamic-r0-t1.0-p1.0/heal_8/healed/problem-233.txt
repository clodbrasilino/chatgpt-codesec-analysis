```c
#include <stdio.h>
#include <math.h>

double lateral_surface_area_cylinder(double radius, double height) {
    return 2.0 * M_PI * radius * height;
}

int main() {
    double radius = 10.0;
    double height = 10.0;
    printf("Lateral Surface Area: %.15f\n", lateral_surface_area_cylinder(radius, height));

    radius = 5.0;
    height = 8.0;
    printf("Lateral Surface Area: %.15f\n", lateral_surface_area_cylinder(radius, height));

    radius = 4.0;
    height = 12.0;
    printf("Lateral Surface Area: %.15f\n", lateral_surface_area_cylinder(radius, height));

    return 0;
}
```
