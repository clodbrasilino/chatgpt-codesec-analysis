```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096
#define MAX_LIST_SIZE 1024

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;
    size_t list_bytes;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    if (list_size > MAX_LIST_SIZE) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    list_bytes = list_size * sizeof(int);

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(len + 1);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->str, str, len);
    t->str[len] = '\0';

    if (list_size > 0) {
        t->list = malloc(list_bytes);
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_bytes);
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->str == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *s = "WEB";
    int values[] = {1, 2};
    Tuple *t;

    t = create_tuple(s, values, sizeof(values) / sizeof(values[0]));
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warnings on both `memcpy` calls are false positives in this code, but the rewrite makes the safety guarantees explicit:

1. **String copy (`t->str`):** The destination is allocated with `malloc(len + 1)`, where `len` is bounded by `strnlen(str, MAX_STR_LEN)` and rejected if it reaches `MAX_STR_LEN`. The `memcpy` copies exactly `len` bytes and null-termination is done manually, so the destination always holds the source data. I changed `len == MAX_STR_LEN` to `len >= MAX_STR_LEN` as a defensive measure, though functionally equivalent since `strnlen` cannot exceed its bound.

2. **List copy (`t->list`):** The size is validated against both `MAX_LIST_SIZE` and `SIZE_MAX / sizeof(int)`, preventing integer overflow in the allocation size. I hoisted `list_size * sizeof(int)` into a single `list_bytes` variable computed *after* the overflow check, so the identical, verified value is used for both `malloc` and `memcpy` — eliminating any chance of a mismatch between allocation and copy sizes.

3. **Other correctness points preserved:** all allocation failures unwind cleanly (no leaks), `free_tuple` is null-safe, and `print_tuple` guards against null pointers. The empty-list case correctly stores `NULL` rather than calling `malloc(0)`.
