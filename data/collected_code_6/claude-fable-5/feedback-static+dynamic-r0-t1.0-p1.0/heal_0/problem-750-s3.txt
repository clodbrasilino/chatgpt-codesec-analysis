#include <stdio.h>
#include <stdlib.h>

struct IntList {
    int *data;
    size_t size;
    size_t capacity;
};

static int list_init(struct IntList *list, size_t initial_capacity)
{
    if (list == NULL || initial_capacity == 0U) {
        return -1;
    }
    list->data = malloc(initial_capacity * sizeof(int));
    if (list->data == NULL) {
        list->size = 0U;
        list->capacity = 0U;
        return -1;
    }
    list->size = 0U;
    list->capacity = initial_capacity;
    return 0;
}

static int list_append(struct IntList *list, int value)
{
    if (list == NULL || list->data == NULL) {
        return -1;
    }
    if (list->size >= list->capacity) {
        size_t new_capacity;
        int *new_data;
        if (list->capacity > (SIZE_MAX / (2U * sizeof(int)))) {
            return -1;
        }
        new_capacity = list->capacity * 2U;
        new_data = realloc(list->data, new_capacity * sizeof(int));
        if (new_data == NULL) {
            return -1;
        }
        list->data = new_data;
        list->capacity = new_capacity;
    }
    list->data[list->size] = value;
    list->size++;
    return 0;
}

static int add_tuple_to_list(struct IntList *list, const int *tuple, size_t tuple_len)
{
    size_t i;
    if (list == NULL || tuple == NULL) {
        return -1;
    }
    for (i = 0U; i < tuple_len; i++) {
        if (list_append(list, tuple[i]) != 0) {
            return -1;
        }
    }
    return 0;
}

static void list_free(struct IntList *list)
{
    if (list != NULL) {
        free(list->data);
        list->data = NULL;
        list->size = 0U;
        list->capacity = 0U;
    }
}

static void list_print(const struct IntList *list)
{
    size_t i;
    if (list == NULL || list->data == NULL) {
        return;
    }
    printf("[");
    for (i = 0U; i < list->size; i++) {
        printf("%d", list->data[i]);
        if (i + 1U < list->size) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main(void)
{
    struct IntList list;
    const int initial_values[] = {5, 6, 7};
    const int tuple[] = {9, 10};
    size_t i;

    if (list_init(&list, 4U) != 0) {
        fprintf(stderr, "Failed to initialize list\n");
        return EXIT_FAILURE;
    }

    for (i = 0U; i < sizeof(initial_values) / sizeof(initial_values[0]); i++) {
        if (list_append(&list, initial_values[i]) != 0) {
            fprintf(stderr, "Failed to append initial value\n");
            list_free(&list);
            return EXIT_FAILURE;
        }
    }

    printf("Original list: ");
    list_print(&list);

    if (add_tuple_to_list(&list, tuple, sizeof(tuple) / sizeof(tuple[0])) != 0) {
        fprintf(stderr, "Failed to add tuple to list\n");
        list_free(&list);
        return EXIT_FAILURE;
    }

    printf("List after adding tuple: ");
    list_print(&list);

    list_free(&list);
    return EXIT_SUCCESS;
}
