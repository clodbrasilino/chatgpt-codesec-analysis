#include <stdio.h>
#include <stdlib.h>

int is_greater_than_all(int number, const int *array, size_t size)
{
    size_t i;

    if (array == NULL && size > 0) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (number <= array[i]) {
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    int number;
    size_t size;
    int *array;
    size_t i;
    int result;

    printf("Enter the number of elements: ");
    if (scanf("%zu", &size) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (size == 0) {
        fprintf(stderr, "Array size must be greater than zero\n");
        return EXIT_FAILURE;
    }

    array = malloc(size * sizeof(*array));
    if (array == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter %zu elements:\n", size);
    for (i = 0; i < size; i++) {
        if (scanf("%d", &array[i]) != 1) {
            fprintf(stderr, "Invalid input\n");
            free(array);
            return EXIT_FAILURE;
        }
    }

    printf("Enter the number to compare: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        free(array);
        return EXIT_FAILURE;
    }

    result = is_greater_than_all(number, array, size);

    if (result) {
        printf("%d is greater than all elements of the array\n", number);
    } else {
        printf("%d is not greater than all elements of the array\n", number);
    }

    free(array);
    return EXIT_SUCCESS;
}
