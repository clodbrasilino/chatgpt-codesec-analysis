#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char* longest_common_prefix(char **strs, int num) {
    if (num == 0) return "";

    char *prefix = strs[0];
    for (int i = 1; i < num; ++i) {
        size_t prefix_len = strlen(prefix);
        size_t str_len = strlen(strs[i]);
        for (size_t j = 0; j < prefix_len && j < str_len; ++j) {
            if (prefix[j] != strs[i][j]) {
                prefix[j] = '\0';
                break;
            }
        }
        if (*prefix == '\0') break;
    }
    return prefix;
}

int main() {
    char *strs[] = {"flower", "flow", "flight"};
    int num = 3;
    char *result = longest_common_prefix(strs, num);
    printf("%s\n", result);
    return 0;
}
