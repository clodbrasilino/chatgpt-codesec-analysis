#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int findKthElement(const int *arr1, size_t size1, const int *arr2, size_t size2, size_t k) {
    if (arr1 == NULL || arr2 == NULL || k == 0 || k > size1 + size2) {
        return INT_MIN;
    }

    if (size1 > size2) {
        return findKthElement(arr2, size2, arr1, size1, k);
    }

    if (size1 == 0) {
        return arr2[k - 1];
    }

    if (k == 1) {
        return (arr1[0] < arr2[0]) ? arr1[0] : arr2[0];
    }

    size_t i = (k / 2 < size1) ? k / 2 : size1;
    size_t j = (k / 2 < size2) ? k / 2 : size2;

    if (arr1[i - 1] > arr2[j - 1]) {
        return findKthElement(arr1, size1, arr2 + j, size2 - j, k - j);
    } else {
        return findKthElement(arr1 + i, size1 - i, arr2, size2, k - i);
    }
}

int main(void) {
    int arr1[] = {2, 3, 6, 7, 9};
    int arr2[] = {1, 4, 8, 10};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);
    size_t k = 5;

    int result = findKthElement(arr1, size1, arr2, size2, k);

    if (result == INT_MIN) {
        fprintf(stderr, "Invalid input or k out of bounds\n");
        return EXIT_FAILURE;
    }

    printf("The %zuth element is: %d\n", k, result);

    return EXIT_SUCCESS;
}
