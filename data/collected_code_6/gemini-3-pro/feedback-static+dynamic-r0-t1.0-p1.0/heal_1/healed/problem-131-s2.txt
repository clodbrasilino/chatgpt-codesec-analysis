#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_STR_LEN 4096

bool is_vowel(char c) {
    c = tolower((unsigned char)c);
    return (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u');
}

void reverse_vowels(char *str) {
    if (str == NULL) {
        return;
    }

    size_t len = 0;
    while (str[len] != '\0' && len < MAX_STR_LEN) {
        len++;
    }

    if (len == 0) {
        return;
    }

    size_t left = 0;
    size_t right = len - 1;

    while (left < right) {
        while (left < right && !is_vowel(str[left])) {
            left++;
        }
        while (left < right && !is_vowel(str[right])) {
            right--;
        }

        if (left < right) {
            char temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            right--;
        }
    }
}

int main(void) {
    char test_str[] = "hello world";
    reverse_vowels(test_str);
    printf("%s\n", test_str);

    char test_str2[] = "Programming is fun";
    reverse_vowels(test_str2);
    printf("%s\n", test_str2);

    return 0;
}
