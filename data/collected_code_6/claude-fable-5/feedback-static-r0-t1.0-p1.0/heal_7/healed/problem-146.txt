```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        sum += (unsigned char)str[i];
    }

    return sum;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? EOF : 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long result;

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    if (strchr(buffer, '\n') == NULL && !feof(stdin)) {
        discard_remaining_input();
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — Flawfinder flags this as a risk when the character read is stored into a buffer without bounds checking. In `discard_remaining_input()`, the return value is never written into a buffer; it is only compared against `'\n'` and `EOF`. This is the canonical safe idiom for draining `stdin`, so no change is required. Correctly using `int ch` (not `char`) ensures `EOF` is reliably detected.

2. **Statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — The buffer is only ever written through `fgets(buffer, sizeof(buffer), stdin)`, which enforces the length limit and always NUL-terminates on success. Defense-in-depth measures already in place:
   - Explicit re-termination via `buffer[sizeof(buffer) - 1] = '\0'`.
   - Detection of truncated (over-long) input by checking for a missing `'\n'` while not at EOF, followed by draining the stream to avoid leftover input affecting subsequent reads.
   - `strcspn` used to strip the newline safely (never reads past the terminator).
   - `ascii_sum` uses `strnlen` with an explicit bound and casts to `unsigned char` before summation, avoiding both out-of-bounds reads and sign-related surprises on platforms where `char` is signed.

The original code was already written defensively; the Flawfinder hits are informational (it flags patterns, not proven bugs). The rewrite preserves all bounds checks, adds an explicit `(int)` cast on the `fgets` size argument for strict conversion-warning cleanliness, and removes the comments as requested.
