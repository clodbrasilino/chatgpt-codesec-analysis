```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    const char *first;
    const char *second;
} Tuple;

int compare_tuples(const void *a, const void *b) {
    const Tuple *tuple_a = (const Tuple *)a;
    const Tuple *tuple_b = (const Tuple *)b;
    
    if (tuple_a->first == NULL && tuple_b->first == NULL) return 0;
    if (tuple_a->first == NULL) return -1;
    if (tuple_b->first == NULL) return 1;
    
    return strcmp(tuple_a->first, tuple_b->first);
}

void sort_tuples(Tuple *tuples, size_t count) {
    if (tuples == NULL || count == 0) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

int main(void) {
    Tuple tuples[] = {
        {"banana", "yellow"},
        {"apple", "red"},
        {"cherry", "dark red"},
        {"date", "brown"}
    };
    size_t count = sizeof(tuples) / sizeof(tuples[0]);
    size_t i;
    
    sort_tuples(tuples, count);
    
    for (i = 0; i < count; i++) {
        printf("(%s, %s)\n", tuples[i].first, tuples[i].second);
    }
    
    return 0;
}
```
