#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

char* remove_words_with_length(const char* input, int k) {
    if (input == NULL || k <= 0) {
        return NULL;
    }

    size_t len = strnlen(input, 4096);
    if (len == 0 || len >= 4096) {
        char* result = (char*)malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    size_t capacity = len + 1;
    char* result = (char*)malloc(capacity);
    if (result == NULL) {
        return NULL;
    }

    size_t i = 0;
    size_t j = 0;
    size_t word_start = 0;
    size_t word_length = 0;
    int in_word = 0;

    while (i <= len) {
        if (i < len && (isalnum((unsigned char)input[i]) || input[i] == '\'' || input[i] == '-')) {
            if (!in_word) {
                word_start = i;
                word_length = 1;
                in_word = 1;
            } else {
                word_length++;
            }
        } else {
            if (in_word) {
                if (word_length != (size_t)k) {
                    if (j > 0) {
                        if (j + 2 > capacity) {
                            if (capacity > SIZE_MAX / 2) {
                                free(result);
                                return NULL;
                            }
                            size_t new_capacity = capacity * 2;
                            char* temp = (char*)realloc(result, new_capacity);
                            if (temp == NULL) {
                                free(result);
                                return NULL;
                            }
                            result = temp;
                            capacity = new_capacity;
                        }
                        result[j++] = ' ';
                    }
                    if (j + word_length + 1 > capacity) {
                        size_t needed = j + word_length + 1;
                        size_t new_capacity = capacity;
                        while (new_capacity < needed) {
                            if (new_capacity > SIZE_MAX / 2) {
                                free(result);
                                return NULL;
                            }
                            new_capacity *= 2;
                        }
                        char* temp = (char*)realloc(result, new_capacity);
                        if (temp == NULL) {
                            free(result);
                            return NULL;
                        }
                        result = temp;
                        capacity = new_capacity;
                    }
                    if (word_length > 0 && j + word_length <= capacity && word_start + word_length <= len) {
                        memcpy(result + j, input + word_start, word_length);
                        j += word_length;
                    }
                }
                in_word = 0;
                word_length = 0;
            }
        }
        i++;
    }

    if (j + 1 > capacity) {
        size_t new_capacity = j + 1;
        char* temp = (char*)realloc(result, new_capacity);
        if (temp == NULL) {
            free(result);
            return NULL;
        }
        result = temp;
        capacity = new_capacity;
    }
    
    if (j < capacity) {
        result[j] = '\0';
    } else {
        free(result);
        return NULL;
    }

    char* final_result = (char*)realloc(result, j + 1);
    if (final_result == NULL) {
        free(result);
        return NULL;
    }

    return final_result;
}

int main() {
    char input[] = "This is a test string with words of different lengths";
    int k = 2;
    
    char* result = remove_words_with_length(input, k);
    if (result != NULL) {
        printf("Original: %s\n", input);
        printf("After removing words of length %d: %s\n", k, result);
        free(result);
    } else {
        printf("Error processing string\n");

}
