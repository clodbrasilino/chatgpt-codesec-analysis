```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}
```

The code already mitigates the Flawfinder-flagged weakness (CWE-119/CWE-120) in `next_bigger`:

- **Bounded write**: `snprintf(digits, sizeof(digits), ...)` is used instead of `sprintf`, guaranteeing the buffer cannot be overflowed during formatting.
- **Truncation check**: The return value of `snprintf` is validated (`len < 0 || (size_t)len >= sizeof(digits)`), so truncated or failed conversions are rejected rather than silently processed.
- **Sufficient buffer size**: `DIGITS_BUF_SIZE` (32) comfortably exceeds the maximum decimal length of a `long long` (19 digits plus sign and NUL terminator, ~21 bytes).
- **Initialization**: `memset` zeroes the buffer, ensuring no uninitialized reads.
- **Bounded indexing**: All loop indices (`i`, `j`) are constrained within `[0, len)`, and `reverse_range` only operates within validated bounds.
- **Overflow-safe reconstruction**: The digit-to-integer loop checks `result > (LLONG_MAX - d) / 10` before multiplying, preventing signed integer overflow (undefined behavior in C).
- **Digit validation**: Each character is verified to be in `'0'`–`'9'` before use, guarding against unexpected buffer contents.

No further changes were required; the statically-sized array is safely used because every access is length-limited and bounds-checked.
