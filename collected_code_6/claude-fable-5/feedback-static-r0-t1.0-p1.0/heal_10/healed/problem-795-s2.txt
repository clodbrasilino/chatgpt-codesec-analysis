```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define ITEM_NAME_LEN 64

typedef struct {
    char name[ITEM_NAME_LEN];
    double price;
} Item;

static void swap_items(Item *a, Item *b)
{
    Item tmp = *a;
    *a = *b;
    *b = tmp;
}

static void sift_down(Item *heap, size_t size, size_t idx)
{
    for (;;) {
        size_t smallest = idx;
        size_t left = 2U * idx + 1U;
        size_t right = 2U * idx + 2U;

        if (left < size && heap[left].price < heap[smallest].price) {
            smallest = left;
        }
        if (right < size && heap[right].price < heap[smallest].price) {
            smallest = right;
        }
        if (smallest == idx) {
            break;
        }
        swap_items(&heap[idx], &heap[smallest]);
        idx = smallest;
    }
}

static void build_min_heap(Item *heap, size_t size)
{
    if (size < 2U) {
        return;
    }
    for (size_t i = size / 2U; i > 0U; i--) {
        sift_down(heap, size, i - 1U);
    }
}

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0U;
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

static int item_set_name(Item *item, const char *name)
{
    if (item == NULL || name == NULL) {
        return -1;
    }

    size_t len = bounded_strlen(name, sizeof(item->name));
    if (len >= sizeof(item->name)) {
        return -1;
    }

    memcpy(item->name, name, len);
    item->name[len] = '\0';
    return 0;
}

static int find_n_cheapest(const Item *items, size_t count, size_t n, Item *result, size_t result_capacity)
{
    if (items == NULL || result == NULL || n == 0U || n > count || n > result_capacity) {
        return -1;
    }

    if (count > SIZE_MAX / sizeof(Item)) {
        return -1;
    }

    Item *heap = malloc(count * sizeof(Item));
    if (heap == NULL) {
        return -1;
    }

    for (size_t i = 0U; i < count; i++) {
        heap[i] = items[i];
        heap[i].name[sizeof(heap[i].name) - 1U] = '\0';
    }

    build_min_heap(heap, count);

    size_t heap_size = count;
    for (size_t i = 0U; i < n; i++) {
        result[i] = heap[0];
        heap_size--;
        heap[0] = heap[heap_size];
        sift_down(heap, heap_size, 0U);
    }

    free(heap);
    return 0;
}

int main(void)
{
    static const struct {
        const char *name;
        double price;
    } raw_data[] = {
        {"Laptop", 999.99},
        {"Mouse", 19.99},
        {"Keyboard", 49.99},
        {"Monitor", 199.99},
        {"USB Cable", 5.99},
        {"Headphones", 89.99},
        {"Webcam", 39.99},
        {"Desk Lamp", 24.99}
    };

    size_t count = sizeof(raw_data) / sizeof(raw_data[0]);
    size_t n = 3U;

    Item dataset[sizeof(raw_data) / sizeof(raw_data[0])];
    memset(dataset, 0, sizeof(dataset));

    for (size_t i = 0U; i < count; i++) {
        if (item_set_name(&dataset[i], raw_data[i].name) != 0) {
            fprintf(stderr, "Item name too long: %s\n", raw_data[i].name);
            return EXIT_FAILURE;
        }
        dataset[i].price = raw_data[i].price;
    }

    if (n > SIZE_MAX / sizeof(Item)) {
        fprintf(stderr, "Requested size too large\n");
        return EXIT_FAILURE;
    }

    Item *cheapest = malloc(n * sizeof(Item));
    if (cheapest == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (find_n_cheapest(dataset, count, n, cheapest, n) != 0) {
        fprintf(stderr, "Failed to find cheapest items\n");
        free(cheapest);
        return EXIT_FAILURE;
    }

    printf("The %zu cheapest items are:\n", n);
    for (size_t i = 0U; i < n; i++) {
        printf("%s: $%.2f\n", cheapest[i].name, cheapest[i].price);
    }

    free(cheapest);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **CWE-119/CWE-120 (statically-sized array `name[ITEM_NAME_LEN]`):** The fixed-size buffer is never written to without bounds enforcement. `item_set_name()` is the single write path for names, and it validates length via `bounded_strlen()` (which itself never reads past `max_len`) before any copy occurs. Names of length ≥ 64 (including the NUL) are rejected with an error return rather than truncated silently or overflowed.

2. **CWE-120 (`memcpy` without overflow check):** This is a false positive in context. The `memcpy` in `item_set_name()` is guarded by `if (len >= sizeof(item->name)) return -1;`, guaranteeing `len ≤ ITEM_NAME_LEN - 1`, so `len` bytes plus the explicit NUL terminator always fit within the destination. This pattern is preferable to `strncpy`, which does not guarantee NUL termination and has misleading padding semantics.

**Additional defensive measures already present and retained:**

- **Integer overflow before allocation:** `count > SIZE_MAX / sizeof(Item)` checks prevent multiplication overflow in `malloc(count * sizeof(Item))` (CWE-190).
- **NULL checks** on all `malloc` results and pointer parameters.
- **Defensive re-termination** of copied names in `find_n_cheapest()` (`heap[i].name[sizeof - 1] = '\0'`) protects against callers passing non-terminated `Item` structs.
- **Capacity validation** (`n > result_capacity`) prevents writing past the caller's result buffer.
- **No leaks:** `heap` and `cheapest` are freed on all paths, including error paths.

The code compiles cleanly with `-Wall -Wextra -Wpedantic` and requires no functional changes — the Flawfinder hits are informational and already mitigated by the existing bounds checks.
