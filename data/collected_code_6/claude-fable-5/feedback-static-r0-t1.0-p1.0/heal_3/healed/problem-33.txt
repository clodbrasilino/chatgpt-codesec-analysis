```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define BINARY_BUFFER_SIZE (CHAR_BIT * sizeof(long) + 2)

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size);

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t start;
    size_t end;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal < 0) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index] = '-';
        index++;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    start = index;

    if (value == 0UL) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index] = '0';
        index++;
    } else {
        while (value > 0UL) {
            if (index + 1 >= buffer_size) {
                return -1;
            }
            buffer[index] = (char)('0' + (char)(value % 2UL));
            index++;
            value /= 2UL;
        }
    }

    if (index >= buffer_size) {
        return -1;
    }
    buffer[index] = '\0';

    end = index - 1;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    long decimal = 0;
    char binary[BINARY_BUFFER_SIZE] = {0};
    int scan_result;

    printf("Enter a decimal number: ");

    scan_result = scanf("%ld", &decimal);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(decimal, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary representation: %s\n", binary);

    return EXIT_SUCCESS;
}
```

The source code is already well-hardened against the identified weakness. Here is my analysis of why:

**Regarding the Flawfinder CWE-119/CWE-120 warning on the `binary` buffer:**

1. **Buffer sizing is mathematically correct**: `CHAR_BIT * sizeof(long) + 2` guarantees room for every bit of a `long` (typically 64 bits), plus one byte for a potential `-` sign and one for the null terminator.

2. **All writes are bounds-checked**: The conversion function receives `buffer_size` explicitly via `sizeof(binary)` and validates `index + 1 >= buffer_size` before every single write, so no overflow is possible regardless of buffer size.

3. **Negative-value handling avoids UB**: The expression `(unsigned long)(-(decimal + 1)) + 1UL` safely converts `LONG_MIN` without triggering signed-overflow undefined behavior that a naive `-decimal` would cause.

4. **Input is validated**: The `scanf` return value is checked, and `%ld` with a `long *` argument is type-correct with no string format vulnerability.

The changes I made in the rewrite are minor defensive additions: a bounds check before writing the `-` sign, an explicit check before writing the null terminator, and extracting the buffer size into a named constant for maintainability. The Flawfinder finding is effectively a false positive here — the static array is used safely because its size is always passed to and honored by the consuming function.
