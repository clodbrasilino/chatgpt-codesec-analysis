```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_LEN 256

static void print_regex_error(int ret, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_LEN];

    memset(errbuf, 0, sizeof(errbuf));
    (void)regerror(ret, regex, errbuf, sizeof(errbuf) - 1U);
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const regex_t *regex, const char *input,
                                  size_t input_len, size_t *offset)
{
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;

    if (regex == NULL || input == NULL || offset == NULL) {
        return NULL;
    }

    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regexec(regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (size_t)end <= input_len - *offset) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    if ((size_t)matches[0].rm_eo <= input_len - *offset) {
                        *offset += (size_t)matches[0].rm_eo;
                    } else {
                        *offset = input_len;
                    }
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, regex, "Regex execution failed");
    }

    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;
    size_t input_len;
    regex_t regex;
    int ret;

    input_len = strnlen(text, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(&regex, text, input_len, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    regfree(&regex);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **`errbuf` static arrays (CWE-119/CWE-120)**: Centralized error handling into `print_regex_error()`, which zeroes the buffer with `memset` before use, passes `sizeof(errbuf) - 1U` to `regerror()` to guarantee room for the terminator, and explicitly null-terminates as a defensive measure.

2. **`memcpy` overflow risk (CWE-120)**: The destination is allocated as exactly `len + 1U` bytes, `len` is validated against `MAX_INPUT_LEN` before allocation, the source bounds are verified (`end <= input_len - *offset` prevents integer underflow since `*offset < input_len` is checked earlier), and the copy is skipped entirely when `len == 0`.

3. **Additional hardening**: The regex is now compiled once in `main()` instead of on every call (eliminating repeated compile/free cycles and a per-call failure path), the offset advancement is bounds-checked to prevent it from exceeding the input length, and input length validation is hoisted out of the loop so `strnlen` runs once.
