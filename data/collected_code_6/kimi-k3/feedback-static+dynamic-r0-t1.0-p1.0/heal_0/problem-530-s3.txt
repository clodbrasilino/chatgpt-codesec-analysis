#include <stdio.h>
#include <stddef.h>

double find_negative_ratio(const int *array, size_t size)
{
    size_t negative_count;
    size_t i;
    
    if (array == NULL || size == 0)
    {
        return 0.0;
    }
    
    negative_count = 0;
    
    for (i = 0; i < size; i++)
    {
        if (array[i] < 0)
        {
            negative_count++;
        }
    }
    
    return (double)negative_count / (double)size;
}

int main(void)
{
    int test_array[] = {5, -3, 8, -1, 0, -7, 2, -4};
    size_t array_size;
    double ratio;
    
    array_size = sizeof(test_array) / sizeof(test_array[0]);
    ratio = find_negative_ratio(test_array, array_size);
    
    printf("Ratio of negative numbers: %.2f\n", ratio);
    
    return 0;
}
