#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool normalize_month(const char *month, char *out, size_t out_size)
{
    size_t len;
    size_t i;

    if (month == NULL || out == NULL || out_size == 0) {
        return false;
    }

    len = strnlen(month, out_size);
    if (len == 0 || len >= out_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        out[i] = (char)tolower((unsigned char)month[i]);
    }
    out[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strcmp(normalized, all_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %s\n", input);
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and fixes applied:**

The original code was already quite defensive, and the Flawfinder hits were mostly informational rather than actual vulnerabilities. Here's my assessment and what I changed:

1. **`char` statically-sized arrays (CWE-119/CWE-120):** These were false positives in practice — every write into `normalized[]` and `input[]` was already bounds-checked (`len >= MAX_INPUT_LEN` rejection, `fgets` with `sizeof(input)`). I kept the arrays but centralized the normalization logic into a single `normalize_month()` helper that takes an explicit buffer size, eliminating code duplication and reducing the chance of a future divergence introducing a bug.

2. **`strlen` on possibly non-terminated strings (CWE-126):** Replaced `strlen` with `strnlen` bounded by the buffer size. Since `fgets` always null-terminates and the callers pass string literals or the `fgets` buffer, this was defensive hardening rather than a real fix — but `strnlen` makes the over-read impossible even if a caller misuses the API. I also added an explicit `input[sizeof(input) - 1] = '\0'` as belt-and-braces termination.

3. **`getchar` in a loop (CWE-120/CWE-20):** The drain loop discards characters and writes nothing to a buffer, so this was a pure false positive. Retained as-is since it correctly flushes oversized input lines.

4. **Minor improvements:** Added `fflush(stdout)` after the unterminated prompt, used `EXIT_SUCCESS`/`EXIT_FAILURE` from `<stdlib.h>`, and kept the `(unsigned char)` cast before `tolower()` — which the original correctly had, avoiding undefined behavior for negative `char` values.
