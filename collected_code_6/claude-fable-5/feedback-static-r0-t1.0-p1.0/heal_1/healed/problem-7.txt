```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[A-Za-z]{4,}", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[256];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    cursor = text;
    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        if (match.rm_so == -1 || match.rm_eo < match.rm_so) {
            break;
        }

        size_t length = (size_t)(match.rm_eo - match.rm_so);
        char *word = malloc(length + 1);
        if (word == NULL) {
            fprintf(stderr, "malloc failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(word, cursor + match.rm_so, length);
        word[length] = '\0';
        printf("%s\n", word);
        free(word);

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        char errbuf[256];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regexec failed: %s\n", errbuf);
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog";

    if (find_long_words(text) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Variable scope reduction** (`variableScope`): Both `word` and `length` are now declared inside the `while` loop body, limiting their scope to exactly where they are used and preventing accidental use of stale values across iterations.

2. **Statically-sized buffer concerns** (CWE-119/CWE-120 on `errbuf`): The buffers remain fixed-size since `regerror` requires a caller-supplied buffer, but bounds are enforced defensively — `sizeof(errbuf)` is passed so `regerror` cannot overflow, and an explicit null terminator is written at the last index to guarantee termination regardless of library behavior.

3. **`memcpy` overflow concern** (CWE-120): The destination is allocated as `length + 1` bytes immediately before the copy of exactly `length` bytes, so the destination is guaranteed to hold the source data. An additional sanity check (`match.rm_eo < match.rm_so`) prevents a negative offset difference from producing a huge unsigned `length`.
