#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

bool contains_z_not_at_edges(const char *word) {
    if (word == NULL) {
        return false;
    }
    
    size_t len = strlen(word);
    
    if (len < 3) {
        return false;
    }
    
    for (size_t i = 1; i < len - 1; i++) {
        if (word[i] == 'z' || word[i] == 'Z') {
            return true;
        }
    }
    
    return false;
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read;
    
    read = getline(&buffer, &buffer_size, stdin);
    
    if (read != -1) {
        if (read > 0 && buffer[read - 1] == '\n') {
            buffer[read - 1] = '\0';
        }
        
        if (contains_z_not_at_edges(buffer)) {
            printf("Match\n");
        } else {
            printf("No match\n");
        }
        
        free(buffer);
    }
    
    return 0;
}
