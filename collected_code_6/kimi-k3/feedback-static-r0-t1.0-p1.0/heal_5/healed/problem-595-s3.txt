#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>
#include <errno.h>

#define MAX_INPUT_LEN 99
#define BUFFER_SIZE (MAX_INPUT_LEN + 2)

static int is_valid_binary_string(const char *str, size_t max_len) {
    if (str == NULL) {
        return 0;
    }
    
    size_t len = strnlen(str, max_len + 1);
    if (len > max_len) {
        return 0;
    }
    
    for (size_t i = 0; i < len; i++) {
        if (str[i] != '0' && str[i] != '1') {
            return 0;
        }
    }
    
    return 1;
}

static int safe_copy_string(char *dest, size_t dest_size, const char *src, size_t max_len) {
    if (dest == NULL || src == NULL || dest_size == 0) {
        return -1;
    }
    
    size_t src_len = strnlen(src, max_len + 1);
    if (src_len > max_len || src_len >= dest_size) {
        return -1;
    }
    
    if (src_len > 0) {
        memcpy(dest, src, src_len);
    }
    dest[src_len] = '\0';
    
    return 0;
}

static int read_input_safely(char *buffer, size_t buffer_size, const char *prompt) {
    if (buffer == NULL || buffer_size == 0 || prompt == NULL) {
        return -1;
    }
    
    printf("%s", prompt);
    fflush(stdout);
    
    if (fgets(buffer, buffer_size, stdin) == NULL) {
        return -1;
    }
    
    size_t len = strnlen(buffer, buffer_size);
    
    if (len == 0) {
        return -1;
    }
    
    if (len == buffer_size - 1 && buffer[len - 1] != '\n') {
        int c;
        int overflow_detected = 0;
        
        while ((c = getchar()) != '\n' && c != EOF) {
            overflow_detected = 1;
        }
        
        if (overflow_detected || c == EOF) {
            return -2;
        }
    }
    
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        len--;
    }
    
    if (len > MAX_INPUT_LEN) {
        return -2;
    }
    
    return 0;
}

int min_swaps(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }
    
    if (!is_valid_binary_string(str1, MAX_INPUT_LEN) ||
        !is_valid_binary_string(str2, MAX_INPUT_LEN)) {
        return -1;
    }
    
    size_t len1 = strnlen(str1, MAX_INPUT_LEN + 1);
    size_t len2 = strnlen(str2, MAX_INPUT_LEN + 1);
    
    if (len1 != len2) {
        return -1;
    }
    
    if (len1 == 0) {
        return 0;
    }
    
    int count0_str1 = 0, count1_str1 = 0;
    int count0_str2 = 0, count1_str2 = 0;
    
    for (size_t i = 0; i < len1; i++) {
        if (str1[i] == '0') {
            count0_str1++;
        } else {
            count1_str1++;
        }
        
        if (str2[i] == '0') {
            count0_str2++;
        } else {
            count1_str2++;
        }
    }
    
    if (count0_str1 != count0_str2 || count1_str1 != count1_str2) {
        return -1;
    }
    
    char *str1_copy = malloc(len1 + 1);
    if (str1_copy == NULL) {
        return -1;
    }
    
    if (safe_copy_string(str1_copy, len1 + 1, str1, len1) != 0) {
        free(str1_copy);
        return -1;
    }
    
    int swaps = 0;
    size_t i = 0, j = 0;
    
    while (i < len1 && j < len2) {
        if (str1_copy[i] == str2[j]) {
            i++;
            j++;
        } else {
            size_t k = i + 1;
            while (k < len1 && str1_copy[k] == str1_copy[i]) {
                k++;
            }
            
            if (k < len1) {
                swaps++;
                char temp = str1_copy[i];
                str1_copy[i] = str1_copy[k];
                str1_copy[k] = temp;
            } else {
                free(str1_copy);
                return -1;
            }
        }
    }
    
    free(str1_copy);
    return swaps;
}

int main(void) {
    char str1[BUFFER_SIZE];
    char str2[BUFFER_SIZE];
    
    int result1 = read_input_safely(str1, sizeof(str1), "Enter first binary string: ");
    if (result1 == -1) {
        printf("Invalid input\n");
        return 1;
    } else if (result1 == -2) {
        printf("Input too long\n");
        return 1;
    }
    
    int result2 = read_input_safely(str2, sizeof(str2), "Enter second binary string: ");
    if (result2 == -1) {
        printf("Invalid input\n");
        return 1;
    } else if (result2 == -2) {
        printf("Input too long\n");
        return 1;
    }
    
    int result = min_swaps(str1, str2);
    
    if (result == -1) {
        printf("Conversion not possible\n");
    } else {
        printf("Minimum swaps required: %d\n", result);
    }
    
    return 0;
}
