#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

size_t length_of_last_word(const char *s)
{
    size_t len;
    size_t end;
    size_t count;

    if (s == NULL) {
        return 0U;
    }

    len = 0U;
    while (len < MAX_INPUT_LEN && s[len] != '\0') {
        len++;
    }

    if (len == 0U) {
        return 0U;
    }

    end = len;
    while (end > 0U && isspace((unsigned char)s[end - 1U])) {
        end--;
    }

    count = 0U;
    while (end > 0U && !isspace((unsigned char)s[end - 1U])) {
        count++;
        end--;
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t result;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

        buffer[0] = '\0';
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
    }

    result = length_of_last_word(buffer);

    if (printf("%lu\n", (unsigned long)result) < 0) {
        return 1;
    }

    if (fflush(stdout) != 0) {

    }

    return 0;

1. **Bounded input reading**: `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow — this is the primary fix for the flagged `char buffer[MAX_INPUT_LEN]` weakness.

2. **Guaranteed NUL termination**: `buffer[sizeof(buffer) - 1U] = '\0'` is a defensive belt-and-suspenders measure, and `memset` zero-initializes the buffer beforehand.

3. **Bounded string operations**: `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, and `length_of_last_word` caps its own scan at `MAX_INPUT_LEN`, so even a non-terminated string cannot cause an out-of-bounds read.

4. **Safe `ctype` usage**: Casting to `(unsigned char)` before calling `isspace` avoids undefined behavior for negative `char` values.

5. **Checked I/O**: Return values of `fgets`, `printf`, and `fflush` are all checked, with graceful failure paths.
