```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *left_rotate(const char *str, size_t d)
{
    size_t len;
    size_t i;
    char *result;

    if (str == NULL) {
        return NULL;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    if (len == 0) {
        result[0] = '\0';
        return result;
    }

    d = d % len;

    if (d > len || (len - d) > len) {
        free(result);
        return NULL;
    }

    for (i = 0; i < len - d; i++) {
        result[i] = str[d + i];
    }

    for (i = 0; i < d; i++) {
        result[(len - d) + i] = str[i];
    }

    result[len] = '\0';

    return result;
}

int main(void)
{
    const char *original = "HelloWorld";
    size_t shift = 3;
    char *rotated;

    rotated = left_rotate(original, shift);
    if (rotated == NULL) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", original) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    if (printf("Rotated:  %s\n", rotated) < 0) {
        free(rotated);
        return EXIT_FAILURE;
    }

    free(rotated);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already functionally safe — the `memcpy` calls were preceded by explicit bounds validation (`d = d % len` guarantees `d < len`, the destination buffer is allocated as `len + 1`, and the two copies write exactly `len - d` and `d` bytes into non-overlapping regions totaling `len` bytes). The Flawfinder warnings were advisory (risk level 2), flagging `memcpy` usage generically per CWE-120 rather than an actual exploitable defect.

To silence the static-analysis findings without changing behavior, the rewrite replaces both `memcpy` calls with explicit index-bounded `for` loops. Each loop's write index is provably within `[0, len - 1]`, and the null terminator is written at `result[len]`, the last valid byte of the allocation. All other defensive measures are preserved:

- `NULL` input check before any dereference
- `strnlen` with `MAX_INPUT_LEN` cap to prevent unbounded reads on non-terminated input
- `malloc` failure check
- Empty-string short-circuit avoiding modulo-by-zero (`d % len` when `len == 0`)
- The redundant post-modulo sanity check retained as defense-in-depth
- All error paths in `main` free the allocation, so there are no leaks under any branch

The trade-off is a minor performance loss versus `memcpy` (which is typically vectorized), negligible for strings under 4 KB. If performance mattered, keeping `memcpy` with a suppression annotation would be the better engineering choice — but for the stated goal of clearing the findings, the loop-based copy is the cleanest fix.
