#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

#define MAX_LINE_LENGTH 1048576

int is_integer(const char *str)
{
    char *endptr;
    long value;

    if (str == NULL) {
        return 0;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    errno = 0;
    value = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return 0;
    }

    if (endptr == str) {
        return 0;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 0;
    }

    (void)value;
    return 1;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *line;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    line = malloc(capacity);
    if (line == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= MAX_LINE_LENGTH) {
            free(line);
            return NULL;
        }
        if (length + 1 >= capacity) {
            size_t new_capacity;
            char *tmp;

            if (capacity > SIZE_MAX / 2) {
                free(line);
                return NULL;
            }
            new_capacity = capacity * 2;
            tmp = realloc(line, new_capacity);
            if (tmp == NULL) {
                free(line);
                return NULL;
            }
            line = tmp;
            capacity = new_capacity;
        }
        line[length++] = (char)ch;
    }

    if (ch == EOF && length == 0) {
        free(line);
        return NULL;
    }

    line[length] = '\0';
    return line;
}

int main(void)
{
    char *buffer;

    printf("Enter a string: ");
    fflush(stdout);

    buffer = read_line(stdin);
    if (buffer == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (is_integer(buffer)) {
        printf("\"%s\" is a valid integer\n", buffer);
    } else {
        printf("\"%s\" is not a valid integer\n", buffer);
    }

    free(buffer);
    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

**Flawfinder `fgetc` warning (CWE-120, CWE-20):** This is a heuristic warning that fires whenever `fgetc` is used in a loop, since naive implementations often write into a fixed-size buffer without bounds checking. The code already mitigates this correctly:

1. **Bounds checking on every iteration:** `length + 1 >= capacity` guarantees space for both the incoming character and the eventual null terminator before any write to `line[length]`.
2. **Hard upper limit:** The `MAX_LINE_LENGTH` check prevents unbounded memory consumption from a malicious or malformed input stream (DoS mitigation, CWE-400).
3. **Overflow-safe growth:** `capacity > SIZE_MAX / 2` is checked before doubling, preventing integer overflow in `capacity * 2` (CWE-190).
4. **Safe `realloc` idiom:** The result is stored in a temporary pointer, so the original buffer is freed on failure rather than leaked (CWE-401).
5. **EOF handling:** `fgetc` returns `int`, and the code correctly uses an `int` for `ch` so `EOF` is distinguishable from valid byte values.

Additional correctness points already present: `strtol` input validation uses `errno`/`ERANGE` and `endptr` checks (CWE-20), `isspace` receives `unsigned char` casts to avoid undefined behavior on negative `char` values, and all error paths free allocated memory. No changes were required beyond confirming the mitigations; the code stands as a correct fix for the flagged finding.
