#include <stdio.h>
#include <stdlib.h>

static int get_max(const int *arr, size_t n)
{
    int max = arr[0];
    size_t i;
    for (i = 1; i < n; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    return max;
}

static int counting_sort_by_digit(int *arr, size_t n, int exp)
{
    int *output;
    int count[10] = {0};
    size_t i;

    output = malloc(n * sizeof(*output));
    if (output == NULL) {
        return -1;
    }

    for (i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }

    for (i = 1; i < 10; i++) {
        count[i] += count[i - 1];
    }

    for (i = n; i > 0; i--) {
        int digit = (arr[i - 1] / exp) % 10;
        output[count[digit] - 1] = arr[i - 1];
        count[digit]--;
    }

    for (i = 0; i < n; i++) {
        arr[i] = output[i];
    }

    free(output);
    return 0;
}

int radix_sort(int *arr, size_t n)
{
    int max;
    int exp;

    if (arr == NULL || n == 0) {
        return -1;
    }

    max = get_max(arr, n);

    for (exp = 1; max / exp > 0; exp *= 10) {
        if (counting_sort_by_digit(arr, n, exp) != 0) {
            return -1;
        }
        if (exp > max / 10) {
            break;
        }
    }

    return 0;
}

int main(void)
{
    int data[] = {170, 45, 75, 90, 802, 24, 2, 66};
    size_t n = sizeof(data) / sizeof(data[0]);
    size_t i;

    if (radix_sort(data, n) != 0) {
        fprintf(stderr, "Sorting failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", data[i]) < 0) {
            return EXIT_FAILURE;
        }
    }
    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
