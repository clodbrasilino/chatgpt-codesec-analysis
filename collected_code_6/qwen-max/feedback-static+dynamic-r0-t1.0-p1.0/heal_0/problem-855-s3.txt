```c
#include <stdio.h>
#include <stdbool.h>

bool check_even_parity(int num) {
    int count = 0;
    while (num) {
        count += num & 1;
        num >>= 1;
    }
    return (count % 2) == 0;
}

int main() {
    int number;
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        return 1;
    }
    if (check_even_parity(number)) {
        printf("The number has even parity.\n");
    } else {
        printf("The number does not have even parity.\n");
    }
    return 0;
}
```
