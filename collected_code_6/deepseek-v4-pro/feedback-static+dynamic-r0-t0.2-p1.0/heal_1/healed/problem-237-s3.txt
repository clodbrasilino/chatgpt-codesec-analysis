#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    char name[50];
    int value;
} Record;

typedef struct {
    Record *records;
    int count;
} RecordGroup;

typedef struct {
    int *occurrences;
    int size;
} OccurrenceList;

static int compare_ints(const void *a, const void *b) {
    int arg1 = *(const int *)a;
    int arg2 = *(const int *)b;
    return (arg1 > arg2) - (arg1 < arg2);
}

static int compare_records(const void *a, const void *b) {
    const Record *ra = (const Record *)a;
    const Record *rb = (const Record *)b;
    if (ra->id != rb->id) return ra->id - rb->id;
    int name_cmp = strcmp(ra->name, rb->name);
    if (name_cmp != 0) return name_cmp;
    return ra->value - rb->value;
}

static int find_record_index(Record *records, int count, const Record *target) {
    for (int i = 0; i < count; i++) {
        if (records[i].id == target->id &&
            strcmp(records[i].name, target->name) == 0 &&
            records[i].value == target->value) {
            return i;
        }
    }
    return -1;
}

static int *get_occurrences(Record *records, int count, int *out_size) {
    if (records == NULL || count <= 0 || out_size == NULL) {
        if (out_size != NULL) *out_size = 0;
        return NULL;
    }

    Record *unique_records = (Record *)malloc((size_t)count * sizeof(Record));
    int *freq = (int *)malloc((size_t)count * sizeof(int));
    if (unique_records == NULL || freq == NULL) {
        free(unique_records);
        free(freq);
        *out_size = 0;
        return NULL;
    }

    int unique_count = 0;
    for (int i = 0; i < count; i++) {
        int idx = find_record_index(unique_records, unique_count, &records[i]);
        if (idx == -1) {
            unique_records[unique_count] = records[i];
            freq[unique_count] = 1;
            unique_count++;
        } else {
            freq[idx]++;
        }
    }

    int *occurrences = (int *)malloc((size_t)unique_count * sizeof(int));
    if (occurrences == NULL) {
        free(unique_records);
        free(freq);
        *out_size = 0;
        return NULL;
    }

    for (int i = 0; i < unique_count; i++) {
        occurrences[i] = freq[i];
    }

    free(unique_records);
    free(freq);
    *out_size = unique_count;
    return occurrences;
}

static int check_similar_occurrences(int *occurrences, int size) {
    if (occurrences == NULL || size <= 0) return 0;
    if (size == 1) return 1;

    qsort(occurrences, (size_t)size, sizeof(int), compare_ints);

    int min_freq = occurrences[0];
    int max_freq = occurrences[size - 1];

    return (max_freq - min_freq) <= 1;
}

int check_record_occurrences(Record *records, int count) {
    if (records == NULL || count <= 0) return 0;

    int occ_size = 0;
    int *occurrences = get_occurrences(records, count, &occ_size);
    if (occurrences == NULL || occ_size <= 0) {
        free(occurrences);
        return 0;
    }

    int result = check_similar_occurrences(occurrences, occ_size);
    free(occurrences);
    return result;
}

int main(void) {
    Record records1[] = {
        {1, "alpha", 10},
        {2, "beta", 20},
        {1, "alpha", 10},
        {3, "gamma", 30},
        {2, "beta", 20},
        {4, "delta", 40}
    };
    int count1 = (int)(sizeof(records1) / sizeof(records1[0]));
    printf("Test 1: %d\n", check_record_occurrences(records1, count1));

    Record records2[] = {
        {1, "alpha", 10},
        {2, "beta", 20},
        {1, "alpha", 10},
        {1, "alpha", 10},
        {3, "gamma", 30}
    };
    int count2 = (int)(sizeof(records2) / sizeof(records2[0]));
    printf("Test 2: %d\n", check_record_occurrences(records2, count2));

    Record records3[] = {
        {1, "alpha", 10},
        {2, "beta", 20},
        {3, "gamma", 30}
    };
    int count3 = (int)(sizeof(records3) / sizeof(records3[0]));
    printf("Test 3: %d\n", check_record_occurrences(records3, count3));

    Record records4[] = {
        {1, "alpha", 10},
        {1, "alpha", 10},
        {2, "beta", 20},
        {2, "beta", 20},
        {3, "gamma", 30},
        {3, "gamma", 30}
    };
    int count4 = (int)(sizeof(records4) / sizeof(records4[0]));
    printf("Test 4: %d\n", check_record_occurrences(records4, count4));

    printf("Test 5: %d\n", check_record_occurrences(NULL, 0));

    return 0;
}
