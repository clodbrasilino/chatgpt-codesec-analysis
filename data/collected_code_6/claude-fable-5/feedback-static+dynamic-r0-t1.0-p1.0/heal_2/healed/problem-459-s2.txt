```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_uppercase(const char *input);

char *remove_uppercase(const char *input)
{
    char *result;
    size_t input_len;
    size_t result_len;
    size_t i;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    result_len = 0;
    for (i = 0; i < input_len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (!isupper(c)) {
            if (result_len >= input_len) {
                free(result);
                return NULL;
            }
            result[result_len] = (char)c;
            result_len++;
        }
    }

    result[result_len] = '\0';
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "helloWORLDthisIS a TEST string",
        "ALLUPPERCASE",
        "nolowercasehere",
        "MixedCASEString"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_uppercase(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: \"%s\" -> Result: \"%s\"\n", test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code failed all test cases (returning no output) most likely because of the regex-based approach's fragile cursor arithmetic. The core bug was in the loop's bookkeeping: `cursor_len` was decremented by `match.rm_eo`, but the sanity check `result_len + (size_t)match.rm_so > input_len` combined with `strnlen(cursor, cursor_len)` on the `REG_NOMATCH` path could produce inconsistent lengths, and any validation failure caused the entire function to return `NULL` — hence "got no output" in all test cases. Additionally, `regexec` on the shifted `cursor` doesn't respect `cursor_len` at all (it reads until the NUL terminator), so the length tracking was both redundant and error-prone.

Key improvements in the rewrite:

1. **Eliminated the regex entirely** — removing uppercase characters is a trivial character-classification task. Using `isupper()` from `<ctype.h>` removes the entire class of pointer-arithmetic and offset-validation bugs, plus the `regcomp`/`regfree` resource-management burden.

2. **Fixed the Flawfinder CWE-120 findings** — the flagged `memcpy` calls are gone. The single-character copy is explicitly bounds-checked (`result_len >= input_len`) before every write, making buffer overflow structurally impossible since the output can never exceed the input length.

3. **Correct `ctype.h` usage** — the character is cast to `unsigned char` before passing to `isupper()`, avoiding undefined behavior for negative `char` values on platforms where `char` is signed.

4. **Preserved the defensive contract** — NULL input check, input-length capping via `strnlen` against `MAX_INPUT_LEN`, allocation failure handling, and guaranteed NUL termination are all retained from the original.
