```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_VALUE_LENGTH 64

typedef struct {
    const char *value;
} DictEntry;

typedef struct {
    const DictEntry *entries;
    size_t count;
} Dictionary;

static char **extract_unique_values(const Dictionary *dict, size_t *unique_count)
{
    size_t i;
    size_t j;
    size_t k;
    size_t found;
    char **unique;

    if (dict == NULL || unique_count == NULL || dict->entries == NULL) {
        return NULL;
    }

    *unique_count = 0;

    if (dict->count == 0) {
        return NULL;
    }

    if (dict->count > SIZE_MAX / sizeof(*unique)) {
        return NULL;
    }

    unique = malloc(dict->count * sizeof(*unique));
    if (unique == NULL) {
        return NULL;
    }

    found = 0;
    for (i = 0; i < dict->count; i++) {
        int is_duplicate = 0;
        const char *value = dict->entries[i].value;
        size_t value_len;

        if (value == NULL) {
            continue;
        }

        value_len = strnlen(value, MAX_VALUE_LENGTH);

        for (j = 0; j < found; j++) {
            if (strncmp(unique[j], value, MAX_VALUE_LENGTH) == 0) {
                is_duplicate = 1;
                break;
            }
        }

        if (!is_duplicate) {
            size_t dest_capacity = value_len + 1;

            unique[found] = malloc(dest_capacity);
            if (unique[found] == NULL) {
                for (k = 0; k < found; k++) {
                    free(unique[k]);
                }
                free(unique);
                return NULL;
            }

            memcpy(unique[found], value, value_len);
            unique[found][value_len] = '\0';
            found++;
        }
    }

    *unique_count = found;
    return unique;
}

int main(void)
{
    static const DictEntry entries[] = {
        {"apple"},
        {"banana"},
        {"apple"},
        {"cherry"},
        {"banana"},
        {"cherry"}
    };
    Dictionary dict;
    char **unique;
    size_t unique_count;
    size_t i;

    dict.entries = entries;
    dict.count = sizeof(entries) / sizeof(entries[0]);
    unique_count = 0;

    unique = extract_unique_values(&dict, &unique_count);
    if (unique == NULL) {
        fprintf(stderr, "Error: failed to extract unique values\n");
        return EXIT_FAILURE;
    }

    printf("Unique values:\n");
    for (i = 0; i < unique_count; i++) {
        printf("%s\n", unique[i]);
    }

    for (i = 0; i < unique_count; i++) {
        free(unique[i]);
    }
    free(unique);

    return EXIT_SUCCESS;
}
```

Key fixes applied:

1. **Reduced variable scope**: `dest_capacity` is now declared inside the `if (!is_duplicate)` block, limiting its scope to where it's actually used.

2. **Safe `memcpy`**: The destination buffer is now allocated with exactly `value_len + 1` bytes (`dest_capacity = value_len + 1`), guaranteeing the destination always holds the source data. Since `strnlen` bounds `value_len` to at most `MAX_VALUE_LENGTH`, the allocation can never exceed `MAX_VALUE_LENGTH + 1`, making the copy safe by construction and eliminating the need for the separate overflow check (which was previously dead code anyway).
