#include <stdio.h>
#include <stdlib.h>

long long countSequences(int n, int k) {
    long long dp[k + 1][n + 1];
    for (int i = 0; i <= k; ++i) {
        for (int j = 0; j <= n; ++j) {
            if (i == 0 || j == 0) {
                dp[i][j] = 0;
            } else if (i == 1 && j == 1) {
                dp[i][j] = 1;
            } else if (j > i) {
                dp[i][j] = 0;
            } else {
                dp[i][j] = dp[i - 1][j - 1] + dp[i - 1][j] * (i - 1);
            }
        }
    }
    return dp[k][n];
}

int main() {
    int n = 3;
    int k = 2;
    printf("Number of sequences: %lld\n", countSequences(n, k));
    return 0;
}
