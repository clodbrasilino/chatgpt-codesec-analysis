#include <stdio.h>
#include <stddef.h>

int remove_leading_zeroes(char *ip, size_t size) {
    if (ip == NULL || size == 0) {
        return -1;
    }

    size_t read = 0;
    size_t write = 0;
    int at_octet_start = 1;

    while (read < size && ip[read] != '\0') {
        if (ip[read] == '.') {
            if (write < size - 1) {
                ip[write++] = '.';
            }
            read++;
            at_octet_start = 1;
            continue;
        }

        if (at_octet_start) {
            while (read < size && ip[read] == '0' &&
                   (read + 1) < size && ip[read + 1] >= '0' && ip[read + 1] <= '9') {
                read++;
            }
            at_octet_start = 0;
        }

        if (read >= size || ip[read] == '\0') {
            break;
        }

        if (write < size - 1) {
            ip[write++] = ip[read++];
        } else {
            break;
        }
    }

    if (write < size) {
        ip[write] = '\0';
    } else if (size > 0) {
        ip[size - 1] = '\0';
    }

    return 0;
}

int main(void) {
    char ip[] = "192.168.001.001";

    if (remove_leading_zeroes(ip, sizeof(ip)) != 0) {
        return 1;
    }

    printf("%s\n", ip);

    return 0;
}
