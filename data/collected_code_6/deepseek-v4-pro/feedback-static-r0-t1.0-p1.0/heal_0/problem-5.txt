#include <stdio.h>

int count_domino_tilings(int n) {
    if (n < 0) return 0;
    if (n == 0) return 1;
    if (n == 1) return 0;
    if (n == 2) return 3;

    int prev2 = 1;
    int prev1 = 0;
    int current = 3;

    for (int i = 3; i <= n; i++) {
        int next = 4 * current - prev2;
        if (next < 0) return 0;
        prev2 = prev1;
        prev1 = current;
        current = next;
    }

    return current;
}

int main(void) {
    int n;
    printf("Enter board length n (3 x n board): ");
    if (scanf("%d", &n) != 1) {
        printf("Invalid input.\n");
        return 1;
    }

    if (n < 0) {
        printf("Board length cannot be negative.\n");
        return 0;
    }

    int result = count_domino_tilings(n);
    printf("Number of ways to tile a 3 x %d board: %d\n", n, result);

    return 0;
}
