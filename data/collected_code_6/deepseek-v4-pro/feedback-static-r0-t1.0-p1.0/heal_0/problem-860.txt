#include <stdio.h>
#include <string.h>
#include <regex.h>

int ends_with_alphanumeric(const char *str) {
    regex_t regex;
    int ret;
    int result = 0;
    const char *pattern = ".*[[:alnum:]]$";
    
    if (str == NULL) {
        return 0;
    }
    
    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return 0;
    }
    
    ret = regexec(&regex, str, 0, NULL, 0);
    if (ret == 0) {
        result = 1;
    }
    
    regfree(&regex);
    return result;
}

int main(void) {
    const char *test1 = "HelloWorld";
    const char *test2 = "Hello123";
    const char *test3 = "Hello!";
    const char *test4 = "Hello World";
    const char *test5 = "123";
    const char *test6 = "";
    
    printf("\"%s\" ends with alphanumeric: %s\n", test1, 
           ends_with_alphanumeric(test1) ? "Yes" : "No");
    printf("\"%s\" ends with alphanumeric: %s\n", test2, 
           ends_with_alphanumeric(test2) ? "Yes" : "No");
    printf("\"%s\" ends with alphanumeric: %s\n", test3, 
           ends_with_alphanumeric(test3) ? "Yes" : "No");
    printf("\"%s\" ends with alphanumeric: %s\n", test4, 
           ends_with_alphanumeric(test4) ? "Yes" : "No");
    printf("\"%s\" ends with alphanumeric: %s\n", test5, 
           ends_with_alphanumeric(test5) ? "Yes" : "No");
    printf("\"%s\" ends with alphanumeric: %s\n", test6, 
           ends_with_alphanumeric(test6) ? "Yes" : "No");
    
    return 0;
}
