#include <stdio.h>
#include <stdlib.h>

int getMaxGold(const int *mine, int rows, int cols);
int maxOfThree(int a, int b, int c);

int maxOfThree(int a, int b, int c)
{
    int max = a;
    if (b > max)
    {
        max = b;
    }
    if (c > max)
    {
        max = c;
    }
    return max;
}

int getMaxGold(const int *mine, int rows, int cols)
{
    int *table = NULL;
    int result = 0;
    int row = 0;
    int col = 0;

    if (mine == NULL || rows <= 0 || cols <= 0)
    {
        return 0;
    }

    table = (int *)calloc((size_t)rows * (size_t)cols, sizeof(int));
    if (table == NULL)
    {
        return -1;
    }

    for (col = cols - 1; col >= 0; col--)
    {
        for (row = 0; row < rows; row++)
        {
            int right = 0;
            int rightUp = 0;
            int rightDown = 0;

            if (col != cols - 1)
            {
                right = table[row * cols + col + 1];
                if (row > 0)
                {
                    rightUp = table[(row - 1) * cols + col + 1];
                }
                if (row < rows - 1)
                {
                    rightDown = table[(row + 1) * cols + col + 1];
                }
            }

            table[row * cols + col] = mine[row * cols + col] +
                                      maxOfThree(right, rightUp, rightDown);
        }
    }

    result = table[0];
    for (row = 1; row < rows; row++)
    {
        if (table[row * cols] > result)
        {
            result = table[row * cols];
        }
    }

    free(table);
    table = NULL;

    return result;
}

int main(void)
{
    int mine[4][4] = {
        {1, 3, 1, 5},
        {2, 2, 4, 1},
        {5, 0, 2, 3},
        {0, 6, 1, 2}
    };
    int maxGold = 0;

    maxGold = getMaxGold(&mine[0][0], 4, 4);
    if (maxGold < 0)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum gold collected: %d\n", maxGold) < 0)
    {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
