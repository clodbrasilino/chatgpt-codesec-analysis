```c
#include <stdio.h>
#include <stdlib.h>

int min_number_to_make_sum_even(const int *arr, size_t n)
{
    size_t i;
    int parity = 0;

    if (arr == NULL) {
        return -1;
    }

    for (i = 0; i < n; i++) {
        parity ^= (arr[i] & 1);
    }

    if (parity == 1) {
        return 1;
    }
    return 2;
}

int main(void)
{
    int arr[] = {1, 2, 3, 4, 5};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result;

    result = min_number_to_make_sum_even(arr, n);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input array\n");
        return EXIT_FAILURE;
    }

    if (printf("Minimum number to add: %d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
