#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node* create_node(int value) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        exit(EXIT_FAILURE);
    }
    new_node->value = value;
    new_node->next = NULL;
    return new_node;
}

void add_to_list(Node **list, int value) {
    Node *new_node = create_node(value);
    if (*list == NULL) {
        *list = new_node;
    } else {
        Node *current = *list;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
}

Node* list_difference(Node *list1, Node *list2) {
    Node *result = NULL;
    for (Node *current1 = list1; current1 != NULL; current1 = current1->next) {
        int found = 0;
        for (Node *current2 = list2; current2 != NULL && !found; current2 = current2->next) {
            if (current2->value == current1->value) {
                found = 1;
            }
        }
        if (!found) {
            add_to_list(&result, current1->value);
        }
    }
    return result;
}

void print_list(Node *list) {
    for (Node *current = list; current != NULL; current = current->next) {
        printf("%d -> ", current->value);
    }
    printf("NULL\n");
}

void free_list(Node *list) {
    Node *current = list;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int main() {
    Node *list1 = NULL;
    Node *list2 = NULL;
    
    add_to_list(&list1, 1);
    add_to_list(&list1, 2);
    add_to_list(&list1, 3);
    add_to_list(&list1, 4);
    
    add_to_list(&list2, 2);
    add_to_list(&list2, 3);
    add_to_list(&list2, 5);
    
    print_list(list1);
    print_list(list2);
    
    Node *diff = list_difference(list1, list2);
    print_list(diff);
    
    free_list(list1);
    free_list(list2);
    free_list(diff);
    
    return 0;
}
