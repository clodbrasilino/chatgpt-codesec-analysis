#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

#define MAX_KEY_LEN 32

typedef struct {
    char key[MAX_KEY_LEN];
    int value;
} DictItem;

typedef struct {
    DictItem *items;
    size_t size;
    size_t capacity;
} Dict;

int dict_init(Dict *dict, size_t capacity);
int dict_add(Dict *dict, const char *key, int value);
int dict_sum(const Dict *dict, long *result);
void dict_free(Dict *dict);

int dict_init(Dict *dict, size_t capacity)
{
    if (dict == NULL || capacity == 0) {
        return -1;
    }
    if (capacity > SIZE_MAX / sizeof(DictItem)) {
        return -1;
    }
    dict->items = calloc(capacity, sizeof(DictItem));
    if (dict->items == NULL) {
        return -1;
    }
    dict->size = 0;
    dict->capacity = capacity;
    return 0;
}

int dict_add(Dict *dict, const char *key, int value)
{
    if (dict == NULL || key == NULL || dict->items == NULL) {
        return -1;
    }
    if (dict->size >= dict->capacity) {
        return -1;
    }

    size_t key_len = strnlen(key, MAX_KEY_LEN);
    if (key_len >= MAX_KEY_LEN) {
        return -1;
    }

    DictItem *item = &dict->items[dict->size];
    if (key_len >= sizeof(item->key)) {
        return -1;
    }

    memset(item->key, 0, sizeof(item->key));
    memcpy(item->key, key, key_len);
    item->key[sizeof(item->key) - 1] = '\0';
    item->value = value;
    dict->size++;
    return 0;
}

int dict_sum(const Dict *dict, long *result)
{
    if (dict == NULL || result == NULL || dict->items == NULL) {
        return -1;
    }
    long sum = 0;
    for (size_t i = 0; i < dict->size; i++) {
        long v = (long)dict->items[i].value;
        if ((v > 0 && sum > LONG_MAX - v) ||
            (v < 0 && sum < LONG_MIN - v)) {
            return -1;
        }
        sum += v;
    }
    *result = sum;
    return 0;
}

void dict_free(Dict *dict)
{
    if (dict != NULL) {
        free(dict->items);
        dict->items = NULL;
        dict->size = 0;
        dict->capacity = 0;
    }
}

int main(void)
{
    Dict dict;
    long sum = 0;

    if (dict_init(&dict, 10) != 0) {
        fprintf(stderr, "Failed to initialize dictionary\n");
        return EXIT_FAILURE;
    }

    if (dict_add(&dict, "apple", 100) != 0 ||
        dict_add(&dict, "banana", 200) != 0 ||
        dict_add(&dict, "cherry", 300) != 0) {
        fprintf(stderr, "Failed to add items\n");
        dict_free(&dict);
        return EXIT_FAILURE;
    }

    if (dict_sum(&dict, &sum) != 0) {
        fprintf(stderr, "Failed to compute sum\n");
        dict_free(&dict);
        return EXIT_FAILURE;
    }

    printf("Sum of all items: %ld\n", sum);

    dict_free(&dict);
    return EXIT_SUCCESS;
}

The code mitigates it properly because every write to `item->key` is bounds-checked before it happens. The fixed-size buffer itself is not a defect as long as all accesses are validated — which they are.

**2. Flawfinder warning on `memcpy` (CWE-120):** This is a false positive here. The copy is safe because of the layered guards preceding it:

- `strnlen(key, MAX_KEY_LEN)` bounds the length measurement itself, so even a non-terminated source string cannot cause an over-read beyond `MAX_KEY_LEN`.
- `key_len >= MAX_KEY_LEN` rejects any key that would fill or overflow the buffer, guaranteeing room for the NUL terminator.
- The redundant `key_len >= sizeof(item->key)` check defends against future divergence between `MAX_KEY_LEN` and the actual field size.
- `memset` pre-zeroes the destination and the explicit final `'\0'` write guarantees termination regardless.

**Other defensive measures already present:**

- `dict_init` checks `capacity > SIZE_MAX / sizeof(DictItem)`, preventing integer overflow in the allocation size, and uses `calloc` for zero-initialized memory.
- `dict_add` checks `size >= capacity`, preventing out-of-bounds writes into the items array.
- `dict_sum` performs pre-addition overflow checks against `LONG_MAX`/`LONG_MIN`, avoiding signed overflow (undefined behavior in C).
- `dict_free` nulls the pointer and resets counters, mitigating use-after-free/double-free risks.
- All public functions validate their pointer arguments before dereferencing.
