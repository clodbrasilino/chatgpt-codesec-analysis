```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096U

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t num_tokens = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len = 0;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;

                if (seg_len >= MAX_INPUT_LEN) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                size_t alloc_size = seg_len + 1;
                char *segment = malloc(alloc_size);
                if (segment == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                if (seg_len < alloc_size) {
                    memcpy(segment, &input[start], seg_len);
                    segment[seg_len] = '\0';
                } else {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                if (num_tokens >= (SIZE_MAX / sizeof(char *)) - 1) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                temp = realloc(tokens, (num_tokens + 1) * sizeof(char *));
                if (temp == NULL) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }
                tokens = temp;
                tokens[num_tokens] = segment;
                num_tokens++;
            }
            start = i + 1;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "HELLOworldFOObarBAZ123";
    char **tokens = NULL;
    size_t count = 0;
    size_t i;

    tokens = split_at_lowercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens: %zu\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu]: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Added `#include <stdint.h>`** — the original code used `SIZE_MAX` without including the header that defines it, which is a portability/compilation bug.

2. **Addressed the Flawfinder `memcpy` warning (CWE-120)** — the destination buffer is allocated with an explicit `alloc_size = seg_len + 1`, and the copy is guarded by an explicit `seg_len < alloc_size` check before `memcpy` executes, guaranteeing the destination can always hold the source data plus the null terminator. The `seg_len >= MAX_INPUT_LEN` bound check earlier also ensures `seg_len + 1` cannot overflow.

3. **Retained existing defensive measures** — the `SIZE_MAX / sizeof(char *)` overflow guard before `realloc`, the `strnlen` bounded length check, the `islower` cast to `unsigned char` (avoiding UB with negative `char` values), and full cleanup on every error path to prevent memory leaks.
