#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define INITIAL_INPUT_CAPACITY 128u
#define MAX_INPUT_LENGTH ((size_t)1024u * 1024u)

static char *read_line(FILE *stream, size_t max_len)
{
    size_t capacity = INITIAL_INPUT_CAPACITY;
    size_t length = 0;
    int ch;
    char *buffer;

    if (stream == NULL || max_len == 0 || max_len >= SIZE_MAX) {
        return NULL;
    }

    if (capacity > max_len + 1u) {
        capacity = max_len + 1u;
    }

    buffer = (char *)malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= max_len) {
            free(buffer);
            return NULL;
        }

        if (length + 1u >= capacity) {
            size_t new_capacity;
            char *new_buffer;

            if (capacity > (max_len + 1u) / 2u) {
                new_capacity = max_len + 1u;
            } else {
                new_capacity = capacity * 2u;
            }

            if (new_capacity <= capacity || new_capacity > max_len + 1u) {
                free(buffer);
                return NULL;
            }

            new_buffer = (char *)realloc(buffer, new_capacity);
            if (new_buffer == NULL) {
                free(buffer);
                return NULL;
            }

            buffer = new_buffer;
            capacity = new_capacity;
        }

        if (length >= capacity || capacity - length < 2u) {
            free(buffer);
            return NULL;
        }

        buffer[length] = (char)ch;
        length++;
    }

    if (ferror(stream)) {
        free(buffer);
        return NULL;
    }

    if (ch == EOF && length == 0) {
        free(buffer);
        return NULL;
    }

    if (length >= capacity) {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';
    return buffer;
}

char *remove_words_of_length(const char *str, size_t k, size_t str_max_len)
{
    size_t len;
    size_t result_capacity;
    char *result;
    size_t i = 0;
    size_t j = 0;
    int first_word = 1;

    if (str == NULL || str_max_len == 0) {
        return NULL;
    }

    len = strnlen(str, str_max_len);
    if (len >= str_max_len || len == SIZE_MAX) {
        return NULL;
    }

    result_capacity = len + 1u;

    result = (char *)malloc(result_capacity);
    if (result == NULL) {
        return NULL;
    }

    while (i < len) {
        size_t word_start;
        size_t word_len;
        size_t prefix;
        size_t needed;

        while (i < len && isspace((unsigned char)str[i])) {
            i++;
        }
        if (i >= len) {
            break;
        }

        word_start = i;
        while (i < len && !isspace((unsigned char)str[i])) {
            i++;
        }
        word_len = i - word_start;

        if (word_len == k) {
            continue;
        }

        prefix = first_word ? 0u : 1u;
        needed = prefix + word_len;

        if (needed < prefix || j >= result_capacity || needed >= result_capacity - j) {
            free(result);
            return NULL;
        }

        if (!first_word) {
            result[j] = ' ';
            j++;
        }

        if (word_len > 0u) {
            if (word_start >= len || word_len > len - word_start) {
                free(result);
                return NULL;
            }

            if (j >= result_capacity || word_len >= result_capacity - j) {
                free(result);
                return NULL;
            }

            memcpy(result + j, str + word_start, word_len);
            j += word_len;
        }

        first_word = 0;
    }

    if (j >= result_capacity) {
        free(result);
        return NULL;
    }
    result[j] = '\0';

    {
        char *shrunk = (char *)realloc(result, j + 1u);
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    return result;
}

int main(void)
{
    size_t k = 0;
    char *input;
    size_t input_len;
    char *modified;

    printf("Enter a string: ");
    fflush(stdout);

    input = read_line(stdin, MAX_INPUT_LENGTH);
    if (input == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the word length to remove: ");
    fflush(stdout);

    if (scanf("%zu", &k) != 1) {
        fprintf(stderr, "Invalid length value.\n");
        free(input);
        return EXIT_FAILURE;
    }

    input_len = strnlen(input, MAX_INPUT_LENGTH + 1u);
    if (input_len > MAX_INPUT_LENGTH) {
        fprintf(stderr, "Input string is not properly terminated.\n");
        free(input);
        return EXIT_FAILURE;
    }

    modified = remove_words_of_length(input, k, input_len + 1u);
    if (modified == NULL) {
        fprintf(stderr, "Failed to process the string.\n");
        free(input);
        return EXIT_FAILURE;
    }

    printf("Result: \"%s\"\n", modified);

    free(modified);
    free(input);
    return EXIT_SUCCESS;
}
