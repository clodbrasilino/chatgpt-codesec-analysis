#include <stdio.h>

long long count_sequences(int m, int n) {
    long long dp[n + 1][m + 1];
    for (int j = 1; j <= m; ++j) {
        dp[1][j] = 1;
    }
    for (int i = 1; i <= n; ++i) {
        dp[i][0] = 0;
    }
    for (int i = 2; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            dp[i][j] = 0;
            for (int k = 1; k * 2 <= j; ++k) {
                dp[i][j] += dp[i - 1][k];
            }
        }
    }
    long long result = 0;
    for (int j = 1; j <= m; ++j) {
        result += dp[n][j];
    }
    return result;
}

int main() {
    int m = 10;
    int n = 4;
    long long result = count_sequences(m, n);
    printf("Number of sequences: %lld\n", result);
    return 0;
}
