```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int match_word_with_z(const char *text, char *result, size_t result_size);

int match_word_with_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    while (text[i] != '\0') {
        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }
        if (text[i] == '\0') {
            break;
        }

        {
            size_t start = i;
            int found = 0;

            while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
                if (tolower((unsigned char)text[i]) == 'z') {
                    found = 1;
                }
                i++;
            }

            if (found) {
                size_t len = i - start;
                if (len >= result_size) {
                    return -1;
                }
                memcpy(result, &text[start], len);
                result[len] = '\0';
                return 1;
            }
        }
    }
    return 0;
}

int main(void)
{
    const char *sentences[] = {
        "The lazy dog sleeps all day.",
        "A quick brown fox jumps.",
        "Zebras graze in the savanna."
    };
    char word[256];
    size_t i;

    for (i = 0; i < sizeof(sentences) / sizeof(sentences[0]); i++) {
        int status = match_word_with_z(sentences[i], word, sizeof(word));
        if (status < 0) {
            fprintf(stderr, "Error processing input: %s\n", sentences[i]);
            return EXIT_FAILURE;
        }
        if (status == 1) {
            if (printf("Input: \"%s\" -> matched word: \"%s\"\n", sentences[i], word) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("Input: \"%s\" -> no word containing 'z' found\n", sentences[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

1. **`variableScope` (cppcheck) in `match_word_with_z`**: The variables `start`, `found`, and `len` were declared at function scope but only used inside the outer `while` loop. They are now declared in a block inside the loop, minimizing their scope and preventing accidental reuse of stale values across iterations.

2. **`variableScope` in `main`**: The `status` variable is now declared inside the `for` loop where it is actually used.

3. **Flawfinder `memcpy` warning (CWE-120)**: This was already mitigated in the original code — the `len >= result_size` check guarantees the destination buffer can hold `len` bytes plus the NUL terminator before `memcpy` is called. The check is preserved as-is, so the warning is a false positive here.

4. **Flawfinder `char` static array warning (CWE-119/CWE-120)**: The `word[256]` buffer is safe because its size is always passed via `sizeof(word)` to `match_word_with_z`, which performs strict bounds checking. No change required beyond keeping that discipline.
