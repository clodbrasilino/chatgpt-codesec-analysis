#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int id;
    double price;
} Item;

typedef struct {
    Item *data;
    int capacity;
    int size;
} MinHeap;

MinHeap* create_heap(int capacity) {
    MinHeap *heap = (MinHeap*)malloc(sizeof(MinHeap));
    if (heap == NULL) return NULL;
    heap->data = (Item*)malloc(capacity * sizeof(Item));
    if (heap->data == NULL) {
        free(heap);
        return NULL;
    }
    heap->capacity = capacity;
    heap->size = 0;
    return heap;
}

void destroy_heap(MinHeap *heap) {
    if (heap != NULL) {
        free(heap->data);
        free(heap);
    }
}

void swap_items(Item *a, Item *b) {
    Item temp = *a;
    *a = *b;
    *b = temp;
}

void heapify_up(MinHeap *heap, int index) {
    while (index > 0) {
        int parent = (index - 1) / 2;
        if (heap->data[index].price >= heap->data[parent].price) break;
        swap_items(&heap->data[index], &heap->data[parent]);
        index = parent;
    }
}

void heapify_down(MinHeap *heap, int index) {
    while (1) {
        int smallest = index;
        int left = 2 * index + 1;
        int right = 2 * index + 2;
        if (left < heap->size && heap->data[left].price < heap->data[smallest].price) smallest = left;
        if (right < heap->size && heap->data[right].price < heap->data[smallest].price) smallest = right;
        if (smallest == index) break;
        swap_items(&heap->data[index], &heap->data[smallest]);
        index = smallest;
    }
}

int heap_insert(MinHeap *heap, Item item) {
    if (heap == NULL || heap->size == heap->capacity) return 0;
    heap->data[heap->size] = item;
    heap->size++;
    heapify_up(heap, heap->size - 1);
    return 1;
}

int heap_extract_min(MinHeap *heap, Item *out) {
    if (heap == NULL || heap->size == 0) return 0;
    if (out != NULL) *out = heap->data[0];
    heap->data[0] = heap->data[heap->size - 1];
    heap->size--;
    heapify_down(heap, 0);
    return 1;
}

Item* find_n_expensive(Item *dataset, int dataset_size, int n, int *result_size) {
    if (dataset == NULL || dataset_size <= 0 || n <= 0) {
        *result_size = 0;
        return NULL;
    }
    if (n > dataset_size) n = dataset_size;
    
    MinHeap *heap = create_heap(n);
    if (heap == NULL) {
        *result_size = 0;
        return NULL;
    }
    
    for (int i = 0; i < dataset_size; i++) {
        if (heap->size < heap->capacity) {
            heap_insert(heap, dataset[i]);
        } else if (dataset[i].price > heap->data[0].price) {
            Item discarded;
            heap_extract_min(heap, &discarded);
            heap_insert(heap, dataset[i]);
        }
    }
    
    Item *result = (Item*)malloc(heap->size * sizeof(Item));
    if (result == NULL) {
        destroy_heap(heap);
        *result_size = 0;
        return NULL;
    }
    
    *result_size = heap->size;
    int current_size = heap->size;
    for (int i = current_size - 1; i >= 0; i--) {
        heap_extract_min(heap, &result[i]);
    }
    
    destroy_heap(heap);
    return result;
}

int main() {
    Item dataset[] = {
        {1, 10.5},
        {2, 25.0},
        {3, 15.75},
        {4, 50.0},
        {5, 5.25},
        {6, 35.1},
        {7, 20.0}
    };
    int dataset_size = sizeof(dataset) / sizeof(dataset[0]);
    int n = 3;
    int result_size = 0;
    
    Item *expensive_items = find_n_expensive(dataset, dataset_size, n, &result_size);
    
    if (expensive_items != NULL && result_size > 0) {
        for (int i = 0; i < result_size; i++) {
            printf("ID: %d, Price: %.2f\n", expensive_items[i].id, expensive_items[i].price);
        }
        free(expensive_items);
    }
    
    return 0;
}
