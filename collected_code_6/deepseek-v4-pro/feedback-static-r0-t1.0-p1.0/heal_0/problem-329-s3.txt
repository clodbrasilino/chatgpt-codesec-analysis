```c
#include <stdio.h>
#include <stdlib.h>

int count_negative_numbers(const int *list, size_t size) {
    int count = 0;
    size_t i;
    
    if (list == NULL) {
        return -1;
    }
    
    for (i = 0; i < size; i++) {
        if (list[i] < 0) {
            count++;
        }
    }
    
    return count;
}

int main(void) {
    int test_list[] = {-3, 5, -8, 0, -1, 7, -2, 4};
    size_t size = sizeof(test_list) / sizeof(test_list[0]);
    int result = count_negative_numbers(test_list, size);
    
    if (result >= 0) {
        printf("Number of negative numbers: %d\n", result);
        return 0;
    }
    
    return 1;
}
```
