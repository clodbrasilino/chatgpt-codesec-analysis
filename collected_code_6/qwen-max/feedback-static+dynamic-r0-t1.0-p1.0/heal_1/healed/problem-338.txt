```c
#include <stdio.h>
#include <string.h>

int countSubstringsWithSameFirstLast(const char *s, size_t len) {
    int count = 0;
    for (size_t i = 0; i < len; ++i) {
        for (size_t j = i + 1; j <= len; ++j) {
            if (s[i] == s[j - 1]) {
                count++;
            }
        }
    }
    return count;
}

int main() {
    const char str[] = "abcab";
    size_t len = strlen(str);
    int result = countSubstringsWithSameFirstLast(str, len);
    printf("%d\n", result);
    return 0;
}
```
