#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int count_pairs_with_diff_k(const int *arr, size_t n, int k, size_t *result)
{
    int *sorted;
    size_t i;
    size_t j;
    size_t count;
    size_t bytes;

    if (result == NULL) {
        return -1;
    }

    *result = 0;

    if (arr == NULL || n < 2 || k < 0) {
        return (arr == NULL && n > 0) ? -1 : 0;
    }

    if (n > SIZE_MAX / sizeof(*sorted)) {
        return -1;
    }

    bytes = n * sizeof(*sorted);

    sorted = malloc(bytes);
    if (sorted == NULL) {
        return -1;
    }

    memcpy(sorted, arr, bytes);
    qsort(sorted, n, sizeof(*sorted), compare_ints);

    count = 0;
    i = 0;
    j = 1;

    while (i < n && j < n) {
        if (i == j) {
            j++;
            continue;
        }

        long diff = (long)sorted[j] - (long)sorted[i];

        if (diff == (long)k) {
            count++;
            i++;
            j++;
            while (i < n && sorted[i] == sorted[i - 1]) {
                i++;
            }
            while (j < n && sorted[j] == sorted[j - 1]) {
                j++;
            }
        } else if (diff < (long)k) {
            j++;
        } else {
            i++;
        }
    }

    free(sorted);
    *result = count;
    return 0;
}

int main(void)
{
    int arr[] = {1, 5, 3, 4, 2, 8, 8, 6};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int k = 2;
    size_t pairs = 0;

    if (count_pairs_with_diff_k(arr, n, k, &pairs) != 0) {
        fprintf(stderr, "Error counting pairs\n");
        return EXIT_FAILURE;
    }

    if (printf("Count of distinct pairs with difference %d: %zu\n", k, pairs) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
