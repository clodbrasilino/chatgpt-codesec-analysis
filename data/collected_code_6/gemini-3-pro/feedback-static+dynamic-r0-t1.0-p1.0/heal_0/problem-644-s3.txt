#include <stdio.h>
#include <stdlib.h>

void reverse_array_upto_position(int *arr, size_t length, size_t position) {
    if (arr == NULL || length == 0 || position == 0) {
        return;
    }

    if (position > length) {
        position = length;
    }

    size_t start = 0;
    size_t end = position - 1;

    while (start < end) {
        int temp = arr[start];
        arr[start] = arr[end];
        arr[end] = temp;
        start++;
        end--;
    }
}

int main(void) {
    size_t length = 6;
    int *arr = (int *)malloc(length * sizeof(int));
    
    if (arr == NULL) {
        return 1;
    }

    arr[0] = 1;
    arr[1] = 2;
    arr[2] = 3;
    arr[3] = 4;
    arr[4] = 5;
    arr[5] = 6;

    size_t position = 4;

    reverse_array_upto_position(arr, length, position);

    for (size_t i = 0; i < length; ++i) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    free(arr);
    return 0;
}
