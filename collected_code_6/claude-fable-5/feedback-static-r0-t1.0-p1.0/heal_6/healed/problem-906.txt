```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "/([0-9]{4})/([0-9]{1,2})/([0-9]{1,2})(/|$)"
#define MAX_FIELD_LEN 8
#define ERRBUF_LEN 128
#define MATCH_COUNT 5
#define MAX_URL_LEN 2048

typedef struct {
    char year[MAX_FIELD_LEN];
    char month[MAX_FIELD_LEN];
    char day[MAX_FIELD_LEN];
} UrlDate;

static int copy_match(const char *source, size_t source_len,
                      const regmatch_t *match,
                      char *dest, size_t dest_size)
{
    size_t length;

    if (source == NULL || match == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }
    if (match->rm_so < 0 || match->rm_eo < match->rm_so) {
        return -1;
    }
    if ((size_t)match->rm_eo > source_len) {
        return -1;
    }

    length = (size_t)(match->rm_eo - match->rm_so);
    if (length >= dest_size) {
        return -1;
    }

    memcpy(dest, source + match->rm_so, length);
    dest[length] = '\0';
    return 0;
}

int extract_date_from_url(const char *url, UrlDate *result)
{
    regex_t regex;
    regmatch_t matches[MATCH_COUNT];
    size_t url_len;
    int ret;

    if (url == NULL || result == NULL) {
        return -1;
    }

    memset(result, 0, sizeof(*result));

    url_len = strnlen(url, MAX_URL_LEN);
    if (url_len >= MAX_URL_LEN) {
        return -1;
    }

    ret = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        memset(errbuf, 0, sizeof(errbuf));
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, url, MATCH_COUNT, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (copy_match(url, url_len, &matches[1], result->year, sizeof(result->year)) != 0 ||
        copy_match(url, url_len, &matches[2], result->month, sizeof(result->month)) != 0 ||
        copy_match(url, url_len, &matches[3], result->day, sizeof(result->day)) != 0) {
        memset(result, 0, sizeof(*result));
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *urls[] = {
        "https://example.com/blog/2023/07/15/some-article",
        "https://news.site.org/2019/1/9",
        "https://example.com/no-date-here"
    };
    size_t count = sizeof(urls) / sizeof(urls[0]);
    size_t i;
    UrlDate date;

    for (i = 0; i < count; i++) {
        if (extract_date_from_url(urls[i], &date) == 0) {
            if (printf("URL: %s\n  Year: %s, Month: %s, Day: %s\n",
                       urls[i], date.year, date.month, date.day) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("URL: %s\n  No date found\n", urls[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **Statically-sized arrays in `UrlDate` (CWE-119/CWE-120):** The `year`, `month`, and `day` buffers are fixed at `MAX_FIELD_LEN` (8 bytes). This is safe because the regex pattern constrains matches to at most 4 digits (`[0-9]{4}`) and the `copy_match` helper enforces `length >= dest_size` rejection before any write, guaranteeing room for the NUL terminator.

2. **`memcpy` in `copy_match` (CWE-120):** The warning is a false positive here. Every precondition is validated before the copy: null pointers, negative or inverted match offsets, matches extending past the source length, and destination capacity including the terminator. The copy can never exceed `dest_size - 1` bytes.

3. **`errbuf` static array (CWE-119/CWE-120):** Safe because `regerror` is passed `sizeof(errbuf)` and truncates internally; the explicit final NUL assignment is defensive redundancy.

4. **Additional hardening already present:** `strnlen` with `MAX_URL_LEN` bounds untrusted input length, `memset` zeroes the result struct on both entry and failure paths (preventing use of stale partial data), and `regfree` is called on all exit paths after successful `regcomp`, avoiding resource leaks.

The code compiles cleanly with `-Wall -Wextra -pedantic` and requires no functional changes — the identified findings were properly mitigated in the original implementation.
