#include <stdio.h>
#include <stdlib.h>

void extract_k_elements(int *arr, int n, int k, int **max_elements, int **min_elements) {
    if (k > n) return;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n - 1 - i; ++j) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
    *max_elements = (int *)malloc(k * sizeof(int));
    *min_elements = (int *)malloc(k * sizeof(int));
    for (int i = 0; i < k; ++i) {
        (*max_elements)[i] = arr[n - 1 - i];
        (*min_elements)[i] = arr[i];
    }
}

int main() {
    int tuple[] = {5, 3, 9, 1, 7, 2, 8, 4, 6};
    int n = 9, k = 3;
    int *max_elements, *min_elements;

    extract_k_elements(tuple, n, k, &max_elements, &min_elements);

    printf("Maximum k elements: ");
    for (int i = 0; i < k; ++i) {
        printf("%d ", max_elements[i]);
    }
    printf("\nMinimum k elements: ");
    for (int i = 0; i < k; ++i) {
        printf("%d ", min_elements[i]);
    }

    free(max_elements);
    free(min_elements);

    return 0;
}
