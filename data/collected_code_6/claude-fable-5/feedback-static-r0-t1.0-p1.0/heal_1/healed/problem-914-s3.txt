```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096U

bool is_alternating(const char *str)
{
    size_t len;
    size_t i;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return false;
    }

    if (len < 2U) {
        return false;
    }

    if (str[0] == str[1]) {
        return false;
    }

    for (i = 2U; i < len; i++) {
        if (str[i] != str[i - 2U]) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *tests[] = {
        "ababab",
        "xyxyxy",
        "aaaa",
        "abcabc",
        "ab",
        "a",
        "",
        "xyxyx"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        if (is_alternating(tests[i])) {
            if (printf("\"%s\": yes\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("\"%s\": no\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Key change:** The flagged `strlen` call (CWE-126, potential over-read on non-NUL-terminated input) has been replaced with `strnlen` bounded by `MAX_INPUT_LEN`. This guarantees the scan never reads past a fixed upper bound, and inputs that reach the bound (i.e., are not provably NUL-terminated within the limit) are rejected safely by returning `false` rather than risking an over-read.

All other defensive properties of the original are preserved: the `NULL` check before any dereference, `size_t` indexing with unsigned literals to avoid signed/unsigned mismatch, the checked `printf` return values, and use of `EXIT_SUCCESS`/`EXIT_FAILURE` for portable exit codes.

Note that `strnlen` is POSIX.1-2008 (and available on all mainstream platforms); if strict C99 portability were required, an equivalent manual bounded loop could be substituted.
