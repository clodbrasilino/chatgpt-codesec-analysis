```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    char *key;
    char *value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dict;

static char *dup_string(const char *src)
{
    size_t len;
    size_t alloc_size;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, SIZE_MAX - 1);
    if (len >= SIZE_MAX - 1) {
        return NULL;
    }

    alloc_size = len + 1;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0 && len < alloc_size) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static Dict *dict_create(size_t capacity)
{
    Dict *d;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(DictEntry)) {
        return NULL;
    }

    d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }

    d->entries = calloc(capacity, sizeof(DictEntry));
    if (d->entries == NULL) {
        free(d);
        return NULL;
    }

    d->size = 0;
    d->capacity = capacity;
    return d;
}

static int dict_add(Dict *d, const char *key, const char *value)
{
    char *key_copy;
    char *value_copy;

    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }

    if (d->size >= d->capacity) {
        return -1;
    }

    key_copy = dup_string(key);
    if (key_copy == NULL) {
        return -1;
    }

    value_copy = dup_string(value);
    if (value_copy == NULL) {
        free(key_copy);
        return -1;
    }

    d->entries[d->size].key = key_copy;
    d->entries[d->size].value = value_copy;
    d->size++;
    return 0;
}

static const char *dict_key_by_index(const Dict *d, size_t index)
{
    if (d == NULL) {
        return NULL;
    }

    if (index >= d->size) {
        return NULL;
    }

    return d->entries[index].key;
}

static const char *dict_value_by_index(const Dict *d, size_t index)
{
    if (d == NULL) {
        return NULL;
    }

    if (index >= d->size) {
        return NULL;
    }

    return d->entries[index].value;
}

static void dict_destroy(Dict *d)
{
    size_t i;

    if (d == NULL) {
        return;
    }

    for (i = 0; i < d->size; i++) {
        free(d->entries[i].key);
        free(d->entries[i].value);
    }

    free(d->entries);
    free(d);
}

int main(void)
{
    Dict *d;
    const char *key;
    size_t i;

    d = dict_create(10);
    if (d == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }

    if (dict_add(d, "apple", "fruit") != 0 ||
        dict_add(d, "carrot", "vegetable") != 0 ||
        dict_add(d, "salmon", "fish") != 0) {
        fprintf(stderr, "Failed to add entries\n");
        dict_destroy(d);
        return EXIT_FAILURE;
    }

    for (i = 0; i < d->size; i++) {
        const char *value;

        key = dict_key_by_index(d, i);
        value = dict_value_by_index(d, i);
        if (key == NULL || value == NULL) {
            fprintf(stderr, "Invalid index access\n");
            dict_destroy(d);
            return EXIT_FAILURE;
        }
        printf("Index %zu: key=%s, value=%s\n", i, key, value);
    }

    key = dict_key_by_index(d, 100);
    if (key == NULL) {
        printf("Index 100 is out of range\n");
    }

    dict_destroy(d);
    return EXIT_SUCCESS;
}
```

Analysis notes on the flagged weakness:

The Flawfinder warning on `memcpy` (CWE-120) was a false positive in the original code. The destination buffer `copy` is allocated with `alloc_size = len + 1` bytes, and exactly `len` bytes are copied, so the destination always holds the source data with room for the null terminator. The rewrite adds an explicit `len < alloc_size` guard before the `memcpy` call, making the bounds relationship between source length and destination capacity verifiable at the call site, which satisfies static analyzers and documents the invariant defensively.

The rest of the code was already robust and required no changes:

- `dup_string` uses `strnlen` with a bounded maximum, guards against `SIZE_MAX` overflow when computing `alloc_size`, and checks the `malloc` result.
- `dict_create` prevents integer overflow in the `calloc` sizing via the `SIZE_MAX / sizeof(DictEntry)` check and cleans up partial allocations on failure.
- `dict_add` validates all pointers, enforces the capacity bound, and frees `key_copy` if the value duplication fails, avoiding a memory leak.
- Index accessors bounds-check against `d->size`, preventing out-of-range reads.
- `dict_destroy` is null-safe and frees all owned allocations exactly once, with no double-free or leak paths.
