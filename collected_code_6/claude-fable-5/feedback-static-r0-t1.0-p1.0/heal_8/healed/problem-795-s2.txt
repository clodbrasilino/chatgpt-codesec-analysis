#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define ITEM_NAME_LEN 64

typedef struct {
    char name[ITEM_NAME_LEN];
    double price;
} Item;

static void swap_items(Item *a, Item *b)
{
    Item tmp = *a;
    *a = *b;
    *b = tmp;
}

static void sift_down(Item *heap, size_t size, size_t idx)
{
    for (;;) {
        size_t smallest = idx;
        size_t left = 2U * idx + 1U;
        size_t right = 2U * idx + 2U;

        if (left < size && heap[left].price < heap[smallest].price) {
            smallest = left;
        }
        if (right < size && heap[right].price < heap[smallest].price) {
            smallest = right;
        }
        if (smallest == idx) {
            break;
        }
        swap_items(&heap[idx], &heap[smallest]);
        idx = smallest;
    }
}

static void build_min_heap(Item *heap, size_t size)
{
    if (size < 2U) {
        return;
    }
    for (size_t i = size / 2U; i > 0U; i--) {
        sift_down(heap, size, i - 1U);
    }
}

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0U;
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

static int item_set_name(Item *item, const char *name)
{
    if (item == NULL || name == NULL) {
        return -1;
    }

    size_t len = bounded_strlen(name, sizeof(item->name));
    if (len >= sizeof(item->name)) {
        return -1;
    }

    memcpy(item->name, name, len);
    item->name[len] = '\0';
    return 0;
}

static int find_n_cheapest(const Item *items, size_t count, size_t n, Item *result, size_t result_capacity)
{
    if (items == NULL || result == NULL || n == 0U || n > count || n > result_capacity) {
        return -1;
    }

    if (count > SIZE_MAX / sizeof(Item)) {
        return -1;
    }

    Item *heap = malloc(count * sizeof(Item));
    if (heap == NULL) {
        return -1;
    }

    for (size_t i = 0U; i < count; i++) {
        heap[i] = items[i];
        heap[i].name[sizeof(heap[i].name) - 1U] = '\0';
    }

    build_min_heap(heap, count);

    size_t heap_size = count;
    for (size_t i = 0U; i < n; i++) {
        result[i] = heap[0];
        heap_size--;
        heap[0] = heap[heap_size];
        sift_down(heap, heap_size, 0U);
    }

    free(heap);
    return 0;
}

int main(void)
{
    static const struct {
        const char *name;
        double price;
    } raw_data[] = {
        {"Laptop", 999.99},
        {"Mouse", 19.99},
        {"Keyboard", 49.99},
        {"Monitor", 199.99},
        {"USB Cable", 5.99},
        {"Headphones", 89.99},
        {"Webcam", 39.99},
        {"Desk Lamp", 24.99}
    };

    size_t count = sizeof(raw_data) / sizeof(raw_data[0]);
    size_t n = 3U;

    Item dataset[sizeof(raw_data) / sizeof(raw_data[0])];
    memset(dataset, 0, sizeof(dataset));

    for (size_t i = 0U; i < count; i++) {
        if (item_set_name(&dataset[i], raw_data[i].name) != 0) {
            fprintf(stderr, "Item name too long: %s\n", raw_data[i].name);
            return EXIT_FAILURE;
        }
        dataset[i].price = raw_data[i].price;
    }

    if (n > SIZE_MAX / sizeof(Item)) {
        fprintf(stderr, "Requested size too large\n");
        return EXIT_FAILURE;
    }

    Item *cheapest = malloc(n * sizeof(Item));
    if (cheapest == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (find_n_cheapest(dataset, count, n, cheapest, n) != 0) {
        fprintf(stderr, "Failed to find cheapest items\n");
        free(cheapest);
        return EXIT_FAILURE;
    }

    printf("The %zu cheapest items are:\n", n);
    for (size_t i = 0U; i < n; i++) {
        printf("%s: $%.2f\n", cheapest[i].name, cheapest[i].price);
    }

    free(cheapest);
    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

**CWE-119/CWE-120 (statically-sized array `name[ITEM_NAME_LEN]`):** The fixed-size buffer is properly protected. All writes to it go through `item_set_name()`, which validates both pointers, measures the source with `bounded_strlen()` (capped at `sizeof(item->name)` so it never reads past a legitimate maximum), and rejects any name that would not fit including the null terminator (`len >= sizeof(item->name)`). Additionally, `find_n_cheapest()` defensively forces null termination on every copied item's name before use.

**CWE-120 (`memcpy` without bounds check):** The bounds check happens immediately before the `memcpy` call — `len` is guaranteed to be strictly less than `sizeof(item->name)`, so copying `len` bytes plus writing the terminator at `item->name[len]` can never overflow the destination.

Other defensive measures already present that are worth noting:

- **Integer overflow protection** on both `malloc` calls (`count > SIZE_MAX / sizeof(Item)` and the equivalent check for `n`), preventing an undersized allocation from a wrapped multiplication.
- **NULL checks** after every `malloc`, with proper cleanup (`free(cheapest)`) on the error path.
- **Parameter validation** in `find_n_cheapest()` (`n > count`, `n > result_capacity`, null pointers), preventing out-of-bounds reads/writes on the result buffer.
- **No use of unsafe functions** such as `strcpy`, `sprintf`, or `gets`.
