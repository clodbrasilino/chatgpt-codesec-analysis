#include <stdio.h>
#include <stdlib.h>

int findKthElement(const int *arr, size_t size, size_t k) {
    if (arr == NULL || k >= size) {
        fprintf(stderr, "Invalid input parameters.\n");
        exit(EXIT_FAILURE);
    }
    return arr[k];
}

int main(void) {
    int arr[] = {1, 3, 5, 7, 9, 2, 4, 6, 8, 10};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    size_t k = 4; 

    int kthElement = findKthElement(arr, size, k);
    printf("%d\n", kthElement);

    return EXIT_SUCCESS;
}
