#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} Tuple;

int init_tuple(Tuple *t, size_t initial_capacity) {
    if (t == NULL || initial_capacity == 0) {
        return -1;
    }
    t->data = malloc(initial_capacity * sizeof(int));
    if (t->data == NULL) {
        return -1;
    }
    t->size = 0;
    t->capacity = initial_capacity;
    return 0;
}

void free_tuple(Tuple *t) {
    if (t != NULL) {
        free(t->data);
        t->data = NULL;
        t->size = 0;
        t->capacity = 0;
    }
}

int add_list_to_tuple(Tuple *t, const int *list, size_t list_size) {
    if (t == NULL || list == NULL || list_size == 0) {
        return -1;
    }
    if (t->size + list_size > t->capacity) {
        size_t new_capacity = t->capacity * 2;
        while (new_capacity < t->size + list_size) {
            new_capacity *= 2;
        }
        int *new_data = realloc(t->data, new_capacity * sizeof(int));
        if (new_data == NULL) {
            return -1;
        }
        t->data = new_data;
        t->capacity = new_capacity;
    }
    for (size_t i = 0; i < list_size; i++) {
        t->data[t->size++] = list[i];
    }
    return 0;
}

int main(void) {
    Tuple t;
    if (init_tuple(&t, 4) != 0) {
        fprintf(stderr, "Failed to initialize tuple\n");
        return EXIT_FAILURE;
    }

    int list[] = {10, 20, 30, 40, 50};
    size_t list_size = sizeof(list) / sizeof(list[0]);

    if (add_list_to_tuple(&t, list, list_size) != 0) {
        fprintf(stderr, "Failed to add list to tuple\n");
        free_tuple(&t);
        return EXIT_FAILURE;
    }

    printf("Tuple contents: ");
    for (size_t i = 0; i < t.size; i++) {
        printf("%d ", t.data[i]);
    }
    printf("\n");

    free_tuple(&t);
    return EXIT_SUCCESS;
}
