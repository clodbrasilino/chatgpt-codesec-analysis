#include <stdio.h>
#include <stdlib.h>

int extract_odd_numbers(const int *input, size_t input_size, int **output, size_t *output_size) {
    if (input == NULL || output == NULL || output_size == NULL) {
        return -1;
    }

    *output_size = 0;
    for (size_t i = 0; i < input_size; ++i) {
        if (input[i] % 2 != 0) {
            (*output_size)++;
        }
    }

    if (*output_size == 0) {
        *output = NULL;
        return 0;
    }

    *output = (int *)malloc(*output_size * sizeof(int));
    if (*output == NULL) {
        *output_size = 0;
        return -1;
    }

    size_t index = 0;
    for (size_t i = 0; i < input_size; ++i) {
        if (input[i] % 2 != 0) {
            (*output)[index++] = input[i];
        }
    }

    return 0;
}

int main(void) {
    int mixed_list[] = {1, 2, 3, 4, 5, -6, -7, 8, 9, 0};
    size_t input_size = sizeof(mixed_list) / sizeof(mixed_list[0]);
    
    int *odd_numbers = NULL;
    size_t odd_size = 0;

    int status = extract_odd_numbers(mixed_list, input_size, &odd_numbers, &odd_size);

    if (status == 0) {
        for (size_t i = 0; i < odd_size; ++i) {
            printf("%d\n", odd_numbers[i]);
        }
    }

    if (odd_numbers != NULL) {
        free(odd_numbers);
        odd_numbers = NULL;
    }

    return status == 0 ? EXIT_SUCCESS : EXIT_FAILURE;
}
