```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <ctype.h>

#define MIN_LENGTH 8
#define MAX_LENGTH 64
#define BUFFER_SIZE 128

static void secure_clear(void *buf, size_t size)
{
    volatile unsigned char *p = (volatile unsigned char *)buf;

    if (buf == NULL) {
        return;
    }

    while (size--) {
        *p++ = 0;
    }
}

static bool drain_stdin(void)
{
    int ch;
    bool drained = false;

    do {
        ch = getchar();
        if (ch != '\n' && ch != EOF) {
            drained = true;
        }
    } while (ch != '\n' && ch != EOF);

    return drained;
}

bool is_valid_password(const char *password, size_t max_len)
{
    size_t length;
    size_t i;
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    bool has_special = false;

    if (password == NULL || max_len == 0) {
        return false;
    }

    length = strnlen(password, max_len);

    if (length >= max_len) {
        return false;
    }

    if (length < MIN_LENGTH || length > MAX_LENGTH) {
        return false;
    }

    for (i = 0; i < length; i++) {
        unsigned char c = (unsigned char)password[i];

        if (!isprint(c)) {
            return false;
        }

        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        } else if (ispunct(c)) {
            has_special = true;
        } else {
            return false;
        }
    }

    return has_upper && has_lower && has_digit && has_special;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t length;
    bool truncated = false;

    printf("Enter password: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    length = strnlen(buffer, sizeof(buffer));

    if (length > 0 && buffer[length - 1] == '\n') {
        buffer[length - 1] = '\0';
    } else {
        truncated = drain_stdin();
    }

    if (truncated) {
        printf("Password is invalid\n");
    } else if (is_valid_password(buffer, sizeof(buffer))) {
        printf("Password is valid\n");
    } else {
        printf("Password is invalid\n");
    }

    secure_clear(buffer, sizeof(buffer));

    return 0;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20) in `drain_stdin`:** This is a false positive in practice here. The `getchar()` loop never writes into a buffer — it only reads and discards characters until newline or EOF. There is no buffer to overflow. The loop terminates deterministically on `'\n'` or `EOF`, so no unbounded recursion or buffer risk exists. No change required; the pattern is the standard safe idiom for flushing residual input.

2. **Flawfinder `char buffer[BUFFER_SIZE]` warning (CWE-119/CWE-120) in `main`:** The statically sized array is already properly bounded:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` limits the read to the buffer size, guaranteeing no overflow.
   - Explicit null termination at `buffer[sizeof(buffer) - 1]` provides defense-in-depth even though `fgets` already terminates.
   - `strnlen` is used instead of `strlen`, bounding the scan to the buffer size.
   - Truncated input (no newline found) is detected via `drain_stdin()` and rejected rather than silently validated, which prevents accepting a partially-read password.

3. **Additional hardening already present and preserved:**
   - `is_valid_password` validates its pointer, checks `length >= max_len` to detect non-terminated input, and casts to `unsigned char` before calling `ctype` functions (avoiding undefined behavior with negative `char` values).
   - `secure_clear` uses a `volatile` pointer so the compiler cannot optimize away the memory wipe of the password (mitigating CWE-14 / sensitive data remanence). I added a defensive `NULL` check to it.

The code as written was already resistant to the flagged CWEs; the rewrite retains those mitigations and adds a minor null-pointer guard in `secure_clear`.
