#include <stdio.h>
#include <stdlib.h>

int sumOfRepeats(const int *arr, int n) {
    int *counts = (int *)calloc(n, sizeof(int));
    if (!counts) return -1;

    for (int i = 0; i < n; ++i)
        counts[arr[i]]++;

    int sum = 0;
    for (int i = 0; i < n; ++i)
        if (counts[i] > 1)
            sum += i * counts[i];

    free(counts);
    return sum;
}

int main() {
    int arr[] = {1, 2, 3, 2, 3, 3, 4};
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = sumOfRepeats(arr, n);

    if (result != -1)
        printf("Sum of repeated elements: %d\n", result);
    else
        printf("Memory allocation failed.\n");

    return 0;
}
