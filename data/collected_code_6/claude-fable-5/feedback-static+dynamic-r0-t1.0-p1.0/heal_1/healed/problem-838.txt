#include <stdio.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

int min_swaps_to_equal(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count10 = 0;
    int count01 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN);
    len2 = strnlen(s2, MAX_INPUT_LEN);

    if (len1 >= MAX_INPUT_LEN || len2 >= MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] == '1' && s2[i] == '0') {
            count10++;
        } else if (s1[i] == '0' && s2[i] == '1') {
            count01++;
        }
    }

    if ((count10 % 2) != (count01 % 2)) {
        return -1;
    }

    return (count10 / 2) + (count01 / 2) + ((count10 % 2) * 2);
}

int main(void)
{
    const char *a1 = "1101";
    const char *b1 = "0011";
    const char *a2 = "1110000";
    const char *b2 = "0001111";
    const char *a3 = "10";
    const char *b3 = "01";
    const char *a4 = "110";
    const char *b4 = "001";
    int result;

    result = min_swaps_to_equal(a1, b1);
    if (printf("Strings: %s, %s -> Result: %d\n", a1, b1, result) < 0) {

**Analysis summary:**

The original code's flagged weakness (CWE-126, buffer over-read via `strlen`) stems from `strlen` reading indefinitely if a string lacks a null terminator. The fix replaces `strlen` with `strnlen`, which bounds the scan to `MAX_INPUT_LEN` bytes. If a string reaches that bound without a terminator, the function safely rejects it by returning `-1` rather than over-reading.

Other defensive properties retained from the original:

- **NULL checks** on both input pointers before any dereference.
- **Length equality check** before iteration, preventing index mismatch.
- **Character validation** (only `'0'`/`'1'` accepted), rejecting malformed input early.
- **Parity check** on mismatch counts (`count10 % 2 != count01 % 2` means the strings have differing numbers of ones and can never be made equal by swaps).
- **Checked `printf` return values** in `main`, returning a nonzero exit code on output failure.

The algorithm itself is correct: paired `1→0`/`0→1` mismatches each cost one swap (`count10/2 + count01/2`), and one leftover mismatch of each kind costs two swaps (`(count10 % 2) * 2`).
