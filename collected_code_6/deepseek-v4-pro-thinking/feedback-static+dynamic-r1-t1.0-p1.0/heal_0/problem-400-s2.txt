#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

typedef struct {
    int a;
    int b;
    int count;
} TupleFreq;

static void normalize_pair(int *a, int *b) {
    if (*a > *b) {
        int tmp = *a;
        *a = *b;
        *b = tmp;
    }
}

int extract_tuple_frequencies(const int tuples[][2], size_t n, TupleFreq **out_freqs, size_t *out_count) {
    if (out_freqs == NULL || out_count == NULL || (tuples == NULL && n > 0)) {
        return -1;
    }

    *out_freqs = NULL;
    *out_count = 0;

    TupleFreq *freqs = NULL;
    size_t capacity = 0;
    size_t count = 0;

    for (size_t i = 0; i < n; ++i) {
        int a = tuples[i][0];
        int b = tuples[i][1];
        normalize_pair(&a, &b);

        int found = 0;
        for (size_t j = 0; j < count; ++j) {
            if (freqs[j].a == a && freqs[j].b == b) {
                ++freqs[j].count;
                found = 1;
                break;
            }
        }

        if (!found) {
            if (count == capacity) {
                size_t new_capacity = capacity == 0 ? 4 : capacity * 2;
                if (new_capacity <= capacity) {
                    free(freqs);
                    return -1;
                }

                TupleFreq *tmp = realloc(freqs, new_capacity * sizeof(*tmp));
                if (tmp == NULL) {
                    free(freqs);
                    return -1;
                }

                freqs = tmp;
                capacity = new_capacity;
            }

            freqs[count].a = a;
            freqs[count].b = b;
            freqs[count].count = 1;
            ++count;
        }
    }

    *out_freqs = freqs;
    *out_count = count;
    return 0;
}

int main(void) {
    int tuples[][2] = {
        {1, 2},
        {2, 1},
        {3, 4},
        {4, 3},
        {1, 2},
        {5, 6}
    };

    size_t n = sizeof(tuples) / sizeof(tuples[0]);
    TupleFreq *freqs = NULL;
    size_t freq_count = 0;

    if (extract_tuple_frequencies(tuples, n, &freqs, &freq_count) != 0) {
        fprintf(stderr, "Failed to extract tuple frequencies\n");
        return 1;
    }

    for (size_t i = 0; i < freq_count; ++i) {
        printf("(%d, %d): %d\n", freqs[i].a, freqs[i].b, freqs[i].count);
    }

    free(freqs);
    return 0;
}
