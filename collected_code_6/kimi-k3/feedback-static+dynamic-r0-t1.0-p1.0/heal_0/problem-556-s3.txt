#include <stdio.h>
#include <stdlib.h>
#include <errno.h>

long long count_xor_odd_pairs(const int *arr, size_t n)
{
    if (arr == NULL || n == 0) {
        return 0;
    }

    size_t even_count = 0;
    size_t odd_count = 0;

    for (size_t i = 0; i < n; i++) {
        if (arr[i] % 2 == 0) {
            even_count++;
        } else {
            odd_count++;
        }
    }

    return (long long)even_count * (long long)odd_count;
}

int main(void)
{
    int n = 0;
    int *arr = NULL;
    long long result = 0;

    printf("Enter the number of elements: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    arr = (int *)malloc((size_t)n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Enter %d elements: ", n);
    for (int i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Invalid input\n");
            free(arr);
            arr = NULL;
            return EXIT_FAILURE;
        }
    }

    result = count_xor_odd_pairs(arr, (size_t)n);
    printf("Number of pairs with odd XOR: %lld\n", result);

    free(arr);
    arr = NULL;

    return EXIT_SUCCESS;
}
