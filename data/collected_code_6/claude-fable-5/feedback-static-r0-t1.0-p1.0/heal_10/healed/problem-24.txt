```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * CHAR_BIT)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = fgetc(stdin);
        guard++;
        if (guard >= (size_t)INT_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);

    return ch;
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        (void)discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and why this code is already safe:**

1. **Flawfinder `fgetc` warning (CWE-120, CWE-20)** in `discard_remaining_input`: The original code already mitigates this correctly. `fgetc` returns an `int` and never writes into a buffer here, so no overflow is possible. The `guard` counter capped at `INT_MAX` prevents an unbounded loop on a pathological stream, and both `'\n'` and `EOF` terminate the loop. No change was required.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** for `input[INPUT_SIZE]`: The buffer is handled defensively throughout:
   - It is zero-initialized at declaration.
   - `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size, guaranteeing no overflow.
   - Explicit null termination at `input[sizeof(input) - 1U]` guards against any edge case.
   - `strnlen` (bounded) is used instead of `strlen`.
   - Overlong input is detected (missing newline with a full buffer), the remainder of the stream is drained, and the program rejects the input rather than truncating silently.

3. **`binary_to_decimal` robustness**: The function validates all pointers, rejects empty or unterminated strings, restricts input to `'0'`/`'1'`, limits bit-length to `sizeof(unsigned long) * CHAR_BIT`, and checks `value > (ULONG_MAX >> 1)` before each shift to prevent unsigned overflow/wraparound.

The Flawfinder hits are level 1–2 informational findings, and both were false positives given the existing mitigations, so the rewritten code preserves the original logic with the comments removed.
