```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

char **remove_duplicates(const char *const *words, size_t count, size_t *out_count);
void free_word_list(char **words, size_t count);

char **remove_duplicates(const char *const *words, size_t count, size_t *out_count)
{
    char **result;
    size_t i;
    size_t j;
    size_t unique;

    if (words == NULL || out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(count * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    unique = 0;
    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;
        size_t alloc_size;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (found) {
            continue;
        }

        if (len > SIZE_MAX - 1) {
            continue;
        }
        alloc_size = len + 1;

        if (alloc_size <= len) {
            continue;
        }

        result[unique] = malloc(alloc_size);
        if (result[unique] == NULL) {
            for (j = 0; j < unique; j++) {
                free(result[j]);
            }
            free(result);
            return NULL;
        }

        memcpy(result[unique], words[i], len);
        result[unique][len] = '\0';
        unique++;
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "cherry"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Original list:\n");
    for (i = 0; i < input_count; i++) {
        printf("  %s\n", input[i]);
    }

    printf("Deduplicated list:\n");
    for (i = 0; i < result_count; i++) {
        printf("  %s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) was a false positive in the original code, but the rewrite makes the safety guarantees fully explicit:

1. **Destination size is provably sufficient**: `result[unique]` is allocated with `alloc_size = len + 1` bytes, and `memcpy` copies exactly `len` bytes, followed by writing the null terminator at index `len` — the last valid byte of the allocation. No overflow is possible.

2. **Source length is bounded**: `strnlen(words[i], MAX_WORD_LEN)` caps the measured length, and entries at or above `MAX_WORD_LEN` are rejected, so `len` can never exceed `MAX_WORD_LEN - 1`.

3. **Integer overflow guards**: The `len > SIZE_MAX - 1` check and the added `alloc_size <= len` defensive check guarantee `len + 1` cannot wrap around to a smaller allocation than the copy size.

4. **Allocation failure handling**: If any `malloc` fails, all previously allocated strings and the pointer array are freed before returning `NULL`, preventing memory leaks (CWE-401).

5. **Additional robustness already present and preserved**: `NULL` parameter checks, `count == 0` handling, overflow check on `count * sizeof(char *)`, bounded comparison via `strncmp`, and a `free_word_list` that tolerates `NULL`.

The code is compliant with C99, avoids all banned functions (`strcpy`, `sprintf`, etc.), and the memcpy invariant (destination capacity ≥ copy length + 1) is now enforced both by construction and by explicit runtime checks.
