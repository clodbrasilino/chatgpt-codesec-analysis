```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and how the code addresses them:**

1. **`char normalized[MAX_INPUT_LEN]` in `has_28_days` (CWE-119/CWE-120):** The array is protected against overflow through multiple layers: `strnlen` bounds the length check to `MAX_INPUT_LEN`, input is rejected if `len >= MAX_INPUT_LEN`, the copy loop is doubly bounded by `i < len && i < sizeof(normalized) - 1`, and null termination is guaranteed after the loop. No overflow is possible.

2. **`getchar` in `flush_stdin` (CWE-120/CWE-20):** The return value is stored in an `int` (not `char`), which is essential for correctly detecting `EOF`. The loop terminates on both newline and `EOF`, so it cannot spin infinitely on a closed stream, and no buffer is written to at all.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Input is read with `fgets` bounded by `sizeof(input)`, the buffer is pre-zeroed with `memset`, explicit null termination is enforced as a defensive measure, and oversized lines are drained from `stdin` via `flush_stdin` to prevent stale data affecting subsequent reads. Additionally, `%s` in `printf` uses a fixed format string with the sanitized buffer as an argument, avoiding format-string vulnerabilities (CWE-134).

The code is already well-hardened; the Flawfinder hits are informational (static buffer declarations trigger them by pattern-matching), and each has appropriate mitigations in place. No functional changes were required.
