#include <stdio.h>
#include <string.h>
#include <regex.h>

void filter_chars(const char *input, char *output, size_t output_size) {
    regex_t regex;
    if (regcomp(&regex, "[^a-zA-Z0-9]", 0) == 0) {
        size_t j = 0;
        for (size_t i = 0; input[i] != '\0' && j < output_size - 1; ++i) {
            if (regexec(&regex, &input[i], 1, NULL, 0) != 0 && j < output_size - 1) {
                output[j++] = input[i];
            }
        }
        output[j] = '\0';
        regfree(&regex);
    } else {
        output[0] = '\0';
    }
}

int main() {
    const char input[] = "Hello, World! 123";
    char output[50] = "";

    filter_chars(input, output, sizeof(output));

    printf("Filtered: %s\n", output);
    return 0;
}
