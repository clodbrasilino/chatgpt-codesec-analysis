```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t out_pos;
    size_t capacity;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    capacity = len + 1U;
    result = malloc(capacity);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        regoff_t start = match[0].rm_so;
        regoff_t char_pos = match[1].rm_so;

        if (start > 0) {
            if (out_pos + (size_t)start >= capacity) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(&result[out_pos], cursor, (size_t)start);
            out_pos += (size_t)start;
        }

        if (out_pos + 1U >= capacity) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        result[out_pos] = (char)toupper((unsigned char)cursor[char_pos]);
        out_pos++;

        cursor += match[0].rm_eo;
    }

    {
        size_t remaining = strnlen(cursor, MAX_INPUT_LEN);

        if (out_pos + remaining >= capacity) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        memcpy(&result[out_pos], cursor, remaining);
        out_pos += remaining;
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "snake_case_string",
        "already",
        "multi_word_example_test"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    int status = EXIT_SUCCESS;

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);

        if (converted == NULL) {
            if (fprintf(stderr, "Conversion failed for: %s\n", tests[i]) < 0) {
                status = EXIT_FAILURE;
                break;
            }
            status = EXIT_FAILURE;
            continue;
        }

        if (printf("%s -> %s\n", tests[i], converted) < 0) {
            free(converted);
            status = EXIT_FAILURE;
            break;
        }

        free(converted);
    }

    return status;
}
```

Key changes made to address the flagged weaknesses:

1. **CWE-126 (`strlen` over-read):** Both `strlen` calls were replaced with `strnlen` bounded by `MAX_INPUT_LEN`. If the input reaches the limit (indicating a possibly unterminated or oversized string), the function rejects it and returns `NULL` rather than risking an over-read.

2. **CWE-120 (`memcpy` unchecked buffer overflow):** Each `memcpy` (and the single-character write) is now preceded by an explicit bounds check against the tracked destination `capacity`. If a copy would exceed the allocated buffer, the function cleans up (`free` + `regfree`) and fails safely instead of overflowing.

3. **Resource safety on new failure paths:** All added early-exit paths release both the allocated buffer and the compiled regex, so no leaks are introduced by the hardening.

The logical behavior for valid, well-formed inputs is unchanged — since camelCase conversion only removes underscores, the output can never exceed the input length, so the checks never fire on legitimate data; they exist purely as defense-in-depth.
