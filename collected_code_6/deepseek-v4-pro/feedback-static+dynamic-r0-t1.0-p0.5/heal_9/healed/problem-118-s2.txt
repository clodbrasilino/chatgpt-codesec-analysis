#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *data;
    struct Node *next;
} Node;

Node *create_node(const char *str) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = (char *)malloc(strlen(str) + 1);
    if (new_node->data == NULL) {
        free(new_node);
        return NULL;
    }
    strcpy(new_node->data, str);
    new_node->next = NULL;
    return new_node;
}

static char *custom_strtok(char *str, const char *delimiter) {
    static char *next_token = NULL;
    char *token_start;
    
    if (str != NULL) {
        next_token = str;
    }
    
    if (next_token == NULL || *next_token == '\0') {
        next_token = NULL;
        return NULL;
    }
    
    while (*next_token != '\0' && strchr(delimiter, *next_token) != NULL) {
        next_token++;
    }
    
    if (*next_token == '\0') {
        next_token = NULL;
        return NULL;
    }
    
    token_start = next_token;
    
    while (*next_token != '\0' && strchr(delimiter, *next_token) == NULL) {
        next_token++;
    }
    
    if (*next_token != '\0') {
        *next_token = '\0';
        next_token++;
    } else {
        next_token = NULL;
    }
    
    return token_start;
}

Node *string_to_list(const char *str, const char *delimiter) {
    if (str == NULL || delimiter == NULL || strlen(delimiter) == 0) {
        return NULL;
    }
    
    Node *head = NULL;
    Node *tail = NULL;
    char *str_copy = (char *)malloc(strlen(str) + 1);
    if (str_copy == NULL) {
        return NULL;
    }
    strcpy(str_copy, str);
    
    char *token = custom_strtok(str_copy, delimiter);
    while (token != NULL) {
        Node *new_node = create_node(token);
        if (new_node == NULL) {
            Node *current = head;
            while (current != NULL) {
                Node *next = current->next;
                free(current->data);
                free(current);
                current = next;
            }
            free(str_copy);
            return NULL;
        }
        
        if (head == NULL) {
            head = new_node;
            tail = new_node;
        } else {
            tail->next = new_node;
            tail = new_node;
        }
        
        token = custom_strtok(NULL, delimiter);
    }
    
    free(str_copy);
    return head;
}

void free_list(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *next = current->next;
        free(current->data);
        free(current);
        current = next;
    }
}

void print_list(Node *head) {
    Node *current = head;
    int first = 1;
    printf("[");
    while (current != NULL) {
        if (!first) {
            printf(", ");
        }
        printf("'%s'", current->data);
        first = 0;
        current = current->next;
    }
    printf("]\n");
}

int main(void) {
    const char *test_string1 = "python programming";
    const char *delimiter1 = " ";
    
    Node *list1 = string_to_list(test_string1, delimiter1);
    if (list1 != NULL) {
        print_list(list1);
        free_list(list1);
    }
    
    const char *test_string2 = "lists,tuples,strings";
    const char *delimiter2 = ",";
    
    Node *list2 = string_to_list(test_string2, delimiter2);
    if (list2 != NULL) {
        print_list(list2);
        free_list(list2);
    }
    
    const char *test_string3 = "write a program";
    const char *delimiter3 = " ";
    
    Node *list3 = string_to_list(test_string3, delimiter3);
    if (list3 != NULL) {
        print_list(list3);
        free_list(list3);
    }
    
    return EXIT_SUCCESS;
}
