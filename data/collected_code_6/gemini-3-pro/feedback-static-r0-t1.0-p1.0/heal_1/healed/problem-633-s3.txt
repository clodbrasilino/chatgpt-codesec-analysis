#include <stdio.h>
#include <stdlib.h>

long long sumXOR(const int arr[], int n) {
    long long sum = 0;
    for (int i = 0; i < 32; i++) {
        int zero_count = 0;
        int one_count = 0;
        for (int j = 0; j < n; j++) {
            if (arr[j] & (1U << i)) {
                one_count++;
            } else {
                zero_count++;
            }
        }
        sum += (long long)zero_count * one_count * (1LL << i);
    }
    return sum;
}

int main(void) {
    const int arr[] = {7, 3, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    long long result = sumXOR(arr, n);
    printf("%lld\n", result);
    
    return 0;
}
