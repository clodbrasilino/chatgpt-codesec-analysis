```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64
#define FLUSH_LIMIT 4096

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (long)(c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (long)(c - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t count = 0;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The original code already mitigated unbounded reading by capping iterations with `FLUSH_LIMIT` and terminating on `\n` or `EOF`. This is preserved — no data is stored, so no buffer can overflow; the counter simply prevents pathological infinite consumption.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer is safe because:
   - `fgets(input, sizeof(input), stdin)` limits the read to the buffer size and always NUL-terminates within bounds.
   - An explicit `input[sizeof(input) - 1] = '\0'` guarantees termination defensively.
   - `strnlen` is used instead of `strlen`, bounding all length computations.
   - Overlong input (no newline found at the buffer boundary) is detected, the remainder of the line is drained via `flush_stdin()`, and the program rejects the input rather than silently truncating.
   - Zero-initialization of the array (`= {0}`) adds defense-in-depth against reading uninitialized memory.

3. **Integer overflow in conversion:** `octal_to_decimal` performs a pre-multiplication overflow check against `LONG_MAX`, rejecting values that would overflow `long`, and validates every character is within `'0'`–`'7'`.

The static-analysis findings were essentially false positives already handled by the original defensive coding; the rewrite retains those safeguards with minor hardening.
