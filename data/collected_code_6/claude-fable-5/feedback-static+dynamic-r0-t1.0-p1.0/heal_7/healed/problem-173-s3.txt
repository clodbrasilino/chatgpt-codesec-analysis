```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

void remove_non_alnum(char *str)
{
    size_t read_idx;
    size_t write_idx;

    if (str == NULL) {
        return;
    }

    read_idx = 0;
    write_idx = 0;

    while (str[read_idx] != '\0') {
        if (isalnum((unsigned char)str[read_idx])) {
            str[write_idx] = str[read_idx];
            write_idx++;
        }
        read_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[256];
    const char *input = "Hello, World! 123 #$% foo_bar";
    size_t input_len;
    int written;

    input_len = strnlen(input, sizeof(buffer));
    if (input_len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, sizeof(buffer), "%s", input);
    if (written < 0 || (size_t)written >= sizeof(buffer)) {
        fprintf(stderr, "Failed to copy input safely\n");
        return EXIT_FAILURE;
    }

    printf("Before: %s\n", buffer);

    remove_non_alnum(buffer);

    printf("After:  %s\n", buffer);

    return EXIT_SUCCESS;
}
```

The code already addresses the flagged weakness (CWE-119/CWE-120) properly:

- **Bounds-checked length validation**: `strnlen(input, sizeof(buffer))` avoids reading beyond a bounded limit and the result is verified against the buffer size before copying.
- **Safe copy**: `snprintf` with `sizeof(buffer)` guarantees no overflow and null termination; its return value is checked for both encoding errors (`< 0`) and truncation (`>= sizeof(buffer)`).
- **Safe `ctype` usage**: `isalnum` receives its argument cast to `unsigned char`, avoiding undefined behavior for negative `char` values.
- **In-place filtering**: `remove_non_alnum` uses a two-index compaction that can only shrink the string (`write_idx <= read_idx`), so it never writes past the original null terminator, and it guards against `NULL` input.

No further changes are required; the statically-sized buffer is safe because every write into it is length-limited and verified.
