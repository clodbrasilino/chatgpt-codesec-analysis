```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static void print_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1U);
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (*offset + (size_t)end) <= input_len) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN &&
                (*offset + (size_t)start + len) <= input_len) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    *offset += (size_t)matches[0].rm_eo;
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "Regex execution failed");
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`char errbuf[ERRBUF_SIZE]` (CWE-119/CWE-120):** This is a false positive in practice. The buffer is zero-initialized with `memset`, `regerror` is called with `sizeof(errbuf) - 1U` guaranteeing space for a terminator, and the final byte is explicitly null-terminated. There is no path where the buffer can overflow.

2. **`memcpy` (CWE-120):** Also mitigated. The destination is allocated with `malloc(len + 1U)`, so it always holds exactly `len` bytes plus the terminator. The source range is validated before the copy: `start >= 0`, `end >= start`, `len < MAX_INPUT_LEN`, and `(*offset + start + len) <= input_len` guarantee the read stays within the input string.

**Additional defensive properties already present:**

- `strnlen` with `MAX_INPUT_LEN` bounds the input scan and rejects oversized/unterminated inputs.
- All offset arithmetic is done in `size_t` with explicit casts from `regoff_t`, with range checks preceding any pointer arithmetic.
- `regfree` is called on every path after successful `regcomp`, avoiding resource leaks.
- `malloc` failure is handled explicitly, and the caller frees every returned buffer.

The code as provided is already hardened; the rewrite preserves those mitigations with the comments removed as requested.
