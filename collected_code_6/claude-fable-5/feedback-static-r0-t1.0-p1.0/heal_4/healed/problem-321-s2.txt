int find_demlo(const char *input, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, (size_t)MAX_INPUT_LEN + 1U);

    if (len == 0U || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    for (i = len; i > 1U; i--)
    {
        output[pos] = (char)('0' + (char)(i - 1U));
        pos++;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';
    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;
        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;
}

1. **`char input[BUFFER_SIZE]` (CWE-119/CWE-120):** Mitigated. Input is read with `fgets(input, (int)sizeof(input), stdin)`, which bounds the read to the buffer size. A defensive `input[sizeof(input) - 1U] = '\0'` guarantees null-termination, and `strnlen` (rather than `strlen`) is used to measure length safely.

2. **`char result[BUFFER_SIZE]` (CWE-119/CWE-120):** Mitigated. `find_demlo` receives the buffer size explicitly (`sizeof(result)`) and validates `output_size < (2U * len)` before writing. Since the maximum output is `2 * MAX_INPUT_LEN - 1` characters plus the terminator (18 bytes total), it fits well within the 64-byte buffer, and the writes are index-bounded by the validated `len`.

3. **`getchar()` in a loop (CWE-120/CWE-20):** Mitigated. The loop never writes into any buffer; it only discards excess characters until newline or `EOF`, so no overflow is possible. The `EOF` check prevents an infinite loop on closed input streams.
