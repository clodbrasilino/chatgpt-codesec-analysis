```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

static int decimal_to_binary(unsigned int number, char *buffer, size_t buffer_size)
{
    size_t index = 0U;
    size_t start;
    size_t end;

    if (buffer == NULL || buffer_size < 2U) {
        return -1;
    }

    if (number == 0U) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (number > 0U && index < (buffer_size - 1U)) {
        buffer[index] = (char)('0' + (number % 2U));
        number /= 2U;
        index++;
    }

    if (number > 0U) {
        return -1;
    }

    buffer[index] = '\0';

    start = 0U;
    end = index - 1U;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    unsigned int value = 0U;
    char binary[MAX_BITS + 1U];
    int scan_result;

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &value);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(value, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary equivalent: %s\n", binary);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning on the statically-sized `binary` array is a low-risk, largely informational finding, and the original code already mitigates it correctly:

1. **Buffer sizing is provably sufficient.** `MAX_BITS + 1U` guarantees room for every bit of an `unsigned int` plus the NUL terminator, so no valid input can exceed the buffer.

2. **Explicit bounds checking exists.** `decimal_to_binary` receives the buffer size via `sizeof(binary)`, validates it (`buffer_size < 2U`), and the conversion loop is bounded by `index < (buffer_size - 1U)`. If truncation would occur, the function fails safely with `-1` rather than overflowing.

3. **No unbounded string functions are used.** There are no `strcpy`/`sprintf`/`gets` calls; the buffer is written only through index-checked assignments.

4. **Input parsing is safe.** `scanf("%u", ...)` writes a fixed-size integer, not a string, so it cannot overflow, and its return value is checked.

The code as written is already the fix — the comment documents a scanner heuristic, not an actual exploitable defect.
