#include <stdio.h>
#include <math.h>

int find_solutions(double a, double b, double c) {
    if (fabs(a) < 1e-9) {
        if (fabs(b) < 1e-9) {
            return -1;
        }
        return 1;
    }
    double discriminant = b * b - 4 * a * c;
    if (discriminant > 0) {
        return 2;
    } else if (fabs(discriminant) < 1e-9) {
        return 1;
    } else {
        return 0;
    }
}

int main(void) {
    double a, b, c;
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return 1;
    }
    int result = find_solutions(a, b, c);
    if (result == -1) {
        printf("Infinite or no solution (not quadratic)\n");
    } else {
        printf("%d\n", result);
    }
    return 0;
}
