bool first_last_equal(const char *str, size_t max_len)

    size_t len;

    if (str == NULL || max_len == 0) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    return str[0] == str[len - 1];
}

        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len == sizeof(buffer)) {
        fprintf(stderr, "Input too long.\n");
        return 1;
    }

        buffer[len - 1] = '\0';
        len--;
    }

    if (len == 0 || buffer[0] == '\0') {
        fprintf(stderr, "Empty string provided.\n");
        return 1;
    }

    if (first_last_equal(buffer, sizeof(buffer))) {
        printf("The first and last characters are equal.\n");

2. **Guaranteed null termination** — explicitly setting `buffer[sizeof(buffer) - 1] = '\0'` ensures the string is terminated even in edge cases.
3. **Bounded length calculation** — `strnlen()` is used instead of `strlen()`, so length computation can never run past the buffer even if termination were somehow missing.
4. **Truncation detection** — checking `len == sizeof(buffer)` rejects oversized input rather than silently processing truncated data.
