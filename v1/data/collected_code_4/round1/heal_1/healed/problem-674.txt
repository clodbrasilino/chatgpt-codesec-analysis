#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_CHAR 100

typedef struct Node {
    char *data;
    struct Node *hashNext;
    struct Node *next;
} Node;

typedef struct Hash {
    int count;
    struct Node **items;
} Hash;

Hash *createHashTable(int size) {
    Hash *hashTable = calloc(1, sizeof(Hash));
    if(!hashTable) return NULL;
    hashTable->count = size;
    hashTable->items = calloc(size, sizeof(Node *));
    if(!hashTable->items) {
        free(hashTable);
        return NULL;
    }
    return hashTable;
}

unsigned long hashFunction(char *str) {
    unsigned long i = 0;
    for(int j = 0; str[j]; j++)
        i += str[j];
    return i;
}

Node *createNode(char *data) {
    Node *newNode = calloc(1, sizeof(Node));
    if(!newNode) return NULL;
    newNode->data = strdup(data);
    newNode->hashNext = NULL;
    newNode->next = NULL;
    return newNode;
}

void hashAdd(Hash *hashTable, char *data) {
    unsigned long hashIndex = hashFunction(data);
    Node *newNode = createNode(data);
    if(!newNode) return;
    newNode->hashNext = hashTable->items[hashIndex];
    hashTable->items[hashIndex] = newNode;
}

int hashSearch(Hash *hashTable, char *data) {
    unsigned long hashIndex = hashFunction(data);
    Node *node = hashTable->items[hashIndex];  

    while(node != NULL) {
        if(strcmp(node->data, data) == 0) return 1;
        node = node->hashNext;
    }
    return 0;
}

void freeHash(Hash *hashTable) {
    for(int i = 0; i < hashTable->count; i++){
        Node *node = hashTable->items[i];
        while(node) {
            Node *temp = node;
            node = node->hashNext;
            free(temp->data);
            free(temp);
        }
    }
    free(hashTable->items);
    free(hashTable);
}

void removeDuplicates(char *str) {
    Hash *hashTable = createHashTable(MAX_CHAR);
    Node *start = NULL, *end = NULL;

    char *word = strtok(str, " ");
    while(word) {
        if(!hashSearch(hashTable, word)) {
            Node *newNode = createNode(word);
            if(!newNode) break;
            hashAdd(hashTable, word);
            if(start == NULL) {
                start = newNode;
                end = start;
            } else {
                end->next = newNode;
                end = newNode;
            }
        }
        word = strtok(NULL, " ");
    }

    Node *temp = start;
    while(temp) {
        printf("%s ", temp->data);
        temp = temp->next;
    }

    printf("\n");
    freeHash(hashTable);
}

int main() {
    char str[MAX_CHAR] = "Hello world. This is a C program. Hello world. This is a C program.";
    removeDuplicates(str);

    return 0;
}
