#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>

#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999
#define MAX_ROMAN_LENGTH 15
#define ROMAN_BUFFER_SIZE (MAX_ROMAN_LENGTH + 2)
#define INPUT_BUFFER_SIZE 64
#define MAX_DISCARD_CHARS 4096

_Static_assert(ROMAN_BUFFER_SIZE > MAX_ROMAN_LENGTH,
               "roman buffer must hold the longest numeral plus its terminator");

int int_to_roman(int value, char *buffer, size_t buffer_size)
{
    static const struct {
        int value;
        const char *numeral;
        size_t length;
    } roman_table[] = {
        {1000, "M", 1},
        {900, "CM", 2},
        {500, "D", 1},
        {400, "CD", 2},
        {100, "C", 1},
        {90, "XC", 2},
        {50, "L", 1},
        {40, "XL", 2},
        {10, "X", 1},
        {9, "IX", 2},
        {5, "V", 1},
        {4, "IV", 2},
        {1, "I", 1}
    };

    size_t position;
    size_t table_size;
    size_t i;
    size_t j;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    if (buffer_size <= MAX_ROMAN_LENGTH) {
        buffer[0] = '\0';
        return -1;
    }

    if (value < MIN_ROMAN_VALUE || value > MAX_ROMAN_VALUE) {
        buffer[0] = '\0';
        return -1;
    }

    position = 0;
    table_size = sizeof(roman_table) / sizeof(roman_table[0]);

    for (i = 0; i < table_size; i++) {
        while (value >= roman_table[i].value) {
            if (position >= buffer_size - 1 ||
                roman_table[i].length > (buffer_size - 1) - position) {
                buffer[0] = '\0';
                return -1;
            }
            for (j = 0; j < roman_table[i].length; j++) {
                buffer[position] = roman_table[i].numeral[j];
                position++;
            }
            value -= roman_table[i].value;
        }
    }

    buffer[position] = '\0';
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t discarded = 0;

    while (discarded < MAX_DISCARD_CHARS) {
        ch = getchar();
        if (ch == EOF) {
            if (feof(stdin)) {
                return 0;
            }
            return -1;
        }
        discarded++;
        if (ch == '\n') {
            return 0;
        }
    }

    return -1;
}

int main(void)
{
    char *roman_buffer = NULL;
    char *input_buffer = NULL;
    char *endptr;
    long number;
    int conversion_result;
    int exit_status = EXIT_FAILURE;

    roman_buffer = calloc(ROMAN_BUFFER_SIZE, sizeof(char));
    if (roman_buffer == NULL) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    input_buffer = calloc(INPUT_BUFFER_SIZE, sizeof(char));
    if (input_buffer == NULL) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        free(roman_buffer);
        return EXIT_FAILURE;
    }

    printf("Enter an integer (1-3999): ");
    fflush(stdout);

    if (fgets(input_buffer, INPUT_BUFFER_SIZE, stdin) == NULL) {
        fprintf(stderr, "Error: Invalid input\n");
        goto cleanup;
    }

    if (strchr(input_buffer, '\n') == NULL && !feof(stdin)) {
        if (discard_remaining_input() != 0) {
            fprintf(stderr, "Error: Input exceeds maximum allowed length\n");
            goto cleanup;
        }
        fprintf(stderr, "Error: Invalid input\n");
        goto cleanup;
    }

    errno = 0;
    number = strtol(input_buffer, &endptr, 10);

    if (errno != 0 || endptr == input_buffer) {
        fprintf(stderr, "Error: Invalid input\n");
        goto cleanup;
    }

    while (*endptr == ' ' || *endptr == '\t' || *endptr == '\n' ||
           *endptr == '\r' || *endptr == '\f' || *endptr == '\v') {
        endptr++;
    }

    if (*endptr != '\0') {
        fprintf(stderr, "Error: Invalid input\n");
        goto cleanup;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        fprintf(stderr, "Error: Number must be between 1 and 3999\n");
        goto cleanup;
    }

    conversion_result = int_to_roman((int)number, roman_buffer,
                                     ROMAN_BUFFER_SIZE);

    if (conversion_result != 0) {
        fprintf(stderr, "Error: Number must be between 1 and 3999\n");
        goto cleanup;
    }

    printf("Roman numeral: %s\n", roman_buffer);
    exit_status = EXIT_SUCCESS;

cleanup:
    free(input_buffer);
    free(roman_buffer);
    return exit_status;
}
