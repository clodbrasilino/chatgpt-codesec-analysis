```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, sizeof(ta->first));
}

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(t->first));
    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static void sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
}

static int print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return -1;
    }

    for (i = 0U; i < count; i++) {
        if (printf("(%.*s, %d)\n", (int)(sizeof(tuples[i].first) - 1U),
                   tuples[i].first, tuples[i].second) < 0) {
            return -1;
        }
    }
    return 0;
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Initialization error\n");
        return EXIT_FAILURE;
    }

    sort_tuples(tuples, count);

    if (print_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Output error\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-hardened, and the flagged weaknesses were essentially false positives given the existing mitigations. The rewrite preserves and slightly strengthens those defenses:

1. **CWE-119/CWE-120 (statically-sized array `first`):** The buffer itself isn't a flaw — the risk is unchecked writes into it. `set_tuple` guards this by measuring the input with `strnlen` bounded to `sizeof(t->first)` and rejecting any name that wouldn't fit including the NUL terminator. No unbounded `strcpy`/`sprintf` ever touches the buffer.

2. **CWE-120 (`memcpy`):** The `memcpy` is safe because `len < sizeof(t->first)` is verified immediately before the copy, and explicit NUL termination follows. I removed the redundant double check (`len >= MAX_NAME_LEN` and `len >= sizeof(t->first)`) and kept a single check tied directly to `sizeof(t->first)`, which stays correct even if the struct definition changes.

3. **Defensive consistency:** `compare_tuples` and `print_tuples` now derive their bounds from `sizeof` the actual field rather than the macro, eliminating any possibility of drift between the macro and the struct. The `%.*s` precision cap in `printf` guarantees bounded output even if a string were somehow left unterminated.

4. **Robustness retained:** NULL-pointer checks in all helpers, `qsort` guarded against zero/NULL input, `printf` return value checked, and proper `EXIT_SUCCESS`/`EXIT_FAILURE` usage in `main`.
