```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 4096

static int reverse_string(char *str)
{
    size_t left;
    size_t right;

    if (str == NULL) {
        return -1;
    }

    left = 0;
    right = strnlen(str, MAX_STRING_LEN);

    if (right == 0) {
        return 0;
    }

    right--;

    while (left < right) {
        char temp = str[left];
        str[left] = str[right];
        str[right] = temp;
        left++;
        right--;
    }

    return 0;
}

static int reverse_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        if (reverse_string(list[i]) != 0) {
            return -1;
        }
    }

    return 0;
}

static void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
        list[i] = NULL;
    }

    free(list);
}

int main(void)
{
    const char *source[] = { "hello", "world", "example", "" };
    size_t count = sizeof(source) / sizeof(source[0]);
    char **list;
    size_t i;

    list = calloc(count, sizeof(char *));
    if (list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        size_t len = strnlen(source[i], MAX_STRING_LEN);

        list[i] = malloc(len + 1);
        if (list[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_string_list(list, count);
            return EXIT_FAILURE;
        }

        memcpy(list[i], source[i], len);
        list[i][len] = '\0';
    }

    if (reverse_string_list(list, count) != 0) {
        fprintf(stderr, "Failed to reverse string list\n");
        free_string_list(list, count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        printf("%s\n", list[i]);
    }

    free_string_list(list, count);

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **Variable scope (`temp`)** — Moved the declaration of `temp` inside the `while` loop body in `reverse_string`, reducing its scope to where it is actually used, as flagged by the static analyzer.

2. **`strlen` over-read risk (CWE-126)** — Replaced both `strlen` calls with `strnlen` bounded by `MAX_STRING_LEN`, guaranteeing the scan stops even if a string were not NUL-terminated. This eliminates the potential over-read.

3. **`memcpy` overflow risk (CWE-120)** — The copy is now bounded by `len` (the same value used to size the allocation `malloc(len + 1)`), and the NUL terminator is written explicitly with `list[i][len] = '\0'`. This makes the invariant "destination always holds the source data plus terminator" explicit and safe even if `strnlen` truncated the length.

The overall structure, error handling (including cleanup via `free_string_list` on all failure paths), and behavior remain unchanged.
