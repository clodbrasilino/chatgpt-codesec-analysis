```c
#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool is_even(long long number)
{
    return (number % 2LL) == 0LL;
}

int main(void)
{
    long long number = 0LL;
    int result = 0;

    printf("Enter an integer: ");
    result = scanf("%lld", &number);

    if (result != 1)
    {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    if (is_even(number))
    {
        printf("%lld is even.\n", number);
    }
    else
    {
        printf("%lld is odd.\n", number);
    }

    return EXIT_SUCCESS;
}
```
