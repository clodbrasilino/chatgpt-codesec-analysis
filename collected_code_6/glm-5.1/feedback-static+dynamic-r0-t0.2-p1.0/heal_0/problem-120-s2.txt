```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int a;
    int b;
} Tuple;

int find_max_product(Tuple *tuples, size_t size) {
    if (tuples == NULL || size == 0) {
        return 0;
    }

    int max_product = tuples[0].a * tuples[0].b;

    for (size_t i = 1; i < size; i++) {
        int current_product = tuples[i].a * tuples[i].b;
        if (current_product > max_product) {
            max_product = current_product;
        }
    }

    return max_product;
}

int main(void) {
    Tuple tuples[] = {{1, 2}, {3, 4}, {-5, -6}, {7, 1}};
    size_t size = sizeof(tuples) / sizeof(tuples[0]);

    int result = find_max_product(tuples, size);

    printf("%d\n", result);

    return 0;
}
```
