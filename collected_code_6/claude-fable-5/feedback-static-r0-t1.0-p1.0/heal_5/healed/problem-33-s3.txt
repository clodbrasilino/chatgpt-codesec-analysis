```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (decimal > 0 && index < buffer_size - 1) {
        buffer[index] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
        index++;
    }

    if (decimal > 0) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[index] = '\0';

    for (i = 0; i < index / 2; i++) {
        char temp = buffer[i];
        buffer[i] = buffer[index - 1 - i];
        buffer[index - 1 - i] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[MAX_BITS + 1];
    int scan_result;

    memset(binary, 0, sizeof(binary));

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &number);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal: %u\n", number);
    printf("Binary: %s\n", binary);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-defended against the Flawfinder warning (CWE-119/CWE-120), which is a heuristic false positive in this case:

1. **Buffer sizing is provably correct**: `MAX_BITS + 1` guarantees space for every bit of an `unsigned int` plus the null terminator, so overflow is impossible for any valid input.

2. **Bounds checking is enforced in `decimal_to_binary`**: the loop condition `index < buffer_size - 1` prevents writing past the buffer, and the `sizeof(binary)` is passed rather than a hard-coded size, keeping the size and the array coupled.

3. **Defensive checks are present**: `NULL` pointer and minimum-size validation, plus an explicit truncation check (`if (decimal > 0)` after the loop) with a safe failure mode (empty string, error return).

4. **Input handling is safe**: `scanf("%u", ...)` writes a fixed-size integer, not a string, so no unbounded read occurs, and its return value is checked.

The only change made is initializing the buffer via `memset` instead of an initializer list — functionally equivalent, but it makes the zeroing explicit at runtime and satisfies some static analyzers that flag partially-initialized aggregate patterns. No genuine weakness required fixing.
