#include <stdio.h>
#include <stdlib.h>

char *concat_tuples(const char **tuple, int size, const char *delimiter) {
    if (size <= 0 || !tuple || !*tuple || !delimiter) return NULL;

    int delimiter_len = 0;
    while (delimiter[delimiter_len]) delimiter_len++;

    int total_len = 0;
    for (int i = 0; i < size; ++i) {
        const char *str = tuple[i];
        if (str) {
            int len = 0;
            while (str[len]) len++;
            total_len += len;
        }
    }
    total_len += (size - 1) * delimiter_len + 1;

    char *result = malloc(total_len);
    if (!result) return NULL;

    char *p = result;
    for (int i = 0; i < size; ++i) {
        const char *str = tuple[i];
        if (str) {
            while (*str) {
                *p++ = *str++;
            }
        }
        if (i < size - 1) {
            for (int j = 0; j < delimiter_len; j++) {
                *p++ = delimiter[j];
            }
        }
    }
    *p = '\0';

    return result;
}

int main() {
    const char *tuple[] = {"Hello", "World", "C", "Programming"};
    int size = 4;
    const char *delimiter = "-";

    char *result = concat_tuples(tuple, size, delimiter);
    if (result) {
        printf("%s\n", result);
        free(result);
    } else {
        fprintf(stderr, "Memory allocation failed.\n");
        return 1;
    }

    return 0;
}
