#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int calculate_min_operations(const int *arr, int n) {
    int max = INT_MIN, operations = 0;
    for (int i = 0; i < n; ++i) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }

    for (int i = 0; i < n; ++i) {
        long diff = (long)max - (long)arr[i];
        if (diff > INT_MAX || diff < 0) {
            fprintf(stderr, "Overflow detected. Cannot compute result.\n");
            exit(1);
        }
        operations += (int)diff;
    }

    return operations;
}

int main() {
    int n, k;
    printf("Enter the number of elements: ");
    scanf("%d", &n);
    int *arr = (int *)malloc(n * sizeof(int));
    if (!arr) {
        fprintf(stderr, "Failed to allocate memory.\n");
        return 1;
    }

    printf("Enter %d elements: ", n);
    for (int i = 0; i < n; ++i) {
        scanf("%d", &arr[i]);
    }

    k = calculate_min_operations(arr, n);
    printf("Minimum operations required: %d\n", k);

    free(arr);
    return 0;
}
