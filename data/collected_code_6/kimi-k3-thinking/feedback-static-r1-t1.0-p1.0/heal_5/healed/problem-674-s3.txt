#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LENGTH 65536

static int safe_copy(void *dst, size_t dst_capacity, const void *src, size_t count)
{
    unsigned char *d = (unsigned char *)dst;
    const unsigned char *s = (const unsigned char *)src;

    if (dst == NULL || (src == NULL && count > 0) || count > dst_capacity) {
        return -1;
    }
    for (size_t i = 0; i < count; i++) {
        d[i] = s[i];
    }
    return 0;
}

static void free_seen(char **seen, size_t count)
{
    if (seen == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(seen[i]);
    }
    free(seen);
}

char *remove_duplicate_words(const char *str)
{
    if (str == NULL) {
        return NULL;
    }

    size_t len = strnlen(str, MAX_INPUT_LENGTH);
    if (len == MAX_INPUT_LENGTH) {
        return NULL;
    }

    size_t result_capacity = len + 1;
    char *result = malloc(result_capacity);
    if (result == NULL) {
        return NULL;
    }
    result[0] = '\0';

    size_t copy_capacity = len + 1;
    char *copy = malloc(copy_capacity);
    if (copy == NULL) {
        free(result);
        return NULL;
    }

    if (safe_copy(copy, copy_capacity, str, len + 1) != 0) {
        free(copy);
        free(result);
        return NULL;
    }

    char **seen = NULL;
    size_t seen_count = 0;
    size_t out_len = 0;

    char *saveptr = NULL;
    char *word = strtok_r(copy, " \t\n", &saveptr);
    while (word != NULL) {
        int found = 0;
        for (size_t i = 0; i < seen_count; i++) {
            if (strcmp(seen[i], word) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            size_t wlen = strnlen(word, copy_capacity);

            if (seen_count + 1 > SIZE_MAX / sizeof(*seen)) {
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }

            char **tmp = realloc(seen, (seen_count + 1) * sizeof(*seen));
            if (tmp == NULL) {
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }
            seen = tmp;

            char *word_copy = malloc(wlen + 1);
            if (word_copy == NULL) {
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }
            if (safe_copy(word_copy, wlen + 1, word, wlen) != 0) {
                free(word_copy);
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }
            word_copy[wlen] = '\0';
            seen[seen_count] = word_copy;
            seen_count++;

            size_t sep = (out_len > 0) ? 1 : 0;
            size_t needed = sep + wlen + 1;

            if (out_len >= result_capacity || needed > result_capacity - out_len) {
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }

            if (sep) {
                result[out_len++] = ' ';
            }

            if (safe_copy(result + out_len, result_capacity - out_len, word, wlen) != 0) {
                free_seen(seen, seen_count);
                free(copy);
                free(result);
                return NULL;
            }
            out_len += wlen;
            result[out_len] = '\0';
        }

        word = strtok_r(NULL, " \t\n", &saveptr);
    }

    free_seen(seen, seen_count);
    free(copy);
    return result;
}

int main(void)
{
    const char *input = "the quick brown fox jumps over the lazy dog the fox";
    char *output = remove_duplicate_words(input);

    if (output == NULL) {
        fprintf(stderr, "Error: failed to process the input string\n");
        return EXIT_FAILURE;
    }

    printf("Input:  %s\n", input);
    printf("Output: %s\n", output);

    free(output);
    return EXIT_SUCCESS;
}
