#include <stdio.h>
#include <stdlib.h>
#include <math.h>

double calculate_discriminant(double a, double b, double c) {
    return b * b - 4.0 * a * c;
}

int main(void) {
    double a, b, c;
    
    printf("Enter coefficients a, b, c: ");
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (fabs(a) < 1e-10) {
        fprintf(stderr, "Error: Not a quadratic equation (a = 0)\n");
        return EXIT_FAILURE;
    }
    
    double discriminant = calculate_discriminant(a, b, c);
    printf("Discriminant: %.6f\n", discriminant);
    
    if (discriminant > 0) {
        double root1 = (-b + sqrt(discriminant)) / (2.0 * a);
        double root2 = (-b - sqrt(discriminant)) / (2.0 * a);
        printf("Roots are real and distinct: %.6f and %.6f\n", root1, root2);
    } else if (fabs(discriminant) < 1e-10) {
        double root = -b / (2.0 * a);
        printf("Roots are real and equal: %.6f\n", root);
    } else {
        double real_part = -b / (2.0 * a);
        double imag_part = sqrt(-discriminant) / (2.0 * a);
        printf("Roots are complex: %.6f + %.6fi and %.6f - %.6fi\n",
               real_part, fabs(imag_part), real_part, fabs(imag_part));
    }
    
    return EXIT_SUCCESS;
}
