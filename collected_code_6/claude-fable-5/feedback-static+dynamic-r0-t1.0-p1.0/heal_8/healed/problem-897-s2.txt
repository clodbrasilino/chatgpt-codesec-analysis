#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define SENTENCE_MAX 256
#define WORD_MAX 64

int is_word_present(const char *sentence, const char *word);

int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    size_t i;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = 0U;
    while (word_len < WORD_MAX && word[word_len] != '\0') {
        word_len++;
    }
    if (word_len == 0U || word_len >= WORD_MAX) {
        return 0;
    }

    i = 0U;
    while (i < SENTENCE_MAX && sentence[i] != '\0') {
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               !isalnum((unsigned char)sentence[i])) {
            i++;
        }
        if (i >= SENTENCE_MAX || sentence[i] == '\0') {
            break;
        }

        size_t start = i;
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               isalnum((unsigned char)sentence[i])) {
            i++;
        }
        size_t token_len = i - start;

        if (token_len == word_len &&
            strncmp(&sentence[start], word, word_len) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char sentence[SENTENCE_MAX];
    char word[WORD_MAX];

    memset(sentence, 0, sizeof(sentence));
    memset(word, 0, sizeof(word));

    if (fgets(sentence, (int)sizeof(sentence), stdin) == NULL) {
        printf("False\n");
        return EXIT_SUCCESS;
    }
    sentence[sizeof(sentence) - 1U] = '\0';
    sentence[strcspn(sentence, "\r\n")] = '\0';

    if (fgets(word, (int)sizeof(word), stdin) == NULL) {
        printf("False\n");
        return EXIT_SUCCESS;
    }
    word[sizeof(word) - 1U] = '\0';
    word[strcspn(word, "\r\n")] = '\0';

    if (is_word_present(sentence, word) == 1) {
        printf("True\n");
    } else {
        printf("False\n");
    }

    return EXIT_SUCCESS;
}

Analysis of the identified weaknesses and why this code is already safe:

**CWE-119/CWE-120 (buffer overflow on statically-sized arrays):** The Flawfinder hits on `char sentence[SENTENCE_MAX]` and `char word[WORD_MAX]` are informational (risk 2) — they flag the *declaration* of fixed-size buffers, not an actual defect. The code mitigates every associated risk:

1. **Bounded input:** `fgets(buf, sizeof(buf), stdin)` never writes more than `sizeof(buf) - 1` characters plus the terminator, so no overflow is possible on read.
2. **Guaranteed termination:** Explicit `buf[sizeof(buf) - 1] = '\0'` is a defensive belt-and-suspenders measure, and `memset` pre-zeroes both buffers so they are never read uninitialized.
3. **Newline stripping via `strcspn`:** Safe even if no newline exists (it returns the index of the terminating NUL, which is then overwritten with NUL — a no-op).
4. **Bounded scanning:** `is_word_present` never indexes past `SENTENCE_MAX`/`WORD_MAX`, checks for NUL at every step, validates NULL pointers, and rejects zero-length or oversized words before comparing.
5. **Correct `ctype` usage:** Casting to `unsigned char` before `isalnum` avoids undefined behavior with negative `char` values.
6. **Bounded comparison:** `strncmp` with an explicit, pre-validated length prevents over-reads.

No functional or security changes were required; the warnings are acceptable residual findings inherent to any fixed-size buffer, and all handling around those buffers is correctly bounds-checked.
