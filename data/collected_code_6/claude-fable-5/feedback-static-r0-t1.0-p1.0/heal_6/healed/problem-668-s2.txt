```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256U

void squeeze_char(char *str, size_t max_len, char ch);
static void discard_remaining_input(void);

static void discard_remaining_input(void)
{
    int input_char;

    do {
        input_char = getchar();
    } while ((input_char != '\n') && (input_char != EOF));
}

void squeeze_char(char *str, size_t max_len, char ch)
{
    size_t read_idx;
    size_t write_idx;
    size_t len;

    if ((str == NULL) || (max_len == 0U)) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }

    write_idx = 0U;

    for (read_idx = 0U; read_idx < len; read_idx++) {
        if ((str[read_idx] == ch) &&
            (read_idx > 0U) &&
            (str[read_idx - 1U] == ch)) {
            continue;
        }
        str[write_idx] = str[read_idx];
        write_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char target;
    int input_char;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    } else if (len == (sizeof(buffer) - 1U)) {
        discard_remaining_input();
    }

    printf("Enter character to squeeze: ");
    input_char = getchar();
    if (input_char == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }

    target = (char)input_char;

    if (input_char != '\n') {
        discard_remaining_input();
    }

    squeeze_char(buffer, sizeof(buffer), target);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The `discard_remaining_input()` function reads characters but never stores them into a buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, preventing an infinite loop on closed/redirected stdin. Using `int` (not `char`) for the return value ensures `EOF` is reliably detected. No fix needed — this is a false positive.

2. **Statically-sized array `buffer` (CWE-119/CWE-120):** The buffer is only ever written to via `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces bounds. Defensive measures already in place:
   - Explicit null termination at `buffer[sizeof(buffer) - 1U]` (belt-and-suspenders; `fgets` already guarantees this on success).
   - `strnlen()` instead of `strlen()`, bounded by `sizeof(buffer)`.
   - Leftover input is drained when the line was truncated, so it can't leak into the next read.

3. **`squeeze_char()` bounds safety:** The function validates `str != NULL` and `max_len != 0`, clamps `len` to `max_len - 1`, and since `write_idx <= read_idx < len < max_len`, the final `str[write_idx] = '\0'` write is always in bounds.

The original code already handled every flagged issue correctly; the rewrite simply removes the analyzer comments while preserving all defensive checks.
