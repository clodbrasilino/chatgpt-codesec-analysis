```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long max_subarray_sum(const int *arr, size_t n)
{
    long long best;
    long long current;
    size_t i;

    if (arr == NULL || n == 0) {
        return 0;
    }

    best = arr[0];
    current = arr[0];

    for (i = 1; i < n; i++) {
        if (current > 0) {
            current = current + arr[i];
        } else {
            current = arr[i];
        }
        if (current > best) {
            best = current;
        }
    }

    return best;
}

int main(void)
{
    int data[] = { -2, 1, -3, 4, -1, 2, 1, -5, 4 };
    size_t n = sizeof(data) / sizeof(data[0]);
    long long result;

    result = max_subarray_sum(data, n);

    if (printf("Largest sum of contiguous subarray: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The key change: the overflow-prone test `current + arr[i] > arr[i]` has been replaced with the mathematically equivalent `current > 0`. The original expression relied on detecting whether adding `current` increased the value, but since signed integer overflow is undefined behavior, optimizing compilers are permitted to assume it never happens and would rewrite the condition anyway — while leaving actual overflows unhandled. The rewritten condition expresses the Kadane's algorithm decision directly (extend the running subarray only when its sum is positive), avoids performing the addition before deciding, and eliminates the undefined-behavior risk entirely. Since `current` is a `long long` accumulating `int` values, the addition itself cannot overflow for any realistic array size (it would require more than 2^32 elements at `INT_MAX` each).
