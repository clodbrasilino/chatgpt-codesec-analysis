```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * 8U)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = getchar();
        guard++;
        if (guard >= (size_t)INT_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);

    return ch;
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        (void)discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the identified weaknesses and how they are mitigated:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — The original loop was already safe since it never writes to a buffer; it only discards characters. To fully satisfy the "check boundaries in loops" concern, I added an iteration guard (`guard`) that bounds the loop even against a pathological stream that never delivers `\n` or `EOF`. The loop still terminates correctly on newline or end-of-file.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — The buffer usage was already properly bounded:
   - `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size.
   - Explicit null-termination at `input[sizeof(input) - 1U]` guarantees a terminated string.
   - `strnlen` with `sizeof(input)` is used instead of `strlen`, so length computation can never run past the buffer.
   - Overlong input is detected (`len == sizeof(input) - 1U` without a trailing newline), remaining stdin data is drained, and the program rejects the input rather than silently truncating.
   - I additionally zero-initialized the array (`= {0}`) as defense-in-depth against any read of uninitialized memory.

3. **Overflow safety in conversion** — The `binary_to_decimal` function validates all pointers, rejects empty or unterminated strings, caps input length at the bit-width of `unsigned long`, validates each character strictly as `'0'`/`'1'`, and checks `value > (ULONG_MAX >> 1)` before shifting, preventing unsigned wraparound (CWE-190).

The remaining flawfinder hits are false positives typical of static pattern matching — the code already performs the bounds checking the tool recommends.
