#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

struct MinHeapNode {
    int element;
    int i;
    int j;
};

void swap(struct MinHeapNode *x, struct MinHeapNode *y) {
    struct MinHeapNode temp = *x;
    *x = *y;
    *y = temp;
}

void minHeapify(struct MinHeapNode harr[], int i, int heap_size) {
    int l = i * 2 + 1;
    int r = i * 2 + 2;
    int smallest = i;

    if (l < heap_size && harr[l].element < harr[i].element) {
        smallest = l;
    }
    if (r < heap_size && harr[r].element < harr[smallest].element) {
        smallest = r;
    }
    if (smallest != i) {
        swap(&harr[i], &harr[smallest]);
        minHeapify(harr, smallest, heap_size);
    }
}

void buildMinHeap(struct MinHeapNode harr[], int n) {
    int i;
    for (i = (n - 2) / 2; i >= 0; i--) {
        minHeapify(harr, i, n);
    }
}

void findSmallestRange(int **arr, int *sizes, int k) {
    int range = INT_MAX;
    int min = INT_MAX, max = INT_MIN;
    int start = -1, end = -1;

    struct MinHeapNode *harr = (struct MinHeapNode *)malloc(k * sizeof(struct MinHeapNode));
    if (harr == NULL) {
        return;
    }

    for (int i = 0; i < k; i++) {
        if (sizes[i] > 0) {
            harr[i].element = arr[i][0];
            harr[i].i = i;
            harr[i].j = 1;
            if (harr[i].element > max) {
                max = harr[i].element;
            }
        } else {
            free(harr);
            return;
        }
    }

    buildMinHeap(harr, k);

    while (1) {
        struct MinHeapNode root = harr[0];
        min = root.element;

        if (range > max - min + 1) {
            range = max - min + 1;
            start = min;
            end = max;
        }

        if (root.j < sizes[root.i]) {
            root.element = arr[root.i][root.j];
            root.j += 1;
            if (root.element > max) {
                max = root.element;
            }
        } else {
            break;
        }

        harr[0] = root;
        minHeapify(harr, 0, k);
    }

    printf("[%d, %d]\n", start, end);
    free(harr);
}

int main(void) {
    int arr1[] = {4, 7, 9, 12, 15};
    int arr2[] = {0, 8, 10, 14, 20};
    int arr3[] = {6, 12, 16, 30, 50};

    int *arr[] = {arr1, arr2, arr3};
    int sizes[] = {5, 5, 5};
    int k = 3;

    findSmallestRange(arr, sizes, k);

    return 0;
}
