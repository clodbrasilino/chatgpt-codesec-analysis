#include <stdio.h>
#include <stdbool.h>
#include <string.h>

typedef enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_DOUBLE,
    TYPE_CHAR,
    TYPE_UNKNOWN
} DataType;

typedef struct {
    DataType type;
    union {
        int i;
        float f;
        double d;
        char c;
    } data;
} TupleElement;

bool check_same_type(const TupleElement *tuple, size_t size) {
    if (tuple == NULL || size == 0) {
        return false;
    }
    
    if (size == 1) {
        return true;
    }
    
    DataType first_type = tuple[0].type;
    
    for (size_t i = 1; i < size; i++) {
        if (tuple[i].type != first_type) {
            return false;
        }
    }
    
    return true;
}

void print_element(const TupleElement *element) {
    if (element == NULL) {
        return;
    }
    
    switch (element->type) {
        case TYPE_INT:
            printf("%d", element->data.i);
            break;
        case TYPE_FLOAT:
            printf("%.2f", element->data.f);
            break;
        case TYPE_DOUBLE:
            printf("%.2lf", element->data.d);
            break;
        case TYPE_CHAR:
            printf("%c", element->data.c);
            break;
        case TYPE_UNKNOWN:
        default:
            printf("UNKNOWN");
            break;
    }
}

void print_tuple(const TupleElement *tuple, size_t size) {
    if (tuple == NULL || size == 0) {
        return;
    }
    
    printf("[");
    for (size_t i = 0; i < size; i++) {
        print_element(&tuple[i]);
        if (i < size - 1) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main(void) {
    TupleElement tuple1[] = {
        {TYPE_INT, {.i = 10}},
        {TYPE_INT, {.i = 20}},
        {TYPE_INT, {.i = 30}}
    };
    
    TupleElement tuple2[] = {
        {TYPE_INT, {.i = 10}},
        {TYPE_FLOAT, {.f = 3.14f}},
        {TYPE_INT, {.i = 30}}
    };
    
    TupleElement tuple3[] = {
        {TYPE_DOUBLE, {.d = 1.5}},
        {TYPE_DOUBLE, {.d = 2.5}},
        {TYPE_DOUBLE, {.d = 3.5}}
    };
    
    TupleElement tuple4[] = {
        {TYPE_CHAR, {.c = 'A'}},
        {TYPE_CHAR, {.c = 'B'}},
        {TYPE_CHAR, {.c = 'C'}}
    };
    
    size_t size1 = sizeof(tuple1) / sizeof(tuple1[0]);
    size_t size2 = sizeof(tuple2) / sizeof(tuple2[0]);
    size_t size3 = sizeof(tuple3) / sizeof(tuple3[0]);
    size_t size4 = sizeof(tuple4) / sizeof(tuple4[0]);
    
    printf("Tuple1 all same type: %s\n", check_same_type(tuple1, size1) ? "true" : "false");
    printf("Tuple2 all same type: %s\n", check_same_type(tuple2, size2) ? "true" : "false");
    printf("Tuple3 all same type: %s\n", check_same_type(tuple3, size3) ? "true" : "false");
    printf("Tuple4 all same type: %s\n", check_same_type(tuple4, size4) ? "true" : "false");
    printf("NULL tuple: %s\n", check_same_type(NULL, 5) ? "true" : "false");
    printf("Empty tuple: %s\n", check_same_type(tuple1, 0) ? "true" : "false");
    
    printf("Tuple1 contents: ");
    print_tuple(tuple1, size1);
    printf("Tuple2 contents: ");
    print_tuple(tuple2, size2);
    printf("Tuple3 contents: ");
    print_tuple(tuple3, size3);
    printf("Tuple4 contents: ");
    print_tuple(tuple4, size4);
    
    return 0;
}
