#include <stdio.h>
#include <stdlib.h>

#define MOD 1000000007

int count_tilings(int n) {
    if (n < 0) return 0;
    if (n == 0) return 1;
    if (n == 1) return 1;
    if (n == 2) return 2;
    
    int *dp = (int*)malloc((n + 1) * sizeof(int));
    if (dp == NULL) {
        return -1;
    }
    
    dp[0] = 1;
    dp[1] = 1;
    dp[2] = 2;
    
    for (int i = 3; i <= n; i++) {
        dp[i] = (dp[i-1] + dp[i-2]) % MOD;
    }
    
    int result = dp[n];
    free(dp);
    return result;
}

int main(void) {
    int n;
    
    printf("Enter the width of the board (2 x n): ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    if (n < 0) {
        fprintf(stderr, "Width must be non-negative\n");
        return 1;
    }
    
    int result = count_tilings(n);
    if (result == -1) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }
    
    printf("Number of ways to tile a 2 x %d board: %d\n", n, result);
    
    return 0;
}
