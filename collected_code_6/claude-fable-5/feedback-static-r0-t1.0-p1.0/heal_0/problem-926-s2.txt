#include <stdio.h>
#include <stdlib.h>

static unsigned long long binomial(unsigned int n, unsigned int k)
{
    unsigned long long result = 1ULL;
    unsigned int i;

    if (k > n) {
        return 0ULL;
    }
    if (k > n - k) {
        k = n - k;
    }
    for (i = 0U; i < k; i++) {
        result = result * (unsigned long long)(n - i);
        result = result / (unsigned long long)(i + 1U);
    }
    return result;
}

static unsigned long long derangement(unsigned int n)
{
    unsigned long long prev2;
    unsigned long long prev1;
    unsigned long long current;
    unsigned int i;

    if (n == 0U) {
        return 1ULL;
    }
    if (n == 1U) {
        return 0ULL;
    }
    prev2 = 1ULL;
    prev1 = 0ULL;
    current = 0ULL;
    for (i = 2U; i <= n; i++) {
        current = (unsigned long long)(i - 1U) * (prev1 + prev2);
        prev2 = prev1;
        prev1 = current;
    }
    return current;
}

static int rencontres(unsigned int n, unsigned int m, unsigned long long *out)
{
    unsigned long long b;
    unsigned long long d;

    if (out == NULL) {
        return -1;
    }
    if (m > n) {
        *out = 0ULL;
        return 0;
    }
    if (n > 20U) {
        return -1;
    }
    b = binomial(n, m);
    d = derangement(n - m);
    if ((b != 0ULL) && (d > (unsigned long long)-1 / b)) {
        return -1;
    }
    *out = b * d;
    return 0;
}

int main(void)
{
    unsigned int n;
    unsigned int m;
    unsigned long long result;
    int scanned;

    printf("Enter n and m: ");
    scanned = scanf("%u %u", &n, &m);
    if (scanned != 2) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    if (rencontres(n, m, &result) != 0) {
        fprintf(stderr, "Computation error or input too large\n");
        return EXIT_FAILURE;
    }
    printf("D(%u, %u) = %llu\n", n, m, result);
    return EXIT_SUCCESS;
}
