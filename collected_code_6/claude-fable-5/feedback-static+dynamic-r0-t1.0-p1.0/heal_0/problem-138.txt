```c
#include <stdio.h>
#include <stdbool.h>

bool canRepresentAsSumOfNonZeroPowersOf2(long long n)
{
    if (n <= 0)
    {
        return false;
    }
    return (n % 2 == 0);
}

int main(void)
{
    long long number;

    printf("Enter a number: ");
    if (scanf("%lld", &number) != 1)
    {
        fprintf(stderr, "Invalid input.\n");
        return 1;
    }

    if (canRepresentAsSumOfNonZeroPowersOf2(number))
    {
        printf("%lld can be represented as sum of non-zero powers of 2.\n", number);
    }
    else
    {
        printf("%lld cannot be represented as sum of non-zero powers of 2.\n", number);
    }

    return 0;
}
```

**Explanation of the logic:** A number can be represented as a sum of non-zero powers of 2 (i.e., 2^1, 2^2, 2^3, ...) if and only if it is a positive even number. Since the smallest non-zero power of 2 is 2 itself, every term in the sum is even, so the total must be even. Conversely, any positive even number `n` can be written as a sum of `n/2` copies of 2^1.
