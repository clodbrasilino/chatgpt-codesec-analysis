#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *create_node(int value)
{
    struct Node *node = malloc(sizeof(struct Node));
    if (node == NULL) {
        return NULL;
    }
    node->data = value;
    node->next = NULL;
    return node;
}

int append(struct Node **head, int value)
{
    struct Node *node;
    struct Node *cur;

    if (head == NULL) {
        return -1;
    }

    node = create_node(value);
    if (node == NULL) {
        return -1;
    }

    if (*head == NULL) {
        *head = node;
        return 0;
    }

    cur = *head;
    while (cur->next != NULL) {
        cur = cur->next;
    }
    cur->next = node;
    return 0;
}

int replace_last(struct Node **head, struct Node *replacement)
{
    struct Node *cur;
    struct Node *prev;

    if (head == NULL || *head == NULL) {
        return -1;
    }

    prev = NULL;
    cur = *head;
    while (cur->next != NULL) {
        prev = cur;
        cur = cur->next;
    }

    free(cur);

    if (prev == NULL) {
        *head = replacement;
    } else {
        prev->next = replacement;
    }
    return 0;
}

void print_list(const struct Node *head)
{
    const struct Node *cur = head;
    while (cur != NULL) {
        printf("%d ", cur->data);
        cur = cur->next;
    }
    printf("\n");
}

void free_list(struct Node *head)
{
    while (head != NULL) {
        struct Node *tmp = head;
        head = head->next;
        free(tmp);
    }
}

int main(void)
{
    struct Node *list1 = NULL;
    struct Node *list2 = NULL;
    int i;

    for (i = 1; i <= 5; i++) {
        if (append(&list1, i) != 0) {
            free_list(list1);
            return EXIT_FAILURE;
        }
    }

    for (i = 10; i <= 30; i += 10) {
        if (append(&list2, i) != 0) {
            free_list(list1);
            free_list(list2);
            return EXIT_FAILURE;
        }
    }

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);

    if (replace_last(&list1, list2) != 0) {
        free_list(list1);
        free_list(list2);
        return EXIT_FAILURE;
    }

    printf("Result: ");
    print_list(list1);

    free_list(list1);
    return EXIT_SUCCESS;
}
