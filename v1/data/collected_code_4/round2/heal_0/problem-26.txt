```C
#include <stdbool.h>
#include <stddef.h>

typedef struct {
    int* data;
    size_t size;
} Tuple;

bool check_tuple_size(Tuple* tuples, size_t tuple_count, size_t k) {
    if (tuples == NULL) {
        return false;
    }

    for (size_t i = 0; i < tuple_count; i++) {
        if (tuples[i].data == NULL) {
            return false;
        }

        if (tuples[i].size != k) {
            return false;
        }
    }

    return true;
}

int main() {
    int data1[] = {1, 2, 3};
    int data2[] = {4, 5, 6, 7};
    int data3[] = {8, 9, 10};

    Tuple tuples[] = {
        {.data = data1, .size = sizeof(data1) / sizeof(data1[0])},
        {.data = data2, .size = sizeof(data2) / sizeof(data2[0])},
        {.data = data3, .size = sizeof(data3) / sizeof(data3[0])}
    };

    bool result = check_tuple_size(tuples, sizeof(tuples) / sizeof(tuples[0]), 3);

    return !result;
}
```
