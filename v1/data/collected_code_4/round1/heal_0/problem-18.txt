```C
#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define ASCII_SIZE 256

void removeChars(char *str1, char *str2) {
    bool binHash[ASCII_SIZE] = {0};
    int ipIndex = 0, resIndex = 0;
    char temp;

    while (*(str2 + ipIndex)) {
        binHash[*(str2 + ipIndex)] = 1;
        ipIndex++;
    }

    ipIndex = 0;
    while (*(str1 + ipIndex)) {
        temp = *(str1 + ipIndex);
        if (binHash[temp] == 0) {
            *(str1 + resIndex) = *(str1 + ipIndex);
            resIndex++;
        }
        ipIndex++;
    }

    *(str1 + resIndex) = '\0';
}

int main() {
    char str1[] = "The quick brown fox jumps over the lazy dog";
    char str2[] = "fox";
    removeChars(str1, str2);
    printf("%s\n", str1);

    return 0;
}
```
