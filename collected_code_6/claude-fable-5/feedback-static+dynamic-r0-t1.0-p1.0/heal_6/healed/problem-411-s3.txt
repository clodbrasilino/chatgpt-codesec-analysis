```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    size_t len;
    size_t i;
    size_t out_pos;
    char *result;
    int capitalize_next;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1U);
    if (result == NULL) {
        return NULL;
    }

    out_pos = 0U;
    capitalize_next = 1;

    for (i = 0U; i < len; i++) {
        char c = input[i];

        if (c == '_') {
            capitalize_next = 1;
            continue;
        }

        if (capitalize_next != 0) {
            result[out_pos] = (char)toupper((unsigned char)c);
            capitalize_next = 0;
        } else {
            result[out_pos] = c;
        }
        out_pos++;
    }

    result[out_pos] = '\0';

    return result;
}

int main(void)
{
    const char *tests[] = {
        "android_tv",
        "google_pixel",
        "apple_watch",
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    setvbuf(stdout, NULL, _IONBF, 0);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s\n", converted);
        fflush(stdout);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The conversion logic in the original `snake_to_camel` function was actually correct — `"android_tv"` properly maps to `"AndroidTv"`, `"google_pixel"` to `"GooglePixel"`, and `"apple_watch"` to `"AppleWatch"`. The buffer sizing is safe as well: since underscores are dropped, the output can never exceed the input length, so `malloc(len + 1U)` leaves adequate room for the null terminator.

The symptom reported in the comments — expected values but `<no output>` received — points not to a logic defect but to an **output delivery failure**. When a program's stdout is redirected to a pipe (as automated test harnesses typically do), stdio switches from line-buffered to fully-buffered mode. If the process terminates abnormally, or the harness reads before the buffer is flushed at exit, the tests observe empty output despite correct computation.

The fix applied:

1. **`setvbuf(stdout, NULL, _IONBF, 0)`** — disables stdout buffering entirely so every `printf` reaches the consumer immediately.
2. **`fflush(stdout)` after each print** — a belt-and-braces guarantee that each result is pushed out before proceeding, protecting against any environment where the `setvbuf` call is ignored.

All other defensive properties of the original are preserved: `NULL` input rejection, bounded length checking via `strnlen`, `malloc` failure handling, safe `toupper` usage via the `unsigned char` cast, and correct memory release with `free`.
