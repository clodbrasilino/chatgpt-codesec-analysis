#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>
#include <errno.h>

#define BINARY_BUFFER_SIZE (sizeof(unsigned int) * CHAR_BIT + 1)

int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size) {
    size_t index;
    size_t i;
    
    if (buffer == NULL || buffer_size < BINARY_BUFFER_SIZE) {
        return -1;
    }
    
    index = 0;
    
    if (decimal == 0) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index++] = '0';
    } else {
        unsigned int temp = decimal;
        while (temp > 0) {
            if (index + 1 >= buffer_size) {
                return -1;
            }
            buffer[index++] = (char)((temp % 2) + '0');
            temp = temp / 2;
        }
    }
    
    buffer[index] = '\0';
    
    for (i = 0; i < index / 2; i++) {
        char temp_char = buffer[i];
        buffer[i] = buffer[index - 1 - i];
        buffer[index - 1 - i] = temp_char;
    }
    
    return 0;
}

int main(void) {
    char binary_buffer[BINARY_BUFFER_SIZE];
    unsigned int decimal_number;
    int result;
    char input_buffer[32];
    char *endptr;
    unsigned long value;
    
    printf("Enter a decimal number: ");
    
    if (fgets(input_buffer, sizeof(input_buffer), stdin) == NULL) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    errno = 0;
    value = strtoul(input_buffer, &endptr, 10);
    
    if (errno != 0 || endptr == input_buffer || value > UINT_MAX) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    decimal_number = (unsigned int)value;
    
    result = decimal_to_binary(decimal_number, binary_buffer, sizeof(binary_buffer));
    
    if (result != 0) {
        fprintf(stderr, "Error: Conversion failed\n");
        return EXIT_FAILURE;
    }
    
    printf("Binary equivalent: %s\n", binary_buffer);
    
    return EXIT_SUCCESS;
}
