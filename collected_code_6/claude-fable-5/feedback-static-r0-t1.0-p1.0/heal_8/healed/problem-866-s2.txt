#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month);
static void discard_remaining_input(void);

static bool has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    memset(lowered, 0, sizeof(lowered));

    for (i = 0; i < len && i < sizeof(lowered) - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(lowered, months_31[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No input provided.\n");
        return EXIT_FAILURE;
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char lowered[MAX_INPUT_LEN]` in `has_31_days` (CWE-119/CWE-120):** This is a false positive in practice. The code uses `strnlen` with an explicit bound, rejects inputs of length `>= MAX_INPUT_LEN`, and the copy loop is doubly guarded by `i < len && i < sizeof(lowered) - 1`, followed by explicit null termination. No overflow is possible.

2. **`getchar` in `discard_remaining_input` (CWE-120/CWE-20):** Also a false positive — the return value is stored in an `int` (correctly, so `EOF` is distinguishable from valid characters), nothing is written to a buffer, and the loop terminates on both `\n` and `EOF`, so it cannot spin forever on a closed stream.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits the read to the buffer size. The code additionally forces null termination, strips the trailing newline safely with `strnlen`, and drains any excess input from `stdin` to avoid leftover data affecting subsequent reads.

The original code already follows defensive-coding best practices (bounded reads, `strnlen` instead of `strlen`, `tolower` fed an `unsigned char` cast to avoid UB on negative `char` values, checked `fflush`/`fgets` return values), so the rewrite preserves the logic unchanged with the comments removed as requested.
