#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void findThreeClosest(const int *arr1, int n1, const int *arr2, int n2, const int *arr3, int n3, int *res1, int *res2, int *res3) {
    if (n1 <= 0 || n2 <= 0 || n3 <= 0) {
        return;
    }

    int i = 0, j = 0, k = 0;
    int min_diff = INT_MAX;

    while (i < n1 && j < n2 && k < n3) {
        int a = arr1[i], b = arr2[j], c = arr3[k];

        int min_val = a < b ? (a < c ? a : c) : (b < c ? b : c);
        int max_val = a > b ? (a > c ? a : c) : (b > c ? b : c);
        int current_diff = max_val - min_val;

        if (current_diff < min_diff) {
            min_diff = current_diff;
            *res1 = a;
            *res2 = b;
            *res3 = c;

            if (min_diff == 0) {
                break;
            }
        }

        if (min_val == a) {
            i++;
        } else if (min_val == b) {
            j++;
        } else {
            k++;
        }
    }
}

int main(void) {
    int arr1[] = {1, 4, 10};
    int arr2[] = {2, 15, 20};
    int arr3[] = {10, 12};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int n3 = sizeof(arr3) / sizeof(arr3[0]);

    int res1, res2, res3;
    findThreeClosest(arr1, n1, arr2, n2, arr3, n3, &res1, &res2, &res3);

    printf("%d %d %d\n", res1, res2, res3);

    return 0;
}
