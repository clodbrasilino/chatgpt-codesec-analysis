```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_LEN 11
#define INPUT_BUF_LEN 64

static int is_valid_format(const char *date)
{
    size_t i;
    size_t len;

    if (date == NULL) {
        return 0;
    }

    len = strnlen(date, DATE_LEN);
    if (len != DATE_LEN - 1) {
        return 0;
    }

    for (i = 0; i < DATE_LEN - 1; i++) {
        if (i == 4 || i == 7) {
            if (date[i] != '-') {
                return 0;
            }
        } else {
            if (!isdigit((unsigned char)date[i])) {
                return 0;
            }
        }
    }

    return 1;
}

static int is_valid_date(int year, int month, int day)
{
    int days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (year < 1 || month < 1 || month > 12 || day < 1) {
        return 0;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        days_in_month[1] = 29;
    }

    if (day > days_in_month[month - 1]) {
        return 0;
    }

    return 1;
}

static int convert_date(const char *input, char *output, size_t output_size)
{
    int year;
    int month;
    int day;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_LEN) {
        return -1;
    }

    if (!is_valid_format(input)) {
        return -1;
    }

    if (sscanf(input, "%4d-%2d-%2d", &year, &month, &day) != 3) {
        return -1;
    }

    if (!is_valid_date(year, month, day)) {
        return -1;
    }

    written = snprintf(output, output_size, "%02d-%02d-%04d", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_LEN];
    char output[DATE_LEN];

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date: %s\n", input);
        return EXIT_FAILURE;
    }

    printf("Converted date: %s\n", output);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

The Flawfinder warnings on `input[INPUT_BUF_LEN]` and `output[DATE_LEN]` are informational (risk level 2) — they flag *any* statically-sized array as a potential CWE-119/CWE-120 risk. In this code, both are already properly mitigated, so no functional changes were required:

1. **`input` buffer:** All reads go through `fgets(input, sizeof(input), stdin)`, which is bounds-limited and guarantees null termination within the buffer. The explicit `input[sizeof(input) - 1] = '\0'` is defensive redundancy, and `strcspn` safely strips the newline without risk of overflow.

2. **`output` buffer:** Writes occur only via `snprintf(output, output_size, ...)`, which truncates rather than overflows. The return value is checked for both encoding errors (`written < 0`) and truncation (`written >= output_size`), and `convert_date` rejects any caller-supplied buffer smaller than `DATE_LEN` before writing.

3. **Parsing safety:** `sscanf` uses width-limited specifiers (`%4d`, `%2d`, `%2d`), and `is_valid_format` uses `strnlen` (bounded) plus per-character validation with the `(unsigned char)` cast for `isdigit`, avoiding undefined behavior on negative `char` values.

The code is a correct example of safely using fixed-size buffers; the static analyzer warnings are false positives given the consistent use of bounds-checked functions throughout.
