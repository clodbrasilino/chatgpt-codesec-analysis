```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096U

typedef struct {
    size_t *positions;
    size_t count;
} SearchResult;

int find_substring_occurrences(const char *text, const char *pattern, SearchResult *result)
{
    const char *current;
    size_t pattern_len;
    size_t text_len;
    size_t capacity;
    size_t *temp;

    if (text == NULL || pattern == NULL || result == NULL) {
        return -1;
    }

    text_len = strnlen(text, MAX_INPUT_LEN);
    if (text_len >= MAX_INPUT_LEN) {
        return -1;
    }

    pattern_len = strnlen(pattern, MAX_INPUT_LEN);
    if (pattern_len == 0 || pattern_len >= MAX_INPUT_LEN) {
        return -1;
    }

    if (pattern_len > text_len) {
        result->positions = NULL;
        result->count = 0;
        return 0;
    }

    result->positions = NULL;
    result->count = 0;
    capacity = 0;

    current = strstr(text, pattern);
    while (current != NULL) {
        if (result->count == capacity) {
            size_t new_capacity = (capacity == 0) ? 8 : capacity * 2;
            if (new_capacity > SIZE_MAX / sizeof(size_t)) {
                free(result->positions);
                result->positions = NULL;
                result->count = 0;
                return -1;
            }
            temp = realloc(result->positions, new_capacity * sizeof(size_t));
            if (temp == NULL) {
                free(result->positions);
                result->positions = NULL;
                result->count = 0;
                return -1;
            }
            result->positions = temp;
            capacity = new_capacity;
        }
        result->positions[result->count] = (size_t)(current - text);
        result->count++;
        current = strstr(current + 1, pattern);
    }

    return 0;
}

void free_search_result(SearchResult *result)
{
    if (result != NULL) {
        free(result->positions);
        result->positions = NULL;
        result->count = 0;
    }
}

int main(void)
{
    const char *text = "the quick brown fox jumps over the lazy dog, the end";
    const char *pattern = "the";
    SearchResult result;
    size_t i;

    if (find_substring_occurrences(text, pattern, &result) != 0) {
        fprintf(stderr, "Error: search failed\n");
        return EXIT_FAILURE;
    }

    printf("Text: \"%s\"\n", text);
    printf("Pattern: \"%s\"\n", pattern);
    printf("Occurrences: %zu\n", result.count);

    for (i = 0; i < result.count; i++) {
        printf("Occurrence %zu at position %zu\n", i + 1, result.positions[i]);
    }

    free_search_result(&result);

    return EXIT_SUCCESS;
}
```

**Key changes made:**

1. **CWE-126 mitigation (the flagged `strlen` issue):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN` for both `pattern` and `text`. If either string is not null-terminated within the bound, the length check (`>= MAX_INPUT_LEN`) rejects the input instead of over-reading past the buffer.

2. **Integer overflow guard:** Added a check that `new_capacity * sizeof(size_t)` cannot overflow (`new_capacity > SIZE_MAX / sizeof(size_t)`) before calling `realloc`, preventing an undersized allocation on capacity doubling.

3. **Early exit optimization/safety:** If the pattern is longer than the text, the function returns immediately with an empty, well-defined result rather than scanning.

4. Added `<stdint.h>` for `SIZE_MAX`. Note that `strnlen` requires POSIX (`_POSIX_C_SOURCE >= 200809L`) or C23; if strict C89/C99 portability is needed, a small local bounded-length helper can be substituted.
