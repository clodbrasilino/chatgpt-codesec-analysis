#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int extract_date_from_url(const char *url, int *year, int *month, int *day) {
    regex_t regex;
    regmatch_t matches[4];
    const char *pattern;
    int ret;
    int y, m, d;
    int i;

    if (url == NULL || year == NULL || month == NULL || day == NULL) {
        return 0;
    }

    pattern = "([0-9]{4})[-/]([0-9]{2})[-/]([0-9]{2})";
    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return 0;
    }

    ret = regexec(&regex, url, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return 0;
    }

    if (matches[1].rm_so < 0 || matches[2].rm_so < 0 || matches[3].rm_so < 0) {
        regfree(&regex);
        return 0;
    }

    y = 0;
    for (i = matches[1].rm_so; i < matches[1].rm_eo; i++) {
        if (url[i] < '0' || url[i] > '9') {
            regfree(&regex);
            return 0;
        }
        y = y * 10 + (url[i] - '0');
    }
    if (y < 1 || y > 9999) {
        regfree(&regex);
        return 0;
    }

    m = 0;
    for (i = matches[2].rm_so; i < matches[2].rm_eo; i++) {
        if (url[i] < '0' || url[i] > '9') {
            regfree(&regex);
            return 0;
        }
        m = m * 10 + (url[i] - '0');
    }
    if (m < 1 || m > 12) {
        regfree(&regex);
        return 0;
    }

    d = 0;
    for (i = matches[3].rm_so; i < matches[3].rm_eo; i++) {
        if (url[i] < '0' || url[i] > '9') {
            regfree(&regex);
            return 0;
        }
        d = d * 10 + (url[i] - '0');
    }
    if (d < 1 || d > 31) {
        regfree(&regex);
        return 0;
    }

    *year = y;
    *month = m;
    *day = d;

    regfree(&regex);
    return 1;
}

int main(void) {
    const char *url = "https://example.com/news/2024/12/25/headline";
    int year = 0;
    int month = 0;
    int day = 0;

    if (extract_date_from_url(url, &year, &month, &day)) {
        printf("%04d-%02d-%02d\n", year, month, day);
    } else {
        printf("No date found\n");
    }

    return 0;
}
