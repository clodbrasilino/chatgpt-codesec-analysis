```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `char input[INPUT_SIZE]` (CWE-119/CWE-120 — statically-sized array):**
- The buffer is only written by `fgets(input, sizeof(input), stdin)`, which strictly bounds the write to `sizeof(input) - 1` characters plus the null terminator. No overflow is possible.
- A defensive `input[sizeof(input) - 1] = '\0'` guarantees termination even in pathological cases.
- Truncated (over-long) input is detected by checking for a missing trailing newline at maximum length; the program then drains stdin and rejects the input rather than silently processing a truncated value.
- `strnlen` is used instead of `strlen`, bounding all length computations to the buffer size.

**2. Flawfinder `getchar` in a loop (CWE-120/CWE-20):**
- `getchar()` writes to no buffer; its return value is stored in an `int` (not `char`), so `EOF` is correctly distinguishable from valid characters.
- The loop terminates on both `'\n'` and `EOF`, so it cannot spin forever on a closed or exhausted stream.

**Additional hardening already present:**
- `octal_to_decimal` validates all pointers, rejects empty and unterminated input, validates every character against the octal range, and performs a pre-multiplication overflow check (`value > (LONG_MAX - digit) / 8`) preventing signed integer overflow (CWE-190).
- `fflush(stdout)` return value is checked, and all error paths return distinct diagnostics on `stderr`.

Both Flawfinder hits are informational (risk 1–2) false positives; the mitigations required by CWE-120, CWE-119, and CWE-20 are already fully implemented, so the rewritten code is functionally identical with the comments removed.
