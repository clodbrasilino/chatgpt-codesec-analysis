#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static int cube_sum_minus_sum(unsigned int n, unsigned long long *result)
{
    unsigned long long s;
    unsigned long long s_squared;

    if (result == NULL) {
        return -1;
    }

    s = ((unsigned long long)n * ((unsigned long long)n + 1ULL)) / 2ULL;

    if (s != 0ULL && s > ULLONG_MAX / s) {
        return -1;
    }

    s_squared = s * s;

    if (s_squared < s) {
        return -1;
    }

    *result = s_squared - s;
    return 0;
}

int main(void)
{
    unsigned int n;
    unsigned long long difference;
    int scan_status;

    printf("Enter a natural number n: ");

    scan_status = scanf("%u", &n);
    if (scan_status != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (n == 0U || n > 100000U) {
        fprintf(stderr, "Error: n must be between 1 and 100000.\n");
        return EXIT_FAILURE;
    }

    if (cube_sum_minus_sum(n, &difference) != 0) {
        fprintf(stderr, "Error: computation overflow.\n");
        return EXIT_FAILURE;
    }

    printf("Difference: %llu\n", difference);

    return EXIT_SUCCESS;
}
