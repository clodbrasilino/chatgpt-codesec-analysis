```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char *replace_spaces(const char *input)
{
    size_t len;
    size_t space_count;
    size_t new_len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && input[MAX_INPUT_LEN - 1] != '\0') {
        return NULL;
    }

    space_count = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == ' ') {
            space_count++;
        }
    }

    if (space_count > (SIZE_MAX - len - 1) / 2) {
        return NULL;
    }

    new_len = len + (space_count * 2);

    result = malloc(new_len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == ' ') {
            result[j] = '%';
            result[j + 1] = '2';
            result[j + 2] = '0';
            j += 3;
        } else {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello World",
        "  leading and trailing  ",
        "nospace",
        ""
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *converted = replace_spaces(test_strings[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for input: \"%s\"\n",
                    test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Input: \"%s\" -> Output: \"%s\"\n",
                   test_strings[i], converted) < 0) {
            free(converted);
            return EXIT_FAILURE;
        }
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

Key changes made:

1. **Added `<stdint.h>`** — the original code used `SIZE_MAX` without including the header that defines it, which would fail to compile (or rely on an accidental transitive include).

2. **Replaced `strlen` with `strnlen`** (CWE-126 mitigation) — bounded the read to `MAX_INPUT_LEN` and rejected inputs that are not NUL-terminated within that bound, preventing an unbounded over-read on non-terminated strings.

3. **Reduced the scope of `converted`** — moved the declaration inside the loop body as flagged by the `[variableScope]` warning. This also removes the need for the redundant `converted = NULL;` reset after `free`.

4. **Retained the overflow guard** — the `(SIZE_MAX - len - 1) / 2` check correctly prevents `size_t` overflow when computing `new_len` and the `malloc(new_len + 1)` allocation size.
