```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>

#define MAX_INPUT_SIZE 1024

static size_t safe_strnlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL)
    {
        return 0;
    }

    for (i = 0; i < max_len; i++)
    {
        if (str[i] == '\0')
        {
            break;
        }
    }

    return i;
}

unsigned long count_substrings(const char *str, size_t max_len)
{
    unsigned long freq[256] = {0};
    unsigned long total = 0;
    size_t i;
    size_t len;

    if (str == NULL)
    {
        return 0;
    }

    len = safe_strnlen(str, max_len);

    for (i = 0; i < len; i++)
    {
        freq[(unsigned char)str[i]]++;
    }

    for (i = 0; i < 256; i++)
    {
        total += (freq[i] * (freq[i] + 1)) / 2;
    }

    return total;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    unsigned long result;
    size_t len;
    char *newline;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL)
    {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    newline = memchr(buffer, '\n', sizeof(buffer));
    if (newline != NULL)
    {
        *newline = '\0';
    }

    len = safe_strnlen(buffer, sizeof(buffer));
    if (len >= sizeof(buffer))
    {
        fprintf(stderr, "Input too long\n");
        return 1;
    }

    result = count_substrings(buffer, sizeof(buffer));

    if (printf("%lu\n", result) < 0)
    {
        return 1;
    }

    return 0;
}
```

**Analysis of the original code and fixes applied:**

The original code was already quite defensive, but the Flawfinder warning (CWE-119/CWE-120) about the statically-sized `buffer` array was mitigated through several existing safeguards that I preserved and strengthened:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` already limits the read to the buffer size, preventing overflow at the read site. This is the primary fix for the CWE-120 concern.

2. **Guaranteed NUL termination** — `buffer[sizeof(buffer) - 1] = '\0'` ensures the buffer is always terminated even in pathological cases.

3. **Bounded length calculation** — The original used `strnlen`, which is POSIX, not ISO C. I added `_POSIX_C_SOURCE 200809L` for portability and provided a `safe_strnlen` fallback implementation with an explicit NULL check, so the code compiles cleanly under strict `-std=c99`/`-std=c11` without implicit declaration warnings — an implicit declaration of `strnlen` would itself be a latent defect (return type assumed `int`).

4. **Bounded search** — `memchr` with `sizeof(buffer)` rather than `strchr` avoids reading past the buffer if termination were ever violated.

5. **NULL guard in `count_substrings`** plus `(unsigned char)` cast before array indexing prevents negative indexing into `freq[]` on platforms where `char` is signed — an important existing protection that was kept.

6. **Length sanity check** (`len >= sizeof(buffer)`) is technically unreachable after the forced termination but serves as defense-in-depth.

No overflow of `total` is realistic: with at most 1023 characters of one value, the maximum is 1023·1024/2 = 523,776, well within `unsigned long` range.
