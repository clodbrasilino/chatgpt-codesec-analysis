#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int minimumTotal(int **triangle, int numRows) {
    if (triangle == NULL || numRows <= 0) {
        return 0;
    }

    int *dp = (int *)malloc((size_t)numRows * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < numRows; i++) {
        dp[i] = triangle[numRows - 1][i];
    }

    for (int row = numRows - 2; row >= 0; row--) {
        for (int col = 0; col <= row; col++) {
            int minChild = (dp[col] < dp[col + 1]) ? dp[col] : dp[col + 1];
            if (triangle[row][col] > INT_MAX - minChild) {
                free(dp);
                return 0;
            }
            dp[col] = triangle[row][col] + minChild;
        }
    }

    int result = dp[0];
    free(dp);
    return result;
}

int main(void) {
    int numRows = 4;

    int **triangle = (int **)malloc((size_t)numRows * sizeof(int *));
    if (triangle == NULL) {
        return EXIT_FAILURE;
    }

    for (int i = 0; i < numRows; i++) {
        triangle[i] = (int *)malloc((size_t)(i + 1) * sizeof(int));
        if (triangle[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(triangle[j]);
            }
            free(triangle);
            return EXIT_FAILURE;
        }
    }

    triangle[0][0] = 2;
    triangle[1][0] = 3;
    triangle[1][1] = 4;
    triangle[2][0] = 6;
    triangle[2][1] = 5;
    triangle[2][2] = 7;
    triangle[3][0] = 4;
    triangle[3][1] = 1;
    triangle[3][2] = 8;
    triangle[3][3] = 3;

    int result = minimumTotal(triangle, numRows);
    printf("Minimum total path sum: %d\n", result);

    for (int i = 0; i < numRows; i++) {
        free(triangle[i]);
    }
    free(triangle);

    return EXIT_SUCCESS;
}
