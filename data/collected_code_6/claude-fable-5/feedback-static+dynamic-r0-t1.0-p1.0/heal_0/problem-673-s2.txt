```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int convert_list_to_integer(const int *list, size_t count, int *result)
{
    size_t i;
    int value;

    if (list == NULL || result == NULL || count == 0U) {
        return -1;
    }

    value = 0;

    for (i = 0U; i < count; i++) {
        if (list[i] < 0 || list[i] > 9) {
            return -1;
        }
        if (value > (INT_MAX - list[i]) / 10) {
            return -1;
        }
        value = (value * 10) + list[i];
    }

    *result = value;
    return 0;
}

int main(void)
{
    int numbers[] = {1, 2, 3, 4, 5};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int combined = 0;
    int status;

    status = convert_list_to_integer(numbers, count, &combined);

    if (status != 0) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Combined integer: %d\n", combined) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
