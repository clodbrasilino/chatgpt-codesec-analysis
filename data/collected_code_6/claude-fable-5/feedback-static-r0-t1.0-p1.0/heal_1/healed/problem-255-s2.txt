```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NUM_COLOURS 3
#define MAX_SELECTION 16

static const char *colours[NUM_COLOURS] = { "Red", "Green", "Blue" };

static void print_combination(const int *indices, size_t count)
{
    size_t i;

    if (indices == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        if ((indices[i] < 0) || (indices[i] >= NUM_COLOURS)) {
            return;
        }
    }

    for (i = 0; i < count; i++) {
        printf("%s", colours[indices[i]]);
        if (i + 1U < count) {
            printf(" ");
        }
    }
    printf("\n");
}

static void generate_combinations(int *indices, size_t depth, size_t count, int start)
{
    int i;

    if (indices == NULL) {
        return;
    }

    if (depth >= count) {
        print_combination(indices, count);
        return;
    }

    if ((start < 0) || (start >= NUM_COLOURS)) {
        return;
    }

    for (i = start; i < NUM_COLOURS; i++) {
        indices[depth] = i;
        generate_combinations(indices, depth + 1U, count, i);
    }
}

static int choose_colours(size_t count)
{
    int *indices;

    if ((count == 0U) || (count > MAX_SELECTION)) {
        return -1;
    }

    if (count > (SIZE_MAX / sizeof(int))) {
        return -1;
    }

    indices = (int *)malloc(count * sizeof(int));
    if (indices == NULL) {
        return -1;
    }

    (void)memset(indices, 0, count * sizeof(int));
    generate_combinations(indices, 0U, count, 0);
    free(indices);
    indices = NULL;

    return 0;
}

int main(void)
{
    size_t count = 2U;
    int result;

    printf("Combinations with repetition choosing %zu colour(s) from %d colours:\n",
           count, NUM_COLOURS);

    result = choose_colours(count);
    if (result != 0) {
        fprintf(stderr, "Error: could not generate combinations.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weakness (CWE-119/CWE-120):

1. **Bounds checking in `print_combination`**: Every index is validated against `[0, NUM_COLOURS)` before being used to access the statically-sized `colours` array, preventing out-of-bounds reads.
2. **Bounds checking in `generate_combinations`**: The `depth` is checked with `>=` instead of `==` to guard against overshoot, and `start` is validated before use as a loop bound.
3. **NULL pointer guards**: Both helper functions verify `indices` is non-NULL before dereferencing.
4. **Integer overflow check**: `count * sizeof(int)` is validated against `SIZE_MAX` before the `malloc` call, preventing an undersized allocation that would lead to a heap buffer overflow.
