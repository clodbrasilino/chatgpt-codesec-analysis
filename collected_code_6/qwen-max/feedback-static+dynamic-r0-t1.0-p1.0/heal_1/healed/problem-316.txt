```c
#include <stdio.h>
#include <stdlib.h>

int find_last_occurrence(const int *array, int size, int target) {
    int index = -1;
    for (int i = 0; i < size; ++i) {
        if (array[i] == target) {
            index = i;
        }
    }
    return index;
}

int main() {
    int array[] = {1, 2, 4, 4, 6, 8, 8, 8, 10};
    int size = sizeof(array) / sizeof(array[0]);
    int target = 8;

    int index = find_last_occurrence(array, size, target);

    if (index != -1) {
        printf("Last occurrence of %d is at index %d\n", target, index);
    } else {
        printf("%d not found in the array\n", target);
    }

    return 0;
}
```
