#include <stdio.h>
#include <string.h>
#include <ctype.h>

void move_numbers(char *str) {
    char result[100] = "";
    char temp[100] = "";
    int j = 0, k = 0;

    for (int i = 0; str[i] != '\0'; i++) {
        if (isdigit(str[i])) {
            result[j++] = str[i];
        } else {
            temp[k++] = str[i];
        }
    }

    while (k > 0) {
        temp[k--] = '\0';
    }

    strcat(result, temp);
    strcpy(str, result);
}

int main() {
    char str[] = "a2b3c4d5e6";
    move_numbers(str);
    printf("%s\n", str);
    return 0;
}
