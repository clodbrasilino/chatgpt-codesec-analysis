```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <string.h>

long long star_number(unsigned int n) {
    if (n == 0) {
        return 0;
    }
    return 6LL * n * (n - 1) + 1;
}

int main(void) {
    char *buffer = NULL;
    size_t buffer_size = 0;
    ssize_t read_len;
    char *endptr;
    long val;
    unsigned int n;
    long long result;

    read_len = getline(&buffer, &buffer_size, stdin);
    if (read_len == -1) {
        fprintf(stderr, "Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (read_len > 0 && buffer[read_len - 1] == '\n') {
        buffer[read_len - 1] = '\0';
        read_len--;
    }

    if (read_len == 0) {
        fprintf(stderr, "Invalid input: not a valid integer\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    errno = 0;
    val = strtol(buffer, &endptr, 10);

    if (errno == ERANGE || val < 0 || val > UINT_MAX) {
        fprintf(stderr, "Invalid input: out of range\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (endptr == buffer || *endptr != '\0') {
        fprintf(stderr, "Invalid input: not a valid integer\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    n = (unsigned int)val;

    if (n > 0 && n > (ULLONG_MAX - 1) / 6 / (n - 1)) {
        fprintf(stderr, "Input too large: would cause overflow\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    result = star_number(n);
    printf("%lld\n", result);

    free(buffer);
    return EXIT_SUCCESS;
}
```
