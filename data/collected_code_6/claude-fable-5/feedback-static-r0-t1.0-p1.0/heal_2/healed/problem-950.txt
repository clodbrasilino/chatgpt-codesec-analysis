```c
#include <stdio.h>
#include <stdlib.h>

#define ZODIAC_COUNT 12
#define MIN_YEAR 1
#define MAX_YEAR 9999

static const char *get_chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        return NULL;
    }

    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return NULL;
    }
    return signs[index];
}

static int display_zodiac(int year)
{
    const char *sign;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        fprintf(stderr, "Error: year must be between %d and %d.\n",
                MIN_YEAR, MAX_YEAR);
        return -1;
    }
    sign = get_chinese_zodiac(year);
    if (sign == NULL) {
        fprintf(stderr, "Error: could not determine zodiac sign.\n");
        return -1;
    }
    if (printf("The Chinese zodiac sign for year %d is: %s\n", year, sign) < 0) {
        return -1;
    }
    return 0;
}

int main(void)
{
    int year;
    int result;

    printf("Enter a year: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }
    result = scanf("%d", &year);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input, expected an integer year.\n");
        return EXIT_FAILURE;
    }
    if (display_zodiac(year) != 0) {
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-defended against the flagged weakness. The Flawfinder warning (CWE-119/CWE-120) on the `signs` array is a heuristic hit on statically-sized arrays — it flags every `char` array declaration regardless of actual usage. In this case it is a false positive because:

1. **The array is read-only** (`static const char *const`) — no writes can occur, so buffer overflow via writing is impossible.
2. **Index computation is bounded**: `year % ZODIAC_COUNT` yields a value in `(-12, 12)`; the negative-adjustment branch normalizes it into `[0, 11]`.
3. **A redundant defensive bounds check** (`index < 0 || index >= ZODIAC_COUNT`) guards the access anyway, returning `NULL` rather than reading out of bounds.

The only change made in the rewrite is a hardening improvement: `get_chinese_zodiac()` now validates the year range itself rather than relying solely on the caller (`display_zodiac`) to enforce it. This makes the function safe even if reused elsewhere without prior validation — defense in depth. With `MIN_YEAR` at 1, the negative-modulo branch also becomes unreachable in practice, but it is retained as a safeguard against future changes to the range constants.

Other correctness points already handled well in the original: `scanf` return value checked, `printf`/`fflush` failures propagated, and `EXIT_SUCCESS`/`EXIT_FAILURE` used consistently.
