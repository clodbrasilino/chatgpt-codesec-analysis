#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

int* tuple_to_set(const int* tuple, size_t tuple_size, size_t* set_size) {
    if (!tuple || !set_size) {
        return NULL;
    }

    if (tuple_size == 0) {
        *set_size = 0;
        return NULL;
    }

    int* set = (int*)malloc(tuple_size * sizeof(int));
    if (!set) {
        return NULL;
    }

    size_t unique_count = 0;

    for (size_t i = 0; i < tuple_size; ++i) {
        bool is_duplicate = false;
        for (size_t j = 0; j < unique_count; ++j) {
            if (tuple[i] == set[j]) {
                is_duplicate = true;
                break;
            }
        }
        if (!is_duplicate) {
            set[unique_count++] = tuple[i];
        }
    }

    *set_size = unique_count;

    int* resized_set = (int*)realloc(set, unique_count * sizeof(int));
    if (!resized_set) {
        return set;
    }

    return resized_set;
}

int main(void) {
    int tuple[] = {1, 2, 2, 3, 4, 4, 5, 1, 6, 3};
    size_t tuple_size = sizeof(tuple) / sizeof(tuple[0]);
    size_t set_size = 0;

    int* set = tuple_to_set(tuple, tuple_size, &set_size);
    
    if (!set && tuple_size > 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < set_size; ++i) {
        printf("%d ", set[i]);
    }
    printf("\n");

    free(set);
    return EXIT_SUCCESS;
}
