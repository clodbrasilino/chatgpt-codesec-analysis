#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

size_t remove_words(char **list, size_t count, const char **words_to_remove, size_t remove_count);

size_t remove_words(char **list, size_t count, const char **words_to_remove, size_t remove_count)
{
    size_t write_index = 0;
    size_t i;
    size_t j;

    if (list == NULL || words_to_remove == NULL) {
        return count;
    }

    for (i = 0; i < count; i++) {
        int match = 0;
        if (list[i] != NULL) {
            for (j = 0; j < remove_count; j++) {
                if (words_to_remove[j] != NULL &&
                    strncmp(list[i], words_to_remove[j], MAX_STRING_LENGTH) == 0) {
                    match = 1;
                    break;
                }
            }
        }
        if (match) {
            free(list[i]);
            list[i] = NULL;
        } else {
            list[write_index] = list[i];
            if (write_index != i) {
                list[i] = NULL;
            }
            write_index++;
        }
    }

    return write_index;
}

char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LENGTH);
    if (len >= MAX_STRING_LENGTH) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *initial_words[] = { "apple", "banana", "cherry", "date", "banana", "fig" };
    const char *words_to_remove[] = { "banana", "date" };
    size_t initial_count = sizeof(initial_words) / sizeof(initial_words[0]);
    size_t remove_count = sizeof(words_to_remove) / sizeof(words_to_remove[0]);
    char **list;
    size_t i;
    size_t new_count;

    list = malloc(initial_count * sizeof(char *));
    if (list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < initial_count; i++) {
        list[i] = duplicate_string(initial_words[i]);
        if (list[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_list(list, i);
            return EXIT_FAILURE;
        }
    }

    printf("Original list:\n");
    for (i = 0; i < initial_count; i++) {
        printf("  %s\n", list[i]);
    }

    new_count = remove_words(list, initial_count, words_to_remove, remove_count);

    printf("List after removal:\n");
    for (i = 0; i < new_count; i++) {
        printf("  %s\n", list[i]);
    }

    free_list(list, new_count);

    return EXIT_SUCCESS;

**Analysis notes:**

The flagged `memcpy` weakness (CWE-120) was already properly mitigated in the original code, and the rewrite preserves those safeguards:

1. **Bounded length measurement**: `strnlen(src, MAX_STRING_LENGTH)` guarantees `len` can never exceed `MAX_STRING_LENGTH`, even if the source string is unterminated within that range.
2. **Explicit rejection of oversized input**: the `len >= MAX_STRING_LENGTH` check returns `NULL` rather than proceeding with a potentially truncated/unterminated string.
3. **Destination sized from measured length**: `malloc(len + 1)` allocates exactly enough for the copy plus the terminator, so `memcpy(copy, src, len)` can never overflow the destination — the destination size is derived from the same `len` used in the copy.
4. **Guaranteed termination**: `copy[len] = '\0'` explicitly null-terminates, avoiding reliance on `strcpy`/`strncpy` semantics.

Other defensive qualities worth noting: `strncmp` is bounded by `MAX_STRING_LENGTH`, all `malloc` results are checked, `free_list` handles partial initialization on failure, freed slots in `remove_words` are set to `NULL` to prevent double-free/dangling pointers, and the compaction correctly returns the new logical count so `free_list(list, new_count)` only frees live entries. No changes were required beyond removing the comments.
