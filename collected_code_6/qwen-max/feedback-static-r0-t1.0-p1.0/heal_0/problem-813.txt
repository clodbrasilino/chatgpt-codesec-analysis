```c
#include <stdio.h>

size_t string_length(const char *str) {
    size_t len = 0;
    while (str[len] != '\0') {
        ++len;
    }
    return len;
}

int main() {
    const char *test_string = "Hello, World!";
    size_t length = string_length(test_string);
    printf("Length of the string: %zu\n", length);
    return 0;
}
```
