#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_LEN 256

static size_t bounded_length(const char *str, size_t max_len)
{
    size_t i = 0;

    if (str == NULL) {
        return 0;
    }

    while (i < max_len && str[i] != '\0') {
        i++;
    }

    return i;
}

int contains_all_vowels(const char *str, size_t max_len)
{
    int found[5] = {0, 0, 0, 0, 0};
    size_t i;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = bounded_length(str, max_len);

    for (i = 0; i < len; i++) {
        int c = tolower((unsigned char)str[i]);
        switch (c) {
        case 'a':
            found[0] = 1;
            break;
        case 'e':
            found[1] = 1;
            break;
        case 'i':
            found[2] = 1;
            break;
        case 'o':
            found[3] = 1;
            break;
        case 'u':
            found[4] = 1;
            break;
        default:
            break;
        }
    }

    return found[0] && found[1] && found[2] && found[3] && found[4];
}

int main(void)
{
    char buffer[MAX_LEN];
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = bounded_length(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (contains_all_vowels(buffer, sizeof(buffer))) {
        printf("Accepted: the string contains all vowels\n");
    } else {
        printf("Rejected: the string does not contain all vowels\n");
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and the applied fixes:**

The flawfinder warning (CWE-119/CWE-120) flagged the statically-sized `char buffer[MAX_LEN]` array. The original code already mitigated this risk well; the rewrite preserves and strengthens those mitigations:

1. **Bounded input reading:** `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow at the read point — the primary defense against the flagged weakness.

2. **Guaranteed null termination:** `buffer[sizeof(buffer) - 1] = '\0'` explicitly ensures termination even in edge cases, so downstream string handling can never run off the end.

3. **Defensive length computation:** `bounded_length()` replaces `strlen()`, capping traversal at `max_len` so even a non-terminated buffer cannot cause an out-of-bounds read.

4. **Buffer zero-initialization (added):** `memset(buffer, 0, sizeof(buffer))` ensures the buffer contains no indeterminate bytes before use, hardening against any code path that might read it before it is fully written.

5. **NULL checks and safe character handling:** Both functions validate their pointer arguments, and `tolower((unsigned char)str[i])` avoids undefined behavior from passing negative `char` values to `ctype.h` functions.

6. **Error handling:** The `fgets` return value is checked, and failure results in a clean `EXIT_FAILURE` rather than operating on uninitialized data.

The code is safe with respect to the identified weakness: all buffer accesses are bounds-checked, input is length-limited, and termination is guaranteed.
