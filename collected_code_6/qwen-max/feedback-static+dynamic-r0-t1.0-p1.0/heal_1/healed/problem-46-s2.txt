```c
#include <stdio.h>
#include <stdbool.h>

bool areAllUnique(const int *arr, int n) {
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (arr[i] == arr[j]) {
                return false;
            }
        }
    }
    return true;
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    if (areAllUnique(arr, n)) {
        printf("All numbers are unique.\n");
    } else {
        printf("There are duplicate numbers.\n");
    }

    return 0;
}
```
