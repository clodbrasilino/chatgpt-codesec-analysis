#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <ctype.h>

#define DATE_LENGTH 11

static int safe_str_to_int(const char *str, int *value) {
    char *endptr;
    long result;
    
    errno = 0;
    result = strtol(str, &endptr, 10);
    
    if (errno == ERANGE || result > INT_MAX || result < INT_MIN) {
        return -1;
    }
    
    if (endptr == str || *endptr != '\0') {
        return -1;
    }
    
    *value = (int)result;
    return 0;
}

int convert_date_format(const char *input, char *output, size_t output_size) {
    int year, month, day;
    
    if (input == NULL || output == NULL) {
        return -1;
    }
    
    if (strnlen(input, DATE_LENGTH) != 10) {
        return -1;
    }
    
    if (input[4] != '-' || input[7] != '-') {
        return -1;
    }
    
    for (int i = 0; i < 10; i++) {
        if (i == 4 || i == 7) {
            continue;
        }
        if (!isdigit((unsigned char)input[i])) {
            return -1;
        }
    }
    
    if (output_size < DATE_LENGTH) {
        return -1;
    }
    
    char year_str[5];
    char month_str[3];
    char day_str[3];
    
    memcpy(year_str, input, 4);
    year_str[4] = '\0';
    
    memcpy(month_str, input + 5, 2);
    month_str[2] = '\0';
    
    memcpy(day_str, input + 8, 2);
    day_str[2] = '\0';
    
    if (safe_str_to_int(year_str, &year) != 0) {
        return -1;
    }
    
    if (safe_str_to_int(month_str, &month) != 0) {
        return -1;
    }
    
    if (safe_str_to_int(day_str, &day) != 0) {
        return -1;
    }
    
    if (month < 1 || month > 12) {
        return -1;
    }
    
    if (day < 1 || day > 31) {
        return -1;
    }
    
    if (snprintf(output, output_size, "%02d-%02d-%04d", day, month, year) >= (int)output_size) {
        return -1;
    }
    
    return 0;
}

int main(void) {
    char input_date[DATE_LENGTH];
    char output_date[DATE_LENGTH];
    
    printf("Enter date (yyyy-mm-dd): ");
    if (fgets(input_date, sizeof(input_date), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }
    
    size_t len = strnlen(input_date, sizeof(input_date));
    if (len > 0 && input_date[len - 1] == '\n') {
        input_date[len - 1] = '\0';
    } else {
        int c;
        while ((c = getchar()) != '\n' && c != EOF);
    }
    
    if (convert_date_format(input_date, output_date, sizeof(output_date)) != 0) {
        fprintf(stderr, "Invalid date format\n");
        return 1;
    }
    
    printf("Converted date: %s\n", output_date);
    
    return 0;
}
