#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>
#include <limits.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    Node *tail;
    size_t size;
} List;

List *list_create(void);
void list_destroy(List *list);
int list_append(List *list, int value);
List *subtract_lists(const List *a, const List *b);
void list_print(const List *list);

List *list_create(void) {
    List *list = (List *)malloc(sizeof(*list));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    return list;
}

void list_destroy(List *list) {
    Node *current;
    if (list == NULL) {
        return;
    }
    current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    free(list);
}

int list_append(List *list, int value) {
    Node *node;
    if (list == NULL) {
        return 0;
    }
    node = (Node *)malloc(sizeof(*node));
    if (node == NULL) {
        return 0;
    }
    node->value = value;
    node->next = NULL;
    if (list->tail == NULL) {
        list->head = node;
        list->tail = node;
    } else {
        list->tail->next = node;
        list->tail = node;
    }
    list->size++;
    return 1;
}

List *subtract_lists(const List *a, const List *b) {
    List *result;
    Node *pa;
    Node *pb;
    if (a == NULL || b == NULL || a->size != b->size) {
        return NULL;
    }
    result = list_create();
    if (result == NULL) {
        return NULL;
    }
    pa = a->head;
    pb = b->head;
    while (pa != NULL && pb != NULL) {
        long long diff = (long long)pa->value - (long long)pb->value;
        if (diff > INT_MAX || diff < INT_MIN) {
            list_destroy(result);
            return NULL;
        }
        if (!list_append(result, (int)diff)) {
            list_destroy(result);
            return NULL;
        }
        pa = pa->next;
        pb = pb->next;
    }
    return result;
}

void list_print(const List *list) {
    Node *current;
    if (list == NULL) {
        return;
    }
    current = list->head;
    while (current != NULL) {
        printf("%d ", current->value);
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    List *a = list_create();
    List *b = list_create();
    List *result;
    if (a == NULL || b == NULL) {
        list_destroy(a);
        list_destroy(b);
        return EXIT_FAILURE;
    }
    if (!list_append(a, 10) || !list_append(a, 20) || !list_append(a, 30)) {
        list_destroy(a);
        list_destroy(b);
        return EXIT_FAILURE;
    }
    if (!list_append(b, 3) || !list_append(b, 5) || !list_append(b, 7)) {
        list_destroy(a);
        list_destroy(b);
        return EXIT_FAILURE;
    }
    result = subtract_lists(a, b);
    if (result == NULL) {
        list_destroy(a);
        list_destroy(b);
        return EXIT_FAILURE;
    }
    list_print(result);
    list_destroy(result);
    list_destroy(a);
    list_destroy(b);
    return EXIT_SUCCESS;
}
