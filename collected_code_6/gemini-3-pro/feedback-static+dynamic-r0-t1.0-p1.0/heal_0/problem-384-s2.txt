#include <stdio.h>
#include <stdlib.h>

int find_frequency_of_smallest(const int *array, size_t size) {
    if (array == NULL || size == 0) {
        return 0;
    }

    int smallest_value = array[0];
    int frequency = 1;

    for (size_t i = 1; i < size; ++i) {
        if (array[i] < smallest_value) {
            smallest_value = array[i];
            frequency = 1;
        } else if (array[i] == smallest_value) {
            frequency++;
        }
    }

    return frequency;
}

int main(void) {
    int array[] = {5, 2, 8, 2, 2, 9, 1, 1, 4, 1};
    size_t size = sizeof(array) / sizeof(array[0]);

    int freq = find_frequency_of_smallest(array, size);

    printf("Frequency of the smallest value: %d\n", freq);

    return 0;
}
