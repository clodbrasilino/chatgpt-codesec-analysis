```c
#include <stdio.h>
#include <stdlib.h>

long long multiply_consecutive(int *arr, int size) {
    if (size <= 0) return 1;
    long long result = 1;
    for (int i = 0; i < size; ++i) {
        result *= arr[i];
    }
    return result;
}

int main() {
    int list[] = {1, 2, 3, 4, 5};
    int size = sizeof(list) / sizeof(list[0]);
    long long result = multiply_consecutive(list, size);
    printf("%lld\n", result);
    return 0;
}
```
