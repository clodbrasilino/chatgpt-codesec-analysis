#include <stdio.h>
#include <stdlib.h>

int countSubsequences(int *arr, int n, int k) {
    int count = 0;
    for (int i = 1; i < (1 << n); i++) {
        int product = 1;
        for (int j = 0; j < n; j++) {
            if (i & (1 << j)) {
                product *= arr[j];
                if (product >= k) break;
            }
        }
        if (product < k) count++;
    }
    return count;
}

int main() {
    int arr[] = {1, 2, 3, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 10;
    int result = countSubsequences(arr, n, k);
    printf("%d\n", result);
    return 0;
}
