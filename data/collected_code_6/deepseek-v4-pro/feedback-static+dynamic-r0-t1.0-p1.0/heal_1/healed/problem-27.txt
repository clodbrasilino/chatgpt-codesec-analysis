#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void remove_digits_from_string(char *str) {
    if (str == NULL) {
        return;
    }
    
    char *src = str;
    char *dst = str;
    
    while (*src) {
        if (!isdigit((unsigned char)*src)) {
            *dst++ = *src;
        }
        src++;
    }
    *dst = '\0';
}

void remove_digits_from_list(char **list, size_t count) {
    if (list == NULL) {
        return;
    }
    
    for (size_t i = 0; i < count; i++) {
        if (list[i] != NULL) {
            remove_digits_from_string(list[i]);
        }
    }
}

int main(void) {
    char buffer1[] = "abc123def456";
    char buffer2[] = "hello42world";
    char buffer3[] = "no_digits_here";
    char buffer4[] = "123456";
    char buffer5[] = "";
    char buffer6[] = "a1b2c3d4e5f6";
    
    char *strings[] = {
        buffer1,
        buffer2,
        buffer3,
        buffer4,
        buffer5,
        buffer6
    };
    size_t count = sizeof(strings) / sizeof(strings[0]);
    
    remove_digits_from_list(strings, count);
    
    for (size_t i = 0; i < count; i++) {
        printf("%s\n", strings[i]);
    }
    
    return 0;
}
