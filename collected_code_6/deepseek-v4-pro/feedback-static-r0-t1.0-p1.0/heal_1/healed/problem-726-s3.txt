#include <stdio.h>
#include <stdlib.h>

int* multiplyAdjacent(const int* tuple, int size, int* resultSize) {
    if (resultSize == NULL) {
        return NULL;
    }

    if (tuple == NULL || size <= 1) {
        *resultSize = 0;
        return NULL;
    }

    *resultSize = size - 1;
    int* result = (int*)malloc(sizeof(int) * (*resultSize));
    if (result == NULL) {
        *resultSize = 0;
        return NULL;
    }

    for (int i = 0; i < *resultSize; i++) {
        result[i] = tuple[i] * tuple[i + 1];
    }

    return result;
}

int main(void) {
    int tuple[] = {2, 3, 4, 5, 6};
    int size = sizeof(tuple) / sizeof(tuple[0]);
    int resultSize = 0;

    int* result = multiplyAdjacent(tuple, size, &resultSize);

    if (result == NULL) {
        printf("Error: unable to compute result\n");
        return 1;
    }

    for (int i = 0; i < resultSize; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);
    return 0;
}
