```c
#include <stdio.h>
#include <stdlib.h>

void print_positive_numbers(const int *list, size_t size) {
    if (list == NULL) {
        return;
    }

    for (size_t i = 0; i < size; i++) {
        if (list[i] > 0) {
            printf("%d\n", list[i]);
        }
    }
}

int main(void) {
    int list[] = {-3, 5, 0, 12, -8, 7, -1, 0, 4};
    size_t size = sizeof(list) / sizeof(list[0]);

    print_positive_numbers(list, size);

    return 0;
}
```
