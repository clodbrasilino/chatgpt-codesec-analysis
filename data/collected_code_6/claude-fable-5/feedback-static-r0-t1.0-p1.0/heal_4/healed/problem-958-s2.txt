```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

typedef struct {
    int value;
    const char *symbol;
    size_t length;
} roman_entry;

int int_to_roman(int number, char *output, size_t output_size)
{
    static const roman_entry table[] = {
        { 1000, "M",  1 },
        {  900, "CM", 2 },
        {  500, "D",  1 },
        {  400, "CD", 2 },
        {  100, "C",  1 },
        {   90, "XC", 2 },
        {   50, "L",  1 },
        {   40, "XL", 2 },
        {   10, "X",  1 },
        {    9, "IX", 2 },
        {    5, "V",  1 },
        {    4, "IV", 2 },
        {    1, "I",  1 }
    };
    size_t used = 0;
    size_t i;

    if (output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(table) / sizeof(table[0]); i++) {
        while (number >= table[i].value) {
            size_t symbol_len = table[i].length;

            if (output_size <= symbol_len ||
                used > output_size - symbol_len - 1) {
                output[0] = '\0';
                return -1;
            }

            memcpy(output + used, table[i].symbol, symbol_len);
            used += symbol_len;
            output[used] = '\0';
            number -= table[i].value;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], buffer, sizeof(buffer)) == 0) {
            if (printf("%d -> %s\n", test_values[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 0\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 4000\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

1. **`memcpy` (CWE-120):** The Flawfinder warning is a heuristic — it flags every `memcpy` call regardless of context. In this code the copy is provably safe because it is guarded by an explicit bounds check immediately before it:
   - `output_size <= symbol_len` guarantees the buffer can hold at least the symbol plus a terminator.
   - `used > output_size - symbol_len - 1` guarantees `used + symbol_len + 1 <= output_size`, so both the copied bytes and the trailing `'\0'` written afterward always fit. The subtraction cannot underflow because the first condition already ensures `output_size > symbol_len`, i.e. `output_size - symbol_len - 1 >= 0`.

2. **Statically-sized `char` array (CWE-119/CWE-120):** Also a heuristic hit. The buffer's size is always passed alongside the pointer via `sizeof(buffer)`, and `int_to_roman` never writes past `output_size`. Additionally, `ROMAN_BUFFER_SIZE` (32) comfortably exceeds the longest possible Roman numeral in the supported range — 3888 → `"MMMDCCCLXXXVIII"` at 15 characters plus terminator.

**Other correctness properties worth noting:**

- Input range is validated (`ROMAN_MIN`–`ROMAN_MAX`), rejecting values Roman numerals cannot represent.
- `NULL` pointer and zero-size buffer are rejected before any write.
- The output is always NUL-terminated, including on every error path (`output[0] = '\0'`), so callers never read stale or uninitialized data.
- Return values of `printf`/`fprintf` are checked, satisfying strict CERT/MISRA-style guidelines.

No functional changes were required — the original code already contained the mitigations; the warnings were false positives from pattern-based static analysis.
