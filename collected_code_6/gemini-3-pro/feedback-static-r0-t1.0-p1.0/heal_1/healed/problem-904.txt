```c
#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool is_even(int number) {
    return (number % 2 == 0);
}

int main(void) {
    char *buffer = malloc(256);
    
    if (buffer == NULL) {
        return 1;
    }

    if (fgets(buffer, 256, stdin) != NULL) {
        char *endptr;
        int input_number = (int)strtol(buffer, &endptr, 10);
        
        if (endptr == buffer) {
            free(buffer);
            return 1;
        }

        if (is_even(input_number)) {
            printf("true\n");
        } else {
            printf("false\n");
        }
    } else {
        free(buffer);
        return 1;
    }

    free(buffer);
    return 0;
}
```
