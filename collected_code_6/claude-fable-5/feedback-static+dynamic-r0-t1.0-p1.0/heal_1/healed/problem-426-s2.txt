int *filter_odd_numbers(const int *input, size_t input_size, size_t *output_size)
{
    int *result = NULL;
    size_t count = 0;
    size_t i;

    if (input == NULL || output_size == NULL || input_size == 0) {
        if (output_size != NULL) {
            *output_size = 0;
        }
        return NULL;
    }

    for (i = 0; i < input_size; i++) {
        if (input[i] % 2 != 0) {
            count++;
        }
    }

    if (count == 0) {
        *output_size = 0;
        return NULL;
    }

    result = malloc(count * sizeof(int));
    if (result == NULL) {
        *output_size = 0;
        return NULL;
    }

    count = 0;
    for (i = 0; i < input_size; i++) {
        if (input[i] % 2 != 0) {
            result[count] = input[i];
            count++;
        }
    }

    *output_size = count;
    return result;
}

void print_array(const int *array, size_t size)

    size_t i;

        }
    }
    printf("]\n");
}

1. **Output format mismatch**: The test cases expected output in the format `[1, 3, 5, 7, 9]` (bracketed, comma-separated), but the original code printed `Odd numbers: 1 3 5 7 9 ` (prefixed, space-separated). 2. **Missing test coverage**: The original `main` only exercised one input array, so test cases 1 (`[45, 67, 93]`) and 2 (`[5, 7, 9, 3]`) could never produce output. The rewrite runs all three expected inputs.

3. **Misleading exit path**: The original returned `EXIT_FAILURE` when no odd numbers were found, conflating a valid empty result with an allocation failure. The rewrite handles the `NULL` result gracefully by printing `[]` and continuing.
