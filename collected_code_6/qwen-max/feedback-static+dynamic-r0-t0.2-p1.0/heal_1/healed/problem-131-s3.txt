#include <stdio.h>
#include <string.h>
#include <ctype.h>

int is_vowel(char c) {
    c = tolower(c);
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u';
}

void reverse_vowels(char *str, size_t len) {
    int left = 0, right = len - 1;
    char temp;

    while (left < right) {
        if (!is_vowel(str[left])) {
            left++;
        } else if (!is_vowel(str[right])) {
            right--;
        } else {
            temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            right--;
        }
    }
}

int main() {
    char str[] = "hello world";
    reverse_vowels(str, sizeof(str) / sizeof(str[0]) - 1);
    printf("%s\n", str);
    return 0;
}
