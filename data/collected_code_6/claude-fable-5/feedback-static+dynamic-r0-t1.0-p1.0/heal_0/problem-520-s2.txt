#include <stdio.h>
#include <stdlib.h>

static unsigned long long gcd_ull(unsigned long long a, unsigned long long b)
{
    while (b != 0ULL) {
        unsigned long long t = a % b;
        a = b;
        b = t;
    }
    return a;
}

static int lcm_ull(unsigned long long a, unsigned long long b, unsigned long long *result)
{
    unsigned long long g;
    unsigned long long q;

    if (result == NULL) {
        return -1;
    }
    if (a == 0ULL || b == 0ULL) {
        *result = 0ULL;
        return 0;
    }
    g = gcd_ull(a, b);
    q = a / g;
    if (q > (~0ULL) / b) {
        return -1;
    }
    *result = q * b;
    return 0;
}

static int lcm_array(const unsigned long long *arr, size_t n, unsigned long long *result)
{
    size_t i;
    unsigned long long acc;

    if (arr == NULL || result == NULL || n == 0U) {
        return -1;
    }
    acc = arr[0];
    for (i = 1U; i < n; i++) {
        if (lcm_ull(acc, arr[i], &acc) != 0) {
            return -1;
        }
    }
    *result = acc;
    return 0;
}

int main(void)
{
    unsigned long long numbers[] = { 4ULL, 6ULL, 8ULL, 10ULL };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    unsigned long long lcm_value = 0ULL;

    if (lcm_array(numbers, count, &lcm_value) != 0) {
        if (fprintf(stderr, "Error: could not compute LCM (invalid input or overflow)\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("LCM = %llu\n", lcm_value) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
