#include <stdio.h>
#include <string.h>
#include <limits.h>

int remove_chars(char *str, const char *mask)
{
    int present[UCHAR_MAX + 1] = {0};
    size_t i;
    size_t j;

    if (str == NULL || mask == NULL) {
        return -1;
    }

    for (i = 0; mask[i] != '\0'; i++) {
        present[(unsigned char)mask[i]] = 1;
    }

    j = 0;
    for (i = 0; str[i] != '\0'; i++) {
        if (!present[(unsigned char)str[i]]) {
            str[j] = str[i];
            j++;
        }
    }
    str[j] = '\0';

    return 0;
}

int main(void)
{
    char first[256];
    char second[256];

    printf("Enter first string: ");
    if (fgets(first, sizeof(first), stdin) == NULL) {
        fprintf(stderr, "Error reading first string\n");
        return 1;
    }
    first[strcspn(first, "\n")] = '\0';

    printf("Enter second string: ");
    if (fgets(second, sizeof(second), stdin) == NULL) {
        fprintf(stderr, "Error reading second string\n");
        return 1;
    }
    second[strcspn(second, "\n")] = '\0';

    if (remove_chars(first, second) != 0) {
        fprintf(stderr, "Error processing strings\n");
        return 1;
    }

    printf("Result: %s\n", first);

    return 0;
}

1. **Bounded input**: `fgets(first, sizeof(first), stdin)` limits reads to the buffer size, writing at most 255 characters plus the null terminator. There is no possibility of overflow, unlike `gets()` or unbounded `scanf("%s", ...)`.

2. **Guaranteed null termination**: `fgets` always null-terminates on success, and both `fgets` calls are checked for `NULL` return, handling EOF and read errors.

3. **Safe newline stripping**: `first[strcspn(first, "\n")] = '\0'` is the idiomatic safe pattern — `strcspn` returns an index within the string bounds whether or not a newline is present, so the write is always in range.

4. **`remove_chars` safety**: The function validates both pointers against `NULL`, indexes the lookup table with `(unsigned char)` casts to avoid negative-index undefined behavior on platforms where `char` is signed, and only compacts the string in place (`j <= i` always), so it can never write past the original string length.
