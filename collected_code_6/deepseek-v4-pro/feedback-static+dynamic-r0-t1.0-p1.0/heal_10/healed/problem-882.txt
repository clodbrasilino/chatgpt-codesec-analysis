```c
#include <stdio.h>

double parallelogram_perimeter(double base, double side) {
    return 2.0 * base + 2.0 * side;
}

int main(void) {
    double base, side, perimeter;

    base = 10.0;
    side = 10.0;
    perimeter = parallelogram_perimeter(base, side);
    printf("%.2f\n", perimeter);

    base = 20.0;
    side = 10.0;
    perimeter = parallelogram_perimeter(base, side);
    printf("%.2f\n", perimeter);

    base = 30.0;
    side = 15.0;
    perimeter = parallelogram_perimeter(base, side);
    printf("%.2f\n", perimeter);

    return 0;
}
```
