#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LENGTH 4096

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0;
    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

char *camel_to_snake(const char *input, size_t max_input_len)
{
    if (input == NULL) {
        return NULL;
    }

    size_t len = bounded_strlen(input, max_input_len);

    if (len > (SIZE_MAX - 1) / 2) {
        return NULL;
    }

    size_t max_len = (len * 2) + 1;

    char *output = malloc(max_len);
    if (output == NULL) {
        return NULL;
    }

    size_t out_idx = 0;
    for (size_t i = 0; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (isupper(c)) {
            if (i > 0) {
                output[out_idx++] = '_';
            }
            output[out_idx++] = (char)tolower(c);
        } else {
            output[out_idx++] = (char)c;
        }
    }
    output[out_idx] = '\0';

    return output;
}

int main(void)
{
    const char *tests[] = {
        "helloWorld",
        "camelCaseString",
        "myVeryLongVariableName",
        "simple",
        "A",
        "already_snake"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = camel_to_snake(tests[i], MAX_INPUT_LENGTH);
        if (result == NULL) {
            fprintf(stderr, "Error: conversion failed for input \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("Input:  %s\n", tests[i]);
        printf("Output: %s\n\n", result);
        free(result);
        result = NULL;
    }

    return EXIT_SUCCESS;
}
