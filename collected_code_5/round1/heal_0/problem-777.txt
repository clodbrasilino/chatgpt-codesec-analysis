```c
#include <stdio.h>
#include <stdlib.h>

#define SIZE 1000

int sumNonRepeated(int arr[], int size) {
    int sum = 0;
    int i, j;

    for(i = 0; i < size; i++) {
        for(j = 0; j < size; j++) {
            if(arr[i] == arr[j] && i != j)
                break;
        }
        if(j == size)
            sum += arr[i];
    }

    return sum;
}

int main() {

    int arr[SIZE];
    int i;

    for(i = 0; i < SIZE; i++) {
        arr[i] = rand() % 100;    
    }

    int sum = sumNonRepeated(arr, SIZE);
    printf("Sum of non repeated elements in array: %d\n", sum);

    return 0;
}
```
