#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int key;
    int value;
} KeyValuePair;

typedef struct {
    KeyValuePair* pairs;
    size_t size;
} Dictionary;

Dictionary* tuple_to_dict(const int* tuple, size_t tuple_size) {
    if (tuple == NULL || tuple_size == 0 || tuple_size % 2 != 0) {
        return NULL;
    }

    Dictionary* dict = (Dictionary*)malloc(sizeof(Dictionary));
    if (dict == NULL) {
        return NULL;
    }

    dict->size = tuple_size / 2;
    dict->pairs = (KeyValuePair*)malloc(dict->size * sizeof(KeyValuePair));
    if (dict->pairs == NULL) {
        free(dict);
        return NULL;
    }

    for (size_t i = 0; i < dict->size; i++) {
        dict->pairs[i].key = tuple[i * 2];
        dict->pairs[i].value = tuple[i * 2 + 1];
    }

    return dict;
}

void free_dict(Dictionary* dict) {
    if (dict != NULL) {
        free(dict->pairs);
        free(dict);
    }
}

int main(void) {
    int tuple[] = {1, 10, 2, 20, 3, 30};
    size_t tuple_size = sizeof(tuple) / sizeof(tuple[0]);

    Dictionary* dict = tuple_to_dict(tuple, tuple_size);

    if (dict != NULL) {
        for (size_t i = 0; i < dict->size; i++) {
            printf("Key: %d, Value: %d\n", dict->pairs[i].key, dict->pairs[i].value);
        }
        free_dict(dict);
    } else {
        printf("Failed to create dictionary.\n");
    }

    return 0;
}
