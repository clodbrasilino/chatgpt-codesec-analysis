```c
#include <stdio.h>
#include <stdbool.h>

#define SIZE 5

bool check_elements(const int first[], const int second[], int size) {
    for (int i = 0; i < size; i++) {
        if (second[i] >= first[i]) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int first[SIZE] = {5, 8, 3, 7, 9};
    int second[SIZE] = {2, 5, 1, 4, 6};
    
    if (check_elements(first, second, SIZE)) {
        printf("All elements in second tuple are smaller than corresponding elements in first tuple.\n");
    } else {
        printf("Not all elements in second tuple are smaller than corresponding elements in first tuple.\n");
    }
    
    return 0;
}
```
