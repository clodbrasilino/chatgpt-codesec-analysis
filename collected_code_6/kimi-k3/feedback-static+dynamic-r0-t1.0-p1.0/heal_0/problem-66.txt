```c
#include <stdio.h>
#include <stddef.h>

size_t count_positive(const int *arr, size_t size)
{
    size_t count = 0;
    size_t i;

    if (arr == NULL) {
        return 0;
    }

    for (i = 0; i < size; i++) {
        if (arr[i] > 0) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    int numbers[] = {-5, 3, -2, 8, 0, -1, 7, 4, -9, 12};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);
    size_t result;

    result = count_positive(numbers, size);

    printf("Number of positive numbers: %zu\n", result);

    return 0;
}
```
