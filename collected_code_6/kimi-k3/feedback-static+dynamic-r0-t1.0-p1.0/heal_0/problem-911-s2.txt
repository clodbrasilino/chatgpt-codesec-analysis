#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} MinHeap;

static void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

static void heapify_down(MinHeap *heap, size_t idx) {
    size_t smallest = idx;
    size_t left = 2 * idx + 1;
    size_t right = 2 * idx + 2;

    if (left < heap->size && heap->data[left] < heap->data[smallest]) {
        smallest = left;
    }
    if (right < heap->size && heap->data[right] < heap->data[smallest]) {
        smallest = right;
    }
    if (smallest != idx) {
        swap(&heap->data[idx], &heap->data[smallest]);
        heapify_down(heap, smallest);
    }
}

static void heapify_up(MinHeap *heap, size_t idx) {
    while (idx > 0) {
        size_t parent = (idx - 1) / 2;
        if (heap->data[parent] <= heap->data[idx]) {
            break;
        }
        swap(&heap->data[parent], &heap->data[idx]);
        idx = parent;
    }
}

static int heap_push(MinHeap *heap, int value) {
    if (heap->size >= heap->capacity) {
        return -1;
    }
    heap->data[heap->size] = value;
    heapify_up(heap, heap->size);
    heap->size++;
    return 0;
}

static int heap_pop(MinHeap *heap) {
    int result = heap->data[0];
    heap->size--;
    heap->data[0] = heap->data[heap->size];
    heapify_down(heap, 0);
    return result;
}

static int heap_peek(const MinHeap *heap) {
    return heap->data[0];
}

static int compare_ints(const void *a, const void *b) {
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    if (ia < ib) return -1;
    if (ia > ib) return 1;
    return 0;
}

int maximum_product_of_three(const int *arr, size_t n, long long *result) {
    if (arr == NULL || result == NULL || n < 3) {
        return -1;
    }

    MinHeap heap;
    heap.capacity = 3;
    heap.size = 0;
    heap.data = (int *)malloc(heap.capacity * sizeof(int));
    if (heap.data == NULL) {
        return -1;
    }

    int min1 = INT_MAX, min2 = INT_MAX;

    for (size_t i = 0; i < n; i++) {
        if (heap.size < 3) {
            if (heap_push(&heap, arr[i]) != 0) {
                free(heap.data);
                return -1;
            }
        } else if (arr[i] > heap_peek(&heap)) {
            heap_pop(&heap);
            if (heap_push(&heap, arr[i]) != 0) {
                free(heap.data);
                return -1;
            }
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    int top3[3];
    for (int i = 2; i >= 0; i--) {
        top3[i] = heap_pop(&heap);
    }

    free(heap.data);
    heap.data = NULL;

    long long product1 = (long long)top3[0] * top3[1] * top3[2];
    long long product2 = (long long)min1 * min2 * top3[2];

    *result = (product1 > product2) ? product1 : product2;
    return 0;
}

int main(void) {
    int arr[] = {-10, -10, 5, 2, 3, -20};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result;

    if (maximum_product_of_three(arr, n, &result) == 0) {
        printf("Maximum product of three numbers: %lld\n", result);
    } else {
        fprintf(stderr, "Error: invalid input or allocation failure\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
