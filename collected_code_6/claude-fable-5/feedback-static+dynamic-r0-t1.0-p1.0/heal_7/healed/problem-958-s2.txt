```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *output, size_t output_size);

int int_to_roman(int number, char *output, size_t output_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t symbol_len = symbol_lengths[i];
            size_t j;

            if (used >= output_size || symbol_len >= output_size - used) {
                output[0] = '\0';
                return -1;
            }

            for (j = 0; j < symbol_len; j++) {
                output[used + j] = symbols[i][j];
            }
            used += symbol_len;

            if (used >= output_size) {
                output[0] = '\0';
                return -1;
            }

            output[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char buffer[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], buffer, sizeof(buffer)) == 0) {
            if (printf("%d -> %s\n", test_values[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 0\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 4000\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness:**

The Flawfinder warning on `char buffer[ROMAN_BUFFER_SIZE]` is a low-risk, informational hit (risk 2). Statically-sized buffers are only dangerous when writes to them are unbounded. In this code, the weakness is already fully mitigated:

1. **Size is always passed explicitly** — every call to `int_to_roman` uses `sizeof(buffer)`, so the callee knows the true capacity rather than assuming it.
2. **Bounds are checked before every write** — the check `used >= output_size || symbol_len >= output_size - used` is written to avoid integer underflow/overflow (it never computes `used + symbol_len`, which could wrap), and guarantees room for both the symbol and the NUL terminator.
3. **NULL and zero-size guards** — `output == NULL || output_size == 0` prevents dereferencing invalid pointers or writing `output[0]` into a zero-length buffer.
4. **Fail-safe output state** — on any error path the buffer is reset to an empty string, so callers never read stale or partially-written data.
5. **Input range validation** — `ROMAN_MIN`/`ROMAN_MAX` bounds ensure the maximum possible output ("MMMDCCCLXXXVIII", 15 chars + NUL) fits comfortably in the 32-byte buffer.
6. **Buffer is zero-initialized** with `memset` before first use.

No functional changes were required; the code as structured already satisfies CWE-119/CWE-120 mitigations, so the static-analysis comment could simply be removed.
