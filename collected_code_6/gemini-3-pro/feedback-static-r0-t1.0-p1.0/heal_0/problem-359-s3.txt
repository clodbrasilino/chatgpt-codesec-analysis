#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool check_root_is_twice(long long a, long long b, long long c);

bool check_root_is_twice(long long a, long long b, long long c) {
    if (a == 0) {
        return false;
    }
    return (2 * b * b) == (9 * a * c);
}

int main(void) {
    long long a = 0;
    long long b = 0;
    long long c = 0;

    if (scanf("%lld %lld %lld", &a, &b, &c) != 3) {
        fprintf(stderr, "Invalid input format.\n");
        return EXIT_FAILURE;
    }

    if (check_root_is_twice(a, b, c)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }

    return EXIT_SUCCESS;
}
