```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char* remove_odd_chars(const char *input) {
    int len = strlen(input);
    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }
    int result_index = 0;
    for (int i = 0; input[i] != '\0'; i++) {
        if ((input[i] - '0') % 2 == 0) {
            result[result_index++] = input[i];
        }
    }
    result[result_index] = '\0';
    return result;
}

int main() {
    const char *input = "1234567890";
    char *output = remove_odd_chars(input);
    if (output != NULL) {
        printf("%s\n", output);
        free(output);
    } else {
        fprintf(stderr, "Memory allocation failed\n");
    }
    return 0;
}
```
