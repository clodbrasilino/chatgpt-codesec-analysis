```c
#include <stdio.h>
#include <ctype.h>
#include <stddef.h>
#include <string.h>

int count_digits_letters(const char *str, size_t *digits, size_t *letters)
{
    size_t i;

    if (str == NULL || digits == NULL || letters == NULL) {
        return -1;
    }

    *digits = 0;
    *letters = 0;

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isdigit(c)) {
            (*digits)++;
        } else if (isalpha(c)) {
            (*letters)++;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[256];
    size_t digits = 0;
    size_t letters = 0;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (count_digits_letters(buffer, &digits, &letters) != 0) {
        fprintf(stderr, "Error counting characters\n");
        return 1;
    }

    printf("Digits: %zu\n", digits);
    printf("Letters: %zu\n", letters);

    return 0;
}
```

**Analysis of the identified weakness (CWE-119/CWE-120, statically-sized buffer):**

The Flawfinder hit on `char buffer[256]` is a **false positive** in this code, and the original source already contains all the necessary mitigations, so the rewritten code preserves them as-is:

1. **Bounded read:** `fgets(buffer, sizeof(buffer), stdin)` limits input to `sizeof(buffer) - 1` characters plus the terminator, so no overflow is possible. Using `sizeof(buffer)` rather than a hardcoded `256` keeps the bound correct if the array size ever changes.

2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0';` is defensive redundancy — `fgets` already NUL-terminates on success, but this guards against future refactoring to a non-terminating read function.

3. **Bounded length computation:** `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so even if termination were somehow lost, the scan cannot run past the buffer.

4. **Safe newline stripping:** the `len > 0` check prevents an out-of-bounds access at `buffer[len - 1]` when the input is an empty line.

5. **Robust counting function:** `count_digits_letters` validates all pointers, and the cast to `unsigned char` before calling `isdigit`/`isalpha` avoids undefined behavior for characters with negative `char` values (e.g., high-bit characters on platforms where `char` is signed).

No functional changes were required — the code as written is already the correct fix for the flagged weakness.
