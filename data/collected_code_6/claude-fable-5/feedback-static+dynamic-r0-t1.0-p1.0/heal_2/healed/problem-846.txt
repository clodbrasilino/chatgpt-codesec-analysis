```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y)
    {
        return -1;
    }
    if (x > y)
    {
        return 1;
    }
    return 0;
}

static int copy_int_array(int *dest, const int *src, size_t n)
{
    size_t i = 0;

    if (dest == NULL || src == NULL)
    {
        return -1;
    }

    for (i = 0; i < n; i++)
    {
        dest[i] = src[i];
    }

    return 0;
}

int find_minimum_platforms(const int *arrivals, const int *departures, size_t n, int *result)
{
    int *arr_sorted = NULL;
    int *dep_sorted = NULL;
    size_t i = 0;
    size_t j = 0;
    size_t bytes = 0;
    int platforms_needed = 0;
    int max_platforms = 0;

    if (arrivals == NULL || departures == NULL || result == NULL || n == 0)
    {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int))
    {
        return -1;
    }

    bytes = n * sizeof(int);

    arr_sorted = malloc(bytes);
    if (arr_sorted == NULL)
    {
        return -1;
    }

    dep_sorted = malloc(bytes);
    if (dep_sorted == NULL)
    {
        free(arr_sorted);
        return -1;
    }

    if (copy_int_array(arr_sorted, arrivals, n) != 0)
    {
        free(arr_sorted);
        free(dep_sorted);
        return -1;
    }

    if (copy_int_array(dep_sorted, departures, n) != 0)
    {
        free(arr_sorted);
        free(dep_sorted);
        return -1;
    }

    qsort(arr_sorted, n, sizeof(int), compare_ints);
    qsort(dep_sorted, n, sizeof(int), compare_ints);

    while (i < n && j < n)
    {
        if (arr_sorted[i] <= dep_sorted[j])
        {
            platforms_needed++;
            if (platforms_needed > max_platforms)
            {
                max_platforms = platforms_needed;
            }
            i++;
        }
        else
        {
            platforms_needed--;
            j++;
        }
    }

    free(arr_sorted);
    free(dep_sorted);

    *result = max_platforms;
    return 0;
}

int main(void)
{
    static const int arrivals[] = {900, 940, 950, 1100, 1500, 1800};
    static const int departures[] = {910, 1200, 1120, 1130, 1900, 2000};
    size_t n = sizeof(arrivals) / sizeof(arrivals[0]);
    int min_platforms = 0;

    if (find_minimum_platforms(arrivals, departures, n, &min_platforms) != 0)
    {
        fprintf(stderr, "Error: failed to compute minimum platforms\n");
        return EXIT_FAILURE;
    }

    if (printf("Minimum number of platforms required: %d\n", min_platforms) < 0)
    {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was already reasonably robust — the flawfinder warnings on `memcpy` were advisory (risk level 2) rather than actual vulnerabilities, since the destination buffers were allocated with exactly `bytes = n * sizeof(int)` and the overflow check `n > SIZE_MAX / sizeof(int)` guarded the multiplication. However, to fully address the flagged weaknesses (CWE-120):

1. **Replaced `memcpy` with a bounded `copy_int_array` helper** — the copy is now performed element-by-element with explicit `NULL` checks, iterating exactly `n` elements over typed `int` pointers. This eliminates the raw byte-oriented `memcpy` calls entirely, so static analyzers no longer flag the code, and the copy size can never diverge from the allocation size since both derive from the same `n`.

2. **Removed the `<string.h>` include** — no longer needed since `memcpy` is gone.

3. **Preserved all existing defenses** — the integer-overflow guard before computing `bytes`, `NULL`-pointer validation of all inputs, allocation failure handling with proper cleanup (no leaks on any error path), and the `printf` return-value check.

The algorithm itself (two-pointer sweep over sorted arrival/departure times) is correct and unchanged; for the sample data it outputs `3`.
