#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int lcs(const char *X, const char *Y, int m, int n) {
    int L[m + 1][n + 1];
    for (int i = 0; i <= m; i++) {
        for (int j = 0; j <= n; j++) {
            if (i == 0 || j == 0)
                L[i][j] = 0;
            else if (X[i - 1] == Y[j - 1])
                L[i][j] = L[i - 1][j - 1] + 1;
            else
                L[i][j] = (L[i - 1][j] > L[i][j - 1]) ? L[i - 1][j] : L[i][j - 1];
        }
    }
    return L[m][n];
}

int longestPalSubseq(const char *str, size_t n) {
    char *revStr = (char *)malloc((n + 1) * sizeof(char));
    if (revStr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(1);
    }
    for (size_t i = 0; i < n; i++)
        revStr[n - 1 - i] = str[i];
    revStr[n] = '\0';

    int result = lcs(str, revStr, n, n);

    free(revStr);
    return result;
}

int main() {
    const char str[] = "agbdba";
    size_t n = strlen(str);
    printf("The length of the longest palindromic subsequence is %d\n", longestPalSubseq(str, n));
    return 0;
}
