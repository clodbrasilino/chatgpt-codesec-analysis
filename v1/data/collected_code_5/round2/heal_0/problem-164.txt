#include <stdio.h>

int find_sum_of_divisors(int num){
    int sum = 0;
    for(int i = 1; i <= num/2; ++i){
        if(num % i == 0)
            sum += i;
    }
    return sum;
}

int check_sum_of_divisors(int num1, int num2){
    int sum1 = find_sum_of_divisors(num1);
    int sum2 = find_sum_of_divisors(num2);
    return sum1 == sum2;
}

int main(){
    int num1, num2;
    printf("Enter two numbers : ");
    scanf("%d %d", &num1, &num2);
    
    if(check_sum_of_divisors(num1, num2)){
        printf("The sum of divisors for %d and %d are the same.\n", num1, num2);
    } 
    else{
        printf("The sum of divisors for %d and %d are not the same.\n", num1, num2);
    }

    return 0;
}
