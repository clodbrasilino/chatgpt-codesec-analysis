```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    const char *cursor;
    size_t out_pos;
    size_t len;
    size_t tail_len;
    size_t buf_size;
    int status;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    status = regcomp(&regex, "_([a-zA-Z])", REG_EXTENDED);
    if (status != 0) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    out_pos = 0U;
    cursor = input;

    while (regexec(&regex, cursor, 1U, &match, 0) == 0) {
        size_t chunk;

        if (match.rm_so < 0 || match.rm_eo < 0 ||
            (size_t)match.rm_eo > strnlen(cursor, buf_size)) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        chunk = (size_t)match.rm_so;

        if (chunk > 0U) {
            if (out_pos >= buf_size || chunk >= buf_size - out_pos) {
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(result + out_pos, cursor, chunk);
            out_pos += chunk;
        }

        if (out_pos + 1U >= buf_size) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        result[out_pos] = (char)toupper((unsigned char)cursor[match.rm_so + 1]);
        out_pos++;
        cursor += match.rm_eo;
    }

    tail_len = strnlen(cursor, buf_size);
    if (out_pos >= buf_size || tail_len >= buf_size - out_pos) {
        free(result);
        regfree(&regex);
        return NULL;
    }
    memcpy(result + out_pos, cursor, tail_len);
    out_pos += tail_len;
    result[out_pos] = '\0';

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world_example",
        "convert_this_snake_case",
        "single",
        "a_b_c_d"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        char *camel = snake_to_camel(tests[i]);
        if (camel == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("%s -> %s\n", tests[i], camel) < 0) {
            free(camel);
            return EXIT_FAILURE;
        }
        free(camel);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive — both `memcpy` calls flagged by Flawfinder were preceded by explicit bounds checks (`chunk >= buf_size - out_pos` and `tail_len >= buf_size - out_pos`), so the CWE-120 warnings were effectively false positives. However, I hardened the rewrite further:

1. **Guarded the subtraction `buf_size - out_pos`** with an additional `out_pos >= buf_size` check before both `memcpy` calls, eliminating any possibility of unsigned integer underflow producing a huge bound.

2. **Validated regex match offsets** (`rm_so`/`rm_eo`) before casting to `size_t`, protecting against a pathological regex implementation returning negative or out-of-range offsets, which would otherwise corrupt cursor arithmetic.

3. **Preserved the existing safe patterns**: `strnlen` instead of `strlen` to bound input scanning, `NULL` checks on input and `malloc`, consistent `regfree`/`free` cleanup on every error path (no leaks), and the `toupper` cast through `unsigned char` to avoid undefined behavior with negative `char` values.

The output buffer can never overflow: camelCase output is always strictly shorter than the snake_case input (each `_X` pair collapses to one character), so `len + 1` is a sufficient allocation, and the runtime checks enforce this invariant even if assumptions break.
