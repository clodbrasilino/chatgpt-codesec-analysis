#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORD_LEN 256
#define MAX_WORDS 1000

char *find_first_repeated_word(const char *str) {
    char (*words)[MAX_WORD_LEN] = NULL;
    char current_word[MAX_WORD_LEN];
    int word_count = 0;
    size_t word_len = 0;
    size_t i;
    int j;
    char *result = NULL;

    if (str == NULL) {
        return NULL;
    }

    words = calloc(MAX_WORDS, sizeof(*words));
    if (words == NULL) {
        return NULL;
    }

    memset(current_word, 0, sizeof(current_word));

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i])) {
            if (word_len < MAX_WORD_LEN - 1) {
                current_word[word_len++] = (char)tolower((unsigned char)str[i]);
            }
        } else {
            if (word_len > 0) {
                current_word[word_len] = '\0';

                for (j = 0; j < word_count; j++) {
                    if (strcmp(words[j], current_word) == 0) {
                        size_t len = strlen(current_word);
                        result = malloc(len + 1);
                        if (result != NULL) {
                            memcpy(result, current_word, len);
                            result[len] = '\0';
                        }
                        free(words);
                        return result;
                    }
                }

                if (word_count < MAX_WORDS) {
                    size_t len = strlen(current_word);
                    if (len < MAX_WORD_LEN) {
                        memcpy(words[word_count], current_word, len);
                        words[word_count][len] = '\0';
                        word_count++;
                    }
                }

                word_len = 0;
                memset(current_word, 0, sizeof(current_word));
            }
        }
    }

    if (word_len > 0) {
        current_word[word_len] = '\0';

        for (j = 0; j < word_count; j++) {
            if (strcmp(words[j], current_word) == 0) {
                size_t len = strlen(current_word);
                result = malloc(len + 1);
                if (result != NULL) {
                    memcpy(result, current_word, len);
                    result[len] = '\0';
                }
                free(words);
                return result;
            }
        }
    }

    free(words);
    return NULL;
}

int main(void) {
    const char *test_string = "The quick brown fox jumps over the lazy dog. The dog was not amused.";
    char *repeated = find_first_repeated_word(test_string);

    if (repeated != NULL) {
        printf("First repeated word: %s\n", repeated);
        free(repeated);
    } else {
        printf("No repeated word found.\n");
    }

    return 0;
}
