```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *remove_lowercase(const char *input);

char *remove_lowercase(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len;
    size_t input_len;
    size_t offset;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[256] = {0};
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return NULL;
    }

    input_len = strlen(input);
    if (input_len == (size_t)-1) {
        regfree(&regex);
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        regfree(&regex);
        fprintf(stderr, "Memory allocation failed\n");
        return NULL;
    }

    result_len = 0;
    offset = 0;

    while (offset < input_len) {
        ret = regexec(&regex, input + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = input_len - offset;
            if (result_len + remaining > input_len) {
                fprintf(stderr, "Internal bounds error\n");
                free(result);
                regfree(&regex);
                return NULL;
            }
            memcpy(result + result_len, input + offset, remaining);
            result_len += remaining;
            break;
        } else if (ret != 0) {
            char errbuf[256] = {0};
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Regex execution failed: %s\n", errbuf);
            free(result);
            regfree(&regex);
            return NULL;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so ||
            (size_t)match.rm_eo > input_len - offset) {
            fprintf(stderr, "Invalid match offsets\n");
            free(result);
            regfree(&regex);
            return NULL;
        }

        if (result_len + (size_t)match.rm_so > input_len) {
            fprintf(stderr, "Internal bounds error\n");
            free(result);
            regfree(&regex);
            return NULL;
        }

        memcpy(result + result_len, input + offset, (size_t)match.rm_so);
        result_len += (size_t)match.rm_so;
        offset += (size_t)match.rm_eo;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "HelloWorldFOO123bar",
        "KDeoALOklOOHserfLoAJSIskdsf",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix3d C4s3 Str1ng"
    };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        char *cleaned = remove_lowercase(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: %s\n", test_strings[i]);
        printf("Cleaned:  %s\n\n", cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Summary of the hardening changes made:**

1. **CWE-119/CWE-120 (static `errbuf` arrays):** Both error buffers are now zero-initialized and explicitly NUL-terminated after `regerror()`, guaranteeing safe use with `fprintf` even if the implementation truncates without terminating.

2. **CWE-126 (`strlen` over-read):** The repeated `strlen(cursor)` inside the loop was eliminated entirely. The input length is measured once, and the remaining length is derived arithmetically via `input_len - offset`, so no re-scanning of potentially problematic memory occurs.

3. **CWE-120 (`memcpy` overflow):** Every `memcpy` is now preceded by an explicit bounds check ensuring `result_len + n` never exceeds the allocated `input_len + 1` buffer. Additionally, `match.rm_so`/`match.rm_eo` values returned by `regexec` are validated (non-negative, ordered, within remaining input) before being used as sizes or offsets, protecting against any anomalous regex library behavior.

4. **Pointer arithmetic replaced with indexed offsets:** Using `offset` instead of advancing a raw `cursor` pointer makes the bounds invariants easier to verify and prevents pointer arithmetic from running past the end of the input.
