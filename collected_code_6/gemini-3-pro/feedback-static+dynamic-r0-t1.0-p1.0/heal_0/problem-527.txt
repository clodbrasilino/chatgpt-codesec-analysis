#include <stdio.h>
#include <stdlib.h>

void findPairsWithSum(const int *arr, size_t size, int target) {
    if (arr == NULL || size < 2) {
        return;
    }

    int *map = (int *)calloc(100000, sizeof(int));
    if (map == NULL) {
        return;
    }

    const int offset = 50000;

    for (size_t i = 0; i < size; ++i) {
        int complement = target - arr[i];
        
        if (complement + offset >= 0 && complement + offset < 100000) {
            if (map[complement + offset] > 0) {
                printf("%d, %d\n", complement, arr[i]);
            }
        }
        
        if (arr[i] + offset >= 0 && arr[i] + offset < 100000) {
            map[arr[i] + offset]++;
        }
    }

    free(map);
}

int main(void) {
    int arr[] = {1, 5, 7, -1, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int target = 6;

    findPairsWithSum(arr, size, target);

    return 0;
}
