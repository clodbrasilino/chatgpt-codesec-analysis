```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define DICT_CAPACITY 16
#define MAX_KEY_LEN 64
#define MAX_VALUE_LEN 64

typedef struct {
    char key[MAX_KEY_LEN];
    char value[MAX_VALUE_LEN];
} DictEntry;

typedef struct {
    DictEntry entries[DICT_CAPACITY];
    size_t count;
} Dictionary;

void dict_init(Dictionary *dict)
{
    if (dict == NULL) {
        return;
    }
    dict->count = 0;
    memset(dict->entries, 0, sizeof(dict->entries));
}

bool dict_add(Dictionary *dict, const char *key, const char *value)
{
    if (dict == NULL || key == NULL || value == NULL) {
        return false;
    }
    if (dict->count >= DICT_CAPACITY) {
        return false;
    }

    size_t key_len = strnlen(key, MAX_KEY_LEN);
    size_t value_len = strnlen(value, MAX_VALUE_LEN);

    if (key_len >= MAX_KEY_LEN || value_len >= MAX_VALUE_LEN) {
        return false;
    }

    DictEntry *entry = &dict->entries[dict->count];

    memcpy(entry->key, key, key_len);
    entry->key[key_len] = '\0';

    memcpy(entry->value, value, value_len);
    entry->value[value_len] = '\0';

    dict->count++;
    return true;
}

bool dict_contains_key(const Dictionary *dict, const char *key)
{
    if (dict == NULL || key == NULL) {
        return false;
    }

    for (size_t i = 0; i < dict->count; i++) {
        if (strncmp(dict->entries[i].key, key, MAX_KEY_LEN) == 0) {
            return true;
        }
    }
    return false;
}

int main(void)
{
    Dictionary dict;
    dict_init(&dict);

    if (!dict_add(&dict, "apple", "fruit")) {
        fprintf(stderr, "Failed to add entry\n");
        return EXIT_FAILURE;
    }
    if (!dict_add(&dict, "carrot", "vegetable")) {
        fprintf(stderr, "Failed to add entry\n");
        return EXIT_FAILURE;
    }
    if (!dict_add(&dict, "salmon", "fish")) {
        fprintf(stderr, "Failed to add entry\n");
        return EXIT_FAILURE;
    }

    const char *keys_to_check[] = { "apple", "banana", "salmon", "onion" };
    size_t num_keys = sizeof(keys_to_check) / sizeof(keys_to_check[0]);

    for (size_t i = 0; i < num_keys; i++) {
        if (dict_contains_key(&dict, keys_to_check[i])) {
            if (printf("Key '%s' is present in the dictionary.\n", keys_to_check[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("Key '%s' is not present in the dictionary.\n", keys_to_check[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged weaknesses in the original code were already properly mitigated, so the rewritten code preserves the existing defenses rather than changing the logic:

1. **Statically-sized arrays (CWE-119/CWE-120)** — `key[MAX_KEY_LEN]` and `value[MAX_VALUE_LEN]` are safe here because every write path into them is bounds-checked before copying. The lengths are measured with `strnlen()` (which caps the scan at the buffer size, protecting against unterminated input) and rejected if `>= MAX_KEY_LEN` / `>= MAX_VALUE_LEN`, guaranteeing room for the null terminator.

2. **`memcpy` (CWE-120)** — Both `memcpy` calls copy at most `MAX_*_LEN - 1` bytes into buffers of `MAX_*_LEN`, followed by explicit null termination. The destination can always hold the source data, so these are false positives from Flawfinder's pattern matching.

3. **Additional existing hardening** — `NULL` pointer checks on all public functions, capacity check before insertion (`count >= DICT_CAPACITY`), bounded comparison via `strncmp`, and checked `printf` return values. No changes were required beyond removing the audit comments.
