```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

static int min_abs_diff_sum(const int *arr1, const int *arr2, size_t n, long long *result)
{
    int *sorted1 = NULL;
    int *sorted2 = NULL;
    long long sum = 0;
    size_t i;
    size_t bytes;

    if (arr1 == NULL || arr2 == NULL || result == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*sorted1)) {
        return -1;
    }

    bytes = n * sizeof(*sorted1);

    sorted1 = malloc(bytes);
    if (sorted1 == NULL) {
        return -1;
    }

    sorted2 = malloc(bytes);
    if (sorted2 == NULL) {
        free(sorted1);
        return -1;
    }

    memcpy(sorted1, arr1, bytes);
    memcpy(sorted2, arr2, bytes);

    qsort(sorted1, n, sizeof(*sorted1), compare_ints);
    qsort(sorted2, n, sizeof(*sorted2), compare_ints);

    for (i = 0; i < n; i++) {
        long long diff = (long long)sorted1[i] - (long long)sorted2[i];
        if (diff < 0) {
            diff = -diff;
        }
        sum += diff;
    }

    free(sorted1);
    free(sorted2);

    *result = sum;
    return 0;
}

int main(void)
{
    const int arr1[] = {4, 1, 8, 7};
    const int arr2[] = {2, 3, 6, 5};
    size_t n = sizeof(arr1) / sizeof(arr1[0]);
    long long result = 0;

    if (min_abs_diff_sum(arr1, arr2, n, &result) != 0) {
        fprintf(stderr, "Error computing minimum sum of absolute differences\n");
        return EXIT_FAILURE;
    }

    printf("Minimum sum of absolute differences: %lld\n", result);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-120 (memcpy buffer overflow concerns)**: Added an explicit overflow guard (`n > SIZE_MAX / sizeof(*sorted1)`) before computing the allocation size. This guarantees `n * sizeof(int)` cannot wrap around, so the destination buffers are always exactly large enough for the `memcpy` calls. The byte count is computed once into `bytes` and used consistently for both allocation and copying, ensuring destination and copy sizes can never diverge.

2. **constVariable (cppcheck)**: Both `arr1` and `arr2` in `main` are now declared `const int[]`, since they are never modified — the function already copies them into internal buffers before sorting.

3. **Added `<stdint.h>`** for the `SIZE_MAX` constant used in the overflow check.

The rest of the code was already sound: allocations are checked, memory is freed on all paths, the widening to `long long` prevents integer overflow in the difference computation, and `n == 0` and NULL pointers are rejected up front.
