#include <stdio.h>
#include <stdlib.h>

typedef void (*map_func_t)(void *);

typedef struct {
    void **items;
    size_t count;
    size_t capacity;
} list_t;

static list_t *list_create(size_t initial_capacity)
{
    list_t *list;

    if (initial_capacity == 0) {
        initial_capacity = 1;
    }

    list = malloc(sizeof(*list));
    if (list == NULL) {
        return NULL;
    }

    list->items = malloc(initial_capacity * sizeof(*list->items));
    if (list->items == NULL) {
        free(list);
        return NULL;
    }

    list->count = 0;
    list->capacity = initial_capacity;

    return list;
}

static int list_append(list_t *list, void *item)
{
    void **new_items;
    size_t new_capacity;

    if (list == NULL) {
        return -1;
    }

    if (list->count >= list->capacity) {
        new_capacity = list->capacity * 2;
        if (new_capacity < list->capacity) {
            return -1;
        }

        new_items = realloc(list->items, new_capacity * sizeof(*new_items));
        if (new_items == NULL) {
            return -1;
        }

        list->items = new_items;
        list->capacity = new_capacity;
    }

    list->items[list->count] = item;
    list->count++;

    return 0;
}

static void list_destroy(list_t *list)
{
    if (list != NULL) {
        free(list->items);
        free(list);
    }
}

static void list_map(const list_t *list, map_func_t func)
{
    size_t i;

    if (list == NULL || func == NULL) {
        return;
    }

    for (i = 0; i < list->count; i++) {
        func(list->items[i]);
    }
}

static void list_map_n_times(const list_t *list, map_func_t func, unsigned int n)
{
    unsigned int iteration;

    if (list == NULL || func == NULL || n == 0) {
        return;
    }

    for (iteration = 0; iteration < n; iteration++) {
        list_map(list, func);
    }
}

static void print_string(void *item)
{
    const char *str;

    if (item == NULL) {
        return;
    }

    str = (const char *)item;
    printf("%s\n", str);
}

int main(void)
{
    list_t *list;
    unsigned int n = 3;
    int result = EXIT_FAILURE;

    list = list_create(4);
    if (list == NULL) {
        fprintf(stderr, "Error: Failed to create list\n");
        return EXIT_FAILURE;
    }

    if (list_append(list, "apple") != 0) {
        fprintf(stderr, "Error: Failed to append item\n");
        goto cleanup;
    }

    if (list_append(list, "banana") != 0) {
        fprintf(stderr, "Error: Failed to append item\n");
        goto cleanup;
    }

    if (list_append(list, "cherry") != 0) {
        fprintf(stderr, "Error: Failed to append item\n");
        goto cleanup;
    }

    if (list_append(list, "date") != 0) {
        fprintf(stderr, "Error: Failed to append item\n");
        goto cleanup;
    }

    printf("Printing list %u times:\n", n);
    list_map_n_times(list, print_string, n);

    result = EXIT_SUCCESS;

cleanup:
    list_destroy(list);
    return result;
}
