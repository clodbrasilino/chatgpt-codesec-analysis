#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

int int_to_roman(int num, char *buffer, size_t buffer_size) {
    const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    const char *symbols[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    size_t pos = 0;

    if (num < 1 || num > 3999 || buffer == NULL || buffer_size < 16) {
        return -1;
    }

    buffer[0] = '\0';

    for (size_t i = 0; i < 13; i++) {
        while (num >= values[i]) {
            const char *sym = symbols[i];
            
            while (*sym != '\0') {
                if (pos + 1 >= buffer_size) {
                    return -1;
                }
                buffer[pos++] = *sym++;
            }
            buffer[pos] = '\0';
            num -= values[i];
        }
    }

    return 0;
}

int main(void) {
    const int test_cases[] = {1, 4, 9, 58, 1994, 3999, -1, 4000, 0};
    size_t num_cases = sizeof(test_cases) / sizeof(test_cases[0]);
    size_t buffer_size = 16;
    char *roman_buffer = (char *)malloc(buffer_size);

    if (roman_buffer == NULL) {
        return 1;
    }

    for (size_t i = 0; i < num_cases; i++) {
        int current_num = test_cases[i];
        
        if (int_to_roman(current_num, roman_buffer, buffer_size) == 0) {
            printf("%d -> %s\n", current_num, roman_buffer);
        } else {
            printf("%d -> Error: Invalid input or buffer size\n", current_num);
        }
    }

    free(roman_buffer);
    return 0;
}
