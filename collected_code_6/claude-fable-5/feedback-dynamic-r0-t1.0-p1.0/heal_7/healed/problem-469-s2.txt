```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    long long candidate;
    long long *buy;
    long long *sell;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k > n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    buy = malloc((size_t)(k + 1) * sizeof(long long));
    if (buy == NULL) {
        return -1;
    }

    sell = malloc((size_t)(k + 1) * sizeof(long long));
    if (sell == NULL) {
        free(buy);
        return -1;
    }

    for (j = 0; j <= k; j++) {
        buy[j] = LLONG_MIN / 2;
        sell[j] = 0;
    }

    for (i = 0; i < n; i++) {
        for (j = 1; j <= k; j++) {
            candidate = sell[j - 1] - (long long)prices[i];
            if (candidate > buy[j]) {
                buy[j] = candidate;
            }
            candidate = buy[j] + (long long)prices[i];
            if (candidate > sell[j]) {
                sell[j] = candidate;
            }
        }
    }

    result = (int)sell[k];

    free(buy);
    free(sell);

    return result;
}

int main(void)
{
    int prices0[] = {3, 2, 6, 5, 0, 3};
    int prices1[] = {2, 4, 1};
    int prices2[] = {1, 2, 4, 2, 5, 7, 2, 4, 9, 0};
    int n0 = (int)(sizeof(prices0) / sizeof(prices0[0]));
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int profit;

    profit = maxProfit(2, prices0, n0);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The algorithm itself is the standard O(n·k) dynamic programming solution for "Best Time to Buy and Sell Stock IV" (at most `k` transactions), and it is implemented correctly:

1. **Guard clauses** — `NULL` pointer, `n <= 1`, and `k <= 0` all correctly return `0` (no profit possible).

2. **Fast path for large `k`** — when `k > n/2`, the transaction cap is effectively unlimited, so the greedy sum of all positive deltas is correct and avoids unnecessary allocation.

3. **Overflow safety** — the DP states use `long long`, and the sentinel is `LLONG_MIN / 2` rather than `LLONG_MIN`, which prevents signed overflow (undefined behavior) when adding `prices[i]` to an "impossible" buy state.

4. **Memory hygiene** — both allocations are checked, and `buy` is freed if the `sell` allocation fails; both buffers are freed before returning.

5. **State update ordering** — updating `buy[j]` before `sell[j]` within the same day permits a same-day buy/sell, which contributes zero profit and never worsens the result, so it is correct.

Running the tests yields `7`, `2`, and `13` respectively, which are the correct maxima for `k = 2` on those price sequences — the "got 5 / got 0" failures described in the comment header correspond to an earlier defective revision (likely an off-by-one in the `j` loop bounds or missing DP initialization), which this version resolves.
