#include <stdio.h>
#include <stdlib.h>

int **generate_spiral(int n) {
    if (n <= 0) {
        return NULL;
    }

    int **matrix = (int **)malloc(n * sizeof(int *));
    if (matrix == NULL) {
        return NULL;
    }

    for (int i = 0; i < n; i++) {
        matrix[i] = (int *)malloc(n * sizeof(int));
        if (matrix[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(matrix[j]);
            }
            free(matrix);
            return NULL;
        }
    }

    int top = 0;
    int bottom = n - 1;
    int left = 0;
    int right = n - 1;
    int val = 1;

    while (top <= bottom && left <= right) {
        for (int i = left; i <= right; i++) {
            matrix[top][i] = val++;
        }
        top++;

        for (int i = top; i <= bottom; i++) {
            matrix[i][right] = val++;
        }
        right--;

        if (top <= bottom) {
            for (int i = right; i >= left; i--) {
                matrix[bottom][i] = val++;
            }
            bottom--;
        }

        if (left <= right) {
            for (int i = bottom; i >= top; i--) {
                matrix[i][left] = val++;
            }
            left++;
        }
    }

    return matrix;
}

int main() {
    int n = 4;
    int **matrix = generate_spiral(n);

    if (matrix != NULL) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                printf("%4d ", matrix[i][j]);
            }
            printf("\n");
            free(matrix[i]);
        }
        free(matrix);
    } else {
        printf("Failed to allocate matrix for n = %d\n", n);
    }

    return 0;
}
