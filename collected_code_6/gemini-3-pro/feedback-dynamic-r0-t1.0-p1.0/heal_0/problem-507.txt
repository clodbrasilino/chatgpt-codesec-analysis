#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char** remove_words(char** word_list, int list_size, const char** words_to_remove, int num_words_to_remove, int* new_size) {
    if (!word_list || !words_to_remove || !new_size || list_size <= 0 || num_words_to_remove < 0) {
        return NULL;
    }

    char** result = (char**)calloc(list_size, sizeof(char*));
    if (!result) {
        return NULL;
    }

    int count = 0;

    for (int i = 0; i < list_size; i++) {
        if (!word_list[i]) {
            continue;
        }

        int should_remove = 0;
        for (int j = 0; j < num_words_to_remove; j++) {
            if (words_to_remove[j] && strcmp(word_list[i], words_to_remove[j]) == 0) {
                should_remove = 1;
                break;
            }
        }

        if (!should_remove) {
            result[count] = strdup(word_list[i]);
            if (!result[count]) {
                for (int k = 0; k < count; ++k) {
                    free(result[k]);
                }
                free(result);
                return NULL;
            }
            count++;
        }
    }

    *new_size = count;
    return result;
}

int main() {
    char* original_list[] = {"apple", "banana", "orange", "grape", "apple", "kiwi"};
    int original_size = sizeof(original_list) / sizeof(original_list[0]);

    const char* to_remove[] = {"apple", "kiwi"};
    int remove_size = sizeof(to_remove) / sizeof(to_remove[0]);

    int new_size = 0;
    char** filtered_list = remove_words(original_list, original_size, to_remove, remove_size, &new_size);

    if (filtered_list) {
        for (int i = 0; i < new_size; i++) {
            printf("%s\n", filtered_list[i]);
            free(filtered_list[i]);
        }
        free(filtered_list);
    }

    return 0;
}
