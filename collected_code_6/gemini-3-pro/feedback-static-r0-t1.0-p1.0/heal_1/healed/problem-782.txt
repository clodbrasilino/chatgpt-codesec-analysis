#include <stdio.h>
#include <stdlib.h>

int sumOddLengthSubarrays(const int* arr, int arrSize) {
    if (arr == NULL || arrSize <= 0) {
        return 0;
    }
    
    int totalSum = 0;
    
    for (int i = 0; i < arrSize; i++) {
        int currentSum = 0;
        for (int j = i; j < arrSize; j++) {
            currentSum += arr[j];
            if ((j - i + 1) % 2 != 0) {
                totalSum += currentSum;
            }
        }
    }
    
    return totalSum;
}

int main(void) {
    int arr[] = {1, 4, 2, 5, 3};
    int arrSize = sizeof(arr) / sizeof(arr[0]);
    
    int result = sumOddLengthSubarrays(arr, arrSize);
    printf("%d\n", result);
    
    return 0;
}
