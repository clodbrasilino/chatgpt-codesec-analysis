#include <stdio.h>
#include <stdlib.h>

#define ZODIAC_COUNT 12

static const char *chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return "Unknown";
    }
    return signs[index];
}

int main(void)
{
    int year;
    int result;

    printf("Enter a year: ");
    result = scanf("%d", &year);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input, expected an integer year.\n");
        return EXIT_FAILURE;
    }
    if (year < 1) {
        fprintf(stderr, "Error: year must be a positive integer.\n");
        return EXIT_FAILURE;
    }

    printf("The Chinese zodiac sign for year %d is: %s\n",
           year, chinese_zodiac(year));

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already quite solid. The Flawfinder warning about the statically-sized array (`signs[12]`) is largely a false positive here, since the index is derived from `year % 12` and corrected for negative values, guaranteeing a range of `[0, 11]`. 2. **Adds an explicit bounds check** (`index < 0 || index >= ZODIAC_COUNT`) before dereferencing the array, returning a safe fallback string if the invariant is ever violated.
3. **Retains the existing input validation** — checking the `scanf` return value and rejecting non-positive years — which already mitigated the common CWE-20 (improper input validation) risks.
