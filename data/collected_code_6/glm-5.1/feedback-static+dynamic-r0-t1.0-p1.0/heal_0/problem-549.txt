#include <stdio.h>
#include <stdlib.h>

unsigned long long sum_fifth_power_odd(int n) {
    if (n < 0) {
        return 0;
    }

    unsigned long long sum = 0;
    int count = 0;
    int current = 1;

    while (count < n) {
        unsigned long long term = (unsigned long long)current * current * current * current * current;
        sum += term;
        current += 2;
        count++;
    }

    return sum;
}

int main(int argc, char *argv[]) {
    int n;

    if (argc != 2) {
        fprintf(stderr, "Usage: %s <n>\n", argv[0]);
        return EXIT_FAILURE;
    }

    if (sscanf(argv[1], "%d", &n) != 1) {
        fprintf(stderr, "Invalid input: not an integer\n");
        return EXIT_FAILURE;
    }

    if (n < 0) {
        fprintf(stderr, "Invalid input: n must be non-negative\n");
        return EXIT_FAILURE;
    }

    unsigned long long result = sum_fifth_power_odd(n);
    printf("%llu\n", result);

    return EXIT_SUCCESS;
}
