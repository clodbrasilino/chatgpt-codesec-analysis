#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

static Node *create_node(int data)
{
    Node *new_node = malloc(sizeof(*new_node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

static int append_node(Node **head, int data)
{
    Node *new_node;
    Node *current;

    if (head == NULL) {
        return -1;
    }

    new_node = create_node(data);
    if (new_node == NULL) {
        return -1;
    }

    if (*head == NULL) {
        *head = new_node;
        return 0;
    }

    current = *head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = new_node;
    return 0;
}

static int list_contains(const Node *head, int data)
{
    const Node *current = head;
    while (current != NULL) {
        if (current->data == data) {
            return 1;
        }
        current = current->next;
    }
    return 0;
}

static void free_list(Node *head)
{
    Node *current = head;
    Node *next;

    while (current != NULL) {
        next = current->next;
        free(current);
        current = next;
    }
}

static int list_difference(const Node *list1, const Node *list2, Node **result)
{
    Node *diff = NULL;
    const Node *current;

    if (result == NULL) {
        return -1;
    }
    *result = NULL;

    current = list1;
    while (current != NULL) {
        if (!list_contains(list2, current->data) &&
            !list_contains(diff, current->data)) {
            if (append_node(&diff, current->data) != 0) {
                free_list(diff);
                return -1;
            }
        }
        current = current->next;
    }

    *result = diff;
    return 0;
}

static void print_list(const Node *head)
{
    const Node *current = head;
    while (current != NULL) {
        printf("%d", current->data);
        if (current->next != NULL) {
            printf(" -> ");
        }
        current = current->next;
    }
    printf("\n");
}

int main(void)
{
    Node *list1 = NULL;
    Node *list2 = NULL;
    Node *diff = NULL;
    int values1[] = {1, 2, 3, 4, 5, 6};
    int values2[] = {4, 5, 6, 7, 8};
    size_t i;

    for (i = 0; i < sizeof(values1) / sizeof(values1[0]); i++) {
        if (append_node(&list1, values1[i]) != 0) {
            fprintf(stderr, "Error: failed to build list 1\n");
            free_list(list1);
            return EXIT_FAILURE;
        }
    }

    for (i = 0; i < sizeof(values2) / sizeof(values2[0]); i++) {
        if (append_node(&list2, values2[i]) != 0) {
            fprintf(stderr, "Error: failed to build list 2\n");
            free_list(list1);
            free_list(list2);
            return EXIT_FAILURE;
        }
    }

    printf("List 1: ");
    print_list(list1);
    printf("List 2: ");
    print_list(list2);

    if (list_difference(list1, list2, &diff) != 0) {
        fprintf(stderr, "Error: failed to compute list difference\n");
        free_list(list1);
        free_list(list2);
        return EXIT_FAILURE;
    }

    printf("Difference (list1 - list2): ");
    print_list(diff);

    free_list(list1);
    free_list(list2);
    free_list(diff);

    return EXIT_SUCCESS;
}
