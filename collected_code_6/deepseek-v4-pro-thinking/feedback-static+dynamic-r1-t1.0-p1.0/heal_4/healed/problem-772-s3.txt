#include <ctype.h>
#include <stddef.h>
#include <stdio.h>
#include <string.h>

void remove_words_of_length_k(char *str, size_t size, int k) {
    if (str == NULL || k <= 0 || size == 0) {
        return;
    }

    const char *end = str + size;
    char *src = str;
    char *dst = str;
    int has_word = 0;

    while (src < end && *src != '\0') {
        while (src < end && *src != '\0' && isspace((unsigned char)*src)) {
            src++;
        }

        if (src >= end || *src == '\0') {
            break;
        }

        const char *word_start = src;

        while (src < end && *src != '\0' && !isspace((unsigned char)*src)) {
            src++;
        }

        size_t word_len = (size_t)(src - word_start);

        if (word_len != (size_t)k) {
            if (has_word) {
                if (dst < end - 1) {
                    *dst++ = ' ';
                } else {
                    break;
                }
            }

            if ((size_t)(end - dst) > word_len) {
                memmove(dst, word_start, word_len);
                dst += word_len;
                has_word = 1;
            } else {
                break;
            }
        }
    }

    if (dst < end) {
        *dst = '\0';
    } else {
        str[size - 1] = '\0';
    }
}

int main(void) {
    char str[] = "  hello   world  is  a test  ";
    remove_words_of_length_k(str, sizeof(str), 2);
    printf("%s\n", str);
    return 0;
}
