#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_STRING_LENGTH 1024

int replace_chars_with_colon(char *str, size_t buffer_size) {
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t str_len;
    size_t result_len = 0;
    size_t pos = 0;
    int ret;

    if (str == NULL || buffer_size == 0) {
        return -1;
    }

    str_len = strnlen(str, buffer_size);
    if (str_len == buffer_size) {
        return -1;
    }

    ret = regcomp(&regex, "[ ,.]", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    result = (char *)malloc(buffer_size);
    if (result == NULL) {
        regfree(&regex);
        return -1;
    }

    result[0] = '\0';

    while (pos < str_len) {
        ret = regexec(&regex, str + pos, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        } else if (ret != 0) {
            free(result);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < 0 || match.rm_so > match.rm_eo) {
            free(result);
            regfree(&regex);
            return -1;
        }

        {
            size_t copy_len = (size_t)match.rm_so;
            if (copy_len > buffer_size - result_len - 2) {
                free(result);
                regfree(&regex);
                return -1;
            }

            if (copy_len > 0) {
                if (result_len + copy_len >= buffer_size) {
                    free(result);
                    regfree(&regex);
                    return -1;
                }
                if (result_len + copy_len <= buffer_size) {
                    memcpy(result + result_len, str + pos, copy_len);
                    result_len += copy_len;
                } else {
                    free(result);
                    regfree(&regex);
                    return -1;
                }
            }
        }
        if (result_len >= buffer_size - 1) {
            free(result);
            regfree(&regex);
            return -1;
        }
        result[result_len++] = ':';
        result[result_len] = '\0';

        pos += (size_t)match.rm_eo;
    }

    if (str_len - pos > buffer_size - result_len - 1) {
        free(result);
        regfree(&regex);
        return -1;
    }

    if (str_len - pos > 0) {
        if (result_len + (str_len - pos) >= buffer_size) {
            free(result);
            regfree(&regex);
            return -1;
        }
        if (result_len + (str_len - pos) <= buffer_size) {
            memcpy(result + result_len, str + pos, str_len - pos);
            result_len += str_len - pos;
        } else {
            free(result);
            regfree(&regex);
            return -1;
        }
    }
    result[result_len] = '\0';

    if (result_len + 1 > buffer_size) {
        free(result);
        regfree(&regex);
        return -1;
    }

    if (result_len + 1 <= buffer_size) {
        memcpy(str, result, result_len + 1);
    } else {
        free(result);
        regfree(&regex);
        return -1;
    }

    free(result);
    regfree(&regex);

    return 0;
}

int main(void) {
    char test_string[MAX_STRING_LENGTH];
    int status;
    int written;

    written = snprintf(test_string, sizeof(test_string), "%s", "Hello, World. This is a test string, with spaces.");
    if (written < 0 || (size_t)written >= sizeof(test_string)) {
        fprintf(stderr, "Error: String too long\n");
        return EXIT_FAILURE;
    }

    printf("Original string: %s\n", test_string);

    status = replace_chars_with_colon(test_string, sizeof(test_string));
    if (status != 0) {
        fprintf(stderr, "Error: Failed to replace characters\n");
        return EXIT_FAILURE;
    }

    printf("Modified string: %s\n", test_string);

    return EXIT_SUCCESS;
}
