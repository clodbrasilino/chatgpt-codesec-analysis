#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void reverse_words(char *str) {
    if (str == NULL) return;
    char *start = str, *end, temp;
    int len = 0;
    while (str[len] != '\0') len++;
    for (start = end = str; start <= str + len && *start != '\0'; start++) {
        if (*start == ' ' || *(start + 1) == '\0') {
            start--;
            while (end < start) {
                temp = *end;
                *end = *start;
                *start = temp;
                end++;
                start--;
            }
            end = start + 2;
        }
    }
    if (end < str + len) {
        while (end < str + len - 1) {
            *end = *(end + 1);
            end++;
        }
        *(end) = '\0';
    }
}

int main() {
    char str[] = "Hello World This is C";
    reverse_words(str);
    printf("%s\n", str);
    return 0;
}
