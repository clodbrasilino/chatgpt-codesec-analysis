```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096

typedef struct {
    char *items;
    size_t count;
} Tuple;

Tuple *string_to_tuple(const char *input)
{
    Tuple *tuple = NULL;
    size_t length = 0;
    char *items = NULL;

    if (input == NULL) {
        return NULL;
    }

    length = strnlen(input, MAX_INPUT_LENGTH);
    if (length == 0 || length >= MAX_INPUT_LENGTH) {
        return NULL;
    }

    items = malloc(length + 1);
    if (items == NULL) {
        return NULL;
    }

    memcpy(items, input, length);
    items[length] = '\0';

    tuple = malloc(sizeof(Tuple));
    if (tuple == NULL) {
        free(items);
        return NULL;
    }

    tuple->items = items;
    tuple->count = length;
    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if (tuple == NULL || tuple->items == NULL) {
        return;
    }

    printf("[");
    for (i = 0; i < tuple->count; i++) {
        printf("'%c'", tuple->items[i]);
        if (i + 1 < tuple->count) {
            printf(", ");
        }
    }
    printf("]\n");
    fflush(stdout);
}

int main(void)
{
    const char *inputs[] = { "python3.0", "item1", "15.10" };
    size_t i;
    size_t n = sizeof(inputs) / sizeof(inputs[0]);

    for (i = 0; i < n; i++) {
        Tuple *tuple = string_to_tuple(inputs[i]);

        if (tuple == NULL) {
            fprintf(stderr, "Failed to convert string to tuple: \"%s\"\n",
                    inputs[i]);
            return EXIT_FAILURE;
        }

        printf("Input string: \"%s\"\n", inputs[i]);
        printf("Tuple: ");
        print_tuple(tuple);

        free_tuple(tuple);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes (as an experienced C developer):**

The original code was functionally close to correct, but the reported test failures (`got <no output>`) and the Flawfinder warning point to a few subtle issues that the rewrite addresses:

1. **Unterminated buffer (root cause of the CWE-120 concern):** `malloc(length)` followed by `memcpy` of exactly `length` bytes produced a buffer that was *not* NUL-terminated. While `print_tuple` iterated by count and didn't technically overrun, any future code treating `tuple->items` as a C string would read out of bounds. The fix allocates `length + 1` bytes and explicitly terminates, guaranteeing the destination can always hold the source data plus terminator.

2. **Missing output flushing:** The `<no output>` test failures are consistent with buffered stdout never being flushed in the test harness environment (e.g., when stdout is a pipe and the process is inspected before exit). Adding `fflush(stdout)` after printing the tuple ensures output is delivered deterministically.

3. **Defensive check in `print_tuple`:** Added a guard against `tuple->items == NULL` to avoid dereferencing a null pointer if a partially constructed tuple were ever passed in.

4. **Bounded string handling retained:** `strnlen` with `MAX_INPUT_LENGTH` remains, correctly rejecting empty and oversized inputs, which keeps the `memcpy` provably safe since `length` is validated before allocation.
