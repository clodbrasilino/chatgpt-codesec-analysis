#include <stdio.h>
#include <stdlib.h>

static void reverse(int *array, size_t start, size_t end) {
    while (start < end) {
        int temp = array[start];
        array[start] = array[end];
        array[end] = temp;
        start++;
        end--;
    }
}

void rotate_right(int *array, size_t size, size_t k) {
    if (array == NULL || size == 0) {
        return;
    }
    k = k % size;
    if (k == 0) {
        return;
    }
    reverse(array, 0, size - 1);
    reverse(array, 0, k - 1);
    reverse(array, k, size - 1);
}

int main(void) {
    size_t size = 7;
    int *arr = malloc(size * sizeof(int));
    if (arr == NULL) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < size; i++) {
        arr[i] = (int)i + 1;
    }

    size_t k = 3;
    rotate_right(arr, size, k);

    for (size_t i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    free(arr);
    return EXIT_SUCCESS;
}
