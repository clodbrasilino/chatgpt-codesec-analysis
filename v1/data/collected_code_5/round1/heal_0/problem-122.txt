#include <stdio.h>
#include <stdbool.h>

#define MAX 10000

bool isSmart(int num) {
    int count = 0;
    for (int i = 1; i <= num; i++) {
        if (num % i == 0){
            count++;
            if(count > 3) {
                break;
            }
        }
    }
    if (count == 3) {
        return true;
    } else {
        return false;
    }
}

int smartNumber(int n){
    int count = 0;
    for (int i = 1;; i++){
        if (isSmart(i)) {
            count++;
            if (count == n) {
                return i;
            }
        }
    }
    return -1;
}

int main() {
    int n;
    printf("Enter the nth smart number you want to find: ");
    scanf("%d", &n);
	if(n<1 || n>MAX) {
        printf("Input is out of range. Please enter a positive number up to %d\n", MAX);
        return 0;
    }

    int sNumber = smartNumber(n);

    if(sNumber != -1) {
        printf("The %dth smart number is: %d\n", n, sNumber);
    } else {
        printf("Could not find the %dth smart number\n", n);
    }

    return 0;
}
