```c
#include <stdio.h>

long nth_star_number(long n) {
    return 6 * n * (n - 1) + 1;
}

int main() {
    long n;
    printf("Enter the value of n: ");
    if(scanf("%ld", &n) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    printf("The %ldth star number is: %ld\n", n, nth_star_number(n));
    return 0;
}
```
