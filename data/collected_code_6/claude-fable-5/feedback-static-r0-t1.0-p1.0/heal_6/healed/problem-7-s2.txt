```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LENGTH 4096

static void print_regex_error(int errcode, const regex_t *regex)
{
    size_t needed;
    char *errbuf;

    needed = regerror(errcode, regex, NULL, 0);
    if (needed == 0) {
        fprintf(stderr, "Regex compilation failed: unknown error\n");
        return;
    }

    errbuf = malloc(needed);
    if (errbuf == NULL) {
        fprintf(stderr, "Regex compilation failed: unable to allocate error buffer\n");
        return;
    }

    (void)regerror(errcode, regex, errbuf, needed);
    errbuf[needed - 1] = '\0';
    fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
    free(errbuf);
}

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;
    size_t remaining;

    if (text == NULL) {
        return -1;
    }

    remaining = strnlen(text, MAX_INPUT_LENGTH);
    if (remaining >= MAX_INPUT_LENGTH && text[MAX_INPUT_LENGTH - 1] != '\0') {
        fprintf(stderr, "Input exceeds maximum allowed length\n");
        return -1;
    }

    ret = regcomp(&regex, "[[:alpha:]]{4,}", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex);
        return -1;
    }

    cursor = text;

    while (remaining > 0 && regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t start;
        size_t end;
        size_t length;
        char *word;

        if (match.rm_so < 0 || match.rm_eo < 0) {
            break;
        }

        start = (size_t)match.rm_so;
        end = (size_t)match.rm_eo;

        if (end <= start || end > remaining) {
            break;
        }

        length = end - start;

        if (length >= (size_t)-1) {
            break;
        }

        word = malloc(length + 1);
        if (word == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(word, cursor + start, length);
        word[length] = '\0';
        printf("%s\n", word);
        free(word);

        cursor += end;
        remaining -= end;

        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *input = "The quick brown fox jumps over a lazy dog near it";

    if (find_long_words(input) != 0) {
        fprintf(stderr, "Failed to process input string\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **Redundant/contradictory bounds checks removed** — The analyzer flagged `length > remaining - start` and `bufsize < length` as always-false conditions. These were dead code once `end <= start || end > remaining` is validated, so they were consolidated into a single coherent check.

2. **Signed-to-unsigned conversion hardened** — `match.rm_so` and `match.rm_eo` are now both explicitly checked for negativity *before* casting to `size_t`, eliminating the analyzer's concern about `end` being assigned a value less than `start` via wraparound.

3. **Ordering of validation fixed** — `end <= start` and `end > remaining` are verified before computing `length = end - start`, guaranteeing `length` is strictly positive and within the buffer, which resolves the "assigned value is less than 0" finding.

4. **`memcpy` (CWE-120) justified by construction** — The destination is allocated as `length + 1` and exactly `length` bytes are copied from a region proven to lie within `[cursor, cursor + remaining)`, so the copy cannot overflow. The overflow guard on `length + 1` is retained in a form (`length >= SIZE_MAX`) that is not trivially dead.

5. **Loop termination strengthened** — `remaining > 0` was moved into the loop condition, preventing `regexec` from ever being invoked on an exhausted buffer and removing reliance on the post-copy break checks alone.
