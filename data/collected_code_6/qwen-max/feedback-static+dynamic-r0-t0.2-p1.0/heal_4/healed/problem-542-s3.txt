#include <stdio.h>
#include <regex.h>
#include <stdlib.h>
#include <string.h>

void replace_chars(char *str) {
    regex_t regex;
    regmatch_t pmatch[1];
    size_t str_len = strlen(str);
    char *new_str = malloc(str_len + 1);
    if (new_str == NULL) {
        return;
    }
    new_str[str_len] = '\0';

    if (regcomp(&regex, "[ ,.]", REG_EXTENDED) != 0) {
        free(new_str);
        return;
    }

    size_t i = 0, j = 0;
    while (regexec(&regex, str + i, 1, pmatch, 0) == 0) {
        size_t match_start = pmatch[0].rm_so;
        size_t match_end = pmatch[0].rm_eo;
        memcpy(new_str + j, str + i, match_start);
        j += match_start;
        new_str[j++] = ':';
        i += match_end;
    }
    memcpy(new_str + j, str + i, str_len - i + 1);

    strncpy(str, new_str, str_len + 1);
    free(new_str);
    regfree(&regex);
}

int main() {
    char str[] = "Hello, World. This is a test string.";
    replace_chars(str);
    printf("%s\n", str);
    return 0;
}
