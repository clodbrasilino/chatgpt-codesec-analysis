#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count);
void free_shortlist(char **list, size_t count);

static void cleanup_partial(char **result, size_t filled)
{
    size_t j;

    for (j = 0; j < filled; j++) {
        free(result[j]);
    }
    free(result);
}

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count)
{
    char **result = NULL;
    size_t matched = 0;
    size_t i;

    if (words == NULL || result_count == NULL) {
        if (result_count != NULL) {
            *result_count = 0;
        }
        return NULL;
    }

    *result_count = 0;

    for (i = 0; i < count; i++) {
        if (words[i] != NULL) {
            size_t len = strnlen(words[i], MAX_WORD_LEN);
            if (len > n && len < MAX_WORD_LEN) {
                matched++;
            }
        }
    }

    if (matched == 0) {
        return NULL;
    }

    if (matched > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(matched * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count && *result_count < matched; i++) {
        if (words[i] != NULL) {
            size_t len = strnlen(words[i], MAX_WORD_LEN);
            if (len > n && len < MAX_WORD_LEN) {
                size_t copy_size;
                char *copy;

                if (len >= SIZE_MAX) {
                    cleanup_partial(result, *result_count);
                    *result_count = 0;
                    return NULL;
                }

                copy_size = len + 1;
                copy = malloc(copy_size);

                if (copy == NULL) {
                    cleanup_partial(result, *result_count);
                    *result_count = 0;
                    return NULL;
                }

                if (len > copy_size - 1) {
                    free(copy);
                    cleanup_partial(result, *result_count);
                    *result_count = 0;
                    return NULL;
                }

                memcpy(copy, words[i], len);
                copy[len] = '\0';

                result[*result_count] = copy;
                (*result_count)++;
            }
        }
    }

    if (*result_count == 0) {
        free(result);
        return NULL;
    }

    return result;
}

void free_shortlist(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *const words[] = {
        "apple", "cat", "banana", "dog", "elephant", "sun", "computer"
    };
    size_t word_count = sizeof(words) / sizeof(words[0]);
    size_t n = 4;
    size_t result_count = 0;
    char **shortlisted;
    size_t i;

    shortlisted = shortlist_words(words, word_count, n, &result_count);

    if (shortlisted == NULL) {
        if (printf("No words longer than %zu found.\n", n) < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("Words longer than %zu characters:\n", n) < 0) {
        free_shortlist(shortlisted, result_count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_count; i++) {
        if (printf("%s\n", shortlisted[i]) < 0) {
            free_shortlist(shortlisted, result_count);
            return EXIT_FAILURE;
        }
    }

    free_shortlist(shortlisted, result_count);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The flagged weakness (Flawfinder, CWE-120 on `memcpy`) was already effectively mitigated in the original code: the destination buffer `copy` is allocated with `len + 1` bytes, and exactly `len` bytes are copied followed by explicit null termination. Flawfinder flags `memcpy` heuristically without data-flow analysis, so this was a false positive.

The rewrite hardens the code further to make the safety invariant explicit and auditable:

1. **Overflow guard on `len + 1`** — a `len >= SIZE_MAX` check prevents any theoretical wraparound when computing `copy_size` (already implicitly bounded by `MAX_WORD_LEN`, but now provable locally).
2. **Explicit bounds check before `memcpy`** — `len > copy_size - 1` verifies the destination can hold the source data plus the null terminator, directly satisfying the CWE-120 recommendation.

Other defensive properties of the original were preserved: `strnlen` bounding against `MAX_WORD_LEN`, the `SIZE_MAX / sizeof(char *)` allocation-overflow check, NULL-pointer validation, full cleanup on partial allocation failure, and checked `printf` return values.
