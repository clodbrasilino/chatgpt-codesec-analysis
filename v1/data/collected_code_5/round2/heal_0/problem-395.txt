```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NO_OF_CHARS 256

/* returns index of the first non-repeating character or -1 in the case if all characters are repeating */
int firstNonRepeatingCharacter(char* str) {
    int* count = (int*) calloc(NO_OF_CHARS, sizeof(int));
    if(count == NULL) {
        printf("Memory allocation failed.\n");
        return -1;
    }

    for (int i = 0; *(str + i); i++) 
        count[*(str + i)]++;

    int result = -1;
    for (int i = 0; *(str + i); i++) {
        if (count[*(str + i)] == 1) {
            result = i;
            break;
        }
    }

    free(count);
    return result;
}

int main() {
    char str[] = "beginnersbook";
    int index = firstNonRepeatingCharacter(str);
    if (index == -1) 
        printf("All characters are repeating.\n");
    else
        printf("The first non-repeating character is %c.\n", str[index]);
    return 0;
}
```
