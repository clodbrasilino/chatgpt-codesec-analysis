```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 4096

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t len = 0;

    while (len < max_len && s[len] != '\0') {
        len++;
    }
    return len;
}

char **extract_strings_by_size(const char **list, size_t list_count, size_t target_size, size_t *result_count)
{
    char **result = NULL;
    size_t count = 0;
    size_t i;

    if (result_count == NULL) {
        return NULL;
    }

    *result_count = 0;

    if (list == NULL || list_count == 0 || target_size >= MAX_STRING_LEN) {
        return NULL;
    }

    for (i = 0; i < list_count; i++) {
        if (list[i] != NULL && bounded_strlen(list[i], MAX_STRING_LEN) == target_size) {
            count++;
        }
    }

    if (count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(count * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    count = 0;
    for (i = 0; i < list_count; i++) {
        if (list[i] != NULL && bounded_strlen(list[i], MAX_STRING_LEN) == target_size) {
            result[count] = malloc(target_size + 1);
            if (result[count] == NULL) {
                size_t j;
                for (j = 0; j < count; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            memcpy(result[count], list[i], target_size);
            result[count][target_size] = '\0';
            count++;
        }
    }

    *result_count = count;
    return result;
}

void free_extracted_strings(char **strings, size_t count)
{
    size_t i;

    if (strings == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(strings[i]);
    }
    free(strings);
}

int main(void)
{
    const char *words[] = { "apple", "cat", "banana", "dog", "grape", "melon", "fig" };
    size_t word_count = sizeof(words) / sizeof(words[0]);
    size_t result_count = 0;
    size_t target_size = 5;
    size_t i;
    char **extracted;

    extracted = extract_strings_by_size(words, word_count, target_size, &result_count);

    if (extracted == NULL) {
        if (printf("No strings of size %zu found.\n", target_size) < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("Strings of size %zu:\n", target_size) < 0) {
        free_extracted_strings(extracted, result_count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_count; i++) {
        if (printf("%s\n", extracted[i]) < 0) {
            free_extracted_strings(extracted, result_count);
            return EXIT_FAILURE;
        }
    }

    free_extracted_strings(extracted, result_count);
    return EXIT_SUCCESS;
}
```

Key changes made to address the Flawfinder findings:

1. **CWE-126 (strlen over-read)**: Replaced all `strlen` calls with a `bounded_strlen` helper that never reads beyond `MAX_STRING_LEN` bytes, so a non-terminated string cannot cause an unbounded over-read.

2. **CWE-120 (memcpy overflow)**: The destination buffer is now allocated with exactly `target_size + 1` bytes, and `memcpy` copies exactly `target_size` bytes with the terminator written explicitly, guaranteeing the destination always holds the copied data.

3. **Integer overflow hardening**: Added a check on `count * sizeof(char *)` before `malloc` to prevent an allocation-size overflow, and rejected `target_size` values at or above the bound to keep `target_size + 1` safe.
