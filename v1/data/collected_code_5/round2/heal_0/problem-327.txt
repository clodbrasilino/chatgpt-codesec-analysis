#include <stdio.h>
#include <stdbool.h>

bool isIsosceles(int a, int b, int c) {
    if(a<=0 || b<=0 || c<=0)
    {
        return false;
    }
    if ((a == b) || (a == c) || (b == c)) {
        return true;
    }
    return false;
}

int main() {
    int side1, side2, side3;
    
    printf("Enter the sides of triangle: ");
    if(scanf("%d %d %d",&side1,&side2,&side3) != 3){
       printf("Invalid input\n");
       return 1;
    }

    if (isIsosceles(side1, side2, side3))
        printf("The triangle is an Isosceles triangle.\n");
    else
        printf("The triangle is not an Isosceles triangle.\n");

    return 0;
}
