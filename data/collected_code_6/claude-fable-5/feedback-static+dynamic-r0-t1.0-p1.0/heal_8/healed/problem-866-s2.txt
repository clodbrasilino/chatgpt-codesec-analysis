```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN] = {0};
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    for (i = 0; i < len && i < MAX_INPUT_LEN - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(lowered, months_31[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN] = {0};
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    } else if (len == sizeof(input) - 1) {
        flush_stdin();
    }

    if (len == 0 || input[0] == '\0') {
        fprintf(stderr, "No input provided.\n");
        return EXIT_FAILURE;
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and their mitigations:**

1. **`char lowered[MAX_INPUT_LEN]` in `has_31_days` (CWE-119/CWE-120, risk 2):** This is a false positive in practice. The original code already mitigates the risk properly: it uses `strnlen` with an explicit bound, rejects inputs of length `>= MAX_INPUT_LEN`, bounds the copy loop with both `len` and `MAX_INPUT_LEN - 1`, and explicitly null-terminates. No overflow is possible. The rewrite preserves these safeguards, plus a `NULL` check.

2. **`getchar()` in `flush_stdin` (CWE-120/CWE-20, risk 1):** Also a false positive — no buffer is written to; characters are simply discarded until newline or `EOF`, and the `EOF` check prevents an infinite loop on closed streams. Kept as-is (restructured as a `do/while` for clarity).

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120, risk 2):** Mitigated because input is read via `fgets(input, sizeof(input), stdin)`, which cannot overflow the buffer. The code also forces null-termination, strips the trailing newline, and drains leftover characters from `stdin` when the input was truncated — preventing stale data from corrupting subsequent reads.

Minor hardening added in the rewrite: the return value of `fflush(stdout)` is checked, and `len` is decremented after stripping the newline so the empty-input check is consistent.
