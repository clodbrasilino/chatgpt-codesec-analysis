```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char *output;
    size_t len;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    output = remove_lowercase(buffer);
    if (output == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", output);
    free(output);

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **CWE-119/CWE-120 (statically-sized buffer):** The original code was already reasonably safe because `fgets(buffer, sizeof(buffer), stdin)` performs bounds checking — it reads at most `sizeof(buffer) - 1` bytes and always null-terminates. The buffer size is tied to `sizeof(buffer)` rather than a repeated magic number, so the read cannot overflow. As defense-in-depth, I added an explicit `buffer[sizeof(buffer) - 1] = '\0';` to guarantee termination regardless of assumptions about `fgets`.

2. **CWE-126 (`strlen` over-read on unterminated strings):** Replaced `strlen(buffer)` with `strnlen(buffer, sizeof(buffer))`. Combined with the forced null terminator, this eliminates any possibility of reading past the end of `buffer`, even in the theoretical case of a non-terminated string.

3. **Existing good practices preserved:**
   - `remove_lowercase` already uses `strnlen` with an explicit cap and rejects oversized input.
   - The `(unsigned char)` cast before `islower()` avoids undefined behavior for negative `char` values.
   - `malloc` return value is checked, allocation size `len + 1` cannot overflow since `len < MAX_INPUT_LEN`, and the result is freed after use.
   - All error paths report to `stderr` and return `EXIT_FAILURE`.
