```c
#include <stdio.h>
#include <ctype.h>
#include <stdlib.h>
#include <string.h>

void remove_non_alnum(char *str)
{
    size_t read_idx;
    size_t write_idx;

    if (str == NULL) {
        return;
    }

    read_idx = 0;
    write_idx = 0;

    while (str[read_idx] != '\0') {
        if (isalnum((unsigned char)str[read_idx])) {
            str[write_idx] = str[read_idx];
            write_idx++;
        }
        read_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    const char *input = "Hello, World! 123 #$% foo_bar";
    size_t input_len;
    char *buffer;
    int written;

    input_len = strlen(input);

    buffer = malloc(input_len + 1);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    written = snprintf(buffer, input_len + 1, "%s", input);
    if (written < 0 || (size_t)written > input_len) {
        fprintf(stderr, "Input too long or encoding error\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    printf("Before: %s\n", buffer);

    remove_non_alnum(buffer);

    printf("After:  %s\n", buffer);

    free(buffer);

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was already quite defensive, but the Flawfinder warning pointed at the statically-sized `char buffer[256]`. The rewrite addresses it as follows:

1. **Dynamic allocation instead of a fixed-size array** — the buffer is sized exactly from `strlen(input) + 1`, so the "improperly restricted fixed-size buffer" concern (CWE-119/CWE-120) is eliminated. No input can exceed the buffer because the buffer is derived from the input's actual length.

2. **`snprintf` with explicit bounds retained** — even though the buffer is exactly sized, `snprintf` is kept with a size limit and its return value is validated for both encoding errors (`< 0`) and truncation (`> input_len`), providing defense in depth.

3. **Allocation failure handling** — `malloc` return value is checked, with a clean error path.

4. **No memory leaks** — `free(buffer)` is called on both the error path after allocation and the success path.

5. **Existing safe practices preserved** — the `isalnum((unsigned char)...)` cast remains, which correctly avoids undefined behavior for negative `char` values, and `remove_non_alnum` still guards against `NULL` and can never write beyond the original string length (write index never exceeds read index).
