```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    char *first;
    char *last;
    size_t tail_len;

    if (str == NULL) {
        return;
    }

    len = strnlen(str, max_len);
    if (len == 0U || len >= max_len) {
        return;
    }

    first = memchr(str, (int)ch, len);
    if (first == NULL) {
        return;
    }

    last = strrchr(str, (int)ch);

    if (last != NULL && last != first) {
        tail_len = len - (size_t)(last - str) - 1U;
        memmove(last, last + 1, tail_len + 1U);
        len -= 1U;
    }

    tail_len = len - (size_t)(first - str) - 1U;
    memmove(first, first + 1, tail_len + 1U);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char target;
    int input_char;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }
    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
    }

    printf("Enter the character to remove: ");
    input_char = getchar();
    if (input_char == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    target = (char)input_char;

    remove_first_last(buffer, sizeof(buffer), target);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (over-read via `strlen`)**: Replaced all `strlen` calls with `strnlen` bounded by the buffer size, and passed `max_len` into `remove_first_last` so the function never scans past the caller's buffer. Guaranteed `\0`-termination explicitly after `fgets` as a defensive measure.

2. **Redundant scans eliminated**: The tail lengths for the `memmove` calls are now computed arithmetically from the already-known string length and pointer offsets, rather than re-scanning with `strlen(last + 1)` / `strlen(first + 1)`. This is both safer and more efficient (O(n) instead of multiple passes).

3. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is defined once via `BUFFER_SIZE`, `fgets` is bounded with `sizeof(buffer)`, and the length check `len >= max_len` in `remove_first_last` rejects any string that isn't provably terminated within bounds.

4. **CWE-120/CWE-20 (`getchar`)**: The single `getchar` call is not in a loop and its `EOF` result is checked before use, so no further mitigation is required.

5. **Used `memchr` for the first occurrence search**: bounded by the validated length, avoiding any reliance on termination for the forward scan; `strrchr` remains safe because termination within `max_len` has already been verified.
