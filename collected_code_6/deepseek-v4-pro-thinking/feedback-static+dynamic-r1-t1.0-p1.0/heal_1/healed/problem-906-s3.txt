#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int extract_date_from_url(const char *url, int *year, int *month, int *day) {
    regex_t regex;
    regmatch_t matches[4];
    char *endptr;
    long y, m, d;
    int ret;

    if (url == NULL || year == NULL || month == NULL || day == NULL) {
        return 0;
    }

    ret = regcomp(&regex, "([0-9]{4})[-/]([0-9]{2})[-/]([0-9]{2})", REG_EXTENDED);
    if (ret != 0) {
        return 0;
    }

    ret = regexec(&regex, url, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return 0;
    }

    if (matches[1].rm_so < 0 || matches[2].rm_so < 0 || matches[3].rm_so < 0) {
        regfree(&regex);
        return 0;
    }

    y = strtol(url + matches[1].rm_so, &endptr, 10);
    if (endptr != url + matches[1].rm_eo) {
        regfree(&regex);
        return 0;
    }

    m = strtol(url + matches[2].rm_so, &endptr, 10);
    if (endptr != url + matches[2].rm_eo) {
        regfree(&regex);
        return 0;
    }

    d = strtol(url + matches[3].rm_so, &endptr, 10);
    if (endptr != url + matches[3].rm_eo) {
        regfree(&regex);
        return 0;
    }

    if (y < 1 || y > 9999 || m < 1 || m > 12 || d < 1 || d > 31) {
        regfree(&regex);
        return 0;
    }

    *year = (int)y;
    *month = (int)m;
    *day = (int)d;

    regfree(&regex);
    return 1;
}

int main(void) {
    const char *url = "https://example.com/news/2024/12/25/headline";
    int year = 0, month = 0, day = 0;

    if (extract_date_from_url(url, &year, &month, &day)) {
        printf("%04d-%02d-%02d\n", year, month, day);
    } else {
        printf("No date found\n");
    }

    return 0;
}
