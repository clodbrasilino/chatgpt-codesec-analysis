```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

int main(void)
{
    char buffer[64];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged weakness (CWE-119/CWE-120, statically-sized `char buffer[64]`) was already properly mitigated in the original code, and those mitigations are preserved:

1. **Bounded read:** `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow — unlike `gets()` or unbounded `scanf("%s", ...)`.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is defensive; `fgets` already null-terminates, but this costs nothing and protects against future refactoring.
3. **Zero-initialization:** `memset` ensures no uninitialized reads if the logic changes later.

Other robustness points already handled correctly:

- **Integer overflow on `n - 1`:** guarded by the `n == INT_MIN` check in `closest_smaller`, avoiding signed underflow (undefined behavior).
- **Safe parsing:** `strtol` with `errno`/`ERANGE` checks, `endptr` validation (rejecting empty and trailing-garbage input), and explicit range clamping before the narrowing cast to `int` — important on platforms where `long` is wider than `int`.
- **Checked I/O:** return values of `fgets` and `printf` are verified.
