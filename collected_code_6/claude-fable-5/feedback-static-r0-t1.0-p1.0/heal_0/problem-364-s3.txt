#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static int count_flips(const char *str, size_t len, char start)
{
    int flips = 0;
    char expected = start;
    size_t i;

    for (i = 0; i < len; i++) {
        if (str[i] != expected) {
            flips++;
        }
        expected = (expected == '0') ? '1' : '0';
    }
    return flips;
}

int min_flips_to_alternate(const char *str, int *result)
{
    size_t len;
    size_t i;
    int flips_start_zero;
    int flips_start_one;

    if (str == NULL || result == NULL) {
        return -1;
    }

    len = strlen(str);
    if (len == 0) {
        *result = 0;
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (str[i] != '0' && str[i] != '1') {
            return -1;
        }
    }

    flips_start_zero = count_flips(str, len, '0');
    flips_start_one = count_flips(str, len, '1');

    *result = (flips_start_zero < flips_start_one) ? flips_start_zero : flips_start_one;
    return 0;
}

int main(void)
{
    const char *tests[] = {"0001010111", "001", "0000", "1010", "0110"};
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    int result;

    for (i = 0; i < num_tests; i++) {
        if (min_flips_to_alternate(tests[i], &result) != 0) {
            fprintf(stderr, "Invalid input: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("String: %s -> Flips required: %d\n", tests[i], result) < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
