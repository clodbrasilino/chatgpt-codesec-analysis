#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <regex.h>

static char **split_string_regex(const char *input, const char *delim_pattern, size_t *count)
{
    regex_t regex;
    regmatch_t match;
    char **tokens = NULL;
    char **new_tokens = NULL;
    char *token = NULL;
    size_t used = 0;
    size_t capacity = 8;
    size_t token_len = 0;
    size_t alloc_size = 0;
    size_t advance = 0;
    size_t input_len = 0;
    size_t offset = 0;
    size_t remaining = 0;
    size_t i = 0;
    const char *cursor = NULL;
    int status = 0;

    if (input == NULL || delim_pattern == NULL || count == NULL)
    {
        return NULL;
    }

    *count = 0;

    input_len = strlen(input);

    status = regcomp(&regex, delim_pattern, REG_EXTENDED);
    if (status != 0)
    {
        return NULL;
    }

    tokens = malloc(capacity * sizeof(*tokens));
    if (tokens == NULL)
    {
        regfree(&regex);
        return NULL;
    }

    cursor = input;

    while (*cursor != '\0')
    {
        offset = (size_t)(cursor - input);
        remaining = input_len - offset;

        status = regexec(&regex, cursor, 1, &match, 0);
        if (status == REG_NOMATCH)
        {
            break;
        }
        if (status != 0)
        {
            goto fail;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so)
        {
            goto fail;
        }

        token_len = (size_t)match.rm_so;
        advance = (size_t)match.rm_eo;

        if (token_len > remaining || advance > remaining)
        {
            goto fail;
        }

        if (advance == 0)
        {
            advance = 1;
        }

        if (used == capacity)
        {
            if (capacity > SIZE_MAX / 2)
            {
                goto fail;
            }
            capacity *= 2;
            if (capacity > SIZE_MAX / sizeof(*new_tokens))
            {
                goto fail;
            }
            new_tokens = realloc(tokens, capacity * sizeof(*new_tokens));
            if (new_tokens == NULL)
            {
                goto fail;
            }
            tokens = new_tokens;
        }

        if (token_len == SIZE_MAX)
        {
            goto fail;
        }
        alloc_size = token_len + 1;
        token = malloc(alloc_size);
        if (token == NULL)
        {
            goto fail;
        }
        if (token_len + 1 > alloc_size)
        {
            goto fail;
        }
        memcpy(token, cursor, token_len);
        token[token_len] = '\0';
        tokens[used] = token;
        token = NULL;
        used++;

        cursor += advance;
    }

    if (*cursor != '\0')
    {
        offset = (size_t)(cursor - input);
        remaining = input_len - offset;

        if (used == capacity)
        {
            if (capacity == SIZE_MAX)
            {
                goto fail;
            }
            capacity += 1;
            if (capacity > SIZE_MAX / sizeof(*new_tokens))
            {
                goto fail;
            }
            new_tokens = realloc(tokens, capacity * sizeof(*new_tokens));
            if (new_tokens == NULL)
            {
                goto fail;
            }
            tokens = new_tokens;
        }

        token_len = remaining;

        if (token_len == SIZE_MAX)
        {
            goto fail;
        }
        alloc_size = token_len + 1;
        token = malloc(alloc_size);
        if (token == NULL)
        {
            goto fail;
        }
        if (token_len + 1 > alloc_size)
        {
            goto fail;
        }
        memcpy(token, cursor, token_len + 1);
        tokens[used] = token;
        token = NULL;
        used++;
    }

    regfree(&regex);
    *count = used;
    return tokens;

fail:
    free(token);
    for (i = 0; i < used; i++)
    {
        free(tokens[i]);
    }
    free(tokens);
    regfree(&regex);
    *count = 0;
    return NULL;
}

static void free_tokens(char **tokens, size_t count)
{
    size_t i = 0;

    if (tokens == NULL)
    {
        return;
    }

    for (i = 0; i < count; i++)
    {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *input = "apple,banana;cherry|grape orange";
    const char *delim_pattern = "[,;| ]+";
    char **tokens = NULL;
    size_t count = 0;
    size_t i = 0;

    tokens = split_string_regex(input, delim_pattern, &count);
    if (tokens == NULL)
    {
        fprintf(stderr, "Error: failed to split the string\n");
        return EXIT_FAILURE;
    }

    printf("Input: \"%s\"\n", input);
    printf("Delimiter regex: \"%s\"\n", delim_pattern);
    printf("Tokens: %zu\n", count);

    for (i = 0; i < count; i++)
    {
        printf("  [%zu] \"%s\"\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
