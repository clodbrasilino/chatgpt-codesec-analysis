#include <stdio.h>
#include <stdlib.h>

double calculate_triangular_prism_volume(double base, double height, double length) {
    if (base < 0.0 || height < 0.0 || length < 0.0) {
        return -1.0;
    }
    return 0.5 * base * height * length;
}

int main(void) {
    double base = 5.0;
    double height = 4.0;
    double length = 10.0;
    
    double volume = calculate_triangular_prism_volume(base, height, length);
    
    if (volume < 0.0) {
        fprintf(stderr, "Error: Invalid dimensions provided.\n");
        return EXIT_FAILURE;
    }
    
    printf("Volume: %.2f\n", volume);
    
    return EXIT_SUCCESS;
}
