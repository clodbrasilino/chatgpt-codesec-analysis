#include<stdio.h>
#include<string.h>
#include<stdlib.h>
#include<regex.h>

#define MAX_MATCHES 100

typedef struct {
    char** array;
    size_t used;
    size_t size;
} Array;

void initArray(Array* a, size_t initialSize) {
    if(a == NULL)
        return;
    
    a->array = malloc(initialSize * sizeof(char*));
    a->used = 0;
    a->size = initialSize;
    
    if(a->array == NULL) 
        exit(1);
}

void insertArray(Array* a, char* element) {
    if(a == NULL)
        return;
        
    if (a->used == a->size) {
        a->size *= 2;
        a->array = realloc(a->array, a->size * sizeof(char*));
        
        if(a->array == NULL) 
            exit(1);
    }
    a->array[a->used++] = element;
}

void freeArray(Array* a) {
    if(a == NULL)
        return;
        
    free(a->array);
    a->array = NULL;
    a->used = a->size = 0;
}

Array* regex_split(const char* string, const char* regex_pattern) {
    if(string == NULL || regex_pattern == NULL)
        return NULL;
        
    regmatch_t matches[MAX_MATCHES];
    regex_t regex_compiled;
    int result;
    const char* p = string;
    size_t offset = 0;
    Array *arr = malloc(sizeof(Array));
    
    if(arr == NULL) 
        return NULL;
    
    initArray(arr, 1);

    if (regcomp(&regex_compiled, regex_pattern, REG_EXTENDED)) {
        printf("Could not compile regular expression.\n");
        free(arr);
        return NULL;
    }
    
    while (1) {
        result = regexec(&regex_compiled, p, MAX_MATCHES, matches, 0);
        if (result != 0)
            break;
        
        if (matches[0].rm_so != 0) {
            char* new_str = malloc(matches[0].rm_so + 1);
            
            if(new_str == NULL) {
                freeArray(arr);
                free(arr);
                regfree(&regex_compiled);
                return NULL;
            }
            
            memcpy(new_str, p, matches[0].rm_so);
            new_str[matches[0].rm_so] = '\0';
            insertArray(arr, new_str);
        }

        p += matches[0].rm_eo;
    }

    regfree(&regex_compiled);
    return arr;
}


int main(int argc, char**argv)
{
    Array* arr = regex_split("TextColor=Yellow", "[=]");
    
    if(arr != NULL) {
        for (int i=0; i<arr->used; i++) {
            printf("%s\n", arr->array[i]);
            free(arr->array[i]); 
        }
        freeArray(arr);
        free(arr);
    }
    
    return 0;
}
