#include <stdio.h>

float find_negative_ratio(const int *array, size_t size) {
    if (array == NULL || size == 0) {
        return 0.0f;
    }

    size_t negative_count = 0;

    for (size_t i = 0; i < size; ++i) {
        if (array[i] < 0) {
            negative_count++;
        }
    }

    return (float)negative_count / (float)size;
}

int main(void) {
    int numbers[] = {1, -2, 3, -4, 5, -6};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);

    float ratio = find_negative_ratio(numbers, count);

    printf("%f\n", ratio);

    return 0;
}
