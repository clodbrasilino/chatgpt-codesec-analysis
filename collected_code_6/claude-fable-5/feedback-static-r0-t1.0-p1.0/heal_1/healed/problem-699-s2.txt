```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int diff10 = 0;
    int diff01 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN);
    len2 = strnlen(s2, MAX_INPUT_LEN);

    if (len1 >= MAX_INPUT_LEN || len2 >= MAX_INPUT_LEN) {
        return -1;
    }

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') || (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] == '1' && s2[i] == '0') {
            diff10++;
        } else if (s1[i] == '0' && s2[i] == '1') {
            diff01++;
        }
    }

    if (diff10 != diff01) {
        return -1;
    }

    return diff10;
}

int main(void)
{
    const char *str1 = "1101";
    const char *str2 = "0011";
    int result;

    result = min_swaps(str1, str2);

    if (result < 0) {
        if (printf("Conversion is not possible\n") < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("Minimum swaps required: %d\n", result) < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the CWE-126 (buffer over-read) weakness flagged by Flawfinder: `strlen` was replaced with `strnlen`, which bounds the scan to `MAX_INPUT_LEN` bytes. If a string is not properly `\0`-terminated within that limit, the function now returns `-1` safely instead of reading past the end of the buffer. The rest of the logic was already defensive (NULL checks, character validation, length comparison) and is preserved unchanged.
