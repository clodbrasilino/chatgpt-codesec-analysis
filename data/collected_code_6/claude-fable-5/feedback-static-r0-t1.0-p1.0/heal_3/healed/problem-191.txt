```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 32

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    for (i = 0; i < len; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No month name provided.\n");
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char normalized[MAX_INPUT_LEN]` in `has_thirty_days` (CWE-119/CWE-120):** This is a false positive in practice. The function uses `strnlen(month, MAX_INPUT_LEN)` to bound the length, rejects inputs where `len >= MAX_INPUT_LEN`, and only then writes `len` characters plus a null terminator. Since `len <= MAX_INPUT_LEN - 1`, the write of `normalized[len] = '\0'` stays in bounds. No overflow is possible.

2. **`getchar()` in a loop (CWE-120/CWE-20):** The loop discards characters without storing them into any buffer, so there is no buffer to overflow. The loop terminates correctly on both `'\n'` and `EOF`, avoiding an infinite loop if stdin is closed.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Input is read with `fgets(input, sizeof(input), stdin)`, which guarantees null termination and never writes beyond the buffer. Oversized lines are detected (no trailing newline while the buffer is full) and the remainder is drained with `discard_remaining_input()`, preventing stale input from leaking into subsequent reads.

The Flawfinder hits are heuristic warnings triggered by the mere presence of fixed-size arrays and `getchar()`. The original code already applies the recommended mitigations (bounded reads via `fgets`/`strnlen`, explicit length checks, guaranteed null termination), so the rewritten version preserves that logic verbatim with the audit comments removed.
