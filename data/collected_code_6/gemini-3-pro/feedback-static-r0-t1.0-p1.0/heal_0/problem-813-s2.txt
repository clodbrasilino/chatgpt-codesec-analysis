#include <stdio.h>
#include <stddef.h>

size_t string_length(const char *str);

size_t string_length(const char *str) {
    size_t length = 0;
    
    if (str == NULL) {
        return 0;
    }
    
    while (str[length] != '\0') {
        length++;
    }
    
    return length;
}

int main(void) {
    const char *test_string = "Hello, World!";
    size_t length;

    length = string_length(test_string);
    printf("%zu\n", length);

    length = string_length(NULL);
    printf("%zu\n", length);

    return 0;
}
