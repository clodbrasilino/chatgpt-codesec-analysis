#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

int is_even_hex(const char *hex) {
    if (hex == NULL) {
        return -1;
    }

    while (isspace((unsigned char)*hex)) {
        hex++;
    }

    if (*hex == '+' || *hex == '-') {
        hex++;
    }

    if (hex[0] == '0' && (hex[1] == 'x' || hex[1] == 'X')) {
        hex += 2;
    }

    if (*hex == '\0') {
        return -1;
    }

    const char *last_digit = NULL;

    while (*hex != '\0') {
        if (isxdigit((unsigned char)*hex)) {
            last_digit = hex;
            hex++;
        } else if (isspace((unsigned char)*hex)) {
            break;
        } else {
            return -1;
        }
    }

    while (isspace((unsigned char)*hex)) {
        hex++;
    }

    if (*hex != '\0') {
        return -1;
    }

    if (last_digit == NULL) {
        return -1;
    }

    char c = *last_digit;

    if (c >= 'A' && c <= 'F') {
        c = (char)(c - 'A' + 'a');
    }

    if (c == '0' || c == '2' || c == '4' || c == '6' || c == '8' ||
        c == 'a' || c == 'c' || c == 'e') {
        return 1;
    }

    return 0;
}

int main(int argc, const char * const argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <hex>\n", argv[0]);
        return EXIT_FAILURE;
    }

    int result = is_even_hex(argv[1]);

    if (result == -1) {
        fprintf(stderr, "Invalid hexadecimal number\n");
        return EXIT_FAILURE;
    }

    printf("%s\n", result == 1 ? "even" : "odd");

    return EXIT_SUCCESS;
}
