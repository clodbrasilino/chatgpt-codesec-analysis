#include <stdio.h>
#include <stdbool.h>
#include <ctype.h>

bool is_integer(const char *str) {
    if (str == NULL) {
        return false;
    }

    if (*str == '\0') {
        return false;
    }

    if (*str == '+' || *str == '-') {
        str++;
        if (*str == '\0') {
            return false;
        }
    }

    while (*str != '\0') {
        if (!isdigit((unsigned char)*str)) {
            return false;
        }
        str++;
    }

    return true;
}

int main(void) {
    const char *test_cases[] = {
        "123",
        "-456",
        "+789",
        "0",
        "12a3",
        "--5",
        "++4",
        "",
        "+",
        "-",
        " 123",
        "123 ",
        NULL
    };

    size_t num_cases = sizeof(test_cases) / sizeof(test_cases[0]);

    for (size_t i = 0; i < num_cases; i++) {
        if (test_cases[i] == NULL) {
            printf("NULL: false\n");
        } else {
            printf("\"%s\": %s\n", test_cases[i], is_integer(test_cases[i]) ? "true" : "false");
        }
    }

    return 0;
}
