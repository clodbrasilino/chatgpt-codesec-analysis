#include <stdio.h>
#include <stdlib.h>

int findOddOccurringElement(const int *arr, size_t size, int *result) {
    if (arr == NULL || size == 0 || result == NULL) {
        return -1;
    }

    int oddElement = 0;
    for (size_t i = 0; i < size; ++i) {
        oddElement ^= arr[i];
    }

    *result = oddElement;
    return 0;
}

int main(void) {
    int arr[] = {2, 3, 5, 4, 5, 2, 4, 3, 5, 2, 4, 4, 2};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int result = 0;

    if (findOddOccurringElement(arr, size, &result) == 0) {
        printf("%d\n", result);
    } else {
        printf("Invalid input.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
