#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
} Pair;

typedef struct {
    int element;
    int pair_element;
} PairResult;

int assign_pair_elements(const Pair *pairs, size_t pair_count, PairResult **results, size_t *result_count)
{
    size_t i;
    size_t j;
    size_t idx;
    PairResult *temp_results;

    if (pairs == NULL || results == NULL || result_count == NULL) {
        return -1;
    }

    if (pair_count == 0) {
        *results = NULL;
        *result_count = 0;
        return 0;
    }

    if (pair_count > SIZE_MAX / (2 * sizeof(PairResult))) {
        return -1;
    }

    temp_results = (PairResult *)malloc(pair_count * 2 * sizeof(PairResult));
    if (temp_results == NULL) {
        return -1;
    }

    idx = 0;
    for (i = 0; i < pair_count; i++) {
        for (j = 0; j < pair_count; j++) {
            if (i != j) {
                if (pairs[i].first == pairs[j].first) {
                    temp_results[idx].element = pairs[i].first;
                    temp_results[idx].pair_element = pairs[j].second;
                    idx++;
                }
                if (pairs[i].first == pairs[j].second) {
                    temp_results[idx].element = pairs[i].first;
                    temp_results[idx].pair_element = pairs[j].first;
                    idx++;
                }
                if (pairs[i].second == pairs[j].first) {
                    temp_results[idx].element = pairs[i].second;
                    temp_results[idx].pair_element = pairs[j].second;
                    idx++;
                }
                if (pairs[i].second == pairs[j].second) {
                    temp_results[idx].element = pairs[i].second;
                    temp_results[idx].pair_element = pairs[j].first;
                    idx++;
                }
            }
        }
    }

    *results = temp_results;
    *result_count = idx;
    return 0;
}

int main(void)
{
    Pair pairs[] = {{1, 2}, {2, 3}, {4, 5}, {1, 6}};
    size_t pair_count = sizeof(pairs) / sizeof(pairs[0]);
    PairResult *results = NULL;
    size_t result_count = 0;
    size_t i;
    int status;

    status = assign_pair_elements(pairs, pair_count, &results, &result_count);
    if (status != 0) {
        fprintf(stderr, "Error: Failed to assign pair elements\n");
        return EXIT_FAILURE;
    }

    printf("Element -> Pair Element\n");
    for (i = 0; i < result_count; i++) {
        printf("%d -> %d\n", results[i].element, results[i].pair_element);
    }

    free(results);
    results = NULL;

    return EXIT_SUCCESS;
}
