```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_multiple_spaces(const char *input);

char *remove_multiple_spaces(const char *input)
{
    size_t input_len;
    size_t i;
    size_t pos;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    input_len = 0;
    while (input_len < MAX_INPUT_LEN && input[input_len] != '\0') {
        input_len++;
    }

    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    pos = 0;
    i = 0;

    while (i < input_len) {
        if (isspace((unsigned char)input[i])) {
            size_t start = i;

            while (i < input_len && isspace((unsigned char)input[i])) {
                i++;
            }

            if (pos >= input_len) {
                free(result);
                return NULL;
            }

            if (i - start > 1) {
                result[pos] = ' ';
            } else {
                result[pos] = input[start];
            }
            pos++;
        } else {
            if (pos >= input_len) {
                free(result);
                return NULL;
            }
            result[pos] = input[i];
            pos++;
            i++;
        }
    }

    result[pos] = '\0';
    return result;
}

int main(void)
{
    const char *tests[] = {
        "Google    Assistant",
        "Quad \t Core",
        "ChromeCast   Built-in",
        "Hello    world,   this  is    a     test."
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < num_tests; i++) {
        char *cleaned = remove_multiple_spaces(tests[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string\n");
            return EXIT_FAILURE;
        }
        if (printf("Original: %s\n", tests[i]) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        if (printf("Cleaned:  %s\n", cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The root cause of the "no output" failures was the use of `strnlen()`. While the algorithm itself was logically sound, `strnlen()` is a POSIX extension rather than standard C (it was only added to ISO C in the optional Annex K as `strnlen_s`). On strict ANSI/ISO compilation environments (e.g., `gcc -std=c99` without `_POSIX_C_SOURCE` defined, or non-POSIX toolchains), the program fails to compile or link, which explains why the test harness received no output at all rather than incorrect output.

The fix replaces `strnlen()` with a portable bounded length scan that preserves the original safety guarantee: it never reads past `MAX_INPUT_LEN` bytes, and inputs of `MAX_INPUT_LEN` or longer are still rejected with `NULL`, so the buffer overflow protections remain intact.

The rest of the logic was retained as-is since it is correct:

- `isspace()` is properly called with an `unsigned char` cast, avoiding undefined behavior for negative `char` values.
- The output buffer can never exceed the input length (whitespace runs only shrink), so `malloc(input_len + 1)` is sufficient and the defensive `pos >= input_len` checks are unreachable-but-harmless guards.
- All allocation paths are freed on error in `main`, so there are no leaks.
