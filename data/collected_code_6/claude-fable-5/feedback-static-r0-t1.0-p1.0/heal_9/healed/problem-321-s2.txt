#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 9
#define BUFFER_SIZE 64

int find_demlo(const char *input, char *output, size_t output_size);

int find_demlo(const char *input, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, (size_t)MAX_INPUT_LEN + 1U);

    if (len == 0U || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    for (i = len; i > 1U; i--)
    {
        output[pos] = (char)('0' + (char)(i - 1U));
        pos++;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';
    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;
        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char input[BUFFER_SIZE]` (CWE-119/CWE-120):** This is a false positive in practice. The buffer is zero-initialized with `memset`, populated exclusively via `fgets(input, sizeof(input), stdin)` which enforces the bound, and defensively null-terminated at `input[sizeof(input) - 1]`. No overflow is possible.

2. **`char result[BUFFER_SIZE]` (CWE-119/CWE-120):** Also safe. `find_demlo` receives the buffer size explicitly and validates `output_size >= 2 * len` before writing. The maximum write is `2 * len - 1` data bytes plus the null terminator at index `2 * len - 1`, which fits exactly within the checked bound. With `MAX_INPUT_LEN = 9`, the worst case needs 18 bytes, well under 64.

3. **`getchar()` in a loop (CWE-120/CWE-20):** This is the standard idiom for draining stdin when input exceeds the buffer. Nothing is stored, so there is no buffer to overflow; the loop terminates on `'\n'` or `EOF`. This is a false positive.

Additional defensive properties already present: NULL-pointer checks on all `find_demlo` parameters, `strnlen` instead of `strlen` to bound length scans, strict input validation (repunit digits only, length 1–9), rejection of oversized input via the `MAX_INPUT_LEN + 1` sentinel in `strnlen`, and checked return values for `fgets` and `find_demlo`. The code is correct as written; no functional changes were required.
