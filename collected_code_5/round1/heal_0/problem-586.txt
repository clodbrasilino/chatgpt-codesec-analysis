#include<stdio.h>

void splitArrayAt(int arr[], int n, int k) {
    int i;
    for (i = 0; i < k; i++) {
        int temp = arr[0];

        for (int j = 0; j < n-1; j++) {
            arr[j] = arr[j + 1];
        }
        
        arr[n-1] = temp;
    }
}

int main() {

    int arr[] = {12, 10, 5, 6, 52, 36};
    int n = sizeof(arr) / sizeof(arr[0]);
    int position = 2;

    splitArrayAt(arr, n, position);

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}
