#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>

typedef struct Node {
    void *data;
    int type;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    size_t size;
} List;

enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_STRING,
    TYPE_DOUBLE
};

List *list_create(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->size = 0;
    return list;
}

void list_destroy(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        if (current->data != NULL) {
            free(current->data);
        }
        free(current);
        current = next;
    }
    free(list);
}

int list_append(List *list, void *data, int type) {
    if (list == NULL || data == NULL) {
        return -1;
    }
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return -1;
    }
    new_node->data = data;
    new_node->type = type;
    new_node->next = NULL;
    
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->size++;
    return 0;
}

int is_valid_float_string(const char *str) {
    if (str == NULL || *str == '\0') {
        return 0;
    }
    
    char *endptr = NULL;
    errno = 0;
    float val = strtof(str, &endptr);
    
    if (errno == ERANGE) {
        return 0;
    }
    if (endptr == str) {
        return 0;
    }
    while (*endptr != '\0') {
        if (!isspace((unsigned char)*endptr)) {
            return 0;
        }
        endptr++;
    }
    return 1;
}

void convert_to_float(List *list) {
    if (list == NULL) {
        return;
    }
    
    Node *current = list->head;
    while (current != NULL) {
        if (current->data == NULL) {
            current = current->next;
            continue;
        }
        
        switch (current->type) {
            case TYPE_INT: {
                int *int_val = (int *)current->data;
                float *float_val = (float *)malloc(sizeof(float));
                if (float_val == NULL) {
                    current = current->next;
                    continue;
                }
                *float_val = (float)(*int_val);
                free(current->data);
                current->data = float_val;
                current->type = TYPE_FLOAT;
                break;
            }
            case TYPE_DOUBLE: {
                double *double_val = (double *)current->data;
                float *float_val = (float *)malloc(sizeof(float));
                if (float_val == NULL) {
                    current = current->next;
                    continue;
                }
                *float_val = (float)(*double_val);
                free(current->data);
                current->data = float_val;
                current->type = TYPE_FLOAT;
                break;
            }
            case TYPE_STRING: {
                char *str_val = (char *)current->data;
                if (is_valid_float_string(str_val)) {
                    float *float_val = (float *)malloc(sizeof(float));
                    if (float_val == NULL) {
                        current = current->next;
                        continue;
                    }
                    *float_val = strtof(str_val, NULL);
                    free(current->data);
                    current->data = float_val;
                    current->type = TYPE_FLOAT;
                }
                break;
            }
            default:
                break;
        }
        current = current->next;
    }
}

void print_list(const List *list) {
    if (list == NULL) {
        return;
    }
    
    Node *current = list->head;
    while (current != NULL) {
        if (current->data == NULL) {
            printf("NULL ");
        } else {
            switch (current->type) {
                case TYPE_INT:
                    printf("%d ", *(int *)current->data);
                    break;
                case TYPE_FLOAT:
                    printf("%.2f ", *(float *)current->data);
                    break;
                case TYPE_DOUBLE:
                    printf("%.2f ", *(double *)current->data);
                    break;
                case TYPE_STRING:
                    printf("%s ", (char *)current->data);
                    break;
                default:
                    printf("UNKNOWN ");
                    break;
            }
        }
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    List *list = list_create();
    if (list == NULL) {
        return 1;
    }
    
    int *int_val = (int *)malloc(sizeof(int));
    if (int_val == NULL) {
        list_destroy(list);
        return 1;
    }
    *int_val = 42;
    list_append(list, int_val, TYPE_INT);
    
    float *float_val = (float *)malloc(sizeof(float));
    if (float_val == NULL) {
        list_destroy(list);
        return 1;
    }
    *float_val = 3.14f;
    list_append(list, float_val, TYPE_FLOAT);
    
    double *double_val = (double *)malloc(sizeof(double));
    if (double_val == NULL) {
        list_destroy(list);
        return 1;
    }
    *double_val = 2.718281828;
    list_append(list, double_val, TYPE_DOUBLE);
    
    char *str_val1 = (char *)malloc(16 * sizeof(char));
    if (str_val1 == NULL) {
        list_destroy(list);
        return 1;
    }
    strcpy(str_val1, "123.45");
    list_append(list, str_val1, TYPE_STRING);
    
    char *str_val2 = (char *)malloc(16 * sizeof(char));
    if (str_val2 == NULL) {
        list_destroy(list);
        return 1;
    }
    strcpy(str_val2, "not_a_number");
    list_append(list, str_val2, TYPE_STRING);
    
    char *str_val3 = (char *)malloc(16 * sizeof(char));
    if (str_val3 == NULL) {
        list_destroy(list);
        return 1;
    }
    strcpy(str_val3, "  67.89  ");
    list_append(list, str_val3, TYPE_STRING);
    
    printf("Before conversion:\n");
    print_list(list);
    
    convert_to_float(list);
    
    printf("After conversion:\n");
    print_list(list);
    
    list_destroy(list);
    
    return 0;
}
