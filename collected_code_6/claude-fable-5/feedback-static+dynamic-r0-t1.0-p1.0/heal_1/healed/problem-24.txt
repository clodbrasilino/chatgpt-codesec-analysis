```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_BUFFER_SIZE 128U
#define MAX_BINARY_DIGITS (sizeof(unsigned long) * 8U)

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced both `strlen` calls with `strnlen`, which takes an explicit maximum length and cannot read past the buffer even if the string is not NUL-terminated. The `binary_to_decimal` function now receives the buffer size (`max_len`) as a parameter so it can bound its own scan, and rejects input where `len >= max_len` (i.e., no terminator was found within bounds).

2. **CWE-119/CWE-120 (statically-sized array):** The buffer size is defined once as a named constant (`INPUT_BUFFER_SIZE`), `fgets` is bounded by `sizeof(input)`, and a defensive `input[sizeof(input) - 1U] = '\0'` guarantees NUL-termination before any string operation, making all subsequent accesses provably in-bounds.

3. **Retained existing hardening:** The overflow guard (`value > (ULONG_MAX >> 1)`) before each shift, the digit-count cap against `MAX_BINARY_DIGITS`, NULL-pointer checks, and strict `'0'`/`'1'` validation are all preserved.

Note: `strnlen` is POSIX / C23; if you must target strict C89/C99 without POSIX, keep `strlen` but the guaranteed NUL-termination after `fgets` already makes it safe in practice.
