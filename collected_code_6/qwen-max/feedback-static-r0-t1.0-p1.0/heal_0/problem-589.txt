```c
#include <stdio.h>
#include <math.h>

void findPerfectSquares(int start, int end) {
    for (int i = start; i <= end; ++i) {
        int sqrtVal = sqrt(i);
        if (sqrtVal * sqrtVal == i) {
            printf("%d ", i);
        }
    }
    printf("\n");
}

int main() {
    int start, end;
    printf("Enter the starting number: ");
    scanf("%d", &start);
    printf("Enter the ending number: ");
    scanf("%d", &end);

    findPerfectSquares(start, end);

    return 0;
}
```
