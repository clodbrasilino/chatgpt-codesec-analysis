#include <stdio.h>
#include <math.h>

typedef struct {
    double x;
    double y;
} Point;

double calculateDistance(Point p1, Point p2) {
    return sqrt(pow(p2.x - p1.x, 2) + pow(p2.y - p1.y, 2));
}

int isValidTriangle(Point p1, Point p2, Point p3) {
    double side1 = calculateDistance(p1, p2);
    double side2 = calculateDistance(p2, p3);
    double side3 = calculateDistance(p3, p1);

    if (side1 <= 0 || side2 <= 0 || side3 <= 0) {
        return 0;
    }

    if ((side1 + side2 > side3) && (side2 + side3 > side1) && (side3 + side1 > side2)) {
        return 1;
    }

    return 0;
}

int main() {
    Point p1, p2, p3;

    printf("Enter coordinates of point 1 (x y): ");
    if (scanf("%lf %lf", &p1.x, &p1.y) != 2) {
        printf("Invalid input for point 1.\n");
        return 1;
    }

    printf("Enter coordinates of point 2 (x y): ");
    if (scanf("%lf %lf", &p2.x, &p2.y) != 2) {
        printf("Invalid input for point 2.\n");
        return 1;
    }

    printf("Enter coordinates of point 3 (x y): ");
    if (scanf("%lf %lf", &p3.x, &p3.y) != 2) {
        printf("Invalid input for point 3.\n");
        return 1;
    }

    if (isValidTriangle(p1, p2, p3)) {
        printf("The points form a valid triangle.\n");
    } else {
        printf("The points do not form a valid triangle.\n");
    }

    return 0;
}
