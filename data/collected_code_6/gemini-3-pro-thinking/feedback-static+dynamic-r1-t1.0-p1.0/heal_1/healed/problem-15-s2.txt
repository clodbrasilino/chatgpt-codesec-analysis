#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char **split_at_lowercase(const char *input, size_t *out_count) {
    if (!input || !out_count) {
        if (out_count) {
            *out_count = 0;
        }
        return NULL;
    }

    size_t count = 0;
    int in_token = 0;

    for (size_t i = 0; input[i] != '\0'; i++) {
        if (!islower((unsigned char)input[i])) {
            if (!in_token) {
                count++;
                in_token = 1;
            }
        } else {
            in_token = 0;
        }
    }

    if (count == 0) {
        *out_count = 0;
        return NULL;
    }

    char **result = (char **)malloc(count * sizeof(char *));
    if (!result) {
        *out_count = 0;
        return NULL;
    }

    size_t current_token = 0;
    const char *token_start = NULL;

    for (size_t i = 0; ; i++) {
        if (input[i] != '\0' && !islower((unsigned char)input[i])) {
            if (!token_start) {
                token_start = &input[i];
            }
        } else {
            if (token_start) {
                size_t len = (size_t)(&input[i] - token_start);
                result[current_token] = (char *)malloc(len + 1);
                if (!result[current_token]) {
                    for (size_t j = 0; j < current_token; j++) {
                        free(result[j]);
                    }
                    free(result);
                    *out_count = 0;
                    return NULL;
                }
                
                for (size_t k = 0; k < len; k++) {
                    result[current_token][k] = token_start[k];
                }
                result[current_token][len] = '\0';
                
                current_token++;
                token_start = NULL;
            }
        }
        if (input[i] == '\0') {
            break;
        }
    }

    *out_count = count;
    return result;
}

int main(void) {
    const char *test_str = "HELLOworldHOWareYOU";
    size_t count = 0;
    char **tokens = split_at_lowercase(test_str, &count);

    if (tokens) {
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", tokens[i]);
            free(tokens[i]);
        }
        free(tokens);
    }

    return 0;
}
