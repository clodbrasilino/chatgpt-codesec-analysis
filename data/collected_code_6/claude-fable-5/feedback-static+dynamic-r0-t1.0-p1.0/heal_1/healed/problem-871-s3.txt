int are_rotations(const char *s1, const char *s2)

    size_t len1;
    size_t len2;
    size_t total;
    char *temp;
    int result;

    if (s1 == NULL || s2 == NULL) {
        return 0;
    }

    len1 = strnlen(s1, MAX_INPUT_LEN);
    len2 = strnlen(s2, MAX_INPUT_LEN);

        return -1;

        return 0;
    }

        return 1;

        return -1;

}
