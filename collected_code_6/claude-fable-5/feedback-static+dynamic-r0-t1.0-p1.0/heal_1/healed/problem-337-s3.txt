```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096

bool matches_word_at_end(const char *str, const char *word)
{
    size_t str_len;
    size_t word_len;
    size_t end;

    if (str == NULL || word == NULL) {
        return false;
    }

    str_len = strnlen(str, MAX_INPUT_LEN);
    word_len = strnlen(word, MAX_INPUT_LEN);

    if (str_len >= MAX_INPUT_LEN || word_len >= MAX_INPUT_LEN) {
        return false;
    }

    if (word_len == 0 || str_len < word_len) {
        return false;
    }

    end = str_len;
    while (end > 0 && ispunct((unsigned char)str[end - 1])) {
        end--;
    }

    if (end < word_len) {
        return false;
    }

    if (strncmp(str + (end - word_len), word, word_len) != 0) {
        return false;
    }

    if (end > word_len) {
        unsigned char before = (unsigned char)str[end - word_len - 1];
        if (isalnum(before) || before == '_') {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *tests[][2] = {
        { "Hello world", "world" },
        { "Hello world!", "world" },
        { "Hello world...", "world" },
        { "Hello worldly", "world" },
        { "world", "world" },
        { "Say hello", "world" },
        { "", "world" }
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        bool result = matches_word_at_end(tests[i][0], tests[i][1]);
        if (printf("\"%s\" ends with \"%s\": %s\n",
                   tests[i][0], tests[i][1],
                   result ? "yes" : "no") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Key changes made:**

- Replaced both `strlen` calls with `strnlen`, bounded by a `MAX_INPUT_LEN` constant. This addresses the CWE-126 (buffer over-read) concern flagged by Flawfinder: if a non-NUL-terminated string is passed in, the read is now capped at a known maximum rather than running off the end of the buffer.
- Added a guard rejecting inputs whose measured length hits the bound (`>= MAX_INPUT_LEN`), since such a result means the string may be unterminated or truncated, and proceeding would be unsafe or produce incorrect results.
- All other defensive measures from the original were preserved: NULL-pointer checks, casting to `unsigned char` before calling `ctype.h` functions (avoiding UB with negative `char` values), and checking the return value of `printf`.

**Note:** `strnlen` is POSIX (and available in C11 via Annex K as `strnlen_s`); on strict C90/C99 toolchains without POSIX, a small portable fallback loop would be needed. The fundamental limitation remains that C cannot verify termination of an arbitrary `const char *` — bounding the scan is the standard mitigation.
