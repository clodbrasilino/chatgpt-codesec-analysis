#include <stdio.h>

double power(double a, int b){
    double result = 1.0; 
    int i = 0;
    
    if(b < 0) {
        for(i = 0; i < -b; i++)
            result *= a;
        if(a!=0) {
            return 1/result;
        }
        else {
            printf("Error: Division by zero\n");
            return -1;
        }
    }
    else {
        for(i = 0; i < b; i++)
            result *= a;
        return result;
    }
}

int main() {
    double a;
    int b;

    printf("Enter base: ");
    if(scanf("%lf", &a) != 1) {
        printf("Error: Invalid input\n");
        return -1;
    }

    printf("Enter exponent: ");
    if(scanf("%d", &b) != 1) {
        printf("Error: Invalid input\n");
        return -1;
    }
      
    double result = power(a, b);

    if(result!=-1)
        printf("Result: %lf\n", result);
    return 0;
}
