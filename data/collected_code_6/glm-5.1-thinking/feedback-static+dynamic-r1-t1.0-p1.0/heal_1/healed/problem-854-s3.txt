#include <stdio.h>
#include <stddef.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void sift_down(int *arr, size_t start, size_t end) {
    size_t root = start;
    size_t child;
    
    while ((child = 2 * root + 1) <= end) {
        size_t swap_idx = root;
        if (arr[swap_idx] < arr[child]) {
            swap_idx = child;
        }
        if (child + 1 <= end && arr[swap_idx] < arr[child + 1]) {
            swap_idx = child + 1;
        }
        if (swap_idx == root) {
            return;
        }
        swap(&arr[root], &arr[swap_idx]);
        root = swap_idx;
    }
}

void heapify(int *arr, size_t n) {
    if (arr == NULL || n <= 1) {
        return;
    }
    size_t start = (n - 2) / 2;
    while (1) {
        sift_down(arr, start, n - 1);
        if (start == 0) {
            break;
        }
        start--;
    }
}

int main(void) {
    int arr[] = {4, 10, 3, 5, 1, 8, 7, 2, 9};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    
    heapify(arr, n);
    
    for (size_t i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
