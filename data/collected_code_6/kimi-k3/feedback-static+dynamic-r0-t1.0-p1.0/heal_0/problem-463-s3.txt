#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProductSubarray(int* nums, int numsSize) {
    if (nums == NULL || numsSize <= 0) {
        return 0;
    }

    int maxProduct = nums[0];
    int currentMax = nums[0];
    int currentMin = nums[0];

    for (int i = 1; i < numsSize; i++) {
        if (nums[i] < 0) {
            int temp = currentMax;
            currentMax = currentMin;
            currentMin = temp;
        }

        currentMax = (nums[i] > currentMax * nums[i]) ? nums[i] : currentMax * nums[i];
        currentMin = (nums[i] < currentMin * nums[i]) ? nums[i] : currentMin * nums[i];

        maxProduct = (currentMax > maxProduct) ? currentMax : maxProduct;
    }

    return maxProduct;
}

int main() {
    int nums[] = {2, 3, -2, 4};
    int numsSize = sizeof(nums) / sizeof(nums[0]);

    int result = maxProductSubarray(nums, numsSize);
    printf("Maximum product subarray: %d\n", result);

    return 0;
}
