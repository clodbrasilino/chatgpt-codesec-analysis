```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int pivot;
    int swap_idx;
    char tmp;
    int lo;
    int hi;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    pivot = -1;
    for (i = len - 2; i >= 0; i--) {
        if (digits[i] < digits[i + 1]) {
            pivot = i;
            break;
        }
    }

    if (pivot == -1) {
        return -1;
    }

    swap_idx = -1;
    for (i = len - 1; i > pivot; i--) {
        if (digits[i] > digits[pivot]) {
            swap_idx = i;
            break;
        }
    }

    if (swap_idx == -1) {
        return -1;
    }

    if (pivot < 0 || pivot >= DIGITS_BUF_SIZE ||
        swap_idx < 0 || swap_idx >= DIGITS_BUF_SIZE) {
        return -1;
    }

    tmp = digits[pivot];
    digits[pivot] = digits[swap_idx];
    digits[swap_idx] = tmp;

    lo = pivot + 1;
    hi = len - 1;
    while (lo < hi && lo >= 0 && hi < DIGITS_BUF_SIZE) {
        tmp = digits[lo];
        digits[lo] = digits[hi];
        digits[hi] = tmp;
        lo++;
        hi--;
    }

    result = 0;
    for (i = 0; i < len && i < DIGITS_BUF_SIZE; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        long long res = next_bigger(inputs[i]);
        if (res == -1) {
            if (printf("%lld -> no bigger number possible\n", inputs[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("%lld -> %lld\n", inputs[i], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fixes applied:**

The flawfinder warning (CWE-119/CWE-120) about the statically-sized `digits` array was already largely mitigated in the original code, and the rewrite preserves and strengthens those mitigations:

1. **Bounded write via `snprintf`**: The buffer is written using `snprintf(digits, sizeof(digits), ...)`, which cannot overflow. The return value is checked both for encoding errors (`len < 0`) and truncation (`(size_t)len >= sizeof(digits)`), so any input that would exceed the buffer is rejected. Since `long long` maxes out at 19 digits plus a sign and NUL terminator, `DIGITS_BUF_SIZE` of 32 is comfortably sufficient.

2. **Zero-initialization**: `memset` clears the buffer before use, so no uninitialized reads can occur even in edge cases.

3. **Explicit index bounds checks**: Every index used to access `digits` (`pivot`, `swap_idx`, `lo`, `hi`, loop variable `i`) is validated against `DIGITS_BUF_SIZE` before dereferencing. The rewrite adds `pivot < 0` and `swap_idx < 0` checks for full defensive symmetry.

4. **Integer overflow protection**: The reconstruction loop checks `result > (LLONG_MAX - d) / 10` before each multiply-add, preventing signed overflow (undefined behavior) when the permuted digits would exceed `LLONG_MAX`.

5. **Digit validation**: Each character is verified to be in `'0'`–`'9'` before conversion, guarding against any corruption of the buffer contents.

6. **Negative input rejection**: `n < 0` is rejected up front, so the sign character never enters the digit-permutation logic.

The code is safe with respect to the flagged weakness; the fixed-size buffer's accesses are all bounds-checked, and truncation is detected rather than silently ignored.
