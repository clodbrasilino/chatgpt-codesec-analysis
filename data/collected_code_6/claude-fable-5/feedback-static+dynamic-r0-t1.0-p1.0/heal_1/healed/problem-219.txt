```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int extract_min_max_k(const int *arr, size_t n, size_t k,
                      int *min_out, int *max_out)
{
    int *copy;
    size_t i;
    size_t copy_size;

    if (arr == NULL || min_out == NULL || max_out == NULL) {
        return -1;
    }
    if (n == 0U || k == 0U || k > n) {
        return -1;
    }
    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy_size = n * sizeof(int);

    copy = malloc(copy_size);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, copy_size);
    qsort(copy, n, sizeof(int), compare_ints);

    for (i = 0U; i < k; i++) {
        min_out[i] = copy[i];
        max_out[i] = copy[n - k + i];
    }

    free(copy);
    return 0;
}

int main(void)
{
    int data[] = { 5, 20, 3, 7, 6, 8, 100, -2, 45, 9 };
    size_t n = sizeof(data) / sizeof(data[0]);
    size_t k = 3U;
    int *min_vals;
    int *max_vals;
    size_t i;
    int status;

    if (k == 0U || k > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Invalid k value\n");
        return EXIT_FAILURE;
    }

    min_vals = malloc(k * sizeof(int));
    if (min_vals == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    max_vals = malloc(k * sizeof(int));
    if (max_vals == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(min_vals);
        return EXIT_FAILURE;
    }

    status = extract_min_max_k(data, n, k, min_vals, max_vals);
    if (status != 0) {
        fprintf(stderr, "Extraction failed\n");
        free(min_vals);
        free(max_vals);
        return EXIT_FAILURE;
    }

    printf("Minimum %zu elements: ", k);
    for (i = 0U; i < k; i++) {
        printf("%d ", min_vals[i]);
    }
    printf("\n");

    printf("Maximum %zu elements: ", k);
    for (i = 0U; i < k; i++) {
        printf("%d ", max_vals[i]);
    }
    printf("\n");

    free(min_vals);
    free(max_vals);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The flagged weakness (Flawfinder CWE-120 on `memcpy`) was a false positive in the original code, but the rewrite hardens it anyway:

1. **`memcpy` safety (CWE-120):** The destination `copy` is allocated with exactly `n * sizeof(int)` bytes — the same size copied from `arr`. The size is computed once into `copy_size` and reused for both `malloc` and `memcpy`, guaranteeing they can never diverge and making the invariant explicit to both reviewers and static analyzers.

2. **Missing include fixed:** The original code used `SIZE_MAX` without including `<stdint.h>`, which is a genuine portability bug (it may compile on glibc via transitive includes, but is non-conforming). `<stdint.h>` is now included.

3. **Integer overflow guard retained:** The `n > SIZE_MAX / sizeof(int)` check prevents multiplication overflow in the allocation size (CWE-190), which would otherwise be the real path to a heap overflow.

4. **Defensive check in `main`:** A guard on `k` before the `k * sizeof(int)` allocations prevents overflow there as well, keeping the caller-side allocations provably large enough for the `k` writes performed by `extract_min_max_k`.

5. **Existing good practices preserved:** NULL-pointer validation, `k <= n` bounds enforcement, checked allocations, and no memory leaks on any error path.
