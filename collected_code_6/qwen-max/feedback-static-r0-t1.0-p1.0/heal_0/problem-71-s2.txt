#include <stdio.h>
#include <stdlib.h>

void combSort(int *arr, int n) {
    int gap = n;
    float shrink = 1.3;
    int swapped = 1;

    while (gap > 1 || swapped) {
        gap = (int)(gap / shrink);
        if (gap < 1) 
            gap = 1;
        
        swapped = 0;
        for (int i = 0; i + gap < n; ++i) {
            if (arr[i] > arr[i + gap]) {
                int temp = arr[i];
                arr[i] = arr[i + gap];
                arr[i + gap] = temp;
                swapped = 1;
            }
        }
    }
}

int main() {
    int arr[] = {5, 3, 9, 1, 4, 8, 2, 7, 6};
    int n = sizeof(arr) / sizeof(arr[0]);

    combSort(arr, n);

    for (int i = 0; i < n; ++i) {
        printf("%d ", arr[i]);
    }
    return 0;
}
