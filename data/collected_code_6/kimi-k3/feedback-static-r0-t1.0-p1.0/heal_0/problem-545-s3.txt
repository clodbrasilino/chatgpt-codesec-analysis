```c
#include <stdio.h>
#include <stdint.h>
#include <limits.h>

uint32_t toggle_first_last_bits(uint32_t num) {
    uint32_t first_bit = 1U << (sizeof(num) * CHAR_BIT - 1);
    uint32_t last_bit = 1U;
    return num ^ first_bit ^ last_bit;
}

int main(void) {
    uint32_t num;
    
    if (scanf("%u", &num) != 1) {
        return 1;
    }
    
    uint32_t result = toggle_first_last_bits(num);
    
    if (printf("%u\n", result) < 0) {
        return 1;
    }
    
    return 0;
}
```
