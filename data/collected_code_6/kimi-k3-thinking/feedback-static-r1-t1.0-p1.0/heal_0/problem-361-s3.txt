#include <stdio.h>
#include <stdlib.h>

typedef struct IntNode {
    int data;
    struct IntNode *next;
} IntNode;

typedef struct ListNode {
    IntNode *list;
    struct ListNode *next;
} ListNode;

static IntNode *create_int_list(const int *values, size_t count)
{
    IntNode *head = NULL;
    IntNode *tail = NULL;
    size_t i;

    if (values == NULL && count > 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        IntNode *node = malloc(sizeof *node);
        if (node == NULL) {
            while (head != NULL) {
                IntNode *tmp = head;
                head = head->next;
                free(tmp);
            }
            return NULL;
        }
        node->data = values[i];
        node->next = NULL;
        if (tail == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }

    return head;
}

static void free_int_list(IntNode *head)
{
    while (head != NULL) {
        IntNode *tmp = head;
        head = head->next;
        free(tmp);
    }
}

static ListNode *append_list(ListNode **head, IntNode *list)
{
    ListNode *node;
    ListNode *cur;

    if (head == NULL) {
        return NULL;
    }

    node = malloc(sizeof *node);
    if (node == NULL) {
        return NULL;
    }
    node->list = list;
    node->next = NULL;

    if (*head == NULL) {
        *head = node;
        return node;
    }

    cur = *head;
    while (cur->next != NULL) {
        cur = cur->next;
    }
    cur->next = node;

    return node;
}

static int add_list(ListNode **head, const int *values, size_t count)
{
    IntNode *inner;

    if (head == NULL) {
        return -1;
    }

    inner = create_int_list(values, count);
    if (inner == NULL && count > 0) {
        return -1;
    }

    if (append_list(head, inner) == NULL) {
        free_int_list(inner);
        return -1;
    }

    return 0;
}

static void remove_empty_lists(ListNode **head)
{
    ListNode *cur;
    ListNode *prev = NULL;

    if (head == NULL) {
        return;
    }

    cur = *head;
    while (cur != NULL) {
        if (cur->list == NULL) {
            ListNode *tmp = cur;
            cur = cur->next;
            if (prev == NULL) {
                *head = cur;
            } else {
                prev->next = cur;
            }
            free(tmp);
        } else {
            prev = cur;
            cur = cur->next;
        }
    }
}

static void free_list_of_lists(ListNode *head)
{
    while (head != NULL) {
        ListNode *tmp = head;
        head = head->next;
        free_int_list(tmp->list);
        free(tmp);
    }
}

static void print_int_list(const IntNode *head)
{
    putchar('[');
    while (head != NULL) {
        printf("%d", head->data);
        if (head->next != NULL) {
            putchar(' ');
        }
        head = head->next;
    }
    putchar(']');
}

static void print_list_of_lists(const ListNode *head)
{
    putchar('{');
    while (head != NULL) {
        print_int_list(head->list);
        if (head->next != NULL) {
            putchar(' ');
        }
        head = head->next;
    }
    puts("}");
}

int main(void)
{
    ListNode *lists = NULL;
    int status = EXIT_SUCCESS;
    int a[] = {1, 2, 3};
    int b[] = {4, 5};
    int c[] = {6};
    int d[] = {7, 8, 9, 10};

    if (add_list(&lists, a, 3) != 0 ||
        add_list(&lists, NULL, 0) != 0 ||
        add_list(&lists, b, 2) != 0 ||
        add_list(&lists, NULL, 0) != 0 ||
        add_list(&lists, c, 1) != 0 ||
        add_list(&lists, NULL, 0) != 0 ||
        add_list(&lists, d, 4) != 0) {
        fprintf(stderr, "failed to build list of lists\n");
        status = EXIT_FAILURE;
    } else {
        printf("Before: ");
        print_list_of_lists(lists);
        remove_empty_lists(&lists);
        printf("After:  ");
        print_list_of_lists(lists);
    }

    free_list_of_lists(lists);
    return status;
}
