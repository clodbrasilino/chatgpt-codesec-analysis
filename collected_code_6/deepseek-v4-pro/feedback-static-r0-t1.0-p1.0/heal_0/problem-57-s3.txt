#include <stdio.h>
#include <stdlib.h>

void counting_sort_desc(int arr[], int n, int exp) {
    int *output = (int *)malloc(n * sizeof(int));
    int count[10] = {0};
    
    if (output == NULL) {
        exit(1);
    }
    
    for (int i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }
    
    for (int i = 8; i >= 0; i--) {
        count[i] += count[i + 1];
    }
    
    for (int i = 0; i < n; i++) {
        output[count[(arr[i] / exp) % 10] - 1] = arr[i];
        count[(arr[i] / exp) % 10]--;
    }
    
    for (int i = 0; i < n; i++) {
        arr[i] = output[i];
    }
    
    free(output);
}

void sort_descending(int arr[], int n) {
    int max = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    
    for (int exp = 1; max / exp > 0; exp *= 10) {
        counting_sort_desc(arr, n, exp);
    }
}

long long largest_number_from_digits(int digits[], int n) {
    if (n <= 0) {
        return 0;
    }
    
    sort_descending(digits, n);
    
    long long result = 0;
    for (int i = 0; i < n; i++) {
        result = result * 10 + digits[i];
    }
    
    return result;
}

int main(void) {
    int digits[] = {3, 1, 7, 4, 9, 2};
    int n = sizeof(digits) / sizeof(digits[0]);
    
    long long result = largest_number_from_digits(digits, n);
    printf("%lld\n", result);
    
    return 0;
}
