#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define BUFFER_SIZE 256U

bool is_binary_string(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0U; (i < max_len) && (str[i] != '\0'); i++) {
        if ((str[i] != '0') && (str[i] != '1')) {
            return false;
        }
    }

    if (i >= max_len) {
        return false;
    }

    return true;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;
    char *newline;

    printf("Enter a string: ");

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    newline = memchr(buffer, '\n', sizeof(buffer));
    if (newline != NULL) {
        *newline = '\0';
    }

    len = strnlen(buffer, sizeof(buffer));
    if (len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return 1;
    }

    if (is_binary_string(buffer, sizeof(buffer))) {
        printf("The string is a binary string\n");
    } else {
        printf("The string is not a binary string\n");
    }

    return 0;
}

**CWE-119/CWE-120 (statically-sized buffer `buffer[256]`):** The Flawfinder warning is a heuristic triggered simply by the presence of a fixed-size `char` array. In this code the risk is already mitigated:

1. **Bounded input:** `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to at most 255 characters plus the null terminator, preventing overflow at the read site. The return value is checked for `NULL` to handle EOF/read errors.

2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1U] = '\0';` is defensive redundancy ensuring the buffer is always null-terminated, even though `fgets` already guarantees this on success.

3. **Bounded scanning:** `memchr` (bounded by `sizeof(buffer)`) is used instead of `strchr`, and `strnlen` instead of `strlen`, so no operation can read past the end of the array even if termination were somehow lost.

4. **Bounded validation:** `is_binary_string` receives an explicit `max_len`, checks for `NULL` and empty strings, and rejects any string that fills the buffer without a terminator (`i >= max_len`).
