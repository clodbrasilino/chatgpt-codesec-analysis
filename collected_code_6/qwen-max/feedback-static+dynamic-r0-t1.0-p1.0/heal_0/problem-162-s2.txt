```c
#include <stdio.h>

int sum_series(int n) {
    int sum = 0;
    for (int i = n; i > 0; i -= 2) {
        sum += i;
    }
    return sum;
}

int main() {
    int n;
    printf("Enter an integer: ");
    scanf("%d", &n);
    int result = sum_series(n);
    printf("The sum of the series is: %d\n", result);
    return 0;
}
```
