#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>

unsigned long long sum_of_cubes(unsigned long long n) {
    unsigned long long sum = 0;
    unsigned long long i;
    
    for (i = 1; i <= n; i++) {
        if (i > ULLONG_MAX / (i * i)) {
            fprintf(stderr, "Error: Integer overflow detected\n");
            exit(EXIT_FAILURE);
        }
        unsigned long long cube = i * i * i;
        if (sum > ULLONG_MAX - cube) {
            fprintf(stderr, "Error: Integer overflow detected\n");
            exit(EXIT_FAILURE);
        }
        sum += cube;
    }
    
    return sum;
}

int main(void) {
    unsigned long long n;
    int scan_result;
    
    printf("Enter a positive integer n: ");
    scan_result = scanf("%llu", &n);
    
    if (scan_result != 1) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (n == 0) {
        fprintf(stderr, "Error: n must be greater than 0\n");
        return EXIT_FAILURE;
    }
    
    printf("Sum of cubes from 1^3 to %llu^3 = %llu\n", n, sum_of_cubes(n));
    
    return EXIT_SUCCESS;
}
