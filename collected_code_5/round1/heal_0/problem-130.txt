```c
#include <stdio.h>
#include <stdlib.h>

#define SIZE 10000

void getMaxFrequency(int* list, int length) {
    int freq[SIZE] = {0};
    int max = 0, maxValue = list[0];

    for (int i = 0; i < length; i++) {
        freq[list[i]]++;
        if (max < freq[list[i]]) {
            max = freq[list[i]];
            maxValue = list[i];
        }
    }

    printf("Element with maximum frequency: %d \n", maxValue);
}

int main() {
    int length;
    printf("Enter the length of list: ");
    scanf("%d", &length);

    if (length <= 0) {
        printf("Invalid list length!\n");
        return 0;
    }

    int* list = (int*) malloc(sizeof(int) * length);

    if (list == NULL) {
        printf("Memory allocation failed!\n");
        return 0;
    }

    printf("Enter the elements of the list: \n");
    for(int i=0; i<length; ++i) {
        scanf("%d", &list[i]);
    }

    getMaxFrequency(list, length);
    free(list);

    return 0;
}
```
