```c
#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)b - *(int*)a);
}

void get_n_largest(int *data, int size, int n, int *largest) {
    qsort(data, size, sizeof(int), compare);
    for (int i = 0; i < n; ++i) {
        largest[i] = data[i];
    }
}

int main() {
    int data[] = {10, 2, 8, 6, 4, 12, 1};
    int size = sizeof(data) / sizeof(data[0]);
    int n = 3;
    int *largest = (int *)malloc(n * sizeof(int));

    if (largest == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    get_n_largest(data, size, n, largest);

    for (int i = 0; i < n; ++i) {
        printf("%d ", largest[i]);
    }
    printf("\n");

    free(largest);
    return 0;
}
```
