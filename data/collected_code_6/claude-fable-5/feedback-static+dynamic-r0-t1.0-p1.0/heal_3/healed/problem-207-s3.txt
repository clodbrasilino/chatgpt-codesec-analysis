```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024

static size_t safe_strnlen(const char *str, size_t max_len)
{
    size_t i;

    for (i = 0; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }
    return i;
}

int longestRepeatingSubsequence(const char *str, size_t max_len)
{
    size_t n;
    size_t i;
    size_t j;
    int **dp;
    int result;

    if (str == NULL) {
        return -1;
    }

    n = safe_strnlen(str, max_len);
    if (n == 0) {
        return 0;
    }

    if (n >= (SIZE_MAX / sizeof(int *)) || n >= (SIZE_MAX / sizeof(int))) {
        return -1;
    }

    dp = malloc((n + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= n; i++) {
        dp[i] = calloc(n + 1, sizeof(int));
        if (dp[i] == NULL) {
            while (i > 0) {
                i--;
                free(dp[i]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= n; i++) {
        for (j = 1; j <= n; j++) {
            if (str[i - 1] == str[j - 1] && i != j) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = (dp[i - 1][j] > dp[i][j - 1]) ? dp[i - 1][j] : dp[i][j - 1];
            }
        }
    }

    result = dp[n][n];

    for (i = 0; i <= n; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = safe_strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = longestRepeatingSubsequence(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Failed to compute longest repeating subsequence\n");
        return EXIT_FAILURE;
    }

    printf("Length of longest repeating subsequence: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Summary of changes:**

1. **Duplicate/overlapping overflow checks (cppcheck `duplicateCondition`):** The two separate `if` statements guarding the `n + 1` multiplication overflow were merged into a single condition. Both bounds (`sizeof(int *)` for the pointer array and `sizeof(int)` for each row) are still verified, using `>=` against `SIZE_MAX / size` which cleanly guarantees `(n + 1) * size` cannot wrap, while eliminating the redundant-condition warning and remaining correct on platforms where `sizeof(int *)` differs from `sizeof(int)`.

2. **Flawfinder CWE-119/CWE-120 (statically-sized buffer):** The existing mitigations are already correct and were preserved: `fgets(buffer, sizeof(buffer), stdin)` bounds the read, explicit NUL termination at `sizeof(buffer) - 1` provides defense in depth, `safe_strnlen` never reads past the buffer, and `longestRepeatingSubsequence` receives the buffer capacity as an explicit `max_len` parameter so it performs its own bounded length computation. This is a Flawfinder heuristic hit rather than an actual defect; the code never writes or reads out of bounds.

3. **Memory-safety properties retained:** the partial-allocation rollback loop on `calloc` failure, full cleanup of `dp` before every return path, and the `NULL` input guard are all unchanged.
