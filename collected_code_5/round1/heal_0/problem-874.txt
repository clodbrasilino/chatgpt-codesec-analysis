#include <stdio.h>
#include <string.h>

int isConcatenation(char* str1, char* str2) {
    
    int str1_len = strlen(str1);
    int str2_len = strlen(str2);

    if(str1_len % str2_len != 0){
        return 0;
    }

    for(int i = 0; i < str1_len; i+=str2_len){
        if(strncmp(str1+i, str2, str2_len) != 0){
            return 0;
        }
    }
    return 1;
}

int main() {
    char* str1 = "appleappleapple";
    char* str2 = "apple";

    if(isConcatenation(str1, str2)){
        printf("The string is a concatenation of another string.\n");
    }
    else {
        printf("The string is not a concatenation of another string.\n");
    }
    return 0;
}
