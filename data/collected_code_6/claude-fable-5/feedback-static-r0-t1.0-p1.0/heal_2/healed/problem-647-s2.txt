```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t capacity;
    size_t num_tokens;
    size_t start;
    size_t i;
    char **tokens;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return NULL;
    }

    capacity = 8;
    if (capacity > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }
    tokens = malloc(capacity * sizeof(char *));
    if (tokens == NULL) {
        return NULL;
    }

    num_tokens = 0;
    start = 0;

    for (i = 0; i <= len; i++) {
        if (i == len || (isupper((unsigned char)input[i]) && i > start)) {
            size_t token_len = i - start;

            if (token_len > 0) {
                char *token;
                size_t alloc_size;

                if (num_tokens == capacity) {
                    size_t new_capacity;
                    char **resized;

                    if (capacity > SIZE_MAX / 2 ||
                        (capacity * 2) > SIZE_MAX / sizeof(char *)) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    new_capacity = capacity * 2;
                    resized = realloc(tokens, new_capacity * sizeof(char *));
                    if (resized == NULL) {
                        free_tokens(tokens, num_tokens);
                        return NULL;
                    }
                    tokens = resized;
                    capacity = new_capacity;
                }

                if (start > len || token_len > len - start ||
                    token_len > SIZE_MAX - 1) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                alloc_size = token_len + 1;
                token = malloc(alloc_size);
                if (token == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                if (token_len >= alloc_size) {
                    free(token);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }

                memcpy(token, &input[start], token_len);
                token[token_len] = '\0';
                tokens[num_tokens] = token;
                num_tokens++;
            }
            start = i;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char *test = "helloWorldThisIsATest";
    size_t count = 0;
    size_t i;
    char **tokens;

    tokens = split_at_uppercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens (%zu):\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu] %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged `memcpy` (CWE-120) was already effectively safe in the original code, since `token` is allocated with `malloc(token_len + 1)` immediately before the copy of exactly `token_len` bytes, leaving room for the null terminator. The rewrite makes this guarantee explicit and auditable:

1. **Explicit overflow guard before allocation** — `token_len > SIZE_MAX - 1` ensures `token_len + 1` cannot wrap around to zero, which would otherwise cause `malloc(0)` followed by a heap overflow in `memcpy`.
2. **Explicit destination-capacity check** — `token_len >= alloc_size` verifies the destination buffer can hold the source bytes plus the terminator before `memcpy` executes, directly satisfying the Flawfinder recommendation.
3. **Source-bounds validation retained** — `start > len || token_len > len - start` guarantees the source range `&input[start]` through `token_len` bytes stays within the validated input string (bounded by `strnlen` against `MAX_INPUT_LEN`).

All other defensive measures from the original are preserved: NULL-parameter checks, capacity-doubling overflow guards on `realloc`, full cleanup via `free_tokens` on every failure path, and the `isupper((unsigned char)...)` cast to avoid undefined behavior with negative `char` values.
