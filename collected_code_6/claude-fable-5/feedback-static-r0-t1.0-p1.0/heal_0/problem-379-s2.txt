```c
#include <stdio.h>
#include <stdlib.h>

double cuboid_surface_area(double length, double width, double height)
{
    return 2.0 * ((length * width) + (width * height) + (height * length));
}

int main(void)
{
    double length;
    double width;
    double height;

    printf("Enter length: ");
    if (scanf("%lf", &length) != 1) {
        fprintf(stderr, "Invalid input for length.\n");
        return EXIT_FAILURE;
    }

    printf("Enter width: ");
    if (scanf("%lf", &width) != 1) {
        fprintf(stderr, "Invalid input for width.\n");
        return EXIT_FAILURE;
    }

    printf("Enter height: ");
    if (scanf("%lf", &height) != 1) {
        fprintf(stderr, "Invalid input for height.\n");
        return EXIT_FAILURE;
    }

    if (length <= 0.0 || width <= 0.0 || height <= 0.0) {
        fprintf(stderr, "Dimensions must be positive values.\n");
        return EXIT_FAILURE;
    }

    printf("Surface area of the cuboid: %.2f\n",
           cuboid_surface_area(length, width, height));

    return EXIT_SUCCESS;
}
```

This implementation validates every `scanf` return value to handle exceptional input conditions, rejects non-positive dimensions, uses `double` to reduce overflow risk, returns standard exit codes, and follows consistent coding standards with `int main(void)` and explicit function prototypes through definition ordering.
