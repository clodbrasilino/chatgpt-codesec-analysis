```c
#include <stdio.h>
#include <ctype.h>
#include <string.h>

void count_characters(const char *str, size_t *alphabets, size_t *digits, size_t *specials)
{
    size_t i;

    if (str == NULL || alphabets == NULL || digits == NULL || specials == NULL)
    {
        return;
    }

    *alphabets = 0;
    *digits = 0;
    *specials = 0;

    for (i = 0; str[i] != '\0'; i++)
    {
        unsigned char c = (unsigned char)str[i];

        if (isalpha(c))
        {
            (*alphabets)++;
        }
        else if (isdigit(c))
        {
            (*digits)++;
        }
        else if (!isspace(c))
        {
            (*specials)++;
        }
    }
}

int main(void)
{
    char buffer[256];
    size_t alphabets;
    size_t digits;
    size_t specials;

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL)
    {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    count_characters(buffer, &alphabets, &digits, &specials);

    printf("Alphabets: %zu\n", alphabets);
    printf("Digits: %zu\n", digits);
    printf("Special characters: %zu\n", specials);

    return 0;
}
```

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow. Using `sizeof(buffer)` instead of a hardcoded literal ensures the bound stays correct if the buffer size changes.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0';` is defensive (technically redundant with `fgets`, which always null-terminates on success, but harmless and robust).

3. **Newline stripping via `strcspn`** — safer than the common `buffer[strlen(buffer) - 1] = '\0'` idiom, which invokes undefined behavior on an empty string.

4. **`ctype.h` safety** — casting to `unsigned char` before calling `isalpha`/`isdigit`/`isspace` avoids undefined behavior when `char` is signed and holds negative values (e.g., extended/UTF-8 bytes).

5. **NULL-pointer guards** in `count_characters` and checking the return value of `fgets` handle error paths gracefully.
