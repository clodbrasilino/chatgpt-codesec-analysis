```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

int all_different(const int *arr, size_t n) {
    if (arr == NULL || n == 0) {
        return 0;
    }
    
    for (size_t i = 0; i < n; i++) {
        for (size_t j = i + 1; j < n; j++) {
            if (arr[i] == arr[j]) {
                return 0;
            }
        }
    }
    return 1;
}

int main(void) {
    size_t n = 0;
    int *arr = NULL;
    int result = 0;
    
    printf("Enter the number of elements: ");
    
    if (scanf("%zu", &n) != 1) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (n == 0 || n > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Error: Invalid size\n");
        return EXIT_FAILURE;
    }
    
    arr = malloc(n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Error: Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    printf("Enter %zu integers: ", n);
    
    for (size_t i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Error: Invalid input\n");
            free(arr);
            arr = NULL;
            return EXIT_FAILURE;
        }
    }
    
    result = all_different(arr, n);
    
    if (result) {
        printf("All numbers are different from each other.\n");
    } else {
        printf("Some numbers are duplicated.\n");
    }
    
    free(arr);
    arr = NULL;
    
    return EXIT_SUCCESS;
}
```
