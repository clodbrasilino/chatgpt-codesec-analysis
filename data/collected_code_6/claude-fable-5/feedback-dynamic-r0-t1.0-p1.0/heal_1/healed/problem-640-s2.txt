#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test1 = "Hello (remove me) World";
    const char *test2 = "Nested (outer (inner) text) example";

    output = remove_parentheses(test1);

The original `remove_parentheses` logic is actually sound: it allocates `len + 1` bytes, tracks nesting depth correctly, only copies characters at depth zero, and terminates the string with `'\0'`. Changes made to harden against those reported failures:

3. **NULL input and allocation-failure checks retained** — every allocation is verified before use, and every buffer is freed exactly once after its final use, preventing use-after-free and leaks.
