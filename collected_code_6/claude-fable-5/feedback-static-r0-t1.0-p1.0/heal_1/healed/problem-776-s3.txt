```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
```

The key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen`, which takes an explicit maximum length. The function signature now accepts a `max_len` parameter so the caller communicates the buffer bound, guaranteeing the scan never reads past the buffer even if a non-terminated string is passed.

2. **CWE-119/CWE-120 (statically-sized array):** The buffer size is defined as a named constant, `fgets` is bounded by `sizeof(buffer)` (which already prevents overflow), and an explicit `buffer[sizeof(buffer) - 1] = '\0';` defensively guarantees null-termination before any string operations. The buffer size is also passed down to the counting function so all bounds are enforced consistently.
