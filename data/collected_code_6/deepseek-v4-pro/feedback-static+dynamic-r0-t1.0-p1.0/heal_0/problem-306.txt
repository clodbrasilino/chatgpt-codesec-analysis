#include <stdio.h>
#include <stdlib.h>

int max_sum_increasing_subsequence_prefix_with_k(int arr[], int n, int k) {
    if (arr == NULL || n <= 0 || k < 0 || k >= n) {
        return 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        dp[i] = arr[i];
    }

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[i] > arr[j] && dp[i] < dp[j] + arr[i]) {
                dp[i] = dp[j] + arr[i];
            }
        }
    }

    int max_sum = 0;
    for (int i = 0; i < k; i++) {
        if (arr[i] < arr[k]) {
            if (dp[i] + arr[k] > max_sum) {
                max_sum = dp[i] + arr[k];
            }
        }
    }

    max_sum = (max_sum > arr[k]) ? max_sum : arr[k];

    free(dp);
    return max_sum;
}

int main() {
    int arr[] = {1, 101, 2, 3, 100, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 5;

    int result = max_sum_increasing_subsequence_prefix_with_k(arr, n, k);
    printf("%d\n", result);

    int arr2[] = {3, 4, 5, 10};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int k2 = 3;

    int result2 = max_sum_increasing_subsequence_prefix_with_k(arr2, n2, k2);
    printf("%d\n", result2);

    int arr3[] = {10, 5, 18, 7, 9};

    int k3 = 4;

    printf("%d\n", result3);

    return 0;
}
