#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

typedef struct {
    int *elements;
    size_t size;
} Tuple;

int find_max_nth_column(const Tuple *tuples, size_t num_tuples, size_t n, int *result) {
    size_t i;
    int max_val;
    int found = 0;

    if (tuples == NULL || result == NULL || num_tuples == 0) {
        return -1;
    }

    max_val = INT_MIN;

    for (i = 0; i < num_tuples; i++) {
        if (tuples[i].elements != NULL && n < tuples[i].size) {
            if (!found || tuples[i].elements[n] > max_val) {
                max_val = tuples[i].elements[n];
                found = 1;
            }
        }
    }

    if (!found) {
        return -1;
    }

    *result = max_val;
    return 0;
}

int main(void) {
    Tuple tuples[4];
    int data0[] = {1, 2, 3};
    int data1[] = {4, 5, 6};
    int data2[] = {7, 8, 9};
    int data3[] = {10, 11, 12};
    size_t n = 1;
    int max_value;
    int status;

    tuples[0].elements = data0;
    tuples[0].size = 3;
    tuples[1].elements = data1;
    tuples[1].size = 3;
    tuples[2].elements = data2;
    tuples[2].size = 3;
    tuples[3].elements = data3;
    tuples[3].size = 3;

    status = find_max_nth_column(tuples, 4, n, &max_value);

    if (status == 0) {
        printf("Maximum of column %zu: %d\n", n, max_value);
    } else {
        fprintf(stderr, "Error: Unable to find maximum for column %zu\n", n);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
