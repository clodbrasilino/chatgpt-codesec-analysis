```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    char *result;
    size_t len;

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "No input provided\n");
        return EXIT_FAILURE;
    }

    len = strlen(input);
    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

The conversion logic in `snake_to_camel` was actually correct — memory allocation was bounded, `toupper` was called safely with an `unsigned char` cast, and the buffer was properly null-terminated with no overflow risk (the output is never longer than the input).

The real weakness, revealed by the failing test cases, was in `main`: the program used a **hardcoded input string** (`"hello_world_example_string"`) and printed decorated output (`"Input: ..."` / `"Output: ..."`). The test harness expected results like `PythonProgram` and `ProgrammingLanguage`, meaning it supplies input via **stdin** and expects the bare converted string on stdout — hence the `<no output>` failures.

The rewrite:
