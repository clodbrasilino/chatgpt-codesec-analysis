#include <stdio.h>
#include <stdlib.h>

static unsigned int countUnsetBits(unsigned int num)
{
    unsigned int count = 0U;
    while (num > 0U)
    {
        if ((num & 1U) == 0U)
        {
            count++;
        }
        num >>= 1U;
    }
    return count;
}

static unsigned long long totalUnsetBits(unsigned int n)
{
    unsigned long long total = 0ULL;
    unsigned int i;
    for (i = 1U; i <= n; i++)
    {
        total += (unsigned long long)countUnsetBits(i);
        if (i == n)
        {
            break;
        }
    }
    return total;
}

int main(int argc, const char *argv[])
{
    unsigned int n = 0U;

    if (argc > 1)
    {
        char extra = '\0';
        if (sscanf(argv[1], "%u %c", &n, &extra) != 1 || n < 1U)
        {
            fprintf(stderr, "Error: invalid input\n");
            return EXIT_FAILURE;
        }
    }
    else
    {
        if (scanf("%u", &n) != 1)
        {
            fprintf(stderr, "Error: invalid input\n");
            return EXIT_FAILURE;
        }
        if (n < 1U)
        {
            fprintf(stderr, "Error: input must be at least 1\n");
            return EXIT_FAILURE;
        }
    }

    printf("%llu\n", totalUnsetBits(n));

    return EXIT_SUCCESS;
}

**Analysis of the original code and the identified weaknesses:**

The failing test cases indicated the program was returning only the unset-bit count of the final number `n` rather than the cumulative total across `1..n`:

- **Test case 1** (`n = 5`): expected `4` — the sum of unset bits in 1 (`1`→0), 2 (`10`→1), 3 (`11`→0), 4 (`100`→2), 5 (`101`→1). Getting `1` means only `countUnsetBits(5)` was returned.
- **Test case 2** (`n = 14`): expected `17` (cumulative sum), but got `1`, which is `countUnsetBits(14)` (`1110` has one zero).

The fix ensures the accumulation loop in `totalUnsetBits` iterates over **every** value from `1` through `n`, adding each value's unset-bit count into `total` before returning. The rewritten loop uses a bounded `for` condition (`i <= n`) combined with the explicit `i == n` break to guarantee correct termination even at `n == UINT_MAX` (where `i++` would otherwise wrap and loop forever).

Other aspects retained because they are already sound:
- `unsigned long long` accumulator prevents overflow for large `n`.
- The `sscanf(argv[1], "%u %c", ...)` pattern rejects trailing garbage in command-line input.
- Input validation rejects non-numeric input and values below 1 on both input paths.
