static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static int has_thirty_days(const char *month)

    static const char *thirty_day_months[] = {

        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN];

        return -1;

        return -1;

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    }

}

1. **Statically-sized array `lowered[MAX_INPUT_LEN]` in `has_thirty_days` (CWE-119/CWE-120):** This is mitigated. Before any copy occurs, `bounded_strlen(month, MAX_INPUT_LEN)` rejects inputs that would not fit, and `to_lower_string` receives `sizeof(lowered)` as an explicit bound, always leaving room for and writing the null terminator. No overflow is possible.

2. **Statically-sized array `input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Mitigated. Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits the read to the buffer size, and the buffer is defensively null-terminated afterward. Length calculations use `bounded_strlen` rather than an unbounded `strlen`.

3. **`getchar` in a loop (CWE-120/CWE-20):** Safe. The loop discards excess characters from `stdin` without storing them into any buffer, so no boundary can be violated. It correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed streams.
