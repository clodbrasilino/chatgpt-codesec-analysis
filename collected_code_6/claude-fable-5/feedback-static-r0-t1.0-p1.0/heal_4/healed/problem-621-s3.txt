#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

static int append_chunk(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || *buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n > SIZE_MAX - 1 || *len > SIZE_MAX - n - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t needed = *len + n + 1;
        size_t newcap = *cap;
        if (newcap == 0) {
            newcap = 1;
        }
        while (newcap < needed) {
            if (newcap > SIZE_MAX / 2) {
                return -1;
            }
            newcap *= 2;
        }
        char *tmp = realloc(*buf, newcap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = newcap;
    }
    if (*len + n + 1 > *cap) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

char *increment_numbers(const char *input, long k)
{
    size_t cap = 64;
    size_t len = 0;
    size_t i = 0;
    char *out;

    if (input == NULL) {
        return NULL;
    }

    out = malloc(cap);
    if (out == NULL) {
        return NULL;
    }
    out[0] = '\0';

    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            char *end = NULL;
            long val;
            long newval;
            char numbuf[32];
            int written;

            errno = 0;
            val = strtol(&input[i], &end, 10);
            if (errno == ERANGE || end == NULL || end == &input[i]) {
                free(out);
                return NULL;
            }
            if ((k > 0 && val > LONG_MAX - k) || (k < 0 && val < LONG_MIN - k)) {
                free(out);
                return NULL;
            }
            newval = val + k;

            written = snprintf(numbuf, sizeof numbuf, "%ld", newval);
            if (written < 0 || (size_t)written >= sizeof numbuf) {
                free(out);
                return NULL;
            }
            if (append_chunk(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;
            }
            i = (size_t)(end - input);
        } else {
            if (append_chunk(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    return out;
}

int main(void)
{
    const char *tests[] = {
        "abc 5 def 10",
        "no numbers here",
        "1,2,3",
        "value=99end",
        "0 and 100"
    };
    long k = 5;
    size_t i;
    size_t count = sizeof tests / sizeof tests[0];

    for (i = 0; i < count; i++) {
        char *result = increment_numbers(tests[i], k);
        if (result == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("Input:  %s\nOutput: %s\n\n", tests[i], result) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }
    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is safe:**

1. **`memcpy` in `append_chunk` (CWE-120, Flawfinder risk 2):** This is a false positive. Before the `memcpy` executes, the function guarantees the destination can hold the source data:
   - Integer-overflow guards (`n > SIZE_MAX - 1`, `*len > SIZE_MAX - n - 1`) prevent wraparound in size arithmetic.
   - The capacity-growth loop ensures `*cap >= *len + n + 1` before copying, with an overflow check on the doubling (`newcap > SIZE_MAX / 2`).
   - A redundant post-realloc verification (`*len + n + 1 > *cap`) acts as a defensive belt-and-suspenders check.
   - The buffer is always NUL-terminated within capacity after the copy.

2. **Statically-sized `char numbuf[32]` (CWE-119/CWE-120, Flawfinder risk 2):** Also a false positive. A `long` (even 64-bit) formatted with `%ld` needs at most 20 digits plus sign plus NUL (22 bytes), so 32 bytes is sufficient. More importantly, `snprintf` with `sizeof numbuf` bounds the write, and the return value is checked for both negative errors and truncation (`(size_t)written >= sizeof numbuf`), so no overflow or silently truncated data can propagate.

**Other robustness points already handled correctly:**
- `strtol` errors detected via `errno == ERANGE` and `end == &input[i]` (no conversion).
- Signed overflow of `val + k` is checked *before* the addition, avoiding undefined behavior.
- `isdigit` is called with an `unsigned char` cast, avoiding UB for negative `char` values.
- All allocation failures free `out` and return `NULL`; `main` frees results and checks `printf`'s return.

The code was already written defensively; the Flawfinder hits are informational and both are properly mitigated, so the rewrite preserves the logic with the comments removed as requested.
