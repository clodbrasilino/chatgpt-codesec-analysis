#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <errno.h>

double calculate_std_dev(const double *data, size_t size, int *error) {
    double sum = 0.0;
    double mean = 0.0;
    double variance_sum = 0.0;
    double std_dev = 0.0;
    size_t i = 0;

    if (data == NULL || size == 0) {
        *error = -1;
        return 0.0;
    }

    *error = 0;

    for (i = 0; i < size; i++) {
        sum += data[i];
    }
    mean = sum / size;

    for (i = 0; i < size; i++) {
        variance_sum += (data[i] - mean) * (data[i] - mean);
    }

    std_dev = sqrt(variance_sum / size);
    return std_dev;
}

int main(void) {
    double *data = NULL;
    size_t size = 0;
    size_t i = 0;
    int error = 0;
    double result = 0.0;

    printf("Enter the number of elements: ");
    if (scanf("%zu", &size) != 1 || size == 0) {
        fprintf(stderr, "Invalid input for size.\n");
        return EXIT_FAILURE;
    }

    data = (double *)malloc(size * sizeof(double));
    if (data == NULL) {
        fprintf(stderr, "Memory allocation failed.\n");
        return EXIT_FAILURE;
    }

    printf("Enter %zu elements:\n", size);
    for (i = 0; i < size; i++) {
        if (scanf("%lf", &data[i]) != 1) {
            fprintf(stderr, "Invalid input for element %zu.\n", i + 1);
            free(data);
            return EXIT_FAILURE;
        }
    }

    result = calculate_std_dev(data, size, &error);
    if (error != 0) {
        fprintf(stderr, "Error calculating standard deviation.\n");
        free(data);
        return EXIT_FAILURE;
    }

    printf("Standard Deviation = %.6f\n", result);

    free(data);
    return EXIT_SUCCESS;
}
