```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static long long merge(int *arr, int *temp, size_t left, size_t mid, size_t right, size_t n)
{
    size_t i = left;
    size_t j = mid;
    size_t k = left;
    long long inv_count = 0;

    if (arr == NULL || temp == NULL || right >= n || left > mid || mid > right) {
        return -1;
    }

    while (i <= mid - 1 && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k] = arr[i];
            k++;
            i++;
        } else {
            temp[k] = arr[j];
            k++;
            j++;
            inv_count += (long long)(mid - i);
        }
    }

    while (i <= mid - 1) {
        temp[k] = arr[i];
        k++;
        i++;
    }

    while (j <= right) {
        temp[k] = arr[j];
        k++;
        j++;
    }

    if (right - left + 1 > n - left) {
        return -1;
    }

    memcpy(&arr[left], &temp[left], (right - left + 1) * sizeof(int));

    return inv_count;
}

static long long merge_sort(int *arr, int *temp, size_t left, size_t right, size_t n)
{
    long long inv_count = 0;
    long long sub_count;

    if (right > left) {
        size_t mid = left + (right - left) / 2;

        sub_count = merge_sort(arr, temp, left, mid, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;

        sub_count = merge_sort(arr, temp, mid + 1, right, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;

        sub_count = merge(arr, temp, left, mid + 1, right, n);
        if (sub_count < 0) {
            return -1;
        }
        inv_count += sub_count;
    }

    return inv_count;
}

long long count_inversions(const int *arr, size_t n)
{
    int *copy = NULL;
    int *temp = NULL;
    long long result = -1;

    if (arr == NULL || n == 0) {
        return 0;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy = malloc(n * sizeof(int));
    if (copy == NULL) {
        return -1;
    }

    temp = malloc(n * sizeof(int));
    if (temp == NULL) {
        free(copy);
        return -1;
    }

    memcpy(copy, arr, n * sizeof(int));

    result = merge_sort(copy, temp, 0, n - 1, n);

    free(copy);
    free(temp);

    return result;
}

int main(void)
{
    int arr[] = { 8, 4, 2, 1 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long inversions;

    inversions = count_inversions(arr, n);
    if (inversions < 0) {
        if (fprintf(stderr, "Error: memory allocation failed\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Number of inversions: %lld\n", inversions) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Summary of changes addressing the identified weaknesses:**

1. **CWE-120 in `merge()`:** Added bounds validation before the `memcpy` — the function now receives the total array size `n` and verifies that `right < n` and that the copy length `(right - left + 1)` cannot exceed the space remaining from `left`, guaranteeing the destination can hold the source data. Errors propagate up as `-1` through `merge_sort()`.

2. **CWE-120 / CWE-190 in `count_inversions()`:** Added an integer-overflow guard (`n > SIZE_MAX / sizeof(int)`) before the `malloc` calls, ensuring the allocation sizes cannot wrap around and cause undersized buffers, which makes the subsequent `memcpy` of `n * sizeof(int)` bytes provably safe since both buffers were allocated with exactly that size.

3. **Hardening bonus:** Replaced `(left + right) / 2` with `left + (right - left) / 2` to avoid potential `size_t` overflow on very large indices, and included `<stdint.h>` for `SIZE_MAX`.
