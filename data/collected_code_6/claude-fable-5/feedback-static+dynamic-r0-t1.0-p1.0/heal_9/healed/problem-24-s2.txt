```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

static int binary_to_decimal(const char *binary, size_t max_len, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_line(int *was_truncated)
{
    int ch;
    unsigned long guard = 0UL;

    if (was_truncated == NULL) {
        return -1;
    }

    *was_truncated = 0;

    while ((ch = getchar()) != '\n' && ch != EOF) {
        *was_truncated = 1;
        guard++;
        if (guard == ULONG_MAX) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error: failed to flush output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int truncated = 0;

        if (flush_stdin_line(&truncated) != 0) {
            fprintf(stderr, "Error: internal failure.\n");
            return EXIT_FAILURE;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — The `flush_stdin_line` function never writes the characters it reads into any buffer; it only counts and discards them. The `guard` counter prevents a pathological infinite loop from spinning forever without detection. No buffer can overflow here, so the Flawfinder hit is a false positive.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — The `input` buffer is protected on every path:
   - `fgets(input, (int)sizeof(input), stdin)` limits the read to the buffer size, including the terminating NUL.
   - The explicit `input[sizeof(input) - 1U] = '\0';` guarantees termination even in edge cases.
   - `strnlen` is used instead of `strlen`, bounding every length computation to `sizeof(input)`.
   - Overlong lines are detected (`truncated`) and rejected rather than silently processed.
   - `binary_to_decimal` re-validates bounds (`len >= max_len` rejection) and checks for multiplication/shift overflow against `ULLONG_MAX >> 1` before each shift.

Both findings are already mitigated; the rewrite above preserves all defensive measures with the comments removed.
