```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        sum += (long)(unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    long result;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened against the flagged weakness (CWE-119/CWE-120, statically-sized buffer). The rewrite preserves all mitigations because they are correct as written:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing the classic `gets()`-style overflow. The return value is checked for `NULL` to handle EOF/read errors.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0';` is defensive; `fgets` already null-terminates, but this makes the invariant explicit and robust against future refactoring.

3. **Length-bounded string operations** — `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, ensuring the scan never runs past the buffer even if termination were somehow lost.

4. **Safe character summation** — the cast chain `(long)(unsigned char)str[i]` avoids sign-extension issues on platforms where `char` is signed (e.g., a byte value of `0xFF` sums as `255`, not `-1`), and accumulating in `long` prevents overflow for any realistic 256-byte input (max sum ≈ 65,280).

5. **NULL guard and error signaling** — `ascii_sum` rejects `NULL` input with a `-1` sentinel, which the caller checks before printing.

6. **Newline stripping with bounds check** — `len > 0` is verified before indexing `buffer[len - 1]`, avoiding an out-of-bounds read on empty input.
