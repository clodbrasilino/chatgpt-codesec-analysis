```c
#include <stdio.h>

unsigned int gcd(unsigned int a, unsigned int b) {
    while (b != 0) {
        unsigned int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int main(void) {
    unsigned int num1, num2;
    
    if (scanf("%u %u", &num1, &num2) != 2) {
        return 1;
    }
    
    if (num1 == 0 || num2 == 0) {
        return 1;
    }
    
    printf("%u\n", gcd(num1, num2));
    
    return 0;
}
```
