static void free_table(size_t **table, size_t rows)
{
    size_t i;

    if (table == NULL) {
        return;
    }
    for (i = 0; i < rows; i++) {
        free(table[i]);
    }
    free(table);
}

static char *read_line(FILE *stream, size_t *out_len)
{
    size_t capacity = 128;
    size_t length = 0;
    char *buffer;
    char *tmp;
    int c;

    if (stream == NULL) {
        return NULL;
    }

    buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    while ((c = fgetc(stream)) != EOF && c != '\n') {
        if (length + 1 >= capacity) {
            if (capacity > SIZE_MAX / 2) {
                free(buffer);
                return NULL;
            }
            capacity *= 2;
            tmp = realloc(buffer, capacity);
            if (tmp == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = tmp;
        }
        buffer[length++] = (char)c;
    }

    if (c == EOF && length == 0) {
        free(buffer);
        return NULL;
    }

    buffer[length] = '\0';

    tmp = realloc(buffer, length + 1);
    if (tmp != NULL) {
        buffer = tmp;
    }

    if (out_len != NULL) {
        *out_len = length;
    }
    return buffer;
}

static char *longest_palindromic_subsequence(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t j;
    size_t len;
    size_t left;
    size_t right;
    size_t lps_len;
    size_t **dp;
    char *result;

    if (s == NULL) {
        return NULL;
    }

    n = strnlen(s, max_len);
    if (n == 0) {
        result = malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    dp = calloc(n, sizeof *dp);
    if (dp == NULL) {
        return NULL;
    }
    for (i = 0; i < n; i++) {
        dp[i] = calloc(n, sizeof **dp);
        if (dp[i] == NULL) {
            free_table(dp, i);
            return NULL;

        dp[i][i] = 1;

                dp[i][j] = dp[i + 1][j - 1] + 2;
            } else {
                dp[i][j] = (dp[i + 1][j] > dp[i][j - 1]) ? dp[i + 1][j] : dp[i][j - 1];

    if (lps_len == 0 || lps_len > n) {

    if (result == NULL) {
        free_table(dp, n);
        return NULL;

    i = 0;
    j = n - 1;
    left = 0;
    right = lps_len - 1;
    while (i <= j) {
        if (i == j) {
            result[left] = s[i];
            break;

            result[left] = s[i];
            result[right] = s[j];
            left++;
            right--;
            i++;
            j--;
        } else if (dp[i + 1][j] >= dp[i][j - 1]) {
            i++;
        } else {
            j--;
        }
    }

    return result;
}

**Analysis summary:**

The flagged weakness (`fgetc` in a loop — CWE-120/CWE-20) was already properly mitigated in the original code, so the rewrite preserves those defenses:

- **Bounds checking on every write:** `length + 1 >= capacity` is verified before each `buffer[length++]` store, guaranteeing room for both the character and the eventual NUL terminator, so no buffer overflow is possible.
- **Overflow-safe capacity growth:** `capacity > SIZE_MAX / 2` is checked before doubling, preventing `size_t` wraparound that could lead to an undersized allocation.
- **Robust `realloc` usage:** results are assigned to a temporary pointer first, avoiding memory leaks on allocation failure; the final shrinking `realloc` tolerates failure by keeping the original buffer.
- **EOF handling:** `c == EOF && length == 0` correctly distinguishes empty input at end-of-stream from valid partial lines.
- **Bounded string operations:** `strnlen` with an explicit maximum is used instead of `strlen`, in both the LPS function and `main`.
- **Consistent memory hygiene:** all error paths free previously allocated resources (`free_table` handles partial DP-table allocation via the row count), and `main` frees `input` and `lps` on every path.
- **Defensive validation:** NULL checks on all pointer parameters, `calloc` used for zero-initialized DP rows, and a sanity check `lps_len > n` before allocating the result buffer.

No functional changes were required — the code as written is already free of the reported weakness.
