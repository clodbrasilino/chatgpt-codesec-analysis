```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096

typedef struct {
    char *items;
    size_t count;
} Tuple;

Tuple *string_to_tuple(const char *input)
{
    Tuple *tuple = NULL;
    size_t length = 0;
    char *items = NULL;

    if (input == NULL) {
        return NULL;
    }

    length = strnlen(input, MAX_INPUT_LENGTH);
    if (length == 0 || length >= MAX_INPUT_LENGTH) {
        return NULL;
    }

    items = malloc(length + 1);
    if (items == NULL) {
        return NULL;
    }

    memcpy(items, input, length);
    items[length] = '\0';

    tuple = malloc(sizeof(Tuple));
    if (tuple == NULL) {
        free(items);
        return NULL;
    }

    tuple->items = items;
    tuple->count = length;
    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if (tuple == NULL || tuple->items == NULL) {
        return;
    }

    printf("[");
    for (i = 0; i < tuple->count; i++) {
        printf("'%c'", tuple->items[i]);
        if (i + 1 < tuple->count) {
            printf(", ");
        }
    }
    printf("]\n");
    fflush(stdout);
}

int main(void)
{
    const char *inputs[] = { "python3.0", "item1", "15.10" };
    size_t i;
    size_t n = sizeof(inputs) / sizeof(inputs[0]);

    for (i = 0; i < n; i++) {
        Tuple *tuple = string_to_tuple(inputs[i]);

        if (tuple == NULL) {
            fprintf(stderr, "Failed to convert string to tuple: \"%s\"\n",
                    inputs[i]);
            return EXIT_FAILURE;
        }

        print_tuple(tuple);

        free_tuple(tuple);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

1. **Test case failures (`<no output>`):** The original program prefixed each result with `Input string: "..."` and `Tuple: ` before printing the tuple. Automated test harnesses expecting exactly `['p', 'y', 't', 'h', 'o', 'n', '3', '.', '0']` on its own line would fail to match. The fix removes the extra decorative output so each line contains only the tuple representation the tests expect.

2. **Flawfinder `memcpy` warning (CWE-120):** This is a false positive in practice — the destination buffer is allocated as `length + 1` and exactly `length` bytes are copied, followed by explicit NUL termination. The bound is derived via `strnlen` with a hard cap of `MAX_INPUT_LENGTH`, so no overflow is possible. The code is retained as-is since it is already correct; Flawfinder flags `memcpy` on pattern-matching alone.

3. **Remaining strengths preserved:** NULL-input guard, empty/oversized input rejection, allocation failure handling with cleanup (no leak of `items` if the `Tuple` allocation fails), NULL-safe `free_tuple` and `print_tuple`, and `fflush(stdout)` to ensure output is emitted even if the process terminates abnormally or output is piped.
