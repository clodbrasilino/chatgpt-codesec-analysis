#include <stdio.h>
#include <stdlib.h>

int countElementsWithSelfModularInverse(const int *array, size_t size, int p) {
    if (array == NULL || size == 0 || p <= 1) {
        return 0;
    }

    int count = 0;
    for (size_t i = 0; i < size; i++) {
        long long val = array[i] % p;
        if (val < 0) {
            val += p;
        }

        if (val != 0 && (val * val) % p == 1) {
            count++;
        }
    }

    return count;
}

int main(void) {
    int array[] = {1, 2, 3, 4, 5, 6, 12, 13};
    size_t size = sizeof(array) / sizeof(array[0]);
    int p = 7;

    int result = countElementsWithSelfModularInverse(array, size, p);
    printf("%d\n", result);

    return 0;
}
