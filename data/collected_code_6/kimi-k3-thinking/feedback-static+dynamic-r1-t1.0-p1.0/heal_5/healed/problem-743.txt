#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int rotate_right(int *arr, size_t size, size_t positions)
{
    int *temp;
    size_t arr_bytes;
    size_t tail_bytes;
    size_t head_bytes;
    size_t temp_capacity;
    size_t arr_capacity;

    if (arr == NULL || size == 0)
    {
        return -1;
    }

    positions %= size;
    if (positions == 0)
    {
        return 0;
    }

    if (size > SIZE_MAX / sizeof(*arr))
    {
        return -1;
    }
    arr_bytes = size * sizeof(*arr);
    arr_capacity = arr_bytes;

    if (positions > SIZE_MAX / sizeof(*temp))
    {
        return -1;
    }
    tail_bytes = positions * sizeof(*temp);

    if (tail_bytes > arr_bytes)
    {
        return -1;
    }

    temp_capacity = tail_bytes;
    temp = (int *)malloc(temp_capacity);
    if (temp == NULL)
    {
        return -1;
    }

    if ((size - positions) > SIZE_MAX / sizeof(*arr))
    {
        free(temp);
        temp = NULL;
        return -1;
    }
    head_bytes = (size - positions) * sizeof(*arr);

    if (head_bytes > arr_bytes || tail_bytes > arr_bytes - head_bytes)
    {
        free(temp);
        temp = NULL;
        return -1;
    }

    if (tail_bytes > temp_capacity)
    {
        free(temp);
        temp = NULL;
        return -1;
    }
    memcpy(temp, arr + (size - positions), tail_bytes);

    if (head_bytes > arr_capacity - tail_bytes)
    {
        free(temp);
        temp = NULL;
        return -1;
    }
    memmove(arr + positions, arr, head_bytes);

    if (tail_bytes > arr_capacity)
    {
        free(temp);
        temp = NULL;
        return -1;
    }
    memcpy(arr, temp, tail_bytes);

    free(temp);
    temp = NULL;

    return 0;
}

int main(void)
{
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    const size_t size = sizeof(arr) / sizeof(arr[0]);
    const size_t positions = 3;
    size_t i;

    printf("Original list: ");
    for (i = 0; i < size; i++)
    {
        printf("%d ", arr[i]);
    }
    printf("\n");

    if (rotate_right(arr, size, positions) != 0)
    {
        fprintf(stderr, "Error: failed to rotate the list.\n");
        return EXIT_FAILURE;
    }

    printf("Rotated list:  ");
    for (i = 0; i < size; i++)
    {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return EXIT_SUCCESS;
}
