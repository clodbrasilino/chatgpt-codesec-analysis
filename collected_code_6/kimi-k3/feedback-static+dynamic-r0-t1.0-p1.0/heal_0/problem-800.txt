```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char *remove_whitespace(const char *input)
{
    if (input == NULL)
    {
        return NULL;
    }

    size_t length = strlen(input);
    char *result = malloc(length + 1);

    if (result == NULL)
    {
        return NULL;
    }

    size_t write_index = 0;

    for (size_t read_index = 0; read_index < length; read_index++)
    {
        if (!isspace((unsigned char)input[read_index]))
        {
            result[write_index] = input[read_index];
            write_index++;
        }
    }

    result[write_index] = '\0';
    return result;
}

int main(void)
{
    char input[256];

    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    char *cleaned = remove_whitespace(input);
    if (cleaned == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Result: %s\n", cleaned);
    free(cleaned);

    return EXIT_SUCCESS;
}
```
