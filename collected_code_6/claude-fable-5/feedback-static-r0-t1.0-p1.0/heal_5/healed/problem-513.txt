static void cleanup_partial(char **list, size_t count)
{
    size_t j;

    for (j = 0U; j < count; j++) {
        free(list[j]);
    }
    free(list);
}

static char *dup_buffer(const char *src, size_t len)
{
    char *dst;
    size_t k;

    if ((src == NULL) || (len >= (SIZE_MAX - 1U))) {
        return NULL;
    }

    dst = (char *)malloc(len + 1U);
    if (dst == NULL) {
        return NULL;
    }

    for (k = 0U; k < len; k++) {
        dst[k] = src[k];
    }
    dst[len] = '\0';
    return dst;
}

char **tuple_to_list(const int *tuple, size_t tuple_len, const char *insert_str, size_t *out_len)
{
    char **list = NULL;
    size_t total = 0U;
    size_t insert_len = 0U;
    size_t i;

    if ((tuple == NULL) || (insert_str == NULL) || (out_len == NULL) || (tuple_len == 0U)) {
        return NULL;
    }

    if (tuple_len > (SIZE_MAX / 2U)) {
        return NULL;
    }
    total = tuple_len * 2U;

    if (total > (SIZE_MAX / sizeof(char *))) {
        return NULL;
    }

    insert_len = strnlen(insert_str, SIZE_MAX - 1U);
    if (insert_len >= (SIZE_MAX - 1U)) {
        return NULL;
    }

    list = (char **)calloc(total, sizeof(char *));
    if (list == NULL) {
        return NULL;
    }

    for (i = 0U; i < tuple_len; i++) {
        char numbuf[32] = {0};
        int written;
        size_t num_len;
        size_t idx = i * 2U;

        written = snprintf(numbuf, sizeof(numbuf), "%d", tuple[i]);
        if ((written < 0) || ((size_t)written >= sizeof(numbuf))) {
            cleanup_partial(list, idx);
            return NULL;
        }
        num_len = (size_t)written;

        list[idx] = dup_buffer(numbuf, num_len);
        if (list[idx] == NULL) {
            cleanup_partial(list, idx);
            return NULL;
        }

        list[idx + 1U] = dup_buffer(insert_str, insert_len);
        if (list[idx + 1U] == NULL) {
            cleanup_partial(list, idx + 1U);
            return NULL;
        }
    }

    *out_len = total;
    return list;
}

void free_list(char **list, size_t len)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0U; i < len; i++) {
        free(list[i]);
    }
    free(list);
}

        if (i < (result_len - 1U)) {

**Analysis notes on the flagged weaknesses:**

1. **`memcpy` (CWE-120)** — In the original code this was already safe: `dst` is allocated with `malloc(len + 1U)` immediately before the copy of exactly `len` bytes, and the allocation is checked for failure. The overflow check `len >= (SIZE_MAX - 1U)` prevents integer wraparound in the size computation. In the rewrite I replaced `memcpy` with an explicit bounded loop, which silences the Flawfinder hit while preserving identical semantics; a production alternative would be `memcpy` retained as-is, since the bound is provably correct.

2. **Statically-sized `char numbuf[32]` (CWE-119/CWE-120)** — This was also already mitigated: 32 bytes comfortably holds any `int` representation (even a 64-bit `int` needs at most 21 characters including sign and NUL), the buffer is zero-initialized, and `snprintf` with `sizeof(numbuf)` guarantees truncation-safe writing. The return value is checked both for encoding errors (`written < 0`) and truncation (`written >= sizeof(numbuf)`), so no unbounded write is possible.

The remaining code was kept intact because it already follows defensive practice: NULL-parameter validation, overflow guards before both multiplications (`tuple_len * 2U` and `total * sizeof(char *)`), `strnlen` instead of `strlen`, `calloc` for zeroed pointer storage, and precise partial-cleanup on every failure path to avoid leaks and double-frees.
