#include <stdio.h>
#include <limits.h>

int is_abundant(int n) {
    if (n <= 1) return 0;

    long long sum = 1;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i * i != n) {
                sum += n / i;
                if (sum > INT_MAX) return 2;
            }
        }
    }
    return sum > n ? 1 : 0;
}

int main(void) {
    int num;
    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        printf("Invalid input\n");
        return 1;
    }

    int result = is_abundant(num);
    if (result == 2) {
        printf("%d causes overflow, cannot determine\n", num);
    } else if (result == 1) {
        printf("%d is an abundant number\n", num);
    } else {
        printf("%d is not an abundant number\n", num);
    }

    return 0;
}
