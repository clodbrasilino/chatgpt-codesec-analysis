#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORDS 100
#define MAX_WORD_LEN 100

int is_duplicate(char words[][MAX_WORD_LEN], int count, const char *word) {
    for (int i = 0; i < count; i++) {
        if (strcmp(words[i], word) == 0) {
            return 1;
        }
    }
    return 0;
}

int remove_duplicates(char *input[], int input_count, char output[][MAX_WORD_LEN]) {
    int output_count = 0;
    
    for (int i = 0; i < input_count; i++) {
        if (input[i] == NULL) {
            continue;
        }
        
        if (!is_duplicate(output, output_count, input[i])) {
            strncpy(output[output_count], input[i], MAX_WORD_LEN - 1);
            output[output_count][MAX_WORD_LEN - 1] = '\0';
            output_count++;
        }
    }
    
    return output_count;
}

int main() {
    char *input[] = {"apple", "banana", "apple", "cherry", "banana", "date", "apple"};
    int input_count = sizeof(input) / sizeof(input[0]);
    char output[MAX_WORDS][MAX_WORD_LEN];
    
    int output_count = remove_duplicates(input, input_count, output);
    
    printf("Original list:\n");
    for (int i = 0; i < input_count; i++) {
        printf("%s ", input[i]);
    }
    printf("\n\n");
    
    printf("List after removing duplicates:\n");
    for (int i = 0; i < output_count; i++) {
        printf("%s ", output[i]);
    }
    printf("\n");
    
    return 0;
}
