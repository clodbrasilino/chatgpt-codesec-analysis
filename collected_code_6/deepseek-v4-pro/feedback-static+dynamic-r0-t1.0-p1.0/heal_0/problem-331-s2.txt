#include <stdio.h>
#include <limits.h>

int count_unset_bits(unsigned int num) {
    int count = 0;
    unsigned int mask = 1;
    
    for (int i = 0; i < sizeof(unsigned int) * CHAR_BIT; i++) {
        if (!(num & mask)) {
            count++;
        }
        mask <<= 1;
    }
    
    return count;
}

int main(void) {
    unsigned int number;
    
    printf("Enter a number: ");
    if (scanf("%u", &number) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    int unset = count_unset_bits(number);
    printf("Number of unset bits: %d\n", unset);
    
    return 0;
}
