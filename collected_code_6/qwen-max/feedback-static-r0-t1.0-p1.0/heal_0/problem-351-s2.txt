```c
#include <stdio.h>
#include <stdlib.h>

int findFirstElementKTimes(int *arr, int n, int k) {
    int *count = (int *)calloc(n, sizeof(int));
    for (int i = 0; i < n; i++) {
        count[arr[i]]++;
        if (count[arr[i]] == k) {
            free(count);
            return arr[i];
        }
    }
    free(count);
    return -1;
}

int main() {
    int arr[] = {1, 2, 3, 2, 4, 2, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 3;
    int result = findFirstElementKTimes(arr, n, k);
    if (result != -1) {
        printf("The first element occurring %d times is: %d\n", k, result);
    } else {
        printf("No element occurs %d times.\n", k);
    }
    return 0;
}
```
