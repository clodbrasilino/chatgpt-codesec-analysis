#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool check_tuple_list(int **list, int num_tuples, int k) {
    if (list == NULL) {
        return false;
    }
    if (k <= 0) {
        return false;
    }
    for (int i = 0; i < num_tuples; i++) {
        if (list[i] == NULL) {
            return false;
        }
    }
    return true;
}

int main(void) {
    int k = 3;
    int num_tuples = 2;
    int **list = malloc(num_tuples * sizeof(int *));
    if (list == NULL) {
        return EXIT_FAILURE;
    }
    for (int i = 0; i < num_tuples; i++) {
        list[i] = malloc(k * sizeof(int));
        if (list[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(list[j]);
            }
            free(list);
            return EXIT_FAILURE;
        }
        for (int j = 0; j < k; j++) {
            list[i][j] = i * k + j;
        }
    }
    bool result = check_tuple_list(list, num_tuples, k);
    printf("%d\n", result);
    for (int i = 0; i < num_tuples; i++) {
        free(list[i]);
    }
    free(list);
    return EXIT_SUCCESS;
}
