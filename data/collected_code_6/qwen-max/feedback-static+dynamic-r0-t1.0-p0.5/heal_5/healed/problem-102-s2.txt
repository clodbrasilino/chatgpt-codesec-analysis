#include <stdio.h>
#include <string.h>
#include <ctype.h>

char* snake_to_camel(char *snake_str) {
    char *camel_str = snake_str;
    int capitalize_next = 1;

    while (*snake_str) {
        if (*snake_str == '_') {
            capitalize_next = 1;
            snake_str++;
            continue;
        }
        if (capitalize_next) {
            *camel_str = toupper((unsigned char)*snake_str);
            capitalize_next = 0;
        } else {
            *camel_str = *snake_str;
        }
        camel_str++;
        snake_str++;
    }
    *camel_str = '\0';
    return snake_str - 1;
}

int main() {
    char snake_case[] = "this_is_a_snake_case_string";
    snake_to_camel(snake_case);
    printf("%s\n", snake_case);

    char test0[] = "python_program";
    snake_to_camel(test0);
    printf("%s\n", test0);

    char test1[] = "python_language";
    snake_to_camel(test1);
    printf("%s\n", test1);

    char test2[] = "programming_language";
    snake_to_camel(test2);
    printf("%s\n", test2);

    return 0;
}
