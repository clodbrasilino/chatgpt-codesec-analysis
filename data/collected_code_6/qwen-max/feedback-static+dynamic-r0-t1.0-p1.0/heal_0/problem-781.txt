#include <stdio.h>
#include <stdbool.h>

bool is_divisor_count_even(int n) {
    int count = 0;
    for (int i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            if (i * i == n) {
                count++;
            } else {
                count += 2;
            }
        }
    }
    return count % 2 == 0;
}

int main() {
    int number;
    printf("Enter a number: ");
    scanf("%d", &number);
    if (is_divisor_count_even(number)) {
        printf("The count of divisors is even.\n");
    } else {
        printf("The count of divisors is odd.\n");
    }
    return 0;
}
