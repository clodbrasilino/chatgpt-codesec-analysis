```c
#include <stdio.h>
#include <stdlib.h>

typedef struct tuple {
    int a;
    int b;
} Tuple;

Tuple *new_tuple(int a, int b) {
    Tuple *t = malloc(sizeof(Tuple));
    if(t == NULL) {
        printf("Memory allocation for Tuple failed\n");
        exit(EXIT_FAILURE);
    } else {
        t->a = a;
        t->b = b;
    }
    return t;
}

Tuple *subtract_tuples(Tuple *t1, Tuple *t2) {
    if(t1 == NULL || t2 == NULL) {
        printf("Input Tuple pointers are NULL\n");
        exit(EXIT_FAILURE);
    } else {
        Tuple *res = new_tuple(t1->a - t2->a, t1->b - t2->b);
        return res;
    }
}

void free_tuple(Tuple *t) {
    if(t != NULL) {
        free(t);
    }
}

int main() {
    Tuple *t1 = new_tuple(20, 15);
    Tuple *t2 = new_tuple(10, 5);
    Tuple *subtracted = subtract_tuples(t1, t2);

    printf("Subtracted tuples: a = %d, b = %d\n", subtracted->a, subtracted->b);

    free_tuple(t1);
    free_tuple(t2);
    free_tuple(subtracted);

    return 0;
}
```
