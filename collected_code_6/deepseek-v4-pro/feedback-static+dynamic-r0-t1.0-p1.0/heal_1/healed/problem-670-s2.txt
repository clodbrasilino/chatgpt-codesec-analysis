#include <stdio.h>
#include <stdbool.h>

bool has_decreasing_trend(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return false;
    }
    
    for (size_t i = 1; i < size; i++) {
        if (arr[i] >= arr[i - 1]) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int arr1[] = {5, 4, 3, 2, 1};
    int arr2[] = {5, 4, 4, 2, 1};
    int arr3[] = {5, 6, 3, 2, 1};
    int arr4[] = {5};
    
    printf("%d\n", has_decreasing_trend(arr1, sizeof(arr1) / sizeof(arr1[0])));
    printf("%d\n", has_decreasing_trend(arr2, sizeof(arr2) / sizeof(arr2[0])));
    printf("%d\n", has_decreasing_trend(arr3, sizeof(arr3) / sizeof(arr3[0])));
    printf("%d\n", has_decreasing_trend(arr4, sizeof(arr4) / sizeof(arr4[0])));
    printf("%d\n", has_decreasing_trend(NULL, 0));
    
    return 0;
}
