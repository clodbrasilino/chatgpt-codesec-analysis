#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool isSubsetSumDivisibleByM(const int *arr, size_t n, int m) {
    if (arr == NULL || m <= 0) {
        return false;
    }
    
    if (n == 0) {
        return false;
    }
    
    bool *dp = calloc((size_t)m, sizeof(bool));
    if (dp == NULL) {
        return false;
    }
    
    dp[0] = true;
    
    for (size_t i = 0; i < n; i++) {
        bool *temp = calloc((size_t)m, sizeof(bool));
        if (temp == NULL) {
            free(dp);
            return false;
        }
        
        int rem = arr[i] % m;
        if (rem < 0) {
            rem += m;
        }
        
        for (int j = 0; j < m; j++) {
            if (dp[j]) {
                temp[j] = true;
                temp[(j + rem) % m] = true;
            }
        }
        
        free(dp);
        dp = temp;
    }
    
    bool result = dp[0];
    free(dp);
    return result;
}

int main(void) {
    int arr[] = {3, 1, 7, 5};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int m = 6;
    
    if (isSubsetSumDivisibleByM(arr, n, m)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }
    
    return 0;
}
