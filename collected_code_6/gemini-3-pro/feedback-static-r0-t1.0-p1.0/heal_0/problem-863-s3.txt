#include <stdio.h>
#include <stdlib.h>

int compareQuery(const void* a, const void* b) {
    return (*(int*)a - *(int*)b);
}

int findLongestConseqSubseq(int arr[], int n) {
    if (n == 0) {
        return 0;
    }

    qsort(arr, n, sizeof(int), compareQuery);

    int ans = 1;
    int count = 1;

    for (int i = 1; i < n; i++) {
        if (arr[i] != arr[i - 1]) {
            if (arr[i] == arr[i - 1] + 1) {
                count++;
            } else {
                if (count > ans) {
                    ans = count;
                }
                count = 1;
            }
        }
    }
    if (count > ans) {
        ans = count;
    }

    return ans;
}

int main(void) {
    int arr[] = {1, 9, 3, 10, 4, 20, 2};
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = findLongestConseqSubseq(arr, n);

    printf("%d\n", result);

    return 0;
}
