#include <stdio.h>
#include <stdlib.h>

int find_kth_element(int *arr, int len, int k) {
    if (k <= 0 || k > len) {
        return -1;
    }
    int odd_count = 0;
    for (int i = 0; i < len; ++i) {
        if (arr[i] % 2 != 0 && ++odd_count == k) {
            return arr[i];
        }
    }
    k -= odd_count;
    for (int i = 0; i < len; ++i) {
        if (arr[i] % 2 == 0 && k-- == 1) {
            return arr[i];
        }
    }
    return -1;
}

int main() {
    int arr[] = {1, 3, 5, 7, 2, 4, 6, 8};
    int len = sizeof(arr) / sizeof(arr[0]);
    int k = 5;
    int result = find_kth_element(arr, len, k);
    printf("The %d-th element is: %d\n", k, result);
    return 0;
}
