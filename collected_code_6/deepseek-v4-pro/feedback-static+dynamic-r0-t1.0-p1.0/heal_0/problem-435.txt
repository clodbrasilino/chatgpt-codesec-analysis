```c
#include <stdio.h>
#include <stdlib.h>

int get_last_digit(int number) {
    return abs(number % 10);
}

int main(void) {
    int number;
    
    printf("Enter a number: ");
    if (scanf("%d", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    printf("Last digit: %d\n", get_last_digit(number));
    
    return EXIT_SUCCESS;
}
```
