```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

char *find_most_frequent_word(const char *const words[], size_t count)
{
    size_t i, j;
    size_t max_count = 0;
    size_t max_index = 0;
    int found = 0;
    char *result;
    size_t word_len;

    if (words == NULL || count == 0) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        size_t current_count = 0;

        if (words[i] == NULL) {
            continue;
        }

        for (j = 0; j < count; j++) {
            if (words[j] != NULL && strcmp(words[i], words[j]) == 0) {
                current_count++;
            }
        }

        if (!found || current_count > max_count) {
            max_count = current_count;
            max_index = i;
            found = 1;
        }
    }

    if (!found || words[max_index] == NULL) {
        return NULL;
    }

    word_len = strnlen(words[max_index], MAX_WORD_LEN);

    result = malloc(word_len + 1);
    if (result == NULL) {
        return NULL;
    }

    memcpy(result, words[max_index], word_len);
    result[word_len] = '\0';

    return result;
}

int main(void)
{
    const char *words[] = {
        "apple", "banana", "apple", "cherry",
        "banana", "apple", "cherry", "banana"
    };
    size_t count = sizeof(words) / sizeof(words[0]);
    char *most_frequent;

    most_frequent = find_most_frequent_word(words, count);
    if (most_frequent == NULL) {
        fprintf(stderr, "Error: unable to determine the most frequent word\n");
        return EXIT_FAILURE;
    }

    printf("Most frequent word: %s\n", most_frequent);

    free(most_frequent);

    return EXIT_SUCCESS;
}
```

**Fixes applied:**

1. **`strlen` (CWE-126, over-read):** Replaced with `strnlen(words[max_index], MAX_WORD_LEN)`, which bounds the scan to `MAX_WORD_LEN` bytes. If the string is not properly NUL-terminated, the function will no longer read past that limit and crash or leak adjacent memory.

2. **`strcpy` (CWE-120, buffer overflow):** Replaced with `memcpy(result, words[max_index], word_len)` using the exact length computed in the previous step, followed by explicit NUL termination (`result[word_len] = '\0'`). Since the length is known and the allocation is exactly `word_len + 1`, the copy is fully bounded and cannot overflow.

3. **Defensive check:** Added `words[max_index] == NULL` verification before dereferencing, ensuring the pointer is valid even if the function logic is later modified.
