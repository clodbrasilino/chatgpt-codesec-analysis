#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* move_numbers(char *str) {
    if (str == NULL) return NULL;
    size_t len = strlen(str);
    char *numStr = (char *)malloc(len + 1);
    if (numStr == NULL) return NULL;
    numStr[0] = '\0';

    char *nonNumStr = (char *)malloc(len + 1);
    if (nonNumStr == NULL) {
        free(numStr);
        return NULL;
    }
    nonNumStr[0] = '\0';

    int numLen = 0, nonNumLen = 0;
    for (size_t i = 0; i < len; ++i) {
        if (isdigit((unsigned char)str[i])) {
            numStr[numLen++] = str[i];
        } else {
            nonNumStr[nonNumLen++] = str[i];
        }
    }
    numStr[numLen] = '\0';
    nonNumStr[nonNumLen] = '\0';

    snprintf(str, len + 1, "%s%s", nonNumStr, numStr);

    free(numStr);
    free(nonNumStr);
    return str;
}

int main() {
    char str[] = "Hello123World456";
    char *result = move_numbers(str);
    if (result != NULL) {
        printf("%s\n", result);
    } else {
        fprintf(stderr, "Memory allocation failed\n");
    }
    return 0;
}
