```C
#include <stdio.h>

double harmonicSum(int n){
    double result = 0.0;

    if(n < 1) {
        printf("Sorry, number has to be greater than 1");
        return -1;
    }

    for(int i = 1; i < n; i++) {
        result += 1.0/i;
    }
    return result;
}

int main(){
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);
    
    double result = harmonicSum(n);
    
    if(result != -1) {
        printf("The harmonic sum of numbers till %d is: %lf\n", n, result);
    }
    
    return 0;
}
```
