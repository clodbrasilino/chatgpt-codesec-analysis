```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64
#define FLUSH_LIMIT 4096

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (long)(c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (long)(c - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t count = 0;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses — both are already mitigated:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The `flush_stdin()` function is safe as written. It never stores the read characters into a buffer, so no overflow is possible. It also terminates correctly on `\n`, `EOF`, and additionally caps iterations at `FLUSH_LIMIT`, preventing unbounded looping on a pathological stream. No change required.

2. **Statically-sized `char input[INPUT_SIZE]` (CWE-119/CWE-120):** The buffer usage is already bounds-safe:
   - Input is read with `fgets(input, sizeof(input), stdin)`, which limits writes to the buffer size.
   - Explicit null termination is enforced at `input[sizeof(input) - 1]`.
   - Length is measured with `strnlen()` (never `strlen()` on potentially unterminated data).
   - Oversized input is detected (`len == sizeof(input) - 1` with no trailing newline), the remainder of the line is drained, and the program rejects the input rather than silently truncating.

3. **Additional robustness already present:** `octal_to_decimal()` validates its pointers, rejects empty and unterminated strings, validates each character is in `'0'`–`'7'`, and performs overflow checking against `LONG_MAX` before each accumulation step (CWE-190 mitigation).
