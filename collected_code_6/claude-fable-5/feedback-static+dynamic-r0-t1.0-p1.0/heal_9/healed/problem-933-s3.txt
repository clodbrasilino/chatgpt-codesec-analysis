#define MAX_INPUT_LEN 4096U

char *camel_to_snake(const char *input)
{
    char *output = NULL;
    size_t input_len = 0U;
    size_t out_capacity;
    size_t out_len = 0U;
    size_t i;

    if (input == NULL) {
        return NULL;
    }

    while ((input_len < MAX_INPUT_LEN) && (input[input_len] != '\0')) {
        input_len++;
    }
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (input_len > ((SIZE_MAX - 1U) / 2U)) {
        return NULL;
    }
    out_capacity = (input_len * 2U) + 1U;

    output = (char *)malloc(out_capacity);
    if (output == NULL) {
        return NULL;
    }

    for (i = 0U; i < input_len; i++) {
        unsigned char c = (unsigned char)input[i];

        if (isupper((int)c) != 0) {
            if (i > 0U) {
                if (out_len >= (out_capacity - 1U)) {
                    free(output);
                    return NULL;
                }
                output[out_len] = '_';
                out_len++;
            }

                free(output);
                return NULL;
            }
            output[out_len] = (char)tolower((int)c);
            out_len++;
        } else {
            if (out_len >= (out_capacity - 1U)) {
                free(output);
                return NULL;
            }
            output[out_len] = (char)c;
            out_len++;
        }
    }

    output[out_len] = '\0';
    return output;
}
