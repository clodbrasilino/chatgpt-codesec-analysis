#include <stdlib.h>
#include <string.h>

char *remove_parenthesis_area(const char *str) {
    if (str == NULL) {
        return NULL;
    }

    size_t len = 0;
    size_t result_len = 0;
    int depth = 0;
    const char *p = str;

    while (*p) {
        if (*p == '(') {
            depth++;
            p++;
            continue;
        }
        if (*p == ')') {
            if (depth > 0) {
                depth--;
            }
            p++;
            continue;
        }
        if (depth == 0) {
            result_len++;
        }
        p++;
    }

    char *result = (char *)malloc(result_len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t j = 0;
    depth = 0;
    p = str;

    while (*p) {
        if (*p == '(') {
            depth++;
            p++;
            continue;
        }
        if (*p == ')') {
            if (depth > 0) {
                depth--;
            }
            p++;
            continue;
        }
        if (depth == 0) {
            result[j++] = *p;
        }
        p++;
    }

    result[j] = '\0';
    return result;
}
