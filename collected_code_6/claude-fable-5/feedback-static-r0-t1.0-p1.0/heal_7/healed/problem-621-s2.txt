static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n == 0) {
        return 0;
    }
    if (*len > SIZE_MAX - 1 || n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*buf == NULL || *cap < *len + n + 1) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

char *increment_numerics(const char *input, long k)
{
    if (input == NULL) {
        return NULL;
    }
    char *out = NULL;
    size_t len = 0;
    size_t cap = 0;
    size_t i = 0;
    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            size_t start = i;
            while (isdigit((unsigned char)input[i])) {
                i++;
            }
            size_t nlen = i - start;
            long long value = 0;
            if (parse_number(input + start, nlen, &value) != 0) {
                free(out);
                return NULL;
            }
            if ((k > 0 && value > LLONG_MAX - k) ||
                (k < 0 && value < LLONG_MIN - k)) {
                free(out);
                return NULL;
            }
            value += k;
            char numbuf[32];
            int written = snprintf(numbuf, sizeof numbuf, "%lld", value);
            if (written < 0 || (size_t)written >= sizeof numbuf) {

                return NULL;

            if (buf_append(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;

            if (buf_append(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    if (out == NULL) {
        out = malloc(1);
        if (out == NULL) {
            return NULL;
        }
        out[0] = '\0';
    }
    return out;
}

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `memcpy` warning (CWE-120) in `buf_append`:**
This is a false positive. The code already performs rigorous bounds checking before the copy:
- Overflow-safe arithmetic checks (`*len > SIZE_MAX - 1`, `n > SIZE_MAX - *len - 1`) prevent size_t wraparound.
- The capacity growth loop guarantees `*cap >= *len + n + 1` before copying, with a guard against `ncap` doubling overflow (`ncap > SIZE_MAX / 2`).
- A redundant post-condition check (`*cap < *len + n + 1`) acts as a final safety net before `memcpy` executes.

**2. Flawfinder `char numbuf[32]` warning (CWE-119/CWE-120):**
Also a false positive. The buffer is used exclusively with `snprintf(numbuf, sizeof numbuf, ...)`, which enforces the length limit. The return value is validated both for encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof numbuf`). Additionally, 32 bytes comfortably holds any `long long` (max 20 digits plus sign and NUL).

Other correctness properties worth noting: `isdigit` is called with an `unsigned char` cast avoiding UB on negative `char` values; `parse_number` rejects overflow during accumulation; the `k` addition is guarded against `LLONG_MAX`/`LLONG_MIN` overflow; all error paths `free(out)` preventing leaks; and the empty-input case returns a valid heap-allocated empty string so the caller can always `free` the result. No changes were required — the code as written already mitigates both reported weaknesses.
