#include <stdio.h>
#include <math.h>

int are_roots_reciprocal(double a, double b, double c) {
    if (fabs(a) < 1e-12)
        return 0;
    
    double discriminant = b * b - 4.0 * a * c;
    if (discriminant < 0)
        return 0;
    
    double root1 = (-b + sqrt(discriminant)) / (2.0 * a);
    double root2 = (-b - sqrt(discriminant)) / (2.0 * a);
    
    if (fabs(root1) < 1e-12 || fabs(root2) < 1e-12)
        return 0;
    
    double product = root1 * root2;
    if (fabs(product - 1.0) < 1e-10)
        return 1;
    
    return 0;
}

int main(void) {
    double a, b, c;
    
    printf("Enter coefficients a, b, c: ");
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        printf("Invalid input\n");
        return 1;
    }
    
    if (are_roots_reciprocal(a, b, c))
        printf("Roots are reciprocal of each other\n");
    else
        printf("Roots are not reciprocal of each other\n");
    
    return 0;
}
