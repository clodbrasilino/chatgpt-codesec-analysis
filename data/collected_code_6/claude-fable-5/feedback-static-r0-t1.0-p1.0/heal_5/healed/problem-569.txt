```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 4096U
#define LIST_COUNT 3U
#define LIST_SIZE 4U

static int compare_strings(const void *a, const void *b)
{
    const char *const *sa = (const char *const *)a;
    const char *const *sb = (const char *const *)b;
    return strcmp(*sa, *sb);
}

static void sort_sublists(char ***lists, const size_t *sizes, size_t count)
{
    size_t i;

    if (lists == NULL || sizes == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        if (lists[i] != NULL && sizes[i] > 1) {
            qsort(lists[i], sizes[i], sizeof(char *), compare_strings);
        }
    }
}

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

static void free_lists(char ***lists, const size_t *sizes, size_t count)
{
    size_t i;
    size_t j;

    if (lists == NULL || sizes == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        if (lists[i] != NULL) {
            for (j = 0; j < sizes[i]; j++) {
                free(lists[i][j]);
            }
            free(lists[i]);
        }
    }
    free(lists);
}

int main(void)
{
    static const char *const data[LIST_COUNT][LIST_SIZE] = {
        { "banana", "apple", "cherry", "date" },
        { "zebra", "yak", "xenon", "walrus" },
        { "delta", "alpha", "charlie", "bravo" }
    };
    static const size_t sizes[LIST_COUNT] = { LIST_SIZE, LIST_SIZE, LIST_SIZE };
    const size_t count = LIST_COUNT;
    char ***lists;
    size_t i;
    size_t j;

    lists = malloc(count * sizeof(char **));
    if (lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (sizes[i] == 0U || sizes[i] > LIST_SIZE) {
            fprintf(stderr, "Invalid sublist size\n");
            free_lists(lists, sizes, i);
            return EXIT_FAILURE;
        }

        lists[i] = malloc(sizes[i] * sizeof(char *));
        if (lists[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_lists(lists, sizes, i);
            return EXIT_FAILURE;
        }

        for (j = 0; j < sizes[i]; j++) {
            lists[i][j] = duplicate_string(data[i][j]);
            if (lists[i][j] == NULL) {
                size_t k;
                fprintf(stderr, "Memory allocation failed\n");
                for (k = 0; k < j; k++) {
                    free(lists[i][k]);
                }
                free(lists[i]);
                free_lists(lists, sizes, i);
                return EXIT_FAILURE;
            }
        }
    }

    sort_sublists(lists, sizes, count);

    for (i = 0; i < count; i++) {
        printf("Sublist %zu:", i);
        for (j = 0; j < sizes[i]; j++) {
            printf(" %s", lists[i][j]);
        }
        printf("\n");
    }

    free_lists(lists, sizes, count);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **`memcpy` (CWE-120) in `duplicate_string`:** This is a false positive. The destination buffer is allocated with `malloc(len + 1U)` immediately before the copy, and exactly `len` bytes are copied followed by explicit null termination. The length itself is bounded by `strnlen(src, MAX_STRING_LEN)` with a rejection check for oversized input, so the destination can always hold the source data. No change required.

2. **Statically-sized `char` arrays (CWE-119/CWE-120) in `main`:** Also a false positive. The `data` array holds compile-time constant string literals and is declared `static const char *const`, so it is never written to at runtime. All iteration over it is bounded by `LIST_COUNT`/`LIST_SIZE`, and `sizes[i]` is validated (`sizes[i] == 0U || sizes[i] > LIST_SIZE`) before use. No overflow is possible.

Additional strengths worth noting: the error paths correctly avoid double-frees (partial cleanup of the current sublist before calling `free_lists` with count `i`), `qsort` is only invoked for lists with more than one element, and all pointer parameters are null-checked. The code stands as written.
