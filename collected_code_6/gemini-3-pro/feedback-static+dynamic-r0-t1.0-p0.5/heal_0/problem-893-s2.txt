#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct Sublist {
    Node* head;
    struct Sublist* next;
} Sublist;

int* get_last_elements(Sublist* list, int* count) {
    if (!list || !count) {
        if (count) *count = 0;
        return NULL;
    }

    int list_count = 0;
    Sublist* current_sublist = list;
    while (current_sublist) {
        list_count++;
        current_sublist = current_sublist->next;
    }

    int* last_elements = (int*)malloc(list_count * sizeof(int));
    if (!last_elements) {
        *count = 0;
        return NULL;
    }

    current_sublist = list;
    int index = 0;
    while (current_sublist) {
        Node* current_node = current_sublist->head;
        if (!current_node) {
            free(last_elements);
            *count = 0;
            return NULL;
        }
        while (current_node->next) {
            current_node = current_node->next;
        }
        last_elements[index++] = current_node->data;
        current_sublist = current_sublist->next;
    }

    *count = list_count;
    return last_elements;
}

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) return NULL;
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

Sublist* create_sublist(Node* head) {
    Sublist* new_sublist = (Sublist*)malloc(sizeof(Sublist));
    if (!new_sublist) return NULL;
    new_sublist->head = head;
    new_sublist->next = NULL;
    return new_sublist;
}

void free_list(Sublist* list) {
    while (list) {
        Node* current_node = list->head;
        while (current_node) {
            Node* temp = current_node;
            current_node = current_node->next;
            free(temp);
        }
        Sublist* temp_sublist = list;
        list = list->next;
        free(temp_sublist);
    }
}

int main() {
    Node* n1 = create_node(1);
    if (!n1) return 1;
    n1->next = create_node(2);
    if (!n1->next) { free(n1); return 1; }
    n1->next->next = create_node(3);
    if (!n1->next->next) { free(n1->next); free(n1); return 1; }

    Node* n2 = create_node(4);
    if (!n2) { free_list(create_sublist(n1)); return 1; }
    n2->next = create_node(5);
    if (!n2->next) { free(n2); free_list(create_sublist(n1)); return 1; }

    Sublist* s1 = create_sublist(n1);
    if (!s1) { free_list(create_sublist(n1)); free_list(create_sublist(n2)); return 1; }
    Sublist* s2 = create_sublist(n2);
    if (!s2) { free_list(s1); free_list(create_sublist(n2)); return 1; }
    s1->next = s2;

    int count = 0;
    int* last_elements = get_last_elements(s1, &count);

    if (last_elements) {
        for (int i = 0; i < count; i++) {
            printf("%d\n", last_elements[i]);
        }
        free(last_elements);
    }

    free_list(s1);

    return 0;
}
