#include <stdio.h>
#include <stdlib.h>

struct Node {
    int value;
    struct Node *next;
};

struct Node *create_node(int value)
{
    struct Node *node = malloc(sizeof(struct Node));
    if (node == NULL) {
        return NULL;
    }
    node->value = value;
    node->next = NULL;
    return node;
}

int append_node(struct Node **head, int value)
{
    struct Node *node;
    struct Node *current;

    if (head == NULL) {
        return -1;
    }

    node = create_node(value);
    if (node == NULL) {
        return -1;
    }

    if (*head == NULL) {
        *head = node;
        return 0;
    }

    current = *head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = node;
    return 0;
}

int get_nth_item(const struct Node *head, size_t n, int *out)
{
    const struct Node *current = head;
    size_t index = 0;

    if (out == NULL) {
        return -1;
    }

    while (current != NULL) {
        if (index == n) {
            *out = current->value;
            return 0;
        }
        index++;
        current = current->next;
    }
    return -1;
}

void free_list(struct Node *head)
{
    struct Node *next;

    while (head != NULL) {
        next = head->next;
        free(head);
        head = next;
    }
}

int main(void)
{
    struct Node *head = NULL;
    int values[] = {10, 20, 30, 40, 50};
    size_t count = sizeof(values) / sizeof(values[0]);
    size_t i;
    size_t n = 2;
    int result;

    for (i = 0; i < count; i++) {
        if (append_node(&head, values[i]) != 0) {
            fprintf(stderr, "Failed to append node\n");
            free_list(head);
            return EXIT_FAILURE;
        }
    }

    if (get_nth_item(head, n, &result) != 0) {
        fprintf(stderr, "Index %zu out of range\n", n);
        free_list(head);
        return EXIT_FAILURE;
    }

    printf("Item at index %zu: %d\n", n, result);

    free_list(head);
    return EXIT_SUCCESS;
}
