```c
#include <stdio.h>
#include <stdlib.h>

int gcd(int a, int b) {
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int are_coprime(int a, int b) {
    return gcd(a, b) == 1;
}

int main() {
    int num1, num2;
    
    if (scanf("%d %d", &num1, &num2) != 2) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (num1 <= 0 || num2 <= 0) {
        fprintf(stderr, "Numbers must be positive integers\n");
        return EXIT_FAILURE;
    }
    
    if (are_coprime(num1, num2)) {
        printf("%d and %d are co-prime\n", num1, num2);
    } else {
        printf("%d and %d are not co-prime\n", num1, num2);
    }
    
    return EXIT_SUCCESS;
}
```
