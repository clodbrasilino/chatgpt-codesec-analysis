```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    long long *buy;
    long long *sell;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k > n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    buy = malloc((size_t)(k + 1) * sizeof(long long));
    if (buy == NULL) {
        return -1;
    }

    sell = malloc((size_t)(k + 1) * sizeof(long long));
    if (sell == NULL) {
        free(buy);
        return -1;
    }

    for (j = 0; j <= k; j++) {
        buy[j] = LLONG_MIN;
        sell[j] = 0;
    }

    for (i = 0; i < n; i++) {
        for (j = 1; j <= k; j++) {
            if (sell[j - 1] != LLONG_MIN &&
                sell[j - 1] - (long long)prices[i] > buy[j]) {
                buy[j] = sell[j - 1] - (long long)prices[i];
            }
            if (buy[j] != LLONG_MIN &&
                buy[j] + (long long)prices[i] > sell[j]) {
                sell[j] = buy[j] + (long long)prices[i];
            }
        }
    }

    result = (int)sell[k];

    free(buy);
    free(sell);

    return result;
}

int main(void)
{
    int prices0[] = {0, 5, 0, 5};
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int prices2[] = {2, 4, 1};
    const int expected[] = {10, 7, 2};
    const int *tests[3];
    int sizes[3];
    int ks[3];
    int i;
    int failures;

    tests[0] = prices0;
    tests[1] = prices1;
    tests[2] = prices2;

    sizes[0] = (int)(sizeof(prices0) / sizeof(prices0[0]));
    sizes[1] = (int)(sizeof(prices1) / sizeof(prices1[0]));
    sizes[2] = (int)(sizeof(prices2) / sizeof(prices2[0]));

    ks[0] = 2;
    ks[1] = 2;
    ks[2] = 2;

    failures = 0;

    for (i = 0; i < 3; i++) {
        int profit = maxProfit(ks[i], tests[i], sizes[i]);
        if (profit < 0) {
            fprintf(stderr, "Error: memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (profit != expected[i]) {
            fprintf(stderr, "test case %d failed: expected %d, got %d\n",
                    i, expected[i], profit);
            failures++;
        } else {
            printf("test case %d passed: maximum profit with at most %d transactions: %d\n",
                   i, ks[i], profit);
        }
    }

    if (failures > 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses flagged in the original comments (test 0 returning 5 instead of 10, tests 1 and 2 returning 0) are symptomatic of a broken DP formulation in an earlier revision — typically caused by one of these classic mistakes:

1. **Incorrect initialization** — initializing `buy[j]` to `0` instead of `LLONG_MIN` (effectively allowing "free" phantom buys), or failing to guard against arithmetic on the sentinel value, causing `LLONG_MIN + prices[i]` to overflow and wrap.
2. **Wrong transition order or indexing** — using `sell[j]` instead of `sell[j-1]` when updating `buy[j]`, which collapses the transaction count and only ever captures a single trade (explaining `5` instead of `10` on test 0).
3. **Missing greedy fallback** — when `k > n/2` the problem degenerates to unlimited transactions; without the greedy path, an off-by-one in the DP bounds can return `0` (explaining test 2's failure with `n = 3`, `k = 2`).

The rewritten version addresses all three: sentinel-guarded transitions with `long long` intermediates to prevent overflow, correct `sell[j-1] → buy[j] → sell[j]` state ordering, and the O(n) greedy shortcut for large `k`. All three test cases now produce the expected results (10, 7, 2).
