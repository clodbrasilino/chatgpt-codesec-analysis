```c
#include <stdio.h>
#include <stdlib.h>

void print_negative_numbers(const int *numbers, size_t length) {
    if (numbers == NULL) {
        return;
    }
    
    for (size_t i = 0; i < length; i++) {
        if (numbers[i] < 0) {
            printf("%d\n", numbers[i]);
        }
    }
}

int main(void) {
    int test_numbers[] = {5, -3, 10, -8, 0, -1, 7, -15, 2};
    size_t length = sizeof(test_numbers) / sizeof(test_numbers[0]);
    
    print_negative_numbers(test_numbers, length);
    
    return 0;
}
```
