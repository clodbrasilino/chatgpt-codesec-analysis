#include <stdio.h>
#include <string.h>
#include <ctype.h>

void capitalizeFirstLast(char *str) {
    int len = strlen(str);
    for (int i = 0, wordStart = 0; i <= len; i++) {
        if (i == 0 || str[i-1] == ' ') {
            wordStart = i;
            str[i] = toupper(str[i]);
        } else if (str[i] == ' ' || str[i] == '\0') {
            if (wordStart < i - 1) {
                str[i-1] = toupper(str[i-1]);
            }
        }
    }
}

int main() {
    char str[] = "hello world this is a test";
    capitalizeFirstLast(str);
    printf("%s\n", str);
    return 0;
}
