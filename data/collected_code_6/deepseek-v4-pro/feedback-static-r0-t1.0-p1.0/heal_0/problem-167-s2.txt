#include <stdio.h>
#include <stdint.h>

unsigned int smallest_power_of_two(unsigned int n) {
    if (n == 0) return 1;
    
    n--;
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    
    return n + 1;
}

int main(void) {
    unsigned int test_values[] = {0, 1, 2, 3, 4, 5, 8, 9, 16, 17, 31, 32, 33, 100, 255, 256, 1024, 1025};
    size_t num_tests = sizeof(test_values) / sizeof(test_values[0]);
    
    for (size_t i = 0; i < num_tests; i++) {
        unsigned int input = test_values[i];
        unsigned int result = smallest_power_of_two(input);
        printf("smallest_power_of_two(%u) = %u\n", input, result);
    }
    
    return 0;
}
