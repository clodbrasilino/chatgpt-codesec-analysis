#include <stdio.h>
#include <stdbool.h>
#include <limits.h>
#include <math.h>

bool is_woodall(unsigned long long n) {
    if (n == 0) return false;
    if (n == 1) return true;
    
    for (int k = 1; k < 64; k++) {
        unsigned long long power = 1ULL << k;
        if (power > (n + 1ULL) / (unsigned long long)k) break;
        
        unsigned long long product = (unsigned long long)k * power;
        if (product - 1ULL == n) return true;
    }
    
    return false;
}

int main(void) {
    unsigned long long num;
    
    printf("Enter a number: ");
    if (scanf("%llu", &num) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    if (is_woodall(num)) {
        printf("%llu is a Woodall number\n", num);
    } else {
        printf("%llu is not a Woodall number\n", num);
    }
    
    return 0;
}
