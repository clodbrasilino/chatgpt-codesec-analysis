```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int pivot;
    int swap_idx;
    char tmp;
    int lo;
    int hi;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    pivot = -1;
    for (i = len - 2; i >= 0; i--) {
        if (digits[i] < digits[i + 1]) {
            pivot = i;
            break;
        }
    }

    if (pivot == -1) {
        return -1;
    }

    swap_idx = -1;
    for (i = len - 1; i > pivot; i--) {
        if (digits[i] > digits[pivot]) {
            swap_idx = i;
            break;
        }
    }

    if (swap_idx == -1) {
        return -1;
    }

    tmp = digits[pivot];
    digits[pivot] = digits[swap_idx];
    digits[swap_idx] = tmp;

    lo = pivot + 1;
    hi = len - 1;
    while (lo < hi) {
        tmp = digits[lo];
        digits[lo] = digits[hi];
        digits[hi] = tmp;
        lo++;
        hi--;
    }

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        long long res = next_bigger(inputs[i]);
        if (res == -1) {
            if (printf("%lld -> no bigger number possible\n", inputs[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("%lld -> %lld\n", inputs[i], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already well-hardened against the flagged weakness (CWE-119/CWE-120, buffer handling with a statically-sized array). The mitigations already in place, which I have preserved, are:

1. **Bounded write via `snprintf`**: The conversion of `n` into `digits` uses `snprintf(digits, sizeof(digits), ...)`, which guarantees no write beyond the buffer. A 64-bit `long long` needs at most 20 characters plus a NUL, so `DIGITS_BUF_SIZE` of 32 is sufficient with margin.

2. **Truncation/error check**: The return value of `snprintf` is validated (`len < 0 || (size_t)len >= sizeof(digits)`), rejecting both encoding errors and truncated output before any indexing occurs.

3. **Buffer initialization**: `memset` zeroes the buffer, ensuring no uninitialized reads even in edge cases.

4. **Index discipline**: All loops index within `[0, len)`, where `len` has been proven strictly less than `sizeof(digits)`.

5. **Overflow-safe reconstruction**: The rebuild of the result checks `result > (LLONG_MAX - d) / 10` before each multiply-add, preventing signed integer overflow (undefined behavior), and validates each digit is in `'0'..'9'`.

6. **Negative input rejection**: `n < 0` is rejected early, avoiding a `'-'` sign appearing in the digit buffer.

7. **Checked I/O in `main`**: `printf` return values are verified.

No functional or security changes were required; the code is a correct and safe implementation of the "next bigger number with the same digits" algorithm.
