#include <stdio.h>
#include <string.h>

void convertDateFormat(char *date) {
    if (strlen(date) == 8) {
        char temp[9];
        strncpy(temp, date, 8);
        temp[8] = '\0';

        if (strlen(date) >= 11) {
            memmove(date + 3, date + 5, 3);
            date[6] = date[0];
            date[7] = date[1];

            if (strlen(temp) >= 2) {
                size_t len = strlen(temp);
                if (len <= 3) {
                    memcpy(date + 6, temp, 2);
                }
            }
            date[8] = '-';
            date[9] = date[2];
            date[10] = date[3];
            date[11] = '\0';
        }
    }
}

int main() {
    char date[] = "2023-04-01";
    convertDateFormat(date);
    printf("%s\n", date);
    return 0;
}
