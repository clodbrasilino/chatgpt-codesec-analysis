#include <stdio.h>
#include <stdlib.h>

double zero_ratio(const int *arr, size_t size) {
    size_t zero_count = 0;
    size_t i;

    if (arr == NULL || size == 0) {
        return 0.0;
    }

    for (i = 0; i < size; i++) {
        if (arr[i] == 0) {
            zero_count++;
        }
    }

    return (double)zero_count / (double)size;
}

int main(void) {
    int arr[] = {1, 0, 3, 0, 5, 0, 7, 8, 9, 10};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    double ratio;

    ratio = zero_ratio(arr, size);

    printf("Ratio of zeroes: %.2f\n", ratio);

    return EXIT_SUCCESS;
}
