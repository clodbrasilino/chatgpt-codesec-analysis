#include <stdio.h>
#include <stdlib.h>

long long sum_of_absolute_differences(int *arr, size_t n) {
    if (arr == NULL || n == 0) {
        return 0;
    }

    long long sum = 0;
    size_t i, j;

    for (i = 0; i < n; i++) {
        for (j = i + 1; j < n; j++) {
            long long diff = (long long)arr[i] - arr[j];
            sum += diff > 0 ? diff : -diff;
        }
    }

    return sum;
}

int main(void) {
    size_t n;
    if (scanf("%zu", &n) != 1) {
        return 1;
    }

    if (n == 0) {
        printf("0\n");
        return 0;
    }

    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        return 1;
    }

    for (size_t i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            free(arr);
            return 1;
        }
    }

    long long result = sum_of_absolute_differences(arr, n);
    printf("%lld\n", result);

    free(arr);
    return 0;
}
