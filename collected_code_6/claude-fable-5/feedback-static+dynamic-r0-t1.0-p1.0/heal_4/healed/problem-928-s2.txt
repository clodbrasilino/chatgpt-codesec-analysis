```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_LEN 11
#define INPUT_BUF_LEN 64

static int is_valid_format(const char *date)
{
    size_t i;
    size_t len;

    if (date == NULL) {
        return 0;
    }

    len = strnlen(date, DATE_LEN);
    if (len != DATE_LEN - 1) {
        return 0;
    }

    for (i = 0; i < DATE_LEN - 1; i++) {
        if (i == 4 || i == 7) {
            if (date[i] != '-') {
                return 0;
            }
        } else {
            if (!isdigit((unsigned char)date[i])) {
                return 0;
            }
        }
    }

    return 1;
}

static int is_valid_date(int year, int month, int day)
{
    int days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (year < 1 || month < 1 || month > 12 || day < 1) {
        return 0;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        days_in_month[1] = 29;
    }

    if (day > days_in_month[month - 1]) {
        return 0;
    }

    return 1;
}

static int convert_date(const char *input, char *output, size_t output_size)
{
    int year;
    int month;
    int day;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_LEN) {
        return -1;
    }

    if (!is_valid_format(input)) {
        return -1;
    }

    if (sscanf(input, "%4d-%2d-%2d", &year, &month, &day) != 3) {
        return -1;
    }

    if (!is_valid_date(year, month, day)) {
        return -1;
    }

    written = snprintf(output, output_size, "%02d-%02d-%04d", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_LEN];
    char output[DATE_LEN];

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date: %s\n", input);
        return EXIT_FAILURE;
    }

    printf("Converted date: %s\n", output);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warnings flagged in the original comments (CWE-119/CWE-120 on the statically-sized `input` and `output` buffers) are already mitigated in this code — no changes to the logic were required:

1. **`input[INPUT_BUF_LEN]`** — Reads are performed with `fgets(input, sizeof(input), stdin)`, which strictly bounds the write to the buffer size and always NUL-terminates. The explicit `input[sizeof(input) - 1] = '\0'` is redundant but harmless defense-in-depth, and `strcspn` safely strips the newline without risk of out-of-bounds access.

2. **`output[DATE_LEN]`** — All writes go through `snprintf(output, output_size, ...)`, with the size passed explicitly via `sizeof(output)`. The return value is checked for both encoding errors (`written < 0`) and truncation (`written >= output_size`), so overflow is impossible.

3. **Additional hardening already present:**
   - `strnlen` bounds the length scan instead of `strlen`.
   - `sscanf` uses field-width limits (`%4d`, `%2d`) and its return value is verified.
   - `isdigit` receives an `unsigned char` cast, avoiding UB on negative `char` values.
   - `convert_date` validates all pointers and the minimum output size before use.
