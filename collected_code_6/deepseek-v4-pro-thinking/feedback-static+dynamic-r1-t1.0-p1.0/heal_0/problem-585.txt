#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int id;
    double price;
} Item;

static int item_less(const Item *a, const Item *b)
{
    return a->price < b->price;
}

static void heap_swap(Item *a, Item *b)
{
    Item temp = *a;
    *a = *b;
    *b = temp;
}

static void heap_sift_down(Item *heap, size_t size, size_t index)
{
    while (index < size) {
        size_t left = index * 2 + 1;
        size_t right = index * 2 + 2;
        size_t smallest = index;

        if (left < size && item_less(&heap[left], &heap[smallest])) {
            smallest = left;
        }
        if (right < size && item_less(&heap[right], &heap[smallest])) {
            smallest = right;
        }
        if (smallest == index) {
            break;
        }
        heap_swap(&heap[index], &heap[smallest]);
        index = smallest;
    }
}

size_t find_most_expensive(const Item *items, size_t num_items, size_t n, Item *result)
{
    if (items == NULL || result == NULL || num_items == 0 || n == 0) {
        return 0;
    }

    size_t count = n < num_items ? n : num_items;
    size_t i;

    for (i = 0; i < count; ++i) {
        result[i] = items[i];
    }

    for (i = count / 2; i > 0; --i) {
        heap_sift_down(result, count, i - 1);
    }

    for (i = count; i < num_items; ++i) {
        if (item_less(&result[0], &items[i])) {
            result[0] = items[i];
            heap_sift_down(result, count, 0);
        }
    }

    for (i = count; i > 1; --i) {
        heap_swap(&result[0], &result[i - 1]);
        heap_sift_down(result, i - 1, 0);
    }

    return count;
}

int main(void)
{
    Item items[] = {
        {1, 19.99},
        {2, 5.49},
        {3, 100.00},
        {4, 42.42},
        {5, 7.25},
        {6, 99.99},
        {7, 12.34}
    };
    size_t num_items = sizeof(items) / sizeof(items[0]);
    size_t n = 3;
    Item *top = malloc(n * sizeof(Item));

    if (top == NULL) {
        return 1;
    }

    size_t count = find_most_expensive(items, num_items, n, top);

    printf("Top %zu most expensive items:\n", count);
    for (size_t i = 0; i < count; ++i) {
        printf("Item %d: %.2f\n", top[i].id, top[i].price);
    }

    free(top);
    return 0;
}
