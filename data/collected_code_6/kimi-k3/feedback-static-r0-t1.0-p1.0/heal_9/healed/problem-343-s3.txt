#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

void count_digits_and_letters(const char *str, int *digit_count, int *letter_count) {
    size_t i;
    *digit_count = 0;
    *letter_count = 0;
    for (i = 0; str[i] != '\0'; i++) {
        if (isdigit((unsigned char)str[i])) {
            (*digit_count)++;
        } else if (isalpha((unsigned char)str[i])) {
            (*letter_count)++;
        }
    }
}

int main(void) {
    char *str = NULL;
    size_t size = 0;
    ssize_t len;
    int digit_count, letter_count;
    
    printf("Enter a string: ");
    
    len = getline(&str, &size, stdin);
    if (len != -1) {
        if (str[len - 1] == '\n') {
            str[len - 1] = '\0';
        }
        count_digits_and_letters(str, &digit_count, &letter_count);
        printf("Number of digits: %d\n", digit_count);
        printf("Number of letters: %d\n", letter_count);
    }
    
    free(str);
    return 0;
}
