#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;
    char tmp;
    char *endptr;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }
    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }
    if (j <= i) {
        return -1;
    }

    tmp = digits[i];
    digits[i] = digits[j];
    digits[j] = tmp;

    for (int a = i + 1, b = len - 1; a < b; a++, b--) {
        tmp = digits[a];
        digits[a] = digits[b];
        digits[b] = tmp;
    }

    errno = 0;
    result = strtoll(digits, &endptr, 10);
    if (errno == ERANGE || endptr == digits || *endptr != '\0') {
        return -1;
    }

    if (result <= n) {
        return -1;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            if (printf("next_bigger(%lld) = no bigger number exists\n", inputs[k]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("next_bigger(%lld) = %lld\n", inputs[k], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

**CWE-119/CWE-120 (buffer overflow risk on `char digits[DIGITS_BUF_SIZE]`):** The Flawfinder warning is a heuristic triggered by any statically-sized character array. In this code the risk is already fully mitigated:

1. **Bounded write:** `snprintf(digits, sizeof(digits), "%lld", n)` limits output to the buffer size and always NUL-terminates. A `long long` maximum (`9223372036854775807`) requires 19 digits plus terminator, well within the 32-byte buffer.
2. **Truncation check:** The return value of `snprintf` is validated (`len < 0 || (size_t)len >= sizeof(digits)`), so any truncation or encoding error causes an early bail-out rather than operating on partial data.
3. **Pre-zeroing:** `memset` guarantees the buffer contains no indeterminate bytes before use.
4. **Index safety:** All loop indices (`i`, `j`, `a`, `b`) are derived from and bounded by `len`, which was verified to be within the buffer, so no out-of-bounds reads or writes occur during the permutation steps.
5. **Safe parsing:** `strtoll` with `errno`/`endptr` validation guards against range errors and malformed conversions, and the final `result <= n` check enforces the function's contract.

No functional or security changes were required; the original bounds-checking logic is correct and the code stands as a properly hardened implementation.
