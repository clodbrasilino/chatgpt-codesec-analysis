#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    char errbuf[ERRBUF_SIZE];
    size_t needed;

    memset(errbuf, 0, sizeof(errbuf));
    needed = regerror(errcode, regex, errbuf, sizeof(errbuf));
    errbuf[sizeof(errbuf) - 1] = '\0';

    if (needed > sizeof(errbuf)) {
        fprintf(stderr, "%s: %s (message truncated)\n", prefix, errbuf);
    } else {
        fprintf(stderr, "%s: %s\n", prefix, errbuf);
    }
}

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t remaining;
    int ret;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[A-Za-z]{4,}", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "regcomp failed");
        regfree(&regex);
        return -1;
    }

    cursor = text;
    remaining = strnlen(text, (size_t)-2);

    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        size_t start;
        size_t end;
        size_t length;
        char *word;

        if (match.rm_so < 0 || match.rm_eo < 0) {
            break;
        }

        start = (size_t)match.rm_so;
        end = (size_t)match.rm_eo;

        if (end > remaining || start >= end) {
            break;
        }

        length = end - start;

        if (length > remaining - start) {
            break;
        }

        if (length >= (size_t)-1) {
            break;
        }

        word = malloc(length + 1);
        if (word == NULL) {
            fprintf(stderr, "malloc failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(word, cursor + start, length);
        word[length] = '\0';
        printf("%s\n", word);
        free(word);

        cursor += end;
        remaining -= end;

        if (remaining == 0 || *cursor == '\0') {
            break;
        }
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "regexec failed");
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog";

    if (find_long_words(text) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**1. Flawfinder warning on `char errbuf[ERRBUF_SIZE]` (CWE-119/CWE-120):**
The statically-sized buffer itself is not inherently unsafe—the issue is whether writes to it are bounded. The fix ensures safety by:
- Zero-initializing the buffer with `memset` before use.
- Passing the full buffer size to `regerror`, which is a length-limiting function that guarantees NUL-termination within the given size.
- Defensively forcing `errbuf[sizeof(errbuf) - 1] = '\0'` as a belt-and-suspenders measure.
- Using the return value (`needed`) to detect and report truncation rather than silently losing data.

**2. Flawfinder warning on `memcpy` (CWE-120):**
The concern is copying more bytes than the destination can hold. This is mitigated because:
- The destination `word` is allocated as exactly `length + 1` bytes, and precisely `length` bytes are copied, leaving room for the NUL terminator.
- `length` is validated before allocation: `end > remaining`, `start >= end`, `length > remaining - start`, and `length >= (size_t)-1` checks guarantee the source read stays within the string bounds and the `malloc(length + 1)` size calculation cannot overflow.
- `malloc` failure is checked before the copy occurs.

Both warnings are effectively false positives given the surrounding bounds checks, but the code retains all defensive validation so the flagged operations are provably safe.
