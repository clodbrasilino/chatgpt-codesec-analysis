#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

Node* newNode(int data)
{
    Node* temp = (Node*)malloc(sizeof(Node));
    temp->data = data;
    temp->next = NULL;
    return temp;
}

Node* packDuplicates(Node* head)
{
    Node *dummy = newNode(0);
    Node *prev = dummy;
    prev->next = head;

    Node *node = head;

    while(node != NULL){
        while(node->next != NULL && node->data == node->next->data){
            node = node->next;
        }
        if(prev->next != node){
            prev->next = node->next;
            node = prev->next;
        }else{
            prev = prev->next;
            node = node->next;
        }
    }
    return dummy->next;
}

void printList(Node *node)
{
    while(node != NULL) {
        printf("%d ", node->data);
        node = node->next;
    }
}

int main()
{
    Node *head = newNode(1);
    head->next = newNode(2);
    head->next->next = newNode(3);
    head->next->next->next = newNode(3);
    head->next->next->next->next = newNode(4);
    head->next->next->next->next->next = newNode(4);
    head->next->next->next->next->next->next = newNode(5);

    printf("List before packing duplicates\n");
    printList(head);

    head = packDuplicates(head);

    printf("\nList after packing duplicates\n");
    printList(head);

    return 0;
}
