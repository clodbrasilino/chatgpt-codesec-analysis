```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            best = max_int(best, dp[i - 1][j] - prices[j]);
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

Analysis notes on the original code and the reported weaknesses:

The DP algorithm itself is the standard O(k·n) solution for the "at most k transactions" stock problem, and tracing it by hand gives the correct results for all three test cases: `{3,2,6,5,0,3}` with k=2 yields 7, `{1,2,3,4,5}` with k=2 yields 4 (also correctly handled by the `k >= n/2` greedy shortcut), and `{2,4,1}` with k=2 yields 2.

The failure messages ("got 0", "got 5") are symptomatic of a version where either the early-return guard incorrectly rejected valid inputs (e.g., returning 0 when `k >= n/2` instead of running the greedy pass), or where `best` was reset incorrectly inside the transaction loop so only a single buy/sell was credited. The rewritten version above ensures:

1. The `k >= n/2` fast path accumulates every upward price delta rather than returning early.
2. `best` is initialized per transaction level as `dp[i-1][0] - prices[0]` and updated before `dp[i][j]` is computed, so the buy decision correctly incorporates profit from prior transactions.
3. All allocation failures are cleaned up without leaks, and `-1` is reserved exclusively for allocation errors so callers can distinguish it from a legitimate zero profit.
