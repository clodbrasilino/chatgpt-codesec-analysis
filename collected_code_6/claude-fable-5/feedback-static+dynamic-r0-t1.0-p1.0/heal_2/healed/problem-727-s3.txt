```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *remove_non_alnum(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t result_len;
    size_t input_len;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    result = malloc(input_len + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    result_len = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t chunk_len;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        chunk_len = (size_t)(match.rm_eo - match.rm_so);
        if (chunk_len > input_len - result_len) {
            free(result);
            regfree(&regex);
            return NULL;
        }
        memcpy(result + result_len, cursor + match.rm_so, chunk_len);
        result_len += chunk_len;
        cursor += match.rm_eo;
    }

    result[result_len] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a-b_c=d+e",
        "!!!@@@###",
        "NoSpecialChars2024"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alnum(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            return EXIT_FAILURE;
        }
        if (printf("Original: %s\nCleaned:  %s\n\n",
                   test_strings[i], cleaned) < 0) {
            free(cleaned);
            return EXIT_FAILURE;
        }
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

Notes on the changes made:

1. **`variableScope` (cppcheck)** — The `chunk_len` variable was declared at function scope but only used inside the `while` loop. It has been moved into the loop body, minimizing its scope as recommended.

2. **`memcpy` / CWE-120 (Flawfinder)** — This is effectively a false positive in the original code, since the destination buffer `result` is allocated with `input_len + 1` bytes and the guard `chunk_len > input_len - result_len` guarantees the copy never exceeds capacity (with no integer underflow, since `result_len <= input_len` is an invariant). The bounds check is retained, so the `memcpy` remains provably safe; the comment is simply removed per your instructions.

All other defensive measures from the original — NULL input check, `strnlen` bounding against `MAX_INPUT_LEN`, `regcomp`/`regexec` error handling, validation of `rm_so`/`rm_eo`, and consistent cleanup of `result` and `regex` on every error path — are preserved unchanged.
