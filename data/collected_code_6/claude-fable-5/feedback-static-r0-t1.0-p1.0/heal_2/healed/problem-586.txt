```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > n * sizeof(int) || tail_bytes > n * sizeof(int)) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

1. **Include fix:** The original code used `SIZE_MAX` but included `<limits.h>`. `SIZE_MAX` is defined in `<stdint.h>`, so the original may not even compile on strict toolchains. This is corrected.

2. **CWE-120 (memcpy buffer overflow) warnings:** These are addressed structurally rather than being genuine defects:
   - `temp` is allocated with exactly `head_bytes`, and the first `memcpy` copies exactly `head_bytes` — the destination always holds the source.
   - Since `k = k % n` guarantees `0 < k < n`, both `arr + k` (source of `memmove`, `tail_bytes` long) and `arr + (n - k)` (destination of the final `memcpy`, `head_bytes` long) stay strictly within the `n * sizeof(int)` bounds of the caller's array.
   - An explicit sanity check (`head_bytes`/`tail_bytes` vs. total array size) was added as defense-in-depth to satisfy static analyzers; it can never fail given the modulo, but documents the invariant.

3. **Integer overflow:** The `n > SIZE_MAX / sizeof(int)` guard already prevents `k * sizeof(int)` and `(n - k) * sizeof(int)` from wrapping, since `k < n`.

4. **Correct use of `memmove`:** The overlapping shift of the tail correctly uses `memmove` rather than `memcpy` — overlapping regions with `memcpy` would be undefined behavior.

5. **Error handling:** `malloc` failure, `NULL` input, and `printf` return values are all checked, and `temp` is freed on the success path with no leak (early returns occur only before allocation).
