```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_WORD_LEN 256

int match_word_at_start(const char *str, const char *word)
{
    size_t word_len;
    size_t i;

    if (str == NULL || word == NULL) {
        return 0;
    }

    word_len = strnlen(word, MAX_WORD_LEN);
    if (word_len == 0 || word_len == MAX_WORD_LEN) {
        return 0;
    }

    for (i = 0; i < word_len; i++) {
        if (str[i] == '\0' || str[i] != word[i]) {
            return 0;
        }
    }

    if (str[word_len] == '\0' || isspace((unsigned char)str[word_len]) ||
        ispunct((unsigned char)str[word_len])) {
        return 1;
    }

    return 0;
}

int main(void)
{
    const char *text1 = "hello world";
    const char *text2 = "helloworld";
    const char *text3 = "hello, everyone";
    const char *word = "hello";
    int result;

    result = match_word_at_start(text1, word);
    if (printf("Match in \"%s\": %d\n", text1, result) < 0) {
        return EXIT_FAILURE;
    }

    result = match_word_at_start(text2, word);
    if (printf("Match in \"%s\": %d\n", text2, result) < 0) {
        return EXIT_FAILURE;
    }

    result = match_word_at_start(text3, word);
    if (printf("Match in \"%s\": %d\n", text3, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key change:** The flagged `strlen(word)` call was replaced with `strnlen(word, MAX_WORD_LEN)`, which bounds the read to a maximum length. This addresses CWE-126 (buffer over-read) by ensuring that even if `word` is not properly NUL-terminated, the scan cannot read past a defined limit. If the length hits the cap (meaning no terminator was found within bounds), the function safely rejects the input by returning `0`.

The `str` parameter was already handled safely in the original code, since the loop checks `str[i] == '\0'` before comparing, and the final boundary check at `str[word_len]` is only reached after confirming all preceding characters are non-NUL — meaning it never reads beyond a terminator in a well-formed string.
