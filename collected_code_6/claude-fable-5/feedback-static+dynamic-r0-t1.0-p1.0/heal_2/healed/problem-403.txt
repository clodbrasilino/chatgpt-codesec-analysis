```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

#define URL_PATTERN "^(https?|ftp)://[A-Za-z0-9]([A-Za-z0-9.-]*[A-Za-z0-9])?(:[0-9]{1,5})?(/[A-Za-z0-9._~%/?#=&+-]*)?$"

int is_valid_url(const char *url)
{
    regex_t regex;
    int ret;

    if (url == NULL) {
        return 0;
    }

    ret = regcomp(&regex, URL_PATTERN, REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        size_t errsize = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errsize);
        if (errbuf != NULL) {
            regerror(ret, &regex, errbuf, errsize);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Regex compilation failed\n");
        }
        return -1;
    }

    ret = regexec(&regex, url, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex execution failed\n");
        return -1;
    }
}

int main(void)
{
    const char *test_urls[] = {
        "https://www.example.com",
        "http://example.com/path/to/page?name=value",
        "ftp://files.example.org",
        "https://example.com:8080/index.html",
        "not_a_url",
        "http://",
        "https://sub.domain.example.com/page#section"
    };
    size_t count = sizeof(test_urls) / sizeof(test_urls[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = is_valid_url(test_urls[i]);
        if (result == 1) {
            printf("True\n");
        } else if (result == 0) {
            printf("False\n");
        } else {
            fprintf(stderr, "Error validating: %s\n", test_urls[i]);
            return EXIT_FAILURE;
        }
        fflush(stdout);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the weaknesses and fixes applied:**

The failed test cases all reported `<no output>` — the test harness expected the literal strings `True` and `False`, but the original program printed `VALID: <url>` and `INVALID: <url>` instead. Since the validation logic itself was sound (the regex correctly accepts the valid URLs and rejects `not_a_url` and `http://`), the actual defect was in the output contract, not the matching.

Changes made:

1. **Output format corrected** — replaced `"VALID:   %s\n"` / `"INVALID: %s\n"` with `"True\n"` / `"False\n"` to match the expected test output.
2. **Added `fflush(stdout)`** after each result — `<no output>` failures can also occur when stdout is block-buffered (e.g., piped to a test harness) and the process is terminated or crashes before the buffer flushes; explicit flushing guarantees each result is emitted immediately.
3. **Preserved the existing robustness** — the `NULL` guard, `regcomp` error reporting via `regerror` with a dynamically sized buffer, `regfree` after `regexec` (no leak), and the three-state return value (`1`/`0`/`-1`) were already correct and were kept intact.
