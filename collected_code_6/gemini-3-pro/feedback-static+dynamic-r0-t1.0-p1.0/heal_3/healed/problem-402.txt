#include <stdio.h>
#include <string.h>

long long nCrModP(long long n, long long r, long long p) {
    if (r > n)
        return 0;
    if (r == 0 || r == n)
        return 1 % p;
    if (r > n - r)
        r = n - r;

    long long C[r + 1];
    for (long long i = 0; i <= r; i++) {
        C[i] = 0;
    }
    C[0] = 1;

    for (long long i = 1; i <= n; i++) {
        for (long long j = (i < r ? i : r); j > 0; j--) {
            C[j] = (C[j] + C[j - 1]) % p;
        }
    }
    return C[r];
}

int main() {
    long long n = 10;
    long long r = 2;
    long long p = 13;
    
    long long result = nCrModP(n, r, p);
    printf("%lld\n", result);
    
    return 0;
}
