#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int frequency_difference(const int *arr, int size) {
    if (arr == NULL || size <= 0) {
        return 0;
    }

    int *freq = (int *)calloc(100001, sizeof(int));
    if (freq == NULL) {
        return 0;
    }

    int i;
    for (i = 0; i < size; i++) {
        if (arr[i] >= 0 && arr[i] <= 100000) {
            freq[arr[i]]++;
        }
    }

    int highest = INT_MIN;
    int least = INT_MAX;
    int found = 0;

    for (i = 0; i <= 100000; i++) {
        if (freq[i] > 0) {
            if (freq[i] > highest) {
                highest = freq[i];
            }
            if (freq[i] < least) {
                least = freq[i];
            }
            found = 1;
        }
    }

    free(freq);

    if (!found) {
        return 0;
    }

    return highest - least;
}

int main(void) {
    int arr[] = {1, 2, 3, 1, 2, 3, 4};
    int size = sizeof(arr) / sizeof(arr[0]);
    int diff = frequency_difference(arr, size);
    printf("%d\n", diff);
    return 0;
}
