#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

char *decimal_to_octal(int decimal) {
    char *octal = malloc(24 * sizeof(char));
    if (octal == NULL) {
        return NULL;
    }

    if (decimal == 0) {
        octal[0] = '0';
        octal[1] = '\0';
        return octal;
    }

    int is_negative = 0;
    unsigned int num;
    if (decimal < 0) {
        is_negative = 1;
        num = (unsigned int)(-(long long)decimal);
    } else {
        num = (unsigned int)decimal;
    }

    size_t temp_size = 24;
    char *temp = malloc(temp_size * sizeof(char));
    if (temp == NULL) {
        free(octal);
        return NULL;
    }

    size_t i = 0;
    while (num > 0 && i < temp_size - 1) {
        temp[i++] = '0' + (num % 8);
        num /= 8;
    }
    temp[i] = '\0';

    size_t index = 0;
    if (is_negative) {
        octal[index++] = '-';
    }

    size_t max_digits = 24 - index - 1;
    if (i > max_digits) {
        i = max_digits;
    }

    size_t j;
    for (j = i; j > 0; j--) {
        size_t pos = j - 1;
        if (pos >= i) {
            break;
        }
        octal[index++] = temp[pos];
    }
    octal[index] = '\0';

    free(temp);
    return octal;
}

int main(void) {
    int test_values[] = {0, 8, 64, -8, -64, 123, -123, 100, -100};
    size_t num_tests = sizeof(test_values) / sizeof(test_values[0]);

    for (size_t i = 0; i < num_tests; i++) {
        char *result = decimal_to_octal(test_values[i]);
        if (result != NULL) {
            printf("Decimal: %d -> Octal: %s\n", test_values[i], result);
            free(result);
        } else {
            printf("Memory allocation failed for decimal: %d\n", test_values[i]);
        }
    }

    return 0;
}
