#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stddef.h>
#include <stdlib.h>

int binary_to_decimal(const char *binary, long *result) {
    if (binary == NULL || result == NULL) {
        errno = EINVAL;
        return -1;
    }

    if (*binary == '\0') {
        errno = EINVAL;
        return -1;
    }

    long decimal = 0;
    const char *p = binary;

    while (*p != '\0') {
        if (*p != '0' && *p != '1') {
            errno = EINVAL;
            return -1;
        }

        if (decimal > (LONG_MAX - (*p - '0')) / 2) {
            errno = ERANGE;
            return -1;
        }

        decimal = decimal * 2 + (*p - '0');
        p++;
    }

    *result = decimal;
    return 0;
}

int main(void) {
    char *input = NULL;
    size_t buffer_size = 0;
    long result;

    printf("Enter a binary number: ");

    ssize_t line_length = getline(&input, &buffer_size, stdin);
    if (line_length == -1) {
        if (ferror(stdin)) {
            perror("Input error");
        }
        free(input);
        return 1;
    }

    if (line_length > 0 && input[line_length - 1] == '\n') {
        input[line_length - 1] = '\0';
        line_length--;
    }

    if (line_length == 0) {
        fprintf(stderr, "Invalid input: please provide a valid binary number\n");
        free(input);
        return 1;
    }

    if (binary_to_decimal(input, &result) != 0) {
        switch (errno) {
            case EINVAL:
                fprintf(stderr, "Invalid input: please provide a valid binary number\n");
                break;
            case ERANGE:
                fprintf(stderr, "Error: result exceeds range of long\n");
                break;
            default:
                fprintf(stderr, "Unexpected error occurred\n");
                break;
        }
        free(input);
        return 1;
    }

    printf("Decimal equivalent: %ld\n", result);
    free(input);
    return 0;
}
