#include <stdio.h>
#include <string.h>

void remove_first_last(char *str, char ch) {
    if (str == NULL || *str == '\0') {
        return;
    }

    size_t len = strnlen(str, 4096);
    if (len == 0) {
        return;
    }

    size_t first = (size_t)-1;
    for (size_t i = 0; i < len; i++) {
        if (str[i] == ch) {
            first = i;
            break;
        }
    }

    if (first == (size_t)-1) {
        return;
    }

    size_t last = (size_t)-1;
    for (size_t i = len; i > 0; i--) {
        if (str[i - 1] == ch) {
            last = i - 1;
            break;
        }
    }

    size_t write = 0;
    for (size_t read = 0; read < len; read++) {
        if (read == first || read == last) {
            continue;
        }
        if (write < read) {
            str[write] = str[read];
        }
        write++;
    }
    str[write] = '\0';
}

int main(void) {
    char str1[] = "hello world";
    remove_first_last(str1, 'o');
    printf("%s\n", str1);

    char str2[] = "a";
    remove_first_last(str2, 'a');
    printf("%s\n", str2);

    char str3[] = "aaa";
    remove_first_last(str3, 'a');
    printf("%s\n", str3);

    char str4[] = "programming";
    remove_first_last(str4, 'g');
    printf("%s\n", str4);

    char str5[] = "test";
    remove_first_last(str5, 'x');
    printf("%s\n", str5);

    return 0;
}
