#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define BUFFER_SIZE 256
#define MAX_DISCARD 1000000

static int is_valid_hex(const char *str, size_t max_len)
{
    size_t i = 0;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);

    if (len == 0 || len >= max_len) {
        return 0;
    }

    if (len > 2 && str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        i = 2;
    }

    if (i >= len) {
        return 0;
    }

    for (; i < len; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

static int is_hex_even(const char *hex, size_t max_len, int *result)
{
    size_t len;
    char last;

    if (hex == NULL || result == NULL) {
        return -1;
    }

    if (!is_valid_hex(hex, max_len)) {
        return -1;
    }

    len = strnlen(hex, max_len);

    if (len == 0 || len >= max_len) {
        return -1;
    }

    last = hex[len - 1];

    if (last >= '0' && last <= '9') {
        *result = ((last - '0') % 2 == 0) ? 1 : 0;
    } else {
        last = (char)tolower((unsigned char)last);
        if (last < 'a' || last > 'f') {
            return -1;
        }
        *result = ((last - 'a' + 10) % 2 == 0) ? 1 : 0;
    }

    return 0;
}

static int discard_remaining_input(void)
{
    int c;
    int discarded = 0;
    size_t guard = 0;

    while ((c = getchar()) != '\n' && c != EOF) {
        discarded = 1;
        guard++;
        if (guard > (size_t)MAX_DISCARD) {
            break;
        }
    }

    return discarded;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;
    int even = 0;

    printf("Enter a hexadecimal number: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));

    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else {
        if (discard_remaining_input()) {
            fprintf(stderr, "Input too long (maximum %d characters)\n",
                    BUFFER_SIZE - 2);
            return EXIT_FAILURE;
        }
    }

    if (is_hex_even(buffer, sizeof(buffer), &even) != 0) {
        fprintf(stderr, "Invalid hexadecimal input\n");
        return EXIT_FAILURE;
    }

    if (even) {
        printf("%s is even\n", buffer);
    } else {
        printf("%s is odd\n", buffer);
    }

    return EXIT_SUCCESS;
}

Flawfinder warning on `getchar` in a loop (CWE-120, CWE-20):** The `discard_remaining_input()` function does not write into any buffer — it only reads and discards characters, so no overflow is possible. The `guard` counter bounded by `MAX_DISCARD` additionally prevents an unbounded loop (e.g., from a malicious pipe feeding infinite data), and the loop correctly terminates on both `'\n'` and `EOF`. No change required.

**2. Flawfinder warning on the statically-sized `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** The buffer is only ever filled via `fgets(buffer, (int)sizeof(buffer), stdin)`, which is length-bounded and always NUL-terminates within the given size. The code additionally:
- Forces termination with `buffer[sizeof(buffer) - 1] = '\0';` (defensive, harmless).
- Uses `strnlen` with an explicit bound instead of `strlen`, so it never scans past the buffer.
- Detects truncated input (no trailing `'\n'`) and rejects it after safely draining stdin, avoiding stale leftover input contaminating subsequent reads.
- Validates all content via `is_valid_hex` with bounded scanning and `(unsigned char)` casts before `isxdigit`/`tolower`, avoiding undefined behavior on negative `char` values.
