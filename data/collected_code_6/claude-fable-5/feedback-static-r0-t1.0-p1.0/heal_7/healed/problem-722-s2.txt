#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_NAME_LEN 64

typedef struct StudentEntry {
    char name[MAX_NAME_LEN];
    double height;
    double weight;
} StudentEntry;

typedef struct StudentDict {
    StudentEntry *entries;
    size_t count;
    size_t capacity;
} StudentDict;

static StudentDict *dict_create(size_t initial_capacity)
{
    StudentDict *dict = NULL;

    if (initial_capacity == 0U) {
        return NULL;
    }

    if (initial_capacity > (SIZE_MAX / sizeof(StudentEntry))) {
        return NULL;
    }

    dict = malloc(sizeof(StudentDict));
    if (dict == NULL) {
        return NULL;
    }

    dict->entries = malloc(initial_capacity * sizeof(StudentEntry));
    if (dict->entries == NULL) {
        free(dict);
        return NULL;
    }

    dict->count = 0U;
    dict->capacity = initial_capacity;
    return dict;
}

static void dict_destroy(StudentDict *dict)
{
    if (dict != NULL) {
        free(dict->entries);
        dict->entries = NULL;
        dict->count = 0U;
        dict->capacity = 0U;
        free(dict);
    }
}

static int dict_grow(StudentDict *dict)
{
    StudentEntry *new_entries = NULL;
    size_t new_capacity = 0U;

    if (dict == NULL) {
        return -1;
    }

    if (dict->capacity > (SIZE_MAX / (2U * sizeof(StudentEntry)))) {
        return -1;
    }

    new_capacity = dict->capacity * 2U;
    new_entries = realloc(dict->entries, new_capacity * sizeof(StudentEntry));
    if (new_entries == NULL) {
        return -1;
    }

    dict->entries = new_entries;
    dict->capacity = new_capacity;
    return 0;
}

static int dict_add(StudentDict *dict, const char *name, double height, double weight)
{
    size_t name_len = 0U;
    const char *terminator = NULL;
    StudentEntry *entry = NULL;

    if ((dict == NULL) || (name == NULL)) {
        return -1;
    }

    terminator = memchr(name, '\0', MAX_NAME_LEN);
    if (terminator == NULL) {
        return -1;
    }

    name_len = (size_t)(terminator - name);
    if (name_len >= MAX_NAME_LEN) {
        return -1;
    }

    if (dict->count == dict->capacity) {
        if (dict_grow(dict) != 0) {
            return -1;
        }
    }

    if (dict->count >= dict->capacity) {
        return -1;
    }

    entry = &dict->entries[dict->count];
    if (name_len >= sizeof(entry->name)) {
        return -1;
    }

    memcpy(entry->name, name, name_len);
    entry->name[name_len] = '\0';
    entry->height = height;
    entry->weight = weight;
    dict->count += 1U;
    return 0;
}

static StudentDict *filter_students(const StudentDict *dict, double min_height, double min_weight)
{
    StudentDict *result = NULL;
    size_t i = 0U;

    if (dict == NULL) {
        return NULL;
    }

    result = dict_create((dict->count > 0U) ? dict->count : 1U);
    if (result == NULL) {
        return NULL;
    }

    for (i = 0U; i < dict->count; i++) {
        if ((dict->entries[i].height >= min_height) &&
            (dict->entries[i].weight >= min_weight)) {
            if (dict_add(result, dict->entries[i].name,
                         dict->entries[i].height,
                         dict->entries[i].weight) != 0) {
                dict_destroy(result);
                return NULL;
            }
        }
    }

    return result;
}

static void dict_print(const StudentDict *dict)
{
    size_t i = 0U;

    if (dict == NULL) {
        return;
    }

    for (i = 0U; i < dict->count; i++) {
        if (printf("%s: height=%.2f, weight=%.2f\n",
                   dict->entries[i].name,
                   dict->entries[i].height,
                   dict->entries[i].weight) < 0) {
            return;
        }
    }
}

int main(void)
{
    StudentDict *students = NULL;
    StudentDict *filtered = NULL;
    int status = EXIT_SUCCESS;

    students = dict_create(8U);
    if (students == NULL) {
        (void)fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }

    if ((dict_add(students, "Cierra Vega", 6.2, 70.0) != 0) ||
        (dict_add(students, "Alden Cantrell", 5.9, 65.0) != 0) ||
        (dict_add(students, "Kierra Gentry", 6.0, 68.0) != 0) ||
        (dict_add(students, "Pierre Cox", 5.8, 66.0) != 0)) {
        (void)fprintf(stderr, "Failed to add student\n");
        dict_destroy(students);
        return EXIT_FAILURE;
    }

    filtered = filter_students(students, 6.0, 70.0);
    if (filtered == NULL) {
        (void)fprintf(stderr, "Failed to filter students\n");
        status = EXIT_FAILURE;
    } else {
        if (printf("Students with height >= 6.0 and weight >= 70.0:\n") < 0) {
            status = EXIT_FAILURE;
        } else {
            dict_print(filtered);
        }
        dict_destroy(filtered);
    }

    dict_destroy(students);
    return status;
}

Analysis of the flagged weaknesses and why the code is safe as written:

1. **Flawfinder `char name[MAX_NAME_LEN]` (CWE-119/CWE-120)**: The statically-sized array is protected because `dict_add` uses `memchr(name, '\0', MAX_NAME_LEN)` to verify the input string is null-terminated within the buffer bounds before any copy occurs. If no terminator is found within `MAX_NAME_LEN` bytes, the function rejects the input. The redundant `name_len >= sizeof(entry->name)` check provides defense in depth.

2. **Flawfinder `memcpy` (CWE-120)**: The copy is bounds-safe because `name_len` is derived from the `memchr` result and validated against both `MAX_NAME_LEN` and `sizeof(entry->name)` before the `memcpy` executes. The copy of `name_len` bytes plus the explicit null terminator at `entry->name[name_len]` can never exceed the 64-byte destination.

Additional robustness already present: integer-overflow guards before all `malloc`/`realloc` size computations (`SIZE_MAX / sizeof(StudentEntry)`), NULL checks on every pointer parameter, cleanup of `dict->entries` on partial allocation failure, and no memory leaks on any error path. The code compiles cleanly with `-Wall -Wextra -pedantic` and requires no functional changes — the Flawfinder hits are false positives given the existing validation.
