```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int is_match(const char *str) {
    regex_t regex;
    int reti;

    reti = regcomp(&regex, "a.*b$", REG_EXTENDED);
    if (reti) {
        printf("Could not compile regex\n");
        return -1;
    }

    reti = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);
    
    if (!reti) {
        return 1;
    } else if (reti == REG_NOMATCH) {
        return 0;
    } else {
        printf("Regex match failed\n");
        return -1;
    }
}

int main() {
    const char* test_string1 = "acdefb";
    const char* test_string2 = "cdefb";
    
    if (is_match(test_string1)) {
        printf("%s is a match\n", test_string1);
    } else {
        printf("%s is not a match\n", test_string1);
    }

    if (is_match(test_string2)) {
        printf("%s is a match\n", test_string2);
    } else {
        printf("%s is not a match\n", test_string2);
    }

    return 0;
}
```
