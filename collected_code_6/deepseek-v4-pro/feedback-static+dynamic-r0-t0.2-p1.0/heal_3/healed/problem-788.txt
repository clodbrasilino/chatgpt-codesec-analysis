#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *str;
    int *list;
    size_t list_len;
} Tuple;

Tuple create_tuple(const char *str, const int *list, size_t list_len) {
    Tuple t;
    size_t str_len = strnlen(str, 1024);
    
    if (str_len >= 1024 || list_len > (SIZE_MAX / sizeof(int))) {
        t.str = NULL;
        t.list = NULL;
        t.list_len = 0;
        return t;
    }
    
    t.str = (char *)malloc(str_len + 1);
    if (t.str == NULL) {
        t.list = NULL;
        t.list_len = 0;
        return t;
    }
    
    if (str_len > 0) {
        memcpy(t.str, str, str_len);
    }
    t.str[str_len] = '\0';
    
    if (list == NULL && list_len > 0) {
        free(t.str);
        t.str = NULL;
        t.list = NULL;
        t.list_len = 0;
        return t;
    }
    
    if (list_len > 0) {
        t.list = (int *)malloc(list_len * sizeof(int));
        if (t.list == NULL) {
            free(t.str);
            t.str = NULL;
            t.list_len = 0;
            return t;
        }
        memcpy(t.list, list, list_len * sizeof(int));
    } else {
        t.list = NULL;
    }
    
    t.list_len = list_len;
    return t;
}

void free_tuple(Tuple *t) {
    if (t != NULL) {
        if (t->str != NULL) {
            free(t->str);
            t->str = NULL;
        }
        if (t->list != NULL) {
            free(t->list);
            t->list = NULL;
        }
        t->list_len = 0;
    }
}

int main(void) {
    const char *str = "hello";
    int list[] = {1, 2, 3, 4, 5};
    size_t list_len = sizeof(list) / sizeof(list[0]);
    
    Tuple t = create_tuple(str, list, list_len);
    
    if (t.str == NULL || t.list == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        free_tuple(&t);
        return EXIT_FAILURE;
    }
    
    printf("String: %s\n", t.str);
    printf("List: ");
    for (size_t i = 0; i < t.list_len; i++) {
        printf("%d ", t.list[i]);
    }
    printf("\n");
    
    free_tuple(&t);
    
    return 0;
}
