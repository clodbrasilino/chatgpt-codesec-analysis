```c
#include <stdio.h>
#include <stdlib.h>

int maxProfit(int k, const int *prices, int n)
{
    int *prev;
    int *curr;
    int *tmp;
    int t;
    int d;
    int result;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (d = 1; d < n; d++) {
            if (prices[d] > prices[d - 1]) {
                result += prices[d] - prices[d - 1];
            }
        }
        return result;
    }

    prev = calloc((size_t)n, sizeof(int));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc((size_t)n, sizeof(int));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    result = 0;

    for (t = 1; t <= k; t++) {
        int maxDiff = prev[0] - prices[0];
        curr[0] = 0;
        for (d = 1; d < n; d++) {
            if (prev[d] - prices[d] > maxDiff) {
                maxDiff = prev[d] - prices[d];
            }
            curr[d] = curr[d - 1];
            if (prices[d] + maxDiff > curr[d]) {
                curr[d] = prices[d] + maxDiff;
            }
        }
        if (curr[n - 1] > result) {
            result = curr[n - 1];
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    free(prev);
    free(curr);

    return result;
}

int main(void)
{
    int prices0[] = {10, 22, 5, 75, 65, 80};
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int prices2[] = {2, 4, 1};

    struct {
        int k;
        const int *prices;
        int n;
        int expected;
    } tests[] = {
        {2, prices0, (int)(sizeof(prices0) / sizeof(prices0[0])), 87},
        {2, prices1, (int)(sizeof(prices1) / sizeof(prices1[0])), 7},
        {2, prices2, (int)(sizeof(prices2) / sizeof(prices2[0])), 2}
    };

    int numTests = (int)(sizeof(tests) / sizeof(tests[0]));
    int i;
    int failures = 0;

    for (i = 0; i < numTests; i++) {
        int actual = maxProfit(tests[i].k, tests[i].prices, tests[i].n);
        if (actual == tests[i].expected) {
            printf("Test %d passed: expected %d, got %d\n",
                   i, tests[i].expected, actual);
        } else {
            printf("Test %d FAILED: expected %d, got %d\n",
                   i, tests[i].expected, actual);
            failures++;
        }
    }

    if (failures > 0) {
        printf("%d of %d tests failed\n", failures, numTests);
        return EXIT_FAILURE;
    }

    printf("All %d tests passed\n", numTests);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses hinted at in the original comments (`expected 2, got 0`, `expected 10, got 5`, `expected 7, got 0`) point to three classic bugs in earlier revisions of this kind of implementation, all of which are properly handled in the version above:

1. **Guard clause rejecting valid inputs** (`got 0` cases): an overly strict early return such as `k <= 1` or `n <= 2` would return 0 for valid small inputs like `{2, 4, 1}` with `k = 2`. The correct guard is `prices == NULL || n <= 1 || k <= 0`.

2. **Missing or broken `k >= n / 2` fast path** (`expected 10, got 5` style failures): when transactions are effectively unlimited, the greedy sum of all positive adjacent differences must be used; capping to a single best transaction undercounts multi-transaction profits.

3. **DP recurrence/state-swap errors**: `maxDiff` must be seeded with `prev[0] - prices[0]` and updated *before* computing `curr[d]`, and the `prev`/`curr` buffers must be swapped each transaction iteration — otherwise later transactions build on stale state and return 0 or partial profit.

The implementation above passes all three test cases (87, 7, 2), frees both heap buffers exactly once regardless of the swap parity, checks both `calloc` results, and avoids overflow on the size computation via the `size_t` cast.
