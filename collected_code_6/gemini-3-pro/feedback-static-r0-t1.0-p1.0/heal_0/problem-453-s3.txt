```c
#include <stdio.h>
#include <stdlib.h>

int sumOfEvenFactors(int n) {
    if (n <= 0) {
        return 0;
    }

    int sum = 0;
    for (int i = 1; i <= n; i++) {
        if (n % i == 0 && i % 2 == 0) {
            sum += i;
        }
    }
    
    return sum;
}

int main(void) {
    int num;
    
    if (scanf("%d", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    int result = sumOfEvenFactors(num);
    printf("%d\n", result);
    
    return EXIT_SUCCESS;
}
```
