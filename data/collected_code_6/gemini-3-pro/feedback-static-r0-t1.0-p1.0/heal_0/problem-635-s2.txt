#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} MinHeap;

MinHeap* create_heap(size_t capacity) {
    if (capacity == 0) {
        return NULL;
    }
    MinHeap *heap = malloc(sizeof(MinHeap));
    if (!heap) {
        return NULL;
    }
    heap->data = malloc(capacity * sizeof(int));
    if (!heap->data) {
        free(heap);
        return NULL;
    }
    heap->size = 0;
    heap->capacity = capacity;
    return heap;
}

void free_heap(MinHeap *heap) {
    if (heap) {
        free(heap->data);
        free(heap);
    }
}

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void heapify_up(MinHeap *heap, size_t index) {
    while (index > 0) {
        size_t parent = (index - 1) / 2;
        if (heap->data[parent] > heap->data[index]) {
            swap(&heap->data[parent], &heap->data[index]);
            index = parent;
        } else {
            break;
        }
    }
}

void heapify_down(MinHeap *heap, size_t index) {
    size_t smallest = index;
    size_t left = 2 * index + 1;
    size_t right = 2 * index + 2;

    if (left < heap->size && heap->data[left] < heap->data[smallest]) {
        smallest = left;
    }
    if (right < heap->size && heap->data[right] < heap->data[smallest]) {
        smallest = right;
    }
    if (smallest != index) {
        swap(&heap->data[index], &heap->data[smallest]);
        heapify_down(heap, smallest);
    }
}

int push_heap(MinHeap *heap, int value) {
    if (!heap || heap->size >= heap->capacity) {
        return -1; 
    }
    heap->data[heap->size] = value;
    heapify_up(heap, heap->size);
    heap->size++;
    return 0;
}

int pop_heap(MinHeap *heap, int *out_value) {
    if (!heap || !out_value || heap->size == 0) {
        return -1;
    }
    *out_value = heap->data[0];
    heap->data[0] = heap->data[heap->size - 1];
    heap->size--;
    if (heap->size > 0) {
        heapify_down(heap, 0);
    }
    return 0;
}

void heap_sort(int *values, size_t count) {
    if (!values || count == 0) {
        return;
    }
    MinHeap *heap = create_heap(count);
    if (!heap) {
        return;
    }

    for (size_t i = 0; i < count; i++) {
        if (push_heap(heap, values[i]) != 0) {
            free_heap(heap);
            return;
        }
    }

    for (size_t i = 0; i < count; i++) {
        if (pop_heap(heap, &values[i]) != 0) {
            free_heap(heap);
            return;
        }
    }

    free_heap(heap);
}

int main(void) {
    int values[] = {9, 4, 7, 1, 3, 2, 8, 5, 6};
    size_t count = sizeof(values) / sizeof(values[0]);

    heap_sort(values, count);

    for (size_t i = 0; i < count; i++) {
        printf("%d ", values[i]);
    }
    printf("\n");

    return 0;
}
