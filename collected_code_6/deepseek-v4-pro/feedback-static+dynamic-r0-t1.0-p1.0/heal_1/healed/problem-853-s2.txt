#include <stdio.h>
#include <stdint.h>

int64_t sumOfOddFactors(int n) {
    int64_t sum = 0;
    int i;

    if (n <= 0) {
        return 0;
    }

    for (i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            if (i % 2 != 0) {
                sum += i;
            }
            if (i != n / i && (n / i) % 2 != 0) {
                sum += n / i;
            }
        }
    }

    return sum;
}

int main(void) {
    int num;

    printf("Enter a number: ");
    if (scanf("%d", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    printf("Sum of odd factors: %ld\n", sumOfOddFactors(num));

    return 0;
}
