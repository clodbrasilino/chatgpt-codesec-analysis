#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_INPUT 1024

static void free_table(size_t **table, size_t rows)
{
    size_t i;

    if (table == NULL) {
        return;
    }
    for (i = 0; i < rows; i++) {
        free(table[i]);
    }
    free(table);
}

static char *longest_palindromic_subsequence(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t j;
    size_t len;
    size_t left;
    size_t right;
    size_t **dp;
    char *result;

    if (s == NULL) {
        return NULL;
    }

    n = strnlen(s, max_len);
    if (n == 0) {
        result = malloc(1);
        if (result != NULL) {
            result[0] = '\0';
        }
        return result;
    }

    if (n > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    dp = calloc(n, sizeof *dp);
    if (dp == NULL) {
        return NULL;
    }
    for (i = 0; i < n; i++) {
        dp[i] = calloc(n, sizeof **dp);
        if (dp[i] == NULL) {
            free_table(dp, i);
            return NULL;
        }
    }

    for (i = 0; i < n; i++) {
        dp[i][i] = 1;
    }

    for (len = 2; len <= n; len++) {
        for (i = 0; i + len - 1 < n; i++) {
            j = i + len - 1;
            if (s[i] == s[j]) {
                dp[i][j] = dp[i + 1][j - 1] + 2;
            } else {
                dp[i][j] = (dp[i + 1][j] > dp[i][j - 1]) ? dp[i + 1][j] : dp[i][j - 1];
            }
        }
    }

    if (dp[0][n - 1] > SIZE_MAX - 1) {
        free_table(dp, n);
        return NULL;
    }

    result = malloc(dp[0][n - 1] + 1);
    if (result == NULL) {
        free_table(dp, n);
        return NULL;
    }

    i = 0;
    j = n - 1;
    left = 0;
    right = dp[0][n - 1] - 1;
    while (i <= j) {
        if (i == j) {
            result[left] = s[i];
            break;
        }
        if (s[i] == s[j]) {
            result[left] = s[i];
            result[right] = s[j];
            left++;
            right--;
            i++;
            j--;
        } else if (dp[i + 1][j] >= dp[i][j - 1]) {
            i++;
        } else {
            j--;
        }
    }
    result[dp[0][n - 1]] = '\0';

    free_table(dp, n);
    return result;
}

{
    char input[MAX_INPUT];
    char *lps;
    size_t len;

    len = strnlen(input, sizeof input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    lps = longest_palindromic_subsequence(input, sizeof input);
    if (lps == NULL) {
        fprintf(stderr, "Failed to compute longest palindromic subsequence\n");
        return EXIT_FAILURE;

    printf("Longest palindromic subsequence: %s\n", lps);

**Analysis summary:**

The original code was already well-hardened against the flagged weakness (CWE-119/CWE-120, statically-sized buffer). Key observations:

1. **Buffer handling (`input[MAX_INPUT]`)** — The Flawfinder warning about the fixed-size array is a false positive here, because:
   - `fgets(input, sizeof input, stdin)` bounds the read to the buffer size.
   - `input[sizeof input - 1] = '\0'` guarantees null termination defensively.
   - `strnlen` is used instead of `strlen` everywhere, so even a hypothetically unterminated buffer can't cause an out-of-bounds read.

2. **Integer overflow protection** — `n > SIZE_MAX / sizeof(size_t)` guards the allocation size, and `dp[0][n - 1] > SIZE_MAX - 1` guards the `+ 1` in the result allocation.

3. **Memory management** — `free_table` correctly handles partial allocation failure (freeing only `i` rows), all error paths release memory, and `calloc` zero-initializes the DP table so `dp[i + 1][j - 1]` is safe for length-2 windows.

4. **DP reconstruction** — The two-pointer fill (`left`/`right`) writes exactly `dp[0][n-1]` characters plus the terminator, so no overflow of `result` is possible.

The only functional addition made is `fflush(stdout)` after the unterminated prompt, ensuring it appears before blocking on `fgets` on fully-buffered streams. No genuine vulnerabilities required fixing; the code follows CERT C practices (bounded string functions, checked allocations, overflow guards).
