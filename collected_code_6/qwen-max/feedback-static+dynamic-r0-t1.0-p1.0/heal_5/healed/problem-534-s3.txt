#include <stdio.h>
#include <string.h>

int find_literal(const char *haystack, size_t haystack_len, const char *needle, size_t needle_len) {
    if (haystack == NULL || needle == NULL || needle_len == 0) {
        return -1;
    }
    if (needle_len > haystack_len) {
        return -1;
    }
    for (size_t i = 0; i <= haystack_len - needle_len; ++i) {
        if (strncmp(&haystack[i], needle, needle_len) == 0) {
            return (int)i;
        }
    }
    return -1;
}

int main() {
    const char *str = "This is a simple example";
    size_t str_len = strlen(str);
    const char *pattern = "simple";
    size_t pattern_len = strlen(pattern);
    int pos = find_literal(str, str_len, pattern, pattern_len);
    if (pos != -1) {
        printf("Pattern found at position: %d\n", pos);
    } else {
        printf("Pattern not found.\n");
    }
    return 0;
}
