```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * CHAR_BIT)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t guard = 0U;

    do {
        ch = fgetc(stdin);
        guard++;
        if (guard >= (size_t)INT_MAX) {
            break;
        }
    } while (ch != '\n' && ch != EOF);

    return ch;
}

int main(void)
{
    char input[INPUT_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == (sizeof(input) - 1U)) {
        (void)discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`fgetc` in a loop (CWE-120, CWE-20):** The `discard_remaining_input()` function reads characters one at a time without writing them into any buffer, so no overflow is possible. The `guard` counter additionally bounds the loop against pathological input streams that never yield `\n` or `EOF`, preventing an unbounded loop. This is already mitigated.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The `input` buffer is:
   - Zero-initialized at declaration.
   - Filled only via `fgets(input, (int)sizeof(input), stdin)`, which enforces the buffer bound.
   - Explicitly NUL-terminated at `input[sizeof(input) - 1U]` as a defensive measure.
   - Measured with `strnlen` (bounded) rather than `strlen`.
   - Checked for truncation: if the input filled the buffer without a trailing newline, the remaining stream is drained and the input is rejected with an error, rather than silently processing a truncated value.

3. **`binary_to_decimal` robustness:** The function validates all pointers, enforces a maximum bit length (`sizeof(unsigned long) * CHar_BIT`), rejects non-binary characters, and guards against overflow before each shift via the `ULONG_MAX >> 1` check.

Both Flawfinder findings are risk-level 1–2 heuristic hits triggered by the mere presence of `fgetc` and a fixed-size `char` array; the surrounding bounds checking already neutralizes them, so the rewrite preserves the logic unchanged with the comments removed.
