#include <stdio.h>
#include <stdlib.h>

unsigned int toggle_middle_bits(unsigned int n) {
    if (n == 0 || n == 1) {
        return n;
    }

    unsigned int temp = n;
    unsigned int msb_pos = 0;
    while (temp > 1) {
        temp >>= 1;
        msb_pos++;
    }

    if (msb_pos < 2) {
        return n;
    }

    unsigned int mask = ((1U << (msb_pos - 1)) - 1) << 1;
    return n ^ mask;
}

int main(void) {
    unsigned int test_val = 11;
    
    unsigned int res = toggle_middle_bits(test_val);
    printf("%u\n", res);

    return 0;
}
