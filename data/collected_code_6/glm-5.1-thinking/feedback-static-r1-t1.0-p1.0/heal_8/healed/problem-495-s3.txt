#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>

char *remove_lowercase_substrings(const char *str) {
    if (!str) {
        return NULL;
    }

    size_t len = strlen(str);

    regex_t regex;
    if (regcomp(&regex, "[[:lower:]]+", REG_EXTENDED) != 0) {
        return NULL;
    }

    char *result = malloc(len + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    const char *p = str;
    size_t out_idx = 0;
    regmatch_t match;

    while (*p) {
        int ret = regexec(&regex, p, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = len - (size_t)(p - str);
            size_t space = len - out_idx;
            size_t safe_remaining = remaining < space ? remaining : space;
            memcpy(result + out_idx, p, safe_remaining);
            out_idx += safe_remaining;
            break;
        } else if (ret != 0) {
            break;
        }

        if (match.rm_so > 0) {
            size_t copy_len = (size_t)match.rm_so;
            size_t space = len - out_idx;
            size_t safe_copy_len = copy_len < space ? copy_len : space;
            memcpy(result + out_idx, p, safe_copy_len);
            out_idx += safe_copy_len;
        }

        p += match.rm_eo;
    }

    result[out_idx] = '\0';
    regfree(&regex);

    char *shrunk = realloc(result, out_idx + 1);
    return shrunk != NULL ? shrunk : result;
}

int main(void) {
    const char *input = "HelloWorldABCdefGHI";
    char *output = remove_lowercase_substrings(input);
    
    if (output != NULL) {
        printf("Original: %s\nResult: %s\n", input, output);
        free(output);
    } else {
        fprintf(stderr, "Memory allocation or regex compilation failed\n");
        return 1;
    }
    
    return 0;
}
