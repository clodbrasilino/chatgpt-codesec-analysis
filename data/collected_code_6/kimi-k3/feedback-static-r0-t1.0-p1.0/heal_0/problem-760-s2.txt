#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

bool has_single_distinct_element(const int arr[], size_t size) {
    if (arr == NULL || size == 0) {
        return false;
    }
    
    const int first_element = arr[0];
    
    for (size_t i = 1; i < size; i++) {
        if (arr[i] != first_element) {
            return false;
        }
    }
    
    return true;
}

int main(void) {
    int arr1[] = {5, 5, 5, 5};
    int arr2[] = {1, 2, 1, 1};
    int arr3[] = {7};
    int arr4[] = {};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);
    size_t size3 = sizeof(arr3) / sizeof(arr3[0]);
    size_t size4 = 0;
    
    printf("Array 1: %s\n", has_single_distinct_element(arr1, size1) ? "true" : "false");
    printf("Array 2: %s\n", has_single_distinct_element(arr2, size2) ? "true" : "false");
    printf("Array 3: %s\n", has_single_distinct_element(arr3, size3) ? "true" : "false");
    printf("Array 4: %s\n", has_single_distinct_element(arr4, size4) ? "true" : "false");
    
    return 0;
}
