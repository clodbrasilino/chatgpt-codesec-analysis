```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define BUFFER_SIZE 256

int find_first_repeated_char(const char *str, size_t max_len, char *result)
{
    unsigned char seen[UCHAR_MAX + 1] = {0};
    size_t i;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char repeated = '\0';
    int status;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    len = strcspn(buffer, "\n");
    if (len >= sizeof(buffer)) {
        len = sizeof(buffer) - 1;
    }
    buffer[len] = '\0';

    if (len == sizeof(buffer) - 1) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    status = find_first_repeated_char(buffer, sizeof(buffer), &repeated);

    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input to function\n");
        return 1;
    }

    return 0;
}
```

The original code was already largely safe — the Flawfinder warnings were informational hits on fixed-size buffers rather than actual vulnerabilities. The rewrite hardens it further by:

- **Explicit bounds in the scan loop**: `find_first_repeated_char` now takes a `max_len` parameter, so it can never read past the buffer even if a non-terminated string is passed (mitigating CWE-119/CWE-120 concerns).
- **Defensive index clamp**: the result of `strcspn` is clamped before being used as an index, guaranteeing the write stays within `buffer`.
- **Stdin drain on truncation**: if the input filled the buffer without a newline, leftover characters are flushed from `stdin` so they can't affect subsequent reads.
- **Retained safe I/O**: `fgets` with `sizeof(buffer)` already bounds the read, and `seen[]` indexed by `unsigned char` cannot exceed `UCHAR_MAX`, so those patterns were kept as-is.
