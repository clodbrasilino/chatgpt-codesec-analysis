```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 1024
#define MAX_WORD_LEN 256
#define MAX_INPUT_LEN 65536

static char *duplicate_string(const char *src, size_t max_len)
{
    size_t len;
    char *copy;

    if (src == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(src, max_len - 1);
    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static int extract_word(const char *str, size_t str_len, size_t *pos,
                        char *word, size_t word_size)
{
    size_t i;
    size_t w = 0;

    if (str == NULL || pos == NULL || word == NULL || word_size == 0) {
        return 0;
    }

    i = *pos;

    while (i < str_len && str[i] != '\0' && !isalnum((unsigned char)str[i])) {
        i++;
    }

    if (i >= str_len || str[i] == '\0') {
        *pos = i;
        word[0] = '\0';
        return 0;
    }

    while (i < str_len && str[i] != '\0' && isalnum((unsigned char)str[i])) {
        if (w + 1 < word_size) {
            word[w] = str[i];
            w++;
        }
        i++;
    }

    word[w] = '\0';
    *pos = i;
    return 1;
}

static int store_word(char (*words)[MAX_WORD_LEN], size_t index,
                      const char *src, size_t dest_size)
{
    size_t len;

    if (words == NULL || src == NULL || dest_size == 0) {
        return -1;
    }

    len = strnlen(src, dest_size - 1);
    if (len > 0) {
        memcpy(words[index], src, len);
    }
    words[index][len] = '\0';
    return 0;
}

char *first_repeated_word(const char *str)
{
    char (*words)[MAX_WORD_LEN];
    char current[MAX_WORD_LEN];
    size_t pos = 0;
    size_t count = 0;
    size_t i;
    size_t str_len;
    char *result;

    if (str == NULL) {
        return duplicate_string("None", MAX_WORD_LEN);
    }

    str_len = strnlen(str, MAX_INPUT_LEN);

    words = calloc((size_t)MAX_WORDS, sizeof(*words));
    if (words == NULL) {
        return NULL;
    }

    while (extract_word(str, str_len, &pos, current, sizeof(current)) == 1) {
        for (i = 0; i < count; i++) {
            if (strncmp(words[i], current, MAX_WORD_LEN) == 0) {
                result = duplicate_string(current, MAX_WORD_LEN);
                free(words);
                return result;
            }
        }
        if (count < MAX_WORDS) {
            if (store_word(words, count, current, sizeof(words[count])) != 0) {
                free(words);
                return NULL;
            }
            count++;
        } else {
            break;
        }
    }

    free(words);
    return duplicate_string("None", MAX_WORD_LEN);
}

static int run_test(const char *input)
{
    char *repeated;

    repeated = first_repeated_word(input);
    if (repeated == NULL) {
        if (printf("Memory allocation failure\n") < 0) {
            return -1;
        }
        return -1;
    }

    if (strncmp(repeated, "None", MAX_WORD_LEN) == 0) {
        if (printf("No repeated word found\n") < 0) {
            free(repeated);
            return -1;
        }
    } else {
        if (printf("First repeated word: %s\n", repeated) < 0) {
            free(repeated);
            return -1;
        }
    }

    free(repeated);
    return 0;
}

int main(void)
{
    const char *test1 = "the quick brown fox jumps over the lazy dog";
    const char *test2 = "hello world foo bar";
    const char *test3 = "one two two three";

    if (run_test(test1) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test2) != 0) {
        return EXIT_FAILURE;
    }

    if (run_test(test3) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The rewritten code addresses the flagged weaknesses as follows:

**`memcpy` in `duplicate_string` (CWE-120):** The original condition `len > 0 && len < max_len` was redundant and slightly confusing since `strnlen(src, max_len - 1)` already guarantees `len <= max_len - 1`. The buffer is allocated as `len + 1`, so the copy of exactly `len` bytes can never overflow. The simplified guard makes the invariant explicit: the destination is always sized from the measured length before copying.

**`memcpy` into `words[count]` (CWE-120):** This copy was extracted into a dedicated `store_word` helper that derives the copy length directly from the destination size via `strnlen(src, dest_size - 1)`, eliminating the manual clamping logic (`len > MAX_WORD_LEN - 1`) that duplicated bounds knowledge in two places. The length is now provably bounded by the destination capacity at the point of the copy.

**Statically-sized arrays `words` and `current` (CWE-119/CWE-120):** These remain fixed-size by design, but every write path is bounds-checked: `extract_word` truncates using `w + 1 < word_size` and always null-terminates within bounds (the dead clamp `w >= word_size` was removed since it was unreachable), and `store_word` never copies more than `dest_size - 1` bytes. Input length is capped by `strnlen(str, MAX_INPUT_LEN)`, so no unbounded reads occur either.

All allocations are checked, all paths free `words` before returning, and truncation semantics are preserved from the original.
