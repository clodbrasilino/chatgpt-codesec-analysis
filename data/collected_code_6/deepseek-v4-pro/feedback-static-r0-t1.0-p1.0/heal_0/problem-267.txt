```c
#include <stdio.h>

int sum_of_squares_odd(int n) {
    int sum = 0;
    int odd = 1;
    for (int i = 0; i < n; i++) {
        sum += odd * odd;
        odd += 2;
    }
    return sum;
}

int main(void) {
    int n;
    printf("Enter n: ");
    if (scanf("%d", &n) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    if (n < 0) {
        printf("n must be non-negative\n");
        return 1;
    }
    printf("Sum of squares of first %d odd natural numbers: %d\n", n, sum_of_squares_odd(n));
    return 0;
}
```
