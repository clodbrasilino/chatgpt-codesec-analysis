#include <stdio.h>

int min_of_three(int a, int b, int c);

int min_of_three(int a, int b, int c)
{
    int min = a;

    if (b < min)
    {
        min = b;
    }

    if (c < min)
    {
        min = c;
    }

    return min;
}

int main(void)
{
    int num1 = 0;
    int num2 = 0;
    int num3 = 0;
    int result = 0;

    printf("Enter three integers: ");

    result = scanf("%d %d %d", &num1, &num2, &num3);

    if (result != 3)
    {
        fprintf(stderr, "Error: Invalid input. Please enter three integers.\n");
        return 1;
    }

    printf("Minimum: %d\n", min_of_three(num1, num2, num3));

    return 0;
}
