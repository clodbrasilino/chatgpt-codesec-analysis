#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool is_woodall(unsigned long long n) {
    if (n == 0) {
        return false;
    }

    unsigned int i = 1;
    while (i < 64) {
        unsigned long long w = i * (1ULL << i) - 1;
        if (w == n) {
            return true;
        }
        if (w > n) {
            break;
        }
        i++;
    }

    return false;
}

int main(void) {
    unsigned long long num = 0;
    
    if (scanf("%llu", &num) != 1) {
        return EXIT_FAILURE;
    }

    if (is_woodall(num)) {
        printf("True\n");
    } else {
        printf("False\n");
    }

    return EXIT_SUCCESS;
}
