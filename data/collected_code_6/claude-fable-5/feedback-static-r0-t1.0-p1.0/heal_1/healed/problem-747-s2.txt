```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LCS_INPUT 1000U

static int max3(int a, int b, int c)
{
    int m = a;
    if (b > m) {
        m = b;
    }
    if (c > m) {
        m = c;
    }
    return m;
}

static char *lcs3(const char *s1, const char *s2, const char *s3)
{
    size_t l1;
    size_t l2;
    size_t l3;
    int ***dp;
    size_t i;
    size_t j;
    size_t k;
    char *result;
    size_t len;

    if (s1 == NULL || s2 == NULL || s3 == NULL) {
        return NULL;
    }

    l1 = strnlen(s1, MAX_LCS_INPUT + 1U);
    l2 = strnlen(s2, MAX_LCS_INPUT + 1U);
    l3 = strnlen(s3, MAX_LCS_INPUT + 1U);

    if (l1 > MAX_LCS_INPUT || l2 > MAX_LCS_INPUT || l3 > MAX_LCS_INPUT) {
        return NULL;
    }

    dp = malloc((l1 + 1U) * sizeof(int **));
    if (dp == NULL) {
        return NULL;
    }

    for (i = 0U; i <= l1; i++) {
        dp[i] = malloc((l2 + 1U) * sizeof(int *));
        if (dp[i] == NULL) {
            for (j = 0U; j < i; j++) {
                for (k = 0U; k <= l2; k++) {
                    free(dp[j][k]);
                }
                free(dp[j]);
            }
            free(dp);
            return NULL;
        }
        for (j = 0U; j <= l2; j++) {
            dp[i][j] = calloc(l3 + 1U, sizeof(int));
            if (dp[i][j] == NULL) {
                for (k = 0U; k < j; k++) {
                    free(dp[i][k]);
                }
                free(dp[i]);
                for (k = 0U; k < i; k++) {
                    size_t m;
                    for (m = 0U; m <= l2; m++) {
                        free(dp[k][m]);
                    }
                    free(dp[k]);
                }
                free(dp);
                return NULL;
            }
        }
    }

    for (i = 1U; i <= l1; i++) {
        for (j = 1U; j <= l2; j++) {
            for (k = 1U; k <= l3; k++) {
                if (s1[i - 1U] == s2[j - 1U] && s2[j - 1U] == s3[k - 1U]) {
                    dp[i][j][k] = dp[i - 1U][j - 1U][k - 1U] + 1;
                } else {
                    dp[i][j][k] = max3(dp[i - 1U][j][k],
                                       dp[i][j - 1U][k],
                                       dp[i][j][k - 1U]);
                }
            }
        }
    }

    len = (size_t)dp[l1][l2][l3];
    result = malloc(len + 1U);
    if (result != NULL) {
        size_t pos;
        result[len] = '\0';
        pos = len;
        i = l1;
        j = l2;
        k = l3;
        while (i > 0U && j > 0U && k > 0U && pos > 0U) {
            if (s1[i - 1U] == s2[j - 1U] && s2[j - 1U] == s3[k - 1U]) {
                pos--;
                result[pos] = s1[i - 1U];
                i--;
                j--;
                k--;
            } else if (dp[i - 1U][j][k] == dp[i][j][k]) {
                i--;
            } else if (dp[i][j - 1U][k] == dp[i][j][k]) {
                j--;
            } else {
                k--;
            }
        }
    }

    for (i = 0U; i <= l1; i++) {
        for (j = 0U; j <= l2; j++) {
            free(dp[i][j]);
        }
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    const char *a = "AGGT12";
    const char *b = "12TXAYB";
    const char *c = "12XBA";
    char *lcs;

    lcs = lcs3(a, b, c);
    if (lcs == NULL) {
        if (fprintf(stderr, "Failed to compute LCS\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("LCS of %s, %s and %s is: %s (length %zu)\n",
               a, b, c, lcs, strnlen(lcs, MAX_LCS_INPUT + 1U)) < 0) {
        free(lcs);
        return EXIT_FAILURE;
    }

    free(lcs);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **`strlen` over-read risk (CWE-126)**: Replaced all `strlen` calls with `strnlen` bounded by `MAX_LCS_INPUT + 1U`. This guarantees reads never exceed the expected maximum even if a non-terminated buffer is passed, and the subsequent length check still correctly rejects oversized (or unterminated) inputs.

2. **Variable scope of `pos`**: Moved the declaration of `pos` into the block where the result string is reconstructed, minimizing its scope as flagged by the static analyzer.

3. **Dead code (`total == 0U` always false)**: Removed the `total` variable and its always-false check entirely. Since `l1`, `l2`, `l3` are each capped at 1000, `(l1+1)*(l2+1)*(l3+1)` cannot overflow `size_t` on any realistic platform, so the check was meaningless.

4. **Magic number**: Introduced the `MAX_LCS_INPUT` macro to replace the hard-coded `1000U`, making the bound consistent across the length checks and `strnlen` calls.

The existing strengths of the code were preserved: correct cleanup on every allocation-failure path, NULL-pointer input validation, checked return values of `printf`/`fprintf`, and full deallocation of the 3D DP table before returning.
