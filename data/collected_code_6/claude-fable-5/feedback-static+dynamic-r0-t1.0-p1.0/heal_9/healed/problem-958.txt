```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 16
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size < ROMAN_BUFFER_SIZE) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used > buffer_size - sym_len - 1) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and applied fixes:**

1. **Flawfinder `memcpy` warning (CWE-120):** The original code was already defensively written — before every `memcpy`, it verifies that `sym_len` fits within `buffer_size` and that `used` will not exceed `buffer_size - sym_len - 1` (leaving room for the null terminator). The subtraction `buffer_size - sym_len - 1` is safe from unsigned underflow because the preceding `sym_len >= buffer_size` check guarantees `sym_len < buffer_size`. This check is preserved, so the `memcpy` can never overflow the destination.

2. **Flawfinder `char` array warning (CWE-119/CWE-120):** The statically sized `roman[ROMAN_BUFFER_SIZE]` buffer is protected because the buffer size is always passed via `sizeof(roman)` and validated inside `int_to_roman`. The fix strengthens the entry guard from `buffer_size < 2` to `buffer_size < ROMAN_BUFFER_SIZE`, which guarantees the caller's buffer can hold the worst case ("MMMDCCCLXXXVIII" for 3888 — 15 characters plus the null terminator, exactly 16 bytes) before any writing begins, failing fast rather than mid-conversion.

3. **Additional soundness points retained:** input range validation (1–3999), `scanf` return-value checking, `NULL` pointer check, buffer zero-initialization via `memset`, and clearing the buffer (`buffer[0] = '\0'`) on any error path so callers never read stale/partial data.

The Flawfinder hits were level-2 informational warnings; both were already mitigated by the existing bounds checks, and the rewrite makes the guarantee explicit and stricter.
