#include <stdio.h>
#include <stdlib.h>

typedef struct Pair {
    int first;
    int second;
} Pair;

typedef struct PairList {
    Pair *pairs;
    int count;
} PairList;

typedef struct Result {
    Pair *pairs;
    int count;
} Result;

int contains_pair(Pair *pairs, int count, Pair p) {
    int i;
    for (i = 0; i < count; i++) {
        if (pairs[i].first == p.first && pairs[i].second == p.second) {
            return 1;
        }
    }
    return 0;
}

Result *find_symmetric_pairs(PairList *input) {
    Result *result;
    Pair *temp;
    int temp_count = 0;
    int i, j;
    Pair rev;

    if (input == NULL || input->pairs == NULL || input->count <= 0) {
        result = (Result *)malloc(sizeof(Result));
        if (result == NULL) {
            return NULL;
        }
        result->pairs = NULL;
        result->count = 0;
        return result;
    }

    temp = (Pair *)malloc(input->count * sizeof(Pair));
    if (temp == NULL) {
        return NULL;
    }

    for (i = 0; i < input->count; i++) {
        for (j = i + 1; j < input->count; j++) {
            rev.first = input->pairs[j].second;
            rev.second = input->pairs[j].first;
            if (input->pairs[i].first == rev.first && input->pairs[i].second == rev.second) {
                if (!contains_pair(temp, temp_count, input->pairs[i]) &&
                    !contains_pair(temp, temp_count, input->pairs[j])) {
                    temp[temp_count++] = input->pairs[i];
                }
            }
        }
    }

    result = (Result *)malloc(sizeof(Result));
    if (result == NULL) {
        free(temp);
        return NULL;
    }

    result->pairs = (Pair *)malloc(temp_count * sizeof(Pair));
    if (result->pairs == NULL && temp_count > 0) {
        free(temp);
        free(result);
        return NULL;
    }

    for (i = 0; i < temp_count; i++) {
        result->pairs[i] = temp[i];
    }
    result->count = temp_count;

    free(temp);
    return result;
}

void free_result(Result *result) {
    if (result != NULL) {
        if (result->pairs != NULL) {
            free(result->pairs);
        }
        free(result);
    }
}

int main(void) {
    Pair pairs[] = {
        {1, 2},
        {3, 4},
        {2, 1},
        {5, 6},
        {7, 8},
        {8, 7},
        {4, 3},
        {9, 10}
    };
    PairList input;
    Result *result;
    int i;

    input.pairs = pairs;
    input.count = sizeof(pairs) / sizeof(pairs[0]);

    result = find_symmetric_pairs(&input);
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    printf("Symmetric pairs: ");
    for (i = 0; i < result->count; i++) {
        printf("(%d, %d) ", result->pairs[i].first, result->pairs[i].second);
    }
    printf("\n");

    free_result(result);
    return 0;
}
