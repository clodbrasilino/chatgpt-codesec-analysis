#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>

typedef struct {
    uint64_t val;
    int prime_idx;
} HeapNode;

typedef struct {
    HeapNode* array;
    int size;
    int capacity;
} MinHeap;

MinHeap* createMinHeap(int capacity) {
    MinHeap* minHeap = (MinHeap*)malloc(sizeof(MinHeap));
    if (!minHeap) return NULL;
    minHeap->capacity = capacity;
    minHeap->size = 0;
    minHeap->array = (HeapNode*)malloc(sizeof(HeapNode) * capacity);
    if (!minHeap->array) {
        free(minHeap);
        return NULL;
    }
    return minHeap;
}

void swap(HeapNode* a, HeapNode* b) {
    HeapNode temp = *a;
    *a = *b;
    *b = temp;
}

void minHeapify(MinHeap* minHeap, int idx) {
    int smallest = idx;
    int left = 2 * idx + 1;
    int right = 2 * idx + 2;

    if (left < minHeap->size && minHeap->array[left].val < minHeap->array[smallest].val)
        smallest = left;

    if (right < minHeap->size && minHeap->array[right].val < minHeap->array[smallest].val)
        smallest = right;

    if (smallest != idx) {
        swap(&minHeap->array[smallest], &minHeap->array[idx]);
        minHeapify(minHeap, smallest);
    }
}

void insertMinHeap(MinHeap* minHeap, uint64_t val, int prime_idx) {
    if (minHeap->size == minHeap->capacity) return;

    minHeap->size++;
    int i = minHeap->size - 1;
    minHeap->array[i].val = val;
    minHeap->array[i].prime_idx = prime_idx;

    while (i != 0 && minHeap->array[(i - 1) / 2].val > minHeap->array[i].val) {
        swap(&minHeap->array[i], &minHeap->array[(i - 1) / 2]);
        i = (i - 1) / 2;
    }
}

HeapNode extractMin(MinHeap* minHeap) {
    if (minHeap->size <= 0) {
        HeapNode empty = {0, -1};
        return empty;
    }
    if (minHeap->size == 1) {
        minHeap->size--;
        return minHeap->array[0];
    }

    HeapNode root = minHeap->array[0];
    minHeap->array[0] = minHeap->array[minHeap->size - 1];
    minHeap->size--;
    minHeapify(minHeap, 0);

    return root;
}

void freeMinHeap(MinHeap* minHeap) {
    if (minHeap) {
        free(minHeap->array);
        free(minHeap);
    }
}

int nthSuperUglyNumber(int n, int* primes, int k) {
    if (n <= 0 || !primes || k <= 0) return 0;

    uint64_t* ugly = (uint64_t*)malloc(sizeof(uint64_t) * n);
    if (!ugly) return 0;
    
    int* indices = (int*)calloc(k, sizeof(int));
    if (!indices) {
        free(ugly);
        return 0;
    }

    MinHeap* minHeap = createMinHeap(k);
    if (!minHeap) {
        free(ugly);
        free(indices);
        return 0;
    }

    ugly[0] = 1;

    for (int i = 0; i < k; i++) {
        insertMinHeap(minHeap, (uint64_t)primes[i], i);
    }

    for (int i = 1; i < n; i++) {
        HeapNode minNode = extractMin(minHeap);
        ugly[i] = minNode.val;
        int p_idx = minNode.prime_idx;

        indices[p_idx]++;
        uint64_t next_val = ugly[indices[p_idx]] * primes[p_idx];
        insertMinHeap(minHeap, next_val, p_idx);

        while (minHeap->size > 0 && minHeap->array[0].val == ugly[i]) {
            HeapNode dupNode = extractMin(minHeap);
            int dup_p_idx = dupNode.prime_idx;
            indices[dup_p_idx]++;
            uint64_t dup_next_val = ugly[indices[dup_p_idx]] * primes[dup_p_idx];
            insertMinHeap(minHeap, dup_next_val, dup_p_idx);
        }
    }

    int result = (int)ugly[n - 1];

    free(ugly);
    free(indices);
    freeMinHeap(minHeap);

    return result;
}

int main(void) {
    int primes[] = {2, 7, 13, 19};
    int k = sizeof(primes) / sizeof(primes[0]);
    int n = 12;

    int result = nthSuperUglyNumber(n, primes, k);
    printf("%d\n", result);

    return 0;
}
