```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;
    char tmp;
    char *endptr;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }
    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }
    if (j <= i || j >= len) {
        return -1;
    }

    tmp = digits[i];
    digits[i] = digits[j];
    digits[j] = tmp;

    for (int a = i + 1, b = len - 1; a < b && a >= 0 && b < len; a++, b--) {
        tmp = digits[a];
        digits[a] = digits[b];
        digits[b] = tmp;
    }

    if ((size_t)len < sizeof(digits)) {
        digits[len] = '\0';
    } else {
        return -1;
    }

    errno = 0;
    result = strtoll(digits, &endptr, 10);
    if (errno == ERANGE || endptr == digits || *endptr != '\0') {
        return -1;
    }

    if (result <= n) {
        return -1;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            if (printf("next_bigger(%lld) = no bigger number exists\n", inputs[k]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("next_bigger(%lld) = %lld\n", inputs[k], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The flagged weakness (CWE-119/CWE-120, statically-sized buffer `digits[DIGITS_BUF_SIZE]`) is already fully mitigated in this code, so the rewrite preserves the existing defenses rather than restructuring them:

1. **Bounded write via `snprintf`**: The buffer is populated with `snprintf(digits, sizeof(digits), ...)`, which cannot overflow. The return value is checked both for encoding errors (`len < 0`) and truncation (`(size_t)len >= sizeof(digits)`), rejecting any input whose decimal representation would not fit. A 64-bit `long long` needs at most 20 characters plus NUL, so 32 bytes is ample headroom.

2. **Zero-initialization**: `memset` guarantees the buffer is NUL-terminated regardless of subsequent operations.

3. **Index bounds discipline**: Every index (`i`, `j`, `a`, `b`) is guarded against underflow and overflow before dereferencing (`i >= 0`, `j >= len` check, loop conditions `a >= 0 && b < len`), so no out-of-bounds read or write is possible.

4. **Explicit NUL termination check**: Before writing `digits[len] = '\0'`, the code re-verifies `len` is in range.

5. **Safe parsing**: `strtoll` is used with `errno`/`endptr` validation instead of `atoll`, catching overflow (`ERANGE`) and malformed results, plus a semantic sanity check (`result <= n`).

6. **Checked I/O in `main`**: `printf` return values are verified.

The code is correct, defensive, and warning-clean; the flawfinder hit is a false positive given the consistent bounds checking.
