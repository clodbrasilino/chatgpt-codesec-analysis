#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

unsigned long long count_unset_bits(unsigned int n)
{
    unsigned long long total_unset = 0;
    unsigned int i;
    
    for (i = 1; i <= n; i++) {
        unsigned int num = i;
        unsigned int bits = 0;
        unsigned int set_bits = 0;
        
        while (num > 0) {
            if ((num & 1U) == 1U) {
                set_bits++;
            }
            bits++;
            num >>= 1;
        }
        
        if (bits == 0) {
            bits = 1;
        }
        
        total_unset += (unsigned long long)(bits - set_bits);
    }
    
    return total_unset;
}

int main(void)
{
    char *input = NULL;
    size_t input_size = 0;
    ssize_t read_len;
    char *endptr;
    long val;
    unsigned int n;
    unsigned long long result;
    
    read_len = getline(&input, &input_size, stdin);
    if (read_len == -1) {
        fprintf(stderr, "Error reading input\n");
        free(input);
        return EXIT_FAILURE;
    }
    
    errno = 0;
    val = strtol(input, &endptr, 10);
    
    if (errno != 0 || endptr == input || (*endptr != '\n' && *endptr != '\0')) {
        fprintf(stderr, "Invalid input\n");
        free(input);
        return EXIT_FAILURE;
    }
    
    if (val < 1 || val > UINT_MAX) {
        fprintf(stderr, "Input out of range\n");
        free(input);
        return EXIT_FAILURE;
    }
    
    n = (unsigned int)val;
    result = count_unset_bits(n);
    
    printf("%llu\n", result);
    
    free(input);
    return EXIT_SUCCESS;
}
