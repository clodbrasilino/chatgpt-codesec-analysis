```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int key;
    int value;
    struct Node* next;
} Node;

typedef struct {
    Node** buckets;
    size_t size;
} HashMap;

HashMap* create_hashmap(size_t size) {
    if (size == 0) return NULL;
    
    HashMap* map = (HashMap*)malloc(sizeof(HashMap));
    if (!map) return NULL;
    
    map->size = size;
    map->buckets = (Node**)calloc(size, sizeof(Node*));
    
    if (!map->buckets) {
        free(map);
        return NULL;
    }
    
    return map;
}

void free_hashmap(HashMap* map) {
    if (!map) return;
    
    for (size_t i = 0; i < map->size; i++) {
        Node* current = map->buckets[i];
        while (current) {
            Node* temp = current;
            current = current->next;
            free(temp);
        }
    }
    
    free(map->buckets);
    free(map);
}

int main(void) {
    HashMap* map = create_hashmap(10);
    
    if (map) {
        free_hashmap(map);
    }
    
    return 0;
}
```
