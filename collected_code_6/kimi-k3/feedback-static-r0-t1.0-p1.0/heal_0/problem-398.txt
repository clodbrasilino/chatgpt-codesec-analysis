#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

static int sum_digits(int n)
{
    int sum = 0;
    unsigned int value;

    if (n < 0) {
        value = (unsigned int)(-(n + 1)) + 1U;
    } else {
        value = (unsigned int)n;
    }

    while (value > 0U) {
        sum += (int)(value % 10U);
        value /= 10U;
    }

    return sum;
}

int compute_digit_sums(const int *numbers, size_t count, int *results)
{
    size_t i;

    if (numbers == NULL || results == NULL) {
        return -1;
    }

    for (i = 0; i < count; ++i) {
        results[i] = sum_digits(numbers[i]);
    }

    return 0;
}

int main(void)
{
    int numbers[] = {123, -456, 789, 0, -1001};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int *results = NULL;
    size_t i;

    results = (int *)malloc(count * sizeof(int));
    if (results == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (compute_digit_sums(numbers, count, results) != 0) {
        fprintf(stderr, "Computation failed\n");
        free(results);
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; ++i) {
        printf("Sum of digits of %d is %d\n", numbers[i], results[i]);
    }

    free(results);
    return EXIT_SUCCESS;
}
