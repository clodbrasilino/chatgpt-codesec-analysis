```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

int main(void)
{
    char buffer[64];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened, and the Flawfinder warning about the statically-sized `char buffer[64]` (CWE-119/CWE-120) is a false positive in this context. Here's why the rewritten code remains safe:

1. **Bounded read**: `fgets(buffer, sizeof(buffer), stdin)` never writes more than 63 characters plus a NUL terminator, so the fixed-size buffer cannot overflow.

2. **Guaranteed termination**: The explicit `buffer[sizeof(buffer) - 1] = '\0'` is defensive redundancy — `fgets` already NUL-terminates, but this makes the invariant explicit.

3. **Robust integer parsing**: `parse_int` uses `strtol` with full error checking — `errno == ERANGE` for overflow, `endptr == str` for empty/non-numeric input, trailing-character validation (allowing only `\n` from `fgets`), and an explicit range check before narrowing `long` to `int` (relevant on LP64 platforms where `long` is wider than `int`).

4. **Integer underflow protection**: `closest_smaller` rejects `INT_MIN`, preventing signed integer underflow (undefined behavior) when computing `n - 1`.

5. **NULL checks and return-value checks throughout**: including the `printf` return value.

No functional changes were required — only the removal of comments as requested. The code follows defense-in-depth practices appropriate for untrusted stdin input.
