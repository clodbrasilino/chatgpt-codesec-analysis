#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <limits.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

int64_t octagonal_number(int64_t n) {
    if (n < 0 || n > 3037000499LL) {
        return -1;
    }
    return n * (3 * n - 2);
}

int main(void) {
    char buffer[64];
    char *endptr;
    int64_t n;
    size_t len;
    bool truncated = false;
    int c;
    
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return 1;
    }
    
    len = strlen(buffer);
    
    if (len == sizeof(buffer) - 1 && buffer[len - 1] != '\n') {
        while ((c = getchar()) != '\n' && c != EOF) {
            truncated = true;
        }
        if (truncated) {
            return 1;
        }
    }
    
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }
    
    errno = 0;
    n = strtoimax(buffer, &endptr, 10);
    
    if (errno != 0 || endptr == buffer || *endptr != '\0') {
        return 1;
    }
    
    int64_t result = octagonal_number(n);
    if (result == -1) {
        return 1;
    }
    
    printf("%" PRId64 "\n", result);
    return 0;
}
