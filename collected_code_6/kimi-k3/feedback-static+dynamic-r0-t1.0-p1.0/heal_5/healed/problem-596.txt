```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int id;
    char *name;
    double value;
} Tuple;

size_t get_tuple_size(const Tuple *t) {
    if (t == NULL) {
        return 0;
    }
    return sizeof(*t);
}

int main(void) {
    Tuple t;
    size_t size;
    const char name_str[] = "example";
    size_t name_len = sizeof(name_str) - 1;
    
    memset(&t, 0, sizeof(Tuple));
    t.id = 1;
    
    t.name = malloc(name_len + 1);
    if (t.name == NULL) {
        return 1;
    }
    snprintf(t.name, name_len + 1, "%s", name_str);
    
    t.value = 3.14;
    
    size = get_tuple_size(&t);
    printf("Size of tuple: %zu bytes\n", size);
    
    free(t.name);
    
    return 0;
}
```
