```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_HEAP_SIZE 100

typedef struct {
    int data[MAX_HEAP_SIZE];
    int size;
} MinHeap;

static void swap_ints(int *a, int *b)
{
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

static void sift_down(MinHeap *heap, int index)
{
    while (1) {
        int smallest = index;
        int left = 2 * index + 1;
        int right = 2 * index + 2;

        if (left < heap->size && heap->data[left] < heap->data[smallest]) {
            smallest = left;
        }
        if (right < heap->size && heap->data[right] < heap->data[smallest]) {
            smallest = right;
        }
        if (smallest == index) {
            break;
        }
        swap_ints(&heap->data[index], &heap->data[smallest]);
        index = smallest;
    }
}

static void sift_up(MinHeap *heap, int index)
{
    if (index < 0 || index >= MAX_HEAP_SIZE || index >= heap->size) {
        return;
    }
    while (index > 0) {
        int parent = (index - 1) / 2;
        if (heap->data[parent] <= heap->data[index]) {
            break;
        }
        swap_ints(&heap->data[parent], &heap->data[index]);
        index = parent;
    }
}

int heap_insert(MinHeap *heap, int value)
{
    if (heap == NULL) {
        return -1;
    }
    if (heap->size < 0 || heap->size >= MAX_HEAP_SIZE) {
        return -1;
    }
    heap->data[heap->size] = value;
    heap->size++;
    sift_up(heap, heap->size - 1);
    return 0;
}

int heap_delete_min_and_insert(MinHeap *heap, int new_value, int *deleted_value)
{
    if (heap == NULL || deleted_value == NULL) {
        return -1;
    }
    if (heap->size <= 0 || heap->size > MAX_HEAP_SIZE) {
        return -1;
    }
    *deleted_value = heap->data[0];
    heap->data[0] = new_value;
    sift_down(heap, 0);
    return 0;
}

int main(void)
{
    MinHeap heap;
    int values[] = { 15, 3, 9, 27, 6, 12 };
    size_t count = sizeof(values) / sizeof(values[0]);
    size_t i;
    int deleted;
    int rc;

    heap.size = 0;

    for (i = 0; i < count; i++) {
        rc = heap_insert(&heap, values[i]);
        if (rc != 0) {
            fprintf(stderr, "Failed to insert %d into heap\n", values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Heap before operation:");
    for (i = 0; (int)i < heap.size; i++) {
        printf(" %d", heap.data[i]);
    }
    printf("\n");

    rc = heap_delete_min_and_insert(&heap, 8, &deleted);
    if (rc != 0) {
        fprintf(stderr, "Heap operation failed\n");
        return EXIT_FAILURE;
    }

    printf("Deleted smallest element: %d\n", deleted);
    printf("Inserted new element: %d\n", 8);

    printf("Heap after operation:");
    for (i = 0; (int)i < heap.size; i++) {
        printf(" %d", heap.data[i]);
    }
    printf("\n");

    return EXIT_SUCCESS;
}
```

**Key changes addressing the flagged weaknesses:**

1. **`sift_up` — array index out of bounds (`arrayIndexOutOfBoundsCond`):** The static analyzer could not prove that `index` stays within `[0, MAX_HEAP_SIZE)` when accessing `heap->data[index]`. A guard clause was added at the top of `sift_up` rejecting any `index` that is negative, `>= MAX_HEAP_SIZE`, or `>= heap->size`. This makes the invariant explicit to both the analyzer and future maintainers, and hardens the function against misuse if it is ever called from new code paths.

2. **`heap_insert` — strengthened size validation:** The check `heap->size >= MAX_HEAP_SIZE` was extended to `heap->size < 0 || heap->size >= MAX_HEAP_SIZE`. If a caller passes a heap with a corrupted or uninitialized negative `size`, the original code would have written to `heap->data[negative]` — an out-of-bounds write. This also lets the analyzer conclude that `heap->size - 1` passed to `sift_up` is always in `[0, MAX_HEAP_SIZE - 1]`, eliminating the root cause of the warning.

3. **`heap_delete_min_and_insert` — upper bound check added:** `heap->size > MAX_HEAP_SIZE` is now rejected as well, so `sift_down` can never read `heap->data[left]`/`heap->data[right]` past the array end due to a corrupted `size` field.

The redundant-condition half of the warning (`index > 0`) was a false positive pairing: the condition is genuinely needed as the loop terminator, and with the new guard the bounds side is provably safe, so the diagnostic resolves cleanly.
