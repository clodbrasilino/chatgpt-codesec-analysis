#include <stdio.h>
#include <stdlib.h>

int max(const int a, const int b) {
    return (a > b) ? a : b;
}

int* calculate_max_sum(const int arr[], int n, int* result) {
    int *msis = (int*)malloc(n * sizeof(int));
    if (msis == NULL) {
        *result = 0;
        return NULL;
    }
    for (int i = 0; i < n; i++) {
        msis[i] = arr[i];
    }
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[i] > arr[j] && msis[i] < msis[j] + arr[i]) {
                msis[i] = msis[j] + arr[i];
            }
        }
    }

    int *msds = (int*)malloc(n * sizeof(int));
    if (msds == NULL) {
        free(msis);
        *result = 0;
        return NULL;
    }
    for (int i = 0; i < n; i++) {
        msds[i] = arr[i];
    }
    for (int i = n - 2; i >= 0; i--) {
        for (int j = n - 1; j > i; j--) {
            if (arr[i] > arr[j] && msds[i] < msds[j] + arr[i]) {
                msds[i] = msds[j] + arr[i];
            }
        }
    }

    *result = 0;
    for (int i = 0; i < n; i++) {
        *result = max(*result, msis[i] + msds[i] - arr[i]);
    }

    free(msis);
    free(msds);
    return NULL;
}

int main() {
    const int arr[] = {1, 15, 51, 45, 33, 100, 12, 18, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    int result = 0;

    calculate_max_sum(arr, n, &result);

    printf("Maximum sum of bi-tonic sub-sequence: %d\n", result);

    return 0;
}
