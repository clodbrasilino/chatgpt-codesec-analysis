#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096
#define MAX_LIST_SIZE 4096

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size);
void free_tuple(Tuple *t);
void print_tuple(const Tuple *t);

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;
    size_t str_bytes;
    size_t list_bytes;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    if (list_size > MAX_LIST_SIZE) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(*list)) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    if (len > SIZE_MAX - 1) {
        return NULL;
    }

    str_bytes = len + 1;

    t = malloc(sizeof(*t));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(str_bytes);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    if (len > 0) {
        memcpy(t->str, str, len);
    }
    t->str[len] = '\0';

    if (list_size > 0) {
        list_bytes = list_size * sizeof(*t->list);
        t->list = malloc(list_bytes);
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_bytes);
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->str == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *str = "hello";
    int list[] = {1, 2, 3, 4, 5};
    size_t list_size = sizeof(list) / sizeof(list[0]);
    Tuple *t;

    t = create_tuple(str, list, list_size);
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses:**

Both `memcpy` warnings flagged by Flawfinder (CWE-120) are **false positives** in this code, and the original code already contained the correct mitigations. Here's why each copy is safe:

1. **String copy (`memcpy(t->str, str, len)`):** The destination buffer is allocated with exactly `len + 1` bytes, where `len` is bounded by `strnlen(str, MAX_STR_LEN)` and validated with `len >= MAX_STR_LEN` rejection. The copy writes exactly `len` bytes, and the null terminator is written explicitly at `t->str[len]`. The destination can always hold the source data by construction.

2. **List copy (`memcpy(t->list, list, list_bytes)`):** The destination is allocated with exactly `list_bytes = list_size * sizeof(*t->list)` bytes, and the same value is used for the copy length. Integer overflow in the multiplication is prevented upfront by the `list_size > SIZE_MAX / sizeof(*list)` check, and `list_size` is additionally capped at `MAX_LIST_SIZE`.

Other defensive properties preserved in the rewrite:

- **NULL parameter validation** before any dereference, including the subtle case of `list == NULL` with a nonzero `list_size`.
- **No memory leaks on error paths** — each allocation failure frees previously allocated resources in reverse order.
- **`strnlen` instead of `strlen`** protects against unterminated input strings.
- **`free_tuple` is NULL-safe**, matching `free()` semantics.
- **Empty-list handling** sets `t->list = NULL` rather than calling `malloc(0)`, avoiding implementation-defined behavior.

No functional changes were required — the code was already hardened against the flagged CWE-120 concerns; the rewrite simply strips the annotations while keeping every guard intact.
