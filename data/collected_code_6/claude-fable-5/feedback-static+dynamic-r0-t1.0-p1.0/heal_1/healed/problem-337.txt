```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096

bool matches_word_at_end(const char *str, const char *word)
{
    size_t str_len;
    size_t word_len;
    size_t end;

    if (str == NULL || word == NULL) {
        return false;
    }

    str_len = strnlen(str, MAX_INPUT_LEN);
    if (str_len == MAX_INPUT_LEN) {
        return false;
    }

    word_len = strnlen(word, MAX_INPUT_LEN);
    if (word_len == MAX_INPUT_LEN) {
        return false;
    }

    if (word_len == 0 || str_len < word_len) {
        return false;
    }

    end = str_len;
    while (end > 0 && ispunct((unsigned char)str[end - 1])) {
        end--;
    }

    if (end < word_len) {
        return false;
    }

    if (strncmp(&str[end - word_len], word, word_len) != 0) {
        return false;
    }

    if (end > word_len) {
        unsigned char before = (unsigned char)str[end - word_len - 1];
        if (isalnum(before) || before == '_') {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *sentences[] = {
        "The quick brown fox jumps over the lazy dog.",
        "Hello, world!!!",
        "No punctuation here world",
        "This does not end with the word",
        "worldly matters...",
        ""
    };
    const char *word = "world";
    size_t i;
    size_t count = sizeof(sentences) / sizeof(sentences[0]);

    for (i = 0; i < count; i++) {
        bool result = matches_word_at_end(sentences[i], word);
        if (printf("\"%s\" ends with \"%s\": %s\n",
                   sentences[i], word,
                   result ? "yes" : "no") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (matches_word_at_end(sentences[0], "dog")) {
        if (printf("First sentence ends with \"dog\"\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **Replaced `strlen` with `strnlen`** (CWE-126 mitigation): Both calls to `strlen` were flagged by Flawfinder because `strlen` will read past the buffer boundary if the string is not null-terminated. `strnlen` bounds the scan to `MAX_INPUT_LEN` bytes, preventing an unbounded over-read.

2. **Added a sentinel check on the bounded length**: If `strnlen` returns exactly `MAX_INPUT_LEN`, no null terminator was found within the bound, so the input is treated as untrusted/malformed and the function safely returns `false` rather than proceeding with a potentially unterminated buffer.

3. **Preserved existing defensive measures**: The original `NULL` checks, the `(unsigned char)` casts before `ispunct`/`isalnum` (avoiding undefined behavior with negative `char` values), the use of bounded `strncmp`, and the `printf` return-value checks were all retained, as they were already correct.

The logic and observable behavior for all valid, null-terminated inputs remain identical to the original program.
