#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

int ends_with_alphanumeric(const char *str)
{
    regex_t regex;
    int ret;
    int result = 0;

    if (str == NULL) {
        return 0;
    }

    ret = regcomp(&regex, "[a-zA-Z0-9]$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        char errbuf[256];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, str, 0, NULL, 0);
    if (ret == 0) {
        result = 1;
    } else if (ret == REG_NOMATCH) {
        result = 0;
    } else {
        char errbuf[256];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        fprintf(stderr, "Regex execution failed: %s\n", errbuf);
        result = -1;
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "world!",
        "test_",
        "abcXYZ",
        "ends with space ",
        "9"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = ends_with_alphanumeric(tests[i]);
        if (result < 0) {
            fprintf(stderr, "Error checking string: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (result == 1) {
            printf("\"%s\" ends with an alphanumeric character.\n", tests[i]);
        } else {
            printf("\"%s\" does not end with an alphanumeric character.\n", tests[i]);
        }
    }

    return EXIT_SUCCESS;
}
