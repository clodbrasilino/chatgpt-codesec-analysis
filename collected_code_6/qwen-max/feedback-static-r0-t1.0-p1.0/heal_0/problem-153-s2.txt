```c
#include <stdio.h>

typedef struct {
    double x;
    double y;
} Point;

Point findParabolaVertex(double a, double b, double c) {
    Point vertex;
    vertex.x = -b / (2 * a);
    vertex.y = c - (b * b) / (4 * a);
    return vertex;
}

int main() {
    double a, b, c;
    printf("Enter the coefficients a, b, and c for the parabola (ax^2 + bx + c): ");
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return 1;
    }
    Point vertex = findParabolaVertex(a, b, c);
    printf("The vertex of the parabola is at: (%.2f, %.2f)\n", vertex.x, vertex.y);
    return 0;
}
```
