#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool has_31_days(const char *month);
static void discard_remaining_input(void);

static bool has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char lowered[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    memset(lowered, 0, sizeof(lowered));

    for (i = 0; i < len && i < sizeof(lowered) - 1; i++) {
        lowered[i] = (char)tolower((unsigned char)month[i]);
    }
    lowered[i] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(lowered, months_31[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No input provided.\n");
        return EXIT_FAILURE;
    }

    if (has_31_days(input)) {
        printf("%s has 31 days.\n", input);
    } else {
        printf("%s does not have 31 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char lowered[MAX_INPUT_LEN]` (CWE-119/CWE-120):** The Flawfinder warning is a false positive here. The buffer is fully mitigated by: `strnlen` bounding the length to `MAX_INPUT_LEN`, an early return when `len >= MAX_INPUT_LEN`, the loop condition `i < sizeof(lowered) - 1` guaranteeing space for the terminator, and explicit null termination after the loop.

2. **`getchar()` in a loop (CWE-120/CWE-20):** No buffer is written to in `discard_remaining_input()`; characters are read and discarded until newline or `EOF`, so there is no overflow risk. The `EOF` check also prevents an infinite loop on closed input streams.

3. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Mitigated by using `fgets` with `sizeof(input)` (which never overflows), forced null termination at `input[sizeof(input) - 1]`, `strnlen` with an explicit bound, and draining leftover stdin data when the input line exceeds the buffer.

The code follows defensive-programming best practices already — bounded reads, explicit termination, `NULL` checks, `EOF` handling, and casting to `unsigned char` before `tolower` to avoid undefined behavior — so the rewrite preserves the logic with the comments removed, as no actual fixes were required.
