#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int max_subarray_sum(const int *arr, int size) {
    if (size == 0) return 0;
    int max_so_far = arr[0];
    int current_max = arr[0];
    for (int i = 1; i < size; i++) {
        if (current_max < 0) {
            current_max = arr[i];
        } else {
            if (arr[i] > INT_MAX - current_max) {
                current_max = arr[i];
            } else {
                current_max += arr[i];
            }
        }
        if (current_max > max_so_far) max_so_far = current_max;
    }
    return max_so_far;
}

int max_sum_of_lists(int * const *lists, const int *sizes, int num_lists) {
    if (lists == NULL || sizes == NULL || num_lists <= 0) return 0;
    int max_sum = INT_MIN;
    for (int i = 0; i < num_lists; i++) {
        if (lists[i] == NULL || sizes[i] <= 0) continue;
        int sum = max_subarray_sum(lists[i], sizes[i]);
        if (sum > max_sum) max_sum = sum;
    }
    return (max_sum == INT_MIN) ? 0 : max_sum;
}

int main(void) {
    int list1[] = {1, -2, 3, 4, -1, 2, 1, -5, 4};
    int list2[] = {-1, -3, -2, -4};
    int list3[] = {5, -1, 2, 3, -2};
    
    int sizes[] = {9, 4, 5};
    int *lists[] = {list1, list2, list3};
    
    int result = max_sum_of_lists(lists, sizes, 3);
    printf("%d\n", result);
    
    return 0;
}
