#include <stdio.h>
#include <stdbool.h>

bool has_increasing_trend(const int *sequence, size_t length) {
    if (sequence == NULL || length < 2) {
        return false;
    }

    for (size_t i = 0; i < length - 1; i++) {
        if (sequence[i] >= sequence[i + 1]) {
            return false;
        }
    }

    return true;
}

int main(void) {
    int seq1[] = {1, 2, 3, 4, 5};
    size_t len1 = sizeof(seq1) / sizeof(seq1[0]);

    int seq2[] = {1, 3, 2, 4, 5};
    size_t len2 = sizeof(seq2) / sizeof(seq2[0]);

    if (has_increasing_trend(seq1, len1)) {
        printf("seq1 has an increasing trend.\n");
    } else {
        printf("seq1 does not have an increasing trend.\n");
    }

    if (has_increasing_trend(seq2, len2)) {
        printf("seq2 has an increasing trend.\n");
    } else {
        printf("seq2 does not have an increasing trend.\n");
    }

    return 0;
}
