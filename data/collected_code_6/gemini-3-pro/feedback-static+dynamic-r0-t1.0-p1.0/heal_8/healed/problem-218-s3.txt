int min_operations(int a, int b) {
    if (a == b) return 0;
    if (a < b) {
        int diff = b - a;
        if (diff % 2 != 0) return 1;
        return 2;
    } else {
        int diff = a - b;
        if (diff % 2 == 0) return 1;
        return 2;
    }
}
