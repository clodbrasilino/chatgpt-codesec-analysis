#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 256

char *remove_even_chars(const char *input, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, max_len);

    result = malloc(len / 2 + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (i % 2 != 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char *output;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    output = remove_even_chars(buffer, sizeof(buffer));
    if (output == NULL) {
        fprintf(stderr, "Error processing string\n");
        return EXIT_FAILURE;
    }

    printf("Result: %s\n", output);
    free(output);

    return EXIT_SUCCESS;
}

**Analysis of the original code and why this version is safe:**

The Flawfinder warning about the statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120) is already properly mitigated, so the code required no functional changes:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing the classic `gets()`-style overflow. Its return value is checked for I/O errors and EOF.

2. **Guaranteed null-termination** — `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure (fgets already null-terminates on success), ensuring the buffer is always a valid C string.

3. **Bounded length calculation** — `strnlen(buffer, sizeof(buffer))` is used instead of `strlen()`, so even a non-terminated buffer could not cause an out-of-bounds read.

4. **Safe allocation in `remove_even_chars`** — the output buffer of `len / 2 + 1` bytes is exactly sufficient: for a string of length `len`, exactly `len / 2` characters have odd indices, plus one byte for the terminator. `malloc` failure and `NULL` input are both handled.

5. **No leaks** — the heap-allocated result is freed, and all error paths return distinct exit codes.
