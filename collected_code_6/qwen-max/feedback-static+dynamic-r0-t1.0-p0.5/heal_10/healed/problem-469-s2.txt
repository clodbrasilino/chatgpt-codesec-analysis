#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int maxProfit(int k, const int* prices, int pricesSize) {
    if (pricesSize == 0 || k == 0) return 0;
    if (2 * k >= pricesSize) {
        int profit = 0;
        for (int i = 1; i < pricesSize; i++) {
            if (prices[i] > prices[i - 1]) {
                profit += prices[i] - prices[i - 1];
            }
        }
        return profit;
    }

    int *buy = (int*)malloc((k + 1) * sizeof(int));
    int *sell = (int*)malloc((k + 1) * sizeof(int));
    for (int i = 0; i <= k; i++) {
        buy[i] = INT_MAX;
        sell[i] = 0;
    }

    for (int i = 0; i < pricesSize; i++) {
        for (int j = 1; j <= k; j++) {
            buy[j] = (buy[j] < prices[i] - sell[j - 1]) ? buy[j] : prices[i];
            sell[j] = (sell[j] > prices[i] - buy[j]) ? sell[j] : prices[i];
        }
    }

    int max_profit = 0;
    for (int i = 1; i <= k; i++) {
        if (sell[i] - buy[i] > max_profit) {
            max_profit = sell[i] - buy[i];
        }
    }

    free(buy);
    free(sell);
    return max_profit;
}

int main() {
    int prices[] = {3, 3, 5, 0, 0, 3, 1, 4};
    int pricesSize = sizeof(prices) / sizeof(prices[0]);
    int k = 2;
    int result = maxProfit(k, prices, pricesSize);
    printf("Maximum Profit: %d\n", result);

    int prices1[] = {1, 2, 3, 4, 5};
    int pricesSize1 = sizeof(prices1) / sizeof(prices1[0]);
    k = 2;
    result = maxProfit(k, prices1, pricesSize1);
    printf("Maximum Profit: %d\n", result);

    int prices2[] = {7, 6, 4, 3, 1};
    int pricesSize2 = sizeof(prices2) / sizeof(prices2[0]);
    k = 2;
    result = maxProfit(k, prices2, pricesSize2);
    printf("Maximum Profit: %d\n", result);

    int prices3[] = {1, 2, 4, 2, 5, 7, 2, 4, 9, 0};
    int pricesSize3 = sizeof(prices3) / sizeof(prices3[0]);
    k = 2;
    result = maxProfit(k, prices3, pricesSize3);
    printf("Maximum Profit: %d\n", result);

    return 0;
}
