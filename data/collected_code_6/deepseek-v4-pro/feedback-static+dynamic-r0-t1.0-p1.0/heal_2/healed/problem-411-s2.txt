#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *snake_to_camel(const char *snake_str) {
    if (snake_str == NULL) {
        return NULL;
    }

    size_t len = strnlen(snake_str, 4096);
    if (len == 4096) {
        return NULL;
    }

    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t src_pos = 0;
    size_t dst_pos = 0;
    int capitalize_next = 0;

    while (src_pos < len) {
        if (snake_str[src_pos] == '_') {
            capitalize_next = 1;
            src_pos++;
        } else {
            if (capitalize_next && snake_str[src_pos] >= 'a' && snake_str[src_pos] <= 'z') {
                result[dst_pos++] = snake_str[src_pos] - 'a' + 'A';
            } else {
                result[dst_pos++] = snake_str[src_pos];
            }
            capitalize_next = 0;
            src_pos++;
        }
    }

    result[dst_pos] = '\0';
    return result;
}

int main(int argc, const char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <snake_case_string>\n", argv[0]);
        return 1;
    }

    char *camel_str = snake_to_camel(argv[1]);
    if (camel_str == NULL) {
        fprintf(stderr, "Error converting string\n");
        return 1;
    }

    printf("%s\n", camel_str);
    free(camel_str);
    return 0;
}
