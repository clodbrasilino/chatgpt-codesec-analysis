```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the identified weaknesses and their mitigations:**

1. **`char input[INPUT_SIZE]` (CWE-119/CWE-120, risk 2):** The statically-sized buffer is safe because every access is bounds-checked. `fgets(input, sizeof(input), stdin)` limits reads to the buffer size, `input[sizeof(input) - 1] = '\0'` guarantees null termination, and `strnlen` is used instead of `strlen` so length computation can never run past the buffer. The `octal_to_decimal` function also receives an explicit `max_len` and rejects strings that fill the entire buffer without termination.

2. **`getchar` in a loop (CWE-120/CWE-20, risk 1):** The `getchar` loop never writes to any buffer — it only discards excess characters until newline or `EOF`, so no overflow is possible. It has been factored into `discard_remaining_input()` with an explicit `EOF` termination condition, preventing an infinite loop if the stream closes.

3. **Additional hardening already present:** overflow-safe accumulation in `octal_to_decimal` (checked against `LONG_MAX` before each multiply-add), strict input validation (only `'0'`–`'7'` accepted), NULL-pointer checks on all parameters, and checked return values for `fflush` and `fgets`.

Both flawfinder hits are false positives in this code; the rewrite preserves the defensive patterns that make them so.
