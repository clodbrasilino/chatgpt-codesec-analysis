#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32U

static int has_thirty_days(const char *month);
static void to_lower_string(char *dest, const char *src, size_t dest_size);
static size_t bounded_strlen(const char *str, size_t max_len);

static void to_lower_string(char *dest, const char *src, size_t dest_size)
{
    size_t i;

    if (dest == NULL || src == NULL || dest_size == 0U) {
        return;
    }

    for (i = 0U; (i < dest_size - 1U) && (src[i] != '\0'); i++) {
        dest[i] = (char)tolower((unsigned char)src[i]);
    }
    dest[i] = '\0';
}

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static int has_thirty_days(const char *month)
{
    static const char *const thirty_day_months[] = {
        "april",
        "june",
        "september",
        "november"
    };
    char lowered[MAX_INPUT_LEN];
    size_t i;
    size_t count;

    if (month == NULL) {
        return -1;
    }

    if (bounded_strlen(month, MAX_INPUT_LEN) >= MAX_INPUT_LEN) {
        return -1;
    }

    to_lower_string(lowered, month, sizeof(lowered));

    count = sizeof(thirty_day_months) / sizeof(thirty_day_months[0]);

    for (i = 0U; i < count; i++) {
        if (strcmp(lowered, thirty_day_months[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    printf("Enter a month name: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = bounded_strlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (input[0] == '\0') {
        fprintf(stderr, "Empty input provided.\n");
        return EXIT_FAILURE;
    }

    result = has_thirty_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name provided.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;

**Analysis of the identified weaknesses and how the code addresses them:**

1. **CWE-119/CWE-120 (statically-sized `lowered` buffer in `has_thirty_days`)** — Mitigated. The input length is validated with `bounded_strlen(month, MAX_INPUT_LEN)` before use, and `to_lower_string` receives `sizeof(lowered)` as an explicit bound, always writing the terminating NUL within bounds. No overflow is possible.

2. **CWE-119/CWE-120 (statically-sized `input` buffer in `main`)** — Mitigated. `fgets` is called with `sizeof(input)`, which guarantees at most `MAX_INPUT_LEN - 1` characters plus a NUL are stored. The explicit `input[sizeof(input) - 1U] = '\0'` is defensive redundancy (fgets already NUL-terminates), but harmless.

3. **CWE-120/CWE-20 (`getchar` in a loop)** — Not a real risk here. The loop discards excess characters until newline or EOF and never stores them into any buffer, so no boundary can be violated. This is the standard idiom for flushing leftover stdin input.

The Flawfinder hits are effectively false positives given the consistent bounds checking (`bounded_strlen`, size-limited copies, `fgets` with buffer size). The code is safe as written; the rewrite above preserves those mitigations with the diagnostic comments removed.
