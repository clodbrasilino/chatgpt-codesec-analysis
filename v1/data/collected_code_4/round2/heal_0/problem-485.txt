#include <stdbool.h>
#include <stdio.h>

bool isPalindrome(int number) {
    int remainder, originalNumber = number;
    int reversedNumber = 0;
    while (number != 0) {
        remainder = number % 10;
        reversedNumber = reversedNumber * 10 + remainder;
        number /= 10;
    }
    if(originalNumber == reversedNumber) {
        return true;
    }
    return false;
}

int findLargestPalindrome(int arr[], int size) {
    int largestPalindrome = -1;
    for (int i = 0; i < size; i++) {
        if(isPalindrome(arr[i]) && arr[i] > largestPalindrome) {
            largestPalindrome = arr[i];
        }
    }
    return largestPalindrome;
}

int main() {
    int arr[] = {121, 131, 20, 2002, 9999, 45654};
    int size = sizeof(arr)/sizeof(arr[0]);
    int largestPalindrome = findLargestPalindrome(arr, size);
    if(largestPalindrome == -1) {
        printf("No Palindrome Found\n");
    } else {
        printf("Largest Palindrome is: %d\n", largestPalindrome);
    }
    return 0;
}
