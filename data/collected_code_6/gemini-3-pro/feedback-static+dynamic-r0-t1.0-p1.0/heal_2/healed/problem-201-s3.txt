#include <stdio.h>
#include <stdbool.h>

bool are_elements_same(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return false;
    }
    
    if (size == 1) {
        return true;
    }

    int first_element = arr[0];

    for (size_t i = 1; i < size; i++) {
        if (arr[i] != first_element) {
            return false;
        }
    }

    return true;
}

void check_and_print(const char *list_name, const int *arr, size_t size) {
    if (are_elements_same(arr, size)) {
        printf("%s: All elements are the same.\n", list_name);
    } else {
        printf("%s: Elements are not all the same or list is invalid.\n", list_name);
    }
}

int main(void) {
    int list1[] = {5, 5, 5, 5, 5};
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    check_and_print("list1", list1, size1);

    int list2[] = {5, 5, 4, 5, 5};
    size_t size2 = sizeof(list2) / sizeof(list2[0]);
    check_and_print("list2", list2, size2);

    const int *list3 = NULL;
    size_t size3 = 0;
    check_and_print("Empty/NULL list", list3, size3);

    return 0;
}
