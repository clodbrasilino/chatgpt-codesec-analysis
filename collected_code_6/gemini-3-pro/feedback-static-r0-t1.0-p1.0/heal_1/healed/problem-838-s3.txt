#include <stdio.h>
#include <stdlib.h>

int min_swaps(const char *s1, const char *s2) {
    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    int count_01 = 0;
    int count_10 = 0;
    int i = 0;

    for (i = 0; s1[i] != '\0' && s2[i] != '\0'; i++) {
        if (s1[i] != '0' && s1[i] != '1') return -1;
        if (s2[i] != '0' && s2[i] != '1') return -1;

        if (s1[i] == '0' && s2[i] == '1') {
            count_01++;
        } else if (s1[i] == '1' && s2[i] == '0') {
            count_10++;
        }
    }

    if (s1[i] != '\0' || s2[i] != '\0') {
        return -1;
    }

    if ((count_01 + count_10) % 2 != 0) {
        return -1;
    }

    int swaps = (count_01 / 2) + (count_10 / 2);

    if (count_01 % 2 != 0) {
        swaps += 2;
    }

    return swaps;
}

int main(void) {
    const char *s1 = "0101";
    const char *s2 = "1010";

    int result = min_swaps(s1, s2);
    
    if (result == -1) {
        printf("Invalid input or impossible to make strings equal.\n");
    } else {
        printf("Minimum swaps: %d\n", result);
    }

    return 0;
}
