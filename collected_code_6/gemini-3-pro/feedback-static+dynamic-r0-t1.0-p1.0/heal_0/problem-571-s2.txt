#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int maxPairSum(int arr[], int n, int k) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    qsort(arr, n, sizeof(int), compare);

    int sum = 0;
    int i = n - 1;

    while (i > 0) {
        if (arr[i] - arr[i - 1] < k) {
            sum += arr[i] + arr[i - 1];
            i -= 2;
        } else {
            i--;
        }
    }

    return sum;
}

int main(void) {
    int arr[] = {3, 5, 10, 15, 17, 12, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 4;

    int result = maxPairSum(arr, n, k);
    printf("%d\n", result);

    return 0;
}
