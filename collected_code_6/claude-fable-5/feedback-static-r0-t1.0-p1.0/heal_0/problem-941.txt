```c
#include <stdio.h>
#include <stdlib.h>

typedef enum {
    ELEMENT_INT,
    ELEMENT_TUPLE
} ElementType;

typedef struct {
    int first;
    int second;
} Tuple;

typedef struct {
    ElementType type;
    union {
        int value;
        Tuple tuple;
    } data;
} Element;

typedef struct {
    Element *items;
    size_t size;
} List;

size_t count_until_tuple(const List *list)
{
    size_t count = 0;
    size_t i;

    if (list == NULL || list->items == NULL) {
        return 0;
    }

    for (i = 0; i < list->size; i++) {
        if (list->items[i].type == ELEMENT_TUPLE) {
            break;
        }
        count++;
    }

    return count;
}

int main(void)
{
    List list;
    Element *elements;
    size_t result;
    size_t n = 5;

    elements = (Element *)malloc(n * sizeof(Element));
    if (elements == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    elements[0].type = ELEMENT_INT;
    elements[0].data.value = 10;

    elements[1].type = ELEMENT_INT;
    elements[1].data.value = 20;

    elements[2].type = ELEMENT_INT;
    elements[2].data.value = 30;

    elements[3].type = ELEMENT_TUPLE;
    elements[3].data.tuple.first = 1;
    elements[3].data.tuple.second = 2;

    elements[4].type = ELEMENT_INT;
    elements[4].data.value = 40;

    list.items = elements;
    list.size = n;

    result = count_until_tuple(&list);

    if (printf("Count until tuple: %zu\n", result) < 0) {
        free(elements);
        return EXIT_FAILURE;
    }

    free(elements);
    list.items = NULL;
    list.size = 0;

    return EXIT_SUCCESS;
}
```
