```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size == 0 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = strnlen(buffer, size);
    if (len > 0 && buffer[len - 1] != '\n' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    return 0;
}

int main(void)
{
    char buffer[64];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this version is safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The flagged loop never writes the character into any buffer — it only discards excess input until a newline or `EOF`. There is no buffer boundary to violate, so the Flawfinder hit is a false positive. I extracted it into `discard_remaining_input()` for clarity, keeping the `EOF` check to prevent an infinite loop on closed stdin.

2. **Statically-sized `char buffer[64]` (CWE-119/CWE-120):** All accesses are properly bounded:
   - `read_line` receives `sizeof(buffer)` and passes it to `fgets`, which limits writes to `size - 1` characters plus the terminator.
   - The size is validated (`size == 0`, `size > INT_MAX`) before the cast to `int`, avoiding truncation/negative-size issues in `fgets`.
   - `buffer[size - 1] = '\0'` guarantees termination, and `strnlen` (rather than `strlen`) bounds the length scan.
   - Overlong input is detected (buffer full without a trailing newline) and rejected rather than silently truncated.

3. **Other defensive properties preserved:** `strtol` with `errno`/`ERANGE` checks, range validation against `INT_MAX`/`INT_MIN` before narrowing, `INT_MIN` guard in `closest_smaller` to prevent signed underflow (UB), NULL-pointer checks on all function parameters, and checked return values for `printf`.

The original code was already well-hardened; both Flawfinder findings were false positives, so the rewrite is structurally identical with the discard loop factored out.
