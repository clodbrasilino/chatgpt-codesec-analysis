char **remove_words(char **words, size_t count, char **targets, size_t target_count, size_t *new_count)
{

    size_t i;
    size_t j;
    size_t kept;
    int match;

    if (words == NULL || new_count == NULL) {
        return NULL;
    }

    result = malloc(sizeof(char *) * (count > 0 ? count : 1));
    if (result == NULL) {
        return NULL;
    }

    kept = 0;
    for (i = 0; i < count; i++) {

            continue;
        }
        match = 0;

                break;
            }

}

static char *read_line(void)
{
    char *buffer;
    size_t capacity;
    size_t length;
    int c;

    capacity = 64;
    length = 0;
    buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    c = getchar();
    if (c == EOF) {
        free(buffer);
        return NULL;
    }

    while (c != EOF && c != '\n') {
        if (length + 1 >= capacity) {
            char *tmp;
            capacity *= 2;
            tmp = realloc(buffer, capacity);
            if (tmp == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = tmp;
        }
        buffer[length] = (char)c;
        length++;
        c = getchar();
    }

    buffer[length] = '\0';
    return buffer;
}

static char **split_line(char *line, size_t *count_out)
{
    char **tokens;
    size_t capacity;
    size_t count;
    char *p;

    if (line == NULL || count_out == NULL) {
        return NULL;
    }

    capacity = 8;
    count = 0;
    tokens = malloc(sizeof(char *) * capacity);
    if (tokens == NULL) {
        return NULL;
    }

    p = line;
    while (*p != '\0') {
        while (*p != '\0' && is_delim(*p)) {
            p++;
        }
        if (*p == '\0') {
            break;
        }
        if (count >= capacity) {
            char **tmp;
            capacity *= 2;
            tmp = realloc(tokens, sizeof(char *) * capacity);
            if (tmp == NULL) {
                free(tokens);
                return NULL;
            }
            tokens = tmp;
        }
        tokens[count] = p;
        count++;
        while (*p != '\0' && !is_delim(*p)) {
            p++;
        }
        if (*p != '\0') {
            *p = '\0';
            p++;
        }
    }

    *count_out = count;
    return tokens;
}
