#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

char* camel_to_snake(const char *str) {
    if (str == NULL) return NULL;
    size_t len = 0;
    const char *s = str;
    while (*s++) len++;
    size_t result_len = len * 2 + 1;
    char *result = (char *)malloc(result_len * sizeof(char));
    if (!result) {
        return NULL;
    }
    int i, j = 0;
    for (i = 0; str[i]; ++i) {
        if (isupper((unsigned char)str[i])) {
            if (i > 0 && str[i-1] != '_') {
                result[j++] = '_';
            }
            result[j++] = tolower((unsigned char)str[i]);
        } else {
            result[j++] = str[i];
        }
    }
    result[j] = '\0';
    return result;
}

int main() {
    const char input1[] = "PythonLanguage";
    const char input2[] = "ProgrammingLanguage";
    const char input3[] = "PythonProgram";

    char *output1 = camel_to_snake(input1);
    char *output2 = camel_to_snake(input2);
    char *output3 = camel_to_snake(input3);

    if (output1) {
        printf("%s\n", output1);
        free(output1);
    }

    if (output2) {
        printf("%s\n", output2);
        free(output2);
    }

    if (output3) {
        printf("%s\n", output3);
        free(output3);
    }

    return 0;
}
