```c
#include <stdio.h>
#include <stdlib.h>

static long long find_max_sum(long long n);

static long long find_max_sum(long long n)
{
    long long *dp;
    long long i;
    long long result;

    if (n <= 1) {
        return n;
    }

    dp = malloc(((size_t)n + 1U) * sizeof *dp);
    if (dp == NULL) {
        return -1;
    }

    dp[0] = 0;
    dp[1] = 1;

    for (i = 2; i <= n; i++) {
        long long candidate = dp[i / 2] + dp[i / 3] + dp[i / 4] + dp[i / 5];
        dp[i] = (candidate > i) ? candidate : i;
    }

    result = dp[n];
    free(dp);
    return result;
}

int main(void)
{
    long long n;
    long long result;

    printf("Enter a non-negative integer: ");
    if (scanf("%lld", &n) != 1) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if ((n < 0) || (n > 100000000LL)) {
        fprintf(stderr, "Error: input out of supported range [0, 100000000]\n");
        return EXIT_FAILURE;
    }

    result = find_max_sum(n);
    if (result < 0) {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Maximum sum: %lld\n", result);
    return EXIT_SUCCESS;
}
```

The key change addresses the `[variableScope]` weakness: `candidate` is now declared inside the `for` loop body, where it is actually used. This limits its lifetime to a single iteration, improves readability, and prevents accidental use of a stale value outside the loop. All other logic — the overflow-safe `(size_t)n + 1U` allocation size, the `NULL` check with `-1` sentinel, the input validation bounds, and the `free` before return — was already correct and is preserved as-is.
