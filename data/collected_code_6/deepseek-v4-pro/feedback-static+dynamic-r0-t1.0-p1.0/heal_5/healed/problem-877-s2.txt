#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int compare_chars(const void *a, const void *b) {
    return (*(const unsigned char *)a - *(const unsigned char *)b);
}

void sort_string(char *str, size_t len) {
    if (str == NULL || len == 0) {
        return;
    }
    qsort(str, len, sizeof(char), compare_chars);
}

int main(void) {
    char *str = NULL;
    size_t size = 1024;
    size_t len;
    
    str = (char *)malloc(size);
    if (str == NULL) {
        return 1;
    }
    
    if (fgets(str, (int)size, stdin) == NULL) {
        free(str);
        return 1;
    }
    
    len = strcspn(str, "\n");
    if (len < size) {
        str[len] = '\0';
    } else {
        str[size - 1] = '\0';
        len = size - 1;
    }
    
    if (len > 0) {
        sort_string(str, len);
    }
    
    printf("%s\n", str);
    
    free(str);
    return 0;
}
