```c
#include <stdio.h>
#include <stdlib.h>

double negative_ratio(const int *arr, size_t size)
{
    size_t i;
    size_t negative_count = 0;

    if (arr == NULL || size == 0) {
        return 0.0;
    }

    for (i = 0; i < size; i++) {
        if (arr[i] < 0) {
            negative_count++;
        }
    }

    return (double)negative_count / (double)size;
}

int main(void)
{
    int numbers[] = { -3, 5, -7, 2, 0, -1, 8, -4 };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    double ratio;

    ratio = negative_ratio(numbers, count);

    if (printf("Ratio of negative numbers: %f\n", ratio) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
