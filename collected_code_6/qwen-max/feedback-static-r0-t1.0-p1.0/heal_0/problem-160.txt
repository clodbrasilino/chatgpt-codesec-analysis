#include <stdio.h>
#include <stdlib.h>

int extended_gcd(int a, int b, int *x, int *y) {
    if (a == 0) {
        *x = 0;
        *y = 1;
        return b;
    }
    int x1, y1;
    int gcd = extended_gcd(b % a, a, &x1, &y1);
    *x = y1 - (b / a) * x1;
    *y = x1;
    return gcd;
}

void find_solution(int a, int b, int n, int *x, int *y) {
    int x0, y0;
    int gcd = extended_gcd(a, b, &x0, &y0);
    if (n % gcd != 0) {
        *x = 0;
        *y = 0;
        return;
    }
    int k = n / gcd;
    *x = x0 * k;
    *y = y0 * k;
}

int main() {
    int a = 35, b = 15, n = 30;
    int x, y;
    find_solution(a, b, n, &x, &y);
    printf("x = %d, y = %d\n", x, y);
    return 0;
}
