#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 10
#define MAX_OUTPUT_LEN (2 * MAX_INPUT_LEN)

static int findDemlo(const char *str, char *result, size_t result_size)
{
    size_t len;
    size_t i;
    size_t pos;

    if (str == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);

    if (len == 0 || len > 9) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (str[i] != '1') {
            return -1;
        }
    }

    if (result_size < (2 * len)) {
        return -1;
    }

    pos = 0;
    for (i = 1; i <= len; i++) {
        if (pos >= result_size - 1) {
            return -1;
        }
        result[pos] = (char)('0' + (int)i);
        pos++;
    }
    for (i = len; i > 1; i--) {
        if (pos >= result_size - 1) {
            return -1;
        }
        result[pos] = (char)('0' + (int)(i - 1));
        pos++;
    }
    result[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN + 2];
    char output[MAX_OUTPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(output, 0, sizeof(output));

    printf("Enter a repunit number (only 1s, max 9 digits): ");
    fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }

    if (len == 0) {
        fprintf(stderr, "Invalid input: empty string\n");
        return EXIT_FAILURE;
    }

    if (len > 9) {
        fprintf(stderr, "Invalid input: must be a repunit of at most 9 ones\n");
        return EXIT_FAILURE;
    }

    if (findDemlo(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid input: must be a repunit of at most 9 ones\n");
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", output);

    return EXIT_SUCCESS;
}

**Analysis of the identified weaknesses and their mitigations:**

The Flawfinder warnings (CWE-119/CWE-120) flag the statically-sized `input` and `output` buffers as *potential* overflow risks. Reviewing the code, these are already properly mitigated, so the rewritten code retains the same structure:

1. **`input` buffer** — sized `MAX_INPUT_LEN + 2` to accommodate up to 10 characters plus newline and null terminator. Reads use `fgets` with `sizeof(input)`, which cannot overflow. A defensive `input[sizeof(input) - 1] = '\0'` guarantees termination, and `strnlen` (rather than `strlen`) bounds all length calculations.

2. **`output` buffer** — sized `MAX_OUTPUT_LEN` (20 bytes), sufficient for the worst-case Demlo number `12345678987654321` (17 chars + null). `findDemlo` receives the buffer size explicitly, validates `result_size < 2 * len` up front, and additionally performs per-write bounds checks (`pos >= result_size - 1`) before every store, so no write can exceed the buffer even if the size arithmetic were wrong.

3. **Additional hardening already present** — NULL-pointer checks on all pointer parameters, input length capped at 9 and content validated to be all `'1'` characters, zero-initialization of both buffers via `memset`, and explicit null termination of the result.

No functional changes were required; the code as written already implements the bounds checking and length-limited functions that the Flawfinder recommendations call for, so the warnings are false positives in this instance.
