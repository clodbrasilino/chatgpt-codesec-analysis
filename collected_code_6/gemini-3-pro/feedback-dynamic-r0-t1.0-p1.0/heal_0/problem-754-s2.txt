#include <stdio.h>
#include <stdlib.h>

void find_common_elements(const int* list1, size_t size1,
                          const int* list2, size_t size2,
                          const int* list3, size_t size3,
                          int** common_elements, size_t* common_size) {
    if (!list1 || !list2 || !list3 || !common_elements || !common_size) {
        return;
    }

    size_t min_size = size1;
    if (size2 < min_size) min_size = size2;
    if (size3 < min_size) min_size = size3;

    *common_elements = (int*)malloc(min_size * sizeof(int));
    if (!*common_elements) {
        *common_size = 0;
        return;
    }

    *common_size = 0;

    for (size_t i = 0; i < min_size; ++i) {
        if (list1[i] == list2[i] && list2[i] == list3[i]) {
            (*common_elements)[*common_size] = list1[i];
            (*common_size)++;
        }
    }

    if (*common_size == 0) {
        free(*common_elements);
        *common_elements = NULL;
    } else {
        int* temp = (int*)realloc(*common_elements, (*common_size) * sizeof(int));
        if (temp) {
            *common_elements = temp;
        }
    }
}

int main(void) {
    int list1[] = {1, 2, 3, 4, 5, 6};
    int list2[] = {0, 2, 3, 9, 5};
    int list3[] = {1, 2, 3, 8, 5, 10, 11};
    
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    size_t size2 = sizeof(list2) / sizeof(list2[0]);
    size_t size3 = sizeof(list3) / sizeof(list3[0]);

    int* common_elements = NULL;
    size_t common_size = 0;

    find_common_elements(list1, size1, list2, size2, list3, size3, &common_elements, &common_size);

    if (common_elements) {
        for (size_t i = 0; i < common_size; ++i) {
            printf("%d ", common_elements[i]);
        }
        printf("\n");
        free(common_elements);
        common_elements = NULL;
    } else {
        printf("\n");
    }

    return 0;
}
