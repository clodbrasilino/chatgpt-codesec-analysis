```c
#include <stdio.h>

long long sum_of_cubes(int n) {
    long long sum = 0;
    for (int i = 1; i <= n; ++i) {
        sum += (long long)i * i * i;
    }
    return sum;
}

int main() {
    int n;
    printf("Enter a positive integer: ");
    if (scanf("%d", &n) != 1 || n < 1) {
        printf("Invalid input. Please enter a positive integer.\n");
        return 1;
    }
    long long result = sum_of_cubes(n);
    printf("The sum of cubes up to %d is: %lld\n", n, result);
    return 0;
}
```
