```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <ctype.h>

#define MIN_LENGTH 8
#define MAX_LENGTH 64
#define BUFFER_SIZE 128

static void secure_clear(void *ptr, size_t size)
{
    volatile unsigned char *p;

    if (ptr == NULL) {
        return;
    }

    p = (volatile unsigned char *)ptr;

    while (size--) {
        *p++ = 0;
    }
}

static bool flush_stdin(void)
{
    int ch;
    bool discarded = false;

    do {
        ch = getchar();
        if (ch != '\n' && ch != EOF) {
            discarded = true;
        }
    } while (ch != '\n' && ch != EOF);

    return discarded;
}

static bool is_valid_password(const char *password, size_t max_size)
{
    size_t length;
    size_t i;
    bool has_upper = false;
    bool has_lower = false;
    bool has_digit = false;
    bool has_special = false;

    if (password == NULL || max_size == 0) {
        return false;
    }

    length = strnlen(password, max_size);

    if (length == max_size) {
        return false;
    }

    if (length < MIN_LENGTH || length > MAX_LENGTH) {
        return false;
    }

    for (i = 0; i < length; i++) {
        unsigned char c = (unsigned char)password[i];

        if (!isprint(c)) {
            return false;
        }

        if (isupper(c)) {
            has_upper = true;
        } else if (islower(c)) {
            has_lower = true;
        } else if (isdigit(c)) {
            has_digit = true;
        } else if (ispunct(c)) {
            has_special = true;
        } else {
            return false;
        }
    }

    return has_upper && has_lower && has_digit && has_special;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t length;
    bool truncated = false;

    printf("Enter password: ");

    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        secure_clear(buffer, sizeof(buffer));
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    length = strnlen(buffer, sizeof(buffer));

    if (length > 0 && buffer[length - 1] == '\n') {
        buffer[length - 1] = '\0';
    } else {
        truncated = flush_stdin();
    }

    if (!truncated && is_valid_password(buffer, sizeof(buffer))) {
        printf("Password is valid\n");
    } else {
        printf("Password is invalid\n");
    }

    secure_clear(buffer, sizeof(buffer));

    return 0;
}
```

**Analysis of the original code and the fixes applied:**

The two Flawfinder findings were both informational (risk levels 1–2) and largely already mitigated in the original code, but the rewrite tightens the remaining gaps:

1. **`getchar` in a loop (CWE-120, CWE-20):** The original `flush_stdin` safely discarded excess input, but the program silently *ignored* the fact that truncation occurred — a password longer than the buffer would be validated on its truncated prefix, potentially accepting a password the user didn't actually enter. The rewritten `flush_stdin` returns whether extra characters were discarded, and `main` rejects the input outright in that case. The loop itself never writes to a buffer, so no overflow is possible.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer is safe because every access is bounds-checked: `fgets` is given `sizeof(buffer)`, an explicit NUL terminator is forced at `buffer[sizeof(buffer) - 1]`, `strnlen` is used instead of `strlen`, and `is_valid_password` receives the buffer capacity and rejects unterminated data (`length == max_size`). `BUFFER_SIZE` (128) also comfortably exceeds `MAX_LENGTH` (64) plus the newline and terminator.

Additional strengths preserved: `secure_clear` uses a `volatile` pointer to prevent the compiler from optimizing away the password wipe, characters are cast to `unsigned char` before the `ctype` calls (avoiding UB on negative `char` values), and the buffer is cleared on every exit path after input has been read.
