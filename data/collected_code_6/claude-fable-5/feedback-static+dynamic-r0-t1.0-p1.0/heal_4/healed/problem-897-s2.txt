#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define SENTENCE_MAX 256
#define WORD_MAX 64

int is_word_present(const char *sentence, const char *word);

int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    size_t i;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = strnlen(word, WORD_MAX);
    if (word_len == 0U || word_len >= WORD_MAX) {
        return 0;
    }

    i = 0U;
    while (i < SENTENCE_MAX && sentence[i] != '\0') {
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               !isalnum((unsigned char)sentence[i])) {
            i++;
        }
        if (i >= SENTENCE_MAX || sentence[i] == '\0') {
            break;
        }

        size_t start = i;
        while (i < SENTENCE_MAX && sentence[i] != '\0' &&
               isalnum((unsigned char)sentence[i])) {
            i++;
        }
        size_t token_len = i - start;

        if (token_len == word_len &&
            strncmp(&sentence[start], word, word_len) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char sentence[SENTENCE_MAX];
    char word[WORD_MAX];

    memset(sentence, 0, sizeof(sentence));
    memset(word, 0, sizeof(word));

    if (fgets(sentence, (int)sizeof(sentence), stdin) == NULL) {
        fprintf(stderr, "Error reading sentence.\n");
        return EXIT_FAILURE;
    }
    sentence[sizeof(sentence) - 1U] = '\0';
    sentence[strcspn(sentence, "\n")] = '\0';

    if (fgets(word, (int)sizeof(word), stdin) == NULL) {
        fprintf(stderr, "Error reading word.\n");
        return EXIT_FAILURE;
    }
    word[sizeof(word) - 1U] = '\0';
    word[strcspn(word, "\n")] = '\0';

    if (is_word_present(sentence, word) == 1) {
        printf("True\n");
    } else {
        printf("False\n");
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe:

1. **Statically-sized arrays (CWE-119/CWE-120)**: The Flawfinder warnings on `char sentence[SENTENCE_MAX]` and `char word[WORD_MAX]` are informational (risk level 2). The code already mitigates the risk properly:
   - `fgets()` is used with `sizeof(buffer)`, which guarantees no more than `size - 1` bytes are written and the result is always null-terminated.
   - Explicit null termination at `buffer[sizeof(buffer) - 1U]` provides defense-in-depth.
   - `strcspn()` safely strips the trailing newline without risk of out-of-bounds access.

2. **Bounds checking in `is_word_present()`**: Every loop condition checks `i < SENTENCE_MAX` before dereferencing `sentence[i]`, preventing reads past the buffer even if a non-terminated string were passed. `strnlen()` with `WORD_MAX` bounds the word-length computation, and the `word_len >= WORD_MAX` check rejects unterminated or oversized input.

3. **`isalnum()` safety**: Characters are cast to `unsigned char` before being passed to `isalnum()`, avoiding undefined behavior for negative `char` values on platforms where `char` is signed.

4. **NULL checks**: Both pointer parameters are validated, and all `fgets()` return values are checked before use.
