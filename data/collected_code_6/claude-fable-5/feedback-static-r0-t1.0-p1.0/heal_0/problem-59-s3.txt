```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long octagonal_number(int n)
{
    if (n < 1) {
        return -1;
    }
    if (n > 1000000000) {
        return -1;
    }
    return (long long)n * (3LL * (long long)n - 2LL);
}

int main(void)
{
    int n = 0;
    long long result = 0;

    printf("Enter n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    result = octagonal_number(n);
    if (result < 0) {
        fprintf(stderr, "n must be a positive integer within range\n");
        return EXIT_FAILURE;
    }

    printf("The %d-th octagonal number is %lld\n", n, result);
    return EXIT_SUCCESS;
}
```
