#include <stdio.h>

typedef struct {
    int integer_part;
    int fractional_part;
} Tuple;

double tuple_to_float(Tuple t) {
    double int_part = (double)t.integer_part;
    double frac_part = (double)t.fractional_part;
    
    int sign = 1;
    if (int_part < 0.0) {
        sign = -1;
    } else if (frac_part < 0.0) {
        sign = -1;
    }
    
    if (int_part < 0.0) {
        int_part = -int_part;
    }
    if (frac_part < 0.0) {
        frac_part = -frac_part;
    }
    
    while (frac_part >= 1.0) {
        frac_part /= 10.0;
    }
    
    return sign * (int_part + frac_part);
}

int main(void) {
    Tuple t1 = {3, 14};
    Tuple t2 = {-2, 5};
    Tuple t3 = {0, -7};
    Tuple t4 = {0, 0};
    
    printf("%f\n", tuple_to_float(t1));
    printf("%f\n", tuple_to_float(t2));
    printf("%f\n", tuple_to_float(t3));
    printf("%f\n", tuple_to_float(t4));
    
    return 0;
}
