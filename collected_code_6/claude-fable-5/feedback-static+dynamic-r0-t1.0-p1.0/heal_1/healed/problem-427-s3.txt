#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_BUFFER_SIZE 11

int convert_date(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t matches[4];
    int ret;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_BUFFER_SIZE) {
        return -1;
    }

    ret = regcomp(&regex, "^([0-9]{4})-([0-9]{2})-([0-9]{2})$", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    ret = regexec(&regex, input, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    written = snprintf(output, output_size, "%.*s-%.*s-%.*s",
                       (int)(matches[3].rm_eo - matches[3].rm_so),
                       input + matches[3].rm_so,
                       (int)(matches[2].rm_eo - matches[2].rm_so),
                       input + matches[2].rm_so,
                       (int)(matches[1].rm_eo - matches[1].rm_so),
                       input + matches[1].rm_so);

    regfree(&regex);

    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *input = "2024-05-17";
    char output[DATE_BUFFER_SIZE];

    memset(output, 0, sizeof(output));

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Failed to convert date: %s\n", input);
        return EXIT_FAILURE;
    }

    if (printf("Input:  %s\nOutput: %s\n", input, output) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

1. **Bounds are enforced at the API boundary**: `convert_date()` rejects any call where `output_size < DATE_BUFFER_SIZE`, so the buffer can never be smaller than what the function requires.

2. **`snprintf` is used with the caller-supplied size**: writes are limited to `output_size`, guaranteeing no overflow and always NUL-terminating the output.

3. **Truncation is detected**: the check `written < 0 || (size_t)written >= output_size` catches both encoding errors and truncated output, returning an error rather than silently producing a partial date.

5. **Regex match lengths are safe**: the pattern anchors (`^...$`) and fixed repetition counts (`{4}`, `{2}`, `{2}`) guarantee the total formatted output is exactly 10 characters, so it fits.
