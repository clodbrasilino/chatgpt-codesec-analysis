#include <stdio.h>
#include <stdlib.h>

unsigned long long merge(int arr[], int temp[], int left, int mid, int right) {
    unsigned long long inv_count = 0;
    int i = left;
    int j = mid;
    int k = left;

    while ((i <= mid - 1) && (j <= right)) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            temp[k++] = arr[j++];
            inv_count = inv_count + (mid - i);
        }
    }

    while (i <= mid - 1) {
        temp[k++] = arr[i++];
    }

    while (j <= right) {
        temp[k++] = arr[j++];
    }

    for (i = left; i <= right; i++) {
        arr[i] = temp[i];
    }

    return inv_count;
}

unsigned long long mergeSort(int arr[], int temp[], int left, int right) {
    unsigned long long inv_count = 0;
    if (right > left) {
        int mid = (right + left) / 2;

        inv_count = mergeSort(arr, temp, left, mid);
        inv_count += mergeSort(arr, temp, mid + 1, right);
        inv_count += merge(arr, temp, left, mid + 1, right);
    }
    return inv_count;
}

unsigned long long countInversions(int arr[], int n) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    int *temp = (int *)malloc(sizeof(int) * n);
    if (temp == NULL) {
        return 0;
    }

    unsigned long long result = mergeSort(arr, temp, 0, n - 1);
    free(temp);
    return result;
}

int main(void) {
    int arr[] = {1, 20, 6, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);

    unsigned long long inversions = countInversions(arr, n);
    printf("Number of inversions are %llu\n", inversions);

    return 0;
}
