```c
#include <stdio.h>
#include <stdlib.h>

typedef struct List {
    int *data;
    size_t size;
} List;

size_t count_lists(List **lists, size_t num_lists) {
    if (lists == NULL) return 0;
    size_t count = 0;
    for (size_t i = 0; i < num_lists; ++i) {
        if (lists[i] != NULL && lists[i]->data != NULL) {
            ++count;
        }
    }
    return count;
}

int main() {
    List *list1 = (List *)malloc(sizeof(List));
    list1->data = (int *)malloc(5 * sizeof(int));
    list1->size = 5;

    List *list2 = (List *)malloc(sizeof(List));
    list2->data = (int *)malloc(3 * sizeof(int));
    list2->size = 3;

    List *list3 = (List *)malloc(sizeof(List));
    list3->data = NULL;
    list3->size = 0;

    List *lists[] = {list1, list2, list3};
    size_t num_lists = 3;

    size_t result = count_lists(lists, num_lists);
    printf("Number of valid lists: %zu\n", result);

    free(list1->data);
    free(list1);
    free(list2->data);
    free(list2);
    free(list3);

    return 0;
}
```
