#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>

typedef struct Node {
    void *data;
    int type;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    size_t size;
} List;

enum {
    TYPE_INT,
    TYPE_FLOAT,
    TYPE_STRING,
    TYPE_DOUBLE
};

List *list_create(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->size = 0;
    return list;
}

void list_destroy(List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        Node *next = current->next;
        free(current->data);
        free(current);
        current = next;
    }
    free(list);
}

int list_append(List *list, void *data, int type) {
    if (list == NULL || data == NULL) {
        return -1;
    }
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return -1;
    }
    new_node->data = data;
    new_node->type = type;
    new_node->next = NULL;
    
    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }
    list->size++;
    return 0;
}

int is_valid_float_string(const char *str) {
    if (str == NULL || *str == '\0') {
        return 0;
    }
    
    char *endptr = NULL;
    errno = 0;
    strtod(str, &endptr);
    
    if (errno == ERANGE) {
        return 0;
    }
    if (endptr == str) {
        return 0;
    }
    while (*endptr != '\0') {
        if (!isspace((unsigned char)*endptr)) {
            return 0;
        }
        endptr++;
    }
    return 1;
}

void convert_to_float(List *list) {
    if (list == NULL) {
        return;
    }
    
    Node *current = list->head;
    while (current != NULL) {
        if (current->type == TYPE_INT) {
            int *int_val = (int *)current->data;
            float *float_val = (float *)malloc(sizeof(float));
            if (float_val == NULL) {
                return;
            }
            *float_val = (float)(*int_val);
            free(current->data);
            current->data = float_val;
            current->type = TYPE_FLOAT;
        } else if (current->type == TYPE_DOUBLE) {
            double *double_val = (double *)current->data;
            float *float_val = (float *)malloc(sizeof(float));
            if (float_val == NULL) {
                return;
            }
            *float_val = (float)(*double_val);
            free(current->data);
            current->data = float_val;
            current->type = TYPE_FLOAT;
        } else if (current->type == TYPE_STRING) {
            char *str_val = (char *)current->data;
            if (is_valid_float_string(str_val)) {
                float *float_val = (float *)malloc(sizeof(float));
                if (float_val == NULL) {
                    return;
                }
                *float_val = strtof(str_val, NULL);
                free(current->data);
                current->data = float_val;
                current->type = TYPE_FLOAT;
            }
        }
        current = current->next;
    }
}

void print_list(const List *list) {
    if (list == NULL) {
        return;
    }
    Node *current = list->head;
    while (current != NULL) {
        if (current->type == TYPE_INT) {
            printf("%d ", *(int *)current->data);
        } else if (current->type == TYPE_FLOAT) {
            printf("%.2f ", *(float *)current->data);
        } else if (current->type == TYPE_DOUBLE) {
            printf("%.2f ", *(double *)current->data);
        } else if (current->type == TYPE_STRING) {
            printf("%s ", (char *)current->data);
        }
        current = current->next;
    }
    printf("\n");
}

int main(void) {
    List *list = list_create();
    if (list == NULL) {
        return EXIT_FAILURE;
    }
    
    int *int_val = (int *)malloc(sizeof(int));
    if (int_val == NULL) {
        list_destroy(list);
        return EXIT_FAILURE;
    }
    *int_val = 42;
    if (list_append(list, int_val, TYPE_INT) != 0) {
        free(int_val);
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    float *float_val = (float *)malloc(sizeof(float));
    if (float_val == NULL) {
        list_destroy(list);
        return EXIT_FAILURE;
    }
    *float_val = 3.14f;
    if (list_append(list, float_val, TYPE_FLOAT) != 0) {
        free(float_val);
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    double *double_val = (double *)malloc(sizeof(double));
    if (double_val == NULL) {
        list_destroy(list);
        return EXIT_FAILURE;
    }
    *double_val = 2.71828;
    if (list_append(list, double_val, TYPE_DOUBLE) != 0) {
        free(double_val);
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    char *str_val1 = (char *)malloc(16 * sizeof(char));
    if (str_val1 == NULL) {
        list_destroy(list);
        return EXIT_FAILURE;
    }
    strcpy(str_val1, "123.45");
    if (list_append(list, str_val1, TYPE_STRING) != 0) {
        free(str_val1);
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    char *str_val2 = (char *)malloc(16 * sizeof(char));
    if (str_val2 == NULL) {
        list_destroy(list);
        return EXIT_FAILURE;
    }
    strcpy(str_val2, "not_a_number");
    if (list_append(list, str_val2, TYPE_STRING) != 0) {
        free(str_val2);
        list_destroy(list);
        return EXIT_FAILURE;
    }
    
    printf("Before conversion: ");
    print_list(list);
    
    convert_to_float(list);
    
    printf("After conversion: ");
    print_list(list);
    
    list_destroy(list);
    return EXIT_SUCCESS;
}
