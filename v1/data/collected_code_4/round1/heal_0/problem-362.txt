```c
#include <stdio.h>

#define SIZE 10000

int mostFrequent(int arr[], int n) {
    int max_count = 0;
    int max_item = -1;
    int count[SIZE] = {0};

    for(int i = 0; i<n; i++) {
        count[arr[i]]++;
        if(max_count < count[arr[i]]) {
            max_count = count[arr[i]];
            max_item = arr[i];
        }
    }

    return max_item;
}

int main() {
    int arr[] = {1, 3, 2, 1, 4, 1};
    int n = sizeof(arr)/sizeof(arr[0]);
    int result;

    result = mostFrequent(arr, n);
    printf("%d is the number which has maximum occurrences in the given array.", result);

    return 0;
}
```
