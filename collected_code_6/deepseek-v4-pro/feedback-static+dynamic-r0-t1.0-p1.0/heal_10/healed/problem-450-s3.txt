#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **extract_strings(char **strings, size_t count, size_t size, size_t *out_count) {
    if (strings == NULL || out_count == NULL || count == 0 || size == 0) {
        if (out_count != NULL) {
            *out_count = 0;
        }
        return NULL;
    }

    size_t valid_count = 0;
    for (size_t i = 0; i < count; i++) {
        if (strings[i] != NULL && strlen(strings[i]) >= size) {
            valid_count++;
        }
    }

    if (valid_count == 0) {
        *out_count = 0;
        return NULL;
    }

    char **result = (char **)malloc(valid_count * sizeof(char *));
    if (result == NULL) {
        *out_count = 0;
        return NULL;
    }

    size_t result_index = 0;
    for (size_t i = 0; i < count && result_index < valid_count; i++) {
        if (strings[i] == NULL) {
            continue;
        }

        size_t len = strlen(strings[i]);
        if (len < size) {
            continue;
        }

        result[result_index] = (char *)malloc((size + 1) * sizeof(char));
        if (result[result_index] == NULL) {
            for (size_t j = 0; j < result_index; j++) {
                free(result[j]);
            }
            free(result);
            *out_count = 0;
            return NULL;
        }

        memcpy(result[result_index], strings[i], size);
        result[result_index][size] = '\0';
        result_index++;
    }

    *out_count = result_index;
    return result;
}

int main(void) {
    char *strings[] = {
        "practice",
        "solution",
        "exercises",
        "Python"
    };
    size_t count = sizeof(strings) / sizeof(strings[0]);
    size_t extract_size = 4;
    size_t result_count = 0;

    char **extracted = extract_strings(strings, count, extract_size, &result_count);

    if (extracted != NULL) {
        for (size_t i = 0; i < result_count; i++) {
            printf("%s\n", extracted[i]);
            free(extracted[i]);
        }
        free(extracted);
    } else {
        printf("No strings extracted or error occurred.\n");
    }

    return 0;
}
