#include <stdio.h>
#include <math.h>

double ellipse_area(double a, double b) {
    return M_PI * a * b;
}

double triangle_area(double s1, double s2, double s3) {
    double s = (s1 + s2 + s3) / 2.0;
    return sqrt(s * (s - s1) * (s - s2) * (s - s3));
}

double largest_inscribed_triangle_area(double a, double b) {
    double e = sqrt(1.0 - (b * b) / (a * a));
    double angle = asin(e);
    double side1 = 2.0 * a * sin(angle);
    double side2 = 2.0 * b * cos(angle);
    return (side1 * side2) / 2.0;
}

int main() {
    double a = 5.0;
    double b = 3.0;
    double area = largest_inscribed_triangle_area(a, b);
    printf("The area of the largest inscribed triangle is: %f\n", area);
    return 0;
}
