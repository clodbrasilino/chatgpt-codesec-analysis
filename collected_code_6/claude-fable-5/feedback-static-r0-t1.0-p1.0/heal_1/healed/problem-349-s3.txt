```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define MAX_INPUT_LENGTH 256U

static bool is_binary_string(const char *str)
{
    size_t i;

    if (str == NULL) {
        return false;
    }

    if (str[0] == '\0') {
        return false;
    }

    for (i = 0U; str[i] != '\0'; i++) {
        if ((str[i] != '0') && (str[i] != '1')) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    char input[MAX_INPUT_LENGTH];
    size_t len;
    char *newline;

    (void)memset(input, 0, sizeof(input));

    printf("Enter a string: ");
    (void)fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        (void)fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    newline = memchr(input, '\n', sizeof(input));
    if (newline != NULL) {
        *newline = '\0';
    }

    len = strnlen(input, sizeof(input));
    if (len >= (sizeof(input) - 1U)) {
        int ch;
        do {
            ch = getchar();
        } while ((ch != '\n') && (ch != EOF));
    }

    if (is_binary_string(input)) {
        printf("The string is a binary string.\n");
    } else {
        printf("The string is not a binary string.\n");
    }

    return EXIT_SUCCESS;
}
```

The key hardening changes made to address the identified weaknesses:

1. **CWE-119/CWE-120 (buffer sizing)**: The buffer is zero-initialized with `memset` before use, `fgets` remains bounded by `sizeof(input)`, and a sentinel null terminator is explicitly forced at the last index so the buffer can never be non-terminated regardless of input behavior.

2. **CWE-126 (over-read via `strlen`)**: Replaced `strlen` with the bounded `strnlen`, and the newline stripping now uses `memchr` bounded by the buffer size instead of indexing based on an unbounded string length.

3. **Additional robustness**: If the input filled the buffer (meaning the user typed more than fits), the remainder of the stdin line is drained so leftover characters don't pollute subsequent reads, and `stdout` is flushed after the prompt to guarantee it appears before blocking on input.
