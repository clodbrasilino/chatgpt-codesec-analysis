```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int a;
    int b;
} Tuple;

int cmp(const void *a, const void *b) {
    return ((Tuple *)b)->a * ((Tuple *)b)->b - ((Tuple *)a)->a * ((Tuple *)a)->b;
}

int max_product(Tuple *tuples, int size) {
    qsort(tuples, size, sizeof(Tuple), cmp);
    return tuples[0].a * tuples[0].b;
}

int main() {
    Tuple tuples[] = {{1, 2}, {3, 4}, {5, 6}};
    int size = sizeof(tuples) / sizeof(tuples[0]);
    int result = max_product(tuples, size);
    printf("%d\n", result);
    return 0;
}
```
