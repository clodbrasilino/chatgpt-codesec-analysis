#include <stdio.h>
#include <stdlib.h>

long long sum_of_odds(long long l, long long r) {
    if (l > r) {
        return 0;
    }
    if (l < 0) {
        l = 1;
    }
    if (l % 2 == 0) {
        l++;
    }
    if (r % 2 == 0) {
        r--;
    }
    if (l > r) {
        return 0;
    }
    long long n = (r - l) / 2 + 1;
    long long sum = n * (l + r) / 2;
    return sum;
}

int main() {
    long long l, r;
    if (scanf("%lld %lld", &l, &r) != 2) {
        return EXIT_FAILURE;
    }
    long long result = sum_of_odds(l, r);
    printf("%lld\n", result);
    return EXIT_SUCCESS;
}
