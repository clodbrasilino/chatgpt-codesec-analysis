```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256
#define INPUT_SIZE 1024

static int count_matches(const char *pattern, const char *str, int *count)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;
    char errbuf[ERRBUF_SIZE];

    if (pattern == NULL || str == NULL || count == NULL) {
        return -1;
    }

    *count = 0;

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return -1;
    }

    cursor = str;
    while (*cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Regex execution failed: %s\n", errbuf);
            regfree(&regex);
            return -1;
        }
        (*count)++;
        if (match.rm_eo <= match.rm_so) {
            cursor += 1;
        } else {
            cursor += match.rm_eo;
        }
    }

    regfree(&regex);
    return 0;
}

static int classify_string(const char *str, int *upper, int *lower,
                           int *digit, int *special)
{
    if (str == NULL || upper == NULL || lower == NULL ||
        digit == NULL || special == NULL) {
        return -1;
    }

    if (count_matches("[A-Z]", str, upper) != 0) {
        return -1;
    }
    if (count_matches("[a-z]", str, lower) != 0) {
        return -1;
    }
    if (count_matches("[0-9]", str, digit) != 0) {
        return -1;
    }
    if (count_matches("[^A-Za-z0-9]", str, special) != 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    int upper = 0;
    int lower = 0;
    int digit = 0;
    int special = 0;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (classify_string(input, &upper, &lower, &digit, &special) != 0) {
        fprintf(stderr, "Classification failed\n");
        return EXIT_FAILURE;
    }

    printf("Uppercase letters: %d\n", upper);
    printf("Lowercase letters: %d\n", lower);
    printf("Numeric digits:    %d\n", digit);
    printf("Special characters: %d\n", special);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the identified weaknesses and the applied fixes:**

1. **`char errbuf[256]` (CWE-119/CWE-120):** The original code was already largely safe because `regerror` receives `sizeof(errbuf)` as the limit. As defense-in-depth, the rewrite explicitly null-terminates the buffer after the call and hoists the buffer to a single declaration with a named constant, eliminating duplication.

2. **`char input[1024]` (CWE-119/CWE-120):** Safe as written since `fgets` bounds the read with `sizeof(input)`. The rewrite adds an explicit terminating null on the last byte as a belt-and-suspenders guarantee.

3. **`strlen` over-read (CWE-126):** `fgets` always null-terminates on success, so the risk was theoretical. Nevertheless, the rewrite replaces the `strlen`/index pattern with `strcspn(input, "\n")`, which both strips the newline and is bounded by the guaranteed terminator — it also correctly handles the edge case of an empty string without a length check.

4. **Additional hardening:** `fflush(stdout)` ensures the prompt is visible before blocking on `fgets`, and helper functions are marked `static` to limit linkage scope. The zero-width match guard (`rm_eo <= rm_so`) from the original is retained, as it correctly prevents infinite loops on patterns that can match empty strings.
