#include <stdio.h>
#include <stdlib.h>

void find_pairs(int *arr, int size, int target_sum) {
    if (arr == NULL || size <= 0) {
        return;
    }

    for (int i = 0; i < size; i++) {
        for (int j = i + 1; j < size; j++) {
            if (arr[i] + arr[j] == target_sum) {
                printf("(%d, %d)\n", arr[i], arr[j]);
            }
        }
    }
}

int main(void) {
    int arr[] = {1, 5, 7, -1, 5};
    int size = sizeof(arr) / sizeof(arr[0]);
    int target_sum = 6;

    find_pairs(arr, size, target_sum);

    return EXIT_SUCCESS;
}
