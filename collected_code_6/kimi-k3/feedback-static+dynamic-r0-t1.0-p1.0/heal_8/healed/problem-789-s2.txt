#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <math.h>
#include <string.h>
#include <limits.h>

static double calculate_perimeter(int sides, double side_length)
{
    return sides * side_length;
}

static int read_line(char *buffer, size_t size)
{
    if (buffer == NULL || size == 0) {
        return 0;
    }
    
    if (fgets(buffer, size, stdin) == NULL) {
        return 0;
    }
    
    buffer[size - 1] = '\0';
    
    size_t len = strnlen(buffer, size);
    if (len == 0) {
        return 0;
    }
    
    if (buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        return 1;
    }
    
    if (len == size - 1 && buffer[len - 1] != '\n') {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) { }
        return -1;
    }
    
    return 1;
}

static int parse_sides(const char *str, long *result)
{
    char *endptr;
    long value;
    
    if (str == NULL || result == NULL) {
        return 0;
    }
    
    if (str[0] == '\0') {
        return 0;
    }
    
    errno = 0;
    value = strtol(str, &endptr, 10);
    
    if (errno == ERANGE) {
        return 0;
    }
    
    if (endptr == str) {
        return 0;
    }
    
    if (*endptr != '\0') {
        return 0;
    }
    
    if (value < 3 || value > 1000) {
        return 0;
    }
    
    *result = value;
    return 1;
}

static int parse_side_length(const char *str, double *result)
{
    char *endptr;
    double value;
    
    if (str == NULL || result == NULL) {
        return 0;
    }
    
    if (str[0] == '\0') {
        return 0;
    }
    
    errno = 0;
    value = strtod(str, &endptr);
    
    if (errno == ERANGE) {
        return 0;
    }
    
    if (endptr == str) {
        return 0;
    }
    
    if (*endptr != '\0') {
        return 0;
    }
    
    if (value <= 0.0) {
        return 0;
    }
    
    if (!isfinite(value)) {
        return 0;
    }
    
    *result = value;
    return 1;
}

int main(void)
{
    char input_buffer[256];
    long sides;
    double side_length;
    double perimeter;
    int status;

    status = read_line(input_buffer, sizeof(input_buffer));
    if (status == 0) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }
    if (status == -1) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (!parse_sides(input_buffer, &sides)) {
        fprintf(stderr, "Invalid number of sides\n");
        return EXIT_FAILURE;
    }

    status = read_line(input_buffer, sizeof(input_buffer));
    if (status == 0) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }
    if (status == -1) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (!parse_side_length(input_buffer, &side_length)) {
        fprintf(stderr, "Invalid side length\n");
        return EXIT_FAILURE;
    }

    if (sides > INT_MAX) {
        fprintf(stderr, "Invalid number of sides\n");
        return EXIT_FAILURE;
    }

    perimeter = calculate_perimeter((int)sides, side_length);
    
    if (!isfinite(perimeter)) {
        fprintf(stderr, "Calculation overflow\n");
        return EXIT_FAILURE;
    }

    printf("%.6f\n", perimeter);
    return EXIT_SUCCESS;
}
