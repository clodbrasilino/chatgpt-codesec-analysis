#include <stdio.h>
#include <stdlib.h>

char ***generate_3d_array(size_t x, size_t y, size_t z);
void free_3d_array(char ***arr, size_t x, size_t y);

char ***generate_3d_array(size_t x, size_t y, size_t z)
{
    char ***arr = NULL;
    size_t i;
    size_t j;
    size_t k;

    if ((x == 0U) || (y == 0U) || (z == 0U)) {
        return NULL;
    }

    arr = malloc(x * sizeof(char **));
    if (arr == NULL) {
        return NULL;
    }

    for (i = 0U; i < x; i++) {
        arr[i] = malloc(y * sizeof(char *));
        if (arr[i] == NULL) {
            free_3d_array(arr, i, y);
            return NULL;
        }
        for (j = 0U; j < y; j++) {
            arr[i][j] = malloc(z * sizeof(char));
            if (arr[i][j] == NULL) {
                size_t m;
                for (m = 0U; m < j; m++) {
                    free(arr[i][m]);
                }
                free(arr[i]);
                free_3d_array(arr, i, y);
                return NULL;
            }
            for (k = 0U; k < z; k++) {
                arr[i][j][k] = '*';
            }
        }
    }

    return arr;
}

void free_3d_array(char ***arr, size_t x, size_t y)
{
    size_t i;
    size_t j;

    if (arr == NULL) {
        return;
    }

    for (i = 0U; i < x; i++) {
        if (arr[i] != NULL) {
            for (j = 0U; j < y; j++) {
                free(arr[i][j]);
            }
            free(arr[i]);
        }
    }
    free(arr);
}

int main(void)
{
    const size_t x = 3U;
    const size_t y = 3U;
    const size_t z = 3U;
    size_t i;
    size_t j;
    size_t k;
    char ***arr;

    arr = generate_3d_array(x, y, z);
    if (arr == NULL) {
        (void)fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0U; i < x; i++) {
        for (j = 0U; j < y; j++) {
            for (k = 0U; k < z; k++) {
                if (printf("%c ", arr[i][j][k]) < 0) {
                    free_3d_array(arr, x, y);
                    return EXIT_FAILURE;
                }
            }
            if (printf("\n") < 0) {
                free_3d_array(arr, x, y);
                return EXIT_FAILURE;
            }
        }
        if (printf("\n") < 0) {
            free_3d_array(arr, x, y);
            return EXIT_FAILURE;
        }
    }

    free_3d_array(arr, x, y);
    return EXIT_SUCCESS;
}
