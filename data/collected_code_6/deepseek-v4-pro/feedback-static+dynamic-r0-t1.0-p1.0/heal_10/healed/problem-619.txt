#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void move_numbers_to_string(const char *input, char *output, size_t output_size) {
    size_t i = 0;
    size_t out_pos = 0;

    if (output == NULL || output_size == 0) {
        return;
    }

    output[0] = '\0';

    if (input == NULL) {
        return;
    }

    while (input[i] != '\0' && out_pos + 1 < output_size) {
        if (isdigit((unsigned char)input[i])) {
            output[out_pos++] = input[i];
        }
        i++;
    }

    output[out_pos] = '\0';
}

int main(void) {
    char input[256];
    char output[256];
    size_t len;
    int has_newline = 0;

    if (fgets(input, sizeof(input), stdin) == NULL) {
        return EXIT_FAILURE;
    }

    len = strlen(input);
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
        has_newline = 1;
    }

    if (!has_newline && len == sizeof(input) - 1) {
        int c;
        while ((c = getchar()) != '\n' && c != EOF) {
            if (c == EOF) break;
        }
        clearerr(stdin);
    }

    if (len > 0) {
        move_numbers_to_string(input, output, sizeof(output));
        if (output[0] != '\0') {
            printf("%s\n", output);
        }
    }

    return EXIT_SUCCESS;
}
