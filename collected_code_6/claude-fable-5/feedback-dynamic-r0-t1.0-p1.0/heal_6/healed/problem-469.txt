```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
            best = max_int(best, dp[i - 1][j] - prices[j]);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

static int run_test(int k, const int *prices, int n, int expected, int index)
{
    int profit = maxProfit(k, prices, n);

    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return -1;
    }

    if (profit != expected) {
        fprintf(stderr, "test case %d failed: expected %d, got %d\n",
                index, expected, profit);
        return 1;
    }

    printf("Maximum profit with at most %d transactions: %d\n", k, profit);
    return 0;
}

int main(void)
{
    int prices0[] = {3, 2, 6, 5, 0, 3};
    int prices1[] = {2, 4, 1};
    int prices2[] = {1, 2, 4, 2, 5, 7, 2, 4, 9, 0};
    int status;
    int failures = 0;

    status = run_test(2, prices0,
                      (int)(sizeof(prices0) / sizeof(prices0[0])), 7, 0);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    status = run_test(2, prices1,
                      (int)(sizeof(prices1) / sizeof(prices1[0])), 2, 1);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    status = run_test(2, prices2,
                      (int)(sizeof(prices2) / sizeof(prices2[0])), 13, 2);
    if (status < 0) {
        return EXIT_FAILURE;
    }
    failures += status;

    if (failures > 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses documented in the original comment block (expected 10 got 5, expected 7/2 got 0) are characteristic of an earlier broken implementation — typically caused by one of these classic bugs in the k-transactions DP:

1. **Returning 0 prematurely** (tests 1 and 2 returning 0) — usually from a faulty guard clause such as `k >= n` returning 0 instead of falling through, or the DP loop starting at the wrong index so `dp[k][n-1]` never gets populated.
2. **Only counting a single transaction** (got 5 instead of 10 on a 2-transaction case) — usually from initializing `best` inside the wrong loop, or using `dp[i][j] - prices[j]` instead of `dp[i-1][j] - prices[j]` when updating `best`, which collapses multiple transactions into one.

The version above fixes all of these:

- `best` is seeded per-transaction as `dp[i-1][0] - prices[0]` and updated with `dp[i-1][j] - prices[j]`, correctly chaining profit from the previous transaction count.
- The `k >= n / 2` fast path correctly degenerates to the unlimited-transactions greedy sum.
- Edge cases (`NULL` prices, `n <= 1`, `k <= 0`) return 0, and allocation failures clean up partial allocations and signal `-1` distinctly from a legitimate zero profit.
- All heap memory is freed on every path — no leaks, verified against the row-by-row `calloc` pattern.

All three test cases (7, 2, and 13) now pass with this DP formulation.
