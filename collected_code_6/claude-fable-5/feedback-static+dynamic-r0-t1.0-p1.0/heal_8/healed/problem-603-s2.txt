#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

static int *get_lucid_numbers(int n, size_t *count);

static int *get_lucid_numbers(int n, size_t *count)
{
    if (count == NULL || n < 1) {
        return NULL;
    }
    if ((size_t)n > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    size_t capacity = (size_t)n;
    size_t len = capacity;

    int *arr = malloc(len * sizeof *arr);
    if (arr == NULL) {
        return NULL;
    }

    int *result = malloc(capacity * sizeof *result);
    if (result == NULL) {
        free(arr);
        return NULL;
    }

    for (size_t i = 0; i < len; i++) {
        arr[i] = (int)(i + 1);
    }

    size_t rcount = 0;
    if (len > 0 && rcount < capacity) {
        result[rcount++] = arr[0];
    }

    size_t start = 1;
    while (start < len && rcount < capacity) {
        int step = arr[start];
        if (step <= 0) {
            break;
        }
        result[rcount++] = step;

        size_t write = start;
        size_t idx = 0;
        for (size_t read = start; read < len; read++, idx++) {
            if (idx % (size_t)step != 0) {
                if (write >= len || write > read || read >= capacity) {
                    break;
                }
                arr[write] = arr[read];
                write++;
            }
        }
        if (write > len) {
            break;
        }
        len = write;
    }

    free(arr);

    if (rcount > 0) {
        int *shrunk = realloc(result, rcount * sizeof *shrunk);
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    *count = rcount;
    return result;
}

int main(int argc, const char *const argv[])
{
    int n = 25;

    if (argc > 1) {
        const char *input = argv[1];
        char *end = NULL;
        errno = 0;
        long value = strtol(input, &end, 10);
        if (errno != 0 || end == input || *end != '\0' ||
            value < 1 || value > INT_MAX) {
            fprintf(stderr, "Invalid input: expected a positive integer\n");
            return EXIT_FAILURE;
        }
        n = (int)value;
    }

    size_t count = 0;
    int *lucid = get_lucid_numbers(n, &count);
    if (lucid == NULL) {
        fprintf(stderr, "Failed to compute lucid numbers\n");
        return EXIT_FAILURE;
    }

    printf("Lucid numbers <= %d:\n", n);
    for (size_t i = 0; i < count; i++) {
        printf("%d ", lucid[i]);
    }
    printf("\n");

    free(lucid);
    return EXIT_SUCCESS;
}

**Analysis summary of the changes:**

1. **Flawfinder buffer warnings (CWE-120, CWE-20):** The original code already had defensive bounds checks (`write >= len || write > read || read >= capacity`) before every `arr[write] = arr[read]` access. These were retained since they correctly prevent out-of-bounds reads and writes; the Flawfinder hits are risk-level-1 informational warnings about array indexing in loops, not actual exploitable defects.

2. **Hardened the initial result write:** The unconditional `result[rcount++] = arr[0]` was guarded with `len > 0 && rcount < capacity`. Although `n >= 1` guarantees this in practice, the explicit check removes any reliance on invariants distant from the access point.

3. **Guarded `len = write`:** Added a check that `write` never exceeds `len` before shrinking the logical array length, preventing any possibility of the loop bound growing.

4. **Existing safe practices preserved:** overflow check against `SIZE_MAX / sizeof(int)` before allocation, `strtol` with full `errno`/end-pointer/range validation instead of `atoi`, `NULL` checks on all allocations, graceful `realloc` failure handling (keeping the original buffer), and matching `free` for every allocation path — no leaks, no use-after-free, no unchecked user input.
