#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

int abs_diff(int a, int b) {
    return (a > b) ? (a - b) : (b - a);
}

int digit_distance(int num1, int num2) {
    char s1[64], s2[64];
    snprintf(s1, sizeof(s1), "%d", num1);
    snprintf(s2, sizeof(s2), "%d", num2);

    size_t len1 = strnlen(s1, sizeof(s1));
    size_t len2 = strnlen(s2, sizeof(s2));
    size_t max_len = (len1 > len2) ? len1 : len2;

    if (max_len >= sizeof(s1) || max_len >= sizeof(s2)) {
        fprintf(stderr, "Input numbers are too large.\n");
        exit(EXIT_FAILURE);
    }

    int distance = 0;
    for (size_t i = 0; i < max_len; ++i) {
        int d1 = (i < len1) ? s1[i] - '0' : 0;
        int d2 = (i < len2) ? s2[i] - '0' : 0;
        distance += abs_diff(d1, d2);
    }
    return distance;
}

int main() {
    const int test_cases[] = {234, 123, 987};
    const int test_cases_b[] = {489, 654, 123};
    const int expected_results[] = {35, 18, 24};

    for (int i = 0; i < 3; ++i) {
        int result = digit_distance(test_cases[i], test_cases_b[i]);
        if (result == expected_results[i]) {
            printf("Test case %d passed\n", i);
        } else {
            printf("Test case %d failed: expected %d, got %d\n", i, expected_results[i], result);
        }
    }

    return 0;
}
