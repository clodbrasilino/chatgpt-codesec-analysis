#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 8192

char* abbreviate_road(const char* input) {
    if (input == NULL) {
        return NULL;
    }

    size_t len = strnlen(input, MAX_INPUT_LEN);
    char* result = (char*)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    const char* target = "road";
    size_t target_len = 4;
    const char* replacement = "rd.";
    size_t replacement_len = 3;

    size_t i = 0;
    size_t j = 0;

    while (i < len) {
        if (tolower((unsigned char)input[i]) == 'r' && len - i >= target_len) {
            int match = 1;
            for (size_t k = 1; k < target_len; ++k) {
                if (tolower((unsigned char)input[i + k]) != target[k]) {
                    match = 0;
                    break;
                }
            }

            if (match) {
                int is_word_boundary_before = (i == 0 || !isalpha((unsigned char)input[i - 1]));
                int is_word_boundary_after = (i + target_len == len || !isalpha((unsigned char)input[i + target_len]));

                if (is_word_boundary_before && is_word_boundary_after) {
                    for (size_t k = 0; k < replacement_len; ++k) {
                        result[j++] = replacement[k];
                    }
                    i += target_len;
                    continue;
                }
            }
        }
        result[j++] = input[i++];
    }
    result[j] = '\0';

    return result;
}

int main(void) {
    const char* test_strings[] = {
        "The road is long.",
        "Abbey Road",
        "Broad street",
        "roadway",
        "A road.",
        NULL
    };

    for (size_t i = 0; test_strings[i] != NULL; ++i) {
        char* abbreviated = abbreviate_road(test_strings[i]);
        if (abbreviated != NULL) {
            printf("Original: '%s' -> Abbreviated: '%s'\n", test_strings[i], abbreviated);
            free(abbreviated);
        } else {
            fprintf(stderr, "Memory allocation failed or input was NULL.\n");
        }
    }

    return 0;
}
