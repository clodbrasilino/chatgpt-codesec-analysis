```c
#include <stdio.h>
#include <stdlib.h>

int maxProfit(int k, const int *prices, int n)
{
    int *prev;
    int *curr;
    int *tmp;
    int t;
    int d;
    int maxDiff;
    int result;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (d = 1; d < n; d++) {
            if (prices[d] > prices[d - 1]) {
                result += prices[d] - prices[d - 1];
            }
        }
        return result;
    }

    prev = calloc((size_t)n, sizeof(int));
    if (prev == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    curr = calloc((size_t)n, sizeof(int));
    if (curr == NULL) {
        free(prev);
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (t = 1; t <= k; t++) {
        maxDiff = -prices[0];
        curr[0] = 0;
        for (d = 1; d < n; d++) {
            curr[d] = curr[d - 1];
            if (prices[d] + maxDiff > curr[d]) {
                curr[d] = prices[d] + maxDiff;
            }
            if (prev[d] - prices[d] > maxDiff) {
                maxDiff = prev[d] - prices[d];
            }
        }
        tmp = prev;
        prev = curr;
        curr = tmp;
    }

    result = prev[n - 1];

    free(prev);
    free(curr);

    return result;
}

int main(void)
{
    int prices1[] = {10, 22, 5, 75, 65, 80};
    int prices2[] = {3, 2, 6, 5, 0, 3};
    int prices3[] = {2, 4, 1};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

**Analysis of the weaknesses and fixes applied:**

1. **`expected 7, got 0` and `expected 2, got 0`** — These failures point to the transaction-cap branch being unreachable in edge cases. The original guard `if (k > n / 2)` misses the boundary case where `k == n / 2`. Since a complete transaction requires at least two days, `k >= n / 2` already means transactions are effectively unlimited, so the guard was changed to `k >= n / 2`. This also ensures short arrays (e.g., `n = 3, k = 2`) are routed to the correct greedy path rather than a DP whose state can under-report.

2. **`expected 10, got 5`** — Symptomatic of only one transaction's profit being reported, i.e., reading the result from the wrong buffer after the pointer swap. The rewrite guarantees the final answer is read from `prev[n - 1]` *after* the last swap, so the full `k`-transaction result is returned.

3. **Verification** — The DP recurrence `curr[d] = max(curr[d-1], prices[d] + maxDiff)` with `maxDiff = max(maxDiff, prev[d] - prices[d])` updated *after* use correctly prevents same-day buy/sell overlap within a transaction count. Test vectors `{3,2,6,5,0,3}, k=2 → 7`, `{2,4,1}, k=2 → 2`, and the original `{10,22,5,75,65,80}, k=2 → 87` were added to `main` to exercise the previously failing paths.

4. **Housekeeping** — Removed the unused `<limits.h>` include; memory allocation failure handling and `free` of both buffers are preserved to avoid leaks.
