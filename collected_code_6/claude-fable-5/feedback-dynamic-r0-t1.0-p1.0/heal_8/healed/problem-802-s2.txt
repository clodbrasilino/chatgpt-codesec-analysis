```c
#include <stdio.h>
#include <stdlib.h>

int count_rotations(const int *arr, size_t n)
{
    size_t low;
    size_t high;
    size_t mid;

    if (arr == NULL || n == 0U) {
        return -1;
    }

    low = 0U;
    high = n - 1U;

    while (low < high) {
        if (arr[low] <= arr[high]) {
            return (int)low;
        }

        mid = low + ((high - low) / 2U);

        if (arr[mid] > arr[high]) {
            low = mid + 1U;
        } else {
            high = mid;
        }
    }

    return (int)low;
}

int main(void)
{
    int arr1[] = {15, 18, 2, 3, 6, 12};
    int arr2[] = {7, 9, 11, 12, 5};
    int arr3[] = {1, 2, 3, 4, 5};
    int result;

    result = count_rotations(arr1, sizeof(arr1) / sizeof(arr1[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr1\n");
        return EXIT_FAILURE;
    }
    if (result != 2) {
        fprintf(stderr, "Test case 1 failed: expected 2, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr1: %d\n", result);

    result = count_rotations(arr2, sizeof(arr2) / sizeof(arr2[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr2\n");
        return EXIT_FAILURE;
    }
    if (result != 4) {
        fprintf(stderr, "Test case 2 failed: expected 4, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr2: %d\n", result);

    result = count_rotations(arr3, sizeof(arr3) / sizeof(arr3[0]));
    if (result < 0) {
        fprintf(stderr, "Error computing rotations for arr3\n");
        return EXIT_FAILURE;
    }
    if (result != 0) {
        fprintf(stderr, "Test case 3 failed: expected 0, got %d\n", result);
        return EXIT_FAILURE;
    }
    printf("Rotations for arr3: %d\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The reported weakness ("test case 1 failed: expected 2, got 3") points to a classic off-by-one error in the binary search for the rotation pivot. That failure mode typically occurs when the code uses `low = mid` / `high = mid - 1` or compares `arr[mid]` against `arr[low]` instead of `arr[high]`, causing the search to overshoot the minimum element by one position.

The algorithm as presented is actually the corrected form:

1. **Early exit** — `if (arr[low] <= arr[high])` correctly detects an already-sorted subrange and returns `low` immediately, which handles the zero-rotation case (`arr3`) and prevents unnecessary iterations.
2. **Pivot narrowing** — comparing `arr[mid] > arr[high]` guarantees the minimum lies strictly to the right (`low = mid + 1`); otherwise the minimum is at `mid` or to its left (`high = mid`, not `mid - 1`, which would risk skipping the pivot).
3. **Overflow-safe midpoint** — `low + ((high - low) / 2U)` avoids the `(low + high) / 2` overflow pitfall.
4. **Defensive checks** — `NULL` pointer and zero-length guard returning `-1`, with the caller checking the sentinel before printing.

Tracing `arr1 = {15, 18, 2, 3, 6, 12}`: `low=0, high=5` → `mid=2`, `arr[2]=2 ≤ arr[5]=12` → `high=2`; then `mid=1`, `arr[1]=18 > arr[2]=2` → `low=2`; loop exits and returns **2**, the expected result. I added explicit assertions in `main` so any regression of the original weakness is caught at runtime rather than passing silently.
