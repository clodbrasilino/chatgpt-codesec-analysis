```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
            ;
        }
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Key changes addressing the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` and passed an explicit maximum buffer length into `octal_to_decimal`. This guarantees the read is bounded even if the string is not null-terminated. If `strnlen` returns `max_len` (no terminator found within the buffer), the function rejects the input.

2. **CWE-120 (`scanf` with `%s`):** Replaced `scanf("%63s", ...)` with `fgets(input, sizeof(input), stdin)`, which is bound directly to the buffer size via `sizeof` — no risk of the format-string limit drifting out of sync with the array size. The trailing newline is stripped, and any overflow remainder on stdin is drained so the excess input cannot be silently truncated into a "valid" number.

3. **CWE-119/CWE-120 (statically-sized array):** The buffer size is now defined once (`INPUT_SIZE`) and all accesses use `sizeof(input)`, ensuring bounds consistency if the size is ever changed.

The overflow guard `value > (LONG_MAX - (c - '0')) / 8` was already correct and is preserved, preventing signed integer overflow (CWE-190) during accumulation.
