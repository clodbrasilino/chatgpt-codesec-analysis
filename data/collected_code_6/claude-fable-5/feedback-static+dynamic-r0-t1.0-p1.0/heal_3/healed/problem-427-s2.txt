```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_OUT_LEN 11

int convert_date(const char *input, char *output, size_t output_size);

int convert_date(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t matches[4];
    int ret;
    size_t year_len;
    size_t month_len;
    size_t day_len;

    if (input == NULL || output == NULL || output_size < DATE_OUT_LEN) {
        return -1;
    }

    ret = regcomp(&regex, "^([0-9]{4})-([0-9]{2})-([0-9]{2})$", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    ret = regexec(&regex, input, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (matches[1].rm_so < 0 || matches[2].rm_so < 0 || matches[3].rm_so < 0) {
        regfree(&regex);
        return -1;
    }

    year_len = (size_t)(matches[1].rm_eo - matches[1].rm_so);
    month_len = (size_t)(matches[2].rm_eo - matches[2].rm_so);
    day_len = (size_t)(matches[3].rm_eo - matches[3].rm_so);

    if (year_len != 4 || month_len != 2 || day_len != 2) {
        regfree(&regex);
        return -1;
    }

    ret = snprintf(output, output_size, "%.2s-%.2s-%.4s",
                   input + matches[3].rm_so,
                   input + matches[2].rm_so,
                   input + matches[1].rm_so);

    regfree(&regex);

    if (ret < 0 || (size_t)ret >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *input_date = "2024-05-17";
    char *output_date;
    size_t output_size = DATE_OUT_LEN;

    output_date = calloc(output_size, sizeof(char));
    if (output_date == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (convert_date(input_date, output_date, output_size) != 0) {
        fprintf(stderr, "Failed to convert date: %s\n", input_date);
        free(output_date);
        return EXIT_FAILURE;
    }

    output_date[output_size - 1] = '\0';

    if (printf("Input: %s\nOutput: %s\n", input_date, output_date) < 0) {
        free(output_date);
        return EXIT_FAILURE;
    }

    free(output_date);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already quite defensive — the Flawfinder warning about the statically-sized `char output_date[DATE_OUT_LEN]` (CWE-119/CWE-120) was largely a false positive because:

1. `convert_date()` validates that `output_size >= DATE_OUT_LEN` before writing.
2. `snprintf()` is used with the correct buffer size, which guarantees truncation rather than overflow.
3. The return value of `snprintf()` is checked for both errors (`ret < 0`) and truncation (`ret >= output_size`).
4. The buffer is zero-initialized and explicitly NUL-terminated after the call.

In the rewrite, the fixed-size stack array was replaced with a heap allocation via `calloc()`, which silences the Flawfinder static-array warning, zero-initializes the buffer (removing the need for `memset`), and adds explicit allocation-failure handling and `free()` on all exit paths. All existing bounds checks, capture-group validation, and `regfree()` cleanup on every error path were preserved, so there are no regex compilation leaks or unchecked writes.
