```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U

typedef struct {
    char *first;
    char *second;
} StringTuple;

static char *concat_strings(const char *a, const char *b)
{
    size_t len_a;
    size_t len_b;
    size_t total_len;
    char *result;

    if (a == NULL || b == NULL) {
        return NULL;
    }

    len_a = strnlen(a, MAX_STRING_LEN);
    len_b = strnlen(b, MAX_STRING_LEN);

    if (len_a >= MAX_STRING_LEN || len_b >= MAX_STRING_LEN) {
        return NULL;
    }

    if (len_a > SIZE_MAX - len_b - 1U) {
        return NULL;
    }

    total_len = len_a + len_b + 1U;

    result = malloc(total_len);
    if (result == NULL) {
        return NULL;
    }

    if (len_a > 0U) {
        if (len_a >= total_len) {
            free(result);
            return NULL;
        }
        memcpy(result, a, len_a);
    }

    if (len_b > 0U) {
        if (len_b >= total_len - len_a) {
            free(result);
            return NULL;
        }
        memcpy(result + len_a, b, len_b);
    }

    result[len_a + len_b] = '\0';

    return result;
}

static int concat_tuples(const StringTuple *t1, const StringTuple *t2, StringTuple *out)
{
    if (t1 == NULL || t2 == NULL || out == NULL) {
        return -1;
    }

    out->first = NULL;
    out->second = NULL;

    out->first = concat_strings(t1->first, t2->first);
    if (out->first == NULL) {
        return -1;
    }

    out->second = concat_strings(t1->second, t2->second);
    if (out->second == NULL) {
        free(out->first);
        out->first = NULL;
        return -1;
    }

    return 0;
}

static void free_tuple(StringTuple *t)
{
    if (t != NULL) {
        free(t->first);
        free(t->second);
        t->first = NULL;
        t->second = NULL;
    }
}

int main(void)
{
    StringTuple t1;
    StringTuple t2;
    StringTuple result;

    t1.first = "Hello, ";
    t1.second = "Good ";
    t2.first = "World!";
    t2.second = "Morning!";

    if (concat_tuples(&t1, &t2, &result) != 0) {
        fprintf(stderr, "Error: tuple concatenation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("(%s, %s)\n", result.first, result.second) < 0) {
        free_tuple(&result);
        return EXIT_FAILURE;
    }

    free_tuple(&result);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two Flawfinder `memcpy` warnings (CWE-120) are false positives in this code, and the rewritten version retains the existing mitigations because they already fully address the concern:

1. **Bounded source lengths** — `strnlen(..., MAX_STRING_LEN)` guarantees neither input length exceeds the cap, and lengths equal to `MAX_STRING_LEN` are rejected, so both strings are known to be NUL-terminated within bounds.

2. **Overflow-safe size arithmetic** — the check `len_a > SIZE_MAX - len_b - 1U` prevents integer overflow when computing `total_len`, so `malloc(total_len)` always requests the correct size.

3. **Destination capacity verified before each copy** — the guards `len_a >= total_len` and `len_b >= total_len - len_a` explicitly prove the destination buffer can hold each copy before `memcpy` executes, which is exactly the mitigation CWE-120 asks for.

4. **No leaks on failure paths** — `result` is freed on every early return, and `concat_tuples` releases `out->first` if the second concatenation fails, leaving the output struct in a consistent NULL state.

Since the flagged copies are provably within an allocation sized as `len_a + len_b + 1`, the code is already memory-safe; no functional changes were required beyond keeping the defensive checks in place.
