```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096U

static size_t bounded_length(const char *str, size_t max_length, int *valid)
{
    size_t length;

    if (str == NULL || valid == NULL) {
        if (valid != NULL) {
            *valid = 0;
        }
        return 0U;
    }

    length = strnlen(str, max_length + 1U);
    if (length > max_length) {
        *valid = 0;
        return 0U;
    }
    *valid = 1;
    return length;
}

static int safe_copy(char *dest, size_t dest_size, size_t offset, const char *src, size_t src_length)
{
    if (dest == NULL || src == NULL) {
        return 0;
    }
    if (src_length == 0U) {
        return 1;
    }
    if (offset >= dest_size) {
        return 0;
    }
    if (src_length > dest_size - offset) {
        return 0;
    }
    memcpy(dest + offset, src, src_length);
    return 1;
}

char *concatenate_tuple(const char *const *elements, size_t count, const char *delimiter)
{
    size_t total_length;
    size_t delimiter_length;
    size_t offset;
    size_t i;
    size_t *lengths;
    char *result;
    int valid;

    if (elements == NULL || delimiter == NULL || count == 0U) {
        return NULL;
    }

    delimiter_length = bounded_length(delimiter, MAX_STRING_LENGTH, &valid);
    if (!valid) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(size_t)) {
        return NULL;
    }

    lengths = malloc(count * sizeof(size_t));
    if (lengths == NULL) {
        return NULL;
    }

    total_length = 0U;

    for (i = 0U; i < count; i++) {
        if (elements[i] == NULL) {
            free(lengths);
            return NULL;
        }
        lengths[i] = bounded_length(elements[i], MAX_STRING_LENGTH, &valid);
        if (!valid) {
            free(lengths);
            return NULL;
        }
        if (lengths[i] > SIZE_MAX - total_length) {
            free(lengths);
            return NULL;
        }
        total_length += lengths[i];
    }

    if (delimiter_length != 0U && count - 1U > (SIZE_MAX - total_length) / delimiter_length) {
        free(lengths);
        return NULL;
    }
    total_length += delimiter_length * (count - 1U);

    if (total_length == SIZE_MAX) {
        free(lengths);
        return NULL;
    }

    result = malloc(total_length + 1U);
    if (result == NULL) {
        free(lengths);
        return NULL;
    }

    offset = 0U;

    for (i = 0U; i < count; i++) {
        if (i > 0U && delimiter_length > 0U) {
            if (!safe_copy(result, total_length + 1U, offset, delimiter, delimiter_length)) {
                free(result);
                free(lengths);
                return NULL;
            }
            offset += delimiter_length;
        }
        if (lengths[i] > 0U) {
            if (!safe_copy(result, total_length + 1U, offset, elements[i], lengths[i])) {
                free(result);
                free(lengths);
                return NULL;
            }
            offset += lengths[i];
        }
    }

    if (offset > total_length) {
        free(result);
        free(lengths);
        return NULL;
    }

    result[offset] = '\0';
    free(lengths);

    return result;
}

int main(void)
{
    const char *tuple[] = {"ID", "is", "4", "UTS"};
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    char *joined;

    joined = concatenate_tuple(tuple, count, "-");
    if (joined == NULL) {
        fprintf(stderr, "Error: concatenation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", joined) < 0) {
        free(joined);
        return EXIT_FAILURE;
    }

    free(joined);
    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes made:**

1. **`safe_copy` hardening (CWE-120 fix):** The flagged `memcpy` was already guarded, but the original bounds check had a subtle flaw — `offset > dest_size` allowed `offset == dest_size`, and the caller passed `total_length` as `dest_size` even though the buffer is actually `total_length + 1` bytes. The rewrite passes the true allocated size (`total_length + 1U`), rejects `offset >= dest_size`, short-circuits zero-length copies, and verifies `src_length <= dest_size - offset` before calling `memcpy`, guaranteeing the destination can always hold the source data.

2. **`bounded_length` null-safety:** Added defensive `NULL` checks on `str` and `valid` so the function cannot dereference a null pointer if misused, while preserving the `strnlen`-based bounded scan that prevents unbounded reads on non-terminated input.

3. **Final write bounds check:** Added an explicit `offset > total_length` guard before writing the terminating `'\0'`, ensuring the NUL byte can never land outside the allocated buffer even if the copy logic were ever modified incorrectly.

4. **Preserved overflow protections:** All existing integer-overflow checks (`count * sizeof(size_t)`, cumulative `total_length`, delimiter multiplication, and the `SIZE_MAX` allocation guard) remain intact, as they correctly prevent CWE-190 style allocation-size wraparound.
