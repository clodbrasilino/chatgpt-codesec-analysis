#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    long long val_a = *(const long long *)a;
    long long val_b = *(const long long *)b;
    if (val_a < val_b) return 1;
    if (val_a > val_b) return -1;
    return 0;
}

long long* find_largest_products(const int *list1, size_t len1, const int *list2, size_t len2, size_t k, size_t *out_len) {
    if (k == 0) {
        *out_len = 0;
        return NULL;
    }

    size_t total_products = len1 * len2;
    size_t result_len = k < total_products ? k : total_products;
    
    long long *products = (long long *)malloc(total_products * sizeof(long long));
    if (!products) {
        *out_len = 0;
        return NULL;
    }

    size_t index = 0;
    for (size_t i = 0; i < len1; ++i) {
        for (size_t j = 0; j < len2; ++j) {
            products[index++] = (long long)list1[i] * list2[j];
        }
    }

    qsort(products, total_products, sizeof(long long), compare);

    long long *result = (long long *)malloc(result_len * sizeof(long long));
    if (!result) {
        free(products);
        *out_len = 0;
        return NULL;
    }

    for (size_t i = 0; i < result_len; ++i) {
        result[i] = products[i];
    }

    free(products);
    *out_len = result_len;
    return result;
}

int main() {
    int list1[] = {1, -5, 10, 3};
    int list2[] = {2, 6, -4};
    size_t len1 = sizeof(list1) / sizeof(list1[0]);
    size_t len2 = sizeof(list2) / sizeof(list2[0]);
    size_t k = 5;
    size_t out_len;

    long long *result = find_largest_products(list1, len1, list2, len2, k, &out_len);

    if (result != NULL) {
        for (size_t i = 0; i < out_len; ++i) {
            printf("%lld\n", result[i]);
        }
        free(result);
    }

    return 0;
}
