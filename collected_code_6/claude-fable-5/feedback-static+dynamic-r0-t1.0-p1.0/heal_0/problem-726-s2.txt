#include <stdio.h>
#include <stdlib.h>

int *multiply_adjacent(const int *arr, size_t n, size_t *out_size)
{
    int *result;
    size_t i;

    if (arr == NULL || out_size == NULL || n < 2) {
        if (out_size != NULL) {
            *out_size = 0;
        }
        return NULL;
    }

    result = malloc((n - 1) * sizeof(int));
    if (result == NULL) {
        *out_size = 0;
        return NULL;
    }

    for (i = 0; i < n - 1; i++) {
        result[i] = arr[i] * arr[i + 1];
    }

    *out_size = n - 1;
    return result;
}

int main(void)
{
    int tuple[] = {1, 5, 7, 8, 10};
    size_t n = sizeof(tuple) / sizeof(tuple[0]);
    size_t result_size = 0;
    size_t i;
    int *result;

    result = multiply_adjacent(tuple, n, &result_size);
    if (result == NULL) {
        fprintf(stderr, "Failed to compute adjacent products\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (i = 0; i < result_size; i++) {
        printf("%d", result[i]);
        if (i < result_size - 1) {
            printf(", ");
        }
    }
    printf(")\n");

    free(result);
    return EXIT_SUCCESS;
}
