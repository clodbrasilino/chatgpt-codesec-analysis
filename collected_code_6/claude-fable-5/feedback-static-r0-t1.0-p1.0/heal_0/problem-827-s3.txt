#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int sum_column(const int **matrix, size_t rows, size_t cols, size_t col, long long *result)
{
    size_t i;
    long long sum = 0;

    if (matrix == NULL || result == NULL) {
        return -1;
    }

    if (col >= cols) {
        return -1;
    }

    for (i = 0; i < rows; i++) {
        if (matrix[i] == NULL) {
            return -1;
        }
        if (matrix[i][col] > 0 && sum > LLONG_MAX - matrix[i][col]) {
            return -1;
        }
        if (matrix[i][col] < 0 && sum < LLONG_MIN - matrix[i][col]) {
            return -1;
        }
        sum += matrix[i][col];
    }

    *result = sum;
    return 0;
}

int main(void)
{
    size_t rows = 3;
    size_t cols = 4;
    size_t i;
    size_t j;
    int **matrix;
    long long result;
    int status;

    matrix = malloc(rows * sizeof(int *));
    if (matrix == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < rows; i++) {
        matrix[i] = malloc(cols * sizeof(int));
        if (matrix[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            for (j = 0; j < i; j++) {
                free(matrix[j]);
            }
            free(matrix);
            return EXIT_FAILURE;
        }
        for (j = 0; j < cols; j++) {
            matrix[i][j] = (int)(i * cols + j + 1);
        }
    }

    status = sum_column((const int **)matrix, rows, cols, 2, &result);
    if (status != 0) {
        fprintf(stderr, "Failed to compute column sum\n");
        for (i = 0; i < rows; i++) {
            free(matrix[i]);
        }
        free(matrix);
        return EXIT_FAILURE;
    }

    printf("Sum of column 2: %lld\n", result);

    for (i = 0; i < rows; i++) {
        free(matrix[i]);
    }
    free(matrix);

    return EXIT_SUCCESS;
}
