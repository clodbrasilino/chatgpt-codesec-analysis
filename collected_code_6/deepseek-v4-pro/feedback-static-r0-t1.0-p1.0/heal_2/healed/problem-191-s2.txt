#include <stdio.h>
#include <string.h>
#include <ctype.h>

int has_30_days(const char *month) {
    if (month == NULL) {
        return 0;
    }
    
    char lower[32];
    size_t len = strlen(month);
    if (len >= sizeof(lower)) {
        return 0;
    }
    
    for (size_t i = 0; i < len; i++) {
        lower[i] = tolower((unsigned char)month[i]);
    }
    lower[len] = '\0';
    
    const char *months_30[] = {
        "april", "june", "september", "november"
    };
    
    for (int i = 0; i < 4; i++) {
        if (strcmp(lower, months_30[i]) == 0) {
            return 1;
        }
    }
    
    return 0;
}

int main(void) {
    const char *test_months[] = {
        "January", "february", "March", "April",
        "May", "June", "July", "August",
        "September", "October", "November", "December",
        "april", "JUNE", "SePtEmBeR", NULL
    };
    
    for (int i = 0; test_months[i] != NULL; i++) {
        printf("%s: %s\n", test_months[i], 
               has_30_days(test_months[i]) ? "has 30 days" : "does not have 30 days");
    }
    
    printf("NULL: %s\n", has_30_days(NULL) ? "has 30 days" : "does not have 30 days");
    
    return 0;
}
