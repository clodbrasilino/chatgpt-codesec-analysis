#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void replace_spaces(char *str, char replacement) {
    if (str == NULL) {
        return;
    }

    while (*str != '\0') {
        if (*str == ' ') {
            *str = replacement;
        }
        str++;
    }
}

int main(void) {
    char *text = NULL;
    size_t size = 100;
    
    text = (char *)malloc(size * sizeof(char));
    if (text == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    snprintf(text, size, "%s", "Hello World! This is a test string.");

    printf("Original: %s\n", text);
    
    replace_spaces(text, '-');
    
    printf("Replaced: %s\n", text);

    free(text);
    text = NULL;

    return 0;
}
