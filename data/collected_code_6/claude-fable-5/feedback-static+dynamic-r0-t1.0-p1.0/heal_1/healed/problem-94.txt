```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 32

typedef struct {
    char name[MAX_NAME_LEN];
    int value;
} Record;

int index_minimum(const Record *records, size_t count, char *result, size_t result_size)
{
    size_t i;
    size_t min_index;
    size_t name_len;

    if (records == NULL || result == NULL || count == 0 || result_size == 0) {
        return -1;
    }

    min_index = 0;
    for (i = 1; i < count; i++) {
        if (records[i].value < records[min_index].value) {
            min_index = i;
        }
    }

    name_len = strnlen(records[min_index].name, MAX_NAME_LEN);
    if (name_len >= MAX_NAME_LEN || name_len >= result_size) {
        return -1;
    }

    memcpy(result, records[min_index].name, name_len);
    result[name_len] = '\0';

    return 0;
}

int main(void)
{
    Record records[] = {
        {"Rash", 143},
        {"Manjeet", 200},
        {"Varsha", 100}
    };
    size_t count = sizeof(records) / sizeof(records[0]);
    char result[MAX_NAME_LEN];

    if (index_minimum(records, count, result, sizeof(result)) != 0) {
        fprintf(stderr, "Error: could not extract minimum record\n");
        return EXIT_FAILURE;
    }

    printf("Record with minimum value: %s\n", result);

    return EXIT_SUCCESS;
}
```

**Key changes addressing the identified weaknesses:**

1. **`strlen` → `strnlen`** (CWE-126): The original `strlen` call could over-read if a `name` field were not null-terminated (e.g., if a record was populated via `strncpy` filling all 32 bytes). `strnlen(name, MAX_NAME_LEN)` bounds the scan to the buffer size, eliminating any possible over-read. The subsequent check `name_len >= MAX_NAME_LEN` also detects and rejects unterminated strings rather than silently truncating them.

2. **`strncpy` → `memcpy` + explicit terminator** (CWE-120): `strncpy` is error-prone — it doesn't guarantee null-termination and zero-pads the remainder of the destination, which is wasted work. Since the length is already validated against both `MAX_NAME_LEN` and `result_size`, a `memcpy` of exactly `name_len` bytes followed by an explicit `'\0'` is both safe and unambiguous.

3. **Statically-sized arrays** (CWE-119/CWE-120): The fixed-size buffers themselves are acceptable here because every access is now provably bounds-checked — the copy length is validated against `result_size` before any write, and reads are capped at `MAX_NAME_LEN`. The existing defensive checks for `NULL` pointers, zero `count`, and zero `result_size` are retained.

The program's observable behavior is unchanged (it prints `Record with minimum value: Varsha`), but all buffer reads and writes are now strictly bounded.
