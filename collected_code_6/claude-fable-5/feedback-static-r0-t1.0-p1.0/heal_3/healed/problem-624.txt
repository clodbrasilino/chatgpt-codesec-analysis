```c
#include <stdio.h>
#include <ctype.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 1024

char *string_to_upper(char *str)
{
    size_t i;

    if (str == NULL) {
        return NULL;
    }

    for (i = 0; str[i] != '\0'; i++) {
        str[i] = (char)toupper((unsigned char)str[i]);
    }

    return str;
}

int main(void)
{
    const char *source = "Hello, World! 123 abc";
    size_t len;
    size_t buffer_size;
    char *buffer;

    len = strnlen(source, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input string too long or not null-terminated\n");
        return EXIT_FAILURE;
    }

    if (len > SIZE_MAX - 1) {
        fprintf(stderr, "Length overflow\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    buffer = malloc(buffer_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (snprintf(buffer, buffer_size, "%s", source) < 0) {
        fprintf(stderr, "Copy failed\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (string_to_upper(buffer) == NULL) {
        fprintf(stderr, "Conversion failed\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    if (printf("Original:  %s\n", source) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }

    if (printf("Uppercase: %s\n", buffer) < 0) {
        free(buffer);
        return EXIT_FAILURE;
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Integer overflow guard for `len + 1`**: Added an explicit check against `SIZE_MAX` before computing `buffer_size`, eliminating any possibility of a wraparound in the size calculation (CWE-190).

2. **Removed the always-false condition**: The `len >= buffer_size` check was dead code flagged by cppcheck (`knownConditionTrueFalse`), since `buffer_size` is by construction `len + 1`. Dead branches obscure logic and should be removed.

3. **Replaced `memcpy` with `snprintf`**: The Flawfinder warning (CWE-120) about `memcpy` is resolved by using `snprintf` with the exact buffer size, which guarantees bounds-checked copying and null termination in a single operation, removing the manual `buffer[len] = '\0'` step.

4. **Added `<stdint.h>`**: Required for the `SIZE_MAX` constant.

The remaining structure (allocation failure check, `toupper` with the `unsigned char` cast to avoid UB on negative `char` values, `printf` return-value checks, and consistent `free` on all exit paths) was already sound and preserved.
