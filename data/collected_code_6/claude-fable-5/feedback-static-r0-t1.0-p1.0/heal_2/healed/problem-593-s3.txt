```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_IP_LEN 64

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t n = 0;

    while (n < max_len && s[n] != '\0') {
        n++;
    }
    return n;
}

int remove_leading_zeroes(const char *input, char *output, size_t out_size)
{
    size_t i = 0;
    size_t j = 0;
    size_t len;

    if (input == NULL || output == NULL || out_size == 0) {
        return -1;
    }

    len = bounded_strlen(input, MAX_IP_LEN);
    if (len >= MAX_IP_LEN) {
        return -1;
    }

    while (i < len) {
        if (isdigit((unsigned char)input[i])) {
            size_t start = i;

            while (i < len && isdigit((unsigned char)input[i])) {
                i++;
            }
            while (start < i - 1 && input[start] == '0') {
                start++;
            }
            while (start < i) {
                if (j + 1 >= out_size) {
                    return -1;
                }
                output[j] = input[start];
                j++;
                start++;
            }
        } else if (input[i] == '.') {
            if (j + 1 >= out_size) {
                return -1;
            }
            output[j] = input[i];
            j++;
            i++;
        } else {
            return -1;
        }
    }

    if (j >= out_size) {
        return -1;
    }
    output[j] = '\0';
    return 0;
}

int main(void)
{
    const char *tests[] = {
        "192.068.001.010",
        "010.020.030.040",
        "000.000.000.000",
        "127.0.0.1"
    };
    char result[MAX_IP_LEN];
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    memset(result, 0, sizeof(result));

    for (i = 0; i < count; i++) {
        if (remove_leading_zeroes(tests[i], result, sizeof(result)) == 0) {
            if (printf("%s -> %s\n", tests[i], result) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Invalid input: %s\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning on `char result[MAX_IP_LEN]` is a low-risk, heuristic-based flag — it fires on any statically-sized array. In this code the risk is already mitigated, so no structural change is required:

1. **Size is always passed explicitly** — `remove_leading_zeroes` receives `sizeof(result)` rather than assuming a fixed capacity, so the callee never writes beyond what the caller owns.
2. **Every write is bounds-checked** — the `j + 1 >= out_size` guards before each character write reserve room for the NUL terminator, and the final `j >= out_size` check protects the terminator write itself.
3. **Input length is bounded** — `bounded_strlen` caps scanning at `MAX_IP_LEN`, rejecting unterminated or oversized inputs instead of reading out of bounds (avoiding the classic `strlen` overread).
4. **Buffer is zero-initialized** with `memset`, so no uninitialized reads occur even on early failure paths.
5. **`isdigit` is called with an `unsigned char` cast**, avoiding undefined behavior for negative `char` values.

The code as written is already defensively correct; the rewrite preserves those mitigations verbatim.
