```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static void print_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1U);
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (*offset + (size_t)end) <= input_len) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN &&
                (*offset + (size_t)start + len) <= input_len) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    *offset += (size_t)matches[0].rm_eo;
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "Regex execution failed");
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two flawfinder findings in the original source were already properly mitigated, so the rewritten code preserves those defenses rather than changing logic:

1. **`errbuf` statically-sized array (CWE-119/CWE-120):** Mitigated by zero-initializing the buffer with `memset`, passing `sizeof(errbuf) - 1U` to `regerror` so it can never write to the final byte, and explicitly NUL-terminating the last byte. Overflow is impossible.

2. **`memcpy` (CWE-120):** Mitigated by a chain of guards before the copy:
   - `strnlen` bounds the input to `MAX_INPUT_LEN`, rejecting unterminated/oversized input.
   - Match offsets are validated (`start >= 0`, `end >= start`).
   - `(*offset + (size_t)end) <= input_len` and `(*offset + (size_t)start + len) <= input_len` guarantee the source range lies within the input.
   - The destination is allocated as exactly `len + 1U` bytes, so the copy of `len` bytes plus the NUL terminator always fits.
   - `malloc` failure is checked before use.

Additional correctness properties retained: `regfree` is called on every path after a successful `regcomp` (no resource leak), `REG_NOMATCH` is distinguished from real errors, the caller frees each returned string, and integer arithmetic is performed in `size_t` after sign checks to avoid signed/unsigned conversion issues with `regoff_t`.
