```C
#include <stdio.h>
#include <stdlib.h>

typedef struct tuple {
    int x;
    int y;
} Tuple;

char* tuple_to_string(Tuple* t) {
    char* buffer = (char*)malloc(sizeof(char) * 64);
    if (buffer == NULL) {
        printf("Memory allocation failed.\n");
        exit(1);
    }
    
    int written = snprintf(buffer, 64, "(%d, %d)", t->x, t->y);
    if (written <= 0 || written >= 64) {
        printf("Conversion to string failed.\n");
        free(buffer);
        exit(1);
    }

    return buffer;
}

int main() {
    Tuple tuple = {5, 10};
    char* str = tuple_to_string(&tuple);
    printf("%s\n", str);
    free(str);
    return 0;
}
```
