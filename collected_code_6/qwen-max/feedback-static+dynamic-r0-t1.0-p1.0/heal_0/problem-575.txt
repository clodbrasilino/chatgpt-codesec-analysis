```c
#include <stdio.h>

int find_nth_non_multiple(int n, int multiple) {
    int count = 0;
    int current = 1;

    while (1) {
        if (current % multiple != 0) {
            count++;
            if (count == n) {
                return current;
            }
        }
        current++;
    }
}

int main() {
    int n = 15;
    int multiple = 4;
    int result = find_nth_non_multiple(n, multiple);
    printf("%d\n", result);
    return 0;
}
```
