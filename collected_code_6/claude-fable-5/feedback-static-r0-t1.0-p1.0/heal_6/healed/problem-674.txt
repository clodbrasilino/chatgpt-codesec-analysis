static int word_seen(char **words, size_t count, const char *word)
{
    size_t i;
    for (i = 0; i < count; i++) {
        if (strcmp(words[i], word) == 0) {
            return 1;
        }
    }
    return 0;
}

static void free_words(char **words, size_t count)
{
    size_t i;
    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

static char *duplicate_string(const char *src, size_t src_len)
{
    char *dest;
    size_t dest_size;

    if (src == NULL || src_len == SIZE_MAX) {
        return NULL;
    }

    dest_size = src_len + 1;
    dest = malloc(dest_size);
    if (dest == NULL) {
        return NULL;
    }

    if (src_len > dest_size - 1) {
        free(dest);
        return NULL;
    }

    memcpy(dest, src, src_len);
    dest[src_len] = '\0';
    return dest;
}

char *remove_duplicate_words(const char *input)
{
    char *copy;
    char *result;
    char **words;
    char *token;
    char *saveptr;
    size_t count;
    size_t capacity;
    size_t len;
    size_t i;
    size_t offset;
    size_t result_size;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, SIZE_MAX - 1);

    copy = duplicate_string(input, len);
    if (copy == NULL) {
        return NULL;
    }

    capacity = 16;
    count = 0;
    words = malloc(capacity * sizeof(char *));
    if (words == NULL) {
        free(copy);
        return NULL;
    }

    token = strtok_r(copy, " \t\n", &saveptr);
    while (token != NULL) {
        if (!word_seen(words, count, token)) {
            size_t token_len;
            if (count == capacity) {
                size_t new_capacity;
                char **tmp;
                if (capacity > SIZE_MAX / 2 / sizeof(char *)) {
                    free_words(words, count);
                    free(copy);
                    return NULL;
                }
                new_capacity = capacity * 2;
                tmp = realloc(words, new_capacity * sizeof(char *));
                if (tmp == NULL) {
                    free_words(words, count);
                    free(copy);
                    return NULL;
                }
                words = tmp;
                capacity = new_capacity;
            }
            token_len = strnlen(token, len);
            words[count] = duplicate_string(token, token_len);
            if (words[count] == NULL) {
                free_words(words, count);
                free(copy);
                return NULL;
            }
            count++;
        }
        token = strtok_r(NULL, " \t\n", &saveptr);
    }

    free(copy);

    result_size = len + 1;
    result = malloc(result_size);
    if (result == NULL) {
        free_words(words, count);
        return NULL;
    }

    offset = 0;
    for (i = 0; i < count; i++) {
        size_t word_len = strnlen(words[i], len);
        if (i > 0) {
            if (offset >= len || offset >= result_size - 1) {
                free_words(words, count);
                free(result);
                return NULL;
            }
            result[offset] = ' ';
            offset++;
        }
        if (word_len > len - offset || offset + word_len > result_size - 1) {
            free_words(words, count);
            free(result);
            return NULL;
        }
        memcpy(result + offset, words[i], word_len);
        offset += word_len;
    }
    result[offset] = '\0';

    free_words(words, count);
    return result;
}

{

        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;

    printf("Original: %s\n", text);
    printf("Deduplicated: %s\n", deduped);

    free(deduped);
    return EXIT_SUCCESS;

**Analysis of the original code and the flagged weaknesses:**

The two Flawfinder warnings both concern `memcpy` (CWE-120, classic buffer overflow). Reviewing them:

1. **`memcpy` in `duplicate_string`** — This was already safe. The destination is allocated as `src_len + 1`, so copying `src_len` bytes plus a NUL terminator always fits. The `src_len == SIZE_MAX` guard prevents integer overflow in `src_len + 1`, and the `src_len > dest_size - 1` check is a defensive (if tautological) belt-and-suspenders guard. No change required.

2. **`memcpy` in `remove_duplicate_words`** — Mostly safe, since the deduplicated output can never exceed the input length. The existing check `offset + word_len > result_size - 1` correctly bounds the copy (and cannot overflow, since both `offset` and `word_len` are bounded by `len`). The one hardening improvement I made: the separator write `result[offset] = ' '` was only guarded by `offset >= len`; I added an explicit `offset >= result_size - 1` check so the bound is verified directly against the actual buffer size rather than indirectly via `len`, making the invariant locally verifiable and robust against future refactoring.

Other correctness points verified: `strtok_r` is used instead of the non-reentrant `strtok`; all allocation failures are handled with full cleanup (no leaks on any error path); the `realloc` growth path checks for multiplication overflow before doubling; and `strnlen` is used throughout to bound string scans.
