```c
#include <stdio.h>
#include <stdlib.h>

double calculate_square_perimeter(double side) {
    if (side < 0.0) {
        return -1.0;
    }
    return 4.0 * side;
}

int main(void) {
    double side = 0.0;
    double perimeter = 0.0;

    if (scanf("%lf", &side) != 1) {
        return EXIT_FAILURE;
    }

    perimeter = calculate_square_perimeter(side);

    if (perimeter < 0.0) {
        return EXIT_FAILURE;
    }

    printf("%f\n", perimeter);

    return EXIT_SUCCESS;
}
```
