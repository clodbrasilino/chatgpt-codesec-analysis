```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LEN 4096

static size_t bounded_strlen(const char *s, size_t maxlen)
{
    size_t len = 0;

    while (len < maxlen && s[len] != '\0') {
        len++;
    }
    return len;
}

char **shortlist_words(const char *const *words, size_t count, size_t n, size_t *result_count)
{
    char **result = NULL;
    size_t matched = 0;
    size_t i;

    if (words == NULL || result_count == NULL) {
        if (result_count != NULL) {
            *result_count = 0;
        }
        return NULL;
    }

    *result_count = 0;

    for (i = 0; i < count; i++) {
        if (words[i] != NULL) {
            size_t len = bounded_strlen(words[i], MAX_WORD_LEN);
            if (len < MAX_WORD_LEN && len > n) {
                matched++;
            }
        }
    }

    if (matched == 0) {
        return NULL;
    }

    if (matched > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = malloc(matched * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        if (words[i] != NULL) {
            size_t len = bounded_strlen(words[i], MAX_WORD_LEN);
            if (len < MAX_WORD_LEN && len > n) {
                char *copy = malloc(len + 1);
                if (copy == NULL) {
                    size_t j;
                    for (j = 0; j < *result_count; j++) {
                        free(result[j]);
                    }
                    free(result);
                    *result_count = 0;
                    return NULL;
                }
                memcpy(copy, words[i], len);
                copy[len] = '\0';
                result[*result_count] = copy;
                (*result_count)++;
            }
        }
    }

    return result;
}

void free_shortlist(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *const words[] = {
        "apple", "cat", "banana", "dog", "elephant", "sun", "computer"
    };
    size_t word_count = sizeof(words) / sizeof(words[0]);
    size_t n = 4;
    size_t result_count = 0;
    char **shortlisted;
    size_t i;

    shortlisted = shortlist_words(words, word_count, n, &result_count);

    if (shortlisted == NULL) {
        if (printf("No words longer than %zu found.\n", n) < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("Words longer than %zu characters:\n", n) < 0) {
        free_shortlist(shortlisted, result_count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < result_count; i++) {
        if (printf("%s\n", shortlisted[i]) < 0) {
            free_shortlist(shortlisted, result_count);
            return EXIT_FAILURE;
        }
    }

    free_shortlist(shortlisted, result_count);

    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **CWE-126 (strlen over-read):** Replaced all `strlen` calls with a `bounded_strlen` helper that never reads past `MAX_WORD_LEN` bytes, so a non-null-terminated input cannot cause an unbounded over-read. Strings that hit the bound without finding a terminator are rejected rather than processed.

2. **CWE-120 (memcpy overflow):** The destination buffer is allocated with `malloc(len + 1)` using the same `len` passed to `memcpy`, and the copy is now `memcpy(copy, words[i], len)` followed by explicit null termination — the destination is guaranteed to hold exactly what is copied, and the length is computed once (eliminating the TOCTOU-style double `strlen` in the original, where the string could theoretically differ between the two calls).

3. **Integer overflow hardening:** Added a `matched > SIZE_MAX / sizeof(char *)` check before the array allocation to prevent multiplication overflow in `malloc`.

4. **Consistency fix:** `*result_count` is zeroed before the counting pass, so the out-parameter is always in a defined state on every return path.

Note: `SIZE_MAX` requires `<stdint.h>` on some older toolchains; on C99+ it's available via `<stdint.h>`, so add `#include <stdint.h>` if your compiler complains (many provide it transitively through `<stdlib.h>`).
