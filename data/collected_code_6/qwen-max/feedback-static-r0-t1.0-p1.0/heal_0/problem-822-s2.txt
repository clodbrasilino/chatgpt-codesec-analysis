#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool is_valid_password(const char *password, size_t min_length) {
    if (password == NULL || min_length <= 0) {
        return false;
    }

    size_t length = strlen(password);

    if (length < min_length) {
        return false;
    }

    bool has_upper = 0, has_lower = 0, has_digit = 0;

    for (size_t i = 0; i < length; ++i) {
        if (password[i] >= 'A' && password[i] <= 'Z') {
            has_upper = 1;
        } else if (password[i] >= 'a' && password[i] <= 'z') {
            has_lower = 1;
        } else if (password[i] >= '0' && password[i] <= '9') {
            has_digit = 1;
        }
    }

    return has_upper && has_lower && has_digit;
}

int main() {
    const char *password = "Test123";
    size_t min_length = 8;

    if (is_valid_password(password, min_length)) {
        printf("Password is valid.\n");
    } else {
        printf("Password is not valid.\n");
    }

    return 0;
}
