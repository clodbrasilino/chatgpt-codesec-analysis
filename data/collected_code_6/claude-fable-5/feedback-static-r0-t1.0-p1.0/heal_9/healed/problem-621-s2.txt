static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n == 0) {
        return 0;
    }
    if (*len > SIZE_MAX - 1 || n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*buf == NULL || *cap < *len + n + 1) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

**1. `memcpy` in `buf_append` (CWE-120, risk 2):** The Flawfinder warning is a false positive here. Before the `memcpy` executes, the function guarantees sufficient capacity through multiple layers of defense: it rejects size arithmetic that would overflow (`*len > SIZE_MAX - 1 || n > SIZE_MAX - *len - 1`), grows the buffer geometrically with an overflow guard (`ncap > SIZE_MAX / 2`), and performs a final post-condition check (`*cap < *len + n + 1`) immediately before copying. The destination can always hold `n` bytes plus the NUL terminator.

**2. Statically-sized `char numbuf[32]` (CWE-119/CWE-120, risk 2):** This is also a false positive. A `long long` (even 64-bit) formats to at most 20 characters including the sign, well within 32 bytes. More importantly, `snprintf` with `sizeof numbuf` bounds the write, and the return value is checked for both encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof numbuf`), so an overflow or silent truncation cannot occur.

Additional strengths worth noting: parsing overflow is guarded in `parse_number`, the `k` addition checks against `LLONG_MAX`/`LLONG_MIN` before adding, `isdigit` receives an `unsigned char` cast avoiding UB with negative `char` values, every error path frees `out`, and all I/O return values are checked in `main`.
