```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_LEN 256U

static bool contains_all_vowels(const char *str, size_t max_size)
{
    bool found[5] = { false, false, false, false, false };
    size_t i;
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_size);

    for (i = 0U; i < len; i++) {
        int c = tolower((unsigned char)str[i]);

        switch (c) {
        case 'a':
            found[0] = true;
            break;
        case 'e':
            found[1] = true;
            break;
        case 'i':
            found[2] = true;
            break;
        case 'o':
            found[3] = true;
            break;
        case 'u':
            found[4] = true;
            break;
        default:
            break;
        }
    }

    return found[0] && found[1] && found[2] && found[3] && found[4];
}

int main(void)
{
    char input[MAX_LEN];
    size_t len;

    (void)memset(input, 0, sizeof(input));

    printf("Enter a string: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if ((len > 0U) && (input[len - 1U] == '\n')) {
        input[len - 1U] = '\0';
    }

    if (contains_all_vowels(input, sizeof(input))) {
        printf("Accepted: the string contains all vowels\n");
    } else {
        printf("Rejected: the string does not contain all vowels\n");
    }

    return 0;
}
```

**Summary of changes made:**

1. **CWE-126 (`strlen` over-read):** Replaced both `strlen` calls with `strnlen`, which takes an explicit maximum size and cannot read past the buffer even if the string is not null-terminated. The `contains_all_vowels` function signature was extended to accept the buffer size so the bound can be enforced inside the function.

2. **CWE-119/CWE-120 (statically-sized `char` array):** The buffer is now zero-initialized with `memset` before use, and an explicit null terminator is forced at `input[sizeof(input) - 1]` after `fgets`. Combined with `fgets` already limiting the read to `sizeof(input)`, this guarantees the array is always properly bounded and terminated.

3. Minor hardening: `contains_all_vowels` is declared `static` to limit linkage scope, and the existing `NULL` check and `unsigned char` cast for `tolower` (which prevents undefined behavior on negative `char` values) were preserved.
