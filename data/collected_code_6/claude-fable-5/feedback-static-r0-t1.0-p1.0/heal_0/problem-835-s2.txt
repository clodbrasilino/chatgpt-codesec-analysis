```c
#include <stdio.h>
#include <math.h>
#include <stdlib.h>

int compute_slope(double x1, double y1, double x2, double y2, double *slope)
{
    double dx = x2 - x1;

    if (slope == NULL) {
        return -1;
    }

    if (fabs(dx) < 1e-12) {
        return -1;
    }

    *slope = (y2 - y1) / dx;
    return 0;
}

int main(void)
{
    double x1;
    double y1;
    double x2;
    double y2;
    double slope;

    printf("Enter x1 y1: ");
    if (scanf("%lf %lf", &x1, &y1) != 2) {
        fprintf(stderr, "Invalid input for first point\n");
        return EXIT_FAILURE;
    }

    printf("Enter x2 y2: ");
    if (scanf("%lf %lf", &x2, &y2) != 2) {
        fprintf(stderr, "Invalid input for second point\n");
        return EXIT_FAILURE;
    }

    if (compute_slope(x1, y1, x2, y2, &slope) != 0) {
        fprintf(stderr, "Slope is undefined (vertical line)\n");
        return EXIT_FAILURE;
    }

    printf("Slope: %f\n", slope);
    return EXIT_SUCCESS;
}
```
