#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256
#define MAX_TEXT_LEN 65536

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(errcode, regex, errbuf, sizeof(errbuf) - 1);
    errbuf[sizeof(errbuf) - 1] = '\0';
    fprintf(stderr, "%s: %s\n", prefix, errbuf);
}

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t remaining;
    int ret;

    if (text == NULL) {
        return -1;
    }

    remaining = strnlen(text, MAX_TEXT_LEN);
    if (remaining >= MAX_TEXT_LEN) {
        fprintf(stderr, "input text too long or not null-terminated\n");
        return -1;
    }

    ret = regcomp(&regex, "[A-Za-z]{4,}", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "regcomp failed");
        return -1;
    }

    cursor = text;

    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            break;
        }

        if ((size_t)match.rm_eo > remaining) {
            break;
        }

        size_t length = (size_t)(match.rm_eo - match.rm_so);

        if ((size_t)match.rm_so > remaining || length > remaining - (size_t)match.rm_so) {
            break;
        }

        char *word = malloc(length + 1);
        if (word == NULL) {
            fprintf(stderr, "malloc failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(word, cursor + match.rm_so, length);
        word[length] = '\0';
        printf("%s\n", word);
        free(word);

        cursor += match.rm_eo;
        remaining -= (size_t)match.rm_eo;

        if (remaining == 0 || *cursor == '\0') {
            break;
        }
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "regexec failed");
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog";

    if (find_long_words(text) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Analysis of the original code and the fixes applied:

**Weakness 1 — `errbuf` statically-sized array (CWE-119/CWE-120):** The original code already handled this correctly. The buffer is zeroed with `memset`, `regerror` is called with `sizeof(errbuf) - 1` so it can never write past the buffer, and the final byte is explicitly null-terminated. This is a defensive triple-guarantee, so the Flawfinder hit is a false positive; the pattern is retained as-is.

**Weakness 2 — `memcpy` without bounds check (CWE-120):** The destination `word` is allocated with exactly `length + 1` bytes, and `length` is derived from validated match offsets. The rewrite strengthens the source-side validation by additionally checking `(size_t)match.rm_so > remaining` before computing `remaining - (size_t)match.rm_so`, eliminating any possibility of unsigned underflow in that subtraction. Combined with the existing checks (`rm_so >= 0`, `rm_eo > rm_so`, `rm_eo <= remaining`), both the source read range and destination write range are fully bounded, so the `memcpy` is safe.

**Other sound practices already present and preserved:**
- `strnlen` with `MAX_TEXT_LEN` guards against unterminated input.
- `malloc` return value is checked, and `regfree` is called on every exit path after successful `regcomp`, preventing resource leaks.
- `regexec` errors other than `REG_NOMATCH` are reported via `regerror` rather than ignored.
- `remaining` is decremented in lockstep with the cursor advance, keeping the bounds invariant valid across loop iterations.
