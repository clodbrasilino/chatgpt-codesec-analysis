#include <stdio.h>
#include <stddef.h>

int swap_first_last(int *arr, size_t n)
{
    int temp;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n == 1) {
        return 0;
    }

    temp = arr[0];
    arr[0] = arr[n - 1];
    arr[n - 1] = temp;

    return 0;
}

int main(void)
{
    int list[] = {12, 35, 9, 56, 24};
    size_t n = sizeof(list) / sizeof(list[0]);
    size_t i;

    if (swap_first_last(list, n) != 0) {
        fprintf(stderr, "Error: invalid list\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", list[i]) < 0) {
            return 1;
        }
    }

    if (printf("\n") < 0) {
        return 1;
    }

    return 0;
}
