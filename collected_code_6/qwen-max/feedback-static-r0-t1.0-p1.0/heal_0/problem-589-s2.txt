```c
#include <stdio.h>
#include <math.h>

void findPerfectSquares(int start, int end) {
    for (int i = start; i <= end; ++i) {
        int sqrt_i = (int)sqrt(i);
        if (sqrt_i * sqrt_i == i) {
            printf("%d\n", i);
        }
    }
}

int main() {
    int start = 1;
    int end = 100;
    findPerfectSquares(start, end);
    return 0;
}
```
