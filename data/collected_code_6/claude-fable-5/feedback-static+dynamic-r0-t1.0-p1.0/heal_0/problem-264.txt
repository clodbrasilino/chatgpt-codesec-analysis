```c
#include <stdio.h>
#include <stdlib.h>

double calculate_dog_years(double human_years)
{
    double dog_years;

    if (human_years < 0.0) {
        return -1.0;
    }

    if (human_years <= 2.0) {
        dog_years = human_years * 10.5;
    } else {
        dog_years = (2.0 * 10.5) + ((human_years - 2.0) * 4.0);
    }

    return dog_years;
}

int main(void)
{
    double human_years = 0.0;
    double dog_years = 0.0;
    int result;

    printf("Enter the dog's age in human years: ");

    result = scanf("%lf", &human_years);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (human_years < 0.0 || human_years > 100.0) {
        fprintf(stderr, "Error: age must be between 0 and 100.\n");
        return EXIT_FAILURE;
    }

    dog_years = calculate_dog_years(human_years);
    if (dog_years < 0.0) {
        fprintf(stderr, "Error: calculation failed.\n");
        return EXIT_FAILURE;
    }

    printf("The dog's age in dog years is: %.2f\n", dog_years);

    return EXIT_SUCCESS;
}
```

This implementation uses the common veterinary approximation where the first two years count as 10.5 dog years each, and every year after that counts as 4 dog years. Key safety measures include:

- **CWE-703 avoidance:** The return value of `scanf` is checked before using the input, and the calculation function's result is validated before printing.
- **CWE-664 avoidance:** No dynamically allocated resources or file handles are used, so there are no lifetime management issues; all variables are initialized before use.
- **CWE-710 avoidance:** The code uses `int main(void)`, `EXIT_SUCCESS`/`EXIT_FAILURE` return codes, explicit input range validation, and consistent error reporting to `stderr`.
