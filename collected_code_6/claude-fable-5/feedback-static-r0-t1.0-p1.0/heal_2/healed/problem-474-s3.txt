```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

size_t replace_char(char *str, size_t max_len, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    static const char original[] = "hello world, hello everyone";
    char *buffer;
    size_t len;
    size_t buffer_size;
    size_t replaced;

    len = strnlen(original, sizeof(original));
    if (len >= sizeof(original)) {
        fprintf(stderr, "Source string is not properly terminated\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    if (buffer_size < len) {
        fprintf(stderr, "Buffer size calculation overflow\n");
        return EXIT_FAILURE;
    }

    buffer = malloc(buffer_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (len >= buffer_size) {
        fprintf(stderr, "Destination buffer too small\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    memcpy(buffer, original, len);
    buffer[len] = '\0';

    printf("Before: %s\n", buffer);

    replaced = replace_char(buffer, buffer_size, 'l', 'L');

    printf("After:  %s\n", buffer);
    printf("Replaced %zu occurrence(s)\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was actually already safe regarding the flagged `memcpy` call — the destination buffer was allocated with `buffer_size = len + 1`, guaranteeing sufficient space for `len` bytes plus the null terminator. Flawfinder flags `memcpy` heuristically (CWE-120) without performing data-flow analysis, so this was a false positive in practice.

However, to fully satisfy the concern and harden the code, the rewrite adds:

1. **Integer overflow guard** on `buffer_size = len + 1` — if `len` were ever `SIZE_MAX`, the addition would wrap to `0`, causing a zero-byte allocation followed by an out-of-bounds write. While impossible here given the static array, this makes the pattern safe if the code is refactored to take arbitrary input.

2. **Explicit destination capacity check** (`len >= buffer_size`) immediately before `memcpy` — this documents and enforces the invariant that the destination can hold the source data, directly addressing the CWE-120 warning. Note the `free(buffer)` in this error path to avoid a memory leak.

Other existing good practices retained: `strnlen` with a bound instead of `strlen`, NULL check in `replace_char`, bounded iteration in the replacement loop, checking `malloc`'s return value, and nulling the pointer after `free` to prevent use-after-free/double-free mistakes.
