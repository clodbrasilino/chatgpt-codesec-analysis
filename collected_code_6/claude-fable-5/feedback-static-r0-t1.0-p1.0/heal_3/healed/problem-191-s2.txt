#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool normalize_month(const char *month, char *out, size_t out_size)
{
    size_t len;
    size_t i;

    if (month == NULL || out == NULL || out_size == 0) {
        return false;
    }

    len = strnlen(month, out_size);
    if (len == 0 || len >= out_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        out[i] = (char)tolower((unsigned char)month[i]);
    }
    out[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strcmp(normalized, all_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %s\n", input);
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and why the code is safe as written:**

1. **`char normalized[MAX_INPUT_LEN]` in `has_thirty_days` / `is_valid_month` (CWE-119/CWE-120):** These are false positives. `normalize_month` receives `sizeof(normalized)` as `out_size`, uses `strnlen` bounded by that size, rejects any string with `len >= out_size`, and only then writes `len` characters plus a NUL terminator. No overflow is possible.

2. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120):** Also a false positive. Input is read with `fgets(input, (int)sizeof(input), stdin)`, which limits the write to the buffer size including the terminator. The explicit `input[sizeof(input) - 1] = '\0'` is defensive redundancy.

3. **`getchar()` loop (CWE-120/CWE-20):** The loop discards excess characters from `stdin` when the input line exceeded the buffer; it writes nothing to any buffer, so there is no overflow risk. It correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop on closed streams.

Additional good practices already present: `tolower` is called with an `unsigned char` cast (avoids UB with negative `char` values), `strnlen` is used instead of `strlen`, all pointer parameters are NULL-checked, and `fflush`/`fgets` return values are verified. The code required no functional changes—only removal of the annotation comments.
