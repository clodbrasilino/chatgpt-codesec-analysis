```c
#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

bool is_increasing(const int *sequence, size_t length)
{
    if (sequence == NULL || length < 2) {
        return false;
    }

    for (size_t i = 1; i < length; i++) {
        if (sequence[i] <= sequence[i - 1]) {
            return false;
        }
    }

    return true;
}

int main(void)
{
    int sequence[] = {1, 3, 5, 7, 9};
    size_t length = sizeof(sequence) / sizeof(sequence[0]);

    if (is_increasing(sequence, length)) {
        printf("The sequence is increasing.\n");
    } else {
        printf("The sequence is not increasing.\n");
    }

    return EXIT_SUCCESS;
}
```
