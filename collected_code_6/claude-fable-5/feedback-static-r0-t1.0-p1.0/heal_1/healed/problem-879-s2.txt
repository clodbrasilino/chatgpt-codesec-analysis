```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

int match_pattern(const char *text)
{
    regex_t regex;
    int ret;

    ret = regcomp(&regex, "^a.*b$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        size_t errbuf_size = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errbuf_size);
        if (errbuf == NULL) {
            fprintf(stderr, "Failed to allocate memory for error message\n");
            regfree(&regex);
            return -1;
        }
        regerror(ret, &regex, errbuf, errbuf_size);
        fprintf(stderr, "Failed to compile regex: %s\n", errbuf);
        free(errbuf);
        return -1;
    }

    ret = regexec(&regex, text, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex match failed\n");
        return -1;
    }
}

int main(void)
{
    const char *tests[] = {
        "ab",
        "acb",
        "a123b",
        "abc",
        "b",
        "a"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = match_pattern(tests[i]);
        if (result < 0) {
            return EXIT_FAILURE;
        }
        printf("\"%s\" -> %s\n", tests[i], result ? "match" : "no match");
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder warning about the statically-sized buffer: instead of using a fixed `char errbuf[256]`, the code now calls `regerror` with a `NULL` buffer and size `0` to determine the exact required buffer size, then dynamically allocates precisely that amount. This guarantees the error message can never be truncated or overflow, regardless of its length. The allocation is checked for failure, the buffer is freed after use, and `regfree` is called on the compiled regex in the error path to avoid a resource leak.
