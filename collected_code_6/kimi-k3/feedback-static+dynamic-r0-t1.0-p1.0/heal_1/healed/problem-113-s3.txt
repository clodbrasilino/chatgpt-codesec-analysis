#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <limits.h>

int is_integer(const char *str) {
    if (str == NULL || *str == '\0') {
        return 0;
    }

    if (*str == '+' || *str == '-') {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    while (*str) {
        if (!isdigit((unsigned char)*str)) {
            return 0;
        }
        str++;
    }

    return 1;
}

int main(void) {
    char *input = NULL;
    size_t buffer_size = 0;
    ssize_t length;

    printf("Enter a string: ");
    
    length = getline(&input, &buffer_size, stdin);
    
    if (length == -1) {
        fprintf(stderr, "Error reading input\n");
        free(input);
        return 1;
    }

    if (length > 0 && input[length - 1] == '\n') {
        input[length - 1] = '\0';
    }

    if (is_integer(input)) {
        printf("The string represents an integer.\n");
    } else {
        printf("The string does not represent an integer.\n");
    }

    free(input);
    return 0;
}
