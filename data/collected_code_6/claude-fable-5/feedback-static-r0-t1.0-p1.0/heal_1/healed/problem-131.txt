```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_STR_LEN 4096U

static bool is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

static int reverse_vowels(char *str, size_t max_len)
{
    size_t left;
    size_t right;
    size_t len;
    char temp;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        return -1;
    }
    if (len < 2U) {
        return 0;
    }

    left = 0U;
    right = len - 1U;

    while (left < right) {
        while ((left < right) && (!is_vowel(str[left]))) {
            left++;
        }
        while ((left < right) && (!is_vowel(str[right]))) {
            right--;
        }
        if (left < right) {
            temp = str[left];
            str[left] = str[right];
            str[right] = temp;
            left++;
            right--;
        }
    }

    return 0;
}

int main(void)
{
    char input1[] = "hello world";
    char input2[] = "Programming";
    char input3[] = "xyz";

    if (reverse_vowels(input1, sizeof(input1)) != 0) {
        fprintf(stderr, "Error reversing vowels\n");
        return EXIT_FAILURE;
    }
    if (printf("%s\n", input1) < 0) {
        return EXIT_FAILURE;
    }

    if (reverse_vowels(input2, sizeof(input2)) != 0) {
        fprintf(stderr, "Error reversing vowels\n");
        return EXIT_FAILURE;
    }
    if (printf("%s\n", input2) < 0) {
        return EXIT_FAILURE;
    }

    if (reverse_vowels(input3, sizeof(input3)) != 0) {
        fprintf(stderr, "Error reversing vowels\n");
        return EXIT_FAILURE;
    }
    if (printf("%s\n", input3) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already quite defensive (null checks, `unsigned char` cast for `tolower`, checked `printf` return values), but the flagged weakness was the unbounded `strlen` call (CWE-126), which can over-read if the input is not NUL-terminated.

**Key changes made:**

1. **Replaced `strlen` with `strnlen`** — the function now takes an explicit `max_len` bound, so it never reads past the caller-provided buffer size, even if the string lacks a NUL terminator.
2. **Added a termination check** — if `strnlen` returns `max_len`, the string was not properly terminated within the buffer, and the function returns an error instead of operating on it.
3. **Callers pass `sizeof(buffer)`** — since the inputs are true arrays (not decayed pointers), `sizeof` gives the exact buffer capacity, making the bound accurate at each call site.

Everything else (the two-pointer vowel-swap algorithm, the `is_vowel` helper, error handling in `main`) was preserved as-is, since it was already correct and safe.
