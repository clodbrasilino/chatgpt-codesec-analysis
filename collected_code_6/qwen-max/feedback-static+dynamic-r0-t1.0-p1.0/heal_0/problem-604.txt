#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void reverse_string(char *str, int start, int end) {
    while (start < end) {
        char temp = str[start];
        str[start] = str[end];
        str[end] = temp;
        start++;
        end--;
    }
}

void reverse_words(char *s) {
    int len = strlen(s);
    int start = 0;
    for (int i = 0; i <= len; i++) {
        if (s[i] == ' ' || s[i] == '\0') {
            reverse_string(s, start, i - 1);
            start = i + 1;
        }
    }
    reverse_string(s, 0, len - 1);
}

int main() {
    char s[] = "Hello World This is C";
    reverse_words(s);
    printf("%s\n", s);
    return 0;
}
