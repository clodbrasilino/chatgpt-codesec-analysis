```c
#include <stdbool.h>
#include <string.h>
#include <stdio.h>

bool areCharsSame(char* str) {
    int i;
    int len = strlen(str);

    if(len == 0) {
        return true;
    }

    for(i = 1; i < len; i++) {
        if(str[i] != str[0]) {
            return false;
        }
    }
    return true;
}

int main() {
    char str[] = "aaaaaaa";

    bool result = areCharsSame(str);
    if(result) {
        printf("All characters are same.\n");
    } else {
        printf("All characters are not same.\n");
    }

    return 0;
}
```
