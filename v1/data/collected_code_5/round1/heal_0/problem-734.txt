```C
#include <stdio.h>

long long product(int arr[], int n)
{
   long long prod = 1;
   for (int i = 0; i < n; i++)
   {
       prod *= arr[i];
   }
   return prod;
}

long long sumOfProducts(int arr[], int n)
{
   long long sum = 0;
   for (int i = 0; i < n; i++)
   {
       for (int j = i; j < n; j++)
       {
           sum += product(arr+i, j-i+1);
       }
   }
   return sum;
}

int main()
{
   int arr[] = {1, 2, 3, 4};
   int n = sizeof(arr) / sizeof(arr[0]);
   printf("%lld", sumOfProducts(arr, n));
   return 0;
}
```
