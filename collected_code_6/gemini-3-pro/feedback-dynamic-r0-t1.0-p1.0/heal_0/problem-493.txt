#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

typedef struct {
    double x;
    double y;
} Point;

typedef struct {
    Point points[6];
} Hexagon;

Hexagon** calculate_hexagon_grid(int rows, int cols, double size, double start_x, double start_y) {
    if (rows <= 0 || cols <= 0 || size <= 0.0) {
        return NULL;
    }

    Hexagon** grid = (Hexagon**)malloc(rows * sizeof(Hexagon*));
    if (!grid) {
        return NULL;
    }

    for (int r = 0; r < rows; r++) {
        grid[r] = (Hexagon*)malloc(cols * sizeof(Hexagon));
        if (!grid[r]) {
            for (int i = 0; i < r; i++) {
                free(grid[i]);
            }
            free(grid);
            return NULL;
        }

        for (int c = 0; c < cols; c++) {
            double x_offset = start_x + c * size * sqrt(3.0);
            if (r % 2 != 0) {
                x_offset += (size * sqrt(3.0)) / 2.0;
            }
            double y_offset = start_y + r * size * 1.5;

            for (int i = 0; i < 6; i++) {
                double angle_deg = 60.0 * i - 30.0;
                double angle_rad = M_PI / 180.0 * angle_deg;
                grid[r][c].points[i].x = x_offset + size * cos(angle_rad);
                grid[r][c].points[i].y = y_offset + size * sin(angle_rad);
            }
        }
    }

    return grid;
}

void free_hexagon_grid(Hexagon** grid, int rows) {
    if (grid) {
        for (int r = 0; r < rows; r++) {
            free(grid[r]);
        }
        free(grid);
    }
}

int main() {
    int rows = 3;
    int cols = 4;
    double size = 10.0;
    
    Hexagon** grid = calculate_hexagon_grid(rows, cols, size, 0.0, 0.0);
    
    if (grid) {
        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                printf("Hexagon [%d][%d]:\n", r, c);
                for (int i = 0; i < 6; i++) {
                    printf("  Point %d: (%.2f, %.2f)\n", i, grid[r][c].points[i].x, grid[r][c].points[i].y);
                }
            }
        }
        free_hexagon_grid(grid, rows);
    } else {
        printf("Failed to generate hexagon grid.\n");
    }

    return 0;
}
