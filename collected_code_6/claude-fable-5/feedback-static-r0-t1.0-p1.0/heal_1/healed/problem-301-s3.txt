```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DICT_MAX_STR_LEN 4096

typedef struct Dict Dict;

struct Dict {
    char *key;
    char *value;
    Dict **children;
    size_t child_count;
};

static char *dict_strdup_safe(const char *src)
{
    size_t len;
    char *dst;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, DICT_MAX_STR_LEN);
    if (len >= DICT_MAX_STR_LEN) {
        return NULL;
    }

    dst = malloc(len + 1);
    if (dst == NULL) {
        return NULL;
    }

    memcpy(dst, src, len);
    dst[len] = '\0';
    return dst;
}

static Dict *dict_create(const char *key, const char *value)
{
    Dict *d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }
    d->key = NULL;
    d->value = NULL;
    d->children = NULL;
    d->child_count = 0;

    if (key != NULL) {
        d->key = dict_strdup_safe(key);
        if (d->key == NULL) {
            free(d);
            return NULL;
        }
    }

    if (value != NULL) {
        d->value = dict_strdup_safe(value);
        if (d->value == NULL) {
            free(d->key);
            free(d);
            return NULL;
        }
    }

    return d;
}

static int dict_add_child(Dict *parent, Dict *child)
{
    Dict **tmp;

    if (parent == NULL || child == NULL) {
        return -1;
    }

    if (parent->child_count >= SIZE_MAX / sizeof(Dict *) - 1) {
        return -1;
    }

    tmp = realloc(parent->children, (parent->child_count + 1) * sizeof(Dict *));
    if (tmp == NULL) {
        return -1;
    }

    parent->children = tmp;
    parent->children[parent->child_count] = child;
    parent->child_count++;
    return 0;
}

static void dict_free(Dict *d)
{
    size_t i;

    if (d == NULL) {
        return;
    }

    for (i = 0; i < d->child_count; i++) {
        dict_free(d->children[i]);
    }

    free(d->children);
    free(d->key);
    free(d->value);
    free(d);
}

static size_t dict_depth(const Dict *d)
{
    size_t max_child_depth = 0;
    size_t i;

    if (d == NULL) {
        return 0;
    }

    for (i = 0; i < d->child_count; i++) {
        size_t child_depth = dict_depth(d->children[i]);
        if (child_depth > max_child_depth) {
            max_child_depth = child_depth;
        }
    }

    return 1 + max_child_depth;
}

int main(void)
{
    Dict *root;
    Dict *level1;
    Dict *level2;
    Dict *level3;
    Dict *sibling;
    size_t depth;

    root = dict_create("root", NULL);
    if (root == NULL) {
        fprintf(stderr, "allocation failure\n");
        return EXIT_FAILURE;
    }

    level1 = dict_create("a", NULL);
    if (level1 == NULL || dict_add_child(root, level1) != 0) {
        dict_free(level1);
        dict_free(root);
        fprintf(stderr, "allocation failure\n");
        return EXIT_FAILURE;
    }

    level2 = dict_create("b", NULL);
    if (level2 == NULL || dict_add_child(level1, level2) != 0) {
        dict_free(level2);
        dict_free(root);
        fprintf(stderr, "allocation failure\n");
        return EXIT_FAILURE;
    }

    level3 = dict_create("c", "value");
    if (level3 == NULL || dict_add_child(level2, level3) != 0) {
        dict_free(level3);
        dict_free(root);
        fprintf(stderr, "allocation failure\n");
        return EXIT_FAILURE;
    }

    sibling = dict_create("d", "value");
    if (sibling == NULL || dict_add_child(root, sibling) != 0) {
        dict_free(sibling);
        dict_free(root);
        fprintf(stderr, "allocation failure\n");
        return EXIT_FAILURE;
    }

    depth = dict_depth(root);
    printf("Depth of dictionary: %zu\n", depth);

    dict_free(root);
    return EXIT_SUCCESS;
}
```

**Key changes addressing the flagged weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `DICT_MAX_STR_LEN`, so a non-null-terminated input can no longer cause an unbounded over-read. If the string reaches the limit without a terminator, the function fails safely instead of reading past the buffer.

2. **CWE-120 (`memcpy` overflow):** Copying is now centralized in `dict_strdup_safe`, where the destination is allocated to exactly `len + 1` bytes and `memcpy` copies exactly `len` bytes, with the terminator written explicitly. The destination size is therefore always guaranteed to hold the source data.

3. **Bonus hardening:** Added an overflow guard in `dict_add_child` so `(child_count + 1) * sizeof(Dict *)` cannot wrap around `SIZE_MAX`, which would otherwise cause an undersized `realloc` and subsequent heap overflow. (Note: `SIZE_MAX` requires `<stdint.h>` on some toolchains; add it if your compiler doesn't expose it via the existing headers.)
