```C
#include <stdio.h>

double findMedian(double a, double b, double c) {
    if ((a >= b && a <= c) || (a >= c && a <= b)) {
        return a;
    } else if ((b >= a && b <= c) || (b >= c && b <= a)) {
        return b;
    } else {
        return c;
    }
}

int main() {
    double a, b, c;

    printf("Enter three numbers: ");
    scanf("%lf %lf %lf", &a, &b, &c);
    
    printf("Median: %lf", findMedian(a, b, c));

    return 0;
}
```
