```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LEN 4096U
#define INT_BUFFER_LEN 32U

static char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0U) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        free(list[i]);
        list[i] = NULL;
    }
    free(list);
}

static char **tuple_to_list_with_string(const int *tuple, size_t tuple_len,
                                        const char *insert, size_t *out_len)
{
    char **result;
    size_t total;
    size_t i;
    size_t idx;

    if (tuple == NULL || insert == NULL || out_len == NULL || tuple_len == 0U) {
        return NULL;
    }
    if (tuple_len > (SIZE_MAX / sizeof(char *)) / 2U) {
        return NULL;
    }

    total = tuple_len * 2U;
    result = calloc(total, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    idx = 0U;
    for (i = 0U; i < tuple_len; i++) {
        char buffer[INT_BUFFER_LEN] = {0};
        int written;

        written = snprintf(buffer, sizeof(buffer), "%d", tuple[i]);
        if (written < 0 || (size_t)written >= sizeof(buffer)) {
            free_string_list(result, idx);
            return NULL;
        }
        buffer[sizeof(buffer) - 1U] = '\0';

        result[idx] = duplicate_string(buffer);
        if (result[idx] == NULL) {
            free_string_list(result, idx);
            return NULL;
        }
        idx++;

        result[idx] = duplicate_string(insert);
        if (result[idx] == NULL) {
            free_string_list(result, idx);
            return NULL;
        }
        idx++;
    }

    *out_len = total;
    return result;
}

int main(void)
{
    const int tuple[] = {5, 6, 7, 4, 9};
    const size_t tuple_len = sizeof(tuple) / sizeof(tuple[0]);
    const char *insert = "FDF";
    char **list;
    size_t list_len;
    size_t i;

    list_len = 0U;
    list = tuple_to_list_with_string(tuple, tuple_len, insert, &list_len);
    if (list == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    printf("[");
    for (i = 0U; i < list_len; i++) {
        printf("'%s'", list[i]);
        if (i + 1U < list_len) {
            printf(", ");
        }
    }
    printf("]\n");

    free_string_list(list, list_len);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **`memcpy` in `duplicate_string` (CWE-120):** This is a false positive in practice. The destination buffer is allocated with `malloc(len + 1U)` immediately before the copy of exactly `len` bytes, so the destination is guaranteed to hold the source data. The length is also bounded by `strnlen(src, MAX_STRING_LEN)` with an explicit rejection of strings at or beyond the cap, preventing unbounded reads on non-terminated input.

2. **Statically-sized `char buffer[INT_BUFFER_LEN]` (CWE-119/CWE-120):** Also mitigated. The buffer is zero-initialized, written only via `snprintf` with `sizeof(buffer)` as the bound, the return value is checked for both encoding errors (`written < 0`) and truncation (`>= sizeof(buffer)`), and a defensive terminating NUL is enforced. 32 bytes comfortably holds any `int` representation (max ~11 characters plus sign and NUL, even for 64-bit `int`).

Additional robustness already present and preserved: overflow guard on `tuple_len * 2U * sizeof(char *)` before `calloc`, NULL-checks on all parameters and allocations, and consistent cleanup via `free_string_list` on every failure path, so no memory leaks or double-frees occur. The code was essentially correct as written; the rewrite retains those mitigations with the comments removed.
