```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char **split_at_lowercase(const char *input, size_t *count);
void free_tokens(char **tokens, size_t count);

char **split_at_lowercase(const char *input, size_t *count)
{
    char **tokens = NULL;
    char **temp = NULL;
    size_t num_tokens = 0;
    size_t start = 0;
    size_t i = 0;
    size_t len = 0;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;
    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || islower((unsigned char)input[i])) {
            if (i > start) {
                size_t seg_len = i - start;
                char *segment = malloc(seg_len + 1);
                if (segment == NULL) {
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }
                memcpy(segment, &input[start], seg_len);
                segment[seg_len] = '\0';

                temp = realloc(tokens, (num_tokens + 1) * sizeof(char *));
                if (temp == NULL) {
                    free(segment);
                    free_tokens(tokens, num_tokens);
                    return NULL;
                }
                tokens = temp;
                tokens[num_tokens] = segment;
                num_tokens++;
            }
            start = i + 1;
        }
    }

    *count = num_tokens;
    return tokens;
}

void free_tokens(char **tokens, size_t count)
{
    size_t i;

    if (tokens == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(tokens[i]);
    }
    free(tokens);
}

int main(void)
{
    const char test[] = "HELLOworldFOObarBAZ123";
    char **tokens = NULL;
    size_t count = 0;
    size_t i;

    tokens = split_at_lowercase(test, &count);
    if (tokens == NULL) {
        fprintf(stderr, "Failed to split string or no tokens found\n");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", test);
    printf("Tokens: %zu\n", count);
    for (i = 0; i < count; i++) {
        printf("  [%zu]: %s\n", i, tokens[i]);
    }

    free_tokens(tokens, count);
    return EXIT_SUCCESS;
}
```

**Summary of fixes applied:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, rejecting inputs that are not properly terminated within the limit. The test string in `main` is also declared as a fixed array (`const char test[]`), guaranteeing null-termination at compile time.

2. **CWE-120 (`memcpy` overflow):** The destination buffer is allocated as `seg_len + 1` immediately before the copy of exactly `seg_len` bytes, so the destination is provably large enough; the bound now derives from a validated length.

3. **`knownConditionTrueFalse` (always-true condition):** Removed the redundant `strlen(test) > 0` check entirely. The error path now simply reports failure and exits with `EXIT_FAILURE`, which also eliminates the second flagged `strlen` call.

4. **Ambiguous failure signaling:** A `NULL` return is now treated uniformly as an error in `main`, avoiding the previous conflation of "allocation failure" with "no tokens found" followed by continued use of a potentially invalid state.
