#include <stdio.h>
#include <stdlib.h>

double findAverage(int **data, int row, int col) {
    double sum = 0.0;
    int count = 0;

    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            sum += data[i][j];
            count++;
        }
    }

    if (count == 0) return 0.0;
    return sum / count;
}

int main() {
    int row = 3;
    int col = 2;
    int data[3][2] = {
        {1, 2},
        {3, 4},
        {5, 6}
    };
    int *pData[3];
    for (int i = 0; i < row; i++) {
        pData[i] = data[i];
    }
    
    double average = findAverage((int **)pData, row, col);
    printf("Average: %f\n", average);
    
    return 0;
}
