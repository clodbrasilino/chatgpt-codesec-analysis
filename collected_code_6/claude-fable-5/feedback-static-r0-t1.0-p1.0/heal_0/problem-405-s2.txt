```c
#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

bool element_exists(const int *tuple, size_t size, int target)
{
    if (tuple == NULL || size == 0U) {
        return false;
    }

    for (size_t i = 0U; i < size; i++) {
        if (tuple[i] == target) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    const int tuple[] = { 10, 20, 30, 40, 50 };
    const size_t size = sizeof(tuple) / sizeof(tuple[0]);
    const int targets[] = { 30, 99 };
    const size_t num_targets = sizeof(targets) / sizeof(targets[0]);

    for (size_t i = 0U; i < num_targets; i++) {
        if (element_exists(tuple, size, targets[i])) {
            if (printf("%d exists in the tuple\n", targets[i]) < 0) {
                return 1;
            }
        } else {
            if (printf("%d does not exist in the tuple\n", targets[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```
