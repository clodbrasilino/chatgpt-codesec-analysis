bool is_palindrome(long num) {

    snprintf(str, sizeof(str), "%ld", num);
    size_t len = strlen(str);
    for (size_t i = 0; i < len / 2; ++i) {

        }
    }
    return true;
}

long next_palindrome(long num) {
    while (1) {
        ++num;
        if (is_palindrome(num)) {
            return num;
        }
    }
}

int main() {

    printf("%ld\n", next_palindrome(num));
