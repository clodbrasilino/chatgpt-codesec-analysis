```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

struct Point {
    double x;
    double y;
};

int isValidTriangle(struct Point p1, struct Point p2, struct Point p3) {
    double area = 0.5 * fabs(p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y));
    
    if (area > 1e-9) {
        return 1;
    }
    return 0;
}

int main(void) {
    struct Point p1, p2, p3;
    
    if (scanf("%lf %lf", &p1.x, &p1.y) != 2) {
        return EXIT_FAILURE;
    }
    if (scanf("%lf %lf", &p2.x, &p2.y) != 2) {
        return EXIT_FAILURE;
    }
    if (scanf("%lf %lf", &p3.x, &p3.y) != 2) {
        return EXIT_FAILURE;
    }
    
    if (isValidTriangle(p1, p2, p3)) {
        printf("Valid Triangle\n");
    } else {
        printf("Invalid Triangle\n");
    }
    
    return EXIT_SUCCESS;
}
```
