#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)
#define LINE_SIZE 64

int decimal_to_binary(unsigned int decimal, char *buffer, size_t buffer_size)
{
    size_t index = 0;
    size_t i;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (decimal > 0 && index < buffer_size - 1) {
        buffer[index] = (char)('0' + (decimal % 2U));
        decimal /= 2U;
        index++;
    }

    if (decimal > 0) {
        buffer[0] = '\0';
        return -1;
    }

    buffer[index] = '\0';

    for (i = 0; i < index / 2; i++) {
        char temp = buffer[i];
        buffer[i] = buffer[index - 1 - i];
        buffer[index - 1 - i] = temp;
    }

    return 0;
}

int main(void)
{
    unsigned int number = 0;
    char binary[MAX_BITS + 1] = {0};
    char line[LINE_SIZE] = {0};
    char *endptr = NULL;
    unsigned long parsed;
    size_t len;

    printf("Enter a non-negative decimal number: ");
    fflush(stdout);

    if (fgets(line, sizeof(line), stdin) == NULL) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    line[sizeof(line) - 1] = '\0';
    len = strnlen(line, sizeof(line));

    if (len == sizeof(line) - 1 && line[len - 1] != '\n') {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        fprintf(stderr, "Error: input too long.\n");
        return EXIT_FAILURE;
    }

    if (len > 0 && line[len - 1] == '\n') {
        line[len - 1] = '\0';
        len--;
    }

    if (len == 0 || line[0] == '\0') {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (line[0] == '-') {
        fprintf(stderr, "Error: number must be non-negative.\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    parsed = strtoul(line, &endptr, 10);

    if (endptr == line || endptr == NULL || *endptr != '\0') {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (errno == ERANGE || parsed > UINT_MAX) {
        fprintf(stderr, "Error: number out of range.\n");
        return EXIT_FAILURE;
    }

    number = (unsigned int)parsed;

    if (decimal_to_binary(number, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal: %u\n", number);
    printf("Binary: %s\n", binary);

    return EXIT_SUCCESS;

**Analysis of the identified weaknesses and applied fixes:**

The original code was already quite defensive — the Flawfinder hits are largely low-risk static-analysis heuristics rather than actual vulnerabilities. Here is how each was addressed:

1. **`char binary[MAX_BITS + 1]` (CWE-119/CWE-120):** This is a false positive in practice. The buffer is sized precisely for the worst case (32 bits + NUL on typical platforms), and `decimal_to_binary` receives the size via `sizeof(binary)` and performs explicit bounds checking (`index < buffer_size - 1`) with a failure path if the value doesn't fit. No change needed beyond keeping the size-passing contract.

2. **`char line[64]` (CWE-119/CWE-120):** Safe because `fgets` is called with `sizeof(line)`, which limits the write. Hardened further by explicitly NUL-terminating (`line[sizeof(line)-1] = '\0'`) as a belt-and-braces measure, and by **draining the remaining stdin** when input is too long — the original left the excess in the buffer, which could affect subsequent reads in a larger program.

3. **`strlen` (CWE-126):** After `fgets` succeeds, the buffer is always NUL-terminated, so the original was safe. To fully satisfy the checker, it was replaced with `strnlen(line, sizeof(line))`, which guarantees the read never exceeds the buffer even if termination were somehow violated.

Additional minor hardening: `fflush(stdout)` after the prompt (guaranteed visibility before blocking on input), an explicit `endptr == NULL` guard, and consistent length tracking after stripping the newline.
