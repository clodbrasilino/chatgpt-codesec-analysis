#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;
    char tmp;
    char *endptr;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }
    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }
    if (j <= i) {
        return -1;
    }

    tmp = digits[i];
    digits[i] = digits[j];
    digits[j] = tmp;

    for (int a = i + 1, b = len - 1; a < b; a++, b--) {
        tmp = digits[a];
        digits[a] = digits[b];
        digits[b] = tmp;
    }

    errno = 0;
    result = strtoll(digits, &endptr, 10);
    if (errno == ERANGE || endptr == digits || *endptr != '\0') {
        return -1;
    }

    if (result <= n) {
        return -1;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            if (printf("next_bigger(%lld) = no bigger number exists\n", inputs[k]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("next_bigger(%lld) = %lld\n", inputs[k], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already largely safe, but the rewrite hardens it against the flagged CWE-119/CWE-120 concerns:

1. **Statically-sized buffer (`digits[32]`)** — The buffer is large enough for any `long long` (max 20 characters plus NUL), and the `snprintf` return value is checked against `sizeof(digits)` to guarantee no truncation is silently accepted. Added `memset` to ensure the buffer is fully initialized before use.

2. **Unbounded backward scan** — The original loop `while (digits[j] <= digits[i]) j--;` relied on the algorithm's invariant that a valid `j` exists. The rewrite adds an explicit `j > i` bound check plus a defensive `j <= i` bail-out, so no out-of-bounds read is possible even if the invariant were violated.

3. **Result validation** — Added a sanity check `result <= n` after `strtoll`, guaranteeing the function's contract (returning a strictly bigger permutation) even in unexpected edge cases, alongside the existing `errno`/`endptr` conversion checks.
