#include <stdio.h>
#include <string.h>
#include <stdint.h>

int all_chars_same(const char *str) {
    if (str == NULL || *str == '\0') {
        return 1;
    }

    char first = str[0];
    for (size_t i = 1; str[i] != '\0'; i++) {
        if (str[i] != first) {
            return 0;
        }
    }
    return 1;
}

int main(void) {
    char input[256];
    int c;
    size_t i = 0;

    printf("Enter a string: ");

    while (i < sizeof(input) - 1) {
        c = getchar();
        if (c == EOF) {
            break;
        }
        if (c == '\n') {
            break;
        }
        input[i] = (char)c;
        i++;
    }
    input[i] = '\0';

    if (i == 0 && c == EOF) {
        return 1;
    }

    if (c != '\n' && c != EOF) {
        int discard;
        while ((discard = getchar()) != '\n' && discard != EOF) {
        }
    }

    if (i == 0) {
        printf("Characters are not all the same.\n");
        return 0;
    }

    if (all_chars_same(input)) {
        printf("All characters are the same.\n");
    } else {
        printf("Characters are not all the same.\n");
    }

    return 0;
}
