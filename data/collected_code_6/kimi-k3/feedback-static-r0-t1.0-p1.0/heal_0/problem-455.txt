#include <stdio.h>

int has31Days(int month) {
    return (month == 1 || month == 3 || month == 5 || 
            month == 7 || month == 8 || month == 10 || month == 12);
}

int main(void) {
    int month;
    int result;
    
    if (printf("Enter month number (1-12): ") < 0) {
        return 1;
    }
    
    if (scanf("%d", &month) != 1) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    if (month < 1 || month > 12) {
        fprintf(stderr, "Month must be between 1 and 12\n");
        return 1;
    }
    
    result = has31Days(month);
    
    if (result) {
        printf("Month %d has 31 days\n", month);
    } else {
        printf("Month %d does not have 31 days\n", month);
    }
    
    return 0;
}
