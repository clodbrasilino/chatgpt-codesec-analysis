#include <stdio.h>
#include <stdlib.h>

void merge_count(int arr[], int temp[], int left, int mid, int right, long long *inv_a, long long *inv_b) {
    int i = left;
    int j = mid + 1;
    int k = left;

    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            long long count = (mid - i + 1);
            if (arr[i] < arr[j]) {
                *inv_a += count;
            } else {
                *inv_b += count;
            }
            temp[k++] = arr[j++];
        }
    }

    while (i <= mid) {
        temp[k++] = arr[i++];
    }

    while (j <= right) {
        temp[k++] = arr[j++];
    }

    for (i = left; i <= right; i++) {
        arr[i] = temp[i];
    }
}

void merge_sort_count(int arr[], int temp[], int left, int right, long long *inv_a, long long *inv_b) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        merge_sort_count(arr, temp, left, mid, inv_a, inv_b);
        merge_sort_count(arr, temp, mid + 1, right, inv_a, inv_b);
        merge_count(arr, temp, left, mid, right, inv_a, inv_b);
    }
}

int check_inversion_counts(int arr[], int n) {
    if (n <= 1) {
        return 1;
    }

    int *temp = (int *)malloc(n * sizeof(int));
    if (temp == NULL) {
        return -1;
    }

    long long inv_a = 0;
    long long inv_b = 0;

    merge_sort_count(arr, temp, 0, n - 1, &inv_a, &inv_b);

    free(temp);

    return inv_a == inv_b;
}

int main() {
    int arr1[] = {3, 2, 1, 0};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int res1 = check_inversion_counts(arr1, n1);
    printf("%d\n", res1);

    int arr2[] = {4, 2, 3, 1};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int res2 = check_inversion_counts(arr2, n2);
    printf("%d\n", res2);

    return 0;
}
