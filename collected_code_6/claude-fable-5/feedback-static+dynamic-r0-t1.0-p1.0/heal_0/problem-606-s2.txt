```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

double degrees_to_radians(double degrees)
{
    return degrees * (M_PI / 180.0);
}

int main(void)
{
    double degrees;
    double radians;
    int result;

    printf("Enter degrees: ");
    result = scanf("%lf", &degrees);

    if (result != 1) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (!isfinite(degrees)) {
        fprintf(stderr, "Error: input is not a finite number\n");
        return EXIT_FAILURE;
    }

    radians = degrees_to_radians(degrees);

    printf("%f degrees = %f radians\n", degrees, radians);

    return EXIT_SUCCESS;
}
```
