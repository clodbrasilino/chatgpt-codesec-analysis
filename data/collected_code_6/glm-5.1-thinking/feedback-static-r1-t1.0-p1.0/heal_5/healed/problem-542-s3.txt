#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *replace_with_colon(const char *str, size_t len) {
    if (str == NULL) {
        return NULL;
    }

    regex_t regex;
    int ret = regcomp(&regex, "[ ,.]", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    char *result = malloc(len + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    size_t res_idx = 0;
    const char *start = str;
    const char *end = str + len;
    regmatch_t pmatch[1];

    while (start < end) {
        ret = regexec(&regex, start, 1, pmatch, 0);
        if (ret == REG_NOMATCH) {
            size_t remaining = end - start;
            size_t copy_len = remaining;
            if (res_idx + copy_len > len) {
                copy_len = len - res_idx;
            }
            memcpy(result + res_idx, start, copy_len);
            res_idx += copy_len;
            break;
        }

        if (ret != 0) {
            free(result);
            regfree(&regex);
            return NULL;
        }

        size_t match_start = (size_t)pmatch[0].rm_so;
        size_t match_end = (size_t)pmatch[0].rm_eo;

        if (match_end <= match_start) {
            if (res_idx < len) {
                result[res_idx++] = *start;
            }
            start++;
            continue;
        }

        if (match_start > 0) {
            size_t copy_len = match_start;
            if (res_idx + copy_len > len) {
                copy_len = len - res_idx;
            }
            memcpy(result + res_idx, start, copy_len);
            res_idx += copy_len;
        }

        if (res_idx < len) {
            result[res_idx++] = ':';
        }

        start += match_end;
    }

    result[res_idx] = '\0';

    regfree(&regex);
    return result;
}

int main(void) {
    const char *input = "Hello world, this.is a test.";
    size_t input_len = strlen(input);
    char *output = replace_with_colon(input, input_len);

    if (output != NULL) {
        printf("%s\n", output);
        free(output);
    }

    return 0;
}
