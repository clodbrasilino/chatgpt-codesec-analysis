#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

typedef enum {
    TYPE_INT,
    TYPE_DOUBLE,
    TYPE_STRING,
    TYPE_NONE
} ElementType;

typedef struct {
    ElementType type;
    union {
        int i;
        double d;
        const char *s;
    } value;
} TupleElement;

typedef struct {
    TupleElement *elements;
    size_t size;
} Tuple;

bool has_none_value(const Tuple *tuple)
{
    if (tuple == NULL || tuple->elements == NULL) {
        return false;
    }

    for (size_t idx = 0; idx < tuple->size; idx++) {
        if (tuple->elements[idx].type == TYPE_NONE) {
            return true;
        }
        if (tuple->elements[idx].type == TYPE_STRING &&
            tuple->elements[idx].value.s == NULL) {
            return true;
        }
    }

    return false;
}

int main(void)
{
    TupleElement elems1[3];
    elems1[0].type = TYPE_INT;
    elems1[0].value.i = 10;
    elems1[1].type = TYPE_NONE;
    elems1[2].type = TYPE_STRING;
    elems1[2].value.s = "hello";

    Tuple tuple1;
    tuple1.elements = elems1;
    tuple1.size = 3;

    TupleElement elems2[3];
    elems2[0].type = TYPE_INT;
    elems2[0].value.i = 7;
    elems2[1].type = TYPE_DOUBLE;
    elems2[1].value.d = 3.14;
    elems2[2].type = TYPE_STRING;
    elems2[2].value.s = "world";

    Tuple tuple2;
    tuple2.elements = elems2;
    tuple2.size = 3;

    if (has_none_value(&tuple1)) {
        if (printf("Tuple 1 contains a None value\n") < 0) {
            return 1;
        }
    } else {
        if (printf("Tuple 1 does not contain a None value\n") < 0) {
            return 1;
        }
    }

    if (has_none_value(&tuple2)) {
        if (printf("Tuple 2 contains a None value\n") < 0) {
            return 1;
        }
    } else {
        if (printf("Tuple 2 does not contain a None value\n") < 0) {
            return 1;
        }
    }

    return 0;
}
