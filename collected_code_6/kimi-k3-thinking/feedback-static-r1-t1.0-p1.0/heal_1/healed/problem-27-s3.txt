```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_STRING_LENGTH 1024

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t len = 0;

    if (str == NULL)
    {
        return 0;
    }

    while (len < max_len && str[len] != '\0')
    {
        len++;
    }

    return len;
}

void remove_digits(char *str)
{
    char *read_ptr;
    char *write_ptr;

    if (str == NULL)
    {
        return;
    }

    read_ptr = str;
    write_ptr = str;

    while (*read_ptr != '\0')
    {
        if (!isdigit((unsigned char)*read_ptr))
        {
            *write_ptr = *read_ptr;
            write_ptr++;
        }
        read_ptr++;
    }
    *write_ptr = '\0';
}

int main(void)
{
    const size_t num_strings = 4;
    const char *input_strings[] = {
        "abc123def",
        "h3ll0 w0rld",
        "no digits here",
        "9876543210"
    };
    char **strings = NULL;
    size_t i;
    size_t j;
    size_t len;

    strings = calloc(num_strings, sizeof(char *));
    if (strings == NULL)
    {
        fprintf(stderr, "Error: failed to allocate string array\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_strings; i++)
    {
        len = bounded_strlen(input_strings[i], MAX_STRING_LENGTH);
        strings[i] = malloc(len + 1);
        if (strings[i] == NULL)
        {
            fprintf(stderr, "Error: failed to allocate string %zu\n", i);
            for (j = 0; j < i; j++)
            {
                free(strings[j]);
                strings[j] = NULL;
            }
            free(strings);
            strings = NULL;
            return EXIT_FAILURE;
        }
        memcpy(strings[i], input_strings[i], len);
        strings[i][len] = '\0';
    }

    for (i = 0; i < num_strings; i++)
    {
        remove_digits(strings[i]);
        if (printf("Result %zu: \"%s\"\n", i, strings[i]) < 0)
        {
            fprintf(stderr, "Error: failed to write output\n");
            for (j = 0; j < num_strings; j++)
            {
                free(strings[j]);
                strings[j] = NULL;
            }
            free(strings);
            strings = NULL;
            return EXIT_FAILURE;
        }
    }

    for (i = 0; i < num_strings; i++)
    {
        free(strings[i]);
        strings[i] = NULL;
    }
    free(strings);
    strings = NULL;

    return EXIT_SUCCESS;
}
```

Key fixes applied:

- **`strlen` (CWE-126)**: Replaced with `bounded_strlen()`, which never reads more than `MAX_STRING_LENGTH` bytes, so an unterminated string cannot cause an over-read.
- **`strcpy` (CWE-120)**: Replaced with `memcpy()` using the exact bounded length, followed by explicit null termination — no possibility of a buffer overflow or unterminated destination.
- **Hardening**: `calloc()` is used for the pointer array so all entries start as `NULL`, making partial cleanup paths safe.
