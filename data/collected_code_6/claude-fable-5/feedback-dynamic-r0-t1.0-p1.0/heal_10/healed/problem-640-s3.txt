#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strlen(input);
    if (len == SIZE_MAX) {
        return NULL;
    }

    result = calloc(len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            if (depth < INT_MAX) {
                depth++;
            }
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            if (j < len) {
                result[j] = input[i];
                j++;
            }
        }
    }
    result[j] = '\0';

    return result;
}

static int run_test(const char *input, const char *expected)
{
    char *output;
    int status;

    if (input == NULL || expected == NULL) {
        fprintf(stderr, "Invalid test parameters\n");
        return -1;
    }

    output = remove_parentheses(input);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return -1;
    }

    status = 0;
    if (strcmp(output, expected) != 0) {
        fprintf(stderr, "Test failed: expected '%s', got '%s'\n",
                expected, output);
        status = -1;
    }

    if (printf("%s\n", output) < 0) {
        status = -1;
    }

    free(output);
    output = NULL;
    return status;
}

int main(void)
{
    const char *test1 = "Hello (remove me) World";
    const char *expected1 = "Hello  World";
    const char *test2 = "abc(def(ghi)jkl)mno";
    const char *expected2 = "abcmno";
    const char *test3 = "no parentheses here";
    const char *expected3 = "no parentheses here";

    if (run_test(test1, expected1) != 0) {
        return EXIT_FAILURE;
    }
    if (run_test(test2, expected2) != 0) {
        return EXIT_FAILURE;
    }
    if (run_test(test3, expected3) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

The garbage output described in the comments (`�!H`, `` `!H ``) is the classic signature of printing an **uninitialized heap buffer** — typically caused by allocating with `malloc` without initializing memory, or forgetting the null terminator when no characters are copied into `result`. The fixed version above guards against all of those failure modes:

1. **Zero-initialized allocation**: `calloc(len + 1, sizeof(char))` guarantees the buffer starts fully zeroed, so even if the copy loop writes nothing (e.g., input is entirely inside parentheses), the result is a valid empty string rather than uninitialized bytes.

2. **Explicit null termination**: `result[j] = '\0';` is always executed after the loop, and `j` is bounded by `j < len`, so the write is always within the `len + 1` allocation — no off-by-one overflow.

3. **NULL handling**: Both `remove_parentheses` and `run_test` reject `NULL` inputs, and the allocation failure path is checked before any dereference.

4. **Integer safety**: `depth` is clamped at `INT_MAX` to prevent signed overflow (undefined behavior) on pathological inputs with billions of `'('` characters, and never decremented below zero so unmatched `')'` characters are simply dropped. The `SIZE_MAX` check prevents `len + 1` from wrapping to zero in `calloc`.

5. **No memory leaks**: `output` is freed on every path after use, including test-failure paths, and the pointer is nulled to avoid accidental reuse.
