```c
#include <stdio.h>

int is_binary_string(const char *str) {
    if (str == NULL) {
        return 0;
    }
    
    while (*str != '\0') {
        if (*str != '0' && *str != '1') {
            return 0;
        }
        str++;
    }
    
    return 1;
}

int main(void) {
    const char *test1 = "101010";
    const char *test2 = "10201";
    const char *test3 = "";
    const char *test4 = "111";
    
    printf("%s: %d\n", test1, is_binary_string(test1));
    printf("%s: %d\n", test2, is_binary_string(test2));
    printf("%s: %d\n", test3, is_binary_string(test3));
    printf("%s: %d\n", test4, is_binary_string(test4));
    printf("NULL: %d\n", is_binary_string(NULL));
    
    return 0;
}
```
