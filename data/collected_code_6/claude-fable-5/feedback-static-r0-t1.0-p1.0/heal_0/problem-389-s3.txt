```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>

static int lucas(unsigned int n, unsigned long long *result)
{
    unsigned long long a;
    unsigned long long b;
    unsigned long long next;
    unsigned int i;

    if (result == NULL) {
        return -1;
    }

    a = 2ULL;
    b = 1ULL;

    if (n == 0U) {
        *result = a;
        return 0;
    }

    if (n == 1U) {
        *result = b;
        return 0;
    }

    for (i = 2U; i <= n; i++) {
        if (b > (ULLONG_MAX - a)) {
            return -1;
        }
        next = a + b;
        a = b;
        b = next;
    }

    *result = b;
    return 0;
}

int main(int argc, char *argv[])
{
    unsigned long parsed;
    unsigned long long value;
    char *endptr;

    if (argc != 2) {
        (void)fprintf(stderr, "Usage: %s <n>\n", argv[0]);
        return EXIT_FAILURE;
    }

    if (argv[1][0] == '-') {
        (void)fprintf(stderr, "Error: n must be a non-negative integer\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    endptr = NULL;
    parsed = strtoul(argv[1], &endptr, 10);

    if ((errno != 0) || (endptr == argv[1]) || (endptr == NULL) || (*endptr != '\0')) {
        (void)fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    if (parsed > (unsigned long)UINT_MAX) {
        (void)fprintf(stderr, "Error: input too large\n");
        return EXIT_FAILURE;
    }

    if (lucas((unsigned int)parsed, &value) != 0) {
        (void)fprintf(stderr, "Error: result would overflow\n");
        return EXIT_FAILURE;
    }

    if (printf("%llu\n", value) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
