```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 10

static int findDemlo(const char *str, char *result, size_t result_size)
{
    size_t len;
    size_t i;
    size_t pos;

    if (str == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);

    if (len == 0 || len > 9) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (str[i] != '1') {
            return -1;
        }
    }

    if (result_size < (2 * len)) {
        return -1;
    }

    pos = 0;
    for (i = 1; i <= len; i++) {
        result[pos] = (char)('0' + (int)i);
        pos++;
    }
    for (i = len; i > 1; i--) {
        result[pos] = (char)('0' + (int)(i - 1));
        pos++;
    }
    result[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN + 2];
    char output[2 * MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(output, 0, sizeof(output));

    printf("Enter a repunit number (only 1s, max 9 digits): ");
    fflush(stdout);

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }

    if (len == 0) {
        fprintf(stderr, "Invalid input: empty string\n");
        return EXIT_FAILURE;
    }

    if (len > 9) {
        fprintf(stderr, "Invalid input: must be a repunit of at most 9 ones\n");
        return EXIT_FAILURE;
    }

    if (findDemlo(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid input: must be a repunit of at most 9 ones\n");
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", output);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already quite defensive, but the rewrite addresses the flagged concerns and one latent bug:

1. **CWE-119/CWE-120 (statically-sized buffers):** Both `input` and `output` are used exclusively with length-bounded functions (`fgets`, `strnlen`, `memset`). The `input` buffer is sized `MAX_INPUT_LEN + 2` to accommodate 10 characters, a newline, and the NUL terminator, and `fgets` is bounded by `sizeof(input)`. The `output` buffer size is validated inside `findDemlo` via the `result_size < (2 * len)` check before any write occurs — the maximum write is `2*len - 1` digits plus the NUL at index `2*len - 1`, so the check is exact.

2. **Unsigned underflow hazard (latent bug fixed):** The original descending loop `for (i = len - 1; i >= 1; i--)` used a `size_t`. While it happened to be safe (it never decrements past 1 because the condition stops it), the pattern `len - 1` with `len == 0` would wrap to `SIZE_MAX` — it was only protected by an earlier `len == 0` guard. The rewrite uses `for (i = len; i > 1; i--)` with `i - 1` inside, which is structurally immune to unsigned wraparound regardless of upstream checks.

3. **Defense in depth:** An explicit `len > 9` check was added in `main` before calling `findDemlo`, and `fflush(stdout)` was added after the unterminated prompt so it reliably appears before blocking on `fgets` when stdout is fully buffered (e.g., piped output).

4. **Guaranteed termination:** `input[sizeof(input) - 1] = '\0'` remains as a belt-and-braces measure, though `fgets` already guarantees NUL termination.
