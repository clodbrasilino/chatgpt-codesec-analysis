#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define NEG_INF (-1000000000000000000LL)

int maxProfit(int k, int* prices, int pricesSize) {
    if (prices == NULL || pricesSize <= 1 || k <= 0) {
        return 0;
    }
    if (k >= pricesSize / 2) {
        long long profit = 0;
        for (int i = 1; i < pricesSize; i++) {
            if (prices[i] > prices[i-1]) {
                profit += (long long)(prices[i] - prices[i-1]);
            }
        }
        if (profit > INT_MAX) profit = INT_MAX;
        return (int)profit;
    }
    long long *cash = (long long*)malloc((k+1) * sizeof(long long));
    long long *hold = (long long*)malloc((k+1) * sizeof(long long));
    if (cash == NULL || hold == NULL) {
        if (cash != NULL) free(cash);
        if (hold != NULL) free(hold);
        return 0;
    }
    cash[0] = 0;
    hold[0] = NEG_INF;
    for (int j = 1; j <= k; j++) {
        cash[j] = 0;
        hold[j] = NEG_INF;
    }
    for (int i = 0; i < pricesSize; i++) {
        long long price = (long long)prices[i];
        for (int j = 1; j <= k; j++) {
            if (cash[j-1] - price > hold[j]) {
                hold[j] = cash[j-1] - price;
            }
            if (hold[j] + price > cash[j]) {
                cash[j] = hold[j] + price;
            }
        }
    }
    long long max_profit = 0;
    for (int j = 1; j <= k; j++) {
        if (cash[j] > max_profit) {
            max_profit = cash[j];
        }
    }
    free(cash);
    free(hold);
    if (max_profit > INT_MAX) max_profit = INT_MAX;
    return (int)max_profit;
}

int main() {
