#include <stdio.h>
#include <stdbool.h>

bool isOneRootTwiceTheOther(int a, int b, int c) {
    if (a == 0) {
        return false;
    }
    
    int expected_c = (2 * b * b);
    int actual_c = (9 * a * c);
    
    return expected_c == actual_c;
}

int main(void) {
    int a = 0;
    int b = 0;
    int c = 0;
    
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }
    
    if (isOneRootTwiceTheOther(a, b, c)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }
    
    return 0;
}
