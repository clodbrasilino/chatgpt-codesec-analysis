static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n == 0) {
        return 0;
    }
    if (*len > SIZE_MAX - 1 || n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*buf == NULL || *cap < *len + n + 1) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

Analysis of the flagged weaknesses and why the code is already safe:

**1. `memcpy` in `buf_append` (CWE-120, risk 2):** This is a false positive. The destination buffer is guaranteed to hold the source data because:
- Integer overflow of `*len + n + 1` is prevented by the `SIZE_MAX` checks before any arithmetic.
- The buffer is grown via `realloc` until `*cap >= *len + n + 1`, with overflow protection on the doubling loop (`ncap > SIZE_MAX / 2`).
- A final defensive re-verification (`*cap < *len + n + 1`) runs immediately before the `memcpy`, so the copy can never exceed the allocation.

**2. Statically-sized `char numbuf[32]` (CWE-119/CWE-120, risk 2):** Also a false positive. A `long long` (64-bit) formatted as decimal requires at most 20 characters plus sign and NUL terminator (22 bytes), so 32 bytes is sufficient. More importantly, `snprintf` with `sizeof numbuf` bounds the write, and the return value is checked for both negative results (encoding errors) and truncation (`written >= sizeof numbuf`), aborting safely on either condition.
