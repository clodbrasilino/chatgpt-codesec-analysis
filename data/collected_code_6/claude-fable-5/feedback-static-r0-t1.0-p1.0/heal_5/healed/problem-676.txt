```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

char *remove_non_alphanumeric(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *result;
    size_t input_len;
    size_t result_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = calloc(input_len + 1, sizeof(char));
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "[[:alnum:]]+", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    result_pos = 0;
    cursor = input;

    while (*cursor != '\0') {
        size_t segment_len;
        size_t remaining;
        size_t i;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret != 0) {
            break;
        }
        if (match.rm_so < 0 || match.rm_eo <= match.rm_so) {
            break;
        }

        segment_len = (size_t)(match.rm_eo - match.rm_so);

        if (result_pos >= input_len + 1) {
            break;
        }
        remaining = input_len - result_pos;
        if (segment_len > remaining) {
            segment_len = remaining;
        }
        if (segment_len == 0) {
            break;
        }

        for (i = 0; i < segment_len && result_pos < input_len; i++) {
            result[result_pos] = cursor[(size_t)match.rm_so + i];
            result_pos++;
        }

        cursor += match.rm_eo;
    }

    result[result_pos] = '\0';
    regfree(&regex);
    return result;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello, World! 123",
        "a@b#c$d%e^1&2*3",
        "!!!???...",
        "AlreadyClean123"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        char *cleaned = remove_non_alphanumeric(test_strings[i]);
        if (cleaned == NULL) {
            fprintf(stderr, "Failed to process string: %s\n",
                    test_strings[i]);
            return EXIT_FAILURE;
        }
        printf("Original: %s\nCleaned:  %s\n\n",
               test_strings[i], cleaned);
        free(cleaned);
    }

    return EXIT_SUCCESS;
}
```

The key changes addressing the flagged `memcpy` weakness (CWE-120):

1. **Replaced `memcpy` with an explicitly bounded copy loop** — each byte written is guarded by `result_pos < input_len`, making it structurally impossible to write past the allocated buffer regardless of what the regex match offsets report.

2. **Clamping instead of breaking** — `segment_len` is clamped to the remaining space rather than trusting arithmetic alone, providing defense in depth if `rm_so`/`rm_eo` were ever inconsistent.

3. **`calloc` instead of `malloc`** — the result buffer is zero-initialized, so even if the NUL-termination logic were ever bypassed, the buffer can never be an unterminated string.

4. **Tightened the overflow guard** — the `result_pos` check is performed before computing `remaining`, avoiding any possibility of unsigned underflow in `input_len - result_pos`.

The original code was actually already safe (bounds were checked before `memcpy`), so these changes primarily harden the code against future modification errors and silence static-analysis findings without changing observable behavior.
