#include <stdio.h>
#include <stdlib.h>

long long min_operations(long long a, long long b) {
    long long diff = a > b ? a - b : b - a;
    return (diff * 2 + 1) / 3; 
}

int main(void) {
    long long a, b;
    
    if (scanf("%lld %lld", &a, &b) == 2) {
        long long result = min_operations(a, b);
        printf("%lld\n", result);
    }
    
    return EXIT_SUCCESS;
}
