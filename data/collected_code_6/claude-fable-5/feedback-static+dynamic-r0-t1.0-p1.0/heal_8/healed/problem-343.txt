#include <stdio.h>
#include <ctype.h>
#include <stddef.h>
#include <string.h>

int count_digits_letters(const char *str, size_t *digits, size_t *letters)
{
    size_t i;

    if (str == NULL || digits == NULL || letters == NULL) {
        return -1;
    }

    *digits = 0;
    *letters = 0;

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isdigit(c)) {
            (*digits)++;
        } else if (isalpha(c)) {
            (*letters)++;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[256];
    size_t digits = 0;
    size_t letters = 0;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (count_digits_letters(buffer, &digits, &letters) != 0) {
        fprintf(stderr, "Error counting characters\n");
        return 1;
    }

    printf("Digits: %zu\n", digits);
    printf("Letters: %zu\n", letters);

    return 0;
}

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning on `char buffer[256]` is a heuristic flag on any statically-sized array. In this code the risk is already fully mitigated, so the rewritten code retains the same structure:

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits the read to at most 255 characters plus the null terminator — no overflow is possible, unlike `gets()` or unbounded `scanf("%s")`.

2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders step (technically redundant after a successful `fgets`, but harmless and protects against future refactoring).

3. **Bounded length computation:** `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so even a hypothetically unterminated buffer cannot cause an out-of-bounds read.

4. **Safe indexing:** The `len > 0` check before accessing `buffer[len - 1]` prevents an underflow when the input is empty.

5. **Robust API contract:** `count_digits_letters` validates all pointers, casts to `unsigned char` before calling `isdigit`/`isalpha` (avoiding undefined behavior for negative `char` values on platforms where `char` is signed), and iterates only up to the null terminator.

The original code was already correct; the Flawfinder hit is a false positive given the consistent bounds checking, so the fix is to keep the mitigations in place rather than change the logic.
