```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PATTERN "/([0-9]{4})/([0-9]{1,2})/([0-9]{1,2})(/|$)"
#define MAX_FIELD_LEN 8
#define MAX_URL_LEN 2048
#define ERRBUF_LEN 128
#define MATCH_COUNT 5

typedef struct {
    char year[MAX_FIELD_LEN];
    char month[MAX_FIELD_LEN];
    char day[MAX_FIELD_LEN];
} UrlDate;

static int copy_match(const char *source, size_t source_len,
                      const regmatch_t *match,
                      char *dest, size_t dest_size)
{
    size_t length;

    if (source == NULL || match == NULL || dest == NULL || dest_size == 0) {
        return -1;
    }
    if (match->rm_so < 0 || match->rm_eo < match->rm_so) {
        return -1;
    }
    if ((size_t)match->rm_eo > source_len) {
        return -1;
    }

    length = (size_t)(match->rm_eo - match->rm_so);
    if (length >= dest_size) {
        return -1;
    }

    memcpy(dest, source + match->rm_so, length);
    dest[length] = '\0';
    return 0;
}

int extract_date_from_url(const char *url, UrlDate *result)
{
    regex_t regex;
    regmatch_t matches[MATCH_COUNT];
    size_t url_len;
    int ret;

    if (url == NULL || result == NULL) {
        return -1;
    }

    memset(result, 0, sizeof(*result));

    url_len = strnlen(url, MAX_URL_LEN);
    if (url_len >= MAX_URL_LEN) {
        return -1;
    }

    ret = regcomp(&regex, DATE_PATTERN, REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, url, MATCH_COUNT, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (copy_match(url, url_len, &matches[1], result->year, sizeof(result->year)) != 0 ||
        copy_match(url, url_len, &matches[2], result->month, sizeof(result->month)) != 0 ||
        copy_match(url, url_len, &matches[3], result->day, sizeof(result->day)) != 0) {
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *urls[] = {
        "https://example.com/blog/2023/07/15/some-article",
        "https://news.site.org/2019/1/9",
        "https://example.com/no-date-here"
    };
    size_t count = sizeof(urls) / sizeof(urls[0]);
    size_t i;
    UrlDate date;

    for (i = 0; i < count; i++) {
        if (extract_date_from_url(urls[i], &date) == 0) {
            if (printf("URL: %s\n  Year: %s, Month: %s, Day: %s\n",
                       urls[i], date.year, date.month, date.day) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("URL: %s\n  No date found\n", urls[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The weaknesses flagged by Flawfinder in the original code were already properly mitigated, so the rewrite preserves those defenses rather than restructuring them:

1. **Statically-sized arrays (`year`, `month`, `day`, `errbuf`) — CWE-119/CWE-120:** These are safe because every write into them is bounds-checked. `copy_match()` receives the destination size via `sizeof()`, rejects any match whose length is `>= dest_size` (reserving room for the NUL terminator), and `regerror()` is given `sizeof(errbuf)` with an explicit defensive NUL termination afterward.

2. **`memcpy` — CWE-120:** The copy is guarded by prior validation: the match offsets are checked for sanity (`rm_so >= 0`, `rm_eo >= rm_so`, `rm_eo <= source_len`), and the computed `length` is verified to fit in the destination before copying. This makes the `memcpy` a false positive.

3. **Additional hardening already present and retained:** `strnlen()` with `MAX_URL_LEN` caps input length, all pointers are NULL-checked, `regfree()` is called on every exit path after successful `regcomp()`, and `printf` return values are checked in `main`.

The code compiles cleanly with `gcc -Wall -Wextra -pedantic` on POSIX systems (required for `regex.h` and `strnlen`).
