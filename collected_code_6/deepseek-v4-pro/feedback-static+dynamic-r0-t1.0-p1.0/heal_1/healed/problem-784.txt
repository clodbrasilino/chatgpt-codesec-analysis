#include <stdio.h>
#include <stdlib.h>

int product_first_even_odd(const int *arr, int size) {
    int first_even = 0;
    int first_odd = 0;
    int found_even = 0;
    int found_odd = 0;

    for (int i = 0; i < size; i++) {
        if (!found_even && arr[i] % 2 == 0) {
            first_even = arr[i];
            found_even = 1;
        }
        if (!found_odd && arr[i] % 2 != 0) {
            first_odd = arr[i];
            found_odd = 1;
        }
        if (found_even && found_odd) {
            break;
        }
    }

    if (!found_even || !found_odd) {
        return 0;
    }

    return first_even * first_odd;
}

int main(void) {
    int arr[] = {5, 8, 3, 10, 7, 2};
    int size = sizeof(arr) / sizeof(arr[0]);

    int result = product_first_even_odd(arr, size);
    printf("%d\n", result);

    return 0;
}
