```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_SIZE];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (*offset + (size_t)end) <= input_len) {
            size_t len = (size_t)(end - start);
            result = malloc(len + 1U);
            if (result != NULL) {
                memcpy(result, input + *offset + (size_t)start, len);
                result[len] = '\0';
                *offset += (size_t)matches[0].rm_eo;
            } else {
                fprintf(stderr, "Memory allocation failed\n");
            }
        }
    } else if (ret != REG_NOMATCH) {
        char errbuf[ERRBUF_SIZE];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex execution failed: %s\n", errbuf);
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`. If the input is not null-terminated within that limit, the function safely bails out instead of over-reading.

2. **CWE-119/CWE-120 (static `errbuf` arrays):** The buffers remain statically sized (which is fine since `regerror` truncates to the given size), but they are now explicitly initialized and defensively null-terminated after the call, and the size is defined via a named constant (`ERRBUF_SIZE`).

3. **CWE-120 (`memcpy` overflow):** Added an explicit bounds check ensuring `*offset + end` does not exceed the validated input length before copying, guaranteeing the source region is within the buffer and the destination (`malloc(len + 1)`) always fits the copied data plus terminator.

4. **`variableScope` (cppcheck):** The `value` pointer declaration was moved inside the `for` loop body, reducing its scope to where it is actually used.

5. **Minor hardening:** `extract_quoted_value` is now `static` since it's only used within this translation unit, and the cast on `start` in the `memcpy` source arithmetic is explicit to avoid signed/unsigned mixing.
