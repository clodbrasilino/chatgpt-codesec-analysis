```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>
#include <errno.h>

#define TEMP_SIZE ((sizeof(unsigned long) * CHAR_BIT + 2) / 3 + 1)
#define OCTAL_BUFFER_SIZE (TEMP_SIZE + 2)
#define INPUT_BUFFER_SIZE 64

int decimal_to_octal(long decimal, char *buffer, size_t buffer_size)
{
    char temp[TEMP_SIZE];
    size_t index = 0;
    size_t i;
    int negative = 0;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));

    if (decimal < 0) {
        negative = 1;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    if (value == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0 && index < sizeof(temp)) {
        temp[index] = (char)('0' + (value % 8UL));
        value /= 8UL;
        index++;
    }

    if (value > 0) {
        return -1;
    }

    if (index + (size_t)negative + 1 > buffer_size) {
        return -1;
    }

    i = 0;
    if (negative == 1) {
        buffer[i] = '-';
        i++;
    }

    while (index > 0 && i < buffer_size - 1) {
        index--;
        buffer[i] = temp[index];
        i++;
    }
    buffer[i] = '\0';

    return 0;
}

int main(void)
{
    long decimal;
    char octal[OCTAL_BUFFER_SIZE];
    char input[INPUT_BUFFER_SIZE];
    char *endptr;

    memset(octal, 0, sizeof(octal));
    memset(input, 0, sizeof(input));

    printf("Enter a decimal number: ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\n")] = '\0';

    if (input[0] == '\0') {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    decimal = strtol(input, &endptr, 10);

    if (errno == ERANGE || endptr == input || *endptr != '\0') {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_octal(decimal, octal, sizeof(octal)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Octal representation: %s\n", octal);

    return EXIT_SUCCESS;
}
```

The rewritten code addresses the flagged weaknesses as follows:

**Analysis of the original code and applied fixes:**

1. **CWE-119/CWE-120 on `temp` buffer** — The original code was already well-defended here: `TEMP_SIZE` is computed to hold the maximum number of octal digits for any `unsigned long` (each octal digit encodes 3 bits), the write loop is bounded by `index < sizeof(temp)`, and there is a post-loop check (`if (value > 0)`) that detects truncation. This defense-in-depth is preserved.

2. **CWE-119/CWE-120 on `octal` buffer in `main`** — Also already mitigated: the buffer size is passed explicitly to `decimal_to_octal`, which validates `buffer_size < 2`, checks `index + negative + 1 > buffer_size` before writing, and bounds the copy loop with `i < buffer_size - 1`. The `LONG_MIN` negation edge case is correctly handled via the `-(decimal + 1) + 1UL` idiom, avoiding signed overflow (undefined behavior).

3. **Additional hardening applied** — The one genuinely weak point in the original was the raw `scanf("%ld", ...)`, which silently clamps or exhibits undefined behavior on out-of-range input on some implementations and leaves trailing garbage unvalidated. I replaced it with `fgets` into a fixed, size-checked buffer followed by `strtol` with full error checking: `errno == ERANGE` detects overflow/underflow, `endptr == input` detects no conversion, and `*endptr != '\0'` rejects trailing junk like `123abc`.
