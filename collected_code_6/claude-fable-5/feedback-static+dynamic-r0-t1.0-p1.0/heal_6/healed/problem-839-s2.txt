```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The two Flawfinder findings in the original code were both false positives, as the mitigations were already in place:

1. **Statically-sized array `first[MAX_NAME_LEN]` (CWE-119/CWE-120):** The array itself is not a defect — the risk is unchecked writes into it. Every write path goes through `set_tuple`, which validates the length with `strnlen(name, MAX_NAME_LEN)` (bounded scan, safe even on non-terminated input), rejects strings of length `>= MAX_NAME_LEN`, and double-checks against `sizeof(t->first)`. This guarantees room for the NUL terminator.

2. **`memcpy` (CWE-120):** The copy is safe because `len` is provably `< sizeof(t->first)` before the `memcpy` executes, and the terminator is written explicitly at `t->first[len]`, which is in bounds.

Additional defensive measures already present and preserved:

- `strncmp(..., MAX_NAME_LEN)` in the comparator bounds the comparison even if a struct were somehow corrupted.
- `printf("%.*s", MAX_NAME_LEN, ...)` bounds output length, preventing over-reads on a missing terminator.
- NULL-pointer and zero-count guards in `set_tuple`, `sort_tuples`, and `print_tuples`.
- `main` checks every `set_tuple` and `sort_tuples` return value and fails cleanly with `EXIT_FAILURE`.

No functional changes were required — the rewrite simply removes the analyzer comments, as the code already correctly addresses the flagged weaknesses.
