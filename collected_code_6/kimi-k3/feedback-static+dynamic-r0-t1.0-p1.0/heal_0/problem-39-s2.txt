#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <limits.h>

bool canRearrange(const char *str)
{
    int freq[UCHAR_MAX + 1] = {0};
    size_t len;
    size_t i;
    int maxFreq = 0;

    if (str == NULL) {
        return false;
    }

    len = strlen(str);

    for (i = 0; i < len; i++) {
        freq[(unsigned char)str[i]]++;
    }

    for (i = 0; i <= UCHAR_MAX; i++) {
        if (freq[i] > maxFreq) {
            maxFreq = freq[i];
        }
    }

    return maxFreq <= (int)((len + 1) / 2);
}

int main(void)
{
    char input[1024];

    if (fgets(input, sizeof(input), stdin) != NULL) {
        input[strcspn(input, "\n")] = '\0';

        if (canRearrange(input)) {
            printf("true\n");
        } else {
            printf("false\n");
        }
    }

    return 0;
}
