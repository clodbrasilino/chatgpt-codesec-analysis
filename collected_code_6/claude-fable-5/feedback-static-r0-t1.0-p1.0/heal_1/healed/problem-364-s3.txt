```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

static int count_flips(const char *str, size_t len, char start)
{
    int flips = 0;
    char expected = start;
    size_t i;

    for (i = 0; i < len; i++) {
        if (str[i] != expected) {
            flips++;
        }
        expected = (expected == '0') ? '1' : '0';
    }
    return flips;
}

int min_flips_to_alternate(const char *str, int *result)
{
    size_t len;
    size_t i;
    int flips_start_zero;
    int flips_start_one;

    if (str == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(str, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN && str[MAX_INPUT_LEN - 1] != '\0') {
        return -1;
    }

    if (len == 0) {
        *result = 0;
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (str[i] != '0' && str[i] != '1') {
            return -1;
        }
    }

    flips_start_zero = count_flips(str, len, '0');
    flips_start_one = count_flips(str, len, '1');

    *result = (flips_start_zero < flips_start_one) ? flips_start_zero : flips_start_one;
    return 0;
}

int main(void)
{
    const char *tests[] = {"0001010111", "001", "0000", "1010", "0110"};
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    int result;

    for (i = 0; i < num_tests; i++) {
        if (min_flips_to_alternate(tests[i], &result) != 0) {
            fprintf(stderr, "Invalid input: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("String: %s -> Flips required: %d\n", tests[i], result) < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was already well-structured — it validated pointers for `NULL`, checked input characters, used `size_t` for lengths, and checked the return value of `printf`. The single flagged weakness was the Flawfinder warning on `strlen` (CWE-126), which can over-read if a string is not NUL-terminated.

The fix replaces `strlen` with `strnlen`, bounding the scan to `MAX_INPUT_LEN` bytes. If the string reaches that bound without a terminator, the function rejects the input with `-1` instead of reading past the buffer. This converts a potential out-of-bounds read into a graceful error return, while preserving identical behavior for all valid, terminated inputs — the test cases in `main` produce the same output as before.
