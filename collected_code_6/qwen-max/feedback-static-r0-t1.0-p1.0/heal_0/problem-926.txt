#include <stdio.h>
#include <stdlib.h>

long rencontres_number(int n, int k) {
    long *D = (long *)malloc((n + 1) * (k + 1) * sizeof(long));
    if (!D) {
        return -1;
    }
    for (int i = 0; i <= n; ++i) {
        D[i * (k + 1)] = 1;
        if (i == 1) {
            D[i * (k + 1) + 1] = 0;
        }
    }
    for (int j = 0; j <= k; ++j) {
        D[j] = (j % 2 == 0) ? 1 : -1;
    }

    for (int i = 2; i <= n; ++i) {
        for (int j = 1; j <= k; ++j) {
            D[i * (k + 1) + j] = (i - 1) * (D[(i - 1) * (k + 1) + j] + D[(i - 1) * (k + 1) + j - 1]);
        }
    }
    
    long result = D[n * (k + 1) + k];
    free(D);
    return result;
}

int main() {
    int n, k;
    printf("Enter n and k: ");
    if (scanf("%d %d", &n, &k) != 2) {
        printf("Invalid input\n");
        return 1;
    }

    long result = rencontres_number(n, k);
    if (result == -1) {
        printf("Memory allocation failed\n");
        return 1;
    }

    printf("The %d-th Rencontres Number is: %ld\n", n, result);
    return 0;
}
