#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 4096

char **split_at_uppercase(const char *input, size_t *count);
void free_split(char **parts, size_t count);

static void cleanup_parts(char **parts, size_t count)
{
    size_t i;

    for (i = 0; i < count; i++) {
        free(parts[i]);
    }
    free(parts);
}

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t capacity = 8;
    size_t num = 0;
    size_t start = 0;
    size_t i;
    char **result;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(capacity * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i <= len; i++) {
        if (i == len || (i > 0 && isupper((unsigned char)input[i]))) {
            size_t seglen = i - start;
            size_t segsize;
            char *segment;

            if (seglen == 0) {
                start = i;
                continue;
            }

            if (start > len || seglen > len - start) {
                cleanup_parts(result, num);
                return NULL;
            }

            if (seglen > SIZE_MAX - 1) {
                cleanup_parts(result, num);
                return NULL;
            }

            segsize = seglen + 1;
            segment = malloc(segsize);
            if (segment == NULL) {
                cleanup_parts(result, num);
                return NULL;
            }

            memcpy(segment, input + start, seglen);
            segment[seglen] = '\0';

            if (num >= capacity) {
                size_t new_capacity;
                char **tmp;

                if (capacity > SIZE_MAX / 2 ||
                    (capacity * 2) > SIZE_MAX / sizeof(char *)) {
                    free(segment);
                    cleanup_parts(result, num);
                    return NULL;
                }

                new_capacity = capacity * 2;
                tmp = realloc(result, new_capacity * sizeof(char *));
                if (tmp == NULL) {
                    free(segment);
                    cleanup_parts(result, num);
                    return NULL;
                }
                result = tmp;
                capacity = new_capacity;
            }

            result[num] = segment;
            num++;
            start = i;
        }
    }

    *count = num;
    return result;
}

void free_split(char **parts, size_t count)
{
    size_t i;

    if (parts == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(parts[i]);
    }
    free(parts);
}

int main(void)
{
    const char *text = "helloWorldThisIsATest";
    size_t count = 0;
    size_t i;
    char **parts;

    parts = split_at_uppercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", parts[i]) < 0) {
            free_split(parts, count);
            return EXIT_FAILURE;
        }
    }

    free_split(parts, count);
    return EXIT_SUCCESS;
}

Key changes from the original:

1. **Removed the dead check `if (seglen >= segsize)`** — since `segsize = seglen + 1` and overflow is ruled out beforehand, this condition was provably always false (as cppcheck flagged with `knownConditionTrueFalse`). Dead code removed.

2. **Replaced `seglen >= SIZE_MAX` with `seglen > SIZE_MAX - 1`** — this is the correct overflow guard for the `seglen + 1` computation, making the intent explicit and ensuring `segsize` can never wrap.

3. **Addressed the Flawfinder `memcpy` warning (CWE-120)** — the destination buffer is allocated as exactly `seglen + 1` bytes immediately before the copy of `seglen` bytes, with the overflow check preceding the allocation, so the destination is guaranteed to hold the source data plus the NUL terminator.

4. **Factored cleanup into a static `cleanup_parts` helper** — the original repeated the free-loop five times, which is error-prone; centralizing it reduces the chance of a future leak or double-free on an error path.

5. **Preserved all defensive properties**: NULL-parameter checks, `strnlen` bounding against `MAX_INPUT_LEN`, `isupper` called on `(unsigned char)` to avoid UB with negative `char` values, capacity-doubling overflow checks before `realloc`, and full cleanup on every failure path.
