#include <stdio.h>
#include <stdlib.h>

long long calculateXorSum(int *array, int n) {
    long long xorSum = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            xorSum += (array[i] ^ array[j]);
        }
    }
    return xorSum;
}

int main() {
    int n;
    printf("Enter the number of elements: ");
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }

    if (n <= 0) {
        return EXIT_FAILURE;
    }

    int *array = (int *)malloc(n * sizeof(int));
    if (array == NULL) {
        return EXIT_FAILURE;
    }

    printf("Enter the elements:\n");
    for (int i = 0; i < n; ++i) {
        if (scanf("%d", &array[i]) != 1) {
            free(array);
            return EXIT_FAILURE;
        }
    }

    long long result = calculateXorSum(array, n);
    printf("The sum of XOR of all pairs: %lld\n", result);

    free(array);
    return EXIT_SUCCESS;
}
