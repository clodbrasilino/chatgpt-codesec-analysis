```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256
#define WORD_SIZE 256
#define MIN_WORD_LEN 3
#define MAX_WORD_LEN 5

int find_words(const char *text);

int find_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    int ret;
    char errbuf[ERRBUF_SIZE];
    char word[WORD_SIZE];
    int found;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, "[[:alpha:]]+", REG_EXTENDED);
    if (ret != 0) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regcomp failed: %s\n", errbuf);
        return -1;
    }

    cursor = text;
    found = 0;

    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            break;
        }

        size_t length = (size_t)(match.rm_eo - match.rm_so);

        if (length >= MIN_WORD_LEN && length <= MAX_WORD_LEN &&
            length < sizeof(word)) {
            memcpy(word, cursor + match.rm_so, length);
            word[length] = '\0';
            printf("%s\n", word);
            found++;
        }

        if (match.rm_eo == 0) {
            break;
        }

        cursor += match.rm_eo;
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "regexec failed: %s\n", errbuf);
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return found;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog nearby";
    int count;

    count = find_words(text);
    if (count < 0) {
        fprintf(stderr, "Error while searching for words\n");
        return EXIT_FAILURE;
    }

    printf("Total words found: %d\n", count);
    return EXIT_SUCCESS;
}
```

1. **`errbuf` (CWE-119/CWE-120):** Both `regerror` calls pass `sizeof(errbuf)`, which guarantees the library never writes past the buffer. The explicit null termination (`errbuf[sizeof(errbuf) - 1] = '\0'`) is a defensive belt-and-suspenders measure, since `regerror` already null-terminates within the given size. No overflow is possible.

2. **`word` (CWE-119/CWE-120):** The buffer is only written via the guarded `memcpy`, and the guard condition ensures safety.

3. **`memcpy` (CWE-120):** Before copying, the code checks `length < sizeof(word)`, which reserves space for the terminating `'\0'` written at `word[length]`. Combined with the `MAX_WORD_LEN` (5) constraint against a 256-byte buffer, the copy can never overflow.
