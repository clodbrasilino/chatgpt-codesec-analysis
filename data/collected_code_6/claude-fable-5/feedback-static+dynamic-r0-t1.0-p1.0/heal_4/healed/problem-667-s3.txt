```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT 1024

size_t count_vowels(const char *str)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        int c = tolower((unsigned char)str[i]);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
            count++;
        }
    }

    return count;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[MAX_INPUT];
    size_t len;
    size_t vowels;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    vowels = count_vowels(buffer);
    printf("Number of vowels: %zu\n", vowels);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two Flawfinder warnings in the original code are both low-risk, and the code as written already mitigates them properly:

1. **`getchar()` in a loop (CWE-120, CWE-20):** This is a false positive in practice. The `flush_stdin()` function reads characters without storing them into any buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop if the stream closes.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The buffer is used safely:
   - `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, guaranteeing no overflow.
   - The explicit `buffer[sizeof(buffer) - 1] = '\0';` is defensive redundancy (fgets already null-terminates), guaranteeing termination.
   - `strnlen()` with a bound is used instead of `strlen()`, so even a hypothetically unterminated buffer can't cause an over-read.
   - The `len == sizeof(buffer) - 1` check detects truncated input and drains the remainder of the line, preventing leftover stdin data from affecting subsequent reads.

3. **Other correctness points already handled:**
   - `tolower((unsigned char)str[i])` correctly avoids undefined behavior for negative `char` values.
   - `NULL` check in `count_vowels` guards against invalid input.
   - Return value of `fgets` is checked for EOF/error.
   - `%zu` is the correct format specifier for `size_t`.
