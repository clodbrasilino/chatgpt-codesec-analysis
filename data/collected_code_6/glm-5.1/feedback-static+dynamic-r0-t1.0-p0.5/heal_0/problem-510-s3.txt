#include <stdio.h>
#include <stdlib.h>

int count_subsequences(int arr[], int n, int k) {
    if (k <= 1) {
        return 0;
    }
    int count = 0;
    long long product = 1;
    int left = 0;
    for (int right = 0; right < n; right++) {
        product *= arr[right];
        while (product >= k && left <= right) {
            product /= arr[left];
            left++;
        }
        count += right - left + 1;
    }
    return count;
}

int main() {
    int arr[] = {1, 2, 3, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 10;
    int result = count_subsequences(arr, n, k);
    printf("%d\n", result);
    return 0;
}
