#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>

typedef struct {
    int first;
    int second;
} Tuple;

static bool all_records_occur_same_times(const Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0) {
        return false;
    }

    if (count > SIZE_MAX / (2 * sizeof(size_t))) {
        return false;
    }

    size_t total = count * 2;

    int *distinct = malloc(total * sizeof(*distinct));
    if (distinct == NULL) {
        return false;
    }

    size_t *occurrences = malloc(total * sizeof(*occurrences));
    if (occurrences == NULL) {
        free(distinct);
        return false;
    }

    size_t distinct_count = 0;

    for (size_t i = 0; i < count; i++) {
        const int elements[2] = {tuples[i].first, tuples[i].second};
        for (size_t k = 0; k < 2; k++) {
            bool found = false;
            for (size_t j = 0; j < distinct_count; j++) {
                if (distinct[j] == elements[k]) {
                    occurrences[j]++;
                    found = true;
                    break;
                }
            }
            if (!found) {
                distinct[distinct_count] = elements[k];
                occurrences[distinct_count] = 1;
                distinct_count++;
            }
        }
    }

    bool same = true;
    for (size_t i = 1; i < distinct_count; i++) {
        if (occurrences[i] != occurrences[0]) {
            same = false;
            break;
        }
    }

    free(occurrences);
    free(distinct);

    return same;
}

int main(void)
{
    const Tuple tuples_a[] = {{1, 2}, {2, 3}, {3, 1}};
    const Tuple tuples_b[] = {{1, 2}, {1, 3}, {4, 5}};

    size_t count_a = sizeof(tuples_a) / sizeof(tuples_a[0]);
    size_t count_b = sizeof(tuples_b) / sizeof(tuples_b[0]);

    printf("Tuples A: %s\n",
           all_records_occur_same_times(tuples_a, count_a) ? "true" : "false");
    printf("Tuples B: %s\n",
           all_records_occur_same_times(tuples_b, count_b) ? "true" : "false");

    return EXIT_SUCCESS;
}
