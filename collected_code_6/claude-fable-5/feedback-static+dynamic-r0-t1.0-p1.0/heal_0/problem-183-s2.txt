```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static int cmp_int(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

int count_pairs_with_diff_k(const int *arr, size_t n, int k, size_t *result)
{
    int *copy;
    size_t count = 0;

    if (arr == NULL || result == NULL || k < 0) {
        return -1;
    }

    *result = 0;

    if (n < 2) {
        return 0;
    }

    if (n > SIZE_MAX / sizeof *copy) {
        return -1;
    }

    copy = malloc(n * sizeof *copy);
    if (copy == NULL) {
        return -1;
    }

    memcpy(copy, arr, n * sizeof *copy);
    qsort(copy, n, sizeof *copy, cmp_int);

    if (k == 0) {
        size_t i = 0;
        while (i < n) {
            size_t j = i + 1;
            while (j < n && copy[j] == copy[i]) {
                j++;
            }
            if ((j - i) >= 2) {
                count++;
            }
            i = j;
        }
    } else {
        size_t i = 0;
        size_t j = 1;
        while (i < n && j < n) {
            if (i == j) {
                j++;
                continue;
            }
            long long diff = (long long)copy[j] - (long long)copy[i];
            if (diff == (long long)k) {
                int vi = copy[i];
                count++;
                while (i < n && copy[i] == vi) {
                    i++;
                }
            } else if (diff < (long long)k) {
                j++;
            } else {
                int vi = copy[i];
                while (i < n && copy[i] == vi) {
                    i++;
                }
            }
        }
    }

    free(copy);
    *result = count;
    return 0;
}

int main(void)
{
    int arr[] = { 1, 5, 3, 4, 2, 5, 8, 8 };
    size_t n = sizeof arr / sizeof arr[0];
    int k = 3;
    size_t result = 0;

    if (count_pairs_with_diff_k(arr, n, k, &result) != 0) {
        if (fprintf(stderr, "Failed to count pairs\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Count of distinct pairs with difference %d: %zu\n", k, result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```
