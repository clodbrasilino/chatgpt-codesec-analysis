```c
#include <stdio.h>
#include <stdlib.h>

struct Tuple {
    int *elements;
    size_t size;
};

struct TupleOfTuples {
    struct Tuple *tuples;
    size_t size;
};

double calculateAverage(const struct TupleOfTuples *tot) {
    if (tot == NULL || tot->size == 0) {
        return 0.0;
    }

    double sum = 0.0;
    size_t count = 0;

    for (size_t i = 0; i < tot->size; i++) {
        if (tot->tuples[i].elements == NULL) {
            continue;
        }
        for (size_t j = 0; j < tot->tuples[i].size; j++) {
            sum += (double)tot->tuples[i].elements[j];
            count++;
        }
    }

    if (count == 0) {
        return 0.0;
    }

    return sum / (double)count;
}

int main(void) {
    int elements1[] = {1, 2, 3};
    int elements2[] = {4, 5, 6, 7};
    int elements3[] = {8, 9};

    struct Tuple tuples[] = {
        {elements1, sizeof(elements1) / sizeof(elements1[0])},
        {elements2, sizeof(elements2) / sizeof(elements2[0])},
        {elements3, sizeof(elements3) / sizeof(elements3[0])}
    };

    struct TupleOfTuples tot = {
        tuples,
        sizeof(tuples) / sizeof(tuples[0])
    };

    double average = calculateAverage(&tot);
    printf("Average: %f\n", average);

    return 0;
}
```
