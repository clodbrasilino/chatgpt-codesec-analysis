```c
#include <stdio.h>
#include <stdlib.h>

static long long mulmod(long long a, long long b, long long mod)
{
    return (long long)((unsigned __int128)a * (unsigned __int128)b %
                       (unsigned __int128)mod);
}

long long power(long long base, long long exp, long long mod)
{
    long long result = 1 % mod;
    base %= mod;
    if (base < 0) {
        base += mod;
    }
    while (exp > 0) {
        if (exp & 1) {
            result = mulmod(result, base, mod);
        }
        exp >>= 1;
        base = mulmod(base, base, mod);
    }
    return result;
}

long long mod_inverse(long long a, long long p)
{
    return power(a, p - 2, p);
}

static long long ncr_small(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }
    if (r == 0 || r == n) {
        return 1 % p;
    }
    if (r > n - r) {
        r = n - r;
    }

    long long numerator = 1 % p;
    long long denominator = 1 % p;

    for (long long i = 0; i < r; i++) {
        numerator = mulmod(numerator, (n - i) % p, p);
        denominator = mulmod(denominator, (i + 1) % p, p);
    }

    return mulmod(numerator, mod_inverse(denominator, p), p);
}

long long ncr_mod_p(long long n, long long r, long long p)
{
    if (r < 0 || r > n) {
        return 0;
    }

    long long result = 1 % p;

    while (n > 0 || r > 0) {
        long long ni = n % p;
        long long ri = r % p;

        if (ri > ni) {
            return 0;
        }

        result = mulmod(result, ncr_small(ni, ri, p), p);

        n /= p;
        r /= p;
    }

    return result;
}

int main(void)
{
    long long n, r, p;

    printf("Enter n, r and prime p: ");
    if (scanf("%lld %lld %lld", &n, &r, &p) != 3) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0 || r < 0 || p <= 1) {
        fprintf(stderr, "Invalid values: require n >= 0, r >= 0, p > 1\n");
        return EXIT_FAILURE;
    }

    long long answer = ncr_mod_p(n, r, p);
    printf("C(%lld, %lld) %% %lld = %lld\n", n, r, p, answer);

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Signed integer overflow in modular multiplication** (in `power` and factorial computation): all modular products now go through `mulmod`, which performs the multiplication in `unsigned __int128`, so intermediate products can never overflow `long long`.

2. **Signed integer overflow of `n + 1` and the heap-buffer-overflow in `malloc((n + 1) * sizeof(long long))`**: the factorial table has been eliminated entirely. `ncr_small` computes the binomial coefficient with an O(r) running product, so there is no allocation that can overflow or be undersized for `LLONG_MAX` inputs.

3. **Wrong result "expected 10, got 0"**: this occurred whenever `p <= n`, because `n!` contains the factor `p`, making `fact[n] % p == 0` and `mod_inverse(0, p)` meaningless. The rewrite implements Lucas' theorem (`ncr_mod_p` decomposes `n` and `r` in base `p`), which is mathematically correct for any prime `p`, including `p <= n`.

4. `power` now normalizes negative bases and computes `1 % mod` for the initial result, keeping behavior well-defined for all valid inputs.
