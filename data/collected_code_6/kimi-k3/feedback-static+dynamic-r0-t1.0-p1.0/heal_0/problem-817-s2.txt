#include <stdio.h>
#include <stdlib.h>

void find_divisible(const int *arr, size_t size, int m, int n)
{
    size_t i;
    int found = 0;

    if (arr == NULL || m == 0 || n == 0) {
        fprintf(stderr, "Invalid input\n");
        return;
    }

    printf("Numbers divisible by %d or %d: ", m, n);
    for (i = 0; i < size; i++) {
        if (arr[i] % m == 0 || arr[i] % n == 0) {
            printf("%d ", arr[i]);
            found = 1;
        }
    }

    if (!found) {
        printf("None");
    }
    printf("\n");
}

int main(void)
{
    int arr[] = {10, 15, 20, 7, 30, 11, 40};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int m = 5;
    int n = 3;

    find_divisible(arr, size, m, n);

    return 0;
}
