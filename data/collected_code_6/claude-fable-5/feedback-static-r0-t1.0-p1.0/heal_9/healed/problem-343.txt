```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stddef.h>

#define BUFFER_SIZE 256

int count_digits_letters(const char *str, size_t *digits, size_t *letters)
{
    size_t i;

    if (str == NULL || digits == NULL || letters == NULL) {
        return -1;
    }

    *digits = 0;
    *letters = 0;

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isdigit(c)) {
            (*digits)++;
        } else if (isalpha(c)) {
            (*letters)++;
        }
    }

    return 0;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t digits = 0;
    size_t letters = 0;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (count_digits_letters(buffer, &digits, &letters) != 0) {
        fprintf(stderr, "Error counting characters\n");
        return 1;
    }

    printf("Digits: %zu\n", digits);
    printf("Letters: %zu\n", letters);

    return 0;
}
```

1. **Bounded input** — `fgets(buffer, sizeof(buffer), stdin)` limits the read to the buffer size, preventing overflow (unlike `gets()` or unbounded `scanf("%s", ...)`).
2. **Guaranteed null-termination** — `buffer[sizeof(buffer) - 1] = '\0';` is a defensive belt-and-suspenders measure (technically redundant since `fgets` always null-terminates on success, but harmless).
3. **Bounded length calculation** — `strnlen(buffer, sizeof(buffer))` avoids reading past the buffer even if termination were somehow missing.
4. **Safe `ctype` usage** — casting to `unsigned char` before calling `isdigit()`/`isalpha()` avoids undefined behavior for negative `char` values.
5. **NULL-pointer validation** — `count_digits_letters` checks all pointer arguments before dereferencing.
6. **Error handling** — the return values of `fgets` and the counting function are checked, with errors reported to `stderr` and a nonzero exit code.
