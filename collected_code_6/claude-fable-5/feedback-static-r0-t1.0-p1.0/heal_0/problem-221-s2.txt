```c
#include <stdio.h>
#include <stddef.h>
#include <stdbool.h>

bool find_first_even(const int *numbers, size_t count, int *result)
{
    if ((numbers == NULL) || (result == NULL))
    {
        return false;
    }

    for (size_t i = 0U; i < count; i++)
    {
        if ((numbers[i] % 2) == 0)
        {
            *result = numbers[i];
            return true;
        }
    }

    return false;
}

int main(void)
{
    const int numbers[] = {7, 3, 9, 4, 5, 8};
    const size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int first_even = 0;

    if (find_first_even(numbers, count, &first_even))
    {
        if (printf("First even number: %d\n", first_even) < 0)
        {
            return 1;
        }
    }
    else
    {
        if (printf("No even number found.\n") < 0)
        {
            return 1;
        }
    }

    return 0;
}
```
