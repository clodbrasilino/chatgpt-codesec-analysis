```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_PART_MAX_LEN 5
#define NUM_MATCH_GROUPS 4

static int extract_date_from_url(const char *url, int *year, int *month, int *day)
{
    regex_t regex;
    regmatch_t matches[NUM_MATCH_GROUPS];
    const char *pattern = "([0-9]{4})[-/]([0-9]{1,2})[-/]([0-9]{1,2})";
    int values[3];
    int status = -1;

    if (url == NULL || year == NULL || month == NULL || day == NULL) {
        return -1;
    }

    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        return -1;
    }

    if (regexec(&regex, url, NUM_MATCH_GROUPS, matches, 0) == 0) {
        int valid = 1;

        for (int i = 1; i < NUM_MATCH_GROUPS && valid; i++) {
            regoff_t start = matches[i].rm_so;
            regoff_t end = matches[i].rm_eo;
            char buffer[DATE_PART_MAX_LEN];
            size_t len;
            char *endptr = NULL;
            long value;

            if (start < 0 || end <= start) {
                valid = 0;
                break;
            }

            len = (size_t)(end - start);
            if (len >= sizeof(buffer)) {
                valid = 0;
                break;
            }

            memcpy(buffer, url + start, len);
            buffer[len] = '\0';

            value = strtol(buffer, &endptr, 10);
            if (endptr == buffer || *endptr != '\0') {
                valid = 0;
                break;
            }

            values[i - 1] = (int)value;
        }

        if (valid &&
            values[0] >= 1900 && values[0] <= 2100 &&
            values[1] >= 1 && values[1] <= 12 &&
            values[2] >= 1 && values[2] <= 31) {
            *year = values[0];
            *month = values[1];
            *day = values[2];
            status = 0;
        }
    }

    regfree(&regex);
    return status;
}

int main(void)
{
    const char *url = "https://example.com/2023/05/15/sample-article";
    int year = 0;
    int month = 0;
    int day = 0;

    if (extract_date_from_url(url, &year, &month, &day) == 0) {
        printf("URL:   %s\n", url);
        printf("Year:  %d\n", year);
        printf("Month: %d\n", month);
        printf("Day:   %d\n", day);
        return EXIT_SUCCESS;
    }

    fprintf(stderr, "Error: no valid date found in URL.\n");
    return EXIT_FAILURE;
}
```

**Summary of the fixes applied:**

1. **`buffer` scope reduced** — moved the declaration from function scope into the `for` loop body, where it is actually used. It is now re-declared on each iteration, minimizing its lifetime and visibility.

2. **`valid` scope reduced** — moved from function scope into the `if (regexec(...) == 0)` block, the only place it is needed.

3. **`i` scope reduced** — declared directly in the `for` loop initializer (`for (int i = 1; ...)`), using C99-style declaration.

4. **Buffer overflow protection for `memcpy`** — the length check now uses `sizeof(buffer)` instead of the bare macro, tying the bounds check directly to the actual destination size. The check `len >= sizeof(buffer)` guarantees the destination always has room for the copied data **plus** the terminating `'\0'`, making the `memcpy` provably safe (CWE-120 mitigated).
