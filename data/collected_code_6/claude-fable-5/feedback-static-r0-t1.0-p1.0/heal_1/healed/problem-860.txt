```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    size_t needed = regerror(errcode, regex, NULL, 0);
    char *errbuf = malloc(needed);

    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate memory for error message\n", prefix);
        return;
    }

    regerror(errcode, regex, errbuf, needed);
    fprintf(stderr, "%s: %s\n", prefix, errbuf);
    free(errbuf);
}

int ends_with_alphanumeric(const char *str)
{
    regex_t regex;
    int ret;
    int result = 0;

    if (str == NULL) {
        return 0;
    }

    ret = regcomp(&regex, "[a-zA-Z0-9]$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return -1;
    }

    ret = regexec(&regex, str, 0, NULL, 0);
    if (ret == 0) {
        result = 1;
    } else if (ret == REG_NOMATCH) {
        result = 0;
    } else {
        print_regex_error(ret, &regex, "Regex execution failed");
        result = -1;
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "world!",
        "test_",
        "abcXYZ",
        "ends with space ",
        "9"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = ends_with_alphanumeric(tests[i]);
        if (result < 0) {
            fprintf(stderr, "Error checking string: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (result == 1) {
            printf("\"%s\" ends with an alphanumeric character.\n", tests[i]);
        } else {
            printf("\"%s\" does not end with an alphanumeric character.\n", tests[i]);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was actually already safe — the Flawfinder warnings on `char errbuf[256]` were false positives. `regerror()` was called with `sizeof(errbuf)` as the size limit, which bounds all writes and guarantees null termination, so no overflow was possible (CWE-119/CWE-120 did not apply in practice).

To address the flagged pattern definitively, the rewrite eliminates the statically-sized buffers entirely:

1. **Dynamic buffer sizing** — `regerror(errcode, regex, NULL, 0)` is called first to obtain the exact required buffer size, then the buffer is allocated with `malloc()`. This removes the fixed-size array that triggered the warning and guarantees the message is never truncated.
2. **Allocation failure handling** — if `malloc()` fails, a fallback diagnostic is printed rather than dereferencing a null pointer.
3. **No leaks** — the allocated buffer is freed after use, and `regfree()` is still called on all paths after successful compilation.
4. **Deduplication** — error reporting is factored into `print_regex_error()`, reducing the surface area for future mistakes in the two previously duplicated blocks.

The remaining logic (null-check on input, distinguishing `REG_NOMATCH` from real execution errors, and the `-1`/`0`/`1` return contract) was already sound and is preserved unchanged.
