static void print_regex_error(int code, const regex_t *regex, const char *prefix)
{
    char errbuf[ERRBUF_LEN];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1);
    errbuf[sizeof(errbuf) - 1] = '\0';
    fprintf(stderr, "Error: %s: %s\n", prefix, errbuf);
}

int find_sequences(const char *input)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor;
    size_t remaining;
    int ret;
    int found;

    if (input == NULL) {
        fprintf(stderr, "Error: input string is NULL\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "failed to compile regex");
        return -1;
    }

    found = 0;
    cursor = input;
    remaining = strlen(input);

    while (*cursor != '\0') {
        size_t length;
        char *buffer;

        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            print_regex_error(ret, &regex, "regex execution failed");
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            fprintf(stderr, "Error: invalid match offsets\n");
            regfree(&regex);
            return -1;
        }

        if ((size_t)match.rm_eo > remaining) {
            fprintf(stderr, "Error: match offsets exceed input bounds\n");
            regfree(&regex);
            return -1;
        }

        length = (size_t)(match.rm_eo - match.rm_so);

        buffer = malloc(length + 1);
        if (buffer == NULL) {
            fprintf(stderr, "Error: memory allocation failed\n");
            regfree(&regex);
            return -1;
        }

        memcpy(buffer, cursor + match.rm_so, length);
        buffer[length] = '\0';
        printf("Found match: %s\n", buffer);
        free(buffer);
        found++;

        if (match.rm_eo == 0) {
            cursor++;
            remaining--;
        } else {
            cursor += match.rm_eo;
            remaining -= (size_t)match.rm_eo;
        }
    }

    regfree(&regex);

    if (found == 0) {
        printf("No matches found.\n");
    }

    return found;
}

**Analysis of the original code and the changes made:**

The original code was already largely safe — the Flawfinder findings were mostly low-risk, pattern-based warnings rather than actual vulnerabilities. Here's how each was addressed:

1. **`char errbuf[ERRBUF_LEN]` (CWE-119/CWE-120)** — `regerror()` already respects the buffer size passed to it, so the original was safe. The rewrite hardens it further by zero-initializing the buffer with `memset`, passing `sizeof(errbuf) - 1` to guarantee space for the terminator, and consolidating the duplicated logic into a single `print_regex_error()` helper, reducing the surface where a mistake could be introduced.

2. **`memcpy` (CWE-120)** — The destination is allocated as `length + 1` immediately before the copy of exactly `length` bytes, so no overflow was possible. The rewrite adds an explicit bounds check (`match.rm_eo > remaining`) validating the match offsets against the remaining input length before the copy, so the destination size is provably sufficient even if the regex library returned inconsistent offsets.

3. **Additional hardening** — Tracking `remaining` alongside the cursor guards against out-of-bounds pointer arithmetic when advancing past matches, and the zero-length-match handling (`rm_eo == 0`) is preserved to prevent infinite loops.
