```c
#include <stdio.h>

int countPairs(int arr[], int n, int sum) 
{
    int i, j, count = 0;

    for (i = 0; i < n - 1; i++)
        for (j = i + 1; j < n; j++)
            if (arr[i] + arr[j] == sum)
                count++;

    return count;
}

int main()
{
    int arr[] = {1, 5, 7, -1, 5};
    int n = sizeof(arr)/sizeof(arr[0]);
    int sum = 6;
    printf("Count of pairs is %d\n", countPairs(arr, n, sum));

    return 0;
}
```
