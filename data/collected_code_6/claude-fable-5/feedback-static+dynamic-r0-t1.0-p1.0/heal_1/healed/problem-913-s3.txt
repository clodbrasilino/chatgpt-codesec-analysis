```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_STR_LEN 4096

bool ends_with_number(const char *str)
{
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len == 0 || len >= MAX_STR_LEN) {
        return false;
    }

    return isdigit((unsigned char)str[len - 1]) != 0;
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "hello",
        "42",
        "",
        "abc9",
        "number one"
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        if (ends_with_number(tests[i])) {
            if (printf("\"%s\" ends with a number\n", tests[i]) < 0) {
                return 1;
            }
        } else {
            if (printf("\"%s\" does not end with a number\n", tests[i]) < 0) {
                return 1;
            }
        }
    }

    return 0;
}
```

Key changes made to address the identified weakness:

1. **`strlen` → `strnlen`**: The Flawfinder warning (CWE-126) concerned a potential buffer over-read if the input string is not null-terminated. `strnlen` bounds the scan to `MAX_STR_LEN` bytes, so it will never read past that limit even on a malformed input.

2. **Bound check on the result**: If `strnlen` returns `MAX_STR_LEN`, no null terminator was found within the bound, so the string is treated as invalid and the function safely returns `false` rather than risking access to unverified memory.

3. Everything else (the `NULL` check, the `unsigned char` cast before `isdigit` to avoid undefined behavior with negative `char` values, and the checked `printf` return values) was already correct and is preserved.

Note: `strnlen` is POSIX / C23; if strict C99 portability is required, an equivalent bounded loop (e.g., using `memchr(str, '\0', MAX_STR_LEN)`) could be substituted.
