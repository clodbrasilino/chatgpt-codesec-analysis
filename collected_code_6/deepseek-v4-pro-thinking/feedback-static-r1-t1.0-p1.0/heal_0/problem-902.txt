#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    int value;
} Entry;

typedef struct {
    Entry *items;
    size_t length;
    size_t capacity;
} Dictionary;

static char *duplicate_string(const char *s) {
    if (!s) {
        return NULL;
    }
    size_t len = strlen(s);
    char *copy = malloc(len + 1);
    if (!copy) {
        return NULL;
    }
    memcpy(copy, s, len + 1);
    return copy;
}

int dictionary_init(Dictionary *dict, size_t initial_capacity) {
    if (!dict) {
        return -1;
    }
    size_t cap = initial_capacity == 0 ? 1 : initial_capacity;
    if (cap > (size_t)-1 / sizeof(Entry)) {
        dict->items = NULL;
        dict->length = 0;
        dict->capacity = 0;
        return -1;
    }
    dict->items = malloc(cap * sizeof(Entry));
    if (!dict->items) {
        dict->length = 0;
        dict->capacity = 0;
        return -1;
    }
    dict->length = 0;
    dict->capacity = cap;
    return 0;
}

void dictionary_free(Dictionary *dict) {
    if (!dict) {
        return;
    }
    for (size_t i = 0; i < dict->length; ++i) {
        free(dict->items[i].key);
    }
    free(dict->items);
    dict->items = NULL;
    dict->length = 0;
    dict->capacity = 0;
}

int dictionary_add(Dictionary *dict, const char *key, int value) {
    if (!dict || !key) {
        return -1;
    }
    for (size_t i = 0; i < dict->length; ++i) {
        if (strcmp(dict->items[i].key, key) == 0) {
            dict->items[i].value += value;
            return 0;
        }
    }
    if (dict->length == dict->capacity) {
        if (dict->capacity > ((size_t)-1 / sizeof(Entry)) / 2) {
            return -1;
        }
        size_t new_capacity = dict->capacity * 2;
        Entry *new_items = realloc(dict->items, new_capacity * sizeof(Entry));
        if (!new_items) {
            return -1;
        }
        dict->items = new_items;
        dict->capacity = new_capacity;
    }
    char *key_copy = duplicate_string(key);
    if (!key_copy) {
        return -1;
    }
    dict->items[dict->length].key = key_copy;
    dict->items[dict->length].value = value;
    ++dict->length;
    return 0;
}

Dictionary combine_dictionaries(const Dictionary *a, const Dictionary *b) {
    Dictionary result = {0};
    if (!a || !b) {
        return result;
    }
    size_t total = a->length + b->length;
    if (total < a->length) {
        return result;
    }
    if (dictionary_init(&result, total) != 0) {
        return result;
    }
    for (size_t i = 0; i < a->length; ++i) {
        if (dictionary_add(&result, a->items[i].key, a->items[i].value) != 0) {
            dictionary_free(&result);
            return result;
        }
    }
    for (size_t i = 0; i < b->length; ++i) {
        if (dictionary_add(&result, b->items[i].key, b->items[i].value) != 0) {
            dictionary_free(&result);
            return result;
        }
    }
    return result;
}

void print_dictionary(const Dictionary *dict) {
    if (!dict) {
        return;
    }
    for (size_t i = 0; i < dict->length; ++i) {
        printf("%s: %d\n", dict->items[i].key, dict->items[i].value);
    }
}

int main(void) {
    Dictionary d1 = {0};
    Dictionary d2 = {0};
    Dictionary combined = {0};
    int rc = 1;

    if (dictionary_init(&d1, 4) != 0) {
        return 1;
    }
    if (dictionary_init(&d2, 4) != 0) {
        dictionary_free(&d1);
        return 1;
    }

    if (dictionary_add(&d1, "apple", 3) != 0) {
        goto cleanup;
    }
    if (dictionary_add(&d1, "banana", 4) != 0) {
        goto cleanup;
    }
    if (dictionary_add(&d2, "apple", 7) != 0) {
        goto cleanup;
    }
    if (dictionary_add(&d2, "cherry", 1) != 0) {
        goto cleanup;
    }

    combined = combine_dictionaries(&d1, &d2);
    if (combined.items == NULL && combined.length == 0 && combined.capacity == 0) {
        goto cleanup;
    }

    print_dictionary(&combined);
    rc = 0;

cleanup:
    dictionary_free(&combined);
    dictionary_free(&d2);
    dictionary_free(&d1);
    return rc;
}
