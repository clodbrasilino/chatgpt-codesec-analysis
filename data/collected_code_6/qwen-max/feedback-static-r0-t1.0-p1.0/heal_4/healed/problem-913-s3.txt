#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

int is_number_at_end(const char *str) {
    if (str == NULL) return 0;

    const char *end = str;
    while (*end && !isspace(*end)) ++end;

    if (*end) {
        while (isspace(*end)) ++end;
        while (isdigit(*end)) ++end;
        if (*end == '\0') {
            while (isdigit(*--end));
            return end < str + strlen(str) - 1 && isspace(*(end + 1));
        }
    } else {
        while (end > str && isdigit(*--end));
        return end >= str && !isdigit(*(end + 1));
    }

    return 0;
}

int main(void) {
    const char *test_strings[] = {"hello123", "123hello", "hello 123", "hello", ""};
    for (size_t i = 0; i < sizeof(test_strings) / sizeof(test_strings[0]); ++i) {
        printf("'%s': %d\n", test_strings[i], is_number_at_end(test_strings[i]));
    }
    return 0;
}
