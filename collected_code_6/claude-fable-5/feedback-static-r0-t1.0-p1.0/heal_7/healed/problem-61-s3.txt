```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U

long long count_substrings(const char *s, size_t max_len);
static int read_line(char *buf, size_t buf_size);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL || max_len == 0U) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n == max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;
    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        long long idx;

        prefix_sum += (long long)(s[i] - '0') - 1;
        idx = prefix_sum + offset;
        if (idx < 0 || (size_t)idx >= table_size) {
            free(freq);
            return -1;
        }
        count += freq[idx];
        freq[idx]++;
    }

    free(freq);
    return count;
}

static int read_line(char *buf, size_t buf_size)
{
    size_t len;

    if (buf == NULL || buf_size < 2U || buf_size > (size_t)INT32_MAX) {
        return -1;
    }

    if (fgets(buf, (int)buf_size, stdin) == NULL) {
        return -1;
    }

    buf[buf_size - 1U] = '\0';

    len = strcspn(buf, "\r\n");
    if (buf[len] == '\0' && len == buf_size - 1U) {
        int c;
        do {
            c = getchar();
        } while (c != '\n' && c != EOF);
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    memset(buffer, 0, sizeof buffer);

    printf("Enter a digit string: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (read_line(buffer, sizeof buffer) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long.\n");
        return EXIT_FAILURE;
    }

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("Number of substrings: %lld\n", result);
    return EXIT_SUCCESS;
}
```

Analysis of the identified weaknesses and how the code addresses them:

**1. `getchar()` in a loop (CWE-120, CWE-20, risk 1):** The flagged loop is actually the *fix* for a buffer boundary issue, not the vulnerability. When `fgets` fills the buffer without encountering a newline, the loop drains the remaining characters from `stdin` and returns `-1`, rejecting over-long input rather than silently truncating it. The characters read by `getchar()` are discarded, never stored, so no buffer can overflow. This is a false positive, and the code is retained as-is.

**2. Statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120, risk 2):** All accesses to the buffer are bounds-checked:
- `memset(buffer, 0, sizeof buffer)` initializes it fully.
- `read_line` receives `sizeof buffer` and validates it (`>= 2`, `<= INT32_MAX`) before passing it to `fgets`, which respects the size limit.
- The buffer is explicitly NUL-terminated at `buf_size - 1U` as a defensive measure.
- `count_substrings` uses `strnlen(s, max_len)` and rejects strings that reach `max_len`, guaranteeing NUL termination within bounds.

Additional defensive properties already present and preserved: overflow guards before `calloc` sizing (`SIZE_MAX` checks), index range validation before every `freq[idx]` access, `isdigit` cast through `unsigned char` to avoid UB on negative `char` values, and consistent `free` on all error paths preventing memory leaks. The only change made was adding a `max_len == 0U` guard in `count_substrings` to harden against a zero-length limit being passed.
