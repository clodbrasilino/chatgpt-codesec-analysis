```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 4096

bool matches_word_at_end(const char *str, const char *word)
{
    size_t str_len;
    size_t word_len;
    size_t end;

    if (str == NULL || word == NULL) {
        return false;
    }

    str_len = strnlen(str, MAX_INPUT_LEN);
    word_len = strnlen(word, MAX_INPUT_LEN);

    if (str_len >= MAX_INPUT_LEN || word_len >= MAX_INPUT_LEN) {
        return false;
    }

    if (word_len == 0 || str_len < word_len) {
        return false;
    }

    end = str_len;
    while (end > 0 && ispunct((unsigned char)str[end - 1])) {
        end--;
    }

    if (end < word_len) {
        return false;
    }

    if (strncmp(&str[end - word_len], word, word_len) != 0) {
        return false;
    }

    if (end > word_len) {
        unsigned char before = (unsigned char)str[end - word_len - 1];
        if (isalnum(before) || before == '_') {
            return false;
        }
    }

    return true;
}

int main(void)
{
    const char *sentences[] = {
        "The quick brown fox jumps over the lazy dog.",
        "Hello, world!!!",
        "No punctuation here world",
        "This does not end with the word",
        "worldly matters...",
        ""
    };
    const char *word = "world";
    size_t i;
    size_t count = sizeof(sentences) / sizeof(sentences[0]);

    for (i = 0; i < count; i++) {
        bool result = matches_word_at_end(sentences[i], word);
        if (printf("\"%s\" ends with \"%s\": %s\n",
                   sentences[i], word,
                   result ? "yes" : "no") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (matches_word_at_end(sentences[0], "dog")) {
        if (printf("First sentence ends with \"dog\"\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Key changes made to address the identified weaknesses:**

1. **Replaced `strlen` with `strnlen`** (CWE-126 mitigation): Both calls to `strlen` were flagged by Flawfinder because they can over-read if given a non-`\0`-terminated buffer. `strnlen` bounds the scan to `MAX_INPUT_LEN` bytes, so a missing terminator can no longer cause an unbounded read or crash.

2. **Added a sanity check on the returned lengths**: If `strnlen` returns `MAX_INPUT_LEN`, the string was not terminated within the allowed bound (or is suspiciously large), so the function safely rejects the input by returning `false` rather than proceeding with potentially truncated/invalid data.

3. **Added `#define _POSIX_C_SOURCE 200809L`**: `strnlen` is a POSIX.1-2008 function, so this feature-test macro ensures its declaration is visible in `<string.h>` on conforming systems.

The rest of the logic was already defensively written (NULL checks, `unsigned char` casts before `ctype.h` calls, bounds-checked indexing, and checked `printf` return values), so it was preserved as-is.
