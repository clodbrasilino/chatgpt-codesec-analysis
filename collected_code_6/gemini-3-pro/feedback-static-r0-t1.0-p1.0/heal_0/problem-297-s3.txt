#include <stdio.h>
#include <stdlib.h>

typedef struct NestedList {
    int isInteger;
    int integer;
    struct NestedList** list;
    size_t listCount;
} NestedList;

typedef struct FlattenedList {
    int* elements;
    size_t size;
    size_t capacity;
} FlattenedList;

FlattenedList* create_flattened_list() {
    FlattenedList* flatList = (FlattenedList*)malloc(sizeof(FlattenedList));
    if (!flatList) return NULL;
    
    flatList->size = 0;
    flatList->capacity = 10;
    flatList->elements = (int*)malloc(flatList->capacity * sizeof(int));
    
    if (!flatList->elements) {
        free(flatList);
        return NULL;
    }
    
    return flatList;
}

void free_flattened_list(FlattenedList* flatList) {
    if (flatList) {
        free(flatList->elements);
        free(flatList);
    }
}

int append_to_flattened_list(FlattenedList* flatList, int value) {
    if (!flatList) return 0;
    
    if (flatList->size >= flatList->capacity) {
        size_t newCapacity = flatList->capacity * 2;
        int* newElements = (int*)realloc(flatList->elements, newCapacity * sizeof(int));
        if (!newElements) return 0;
        
        flatList->elements = newElements;
        flatList->capacity = newCapacity;
    }
    
    flatList->elements[flatList->size++] = value;
    return 1;
}

int flatten_nested_list_recursive(NestedList* nested, FlattenedList* flatList) {
    if (!nested || !flatList) return 0;
    
    if (nested->isInteger) {
        return append_to_flattened_list(flatList, nested->integer);
    } else {
        for (size_t i = 0; i < nested->listCount; ++i) {
            if (!flatten_nested_list_recursive(nested->list[i], flatList)) {
                return 0;
            }
        }
    }
    return 1;
}

FlattenedList* flatten(NestedList* nested) {
    FlattenedList* flatList = create_flattened_list();
    if (!flatList) return NULL;
    
    if (!flatten_nested_list_recursive(nested, flatList)) {
        free_flattened_list(flatList);
        return NULL;
    }
    
    return flatList;
}

int main() {
    NestedList inner1 = {1, 1, NULL, 0};
    NestedList inner2 = {1, 2, NULL, 0};
    
    NestedList* innerList[] = {&inner1, &inner2};
    NestedList middle = {0, 0, innerList, 2};
    NestedList outerInner = {1, 3, NULL, 0};
    
    NestedList* outerList[] = {&middle, &outerInner};
    NestedList outer = {0, 0, outerList, 2};
    
    FlattenedList* result = flatten(&outer);
    
    if (result) {
        for (size_t i = 0; i < result->size; ++i) {
            printf("%d ", result->elements[i]);
        }
        printf("\n");
        free_flattened_list(result);
    }
    
    return 0;
}
