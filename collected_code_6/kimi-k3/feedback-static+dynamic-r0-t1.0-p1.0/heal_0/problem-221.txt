#include <stdio.h>

int find_first_even(const int *numbers, size_t count, int *result)
{
    size_t i;

    if (numbers == NULL || result == NULL || count == 0U)
    {
        return 0;
    }

    for (i = 0U; i < count; i++)
    {
        if (numbers[i] % 2 == 0)
        {
            *result = numbers[i];
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    int numbers[] = {3, 7, 9, 12, 15};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int result = 0;

    if (find_first_even(numbers, count, &result))
    {
        printf("%d\n", result);
    }
    else
    {
        printf("No even number found\n");
    }

    return 0;
}
