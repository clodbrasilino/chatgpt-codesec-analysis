```c
#include <stdio.h>
#include <math.h>

int roundListAndMultiplyLength(int *arr, int len) {
    int sum = 0;
    for (int i = 0; i < len; i++) {
        sum += round(arr[i]);
    }
    return sum * len;
}

int main() {
    int numbers[] = {1.6, 2.4, 3.5, 4.5};
    int length = sizeof(numbers) / sizeof(numbers[0]);
    printf("%d\n", roundListAndMultiplyLength(numbers, length));
    return 0;
}
```
