```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > n) {
        return -1;
    }

    total_bytes = n * sizeof(int);
    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes ||
        head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    if (head_bytes <= total_bytes && k <= n) {
        memcpy(temp, arr, head_bytes);
    } else {
        free(temp);
        return -1;
    }

    if (tail_bytes <= total_bytes) {
        memmove(arr, arr + k, tail_bytes);
    } else {
        free(temp);
        return -1;
    }

    if ((n - k) <= n && head_bytes <= total_bytes - tail_bytes) {
        memcpy(arr + (n - k), temp, head_bytes);
    } else {
        free(temp);
        return -1;
    }

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The rewritten code addresses the flagged CWE-120 concerns by adding explicit bounds validation immediately before each `memcpy` and `memmove` call, guaranteeing that the destination buffer can always hold the source data:

1. **First `memcpy` (arr → temp):** `temp` is allocated with exactly `head_bytes`, and the check confirms `head_bytes <= total_bytes` and `k <= n`, so copying `head_bytes` from `arr` never over-reads the source or overflows the destination.

2. **`memmove` (tail shift):** verified that `tail_bytes <= total_bytes`, so writing `tail_bytes` starting at `arr` stays within the array.

3. **Second `memcpy` (temp → arr tail):** verified that the destination offset `n - k` is valid and that `head_bytes` fits in the remaining space (`total_bytes - tail_bytes`), preventing any write past the end of `arr`.

Each failed check releases `temp` before returning, avoiding memory leaks on the error paths. The existing overflow guard (`n > SIZE_MAX / sizeof(int)`) and the arithmetic consistency check (`head_bytes + tail_bytes == total_bytes`) are retained, ensuring no integer overflow can undermine the size calculations that the copy operations depend on.
