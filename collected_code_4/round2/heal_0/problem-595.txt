#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int minSwap(char *a, char *b, int n) {
    int swaps = 0;
    int i = 0, j = 0;
  
    while (i < n) {
        j = i;

        while (a[j] != b[i]) {
            j += 1;
        }

        if (j != i) {
            while (j > i) {
                char temp;
                temp = a[j];
                a[j] = a[j - 1];
                a[j - 1] = temp;
                j -= 1;
                swaps += 1;
            }
        }
        i += 1;
    }

    return swaps;
}

int main() {
    char str1[100], str2[100];
    int n1, n2;
  
    printf("Enter first binary string: ");
    scanf("%s", str1);
    printf("Enter second binary string: ");
    scanf("%s", str2);

    n1 = strlen(str1);
    n2 = strlen(str2);

    if(n1 != n2) {
        printf("Binary strings must be of equal length.\n");
        return 0;
    }

    printf("Minimum swaps required = %d\n", minSwap(str1, str2, n1));

    return 0;
}
