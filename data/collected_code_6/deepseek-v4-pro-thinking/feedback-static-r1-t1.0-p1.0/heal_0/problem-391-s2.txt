static DictNode *create_dict_node(const char *key, int value) {
    DictNode *node = malloc(sizeof(*node));
    if (node == NULL) {
        return NULL;
    }

    node->key = malloc(strlen(key) + 1);
    if (node->key == NULL) {
        free(node);
        return NULL;
    }

    strcpy(node->key, key);
    node->value = value;
    node->child = NULL;
    node->next = NULL;

    return node;
}
