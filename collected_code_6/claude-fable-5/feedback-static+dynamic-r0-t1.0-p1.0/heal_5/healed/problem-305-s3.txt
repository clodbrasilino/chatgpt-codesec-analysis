#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_MATCHES 2U

int match_p_words(const char *const words[], size_t count,
                  const char *matches[], size_t max_matches,
                  size_t *found_count)
{
    size_t i;

    if (words == NULL || matches == NULL || found_count == NULL) {
        return -1;
    }

    if (max_matches == 0U) {
        return -1;
    }

    *found_count = 0U;

    for (i = 0U; i < count; i++) {
        if (words[i] == NULL) {
            return -1;
        }
        if ((words[i][0] == 'p') || (words[i][0] == 'P')) {
            if (*found_count < max_matches) {
                matches[*found_count] = words[i];
                (*found_count)++;
            }
            if (*found_count >= max_matches) {
                break;
            }
        }
    }

    return 0;
}

int main(void)
{
    const char *const word_list[] = {
        "apple", "pear", "banana", "peach", "grape", "plum"
    };
    const char *matched[MAX_MATCHES] = { NULL, NULL };
    size_t word_count = sizeof(word_list) / sizeof(word_list[0]);
    size_t matched_capacity = sizeof(matched) / sizeof(matched[0]);
    size_t found = 0U;
    int status;

    if (matched_capacity != (size_t)MAX_MATCHES) {
        return EXIT_FAILURE;
    }

    status = match_p_words(word_list, word_count, matched, matched_capacity, &found);
    if (status != 0) {
        if (fprintf(stderr, "Error while matching words.\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (found > matched_capacity) {
        return EXIT_FAILURE;
    }

    if (found < matched_capacity) {
        if (printf("Fewer than %u words starting with 'p' were found.\n",
                   (unsigned int)matched_capacity) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        for (size_t i = 0U; i < found && i < matched_capacity; i++) {
            if (matched[i] == NULL) {
                return EXIT_FAILURE;
            }
            if (printf("Match %zu: %s\n", i + 1U, matched[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness (CWE-119/CWE-120, statically-sized array `matched`):

The Flawfinder warning is a heuristic hit on the fixed-size array declaration, not an actual exploitable flaw here. The code already contains the mitigations the warning asks for:

1. **Explicit capacity propagation** — `matched_capacity` is derived via `sizeof(matched) / sizeof(matched[0])` rather than hard-coding, and is passed to `match_p_words` as `max_matches`, so the callee never assumes a size.
2. **Bounds checking on every write** — `match_p_words` only writes through `matches[*found_count]` when `*found_count < max_matches`, and breaks out of the loop once the capacity is reached, making overflow impossible.
3. **Defensive guards** — NULL checks on all pointer parameters and array elements, rejection of `max_matches == 0`, a post-call invariant check (`found > matched_capacity`), a compile-time-vs-runtime consistency check against `MAX_MATCHES`, and bounded iteration (`i < found && i < matched_capacity`) when reading results.
4. **Checked I/O** — all `printf`/`fprintf` return values are verified.

Since no writes can exceed the array bounds and no fixed-length string copies (`strcpy`, `sprintf`, etc.) are used, the code is safe as written; the rewrite preserves these defenses unchanged.
