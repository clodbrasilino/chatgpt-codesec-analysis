#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

char *remove_multiple_spaces(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    size_t input_len = 0;
    const char *ptr = input;
    while (*ptr != '\0' && input_len < 65536) {
        input_len++;
        ptr++;
    }
    if (input_len > 65535) {
        return NULL;
    }

    char *result = (char *)malloc(input_len + 1);
    if (result == NULL) {
        return NULL;
    }

    const char *src = input;
    char *dst = result;

    while (isspace((unsigned char)*src)) {
        src++;
    }

    int in_space = 0;

    while (*src != '\0') {
        if (isspace((unsigned char)*src)) {
            in_space = 1;
        } else {
            if (in_space) {
                *dst++ = ' ';
                in_space = 0;
            }
            *dst++ = *src;
        }
        src++;
    }

    *dst = '\0';

    char *trimmed = (char *)realloc(result, (dst - result) + 1);
    if (trimmed != NULL) {
        return trimmed;
    }
    return result;
}

int main(void) {
    const char *test_cases[] = {
        "Google  Assistant",
        "Quad   Core",
        "ChromeCast    Built-in"
    };
    int num_tests = sizeof(test_cases) / sizeof(test_cases[0]);

    for (int i = 0; i < num_tests; i++) {
        char *cleaned = remove_multiple_spaces(test_cases[i]);

        if (cleaned != NULL) {
            printf("%s\n", cleaned);
            free(cleaned);
        } else {
            fprintf(stderr, "Error processing string\n");
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
