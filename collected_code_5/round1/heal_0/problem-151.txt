```C
#include <stdio.h>

int gcd(int a, int b) {
  while (b != 0) {
    int temp = b;
    b = a % b;
    a = temp;
  }
  return a;
}

int areCoprime(int a, int b) { return (gcd(a, b) == 1); }

int main() {
  int num1, num2;
  printf("Enter two numbers: ");
  scanf("%d %d", &num1, &num2);

  if (areCoprime(num1, num2)) {
    printf("The numbers are co-prime\n");
  } else {
    printf("The numbers are not co-prime\n");
  }
  return 0;
}
```
