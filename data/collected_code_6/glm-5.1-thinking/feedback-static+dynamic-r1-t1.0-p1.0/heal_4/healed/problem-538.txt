#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char **items;
    size_t len;
} Tuple;

void free_tuple(Tuple *tuple) {
    if (tuple == NULL) {
        return;
    }
    if (tuple->items != NULL) {
        for (size_t i = 0; i < tuple->len; i++) {
            free(tuple->items[i]);
        }
        free(tuple->items);
    }
    free(tuple);
}

Tuple *string_list_to_tuple(const char *str) {
    if (str == NULL) {
        return NULL;
    }
    size_t len = 0;
    while (len < 65536 && str[len] != '\0') {
        len++;
    }
    if (len == 65536) {
        return NULL;
    }
    Tuple *tuple = calloc(1, sizeof(Tuple));
    if (tuple == NULL) {
        return NULL;
    }
    tuple->len = len;
    if (len == 0) {
        tuple->items = NULL;
        return tuple;
    }
    tuple->items = calloc(len, sizeof(char *));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }
    for (size_t i = 0; i < len; i++) {
        tuple->items[i] = malloc(2 * sizeof(char));
        if (tuple->items[i] == NULL) {
            for (size_t j = 0; j < i; j++) {
                free(tuple->items[j]);
            }
            free(tuple->items);
            free(tuple);
            return NULL;
        }
        tuple->items[i][0] = str[i];
        tuple->items[i][1] = '\0';
    }
    return tuple;
}

int main(int argc, const char *argv[]) {
    if (argc > 1) {
        for (int i = 1; i < argc; i++) {
            Tuple *tuple = string_list_to_tuple(argv[i]);
            if (tuple != NULL) {
                printf("[");
                for (size_t j = 0; j < tuple->len; j++) {
                    printf("'%s'", tuple->items[j]);
                    if (j + 1 < tuple->len) {
                        printf(", ");
                    }
                }
                printf("]\n");
                fflush(stdout);
                free_tuple(tuple);
            } else {
                printf("[]\n");
                fflush(stdout);
            }
        }
    } else {
        char *buffer = malloc(65536);
        if (buffer == NULL) {
            return EXIT_FAILURE;
        }
        while (fgets(buffer, 65536, stdin)) {
            buffer[strcspn(buffer, "\r\n")] = '\0';
            Tuple *tuple = string_list_to_tuple(buffer);
            if (tuple != NULL) {
                printf("[");
                for (size_t j = 0; j < tuple->len; j++) {
                    printf("'%s'", tuple->items[j]);
                    if (j + 1 < tuple->len) {
                        printf(", ");
                    }
                }
                printf("]\n");
                fflush(stdout);
                free_tuple(tuple);
            } else {
                printf("[]\n");
                fflush(stdout);
            }
        }
        free(buffer);
    }
    return EXIT_SUCCESS;
}
