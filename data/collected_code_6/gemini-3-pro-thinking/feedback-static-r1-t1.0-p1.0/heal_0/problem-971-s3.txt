#include <stdio.h>
#include <stdlib.h>

int get_max_segments(int n, int a, int b, int c);

int get_max_segments(int n, int a, int b, int c) {
    if (n < 0 || a <= 0 || b <= 0 || c <= 0) {
        return -1;
    }

    int *dp = (int *)malloc((n + 1) * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        dp[i] = -1;
    }

    for (int i = 1; i <= n; i++) {
        if (i >= a && dp[i - a] != -1) {
            if (dp[i - a] + 1 > dp[i]) {
                dp[i] = dp[i - a] + 1;
            }
        }
        if (i >= b && dp[i - b] != -1) {
            if (dp[i - b] + 1 > dp[i]) {
                dp[i] = dp[i - b] + 1;
            }
        }
        if (i >= c && dp[i - c] != -1) {
            if (dp[i - c] + 1 > dp[i]) {
                dp[i] = dp[i - c] + 1;
            }
        }
    }

    int result = dp[n];
    free(dp);

    return result;
}

int main(void) {
    int n = 7;
    int a = 5;
    int b = 2;
    int c = 2;

    int result = get_max_segments(n, a, b, c);

    if (result == -1) {
        fprintf(stderr, "Error computing segments or no valid segments can be formed.\n");
        return EXIT_FAILURE;
    }

    printf("%d\n", result);

    return EXIT_SUCCESS;
}
