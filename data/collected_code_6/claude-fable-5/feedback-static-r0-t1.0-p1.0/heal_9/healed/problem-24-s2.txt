```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_LEN 128U

static int binary_to_decimal(const char *binary, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, MAX_INPUT_LEN);
    if (len == 0U || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_check_truncation(void)
{
    int truncated = 0;
    unsigned long guard = 0UL;
    int ch;

    do {
        ch = fgetc(stdin);
        if (ch == '\n' || ch == EOF) {
            break;
        }
        truncated = 1;
        guard++;
    } while (guard < ULONG_MAX);

    return truncated;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        if (flush_stdin_check_truncation() == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

1. **Flawfinder `fgetc` warning (CWE-120, CWE-20)** — The `fgetc` loop in `flush_stdin_check_truncation()` never writes into a buffer; it only discards characters until newline or EOF. The `guard` counter bounds the loop against pathological infinite streams, so no buffer boundary issue exists. This is a false positive, and the mitigation is already in place.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** — The statically sized `input[MAX_INPUT_LEN]` buffer is properly protected:
   - `fgets` is used with `sizeof(input)`, which limits the read and always null-terminates within bounds.
   - A defensive explicit null terminator is written at `input[sizeof(input) - 1U]`.
   - `strnlen` (bounded) is used instead of `strlen`.
   - Truncated input (no newline consumed) is detected and rejected rather than silently processed.

3. **Additional hardening already present:**
   - `binary_to_decimal` validates NULL pointers, rejects empty/oversized input, validates each character strictly as `'0'`/`'1'`, and checks `value > (ULLONG_MAX >> 1)` before shifting to prevent unsigned overflow/wraparound (CWE-190).
   - All return values (`fgets`, conversion result) are checked, with errors reported to `stderr` and distinct exit codes.

The code as written already addresses both Flawfinder findings; the rewrite above is functionally identical with the comments removed.
