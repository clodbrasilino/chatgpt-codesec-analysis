```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        sum += (unsigned char)str[i];
    }

    return sum;
}

static int discard_remaining_input(void)
{
    int ch;
    size_t discarded = 0;
    const size_t discard_limit = 1048576;

    do {
        ch = getchar();
        discarded++;
        if (discarded > discard_limit) {
            return -1;
        }
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? EOF : 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long result;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    if (strchr(buffer, '\n') == NULL && !feof(stdin)) {
        if (discard_remaining_input() < 0) {
            fprintf(stderr, "Error discarding input\n");
            return EXIT_FAILURE;
        }
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and applied fixes:**

The original code was already well-hardened, and both Flawfinder findings were essentially false positives — but here is how each one was addressed:

1. **`getchar()` in a loop (CWE-120, CWE-20):** Flawfinder flags this because unbounded read loops can be abused. The original loop was safe since it discards bytes without storing them, so no buffer overflow was possible. To fully satisfy the hint, I added an upper bound (`discard_limit`) so the drain loop cannot spin indefinitely on a malicious or infinite input stream, and the caller now checks its return value.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The original already mitigated this correctly: `fgets` is bounded by `sizeof(buffer)`, an explicit NUL terminator is forced at `buffer[sizeof(buffer) - 1]`, overly long input is detected via the missing `\n` and rejected, and `ascii_sum` uses `strnlen` with an explicit bound rather than `strlen`. I added a `memset` to zero-initialize the buffer as defense-in-depth against any read of uninitialized memory.

**Other correctness points already present and preserved:**

- `(unsigned char)str[i]` cast prevents sign-extension issues with high-bit characters.
- `strcspn` safely strips the newline even if none exists.
- `fflush(stdout)` return value is checked before reading input.
- The `len == max_len` check in `ascii_sum` rejects unterminated strings.
- All error paths return `EXIT_FAILURE` with diagnostics on `stderr`.
