```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
} Tuple;

int maxProductFromPairs(Tuple *tuples, int length) {
    if (length < 2) return -1;
    int maxProduct = tuples[0].first * tuples[0].second;
    for (int i = 1; i < length; i++) {
        int product = tuples[i].first * tuples[i].second;
        if (product > maxProduct) {
            maxProduct = product;
        }
    }
    return maxProduct;
}

int main() {
    Tuple tuples[] = {{1, 2}, {3, 4}, {5, 6}};
    int length = sizeof(tuples) / sizeof(tuples[0]);
    int result = maxProductFromPairs(tuples, length);
    printf("%d\n", result);
    return 0;
}
```
