```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (isupper((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];

    memset(buffer, 0, sizeof(buffer));

    while (fgets(buffer, (int)sizeof(buffer), stdin) != NULL) {
        buffer[sizeof(buffer) - 1] = '\0';

        size_t len = strnlen(buffer, sizeof(buffer));

        while (len > 0 && (buffer[len - 1] == '\n' || buffer[len - 1] == '\r')) {
            buffer[len - 1] = '\0';
            len--;
        }

        char *output = remove_lowercase(buffer);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        printf("%s\n", output);
        fflush(stdout);
        free(output);

        memset(buffer, 0, sizeof(buffer));
    }

    return EXIT_SUCCESS;
}
```

The original code was already largely safe regarding the flagged weakness: `fgets` is bounded by `sizeof(buffer)`, `strnlen` limits length scanning, and `isupper` receives an `unsigned char` cast to avoid undefined behavior on negative values. The rewrite hardens it further by explicitly NUL-terminating the buffer after each read as a defensive guarantee, and zero-initializing the buffer before use and after each iteration so no stale data can ever be interpreted as input, fully mitigating the CWE-119/CWE-120 concern raised by Flawfinder.
