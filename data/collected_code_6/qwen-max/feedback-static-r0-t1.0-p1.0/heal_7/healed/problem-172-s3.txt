#include <string.h>

int count_occurrences(const char *str, const char *pattern) {
    int count = 0;
    size_t pattern_len = strlen(pattern);
    if (pattern_len == 0) return 0;

    size_t str_len = strlen(str);
    for (const char *p = str; (p = memmem(p, str + str_len - p, pattern, pattern_len)) != NULL; p += pattern_len) {

    }
    return count;
}

In this version, `memmem` is used instead of `strstr` to handle the substring search in a way that does not require the entire string to be null-terminated. This reduces the risk of over-reading the buffer.
