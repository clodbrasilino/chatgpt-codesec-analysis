```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_largest_positive(const int *array, size_t size) {
    if (array == NULL || size == 0) {
        return -1;
    }

    int max_val = -1;

    for (size_t i = 0; i < size; i++) {
        if (array[i] > 0) {
            if (max_val == -1 || array[i] > max_val) {
                max_val = array[i];
            }
        }
    }

    return max_val;
}

int main(void) {
    int data[] = {-5, -2, 0, 3, 12, -8, 7, 0, -1, 15};
    size_t size = sizeof(data) / sizeof(data[0]);

    int result = find_largest_positive(data, size);

    printf("%d\n", result);

    return 0;
}
```
