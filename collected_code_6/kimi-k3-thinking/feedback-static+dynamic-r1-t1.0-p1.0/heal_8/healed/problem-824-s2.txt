```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <limits.h>
#include <errno.h>

size_t remove_even(int *arr, size_t size)
{
    size_t read;
    size_t write;

    if (arr == NULL || size == 0) {
        return 0;
    }

    write = 0;
    for (read = 0; read < size; read++) {
        if (write > read || write >= size) {
            break;
        }
        if (arr[read] % 2 != 0) {
            arr[write] = arr[read];
            write++;
        }
    }

    return write;
}

int main(int argc, const char *argv[])
{
    int *arr;
    size_t size;
    size_t new_size;
    size_t i;
    char *end;
    unsigned long value;

    if (argv == NULL || (argc > 0 && argv[0] == NULL)) {
        fprintf(stderr, "Invalid invocation\n");
        return EXIT_FAILURE;
    }

    if (argc != 2 || argv[1] == NULL) {
        fprintf(stderr, "Usage: %s <array_size>\n", argv[0]);
        return EXIT_FAILURE;
    }

    errno = 0;
    end = NULL;
    value = strtoul(argv[1], &end, 10);
    if (errno != 0 || end == argv[1] || *end != '\0' || value > (unsigned long)SIZE_MAX) {
        fprintf(stderr, "Invalid array size\n");
        return EXIT_FAILURE;
    }

    size = (size_t)value;

    if (size == 0 || size > SIZE_MAX / sizeof(*arr) || size > (size_t)INT_MAX) {
        fprintf(stderr, "Invalid array size\n");
        return EXIT_FAILURE;
    }

    arr = malloc(size * sizeof(*arr));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < size; i++) {
        arr[i] = (int)(i + 1);
    }

    printf("Original list: ");
    for (i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    new_size = remove_even(arr, size);

    if (new_size > size) {
        fprintf(stderr, "Unexpected result size\n");
        free(arr);
        arr = NULL;
        return EXIT_FAILURE;
    }

    printf("List after removing even numbers: ");
    for (i = 0; i < new_size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    free(arr);
    arr = NULL;

    return EXIT_SUCCESS;
}
```

Key hardening applied relative to the flagged weaknesses:

- **In `remove_even`**: added an explicit invariant guard (`write > read || write >= size`) inside the loop so the write index can never exceed the buffer bounds, even if the function is later modified or called with inconsistent arguments. The `read < size` loop condition already bounds the read index.
- **In `main`**: validated `argv`/`argv[0]` before dereferencing for the usage message; retained strict `strtoul` validation (errno, empty input, trailing characters, range); kept the overflow check `size > SIZE_MAX / sizeof(*arr)` before `malloc` and the `INT_MAX` cap so the `(int)(i + 1)` cast and all loop indices stay in range.
- All loops that index `arr` (`i < size`, `i < new_size`, `read < size`) are bounded by the allocated/validated sizes, and `new_size > size` is checked before the output loop.
