#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int is_none;
    int value;
} Element;

typedef struct {
    Element *elements;
    size_t length;
} Tuple;

static int tuple_all_none(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->elements == NULL || t->length == 0U) {
        return 1;
    }
    for (i = 0U; i < t->length; i++) {
        if (t->elements[i].is_none == 0) {
            return 0;
        }
    }
    return 1;
}

static void tuple_free(Tuple *t)
{
    if (t != NULL) {
        free(t->elements);
        t->elements = NULL;
        t->length = 0U;
    }
}

static size_t remove_all_none_tuples(Tuple *tuples, size_t count)
{
    size_t write_index = 0U;
    size_t read_index;

    if (tuples == NULL) {
        return 0U;
    }
    for (read_index = 0U; read_index < count; read_index++) {
        if (tuple_all_none(&tuples[read_index]) != 0) {
            tuple_free(&tuples[read_index]);
        } else {
            if (write_index != read_index) {
                tuples[write_index] = tuples[read_index];
                tuples[read_index].elements = NULL;
                tuples[read_index].length = 0U;
            }
            write_index++;
        }
    }
    return write_index;
}

static int tuple_init(Tuple *t, const int *values, const int *none_flags, size_t length)
{
    size_t i;

    if (t == NULL) {
        return -1;
    }
    t->elements = NULL;
    t->length = 0U;
    if (length == 0U) {
        return 0;
    }
    if (values == NULL || none_flags == NULL) {
        return -1;
    }
    if (length > (SIZE_MAX / sizeof(Element))) {
        return -1;
    }
    t->elements = (Element *)malloc(length * sizeof(Element));
    if (t->elements == NULL) {
        return -1;
    }
    for (i = 0U; i < length; i++) {
        t->elements[i].is_none = none_flags[i];
        t->elements[i].value = values[i];
    }
    t->length = length;
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;
    size_t j;

    if (tuples == NULL) {
        return;
    }
    printf("[");
    for (i = 0U; i < count; i++) {
        printf("(");
        for (j = 0U; j < tuples[i].length; j++) {
            if (tuples[i].elements[j].is_none != 0) {
                printf("None");
            } else {
                printf("%d", tuples[i].elements[j].value);
            }
            if ((j + 1U) < tuples[i].length) {
                printf(", ");
            }
        }
        printf(")");
        if ((i + 1U) < count) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main(void)
{
    Tuple tuples[5];
    size_t count = 5U;
    size_t new_count;
    size_t i;

    const int v0[] = {0, 0};
    const int n0[] = {1, 1};
    const int v1[] = {1, 2};
    const int n1[] = {0, 0};
    const int v2[] = {0, 0, 0};
    const int n2[] = {1, 1, 1};
    const int v3[] = {3, 0};
    const int n3[] = {0, 1};
    const int v4[] = {0};
    const int n4[] = {1};

    if (tuple_init(&tuples[0], v0, n0, 2U) != 0) {
        return EXIT_FAILURE;
    }
    if (tuple_init(&tuples[1], v1, n1, 2U) != 0) {
        tuple_free(&tuples[0]);
        return EXIT_FAILURE;
    }
    if (tuple_init(&tuples[2], v2, n2, 3U) != 0) {
        tuple_free(&tuples[0]);
        tuple_free(&tuples[1]);
        return EXIT_FAILURE;
    }
    if (tuple_init(&tuples[3], v3, n3, 2U) != 0) {
        tuple_free(&tuples[0]);
        tuple_free(&tuples[1]);
        tuple_free(&tuples[2]);
        return EXIT_FAILURE;
    }
    if (tuple_init(&tuples[4], v4, n4, 1U) != 0) {
        tuple_free(&tuples[0]);
        tuple_free(&tuples[1]);
        tuple_free(&tuples[2]);
        tuple_free(&tuples[3]);
        return EXIT_FAILURE;
    }

    printf("Before: ");
    print_tuples(tuples, count);

    new_count = remove_all_none_tuples(tuples, count);

    printf("After:  ");
    print_tuples(tuples, new_count);

    for (i = 0U; i < new_count; i++) {
        tuple_free(&tuples[i]);
    }

    return EXIT_SUCCESS;
}
