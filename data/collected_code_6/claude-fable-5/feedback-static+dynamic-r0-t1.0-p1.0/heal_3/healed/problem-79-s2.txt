#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_WORD_LEN 256

static bool is_length_odd(const char *word, size_t max_len)
{
    size_t len;

    if (word == NULL) {
        return false;
    }

    len = strnlen(word, max_len);
    return (len % 2U) != 0U;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char buffer[MAX_WORD_LEN] = {0};
    size_t len;

    printf("Enter a word: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));

    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
    } else if (len == sizeof(buffer) - 1U) {
        discard_remaining_input();
    }

    if (buffer[0] == '\0') {
        fprintf(stderr, "No word entered\n");
        return EXIT_FAILURE;
    }

    if (is_length_odd(buffer, sizeof(buffer))) {
        printf("The word \"%s\" has an odd length.\n", buffer);
    } else {
        printf("The word \"%s\" has an even length.\n", buffer);
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses:**

Both Flawfinder findings are already properly mitigated in this code — the rewrite above preserves those mitigations:

1. **`getchar()` in a loop (CWE-120, CWE-20, risk 1):** This is a false positive in practical terms. The `discard_remaining_input()` function reads characters into an `int` and discards them — nothing is written to a buffer, so no overflow is possible. The loop correctly terminates on both `'\n'` and `EOF`, avoiding an infinite loop if the stream ends without a newline. Using `int ch` (not `char`) is correct, since `EOF` cannot be reliably distinguished from valid characters in a `char`.

2. **Statically-sized `char` array (CWE-119/CWE-120, risk 2):** The buffer is handled defensively:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size and guarantees null termination on success.
   - The explicit `buffer[sizeof(buffer) - 1U] = '\0';` is belt-and-suspenders termination.
   - `strnlen` (bounded) is used instead of `strlen`, so length computation cannot run past the buffer even if termination were somehow lost.
   - Overly long input lines are detected (`len == sizeof(buffer) - 1U` with no newline) and the remainder is drained, preventing stale input from leaking into subsequent reads.
   - The `fgets` return value is checked, and the empty-input case is handled with a proper error exit.

No functional changes were required — the original code already implements the appropriate bounds checking, input validation, and stream-hygiene fixes that address both warnings.
