#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int find_max_product_pair(const int *arr, size_t n, int *res1, int *res2) {
    if (arr == NULL || res1 == NULL || res2 == NULL || n < 2) {
        return -1;
    }

    long long max1, max2;
    long long min1, min2;

    if (arr[0] > arr[1]) {
        max1 = arr[0];
        max2 = arr[1];
        min1 = arr[1];
        min2 = arr[0];
    } else {
        max1 = arr[1];
        max2 = arr[0];
        min1 = arr[0];
        min2 = arr[1];
    }

    for (size_t i = 2; i < n; i++) {
        long long val = arr[i];
        
        if (val > max1) {
            max2 = max1;
            max1 = val;
        } else if (val > max2) {
            max2 = val;
        }

        if (val < min1) {
            min2 = min1;
            min1 = val;
        } else if (val < min2) {
            min2 = val;
        }
    }

    if ((max1 * max2) > (min1 * min2)) {
        *res1 = (int)max1;
        *res2 = (int)max2;
    } else {
        *res1 = (int)min1;
        *res2 = (int)min2;
    }

    return 0;
}

int main(void) {
    int arr[] = {1, -4, 3, -6, 2, 7, -5};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int a, b;

    if (find_max_product_pair(arr, n, &a, &b) == 0) {
        printf("Pair with highest product: %d and %d\n", a, b);
    } else {
        printf("Invalid input or array size too small.\n");
    }

    return 0;
}
