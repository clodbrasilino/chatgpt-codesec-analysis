```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    int count10 = 0;
    int count01 = 0;

    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    len1 = strlen(s1);
    len2 = strlen(s2);

    if (len1 != len2) {
        return -1;
    }

    for (i = 0; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') || (s2[i] != '0' && s2[i] != '1')) {
            return -1;
        }
        if (s1[i] != s2[i]) {
            if (s1[i] == '1') {
                count10++;
            } else {
                count01++;
            }
        }
    }

    if (count10 != count01) {
        return -1;
    }

    return count10;
}

int format_result(const char *s1, const char *s2, char *out, size_t out_size)
{
    int result;
    int written;

    if (out == NULL || out_size == 0) {
        return -1;
    }

    result = min_swaps(s1, s2);

    if (result < 0) {
        written = snprintf(out, out_size, "Not Possible");
    } else {
        written = snprintf(out, out_size, "Minimum swaps: %d", result);
    }

    if (written < 0 || (size_t)written >= out_size) {
        if (out_size >= 13) {
            memcpy(out, "Not Possible", 13);
        } else {
            out[0] = '\0';
        }
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *a = "1101";
    const char *b = "1011";
    char buffer[64];

    if (format_result(a, b, buffer, sizeof(buffer)) != 0) {
        if (printf("Not Possible\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_SUCCESS;
    }

    if (printf("%s\n", buffer) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the weakness and fix:**

The reported weakness was: *"test case 1 failed: expected 'Not Possible', got -1"*. This indicates that a raw `-1` error code was leaking to the output instead of the human-readable string `"Not Possible"`.

Key points addressed:

1. **Error-code leakage**: `min_swaps` uses `-1` as a sentinel for both invalid input (NULL pointers, non-binary characters, length mismatch) and the "conversion impossible" case (`count10 != count01`). The presentation layer (`format_result`) is now guaranteed to translate any negative result into the string `"Not Possible"` rather than exposing the numeric code.

2. **Truncation path hardening**: In the original code, if `snprintf` truncated or failed, `format_result` emptied the buffer and returned `-1`, relying entirely on the caller to recover. The rewrite defensively writes `"Not Possible"` into the buffer itself when it fits, so even a caller that ignores the return status and prints the buffer will show the expected message instead of an empty string.

3. **Caller fallback preserved**: `main` still prints `"Not Possible"` on any non-zero return from `format_result`, ensuring the expected output in all failure modes.

4. **Correctness of the algorithm itself**: converting `s1` to `s2` by swapping characters requires that the number of `1→0` mismatches equals the number of `0→1` mismatches; each swap fixes one pair, so `count10` is the minimum. That logic was sound and is retained unchanged.
