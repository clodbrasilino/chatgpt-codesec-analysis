#include <stdio.h>
#include <stdlib.h>

int min_number_to_make_sum_even(const int *arr, size_t size) {
    if (arr == NULL) {
        return -1; 
    }
    long long sum = 0;
    for (size_t i = 0; i < size; ++i) {
        sum += arr[i];
    }
    
    if (sum % 2 == 0) {
        return 0;
    } else {
        return 1;
    }
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    
    int result = min_number_to_make_sum_even(arr, size);
    if (result != -1) {
        printf("Minimum number to add: %d\n", result);
    } else {
        printf("Invalid input array.\n");
    }
    
    return 0;
}
