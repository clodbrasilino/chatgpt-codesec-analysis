#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    Node *tail;
} List;

Node* createNode(int value) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) return NULL;
    node->value = value;
    node->next = NULL;
    return node;
}

void appendToList(List *list, int value) {
    Node *newNode = createNode(value);
    if (list->head == NULL) {
        list->head = newNode;
        list->tail = newNode;
    } else {
        list->tail->next = newNode;
        list->tail = newNode;
    }
}

List* createList(void) {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) return NULL;
    list->head = NULL;
    list->tail = NULL;
    return list;
}

List* mergeFirstAndLast(List *lists, int size) {
    if (size < 1) return NULL;
    List *resultList = createList();
    for (int i = 0; i < size; ++i) {
        if (lists[i].head != NULL) {
            appendToList(resultList, lists[i].head->value);
        }
        if (lists[i].tail != NULL && (lists[i].head == NULL || lists[i].head != lists[i].tail)) {
            appendToList(resultList, lists[i].tail->value);
        }
    }
    return resultList;
}

void freeList(List *list) {
    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

void printList(List *list) {
    for (Node *current = list->head; current != NULL; current = current->next) {
        printf("%d ", current->value);
    }
    printf("\n");
}

int main() {
    List *lists[5];
    for (int i = 0; i < 5; ++i) {
        lists[i] = createList();
        appendToList(lists[i], 2 * i + 1);
        appendToList(lists[i], 2 * i + 2);
        appendToList(lists[i], 2 * i + 3);
    }

    List *mergedList = mergeFirstAndLast(lists, 5);

    printList(mergedList);

    for (int i = 0; i < 5; ++i) {
        freeList(lists[i]);
    }
    freeList(mergedList);

    return 0;
}
