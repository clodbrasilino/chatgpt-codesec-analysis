```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64
#define FLUSH_LIMIT 4096

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t count = 0;

    do {
        ch = getchar();
        count++;
    } while (ch != '\n' && ch != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis summary:**

The original code was already well-hardened, and the Flawfinder hits were largely informational rather than actual vulnerabilities:

1. **`char input[INPUT_SIZE]` (CWE-119/CWE-120):** Already mitigated — the code uses `fgets(input, sizeof(input), stdin)` which enforces the buffer boundary, forcibly null-terminates the last byte, and uses `strnlen` with an explicit bound instead of `strlen`. No change needed; the warning is a false positive.

2. **`getchar` in a loop (CWE-120/CWE-20):** `getchar` itself cannot overflow a buffer since nothing is stored, but an adversarial stream without a newline or EOF could theoretically spin the loop. I added a `FLUSH_LIMIT` counter as defense-in-depth so the drain loop is provably bounded.

Other correctness/safety properties preserved:

- **Integer overflow protection** in `octal_to_decimal` via the `value > (LONG_MAX - digit) / 8` pre-check before multiplication.
- **NULL and zero-length parameter validation** on the conversion function.
- **Truncation detection:** if input fills the buffer without a trailing newline, the excess is drained and the input rejected rather than silently truncated.
- **Strict character validation** rejecting anything outside `'0'`–`'7'`.
