#include <stdio.h>
#include <limits.h>

long long find_largest_k(long long x, long long y) {
    if (x <= 0 || y < 0 || y >= x) {
        return -1;
    }
    
    if (LLONG_MAX % x >= y) {
        return LLONG_MAX - (LLONG_MAX % x) + y;
    } else {
        return LLONG_MAX - (LLONG_MAX % x) + y - x;
    }
}

int main(void) {
    long long x, y;
    
    if (scanf("%lld %lld", &x, &y) != 2) {
        return 1;
    }
    
    long long result = find_largest_k(x, y);
    printf("%lld\n", result);
    
    return 0;
}
