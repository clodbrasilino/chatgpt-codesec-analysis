#include <stdio.h>
#include <stdlib.h>

static void reverse_range(int *array, size_t start, size_t end)
{
    while (start < end)
    {
        int temp = array[start];
        array[start] = array[end];
        array[end] = temp;
        start++;
        end--;
    }
}

int rotate_right(int *array, size_t size, size_t positions)
{
    if (array == NULL || size == 0U)
    {
        return -1;
    }

    positions %= size;

    if (positions == 0U)
    {
        return 0;
    }

    reverse_range(array, 0U, size - 1U);
    reverse_range(array, 0U, positions - 1U);
    reverse_range(array, positions, size - 1U);

    return 0;
}

static void print_array(const int *array, size_t size)
{
    size_t i;

    for (i = 0U; i < size; i++)
    {
        printf("%d ", array[i]);
    }
    putchar('\n');
}

int main(void)
{
    int numbers[] = {1, 2, 3, 4, 5, 6, 7};
    const size_t size = sizeof(numbers) / sizeof(numbers[0]);
    const size_t positions = 3U;

    printf("Original: ");
    print_array(numbers, size);

    if (rotate_right(numbers, size, positions) != 0)
    {
        fprintf(stderr, "Error: failed to rotate the array\n");
        return EXIT_FAILURE;
    }

    printf("Rotated right by %zu: ", positions);
    print_array(numbers, size);

    return EXIT_SUCCESS;
}
