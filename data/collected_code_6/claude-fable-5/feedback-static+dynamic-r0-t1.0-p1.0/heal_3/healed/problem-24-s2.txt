#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        int ch;
        int truncated = 0;

        while ((ch = getchar()) != '\n' && ch != EOF) {
            truncated = 1;
        }
        if (truncated == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}

**Analysis of the identified weaknesses and their fixes:**

1. **Flawfinder `char` warning (CWE-119/CWE-120) — statically-sized buffer:** The risk of overflowing `input[INPUT_SIZE]` is mitigated because all reads and string operations are bounded. `fgets` is called with `(int)sizeof(input)`, so it never writes past the buffer. Additionally, `input[sizeof(input) - 1U] = '\0'` guarantees null termination, and `strnlen` (rather than `strlen`) is used with an explicit maximum, preventing out-of-bounds reads even if termination were somehow missing.

2. **Flawfinder `getchar` warning (CWE-120/CWE-20) — unchecked loop input:** The `getchar` loop never stores the characters read; it only drains stdin until a newline or `EOF`, so no buffer can overflow. The loop also correctly terminates on `EOF`, avoiding an infinite loop on closed input streams, and the `truncated` flag ensures oversized input is rejected rather than silently processed as a partial value.

3. **Additional hardening already present:** `binary_to_decimal` validates its pointers, rejects empty or unterminated input (`len >= max_len`), validates every character strictly against `'0'`/`'1'`, and checks `value > (ULLONG_MAX >> 1)` before each shift to prevent unsigned overflow/silent wraparound — meaning inputs longer than 64 significant bits are rejected rather than producing an incorrect result.
