#include <stdio.h>
#include <stdlib.h>

char *int_to_roman(int num) {
    if (num < 1 || num > 3999) {
        return NULL;
    }

    const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    const char *symbols[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};

    char *result = malloc(32);
    if (result == NULL) {
        return NULL;
    }

    char *ptr = result;

    for (int i = 0; i < 13; i++) {
        while (num >= values[i]) {
            const char *sym = symbols[i];
            while (*sym) {
                *ptr++ = *sym++;
            }
            num -= values[i];
        }
    }
    *ptr = '\0';

    return result;
}

int main(void) {
    int test_cases[] = {3, 4, 9, 58, 1994, 3999, 0, 4000};
    int num_cases = sizeof(test_cases) / sizeof(test_cases[0]);

    for (int i = 0; i < num_cases; i++) {
        int num = test_cases[i];
        char *roman = int_to_roman(num);
        if (roman != NULL) {
            printf("%d: %s\n", num, roman);
            free(roman);
        } else {
            printf("%d: Invalid\n", num);
        }
    }

    return 0;
}
