```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256U

void squeeze_char(char *str, size_t max_len, char ch);
static void discard_remaining_input(void);

static void discard_remaining_input(void)
{
    int input_char;

    do {
        input_char = getchar();
    } while ((input_char != '\n') && (input_char != EOF));
}

void squeeze_char(char *str, size_t max_len, char ch)
{
    size_t read_idx;
    size_t write_idx;
    size_t len;

    if ((str == NULL) || (max_len == 0U)) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }

    write_idx = 0U;

    for (read_idx = 0U; read_idx < len; read_idx++) {
        if ((str[read_idx] == ch) &&
            (read_idx > 0U) &&
            (str[read_idx - 1U] == ch)) {
            continue;
        }
        str[write_idx] = str[read_idx];
        write_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char target;
    int input_char;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    } else if (len == (sizeof(buffer) - 1U)) {
        discard_remaining_input();
    }

    printf("Enter character to squeeze: ");
    input_char = getchar();
    if (input_char == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }

    target = (char)input_char;

    if (input_char != '\n') {
        discard_remaining_input();
    }

    squeeze_char(buffer, sizeof(buffer), target);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the identified weaknesses:**

1. **Flawfinder `getchar` warnings (CWE-120, CWE-20):** These are false positives in this code. In `discard_remaining_input()`, `getchar()` is used in a loop but the return value is only compared against `'\n'` and `EOF` — nothing is written to a buffer, so no overflow is possible. The return value is correctly stored in an `int` (not `char`), which is essential for reliable `EOF` detection. In `main()`, the single `getchar()` call is checked for `EOF` before use.

2. **Flawfinder `char buffer[BUFFER_SIZE]` warning (CWE-119/CWE-120):** This is mitigated correctly. Input is read with `fgets(buffer, (int)sizeof(buffer), stdin)`, which limits the read length. The buffer is defensively NUL-terminated at `sizeof(buffer) - 1U`, and length is measured with the bounded `strnlen` rather than `strlen`. When input fills the buffer without a newline, the remainder of the line is drained to prevent it from corrupting the next read.

3. **`squeeze_char` robustness:** The function guards against `NULL` and zero-length buffers, clamps `len` to `max_len - 1U`, and the in-place compaction guarantees `write_idx <= read_idx < len`, so the final `str[write_idx] = '\0'` is always in bounds.

No functional changes were required — the original code already contained the proper mitigations for every flagged item; the flagged lines are inherent Flawfinder heuristics (it flags every `getchar` and statically-sized `char` array regardless of context). The rewrite above simply removes the annotations while preserving all bounds checks, `EOF` handling, and input-draining logic.
