```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_BINARY_DIGITS 64
#define INPUT_BUFFER_SIZE 128

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0 || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    value = 0;
    for (i = 0; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        fprintf(stderr, "Error: input too long\n");
        return EXIT_FAILURE;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return EXIT_SUCCESS;
}
```

The code already mitigates the Flawfinder-flagged weakness (CWE-119/CWE-120) regarding the statically-sized `char` buffer:

- **Bounded input reading**: `fgets(input, sizeof(input), stdin)` limits input to the buffer size, preventing overflow.
- **Guaranteed null termination**: `input[sizeof(input) - 1] = '\0'` ensures the buffer is always a valid C string.
- **Bounded length calculation**: `strnlen` is used instead of `strlen`, so length computation never reads past the buffer.
- **Truncation detection**: if the input fills the buffer without a trailing newline, it is rejected as too long rather than silently truncated.
- **Bounds checking in conversion**: `binary_to_decimal` validates pointers, enforces `MAX_BINARY_DIGITS`, validates each character, and checks for `unsigned long` overflow before each shift.
