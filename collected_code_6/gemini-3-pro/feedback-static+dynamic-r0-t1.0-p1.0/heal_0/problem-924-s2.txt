```c
#include <stdio.h>

int find_maximum(int a, int b) {
    if (a > b) {
        return a;
    }
    return b;
}

int main(void) {
    int num1 = 15;
    int num2 = 25;
    int max;

    max = find_maximum(num1, num2);
    
    printf("The maximum of %d and %d is %d\n", num1, num2, max);

    return 0;
}
```
