```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int maxLenSubsequence(int arr[], int n, int diff) {
    if (n == 0) {
        return 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    int maxLen = 1;

    for (int i = 0; i < n; i++) {
        dp[i] = 1;
    }

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (abs(arr[i] - arr[j]) == diff && dp[i] < dp[j] + 1) {
                dp[i] = dp[j] + 1;
            }
        }
        if (maxLen < dp[i]) {
            maxLen = dp[i];
        }
    }

    free(dp);
    return maxLen;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5, 3, 2};
    int n = sizeof(arr) / sizeof(arr[0]);
    int diff = 1;

    int result = maxLenSubsequence(arr, n, diff);
    printf("Maximum length of subsequence with difference %d is %d\n", diff, result);

    return 0;
}
```
