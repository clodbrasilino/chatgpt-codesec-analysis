```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_SIZE 256

long ascii_sum(const char *str, size_t max_len)
{
    long sum = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    for (i = 0; i < len; i++) {
        sum += (long)(unsigned char)str[i];
    }

    return sum;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    long result;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = ascii_sum(buffer, sizeof(buffer));
    if (result < 0) {
        fprintf(stderr, "Invalid string\n");
        return EXIT_FAILURE;
    }

    printf("Sum of ASCII values: %ld\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120), which is a heuristic flag on any statically-sized `char` array rather than a confirmed vulnerability. The mitigations already in place:

1. **Bounded input**: `fgets(buffer, sizeof(buffer), stdin)` limits reads to the buffer size, preventing overflow — unlike `gets()` or unbounded `scanf("%s", ...)`.
2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant since `fgets` always null-terminates on success, but harmless belt-and-suspenders).
3. **Bounded length calculation**: `strnlen` instead of `strlen` caps traversal at the buffer size, protecting against non-terminated strings.
4. **Safe summation**: The cast to `(unsigned char)` before `(long)` prevents sign-extension issues with characters ≥ 0x80 on platforms where `char` is signed.
5. **NULL check and error propagation**: `ascii_sum` validates its input, and `main` checks both `fgets` failure and the sentinel return value.

The only change worth making is adding `#define _POSIX_C_SOURCE 200809L`, since `strnlen` is POSIX.1-2008 rather than standard C, ensuring its prototype is visible under strict compilation flags (`-std=c99 -pedantic`), avoiding implicit-declaration issues.
