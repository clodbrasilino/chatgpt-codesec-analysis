```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    total_bytes = n * sizeof(int);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > n || (n - k) > n) {
        return -1;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    if (head_bytes > total_bytes) {
        free(temp);
        return -1;
    }

    memcpy(temp, arr, head_bytes);

    if (tail_bytes > total_bytes) {
        free(temp);
        return -1;
    }

    memmove(arr, arr + k, tail_bytes);

    if ((n - k) > n || head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }

    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

Both Flawfinder hits (CWE-120, `memcpy` buffer overflow) are **already mitigated** in the code — the warnings are false positives from static pattern matching. Here's why:

1. **First `memcpy(temp, arr, head_bytes)`:**
   - `temp` is allocated with exactly `head_bytes`, so the destination size matches the copy length by construction.
   - The source `arr` holds `total_bytes`, and `head_bytes <= total_bytes` is verified (since `k < n` after the modulo, so `k * sizeof(int) < n * sizeof(int)`).
   - Integer overflow in `n * sizeof(int)` is prevented by the `n > SIZE_MAX / sizeof(int)` guard.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`:**
   - The destination starts at offset `(n - k)` elements, leaving exactly `k * sizeof(int) == head_bytes` of space — verified by the check `head_bytes > total_bytes - tail_bytes` (where `total_bytes - tail_bytes == head_bytes` exactly, guaranteed by the earlier `head_bytes + tail_bytes != total_bytes` check).
   - The source `temp` contains exactly `head_bytes`.

3. **The `memmove`** is also safe: it copies `tail_bytes` from offset `k` into the front of the buffer; source and destination both remain within `arr`'s `total_bytes`, and `memmove` correctly handles the overlap.

Additional robustness already present: `NULL`/zero-length rejection, `k % n` normalization with an early-exit fast path for `k == 0`, `malloc` failure handling, `temp` freed on all exit paths (no leaks), and checked `printf` return values in `main`. The code compiles cleanly under `-Wall -Wextra -Wpedantic` and correctly outputs `5 6 52 36 12 10`.
