#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256

static int compare_chars(const void *a, const void *b)
{
    unsigned char ca = *(const unsigned char *)a;
    unsigned char cb = *(const unsigned char *)b;

    if (ca < cb) {
        return -1;
    }
    if (ca > cb) {
        return 1;
    }
    return 0;
}

int sort_string(char *str, size_t max_len)
{
    size_t len;

    if (str == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    if (len > 1) {
        qsort(str, len, sizeof(char), compare_chars);
    }

    return 0;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (sort_string(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error sorting string\n");
        return EXIT_FAILURE;
    }

    printf("Sorted string: %s\n", buffer);

    return EXIT_SUCCESS;
}

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow at the read site — the classic CWE-120 vector.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant after `fgets`, but harmless and robust against future refactoring).
3. **Bounded length calculation:** `strnlen` is used instead of `strlen`, so even a non-terminated buffer cannot cause an out-of-bounds read.
