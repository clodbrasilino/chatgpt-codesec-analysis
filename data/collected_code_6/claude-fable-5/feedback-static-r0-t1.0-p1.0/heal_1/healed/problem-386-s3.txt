```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_SIZE 1024U

long min_swaps_for_balancing(const char *s, size_t max_len);

long min_swaps_for_balancing(const char *s, size_t max_len)
{
    long open_count = 0;
    long close_count = 0;
    long swaps = 0;
    long imbalance = 0;
    size_t i;
    size_t len;

    if (s == NULL) {
        return -1L;
    }

    len = strnlen(s, max_len);
    if (len >= max_len) {
        return -1L;
    }

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            open_count++;
        } else if (s[i] == ']') {
            close_count++;
        } else {
            return -1L;
        }
    }

    if (open_count != close_count) {
        return -1L;
    }

    open_count = 0;
    close_count = 0;

    for (i = 0U; i < len; i++) {
        if (s[i] == '[') {
            open_count++;
            if (imbalance > 0) {
                swaps += imbalance;
                imbalance--;
            }
        } else {
            close_count++;
            imbalance = close_count - open_count;
        }
    }

    return swaps;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    long result;
    size_t len;

    (void)memset(buffer, 0, sizeof(buffer));

    printf("Enter a bracket string (only '[' and ']'): ");

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    } else if (len == (sizeof(buffer) - 1U)) {
        int ch;
        int truncated = 0;
        while ((ch = getchar()) != EOF && ch != '\n') {
            truncated = 1;
        }
        if (truncated != 0) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    result = min_swaps_for_balancing(buffer, sizeof(buffer));

    if (result < 0) {
        fprintf(stderr, "Error: invalid or unbalanceable input string.\n");
        return EXIT_FAILURE;
    }

    printf("Minimum number of swaps required: %ld\n", result);

    return EXIT_SUCCESS;
}
```

Key changes addressing the flagged weaknesses:

1. **CWE-126 (`strlen` over-read):** Both calls to `strlen` were replaced with `strnlen`, bounded by the known buffer size. The function `min_swaps_for_balancing` now takes an explicit `max_len` parameter and rejects strings that are not terminated within bounds, so it can never over-read even if handed a non-terminated buffer.

2. **CWE-119/CWE-120 (statically-sized buffer):** The buffer size is defined once via `MAX_INPUT_SIZE`, the buffer is zero-initialized, an explicit terminator is written at the last index as a defensive guarantee, and input longer than the buffer is detected (by draining `stdin` and reporting truncation) rather than silently processing a partial string. `fgets` remains the bounded read primitive, which already prevents overflow.

3. **Robustness:** The algorithm itself (two-pass validation plus greedy swap counting) was correct and left intact; only the boundary and termination handling was hardened.
