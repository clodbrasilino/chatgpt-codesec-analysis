#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_SIZE 1024

long long countSubstrings(const char *str, size_t max_len)
{
    long long counts[256] = {0};
    long long result = 0;
    size_t i;
    size_t len;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);

    for (i = 0; i < len; i++) {
        counts[(unsigned char)str[i]]++;
    }

    for (i = 0; i < 256; i++) {
        result += (counts[i] * (counts[i] + 1)) / 2;
    }

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    size_t len;
    long long result;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    result = countSubstrings(buffer, sizeof(buffer));

    if (printf("Count of substrings: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

Analysis of the original code and how the rewrite addresses the flagged weakness:

**The Flawfinder warning (CWE-119/CWE-120, statically-sized buffer):** The warning on `char buffer[MAX_INPUT_SIZE]` is a low-risk heuristic hit. The code already mitigates it correctly, and the rewrite preserves all of those mitigations:

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits reads to the buffer size, unlike `gets()` or unbounded `scanf("%s")`. This is the primary defense against overflow.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is defensive belt-and-suspenders — `fgets` already null-terminates, but this makes the invariant explicit.
3. **Bounded length calculation:** `strnlen(buffer, sizeof(buffer))` instead of `strlen()` ensures no read past the buffer even if termination were somehow violated.
4. **Zero-initialization:** `memset(buffer, 0, sizeof(buffer))` prevents any use of uninitialized stack memory.
5. **Consistent use of `sizeof(buffer)`** rather than repeating the magic constant, so the size can't drift out of sync.

**Other correctness points worth noting:**

- The cast `(unsigned char)str[i]` is essential — indexing `counts[]` with a plain `char` would be undefined/negative indexing on platforms where `char` is signed and input contains bytes ≥ 0x80.
- `countSubstrings` guards against `NULL` and uses `strnlen` with a caller-supplied bound, making it safe even if handed a non-terminated buffer of known capacity.
- Overflow of the `long long` arithmetic is not a practical concern: with `len ≤ 1023`, the maximum result is `1023 * 1024 / 2`, far below `LLONG_MAX`.
- Return values of `fgets` and `printf` are checked, and `EXIT_SUCCESS`/`EXIT_FAILURE` are used for portable exit codes.
