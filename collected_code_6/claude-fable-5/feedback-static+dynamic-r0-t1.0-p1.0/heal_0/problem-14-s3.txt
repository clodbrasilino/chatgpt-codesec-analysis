```c
#include <stdio.h>
#include <stdlib.h>

double triangular_prism_volume(double base, double height, double length)
{
    return 0.5 * base * height * length;
}

int main(void)
{
    double base;
    double height;
    double length;

    printf("Enter the base of the triangle: ");
    if (scanf("%lf", &base) != 1) {
        fprintf(stderr, "Invalid input for base.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the height of the triangle: ");
    if (scanf("%lf", &height) != 1) {
        fprintf(stderr, "Invalid input for height.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the length of the prism: ");
    if (scanf("%lf", &length) != 1) {
        fprintf(stderr, "Invalid input for length.\n");
        return EXIT_FAILURE;
    }

    if (base <= 0.0 || height <= 0.0 || length <= 0.0) {
        fprintf(stderr, "All dimensions must be positive values.\n");
        return EXIT_FAILURE;
    }

    printf("Volume of the triangular prism: %.2f\n",
           triangular_prism_volume(base, height, length));

    return EXIT_SUCCESS;
}
```
