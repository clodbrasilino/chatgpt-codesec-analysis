```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MONTH_BUF_SIZE 32
#define INPUT_BUF_SIZE 64

static int has_28_days(const char *month)
{
    char normalized[MONTH_BUF_SIZE];
    size_t len;
    size_t i;

    if (month == NULL) {
        return 0;
    }

    len = strnlen(month, sizeof(normalized));
    if (len == 0 || len >= sizeof(normalized)) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);
}

int main(void)
{
    char input[INPUT_BUF_SIZE];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
    }

    if (has_28_days(input)) {
        printf("%s has 28 days.\n", input);
    } else {
        printf("%s does not have exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and how the code mitigates them:**

1. **`char normalized[MONTH_BUF_SIZE]` in `has_28_days` (CWE-119/CWE-120):** The code uses `strnlen` bounded by `sizeof(normalized)` and explicitly rejects any input where `len >= sizeof(normalized)`, guaranteeing the copy loop and the null terminator write stay within bounds. A `NULL` check also guards against invalid pointers.

2. **`getchar` in a loop (CWE-120/CWE-20):** `getchar` here never writes to a buffer — it only discards characters until newline or `EOF`. The `EOF` check prevents an infinite loop on closed streams. This is a false positive, and the implementation is safe as written.

3. **`char input[INPUT_BUF_SIZE]` in `main` (CWE-119/CWE-120):** Input is read with `fgets` bounded by `sizeof(input)`, which cannot overflow. The code defensively forces null termination, strips the newline only after verifying `len > 0`, and drains excess input from `stdin` when the buffer was filled, preventing leftover data from affecting subsequent reads.
