```c
#include <stdio.h>
#include <stdlib.h>

int sum_of_divisors(int n) {
    if (n <= 0) {
        return 0;
    }
    
    int sum = 0;
    for (int i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i != n / i) {
                sum += n / i;
            }
        }
    }
    return sum;
}

int main(void) {
    int numbers[] = {1, 6, 12, 28, 100, 0, -15};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    
    for (size_t i = 0; i < count; i++) {
        int result = sum_of_divisors(numbers[i]);
        printf("Sum of divisors of %d: %d\n", numbers[i], result);
    }
    
    return 0;
}
```
