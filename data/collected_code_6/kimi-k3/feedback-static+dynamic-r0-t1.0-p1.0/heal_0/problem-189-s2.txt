#include <stdio.h>
#include <stdlib.h>

int firstMissingPositive(int *nums, int numsSize) {
    if (nums == NULL || numsSize <= 0) {
        return 1;
    }
    
    for (int i = 0; i < numsSize; i++) {
        while (nums[i] > 0 && nums[i] <= numsSize && nums[nums[i] - 1] != nums[i]) {
            int temp = nums[nums[i] - 1];
            nums[nums[i] - 1] = nums[i];
            nums[i] = temp;
        }
    }
    
    for (int i = 0; i < numsSize; i++) {
        if (nums[i] != i + 1) {
            return i + 1;
        }
    }
    
    return numsSize + 1;
}

int main(void) {
    int arr1[] = {1, 2, 0};
    int arr2[] = {3, 4, -1, 1};
    int arr3[] = {7, 8, 9, 11, 12};
    int arr4[] = {1, 2, 3};
    
    printf("First missing positive in [1, 2, 0]: %d\n", firstMissingPositive(arr1, 3));
    printf("First missing positive in [3, 4, -1, 1]: %d\n", firstMissingPositive(arr2, 4));
    printf("First missing positive in [7, 8, 9, 11, 12]: %d\n", firstMissingPositive(arr3, 5));
    printf("First missing positive in [1, 2, 3]: %d\n", firstMissingPositive(arr4, 3));
    
    return 0;
}
