```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define SENTENCE_MAX 1024
#define WORD_MAX 256

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0U;
    }

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    size_t sent_len;
    size_t i;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = bounded_strlen(word, WORD_MAX);
    if (word_len == 0U || word_len >= WORD_MAX) {
        return 0;
    }

    sent_len = bounded_strlen(sentence, SENTENCE_MAX);
    if (sent_len >= SENTENCE_MAX || sent_len < word_len) {
        return 0;
    }

    for (i = 0U; i + word_len <= sent_len; i++) {
        int start_ok;
        int end_ok;

        if (strncmp(&sentence[i], word, word_len) != 0) {
            continue;
        }

        start_ok = (i == 0U) ||
                   (isalnum((unsigned char)sentence[i - 1U]) == 0);
        end_ok = (sentence[i + word_len] == '\0') ||
                 (isalnum((unsigned char)sentence[i + word_len]) == 0);

        if ((start_ok != 0) && (end_ok != 0)) {
            return 1;
        }
    }

    return 0;
}

static void strip_newline(char *str, size_t max_len)
{
    size_t len;

    if (str == NULL) {
        return;
    }

    len = bounded_strlen(str, max_len);
    if (len > 0U && str[len - 1U] == '\n') {
        str[len - 1U] = '\0';
    }
}

static int read_line(char *buffer, size_t buffer_size, const char *prompt)
{
    if (buffer == NULL || buffer_size == 0U || buffer_size > (size_t)INT_MAX) {
        return -1;
    }

    if (prompt != NULL) {
        printf("%s", prompt);
        (void)fflush(stdout);
    }

    if (fgets(buffer, (int)buffer_size, stdin) == NULL) {
        return -1;
    }

    buffer[buffer_size - 1U] = '\0';
    strip_newline(buffer, buffer_size);

    return 0;
}

int main(void)
{
    char sentence[SENTENCE_MAX];
    char word[WORD_MAX];

    memset(sentence, 0, sizeof(sentence));
    memset(word, 0, sizeof(word));

    if (read_line(sentence, sizeof(sentence), "Enter a sentence: ") != 0) {
        fprintf(stderr, "Error reading sentence.\n");
        return EXIT_FAILURE;
    }

    if (read_line(word, sizeof(word), "Enter a word: ") != 0) {
        fprintf(stderr, "Error reading word.\n");
        return EXIT_FAILURE;
    }

    if (is_word_present(sentence, word) != 0) {
        printf("The word \"%s\" is present in the sentence.\n", word);
    } else {
        printf("The word \"%s\" is not present in the sentence.\n", word);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary as an experienced C developer:**

The original code was already well-hardened against the Flawfinder warnings (CWE-119/CWE-120). The warnings flagged are informational (risk level 2) because statically-sized buffers *can* be misused, but here they are not:

1. **Bounded reads** – `fgets()` is used with `sizeof(buffer)`, preventing buffer overflow on input; there's no `gets()`, `strcpy()`, or unbounded `scanf("%s")`.
2. **Guaranteed termination** – buffers are zeroed with `memset()` and explicitly NUL-terminated after `fgets()`, so no unterminated-string reads occur.
3. **Bounded length computation** – `bounded_strlen()` never scans past the buffer capacity, avoiding out-of-bounds reads even if termination were somehow lost.
4. **Bounded comparison** – `strncmp()` with a validated `word_len`, and the loop condition `i + word_len <= sent_len` guarantees `sentence[i + word_len]` is a valid, in-bounds access (at worst the terminator).
5. **Safe `ctype` usage** – casting to `unsigned char` before `isalnum()` avoids undefined behavior with negative `char` values.

The rewrite keeps all these mitigations and additionally centralizes input handling in `read_line()`, validating the buffer size against `INT_MAX` before the cast to `int` required by `fgets()`, which removes a theoretical truncation issue in the original casts.
