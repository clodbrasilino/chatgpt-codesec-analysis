```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096

static int append_bytes(char *output, size_t output_size, size_t *out_pos,
                        const char *src, size_t len)
{
    if (output == NULL || out_pos == NULL || src == NULL) {
        return -1;
    }
    if (len == 0) {
        return 0;
    }
    if (output_size == 0 || *out_pos >= output_size - 1 ||
        len > output_size - *out_pos - 1) {
        return -1;
    }
    memmove(output + *out_pos, src, len);
    *out_pos += len;
    return 0;
}

int remove_whitespace(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    int ret;
    size_t out_pos = 0;
    size_t input_len;
    size_t remaining_len;
    const char *cursor;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN || input_len >= output_size) {
        return -1;
    }

    ret = regcomp(&regex, "[[:space:]]+", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    remaining_len = input_len;

    while (remaining_len > 0 && *cursor != '\0') {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            if (append_bytes(output, output_size, &out_pos, cursor,
                             remaining_len) != 0) {
                regfree(&regex);
                return -1;
            }
            break;
        }
        if (ret != 0) {
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo <= match.rm_so ||
            (size_t)match.rm_eo > remaining_len) {
            regfree(&regex);
            return -1;
        }

        if ((size_t)match.rm_so > 0) {
            if (append_bytes(output, output_size, &out_pos, cursor,
                             (size_t)match.rm_so) != 0) {
                regfree(&regex);
                return -1;
            }
        }

        cursor += (size_t)match.rm_eo;
        remaining_len -= (size_t)match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        return -1;
    }

    output[out_pos] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char test[] = "  Hello,   World!\tThis is\na test string.  ";
    size_t buffer_size;
    char *result;

    if (sizeof(test) == 0 || sizeof(test) > MAX_INPUT_LEN) {
        fprintf(stderr, "Input too large for output buffer\n");
        return EXIT_FAILURE;
    }

    buffer_size = sizeof(test);
    result = calloc(buffer_size, sizeof(char));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (remove_whitespace(test, result, buffer_size) != 0) {
        fprintf(stderr, "Failed to remove whitespace\n");
        free(result);
        return EXIT_FAILURE;
    }

    if (printf("Original: \"%s\"\n", test) < 0) {
        free(result);
        return EXIT_FAILURE;
    }
    if (printf("Result:   \"%s\"\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

The key change addressing the Flawfinder finding: the statically-sized `char result[MAX_INPUT_LEN]` array in `main` was replaced with a heap allocation sized exactly to the known input length (`sizeof(test)`), with the allocation checked for failure and freed on every exit path. The size is validated against `MAX_INPUT_LEN` before allocation, and `remove_whitespace` already performs strict bounds checking via `strnlen`, the `output_size` guard, and the length-checked `append_bytes` helper, so no unchecked writes into the buffer are possible.
