```c
#include <stdio.h>

int count_positive_numbers(const int *list, size_t length) {
    if (list == NULL) {
        return -1;
    }

    int count = 0;
    for (size_t i = 0; i < length; i++) {
        if (list[i] > 0) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int numbers[] = {-5, 3, 2, -1, 0, 8, -4, 7};
    size_t length = sizeof(numbers) / sizeof(numbers[0]);

    int result = count_positive_numbers(numbers, length);

    if (result >= 0) {
        printf("Positive numbers count: %d\n", result);
    } else {
        printf("Error: Invalid input list.\n");
        return 1;
    }

    return 0;
}
```
