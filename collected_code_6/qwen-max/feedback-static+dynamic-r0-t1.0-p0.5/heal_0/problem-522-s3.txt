#include <stdio.h>
#include <stdlib.h>

int *lis(int arr[], int n) {
    int *lis, i, j, max = 0;
    lis = (int *)malloc(n * sizeof(int));
    for (i = 0; i < n; i++) {
        lis[i] = 1;
    }
    for (i = 1; i < n; i++) {
        for (j = 0; j < i; j++) {
            if (arr[i] > arr[j] && lis[i] < lis[j] + 1) {
                lis[i] = lis[j] + 1;
            }
        }
    }
    return lis;
}

int *lds(int arr[], int n) {
    int *lds, i, j, max = 0;
    lds = (int *)malloc(n * sizeof(int));
    for (i = 0; i < n; i++) {
        lds[i] = 1;
    }
    for (i = n - 2; i >= 0; i--) {
        for (j = n - 1; j > i; j--) {
            if (arr[i] > arr[j] && lds[i] < lds[j] + 1) {
                lds[i] = lds[j] + 1;
            }
        }
    }
    return lds;
}

int longestBitonicSubsequence(int arr[], int n) {
    int *inc, *dec, i, max = 0;
    inc = lis(arr, n);
    dec = lds(arr, n);
    for (i = 0; i < n; i++) {
        if (max < inc[i] + dec[i] - 1) {
            max = inc[i] + dec[i] - 1;
        }
    }
    free(inc);
    free(dec);
    return max;
}

int main() {
    int arr[] = {1, 11, 2, 10, 4, 5, 2, 1};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("Length of Longest Bitonic Subsequence is %d\n", longestBitonicSubsequence(arr, n));
    return 0;
}
