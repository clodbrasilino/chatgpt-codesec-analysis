```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_SIZE 256

static void print_regex_error(int code, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_SIZE];

    memset(errbuf, 0, sizeof(errbuf));
    regerror(code, regex, errbuf, sizeof(errbuf) - 1U);
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (*offset + (size_t)end) <= input_len) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    *offset += (size_t)matches[0].rm_eo;
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "Regex execution failed");
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and fixes applied:**

1. **Flawfinder `char errbuf[ERRBUF_SIZE]` warnings (CWE-119/CWE-120):** These were largely false positives since the original code already passed `sizeof(errbuf)` to `regerror()` and null-terminated defensively. The rewrite hardens this further by zero-initializing the entire buffer with `memset()`, passing `sizeof(errbuf) - 1` to guarantee space for the terminator, and consolidating the duplicated error-handling logic into a single `print_regex_error()` helper — reducing the attack surface and eliminating code duplication.

2. **Flawfinder `memcpy` warning (CWE-120):** The original code was already safe because `len` is derived from validated match offsets and the destination is allocated as `len + 1`. The rewrite adds two extra defensive measures: an explicit upper-bound sanity check (`len < MAX_INPUT_LEN`) before allocation, and a guard against calling `memcpy` with zero-length copies.

3. **Existing good practices preserved:** NULL-pointer checks on parameters, `strnlen()` with an explicit cap instead of unbounded `strlen()`, validation of `rm_so`/`rm_eo` before arithmetic, bounds verification that the match lies within the input, `regfree()` on all exit paths after compilation, and `free()` of each returned allocation in `main()`. No memory leaks or unbounded operations remain.
