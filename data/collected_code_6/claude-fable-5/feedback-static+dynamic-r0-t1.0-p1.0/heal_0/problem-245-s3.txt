#include <stdio.h>
#include <stdlib.h>

int max_sum_bitonic_subsequence(const int *arr, size_t n)
{
    int *inc = NULL;
    int *dec = NULL;
    int max_sum = 0;
    size_t i;
    size_t j;

    if (arr == NULL || n == 0U) {
        return 0;
    }

    inc = malloc(n * sizeof(int));
    if (inc == NULL) {
        return -1;
    }

    dec = malloc(n * sizeof(int));
    if (dec == NULL) {
        free(inc);
        return -1;
    }

    for (i = 0; i < n; i++) {
        inc[i] = arr[i];
    }

    for (i = 1; i < n; i++) {
        for (j = 0; j < i; j++) {
            if (arr[i] > arr[j] && inc[i] < inc[j] + arr[i]) {
                inc[i] = inc[j] + arr[i];
            }
        }
    }

    for (i = 0; i < n; i++) {
        dec[i] = arr[i];
    }

    for (i = n; i-- > 0U;) {
        for (j = n - 1U; j > i; j--) {
            if (arr[i] > arr[j] && dec[i] < dec[j] + arr[i]) {
                dec[i] = dec[j] + arr[i];
            }
        }
    }

    max_sum = inc[0] + dec[0] - arr[0];
    for (i = 1; i < n; i++) {
        int candidate = inc[i] + dec[i] - arr[i];
        if (candidate > max_sum) {
            max_sum = candidate;
        }
    }

    free(inc);
    free(dec);

    return max_sum;
}

int main(void)
{
    int arr[] = {1, 15, 51, 45, 33, 100, 12, 18, 9};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result;

    result = max_sum_bitonic_subsequence(arr, n);
    if (result < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum sum of bitonic subsequence: %d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
