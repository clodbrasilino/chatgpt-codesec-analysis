#include <stdio.h>
#include <stdlib.h>

static unsigned long long delannoy(int n)
{
    unsigned long long **table;
    unsigned long long result;
    int i;
    int j;

    if (n < 0) {
        return 0ULL;
    }

    table = malloc((size_t)(n + 1) * sizeof(*table));
    if (table == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (i = 0; i <= n; i++) {
        table[i] = malloc((size_t)(n + 1) * sizeof(**table));
        if (table[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(table[j]);
            }
            free(table);
            fprintf(stderr, "Memory allocation failed\n");
            exit(EXIT_FAILURE);
        }
    }

    for (i = 0; i <= n; i++) {
        table[i][0] = 1ULL;
        table[0][i] = 1ULL;
    }

    for (i = 1; i <= n; i++) {
        for (j = 1; j <= n; j++) {
            table[i][j] = table[i - 1][j] + table[i - 1][j - 1] + table[i][j - 1];
        }
    }

    result = table[n][n];

    for (i = 0; i <= n; i++) {
        free(table[i]);
    }
    free(table);

    return result;
}

int main(void)
{
    int n;
    int status;
    unsigned long long value;

    printf("Enter n: ");
    status = scanf("%d", &n);
    if (status != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0 || n > 30) {
        fprintf(stderr, "n must be between 0 and 30\n");
        return EXIT_FAILURE;
    }

    value = delannoy(n);
    printf("Delannoy number D(%d, %d) = %llu\n", n, n, value);

    return EXIT_SUCCESS;
}
