#include <stdio.h>
#include <stdlib.h>

long long countSubsequences(const int arr[], int n, int k) {
    if (k <= 1) return 0;
    long long count = 0;
    for (int i = 1; i < (1 << n); i++) {
        long long product = 1;
        for (int j = 0; j < n; j++) {
            if (i & (1 << j)) {
                product *= arr[j];
                if (product >= k) break;
            }
        }
        if (product < k) count++;
    }
    return count;
}

int main() {
    const int arr[] = {1, 2, 3, 4};
    int n = 4;
    int k = 10;
    printf("Number of subsequences with product less than %d: %lld\n", k, countSubsequences(arr, n, k));
    return 0;
}
