#include <stdio.h>
#include <stdlib.h>

int entringer(int n, int k) {
    if (n <= 0 || k > n) return 0;
    if (n == 1 && k == 0) return 1;

    int **e = (int **)malloc((n + 1) * sizeof(int *));
    for (int i = 0; i <= n; ++i) {
        e[i] = (int *)malloc((k + 1) * sizeof(int));
    }

    e[1][0] = 1;
    for (int j = 1; j <= k; ++j) {
        e[1][j] = 0;
    }

    for (int i = 2; i <= n; ++i) {
        e[i][0] = 1;
        for (int j = 1; j <= k; ++j) {
            e[i][j] = e[i-1][j] + e[i][j-1];
        }
    }

    int result = e[n][k];

    for (int i = 0; i <= n; ++i) {
        free(e[i]);
    }
    free(e);

    return result;
}

int main() {
    int n = 5;
    int k = 3;
    printf("The Entringer number E(%d, %d) is: %d\n", n, k, entringer(n, k));
    return 0;
}
