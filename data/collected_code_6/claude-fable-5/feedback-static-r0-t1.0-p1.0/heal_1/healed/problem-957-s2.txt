```c
#include <stdio.h>
#include <stdlib.h>

int get_rightmost_set_bit_position(unsigned int n)
{
    int position;

    if (n == 0U)
    {
        return -1;
    }

    position = 1;
    while ((n & 1U) == 0U)
    {
        n >>= 1U;
        position++;
    }

    return position;
}

int main(void)
{
    unsigned int values[] = {18U, 12U, 7U, 0U, 128U};
    size_t count = sizeof(values) / sizeof(values[0]);
    size_t i;

    for (i = 0; i < count; i++)
    {
        int result = get_rightmost_set_bit_position(values[i]);
        if (result == -1)
        {
            if (printf("Number %u has no set bits\n", values[i]) < 0)
            {
                return EXIT_FAILURE;
            }
        }
        else
        {
            if (printf("Rightmost set bit of %u is at position %d\n", values[i], result) < 0)
            {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the `[variableScope]` finding: the `result` variable is now declared inside the `for` loop body rather than at function scope. This limits its lifetime and visibility to exactly where it is used, preventing accidental reuse of stale values across iterations and making the code easier to reason about. All other defensive measures from the original — the `-1` sentinel for zero input, unsigned literals in bitwise operations, `printf` return-value checks, and `size_t` for the loop index — remain intact.
