#include <stdio.h>

int findLastOccurrence(int *arr, int length, int target) {
    int start = 0, end = length - 1, result = -1;

    while (start <= end) {
        int mid = (start + end) / 2;

        if (arr[mid] == target) {
            result = mid;
            start = mid + 1;
        }
        else if (arr[mid] > target)
            end = mid - 1;
        else
            start = mid + 1;
    }
    return result;
}

int main() {
    int arr[] = {1, 2, 2, 2, 3, 4, 5, 5, 5, 9}; 
    int n = sizeof(arr) / sizeof(arr[0]); 
    int x = 2; 
    int index = findLastOccurrence(arr, n, x); 

    if (index != -1) 
        printf("Element %d last occurs at index %d\n", x, index); 
    else
        printf("Element %d is not present in array\n", x); 

    return 0; 
}
