```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

static double binomial_probability(unsigned int n, unsigned int k, double p);
static int read_uint(const char *prompt, unsigned int *out);
static int read_double(const char *prompt, double *out);

static double binomial_probability(unsigned int n, unsigned int k, double p)
{
    double log_coeff = 0.0;
    double log_prob;
    unsigned int i;

    if (k > n) {
        return 0.0;
    }

    if (p == 0.0) {
        return (k == 0U) ? 1.0 : 0.0;
    }

    if (p == 1.0) {
        return (k == n) ? 1.0 : 0.0;
    }

    for (i = 1U; i <= k; i++) {
        log_coeff += log((double)(n - k + i)) - log((double)i);
    }

    log_prob = log_coeff + (double)k * log(p) + (double)(n - k) * log(1.0 - p);

    return exp(log_prob);
}

static int read_uint(const char *prompt, unsigned int *out)
{
    long value;
    char buffer[64];
    char *endptr;

    printf("%s", prompt);
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return -1;
    }

    errno = 0;
    value = strtol(buffer, &endptr, 10);
    if (endptr == buffer || value < 0L || errno != 0) {
        return -1;
    }

    *out = (unsigned int)value;
    return 0;
}

static int read_double(const char *prompt, double *out)
{
    double value;
    char buffer[64];
    char *endptr;

    printf("%s", prompt);
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        return -1;
    }

    errno = 0;
    value = strtod(buffer, &endptr);
    if (endptr == buffer || errno != 0) {
        return -1;
    }

    *out = value;
    return 0;
}

int main(void)
{
    unsigned int n;
    unsigned int k;
    double p;
    double result;

    if (read_uint("Enter number of trials (n): ", &n) != 0) {
        fprintf(stderr, "Invalid input for n\n");
        return EXIT_FAILURE;
    }

    if (read_uint("Enter number of successes (k): ", &k) != 0) {
        fprintf(stderr, "Invalid input for k\n");
        return EXIT_FAILURE;
    }

    if (k > n) {
        fprintf(stderr, "k must not exceed n\n");
        return EXIT_FAILURE;
    }

    if (read_double("Enter probability of success (p): ", &p) != 0) {
        fprintf(stderr, "Invalid input for p\n");
        return EXIT_FAILURE;
    }

    if (p < 0.0 || p > 1.0) {
        fprintf(stderr, "p must be between 0 and 1\n");
        return EXIT_FAILURE;
    }

    result = binomial_probability(n, k, p);
    printf("Binomial probability P(X = %u) = %.10f\n", k, result);

    return EXIT_SUCCESS;
}
```
