```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096

static char *duplicate_word(const char *word, size_t len)
{
    char *copy;
    size_t alloc_size;

    if (word == NULL || len >= MAX_WORD_LEN) {
        return NULL;
    }

    if (len > SIZE_MAX - 1) {
        return NULL;
    }

    alloc_size = len + 1;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0) {
        if (len > alloc_size - 1) {
            free(copy);
            return NULL;
        }
        memcpy(copy, word, len);
    }
    copy[len] = '\0';

    return copy;
}

char **remove_duplicates(const char **words, size_t count, size_t *out_count)
{
    char **result = NULL;
    size_t unique = 0;
    size_t i;
    size_t j;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    result = calloc(count, sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        int found = 0;
        size_t len;

        if (words[i] == NULL) {
            continue;
        }

        len = strnlen(words[i], MAX_WORD_LEN);
        if (len >= MAX_WORD_LEN) {
            continue;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(result[j], words[i], MAX_WORD_LEN) == 0) {
                found = 1;
                break;
            }
        }

        if (!found) {
            result[unique] = duplicate_word(words[i], len);
            if (result[unique] == NULL) {
                for (j = 0; j < unique; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            unique++;
        }
    }

    *out_count = unique;
    return result;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
    }
    free(words);
}

int main(void)
{
    const char *input[] = {
        "apple", "banana", "apple", "cherry", "banana", "date", "apple"
    };
    size_t input_count = sizeof(input) / sizeof(input[0]);
    size_t result_count = 0;
    char **result;
    size_t i;

    result = remove_duplicates(input, input_count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to remove duplicates\n");
        return EXIT_FAILURE;
    }

    printf("Unique words:\n");
    for (i = 0; i < result_count; i++) {
        printf("%s\n", result[i]);
    }

    free_word_list(result, result_count);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weakness and why the code is safe as written:

**CWE-120 (memcpy buffer overflow) — already mitigated.** The Flawfinder warning on `memcpy` is a heuristic hit, not an actual vulnerability here, because the destination buffer size is provably sufficient at the point of the copy:

1. `len` is bounded by the caller via `strnlen(words[i], MAX_WORD_LEN)` and rejected if `len >= MAX_WORD_LEN`, so the source length is validated before `duplicate_word` is ever called.
2. Inside `duplicate_word`, the guard `len >= MAX_WORD_LEN` re-validates independently, so the function is safe even if called from a new code path.
3. `len > SIZE_MAX - 1` prevents integer overflow in `alloc_size = len + 1`.
4. The allocation is exactly `len + 1` bytes, and the redundant check `len > alloc_size - 1` guarantees `memcpy(copy, word, len)` writes at most `alloc_size - 1` bytes, leaving room for the NUL terminator written at `copy[len]`.

Other defensive properties worth noting: `calloc` overflow is prevented by the `count > SIZE_MAX / sizeof(char *)` check; all comparisons use bounded `strncmp`/`strnlen`; allocation failure mid-loop correctly frees all previously allocated strings before returning, avoiding leaks; and `free_word_list` is NULL-tolerant. The code stands as a correct fix for the flagged issue.
