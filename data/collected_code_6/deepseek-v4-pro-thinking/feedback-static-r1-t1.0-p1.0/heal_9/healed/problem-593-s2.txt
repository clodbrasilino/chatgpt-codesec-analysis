#include <stdio.h>
#include <stddef.h>

int remove_leading_zeroes(char *ip, size_t size) {
    if (ip == NULL || size == 0) {
        return -1;
    }

    size_t read = 0;
    size_t write = 0;
    int new_octet = 1;

    while (read < size) {
        char c = ip[read];
        if (c == '\0') {
            break;
        }

        if (c == '.') {
            if (write + 1 < size) {
                ip[write] = '.';
                write++;
            }
            read++;
            new_octet = 1;
            continue;
        }

        if (c < '0' || c > '9') {
            return -1;
        }

        if (new_octet) {
            while (read < size && ip[read] == '0') {
                if (read + 1 < size &&
                    ip[read + 1] >= '0' &&
                    ip[read + 1] <= '9') {
                    read++;
                } else {
                    break;
                }
            }
            new_octet = 0;
            if (read >= size) {
                break;
            }
            c = ip[read];
        }

        if (c == '\0') {
            break;
        }

        if (write + 1 >= size) {
            break;
        }

        ip[write] = c;
        write++;
        read++;
    }

    if (write < size) {
        ip[write] = '\0';
    } else if (size > 0) {
        ip[size - 1] = '\0';
    }

    return 0;
}

int main(void) {
    char ip[] = "192.168.001.001";

    if (remove_leading_zeroes(ip, sizeof(ip)) != 0) {
        return 1;
    }

    printf("%s\n", ip);
    return 0;
}
