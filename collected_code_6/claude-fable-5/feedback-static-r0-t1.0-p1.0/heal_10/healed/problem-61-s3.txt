```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U
#define MAX_DISCARD_ITERATIONS 65536U

long long count_substrings(const char *s, size_t max_len);
static int read_line(char *buf, size_t buf_size);
static void discard_remaining_input(void);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL || max_len == 0U) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n == max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;
    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        long long idx;

        prefix_sum += (long long)(s[i] - '0') - 1;
        idx = prefix_sum + offset;
        if (idx < 0 || (size_t)idx >= table_size) {
            free(freq);
            return -1;
        }
        count += freq[idx];
        freq[idx]++;
    }

    free(freq);
    return count;
}

static void discard_remaining_input(void)
{
    int c;
    size_t iterations;

    iterations = 0U;
    do {
        c = fgetc(stdin);
        iterations++;
    } while (c != '\n' && c != EOF && iterations < MAX_DISCARD_ITERATIONS);
}

static int read_line(char *buf, size_t buf_size)
{
    size_t len;

    if (buf == NULL || buf_size < 2U || buf_size > (size_t)INT32_MAX) {
        return -1;
    }

    if (fgets(buf, (int)buf_size, stdin) == NULL) {
        return -1;
    }

    buf[buf_size - 1U] = '\0';

    len = strcspn(buf, "\r\n");
    if (buf[len] == '\0' && len == buf_size - 1U) {
        discard_remaining_input();
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    memset(buffer, 0, sizeof buffer);

    printf("Enter a digit string: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (read_line(buffer, sizeof buffer) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long.\n");
        return EXIT_FAILURE;
    }

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("Number of substrings: %lld\n", result);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**1. Flawfinder `fgetc` warning (CWE-120, CWE-20) in `discard_remaining_input`:**
The loop is already properly bounded. It never writes into a buffer — it only consumes and discards characters — so no overflow is possible. Additionally, the `MAX_DISCARD_ITERATIONS` cap prevents an unbounded loop on a pathological stream, and both `'\n'` and `EOF` terminate the loop correctly (`c` is declared `int`, which is required to distinguish `EOF` from valid bytes).

**2. Flawfinder `char buffer[MAX_INPUT_LEN]` warning (CWE-119/CWE-120) in `main`:**
The statically-sized buffer is used safely throughout:
- `read_line` receives `sizeof buffer` and passes it to `fgets`, which limits the write to `buf_size - 1` characters plus the terminator.
- The explicit `buf[buf_size - 1U] = '\0'` guarantees termination even in edge cases.
- Truncated (over-long) input is detected via the `strcspn` check and rejected, with the leftover input drained so it cannot pollute subsequent reads.
- `count_substrings` uses `strnlen` bounded by `max_len` and rejects strings that fill the entire buffer, so it never reads past the allocation.

**Other defensive measures already present:**
- Overflow checks before computing `table_size` and before the `calloc` multiplication (`SIZE_MAX / sizeof *freq`).
- Index bounds validation (`idx < 0 || (size_t)idx >= table_size`) before every table access, with `freq` freed on the error path (no leak).
- `isdigit` is called with the value cast to `unsigned char`, avoiding undefined behavior on negative `char` values.
- `buf_size` is validated against `INT32_MAX` before the cast to `int` for `fgets`.

No functional changes were required; the code as structured already mitigates both Flawfinder findings, so the rewrite simply removes the comments as requested.
