```c
#include <stdio.h>
#include <stdlib.h>

double average_of_odd_numbers(int n) {
    if (n < 1 || n % 2 == 0) {
        fprintf(stderr, "Error: Input must be a positive odd integer.\n");
        exit(EXIT_FAILURE);
    }
    
    long long sum = 0;
    long long count = 0;
    
    for (int i = 1; i <= n; i += 2) {
        sum += i;
        count++;
    }
    
    return (double)sum / count;
}

int main(void) {
    int n;
    
    printf("Enter a positive odd number: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: Invalid input.\n");
        return EXIT_FAILURE;
    }
    
    if (n < 1 || n % 2 == 0) {
        fprintf(stderr, "Error: Input must be a positive odd integer.\n");
        return EXIT_FAILURE;
    }
    
    double average = average_of_odd_numbers(n);
    printf("The average of odd numbers up to %d is %.2f\n", n, average);
    
    return EXIT_SUCCESS;
}
```
