#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int areRotations(const char *str1, const char *str2) {
    size_t size1 = 0;
    const char *ptr = str1;
    while (*ptr != '\0') {
        ++size1;
        ++ptr;
    }
    if (str1[size1-1] != '\0') return 0;

    size_t size2 = 0;
    ptr = str2;
    while (*ptr != '\0') {
        ++size2;
        ++ptr;
    }
    if (str2[size2-1] != '\0') return 0;

    if (size1 != size2) return 0;

    char *temp = malloc(2 * size1 + 1);
    if (temp == NULL) return 0;
    snprintf(temp, 2 * size1 + 1, "%s%s", str1, str1);
    int result = strstr(temp, str2) != NULL;
    free(temp);
    return result;
}

int main() {
    const char s1[] = "abcd";
    const char s2[] = "cdab";
    if (areRotations(s1, s2)) {
        printf("Given strings are rotations of each other.\n");
    } else {
        printf("Given strings are not rotations of each other.\n");
    }
    return 0;
}
