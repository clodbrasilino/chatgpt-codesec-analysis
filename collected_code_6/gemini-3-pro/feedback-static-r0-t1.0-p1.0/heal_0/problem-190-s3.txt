#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long long count_points_inside_square(long long x1, long long y1, long long x2, long long y2, long long x3, long long y3, long long x4, long long y4) {
    long long dx = x1 - x2;
    long long dy = y1 - y2;
    long long side_squared = dx * dx + dy * dy;
    
    if (side_squared < 0) {
        return 0;
    }
    
    double side = sqrt((double)side_squared);
    long long n = (long long)round(side);
    
    if (n <= 1) {
        return 0;
    }
    
    return (n - 1) * (n - 1);
}

int main(void) {
    long long x1 = 0, y1 = 2;
    long long x2 = 2, y2 = 0;
    long long x3 = 4, y3 = 2;
    long long x4 = 2, y4 = 4;
    
    long long points = count_points_inside_square(x1, y1, x2, y2, x3, y3, x4, y4);
    
    if (points >= 0) {
        printf("%lld\n", points);
    } else {
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
