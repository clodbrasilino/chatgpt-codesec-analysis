#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>

static int compare_ints(const void *a, const void *b);
static bool find_max_frequency(const int *list, size_t size, int *result);

static int compare_ints(const void *a, const void *b) {
    int arg1 = *(const int *)a;
    int arg2 = *(const int *)b;
    
    if (arg1 < arg2) {
        return -1;
    }
    if (arg1 > arg2) {
        return 1;
    }
    return 0;
}

static bool find_max_frequency(const int *list, size_t size, int *result) {
    if (list == NULL || result == NULL || size == 0) {
        return false;
    }

    int *copy = malloc(size * sizeof(int));
    if (copy == NULL) {
        return false;
    }

    for (size_t i = 0; i < size; i++) {
        copy[i] = list[i];
    }

    qsort(copy, size, sizeof(int), compare_ints);

    int max_count = 1;
    int current_count = 1;
    int max_element = copy[0];

    for (size_t i = 1; i < size; i++) {
        if (copy[i] == copy[i - 1]) {
            current_count++;
        } else {
            if (current_count > max_count) {
                max_count = current_count;
                max_element = copy[i - 1];
            }
            current_count = 1;
        }
    }

    if (current_count > max_count) {
        max_element = copy[size - 1];
    }

    *result = max_element;
    
    free(copy);
    copy = NULL;

    return true;
}

int main(void) {
    const int data[] = {4, 1, 3, 2, 4, 1, 4, 3, 4, 5};
    size_t size = sizeof(data) / sizeof(data[0]);
    int most_frequent = 0;

    if (find_max_frequency(data, size, &most_frequent)) {
        printf("%d\n", most_frequent);
    } else {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
