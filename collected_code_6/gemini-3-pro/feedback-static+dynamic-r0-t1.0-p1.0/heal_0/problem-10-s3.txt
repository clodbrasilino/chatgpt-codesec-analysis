#include <stdio.h>
#include <stdlib.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void max_heapify(int arr[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest]) {
        largest = left;
    }

    if (right < n && arr[right] > arr[largest]) {
        largest = right;
    }

    if (largest != i) {
        swap(&arr[i], &arr[largest]);
        max_heapify(arr, n, largest);
    }
}

void build_max_heap(int arr[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) {
        max_heapify(arr, n, i);
    }
}

int* get_n_smallest(const int* dataset, size_t dataset_size, size_t n, size_t* out_size) {
    if (!dataset || !out_size) {
        return NULL;
    }

    if (n == 0 || dataset_size == 0) {
        *out_size = 0;
        return malloc(1);
    }

    if (n > dataset_size) {
        n = dataset_size;
    }

    int* heap = (int*)malloc(n * sizeof(int));
    if (!heap) {
        return NULL;
    }

    for (size_t i = 0; i < n; i++) {
        heap[i] = dataset[i];
    }

    build_max_heap(heap, n);

    for (size_t i = n; i < dataset_size; i++) {
        if (dataset[i] < heap[0]) {
            heap[0] = dataset[i];
            max_heapify(heap, n, 0);
        }
    }

    *out_size = n;
    return heap;
}

int main(void) {
    int dataset[] = {10, 5, 3, 2, 8, 7, 6, 1, 9, 4};
    size_t dataset_size = sizeof(dataset) / sizeof(dataset[0]);
    size_t n = 4;
    size_t out_size = 0;

    int* smallest_items = get_n_smallest(dataset, dataset_size, n, &out_size);

    if (smallest_items) {
        for (size_t i = 0; i < out_size; i++) {
            printf("%d ", smallest_items[i]);
        }
        printf("\n");
        free(smallest_items);
    } else {
        fprintf(stderr, "Memory allocation failed or invalid input.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
