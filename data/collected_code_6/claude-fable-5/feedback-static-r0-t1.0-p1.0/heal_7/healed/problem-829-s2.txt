```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_LEN 100

struct entry {
    char str[MAX_LEN];
    int count;
};

static const char *second_most_frequent(const char *arr[], size_t n,
                                        struct entry *table, size_t *table_size)
{
    size_t i;
    size_t j;
    size_t unique = 0;
    size_t first_idx;
    size_t second_idx;
    size_t len;
    int found_second = 0;

    if (arr == NULL || table == NULL || table_size == NULL || n == 0) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        int found = 0;

        if (arr[i] == NULL) {
            return NULL;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(table[j].str, arr[i], MAX_LEN) == 0) {
                if (table[j].count == INT_MAX) {
                    return NULL;
                }
                table[j].count++;
                found = 1;
                break;
            }
        }

        if (found == 0) {
            len = strnlen(arr[i], MAX_LEN);
            if (len >= MAX_LEN) {
                return NULL;
            }
            if (len >= sizeof(table[unique].str)) {
                return NULL;
            }
            memcpy(table[unique].str, arr[i], len);
            table[unique].str[len] = '\0';
            table[unique].count = 1;
            unique++;
        }
    }

    *table_size = unique;

    if (unique < 2) {
        return NULL;
    }

    first_idx = 0;
    second_idx = 0;

    for (i = 1; i < unique; i++) {
        if (table[i].count > table[first_idx].count) {
            first_idx = i;
        }
    }

    for (i = 0; i < unique; i++) {
        if (i == first_idx) {
            continue;
        }
        if (found_second == 0 || table[i].count > table[second_idx].count) {
            second_idx = i;
            found_second = 1;
        }
    }

    if (found_second == 0) {
        return NULL;
    }

    return table[second_idx].str;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "orange",
        "banana", "apple", "orange", "banana", "banana"
    };
    size_t n = sizeof(sequence) / sizeof(sequence[0]);
    struct entry *table;
    size_t table_size = 0;
    const char *result;

    table = calloc(n, sizeof(struct entry));
    if (table == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    result = second_most_frequent(sequence, n, table, &table_size);

    if (result == NULL) {
        fprintf(stderr, "No second most frequent string found\n");
        free(table);
        return EXIT_FAILURE;
    }

    if (printf("Second most frequent string: %s\n", result) < 0) {
        free(table);
        return EXIT_FAILURE;
    }

    free(table);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

1. **CWE-119/CWE-120 (statically-sized `char str[MAX_LEN]`):** The fixed-size buffer is safe because every write path into it is bounds-checked. Before any copy occurs, the code verifies `len < MAX_LEN` via `strnlen` (which itself is bounded and cannot over-read an unterminated source), and additionally verifies `len < sizeof(table[unique].str)` as a defense-in-depth check. Comparisons use `strncmp` with an explicit `MAX_LEN` bound rather than unbounded `strcmp`.

2. **CWE-120 (`memcpy` without bounds check):** This is a false positive in context. The `memcpy` copies exactly `len` bytes, and `len` is guaranteed to be strictly less than the destination size by the two preceding guards, leaving room for the explicit null terminator written afterward. The destination can always hold the source data.

Additional robustness already present: `NULL` checks on all pointer parameters and array elements, `INT_MAX` overflow guard on the counter increment, `calloc` return checked, `printf` return checked, and `table` is freed on every exit path. No changes to logic were required; the warnings are mitigated by the existing guards.
