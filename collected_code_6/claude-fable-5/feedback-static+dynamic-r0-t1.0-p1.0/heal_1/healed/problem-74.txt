```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

static bool follows_pattern(const char *pattern, const char *text)
{
    bool result = true;
    size_t plen;
    size_t tlen;
    size_t i = 0;
    size_t j;
    char *copy = NULL;
    char *token = NULL;
    char *map[256] = { NULL };

    if (pattern == NULL || text == NULL) {
        return false;
    }

    plen = strnlen(pattern, 4096U);
    tlen = strnlen(text, 4096U);

    if (plen >= 4096U || tlen >= 4096U) {
        return false;
    }

    copy = malloc(tlen + 1U);
    if (copy == NULL) {
        return false;
    }

    memcpy(copy, text, tlen);
    copy[tlen] = '\0';

    token = strtok(copy, " ");
    while (token != NULL) {
        if (i >= plen) {
            result = false;
            break;
        }
        unsigned char c = (unsigned char)pattern[i];
        if (map[c] != NULL) {
            if (strcmp(map[c], token) != 0) {
                result = false;
                break;
            }
        } else {
            for (j = 0U; j < 256U; j++) {
                if ((map[j] != NULL) && (strcmp(map[j], token) == 0)) {
                    result = false;
                    break;
                }
            }
            if (!result) {
                break;
            }
            map[c] = token;
        }
        i++;
        token = strtok(NULL, " ");
    }

    if (result && (i != plen)) {
        result = false;
    }

    free(copy);
    return result;
}

int main(void)
{
    const char *patterns[] = { "abba", "abba", "aaaa", "abba" };
    const char *texts[] = {
        "dog cat cat dog",
        "dog cat cat fish",
        "dog cat cat dog",
        "dog dog dog dog"
    };
    size_t count = sizeof(patterns) / sizeof(patterns[0]);
    size_t i;

    for (i = 0U; i < count; i++) {
        bool ok = follows_pattern(patterns[i], texts[i]);
        if (printf("Pattern \"%s\" with \"%s\": %s\n",
                   patterns[i], texts[i], ok ? "matches" : "does not match") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Summary of the changes made to address the identified weaknesses:**

1. **`strcpy` (CWE-120, risk 4)** — Replaced with `memcpy` using an explicitly computed length followed by manual null-termination. Since the buffer is allocated as `tlen + 1`, the copy is provably bounded and cannot overflow.

2. **`strlen` over-read (CWE-126, risk 1)** — Replaced both `strlen` calls with `strnlen` bounded to a sane maximum (4096). If the input is not null-terminated within that bound (or exceeds it), the function rejects the input rather than over-reading.

3. **Statically-sized array `map[256]` (CWE-119/CWE-120, risk 2)** — Retained but made safe by construction: the index is always cast to `unsigned char`, which guarantees a value in `[0, 255]`, so out-of-bounds access is impossible. The bounded pattern length check further constrains inputs.

The program's logic and output remain identical: it verifies whether a space-separated word sequence follows a given character pattern (the classic "word pattern" bijection problem), correctly handling both directions of the mapping constraint.
