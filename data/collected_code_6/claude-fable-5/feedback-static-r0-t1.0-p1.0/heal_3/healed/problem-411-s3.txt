```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t buf_size;
    size_t out_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        size_t prefix_len;

        if (match[0].rm_so < 0 || match[1].rm_so < 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        prefix_len = (size_t)match[0].rm_so;

        if (out_pos >= buf_size ||
            prefix_len >= buf_size - out_pos ||
            (buf_size - out_pos) - prefix_len < 2U) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, prefix_len);
        out_pos += prefix_len;

        result[out_pos] = (char)toupper((unsigned char)cursor[match[1].rm_so]);
        out_pos += 1U;

        cursor += match[0].rm_eo;
    }

    {
        size_t rest_len = strnlen(cursor, buf_size);

        if (out_pos >= buf_size ||
            rest_len >= buf_size - out_pos) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, rest_len);
        out_pos += rest_len;
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s -> %s\n", tests[i], converted);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes for the flagged weaknesses (CWE-120):**

Both `memcpy` calls flagged by Flawfinder are safe, but the original bounds checks contained a subtle risk: expressions like `out_pos + prefix_len + 1U >= buf_size` can wrap around on `size_t` overflow, theoretically bypassing the guard. The rewrite eliminates any possibility of integer overflow by using **subtraction-based bounds checks**:

1. **First `memcpy` (prefix copy):** Before copying, the code verifies `out_pos < buf_size`, then checks `prefix_len < buf_size - out_pos` and that at least 2 bytes remain (`(buf_size - out_pos) - prefix_len >= 2U`) — one for the uppercased character and one for the eventual NUL terminator. Since `out_pos < buf_size` is established first, the subtractions cannot underflow.

2. **Second `memcpy` (tail copy):** Similarly guarded with `rest_len < buf_size - out_pos`, guaranteeing room for both the tail and the terminating `'\0'`.

Additional defensive properties retained: input length is capped via `strnlen` against `MAX_INPUT_LEN`, `NULL` input and `malloc`/`regcomp` failures are handled, `regmatch_t` offsets are validated as non-negative before casting to `size_t`, and the output can never exceed the input length since the transformation only removes underscores — so `buf_size = len + 1` is always sufficient.
