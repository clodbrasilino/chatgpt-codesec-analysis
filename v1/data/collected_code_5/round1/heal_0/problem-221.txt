```C
#include <stdio.h>

int findFirstEvenNumber(int arr[], int size) {
    int i = 0;
    
    while(i < size) {
        if(arr[i] % 2 == 0) {
            return arr[i];
        }
        i++;
    }

    return -1;
}

int main() {
    int numbers[] = {1, 3, 5, 7, 9, 10};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    int firstEvenNumber = findFirstEvenNumber(numbers, size);

    if (firstEvenNumber != -1) {
        printf("First even number: %d\n", firstEvenNumber);
    } else {
        printf("No even numbers found\n");
    }   

    return 0;
}
```
