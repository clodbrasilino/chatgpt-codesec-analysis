```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static void print_regex_error(int errcode, const regex_t *regex, const char *prefix)
{
    size_t errbuf_size;
    char *errbuf;

    errbuf_size = regerror(errcode, regex, NULL, 0);
    errbuf = malloc(errbuf_size);
    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate memory for error message\n", prefix);
        return;
    }
    regerror(errcode, regex, errbuf, errbuf_size);
    fprintf(stderr, "%s: %s\n", prefix, errbuf);
    free(errbuf);
}

int search_pattern(const char *text, const char *pattern, regoff_t *start, regoff_t *end)
{
    regex_t regex;
    regmatch_t match;
    int ret;

    if (text == NULL || pattern == NULL || start == NULL || end == NULL) {
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return -1;
    }

    ret = regexec(&regex, text, 1, &match, 0);
    if (ret == 0) {
        *start = match.rm_so;
        *end = match.rm_eo;
        regfree(&regex);
        return 0;
    } else if (ret == REG_NOMATCH) {
        regfree(&regex);
        return 1;
    } else {
        print_regex_error(ret, &regex, "Regex execution failed");
        regfree(&regex);
        return -1;
    }
}

int main(void)
{
    const char *text = "The quick brown fox jumps over the lazy dog";
    const char *pattern = "brown fox";
    regoff_t start = 0;
    regoff_t end = 0;
    int result;

    result = search_pattern(text, pattern, &start, &end);

    if (result == 0) {
        printf("Pattern \"%s\" found in string.\n", pattern);
        printf("Match starts at index %ld and ends at index %ld.\n",
               (long)start, (long)end);
        printf("Matched substring: \"%.*s\"\n",
               (int)(end - start), text + start);
    } else if (result == 1) {
        printf("Pattern \"%s\" not found in string.\n", pattern);
    } else {
        fprintf(stderr, "An error occurred during pattern search.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The original code was flagged by Flawfinder for the statically-sized `char errbuf[256]` arrays (CWE-119/CWE-120). While the original usage was technically safe—`regerror()` respects the `sizeof(errbuf)` bound and always null-terminates—the fixed size carried two residual concerns: potential truncation of long error messages, and the general risk pattern of fixed-size stack buffers.

The rewritten version eliminates the static buffers entirely by:

1. **Two-pass `regerror()` usage** — Calling `regerror(errcode, regex, NULL, 0)` first returns the exact buffer size needed (including the null terminator), so the message is never truncated regardless of length.

2. **Dynamic allocation with failure handling** — The buffer is heap-allocated to the exact required size, with a graceful fallback message if `malloc()` fails, and is freed immediately after use to avoid leaks.

3. **Factored error reporting** — Both error paths (compilation and execution failures) previously duplicated the flagged code; consolidating into `print_regex_error()` removes the duplication and ensures a single, audited code path.

All original behavior is preserved: null-pointer parameter validation, correct `regfree()` on every exit path after successful compilation, the 0/1/-1 return convention, and the bounded `%.*s` substring printing in `main()`.
