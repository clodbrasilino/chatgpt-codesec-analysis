#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size);

int remove_lowercase(const char *input, size_t input_size, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_len = 0;
    size_t consumed = 0;
    int rc;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    if (strnlen(input, input_size) >= input_size) {
        return -1;
    }

    rc = regcomp(&regex, "[a-z]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (consumed < input_size && *cursor != '\0') {
        rc = regexec(&regex, cursor, 1, &match, 0);
        if (rc == REG_NOMATCH) {
            size_t remaining = strnlen(cursor, input_size - consumed);
            if (out_len >= output_size || remaining >= output_size - out_len) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_len, cursor, remaining);
            out_len += remaining;
            break;
        }
        if (rc != 0) {
            regfree(&regex);
            return -1;
        }
        if (match.rm_so > 0) {
            size_t prefix_len = (size_t)match.rm_so;
            if (out_len >= output_size || prefix_len >= output_size - out_len) {
                regfree(&regex);
                return -1;
            }
            memcpy(output + out_len, cursor, prefix_len);
            out_len += prefix_len;
        }
        if (match.rm_eo <= 0) {
            regfree(&regex);
            return -1;
        }
        if ((size_t)match.rm_eo > input_size - consumed) {
            regfree(&regex);
            return -1;
        }
        cursor += match.rm_eo;
        consumed += (size_t)match.rm_eo;
    }

    if (out_len >= output_size) {
        regfree(&regex);
        return -1;
    }
    output[out_len] = '\0';
    regfree(&regex);
    return 0;
}

int main(void)
{
    static const char *test_strings[] = {
        "HelloWORLDfooBAR",
        "abcDEFghiJKL",
        "ALLUPPERCASE",
        "alllowercase",
        "Mix123edCaseWith456numbers"
    };
    enum { MAX_INPUT_LEN = 256 };
    size_t count = sizeof(test_strings) / sizeof(test_strings[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        size_t input_len = strnlen(test_strings[i], MAX_INPUT_LEN);
        size_t buf_size;
        char *result;

        if (input_len >= MAX_INPUT_LEN) {
            fprintf(stderr, "Input string too long\n");
            return EXIT_FAILURE;
        }

        buf_size = input_len + 1;
        result = malloc(buf_size);
        if (result == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (remove_lowercase(test_strings[i], input_len + 1, result, buf_size) == 0) {
            printf("Input:  \"%s\"\n", test_strings[i]);
            printf("Output: \"%s\"\n\n", result);
        } else {
            fprintf(stderr, "Failed to process string: %s\n", test_strings[i]);
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }

    return EXIT_SUCCESS;
}

The original code was already well-defended against the Flawfinder warnings. Both flagged `memcpy` calls are preceded by explicit bounds checks (`out_len >= output_size || len >= output_size - out_len`) written in overflow-safe form — the subtraction is only performed after confirming `out_len < output_size`, so no unsigned wraparound can occur. Flawfinder flags `memcpy` unconditionally as a pattern match (CWE-120); these are false positives here.

Key safety properties preserved and strengthened in the rewrite:

1. **Input validation** — `NULL` checks, zero-size output rejection, and `strnlen` verification that the input is properly NUL-terminated within `input_size`.
2. **Bounds-checked copies** — every `memcpy` destination write is guaranteed to fit, with room reserved for the trailing NUL (`>=` comparisons rather than `>`).
3. **Added hardening** — an extra check that `match.rm_eo` does not advance the cursor beyond `input_size - consumed`, preventing any pathological regex offset from causing an out-of-bounds cursor read.
4. **Resource management** — `regfree` is called on every exit path after `regcomp` succeeds, and `main` frees allocations on both success and failure paths, so there are no leaks.
5. **Guaranteed NUL termination** — the output buffer is always terminated before returning success.
