#include <stdio.h>
#include <limits.h>

int max_volume_cuboid(long sum) {
    long maxVol = 0;
    for (long i = 1; i < sum - 2; i++) {
        for (long j = 1; j < sum - i - 1; j++) {
            long k = sum - i - j;
            if (k > 0) {
                long vol = i * j * k;
                if (vol > maxVol) {
                    maxVol = vol;
                }
            }
        }
    }
    return maxVol;
}

int main() {
    int sum;
    printf("Enter the sum of the sides: ");
    if (scanf("%d", &sum) != 1 || sum < 3) {
        printf("Invalid input\n");
        return 1;
    }
    long maxVol = max_volume_cuboid(sum);
    printf("Maximum volume: %ld\n", maxVol);
    return 0;
}
