#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (src == NULL) {
        return -1;
    }
    if (n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*len + n + 1 > *cap) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

char *increment_numerics(const char *input, long k)
{
    if (input == NULL) {
        return NULL;
    }
    char *out = NULL;
    size_t len = 0;
    size_t cap = 0;
    size_t i = 0;
    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            size_t start = i;
            while (isdigit((unsigned char)input[i])) {
                i++;
            }
            size_t nlen = i - start;
            long long value = 0;
            if (parse_number(input + start, nlen, &value) != 0) {
                free(out);
                return NULL;
            }
            if ((k > 0 && value > LLONG_MAX - k) ||
                (k < 0 && value < LLONG_MIN - k)) {
                free(out);
                return NULL;
            }
            value += k;
            char numbuf[32];
            int written = snprintf(numbuf, sizeof numbuf, "%lld", value);
            if (written < 0 || (size_t)written >= sizeof numbuf) {
                free(out);
                return NULL;
            }
            if (buf_append(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;
            }
        } else {
            if (buf_append(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    if (out == NULL) {
        out = malloc(1);
        if (out == NULL) {
            return NULL;
        }
        out[0] = '\0';
    }
    return out;
}

int main(void)
{
    const char *tests[] = {
        "item 12 costs 45 dollars",
        "abc100def200",
        "no numbers here",

    long k = 5;
    size_t count = sizeof tests / sizeof tests[0];
    size_t i;

    for (i = 0; i < count; i++) {
        char *result = increment_numerics(tests[i], k);
        if (result == NULL) {
            if (fprintf(stderr, "Failed to process: %s\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
            return EXIT_FAILURE;
        }
        if (printf("Input : %s\nOutput: %s\n\n", tests[i], result) < 0) {
            free(result);
            return EXIT_FAILURE;

        free(result);

**Analysis of the original code and the fixes applied:**

**Weakness 1 — `memcpy` in `buf_append` (CWE-120):** The original code was already safe because capacity is guaranteed before copying, but I added a defensive post-condition check (`*len + n + 1 > *cap`) after the growth logic and a `NULL` guard on `src`, so the `memcpy` destination is provably large enough on every path.

**Weakness 2 — `memcpy` into `numstr` (CWE-120):** This was the more interesting one. The original code heap-allocated a temporary string just to feed `strtoll`. I eliminated the allocation, the `memcpy`, and the `strtoll` entirely by parsing the digit run in place with `parse_number`, which performs explicit per-digit overflow checking against `LLONG_MAX`. This removes the flagged copy, removes a `malloc`/`free` pair (and two error-path leak opportunities), and avoids `errno`-based error handling.

**Weakness 3 — statically-sized `char numbuf[32]` (CWE-119/CWE-120):** The buffer is safe: `LLONG_MIN` formats to 20 characters plus the NUL, well under 32. The `snprintf` return value is checked for both negative results and truncation (`written >= sizeof numbuf`), which is the correct mitigation, so this pattern is retained.

**Other notes:** All error paths free `out` before returning, the addition of `k` is overflow-checked in both directions, and empty input still yields a valid heap-allocated empty string that the caller can `free`. The code remains strictly C99-portable.
