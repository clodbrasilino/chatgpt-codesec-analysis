```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;
    char *shrunk;

    if (input == NULL) {
        return NULL;
    }

    len = 0;
    while (len < MAX_INPUT_LEN && input[len] != '\0') {
        len++;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    shrunk = realloc(result, j + 1);
    if (shrunk != NULL) {
        result = shrunk;
        result[j] = '\0';
    }

    return result;
}

int main(void)
{
    const char *tests[] = {
        "py(remove me)thon",
        "str(nested (inner) text)ing",
        "al(unwanted)pha"
    };
    const char *expected[] = {
        "python",
        "string",
        "alpha"
    };
    size_t i;
    size_t count;
    int status;

    count = sizeof(tests) / sizeof(tests[0]);
    status = EXIT_SUCCESS;

    for (i = 0; i < count; i++) {
        char *output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (strcmp(output, expected[i]) != 0) {
            fprintf(stderr, "test case %zu failed: expected '%s', got '%s'\n",
                    i, expected[i], output);
            status = EXIT_FAILURE;
        } else {
            printf("%s\n", output);
        }
        free(output);
    }

    return status;
}
```

**Analysis notes:**

The core algorithm in `remove_parentheses` is actually sound — it correctly tracks nesting depth, copies only characters at depth zero, and handles nested parentheses like `"str(nested (inner) text)ing"` → `"string"`. Tracing through all three test cases yields `python`, `string`, and `alpha` as expected.

The "test case failed... got <no output>" messages in the comment suggest the original failing version likely had one of these defects, all of which are guarded against here:

1. **Missing null terminator** — `result[j] = '\0'` is written both after the copy loop and again after a successful `realloc`, so the string is always terminated even if `realloc` moves the block.
2. **Unchecked `realloc` clobbering the pointer** — the shrink result is stored in `shrunk` first; on failure the original valid buffer is kept rather than leaking or dereferencing `NULL`.
3. **Unbounded input scan** — length is capped at `MAX_INPUT_LEN`, preventing runaway reads on non-terminated input.
4. **Unbalanced closing parens** — `depth` is only decremented when positive, so a stray `)` can't drive depth negative and suppress subsequent output.
5. **NULL input and allocation failures** — both return `NULL`, and `main` checks for it before use, freeing every allocation to avoid leaks.

The program returns `EXIT_FAILURE` if any test mismatches, making it suitable for automated test harnesses.
