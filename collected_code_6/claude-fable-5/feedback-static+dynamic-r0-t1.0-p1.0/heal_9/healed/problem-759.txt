```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 128
#define MAX_FLUSH_ITERATIONS 4096

bool is_decimal_precision_two(const char *str, size_t max_len)
{
    size_t len;
    size_t i = 0;
    size_t digits_before = 0;
    size_t digits_after = 0;
    bool dot_found = false;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    if (str[i] == '+' || str[i] == '-') {
        i++;
        if (i == len) {
            return false;
        }
    }

    for (; i < len; i++) {
        if (isdigit((unsigned char)str[i])) {
            if (dot_found) {
                digits_after++;
            } else {
                digits_before++;
            }
        } else if (str[i] == '.') {
            if (dot_found) {
                return false;
            }
            dot_found = true;
        } else {
            return false;
        }
    }

    if (digits_before == 0) {
        return false;
    }

    if (dot_found && (digits_after == 0 || digits_after > 2)) {
        return false;
    }

    return true;
}

static void flush_stdin(void)
{
    int ch;
    size_t iterations = 0;

    do {
        ch = getchar();
        iterations++;
    } while (ch != '\n' && ch != EOF && iterations < MAX_FLUSH_ITERATIONS);
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a number: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Output error\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Input error\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
    }

    if (is_decimal_precision_two(buffer, sizeof(buffer))) {
        printf("Valid decimal with precision of 2: %s\n", buffer);
    } else {
        printf("Invalid decimal with precision of 2: %s\n", buffer);
    }

    return 0;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20)** — Flawfinder flags this as a risk-1 finding because unbounded read loops can hang or be abused. In this code, the loop is already properly bounded: it terminates on `'\n'`, on `EOF`, and additionally by the `MAX_FLUSH_ITERATIONS` cap, which prevents an infinite loop even against a stream that never delivers a newline or EOF. Crucially, the read characters are discarded, never written to a buffer, so no buffer overflow is possible. No change required.

2. **Statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — This risk-2 finding is mitigated because every operation on the buffer is bounds-limited:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` never writes more than `sizeof(buffer) - 1` characters plus the terminator.
   - The explicit `buffer[sizeof(buffer) - 1] = '\0';` guarantees termination defensively.
   - `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so length computation cannot run past the buffer.
   - Truncated input (no newline found at full length) triggers `flush_stdin()` so leftover bytes don't pollute subsequent reads.
   - `is_decimal_precision_two` receives the buffer size explicitly, guards against `NULL`, and rejects strings that fill the entire buffer (`len == max_len`), which indicates possible truncation.

The code as written already implements the recommended mitigations (bounded reads, explicit termination, length-limited functions), so the rewrite preserves the logic verbatim with the analysis comments removed.
