#include <stdio.h>
#include <string.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_WORDS 1000
#define MAX_WORD_LENGTH 50

char* find_first_repeated_word(const char *str) {
    char **words = (char **)malloc(MAX_WORDS * sizeof(char *));
    if (words == NULL) return NULL;
    for (int i = 0; i < MAX_WORDS; i++) {
        words[i] = (char *)malloc(MAX_WORD_LENGTH + 1);
        if (words[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(words[j]);
            }
            free(words);
            return NULL;
        }
    }

    int word_count = 0;

    const char *p = str;
    while (*p) {
        if (word_count >= MAX_WORDS) break;

        const char *start = p;
        while (*p && *p != ' ') p++;

        int len = p - start;
        if (len > 0 && len < MAX_WORD_LENGTH) {
            for (int i = 0; i < word_count; i++) {
                if (strncmp(words[i], start, len) == 0 && words[i][len] == '\0') {
                    for (int j = 0; j < word_count; j++) {
                        free(words[j]);
                    }
                    free(words);
                    return (char *)start;
                }
            }
            if (len + 1 <= MAX_WORD_LENGTH) {
                strncpy(words[word_count], start, len);
                words[word_count][len] = '\0';
                word_count++;
            }
        }

        while (*p == ' ') p++;
    }

    for (int i = 0; i < word_count; i++) {
        free(words[i]);
    }
    free(words);

    return NULL;
}

int main() {
    const char *input = "This is a test string with a repeated word and the word is repeated";
    char *result = find_first_repeated_word(input);

    if (result) {
        printf("First repeated word: %s\n", result);
    } else {
        printf("No repeated word found.\n");
    }

    return 0;
}
