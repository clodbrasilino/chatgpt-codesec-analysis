#include <stdio.h>
#include <limits.h>

int calculate_dog_years(int human_years) {
    if (human_years < 0) {
        return -1;
    }
    if (human_years <= 2) {
        return human_years * 10.5;
    }
    int additional_years = human_years - 2;
    if (additional_years > (INT_MAX - 21) / 4) {
        return -1;
    }
    return 21 + additional_years * 4;
}

int main() {
    int human_age;
    printf("Enter the dog's age in human years: ");
    if (scanf("%d", &human_age) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    long dog_age = calculate_dog_years(human_age);
    if (dog_age == -1) {
        printf("Age cannot be negative or too large to compute.\n");
    } else {
        printf("The dog's age in dog years is: %ld\n", dog_age);
    }
    return 0;
}
