#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

bool is_woodall(unsigned long long n) {
    if (n < 1) {
        return false;
    }
    
    if (n == 1) {
        return true;
    }

    unsigned int i = 1;

    while (i < 60) {
        unsigned long long power = 1ULL << i;
        unsigned long long limit = ULLONG_MAX / i;
        
        if (power > limit) {
            break;
        }
        
        unsigned long long w = i * power;
        
        if (w == 0) {
            break;
        }
        
        w -= 1;
        
        if (w == n) {
            return true;
        }
        
        if (w > n) {
            break;
        }
        
        i++;
    }

    return false;
}

int main(void) {
    unsigned long long num = 0;
    
    if (scanf("%llu", &num) != 1) {
        return EXIT_FAILURE;
    }

    if (is_woodall(num)) {
        printf("1\n");
    } else {
        printf("0\n");
    }

    return EXIT_SUCCESS;
}
