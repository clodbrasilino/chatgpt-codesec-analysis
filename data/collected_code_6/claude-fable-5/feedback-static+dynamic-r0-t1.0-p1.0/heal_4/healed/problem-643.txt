#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define BUFFER_SIZE 256

int contains_interior_z(const char *word, size_t len)
{
    size_t i;

    if (word == NULL || len < 3) {
        return 0;
    }

    for (i = 1; i < len - 1; i++) {
        if (word[i] == 'z' || word[i] == 'Z') {
            return 1;
        }
    }

    return 0;
}

int find_word_with_interior_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    result[0] = '\0';

    while (text[i] != '\0') {
        size_t start;
        size_t len;

        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }

        start = i;

        while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
            i++;
        }

        len = i - start;

        if (len > 0 && contains_interior_z(text + start, len)) {
            if (len >= result_size) {
                return -1;
            }
            memcpy(result, text + start, len);
            result[len] = '\0';
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    const char *tests[] = {
        "the lazy dog jumped",
        "zebra runs fast",
        "jazz music is nice",
        "no matching words here",
        "amazing puzzle pieces"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    char buffer[BUFFER_SIZE];
    size_t i;

    for (i = 0; i < num_tests; i++) {
        int status = find_word_with_interior_z(tests[i], buffer, sizeof(buffer));
        if (status == 0) {
            printf("Input: \"%s\" -> Match: \"%s\"\n", tests[i], buffer);
        } else if (status == 1) {
            printf("Input: \"%s\" -> No match found\n", tests[i]);
        } else {
            fprintf(stderr, "Error processing input: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why this code is safe:

**1. Flawfinder `memcpy` warning (CWE-120) in `find_word_with_interior_z`:**
This is a false positive. The copy is properly guarded by the check `if (len >= result_size) { return -1; }` immediately before the `memcpy`. Using `>=` (rather than `>`) correctly reserves one byte for the null terminator written afterward via `result[len] = '\0'`. The destination size is passed explicitly by the caller via `result_size`, and the function also validates `result != NULL` and `result_size != 0` up front, so no overflow is possible.

**2. Flawfinder `char buffer[BUFFER_SIZE]` warning (CWE-119/CWE-120) in `main`:**
Also a false positive. The statically sized buffer is never written to directly in `main`; it is only passed to `find_word_with_interior_z` together with its true size via `sizeof(buffer)`. Since the callee performs bounds checking against that size, the array cannot be overrun.

Additional defensive measures already present that are worth noting:

- `isalpha` is called with the argument cast to `unsigned char`, avoiding undefined behavior for negative `char` values on platforms where `char` is signed.
- `contains_interior_z` guards against `NULL` and short words (`len < 3`), so the loop bounds `1` to `len - 1` cannot underflow.
- `result[0] = '\0'` ensures the output buffer is always a valid string even when no match is found or an early error path is taken.
- Return codes distinguish success (`0`), no-match (`1`), and error (`-1`), and `main` checks all three.

The code as provided is already correct with respect to the Flawfinder findings; both hits are informational warnings about potentially risky constructs, and the surrounding bounds checks fully mitigate them.
