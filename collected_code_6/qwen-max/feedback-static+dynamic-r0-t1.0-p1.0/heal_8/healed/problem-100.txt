```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <limits.h>

bool isPalindrome(int n) {
    char numStr[25];
    snprintf(numStr, sizeof(numStr), "%d", n);
    size_t len = strnlen(numStr, sizeof(numStr));
    for (size_t i = 0; i < len / 2; ++i) {
        if (numStr[i] != numStr[len - i - 1]) return false;
    }
    return true;
}

int nextPalindromicNumber(int n) {
    n++;
    while (!isPalindrome(n)) n++;
    return n;
}

int main() {
    int n;
    if (scanf("%d", &n) != 1) return 1;
    int result = nextPalindromicNumber(n);
    printf("%d\n", result);
    return 0;
}
```

The `snprintf` function ensures that the buffer is not overrun, and `strnlen` is used to safely determine the length of the string. The identified potential weakness in the comment is not applicable here, as the buffer size is sufficient to handle the conversion of any integer to a string.
