#include <stdio.h>
#include <stdlib.h>

void rotate_list(int *arr, int n, int k) {
    if (n == 0 || k % n == 0) return;
    k = k % n;
    int *temp = malloc(k * sizeof(int));
    if (temp == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(1);
    }
    for (int i = 0; i < k; ++i) {
        temp[i] = arr[n - k + i];
    }
    for (int i = n - k - 1; i >= 0; --i) {
        arr[i + k] = arr[i];
    }
    for (int i = 0; i < k; ++i) {
        arr[i] = temp[i];
    }
    free(temp);
}

int main() {
    int arr[] = {1, 2, 3, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 2;

    rotate_list(arr, n, k);

    for (int i = 0; i < n; ++i) {
        printf("%d ", arr[i]);
    }

    return 0;
}
