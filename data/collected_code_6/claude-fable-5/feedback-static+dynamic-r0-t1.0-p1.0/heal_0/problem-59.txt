```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long octagonal_number(int n);

long long octagonal_number(int n)
{
    long long ln = (long long)n;
    return ln * (3LL * ln - 2LL);
}

int main(int argc, char *argv[])
{
    int n = 0;

    if (argc != 2) {
        (void)fprintf(stderr, "Usage: %s <n>\n", argv[0]);
        return EXIT_FAILURE;
    }

    {
        char *endptr = NULL;
        long value = strtol(argv[1], &endptr, 10);

        if (endptr == argv[1] || *endptr != '\0') {
            (void)fprintf(stderr, "Error: invalid input\n");
            return EXIT_FAILURE;
        }

        if (value < 1L || value > 1000000L) {
            (void)fprintf(stderr, "Error: n must be between 1 and 1000000\n");
            return EXIT_FAILURE;
        }

        n = (int)value;
    }

    {
        long long result = octagonal_number(n);
        if (printf("The %d%s octagonal number is %lld\n",
                   n,
                   (n % 10 == 1 && n % 100 != 11) ? "st" :
                   (n % 10 == 2 && n % 100 != 12) ? "nd" :
                   (n % 10 == 3 && n % 100 != 13) ? "rd" : "th",
                   result) < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```
