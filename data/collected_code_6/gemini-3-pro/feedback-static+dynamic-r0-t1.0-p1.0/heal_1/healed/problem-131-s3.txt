#include <stdio.h>
#include <stdlib.h>

int is_vowel(char c) {
    if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
        c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U') {
        return 1;
    }
    return 0;
}

void reverse_vowels(char *s) {
    if (s == NULL) {
        return;
    }

    int left = 0;
    int right = 0;
    
    while (s[right] != '\0') {
        right++;
    }
    
    if (right == 0) {
        return;
    }
    
    right--;

    while (left < right) {
        if (!is_vowel(s[left])) {
            left++;
            continue;
        }
        if (!is_vowel(s[right])) {
            right--;
            continue;
        }

        char temp = s[left];
        s[left] = s[right];
        s[right] = temp;

        left++;
        right--;
    }
}

int main(void) {
    char str1[] = "hello";
    char str2[] = "leetcode";

    reverse_vowels(str1);
    printf("%s\n", str1);

    reverse_vowels(str2);
    printf("%s\n", str2);

    return 0;
}
