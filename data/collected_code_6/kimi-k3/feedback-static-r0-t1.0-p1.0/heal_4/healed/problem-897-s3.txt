#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_BUFFER_SIZE 256

int isWordPresent(const char *sentence, size_t sentenceLen, const char *word, size_t wordLen) {
    if (sentence == NULL || word == NULL || wordLen == 0 || sentenceLen == 0) {
        return 0;
    }

    if (wordLen > sentenceLen) {
        return 0;
    }

    size_t i = 0;
    while (i + wordLen <= sentenceLen) {
        if (memcmp(sentence + i, word, wordLen) == 0) {
            int leftOk = (i == 0) || !isalnum((unsigned char)sentence[i - 1]);
            int rightOk = (i + wordLen >= sentenceLen) || !isalnum((unsigned char)sentence[i + wordLen]);
            if (leftOk && rightOk) {
                return 1;
            }
        }
        i++;
    }

    return 0;
}

int main(void) {
    char *sentence = NULL;
    char *word = NULL;
    size_t sentenceCap = 0;
    size_t wordCap = 0;
    ssize_t sentenceRead;
    ssize_t wordRead;
    size_t sentenceLen;
    size_t wordLen;

    printf("Enter a sentence: ");
    sentenceRead = getline(&sentence, &sentenceCap, stdin);
    if (sentenceRead == -1) {
        fprintf(stderr, "Error reading sentence\n");
        free(sentence);
        return 1;
    }

    printf("Enter a word to search: ");
    wordRead = getline(&word, &wordCap, stdin);
    if (wordRead == -1) {
        fprintf(stderr, "Error reading word\n");
        free(sentence);
        free(word);
        return 1;
    }

    sentenceLen = (size_t)sentenceRead;
    if (sentenceLen > 0 && sentence[sentenceLen - 1] == '\n') {
        sentence[sentenceLen - 1] = '\0';
        sentenceLen--;
    }

    wordLen = (size_t)wordRead;
    if (wordLen > 0 && word[wordLen - 1] == '\n') {
        word[wordLen - 1] = '\0';
        wordLen--;
    }

    if (isWordPresent(sentence, sentenceLen, word, wordLen)) {
        printf("The word is present in the sentence.\n");
    } else {
        printf("The word is not present in the sentence.\n");
    }

    free(sentence);
    free(word);

    return 0;
}
