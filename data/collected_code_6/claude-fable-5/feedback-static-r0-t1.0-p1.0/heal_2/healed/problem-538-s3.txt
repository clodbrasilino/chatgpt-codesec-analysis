```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 4096

typedef struct {
    char **items;
    size_t count;
} Tuple;

static char *safe_string_dup(const char *src) {
    size_t len;
    char *dst;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    dst = malloc(len + 1);
    if (dst == NULL) {
        return NULL;
    }

    memcpy(dst, src, len);
    dst[len] = '\0';
    return dst;
}

Tuple *list_to_tuple(const char **list, size_t count) {
    Tuple *t = NULL;

    if (list == NULL && count > 0) {
        return NULL;
    }

    t = malloc(sizeof(Tuple));
    if (t == NULL) {
        return NULL;
    }

    t->count = count;
    t->items = NULL;

    if (count > 0) {
        t->items = calloc(count, sizeof(char *));
        if (t->items == NULL) {
            free(t);
            return NULL;
        }

        for (size_t i = 0; i < count; i++) {
            if (list[i] == NULL) {
                t->items[i] = NULL;
                continue;
            }

            t->items[i] = safe_string_dup(list[i]);
            if (t->items[i] == NULL) {
                for (size_t j = 0; j < i; j++) {
                    free(t->items[j]);
                }
                free(t->items);
                free(t);
                return NULL;
            }
        }
    }

    return t;
}

void free_tuple(Tuple *t) {
    if (t == NULL) {
        return;
    }
    if (t->items != NULL) {
        for (size_t i = 0; i < t->count; i++) {
            free(t->items[i]);
        }
        free(t->items);
    }
    free(t);
}

void print_tuple(const Tuple *t) {
    if (t == NULL) {
        printf("()\n");
        return;
    }
    printf("(");
    for (size_t i = 0; i < t->count; i++) {
        if (t->items[i] != NULL) {
            printf("'%s'", t->items[i]);
        } else {
            printf("NULL");
        }
        if (i + 1 < t->count) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void) {
    const char *list[] = {"apple", "banana", "cherry"};
    size_t count = sizeof(list) / sizeof(list[0]);

    Tuple *t = list_to_tuple(list, count);
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the original weakness:**

The Flawfinder warning on `memcpy` (CWE-120) was a low-risk, largely false-positive finding, because the destination buffer was allocated as `malloc(len + 1)` immediately before copying exactly `len` bytes — the destination is guaranteed to hold the source data by construction.

The rewrite still addresses the concern properly:

1. **Encapsulated bounded duplication** — the copy logic is isolated in `safe_string_dup()`, so the invariant "allocation size = copy length + 1" lives in one place and cannot drift apart during future maintenance (the typical way this class of bug is actually introduced).
2. **Bounded length measurement** — `strnlen(src, MAX_STRING_LEN)` is retained, guaranteeing the copy length can never exceed `MAX_STRING_LEN` even for unterminated input, which caps both the allocation and the `memcpy` size.
3. **Guaranteed NUL termination** — `dst[len] = '\0'` ensures the result is always a valid C string even when the source was truncated at `MAX_STRING_LEN`.
4. **Full error-path cleanup preserved** — on any allocation failure, all previously allocated strings, the item array, and the struct are freed, avoiding leaks (CWE-401) and use of partially initialized structures.
