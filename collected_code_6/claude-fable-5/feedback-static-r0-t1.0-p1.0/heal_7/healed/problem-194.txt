```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 64

int octal_to_decimal(const char *octal, size_t max_len, long *result)
{
    size_t len;
    size_t i;
    long value = 0;

    if (octal == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(octal, max_len);
    if (len == 0 || len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        char c = octal[i];
        if (c < '0' || c > '7') {
            return -1;
        }
        if (value > (LONG_MAX - (c - '0')) / 8) {
            return -1;
        }
        value = value * 8 + (c - '0');
    }

    *result = value;
    return 0;
}

static int discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);

    return (ch == EOF) ? -1 : 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    long decimal = 0;
    size_t len;

    printf("Enter an octal number: ");
    if (fflush(stdout) != 0) {
        return 1;
    }

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (octal_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid octal number or value out of range.\n");
        return 1;
    }

    printf("Decimal value: %ld\n", decimal);

    return 0;
}
```

**Analysis of the flagged weaknesses:**

1. **`getchar` in a loop (CWE-120, CWE-20)** — This is a false positive in this context. The `discard_remaining_input()` function reads and discards characters without storing them into any buffer, so there is no buffer to overflow. The loop terminates correctly on both `'\n'` and `EOF`, avoiding an infinite loop if the stream is closed.

2. **Statically-sized `char` array (CWE-119/CWE-120)** — Already mitigated. The buffer is only written via `fgets(input, sizeof(input), stdin)`, which enforces the bound. Defensive measures already in place:
   - Explicit null termination at `input[sizeof(input) - 1]`.
   - `strnlen` used instead of `strlen`, bounded by the buffer size.
   - Overlong input detection (no trailing newline with a full buffer) with the rest of the line flushed via `discard_remaining_input()`.
   - `octal_to_decimal` validates its arguments, bounds its scan with `max_len`, rejects non-octal characters, and checks for `long` overflow **before** multiplying (`value > (LONG_MAX - digit) / 8`), preventing signed integer overflow (undefined behavior, CWE-190).

No functional changes were required — the code as written already contains the fixes for both flagged issues; the rewrite simply removes the analysis comments.
