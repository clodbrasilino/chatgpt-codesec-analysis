```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_MAX_LEN 64U

typedef struct {
    char name[NAME_MAX_LEN];
    int value;
} Tuple;

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;

    if (ta->value < tb->value) {
        return -1;
    }
    if (ta->value > tb->value) {
        return 1;
    }
    return 0;
}

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strlen(name);
    if (len >= NAME_MAX_LEN) {
        return -1;
    }

    memcpy(t->name, name, len);
    t->name[len] = '\0';
    t->value = value;

    return 0;
}

static Tuple *find_min_k_records(const Tuple *records, size_t n, size_t k)
{
    Tuple *sorted = NULL;
    Tuple *result = NULL;

    if (records == NULL || n == 0U || k == 0U || k > n) {
        return NULL;
    }

    if (n > SIZE_MAX / sizeof(Tuple)) {
        return NULL;
    }

    sorted = malloc(n * sizeof(Tuple));
    if (sorted == NULL) {
        return NULL;
    }

    memcpy(sorted, records, n * sizeof(Tuple));
    qsort(sorted, n, sizeof(Tuple), compare_tuples);

    result = malloc(k * sizeof(Tuple));
    if (result == NULL) {
        free(sorted);
        return NULL;
    }

    memcpy(result, sorted, k * sizeof(Tuple));
    free(sorted);

    return result;
}

int main(void)
{
    Tuple records[5];
    size_t n = sizeof(records) / sizeof(records[0]);
    size_t k = 2U;
    Tuple *min_records = NULL;
    size_t i;

    if (set_tuple(&records[0], "Manjeet", 10) != 0 ||
        set_tuple(&records[1], "Akshat", 4) != 0 ||
        set_tuple(&records[2], "Akash", 2) != 0 ||
        set_tuple(&records[3], "Nikhil", 8) != 0 ||
        set_tuple(&records[4], "Rohit", 6) != 0) {
        if (fprintf(stderr, "Failed to initialize records\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    min_records = find_min_k_records(records, n, k);
    if (min_records == NULL) {
        if (fprintf(stderr, "Failed to find minimum records\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Minimum %zu records:\n", k) < 0) {
        free(min_records);
        return EXIT_FAILURE;
    }

    for (i = 0U; i < k; i++) {
        if (printf("(%s, %d)\n", min_records[i].name, min_records[i].value) < 0) {
            free(min_records);
            return EXIT_FAILURE;
        }
    }

    free(min_records);
    return EXIT_SUCCESS;
}
```

Key changes addressing the identified weaknesses:

1. **Statically-sized array (CWE-119/CWE-120)**: The `name[64]` buffer is now sized via a named constant `NAME_MAX_LEN`, and all writes to it go through `set_tuple()`, which validates the source string length before copying and guarantees null termination. This removes the risk of unchecked initialization overflowing the fixed buffer.

2. **First `memcpy` (CWE-120)**: Added an explicit integer-overflow guard (`n > SIZE_MAX / sizeof(Tuple)`) before the `malloc`, so the allocation size cannot wrap around and produce an undersized destination buffer. The copy size then provably matches the allocated size.

3. **Second `memcpy` (CWE-120)**: The `k <= n` precondition combined with the overflow check ensures `k * sizeof(Tuple)` cannot overflow either, and `result` is allocated with exactly that size, so the destination always holds the copied data.
