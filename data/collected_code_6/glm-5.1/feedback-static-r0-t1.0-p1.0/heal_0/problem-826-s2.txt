#include <stdio.h>
#include <stdlib.h>

typedef enum {
    INVALID,
    EQUILATERAL,
    ISOSCELES,
    SCALENE
} TriangleType;

int compare(const void *a, const void *b) {
    double da = *(const double *)a;
    double db = *(const double *)b;
    if (da < db) return -1;
    if (da > db) return 1;
    return 0;
}

TriangleType find_triangle_type(double a, double b, double c) {
    if (a <= 0.0 || b <= 0.0 || c <= 0.0) {
        return INVALID;
    }

    double sides[3] = {a, b, c};
    qsort(sides, 3, sizeof(double), compare);

    if (sides[0] + sides[1] <= sides[2]) {
        return INVALID;
    }

    if (a == b && b == c) {
        return EQUILATERAL;
    } else if (a == b || b == c || a == c) {
        return ISOSCELES;
    } else {
        return SCALENE;
    }
}

int main(void) {
    double a, b, c;
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return EXIT_FAILURE;
    }

    TriangleType type = find_triangle_type(a, b, c);

    switch (type) {
        case EQUILATERAL:
            printf("EQUILATERAL\n");
            break;
        case ISOSCELES:
            printf("ISOSCELES\n");
            break;
        case SCALENE:
            printf("SCALENE\n");
            break;
        case INVALID:
        default:
            printf("INVALID\n");
            break;
    }

    return EXIT_SUCCESS;
}
