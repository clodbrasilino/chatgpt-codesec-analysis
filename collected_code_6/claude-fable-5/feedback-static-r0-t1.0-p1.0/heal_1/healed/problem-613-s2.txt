```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_RECORDS 10

struct TupleEntry {
    int records[MAX_RECORDS];
    size_t record_count;
};

static int find_max_record(const struct TupleEntry *entries, size_t entry_count, int *max_out)
{
    size_t i;
    size_t j;
    int found;
    int max_val;

    if ((entries == NULL) || (max_out == NULL) || (entry_count == 0U)) {
        return -1;
    }

    found = 0;
    max_val = INT_MIN;

    for (i = 0U; i < entry_count; i++) {
        if (entries[i].record_count > (size_t)MAX_RECORDS) {
            return -1;
        }
        for (j = 0U; j < entries[i].record_count; j++) {
            if ((found == 0) || (entries[i].records[j] > max_val)) {
                max_val = entries[i].records[j];
                found = 1;
            }
        }
    }

    if (found == 0) {
        return -1;
    }

    *max_out = max_val;
    return 0;
}

int main(void)
{
    const struct TupleEntry entries[3] = {
        { { 10, 4, 5, 0, 0, 0, 0, 0, 0, 0 }, 3U },
        { { 1, 7, 8, 6, 0, 0, 0, 0, 0, 0 }, 4U },
        { { 0, 5, 6, 0, 0, 0, 0, 0, 0, 0 }, 3U }
    };
    int max_val;
    int status;

    status = find_max_record(entries, 3U, &max_val);
    if (status != 0) {
        (void)fprintf(stderr, "Error: unable to compute maximum value\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum value in record lists: %d\n", max_val) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The two identified weaknesses were addressed as follows:

1. **`unusedStructMember` (`TupleEntry::key`)** — The `key` member was never read or written meaningfully anywhere in the program, so it was removed from the struct definition, and the initializers in `main` were updated accordingly. This shrinks the struct and eliminates dead data.

2. **`constVariable` (`entries`)** — The `entries` array in `main` is never modified after initialization and is only passed to a function taking a `const struct TupleEntry *`, so it is now declared `const`. This lets the compiler enforce immutability and may allow the data to be placed in read-only storage.

All other defensive checks (NULL pointer validation, bounds check on `record_count`, empty-input handling, and checked return values for `printf`/`fprintf`) were already sound and have been preserved.
