```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int compare_desc(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return 1;
    }
    if (x > y) {
        return -1;
    }
    return 0;
}

int get_n_largest(const int *data, size_t data_len, size_t n,
                  int *result, size_t result_capacity)
{
    int *copy;
    size_t copy_size;
    size_t result_size;
    size_t i;

    if (data == NULL || result == NULL) {
        return -1;
    }
    if (n == 0 || data_len == 0 || n > data_len) {
        return -1;
    }
    if (n > result_capacity) {
        return -1;
    }
    if (data_len > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    copy_size = data_len * sizeof(int);
    result_size = n * sizeof(int);

    if (result_size > copy_size) {
        return -1;
    }

    copy = malloc(copy_size);
    if (copy == NULL) {
        return -1;
    }

    for (i = 0; i < data_len; i++) {
        copy[i] = data[i];
    }

    qsort(copy, data_len, sizeof(int), compare_desc);

    for (i = 0; i < n; i++) {
        result[i] = copy[i];
    }

    free(copy);
    return 0;
}

int main(void)
{
    int data[] = { 42, 7, 19, 88, 3, 56, 91, 23, 65, 10 };
    size_t data_len = sizeof(data) / sizeof(data[0]);
    size_t n = 3;
    int *result;
    size_t i;

    if (n > SIZE_MAX / sizeof(int)) {
        fprintf(stderr, "Requested size too large\n");
        return EXIT_FAILURE;
    }

    result = malloc(n * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (get_n_largest(data, data_len, n, result, n) != 0) {
        fprintf(stderr, "Failed to get largest items\n");
        free(result);
        return EXIT_FAILURE;
    }

    printf("The %zu largest items are:\n", n);
    for (i = 0; i < n; i++) {
        printf("%d\n", result[i]);
    }

    free(result);
    return EXIT_SUCCESS;
}
```

The key changes made to address the identified weaknesses:

1. **First `memcpy` (CWE-120):** Replaced with an explicit bounded loop copying exactly `data_len` elements into a buffer allocated with the same computed size, making the bounds relationship between source and destination explicit and verifiable.

2. **Second `memcpy` (CWE-120):** The original function had no way to verify the caller's `result` buffer was large enough — it trusted the caller entirely. The function signature now takes a `result_capacity` parameter, and the function returns an error if `n > result_capacity`. The copy itself is a bounded loop of exactly `n` elements.

3. **Caller updated** to pass the actual allocated capacity (`n`), so the callee can validate the destination size before writing — the proper fix for "make sure destination can always hold the source data."
