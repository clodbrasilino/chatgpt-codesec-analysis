```c
#include <stdio.h>
#include <stdlib.h>

int modulo_lists(const int *list1, const int *list2, int *result, size_t size)
{
    size_t i;

    if (list1 == NULL || list2 == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; i < size; i++) {
        if (list2[i] == 0) {
            return -2;
        }
        result[i] = list1[i] % list2[i];
    }

    return 0;
}

int main(void)
{
    const int list1[] = {10, 4, 5, 6};
    const int list2[] = {5, 6, 7, 5};
    size_t size = sizeof(list1) / sizeof(list1[0]);
    int *result;
    int status;
    size_t i;

    result = malloc(size * sizeof(int));
    if (result == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    status = modulo_lists(list1, list2, result, size);
    if (status != 0) {
        fprintf(stderr, "Error: invalid input or division by zero\n");
        free(result);
        return EXIT_FAILURE;
    }

    for (i = 0; i < size; i++) {
        printf("%d ", result[i]);
    }
    printf("\n");

    free(result);
    return EXIT_SUCCESS;
}
```
