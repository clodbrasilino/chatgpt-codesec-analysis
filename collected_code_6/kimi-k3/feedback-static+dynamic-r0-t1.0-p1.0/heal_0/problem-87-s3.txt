#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    int value;
} DictEntry;

typedef struct {
    DictEntry *entries;
    size_t size;
    size_t capacity;
} Dictionary;

int dict_init(Dictionary *dict, size_t capacity) {
    if (dict == NULL || capacity == 0) {
        return -1;
    }
    dict->entries = calloc(capacity, sizeof(DictEntry));
    if (dict->entries == NULL) {
        return -1;
    }
    dict->size = 0;
    dict->capacity = capacity;
    return 0;
}

int dict_add(Dictionary *dict, const char *key, int value) {
    if (dict == NULL || key == NULL || dict->size >= dict->capacity) {
        return -1;
    }
    dict->entries[dict->size].key = malloc(strlen(key) + 1);
    if (dict->entries[dict->size].key == NULL) {
        return -1;
    }
    strcpy(dict->entries[dict->size].key, key);
    dict->entries[dict->size].value = value;
    dict->size++;
    return 0;
}

int dict_find(const Dictionary *dict, const char *key) {
    if (dict == NULL || key == NULL) {
        return -1;
    }
    for (size_t i = 0; i < dict->size; i++) {
        if (strcmp(dict->entries[i].key, key) == 0) {
            return (int)i;
        }
    }
    return -1;
}

int dict_merge(Dictionary *dest, const Dictionary *src) {
    if (dest == NULL || src == NULL) {
        return -1;
    }
    for (size_t i = 0; i < src->size; i++) {
        int idx = dict_find(dest, src->entries[i].key);
        if (idx >= 0) {
            dest->entries[idx].value = src->entries[i].value;
        } else {
            if (dict_add(dest, src->entries[i].key, src->entries[i].value) != 0) {
                return -1;
            }
        }
    }
    return 0;
}

int dict_merge_three(Dictionary *result, const Dictionary *dict1, 
                     const Dictionary *dict2, const Dictionary *dict3) {
    if (result == NULL || dict1 == NULL || dict2 == NULL || dict3 == NULL) {
        return -1;
    }
    if (dict_merge(result, dict1) != 0) {
        return -1;
    }
    if (dict_merge(result, dict2) != 0) {
        return -1;
    }
    if (dict_merge(result, dict3) != 0) {
        return -1;
    }
    return 0;
}

void dict_free(Dictionary *dict) {
    if (dict == NULL) {
        return;
    }
    for (size_t i = 0; i < dict->size; i++) {
        free(dict->entries[i].key);
    }
    free(dict->entries);
    dict->entries = NULL;
    dict->size = 0;
    dict->capacity = 0;
}

void dict_print(const Dictionary *dict) {
    if (dict == NULL) {
        return;
    }
    printf("{ ");
    for (size_t i = 0; i < dict->size; i++) {
        printf("%s: %d", dict->entries[i].key, dict->entries[i].value);
        if (i < dict->size - 1) {
            printf(", ");
        }
    }
    printf(" }\n");
}

int main(void) {
    Dictionary dict1, dict2, dict3, merged;
    
    if (dict_init(&dict1, 10) != 0) {
        fprintf(stderr, "Failed to initialize dict1\n");
        return EXIT_FAILURE;
    }
    if (dict_init(&dict2, 10) != 0) {
        fprintf(stderr, "Failed to initialize dict2\n");
        dict_free(&dict1);
        return EXIT_FAILURE;
    }
    if (dict_init(&dict3, 10) != 0) {
        fprintf(stderr, "Failed to initialize dict3\n");
        dict_free(&dict1);
        dict_free(&dict2);
        return EXIT_FAILURE;
    }
    if (dict_init(&merged, 30) != 0) {
        fprintf(stderr, "Failed to initialize merged\n");
        dict_free(&dict1);
        dict_free(&dict2);
        dict_free(&dict3);
        return EXIT_FAILURE;
    }
    
    if (dict_add(&dict1, "a", 1) != 0 || dict_add(&dict1, "b", 2) != 0) {
        fprintf(stderr, "Failed to add to dict1\n");
        goto cleanup;
    }
    if (dict_add(&dict2, "c", 3) != 0 || dict_add(&dict2, "d", 4) != 0) {
        fprintf(stderr, "Failed to add to dict2\n");
        goto cleanup;
    }
    if (dict_add(&dict3, "e", 5) != 0 || dict_add(&dict3, "a", 10) != 0) {
        fprintf(stderr, "Failed to add to dict3\n");
        goto cleanup;
    }
    
    if (dict_merge_three(&merged, &dict1, &dict2, &dict3) != 0) {
        fprintf(stderr, "Failed to merge dictionaries\n");
        goto cleanup;
    }
    
    printf("Merged dictionary: ");
    dict_print(&merged);
    
    dict_free(&dict1);
    dict_free(&dict2);
    dict_free(&dict3);
    dict_free(&merged);
    
    return EXIT_SUCCESS;
    
cleanup:
    dict_free(&dict1);
    dict_free(&dict2);
    dict_free(&dict3);
    dict_free(&merged);
    return EXIT_FAILURE;
}
