double sum_of_gp(double a, double r, int n)
{
    if (n <= 0) {
        return 0.0;
    }

    if (fabs(r - 1.0) < 1e-12) {
        return a * (double)n;
    }

    return a * (pow(r, (double)n) - 1.0) / (r - 1.0);
}

int main(void)
{

    double r;
    int n;
    double result;

    if (n < 0) {
