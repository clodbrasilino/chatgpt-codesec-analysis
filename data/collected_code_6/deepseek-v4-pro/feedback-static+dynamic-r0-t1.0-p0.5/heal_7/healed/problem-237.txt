#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define NAME_SIZE 50

typedef struct {
    int id;
    char name[NAME_SIZE];
    int value;
} Record;

typedef struct {
    Record *records;
    size_t count;
} RecordGroup;

static int compare_ints(const void *a, const void *b) {
    int arg1 = *(const int *)a;
    int arg2 = *(const int *)b;
    return (arg1 > arg2) - (arg1 < arg2);
}

static int count_occurrences(const Record *records, size_t count, const Record *target) {
    int occurrences = 0;
    for (size_t i = 0; i < count; i++) {
        if (records[i].id == target->id &&
            strncmp(records[i].name, target->name, NAME_SIZE) == 0 &&
            records[i].value == target->value) {
            occurrences++;
        }
    }
    return occurrences;
}

static int is_duplicate(const Record *unique_records, size_t unique_count, const Record *target) {
    for (size_t i = 0; i < unique_count; i++) {
        if (unique_records[i].id == target->id &&
            strncmp(unique_records[i].name, target->name, NAME_SIZE) == 0 &&
            unique_records[i].value == target->value) {
            return 1;
        }
    }
    return 0;
}

static int check_similar_occurrences(const Record *records, size_t count) {
    if (records == NULL || count == 0) {
        return 0;
    }

    if (count > SIZE_MAX / sizeof(Record)) {
        return -1;
    }

    Record *unique_records = (Record *)malloc(count * sizeof(Record));
    if (unique_records == NULL) {
        return -1;
    }

    size_t unique_count = 0;
    for (size_t i = 0; i < count; i++) {
        if (!is_duplicate(unique_records, unique_count, &records[i])) {
            if (unique_count < count) {
                Record *dest = &unique_records[unique_count];
                dest->id = records[i].id;
                dest->value = records[i].value;
                strncpy(dest->name, records[i].name, NAME_SIZE - 1);
                dest->name[NAME_SIZE - 1] = '\0';
                unique_count++;
            } else {
                free(unique_records);
                return -1;
            }
        }
    }

    if (unique_count == 0) {
        free(unique_records);
        return 1;
    }

    if (unique_count > SIZE_MAX / sizeof(int)) {
        free(unique_records);
        return -1;
    }

    int *occurrences = (int *)malloc(unique_count * sizeof(int));
    if (occurrences == NULL) {
        free(unique_records);
        return -1;
    }

    for (size_t i = 0; i < unique_count; i++) {
        occurrences[i] = count_occurrences(records, count, &unique_records[i]);
    }

    qsort(occurrences, unique_count, sizeof(int), compare_ints);

    int result = 1;
    for (size_t i = 1; i < unique_count; i++) {
        if (occurrences[i] != occurrences[0]) {
            result = 0;
            break;
        }
    }

    free(unique_records);
    free(occurrences);
    return result;
}

int main(void) {
    Record records[] = {
        {1, "alpha", 100},
        {2, "beta", 200},
        {1, "alpha", 100},
        {3, "gamma", 300},
        {2, "beta", 200},
        {3, "gamma", 300}
    };
    size_t count = sizeof(records) / sizeof(records[0]);

    int result = check_similar_occurrences(records, count);
    if (result == 1) {
        printf("All records occur similar times\n");
    } else if (result == 0) {
        printf("Records do not occur similar times\n");
    } else {
        printf("Error processing records\n");
    }

    Record records2[] = {
        {1, "alpha", 100},
        {2, "beta", 200},
        {1, "alpha", 100},
        {3, "gamma", 300}
    };
    count = sizeof(records2) / sizeof(records2[0]);

    result = check_similar_occurrences(records2, count);
    if (result == 1) {
        printf("All records occur similar times\n");
    } else if (result == 0) {
        printf("Records do not occur similar times\n");
    } else {
        printf("Error processing records\n");
    }

    return 0;
}
