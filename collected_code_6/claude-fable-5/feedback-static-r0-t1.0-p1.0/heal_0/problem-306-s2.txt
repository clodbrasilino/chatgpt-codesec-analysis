#include <stdio.h>
#include <stdlib.h>

static int maxSumIncludingK(const int arr[], int n, int i, int k)
{
    int *dp;
    int result;
    int j;
    int m;

    if (arr == NULL || n <= 0 || i < 0 || k <= i || k >= n) {
        return -1;
    }

    dp = (int *)malloc((size_t)n * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    for (j = 0; j <= i; j++) {
        dp[j] = 0;
        if (arr[j] < arr[k]) {
            dp[j] = arr[j];
            for (m = 0; m < j; m++) {
                if (arr[m] < arr[j] && dp[m] > 0) {
                    if (dp[m] + arr[j] > dp[j]) {
                        dp[j] = dp[m] + arr[j];
                    }
                }
            }
        }
    }

    result = 0;
    for (j = 0; j <= i; j++) {
        if (dp[j] > result) {
            result = dp[j];
        }
    }

    result += arr[k];
    free(dp);
    return result;
}

int main(void)
{
    int arr[] = {1, 101, 2, 3, 100, 4, 5};
    int n = (int)(sizeof(arr) / sizeof(arr[0]));
    int i = 4;
    int k = 6;
    int ans;

    ans = maxSumIncludingK(arr, n, i, k);
    if (ans < 0) {
        fprintf(stderr, "Invalid input or memory allocation failure\n");
        return EXIT_FAILURE;
    }

    if (printf("Maximum sum: %d\n", ans) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
