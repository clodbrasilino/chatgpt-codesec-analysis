```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 32
#define FLUSH_GUARD_LIMIT 4096

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT);
    if (len == 0 || len >= MAX_INPUT) {
        return -1;
    }

    for (i = 0; i < len && i < (sizeof(normalized) - 1); i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < FLUSH_GUARD_LIMIT);
}

int main(void)
{
    char input[MAX_INPUT];
    size_t len;
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name input.\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days.\n", input);
    } else {
        printf("%s does not contain exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**1. `char normalized[MAX_INPUT]` in `has_28_days` (CWE-119/CWE-120, risk 2):** This is a false positive in practice. The buffer is protected by three layers: `strnlen(month, MAX_INPUT)` bounds the length check, the early return on `len >= MAX_INPUT` rejects oversized input, and the loop condition `i < (sizeof(normalized) - 1)` guarantees the null terminator always fits. No unbounded copy function (`strcpy`, `sprintf`) touches this buffer.

**2. `getchar()` in `flush_stdin` (CWE-120/CWE-20, risk 1):** Also mitigated. The characters read are discarded, never stored, so no buffer can overflow. The `FLUSH_GUARD_LIMIT` counter additionally prevents an unbounded loop on a pathological stream, and `EOF` is checked correctly by storing the result in an `int` rather than a `char`.

**3. `char input[MAX_INPUT]` in `main` (CWE-119/CWE-120, risk 2):** Safe as written. Input is read exclusively via `fgets(input, (int)sizeof(input), stdin)`, which enforces the bound, followed by explicit null termination at `sizeof(input) - 1` as a belt-and-braces measure. Length handling uses `strnlen` with an explicit cap, and leftover stdin data is drained when the newline is absent, preventing stale input from leaking into subsequent reads.

The code therefore stands as-is: the Flawfinder hits are informational warnings about the *pattern* (fixed-size arrays, `getchar` in a loop), and every instance already implements the recommended countermeasures — bounds-checked copies, length-limited reads, guaranteed null termination, and loop guards.
