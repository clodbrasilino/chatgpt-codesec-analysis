#include <stdio.h>
#include <stdbool.h>
#include <stdlib.h>

typedef struct {
    int *elements;
    int size;
} Tuple;

bool tuple_contains_k(Tuple *tuples, int tuple_count, int k) {
    if (tuples == NULL || tuple_count <= 0) {
        return false;
    }
    
    for (int i = 0; i < tuple_count; i++) {
        if (tuples[i].elements == NULL || tuples[i].size <= 0) {
            continue;
        }
        
        for (int j = 0; j < tuples[i].size; j++) {
            if (tuples[i].elements[j] == k) {
                return true;
            }
        }
    }
    
    return false;
}

Tuple* create_tuples(int tuple_count, int *sizes, int **elements) {
    if (tuple_count <= 0 || sizes == NULL || elements == NULL) {
        return NULL;
    }
    
    Tuple *tuples = (Tuple*)malloc(sizeof(Tuple) * tuple_count);
    if (tuples == NULL) {
        return NULL;
    }
    
    for (int i = 0; i < tuple_count; i++) {
        if (sizes[i] <= 0 || elements[i] == NULL) {
            tuples[i].elements = NULL;
            tuples[i].size = 0;
            continue;
        }
        
        tuples[i].elements = (int*)malloc(sizeof(int) * sizes[i]);
        if (tuples[i].elements == NULL) {
            for (int j = 0; j < i; j++) {
                free(tuples[j].elements);
            }
            free(tuples);
            return NULL;
        }
        
        tuples[i].size = sizes[i];
        for (int j = 0; j < sizes[i]; j++) {
            tuples[i].elements[j] = elements[i][j];
        }
    }
    
    return tuples;
}

void free_tuples(Tuple *tuples, int tuple_count) {
    if (tuples == NULL) {
        return;
    }
    
    for (int i = 0; i < tuple_count; i++) {
        if (tuples[i].elements != NULL) {
            free(tuples[i].elements);
        }
    }
    
    free(tuples);
}

int main(void) {
    int sizes[] = {3, 4, 2};
    int **elements = (int**)malloc(sizeof(int*) * 3);
    if (elements == NULL) {
        return 1;
    }
    
    elements[0] = (int*)malloc(sizeof(int) * 3);
    elements[1] = (int*)malloc(sizeof(int) * 4);
    elements[2] = (int*)malloc(sizeof(int) * 2);
    
    if (elements[0] == NULL || elements[1] == NULL || elements[2] == NULL) {
        for (int i = 0; i < 3; i++) {
            if (elements[i] != NULL) {
                free(elements[i]);
            }
        }
        free(elements);
        return 1;
    }
    
    int temp0[] = {1, 2, 3};
    int temp1[] = {4, 5, 6, 7};
    int temp2[] = {8, 9};
    
    for (int i = 0; i < 3; i++) elements[0][i] = temp0[i];
    for (int i = 0; i < 4; i++) elements[1][i] = temp1[i];
    for (int i = 0; i < 2; i++) elements[2][i] = temp2[i];
    
    Tuple *tuples = create_tuples(3, sizes, elements);
    if (tuples == NULL) {
        for (int i = 0; i < 3; i++) {
            free(elements[i]);
        }
        free(elements);
        return 1;
    }
    
    int k = 5;
    if (tuple_contains_k(tuples, 3, k)) {
        printf("Found %d in tuples\n", k);
    } else {
        printf("%d not found in tuples\n", k);
    }
    
    k = 10;
    if (tuple_contains_k(tuples, 3, k)) {
        printf("Found %d in tuples\n", k);
    } else {
        printf("%d not found in tuples\n", k);
    }
    
    free_tuples(tuples, 3);
    
    for (int i = 0; i < 3; i++) {
        free(elements[i]);
    }
    free(elements);
    
    return 0;
}
