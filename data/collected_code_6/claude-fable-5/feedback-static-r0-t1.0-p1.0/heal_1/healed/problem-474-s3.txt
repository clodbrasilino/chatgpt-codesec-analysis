```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

size_t replace_char(char *str, size_t max_len, char old_char, char new_char)
{
    size_t count = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len && str[i] != '\0'; i++) {
        if (str[i] == old_char) {
            str[i] = new_char;
            count++;
        }
    }

    return count;
}

int main(void)
{
    static const char original[] = "hello world, hello everyone";
    char *buffer;
    size_t len;
    size_t buffer_size;
    size_t replaced;

    len = strnlen(original, sizeof(original));
    if (len >= sizeof(original)) {
        fprintf(stderr, "Source string is not properly terminated\n");
        return EXIT_FAILURE;
    }

    buffer_size = len + 1;
    buffer = malloc(buffer_size);
    if (buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    memcpy(buffer, original, len);
    buffer[len] = '\0';

    printf("Before: %s\n", buffer);

    replaced = replace_char(buffer, buffer_size, 'l', 'L');

    printf("After:  %s\n", buffer);
    printf("Replaced %zu occurrence(s)\n", replaced);

    free(buffer);
    buffer = NULL;

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read):** Replaced `strlen` with `strnlen`, bounded by `sizeof(original)`. This is possible because `original` was changed from a `const char *` pointer to a `static const char[]` array, so the compiler knows its exact size. A verification check ensures the string is actually null-terminated within its bounds before proceeding.

2. **CWE-120 (unchecked memcpy):** The destination buffer size is now explicitly tracked in `buffer_size`, allocated to exactly `len + 1`. The `memcpy` copies only `len` bytes (guaranteed to fit), and the null terminator is written explicitly at a known-valid index, making the copy provably safe rather than relying on the source's terminator.

3. **Defense in depth in `replace_char`:** The function now takes an explicit `max_len` parameter and bounds its loop with `i < max_len`, so even if a non-terminated buffer were passed, it cannot read or write past the caller-specified limit.
