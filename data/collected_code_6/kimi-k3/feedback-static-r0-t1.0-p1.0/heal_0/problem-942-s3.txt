#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

int is_present(Node* head, int target) {
    Node* current = head;
    while (current != NULL) {
        if (current->data == target) {
            return 1;
        }
        current = current->next;
    }
    return 0;
}

void free_list(Node* head) {
    Node* current = head;
    Node* next_node;
    while (current != NULL) {
        next_node = current->next;
        free(current);
        current = next_node;
    }
}

int main(void) {
    Node* head = NULL;
    Node* temp = NULL;
    int values[] = {10, 20, 30, 40, 50};
    int target = 30;
    int i;

    for (i = 0; i < 5; i++) {
        temp = create_node(values[i]);
        if (temp == NULL) {
            free_list(head);
            return EXIT_FAILURE;
        }
        temp->next = head;
        head = temp;
    }

    if (is_present(head, target)) {
        printf("%d is present in the list.\n", target);
    } else {
        printf("%d is not present in the list.\n", target);
    }

    free_list(head);
    return EXIT_SUCCESS;
}
