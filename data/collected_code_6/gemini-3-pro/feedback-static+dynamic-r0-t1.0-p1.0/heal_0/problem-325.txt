#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int min_squares(int n) {
    if (n < 0) {
        return -1;
    }
    if (n <= 3) {
        return n;
    }

    int *dp = (int *)malloc((n + 1) * sizeof(int));
    if (dp == NULL) {
        return -1;
    }

    dp[0] = 0;
    dp[1] = 1;
    dp[2] = 2;
    dp[3] = 3;

    for (int i = 4; i <= n; i++) {
        dp[i] = i;
        for (int x = 1; x * x <= i; x++) {
            int temp = x * x;
            if (1 + dp[i - temp] < dp[i]) {
                dp[i] = 1 + dp[i - temp];
            }
        }
    }

    int result = dp[n];
    free(dp);
    return result;
}

int main(void) {
    int n;
    
    if (scanf("%d", &n) != 1) {
        return 1;
    }

    int result = min_squares(n);
    if (result == -1) {
        return 1;
    }

    printf("%d\n", result);

    return 0;
}
