```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_LEN 256

static char *extract_quoted_value(const char *input, size_t *offset)
{
    regex_t regex;
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;
    size_t input_len;

    if (input == NULL || offset == NULL) {
        return NULL;
    }

    input_len = strnlen(input, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1U] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return NULL;
    }

    ret = regexec(&regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (size_t)end <= input_len - *offset) {
            size_t len = (size_t)(end - start);
            result = malloc(len + 1U);
            if (result != NULL) {
                memcpy(result, input + *offset + (size_t)start, len);
                result[len] = '\0';
                *offset += (size_t)matches[0].rm_eo;
            } else {
                fprintf(stderr, "Memory allocation failed\n");
            }
        }
    } else if (ret != REG_NOMATCH) {
        char errbuf[ERRBUF_LEN];
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1U] = '\0';
        fprintf(stderr, "Regex execution failed: %s\n", errbuf);
    }

    regfree(&regex);
    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(text, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced `strlen` with `strnlen` bounded by `MAX_INPUT_LEN`, and the length is computed once and reused. Inputs at or beyond the cap are rejected, so a non-terminated buffer can no longer cause an unbounded read.

2. **CWE-119/CWE-120 (static `errbuf` arrays)**: The buffers remain fixed-size (required by the `regerror` API), but the size is passed via `sizeof(errbuf)` and an explicit null terminator is enforced defensively after the call.

3. **CWE-120 (memcpy)**: Added a bounds check ensuring `end` does not exceed the remaining input length before copying, guaranteeing the source read stays within the input buffer; the destination is allocated exactly `len + 1` bytes, so the copy can never overflow it.

4. **[variableScope]**: The `value` pointer in `main` was moved inside the loop, reducing its scope to where it is actually used.

5. Minor hardening: `extract_quoted_value` is now `static` since it has internal linkage in a single translation unit, and the `start` offset cast is explicit in the pointer arithmetic.
