#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int product;
    int index1;
    int index2;
} ProductInfo;

ProductInfo* find_largest_products(const int* list1, int size1,
                                   const int* list2, int size2,
                                   int k, int* result_count) {
    if (list1 == NULL || list2 == NULL || size1 <= 0 || size2 <= 0 || k <= 0) {
        *result_count = 0;
        return NULL;
    }

    int total_products = size1 * size2;
    if (k > total_products) {
        k = total_products;
    }

    int capacity = total_products;
    ProductInfo* all_products = (ProductInfo*)malloc(sizeof(ProductInfo) * capacity);
    if (all_products == NULL) {
        *result_count = 0;
        return NULL;
    }

    int idx = 0;
    for (int i = 0; i < size1; i++) {
        for (int j = 0; j < size2; j++) {
            all_products[idx].product = list1[i] * list2[j];
            all_products[idx].index1 = i;
            all_products[idx].index2 = j;
            idx++;
        }
    }

    for (int i = 0; i < total_products - 1; i++) {
        for (int j = 0; j < total_products - i - 1; j++) {
            if (all_products[j].product < all_products[j + 1].product) {
                ProductInfo temp = all_products[j];
                all_products[j] = all_products[j + 1];
                all_products[j + 1] = temp;
            }
        }
    }

    ProductInfo* result = (ProductInfo*)malloc(sizeof(ProductInfo) * k);
    if (result == NULL) {
        free(all_products);
        *result_count = 0;
        return NULL;
    }

    for (int i = 0; i < k; i++) {
        result[i] = all_products[i];
    }

    free(all_products);
    *result_count = k;
    return result;
}

void print_products(const ProductInfo* products, int count) {
    if (products == NULL || count <= 0) {
        printf("No products to display.\n");
        return;
    }
    
    printf("Top %d products:\n", count);
    for (int i = 0; i < count; i++) {
        printf("Product: %d (list1[%d]=%d, list2[%d]=%d)\n",
               products[i].product,
               products[i].index1, products[i].index1 >= 0 ? products[i].index1 : 0,
               products[i].index2, products[i].index2 >= 0 ? products[i].index2 : 0);
    }
}

int main(void) {
    int list1[] = {3, 5, 2, 8};
    int list2[] = {4, 1, 7, 2};
    int size1 = sizeof(list1) / sizeof(list1[0]);
    int size2 = sizeof(list2) / sizeof(list2[0]);
    int k = 5;
    int result_count = 0;

    ProductInfo* top_products = find_largest_products(list1, size1,
                                                       list2, size2,
                                                       k, &result_count);

    print_products(top_products, result_count);

    free(top_products);
    return 0;
}
