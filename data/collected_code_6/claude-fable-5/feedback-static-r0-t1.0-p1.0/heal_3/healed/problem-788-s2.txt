```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STR_LEN 4096
#define MAX_LIST_SIZE 4096

typedef struct {
    char *str;
    int *list;
    size_t list_size;
} Tuple;

Tuple *create_tuple(const char *str, const int *list, size_t list_size);
void free_tuple(Tuple *t);
void print_tuple(const Tuple *t);

Tuple *create_tuple(const char *str, const int *list, size_t list_size)
{
    Tuple *t;
    size_t len;

    if (str == NULL || (list == NULL && list_size > 0)) {
        return NULL;
    }

    if (list_size > MAX_LIST_SIZE) {
        return NULL;
    }

    if (list_size > SIZE_MAX / sizeof(*list)) {
        return NULL;
    }

    len = strnlen(str, MAX_STR_LEN);
    if (len >= MAX_STR_LEN) {
        return NULL;
    }

    t = malloc(sizeof(*t));
    if (t == NULL) {
        return NULL;
    }

    t->str = malloc(len + 1);
    if (t->str == NULL) {
        free(t);
        return NULL;
    }

    memcpy(t->str, str, len);
    t->str[len] = '\0';

    if (list_size > 0) {
        size_t list_bytes = list_size * sizeof(*t->list);
        t->list = malloc(list_bytes);
        if (t->list == NULL) {
            free(t->str);
            free(t);
            return NULL;
        }
        memcpy(t->list, list, list_bytes);
    } else {
        t->list = NULL;
    }

    t->list_size = list_size;
    return t;
}

void free_tuple(Tuple *t)
{
    if (t != NULL) {
        free(t->str);
        free(t->list);
        free(t);
    }
}

void print_tuple(const Tuple *t)
{
    size_t i;

    if (t == NULL || t->str == NULL) {
        return;
    }

    printf("(\"%s\", [", t->str);
    for (i = 0; i < t->list_size; i++) {
        printf("%d", t->list[i]);
        if (i + 1 < t->list_size) {
            printf(", ");
        }
    }
    printf("])\n");
}

int main(void)
{
    const char *str = "hello";
    int list[] = {1, 2, 3, 4, 5};
    size_t list_size = sizeof(list) / sizeof(list[0]);
    Tuple *t;

    t = create_tuple(str, list, list_size);
    if (t == NULL) {
        fprintf(stderr, "Failed to create tuple\n");
        return EXIT_FAILURE;
    }

    print_tuple(t);
    free_tuple(t);

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **`variableScope` (cppcheck):** The declaration of `list_bytes` was moved from function scope into the `if (list_size > 0)` block where it is actually used. This is the only place the variable is referenced, so narrowing its scope improves readability and reduces the chance of accidental misuse.

2. **`memcpy` / CWE-120 (Flawfinder):** Both Flawfinder hits are false positives, and no code change is required beyond keeping the existing guards:
   - For the string copy: the destination is allocated with exactly `len + 1` bytes, `len` is bounded by `strnlen(str, MAX_STR_LEN)` with an explicit rejection when `len >= MAX_STR_LEN`, and exactly `len` bytes are copied before the manual NUL termination. The destination always holds the source data.
   - For the list copy: `list_size` is validated against both `MAX_LIST_SIZE` and `SIZE_MAX / sizeof(*list)` (preventing multiplication overflow), and the destination is allocated with exactly `list_bytes`. The copy size and allocation size are derived from the same expression, so they cannot diverge.

3. **Other correctness properties preserved:** NULL-parameter validation, full cleanup on every allocation-failure path (no leaks), `free_tuple` tolerating NULL, and `print_tuple` guarding against NULL input all remain intact.
