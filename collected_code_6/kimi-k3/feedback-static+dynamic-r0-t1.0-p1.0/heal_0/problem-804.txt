#include <stdio.h>
#include <limits.h>

int is_product_even(int num1, int num2) {
    if (num1 % 2 == 0 || num2 % 2 == 0) {
        return 1;
    }
    return 0;
}

int main(void) {
    int num1 = 0;
    int num2 = 0;
    int result = 0;

    printf("Enter first number: ");
    if (scanf("%d", &num1) != 1) {
        fprintf(stderr, "Invalid input for first number\n");
        return 1;
    }

    printf("Enter second number: ");
    if (scanf("%d", &num2) != 1) {
        fprintf(stderr, "Invalid input for second number\n");
        return 1;
    }

    result = is_product_even(num1, num2);

    if (result == 1) {
        printf("The product is even\n");
    } else {
        printf("The product is odd\n");
    }

    return 0;
}
