```c
#include <stdio.h>
#include <stdbool.h>

bool areAllCharactersUnique(const char *str) {
    if (str == NULL) {
        return false;
    }

    bool char_set[256] = {false};

    for (int i = 0; str[i] != '\0'; i++) {
        unsigned char val = (unsigned char)str[i];
        if (char_set[val]) {
            return false;
        }
        char_set[val] = true;
    }

    return true;
}

int main(void) {
    const char *tests[] = {
        "abcdefg",
        "hello",
        "",
        NULL
    };

    for (int i = 0; i < 4; i++) {
        if (areAllCharactersUnique(tests[i])) {
            printf("test%d: Unique\n", i + 1);
        } else {
            printf("test%d: Not Unique\n", i + 1);
        }
    }

    return 0;
}
```
