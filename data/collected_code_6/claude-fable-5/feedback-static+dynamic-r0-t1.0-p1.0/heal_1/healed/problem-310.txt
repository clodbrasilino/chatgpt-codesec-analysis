```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *items;
    size_t count;
} Tuple;

Tuple *string_to_tuple(const char *input)
{
    Tuple *tuple = NULL;
    size_t length = 0;
    char *items = NULL;

    if (input == NULL) {
        return NULL;
    }

    length = strlen(input);
    if (length == 0) {
        return NULL;
    }

    items = malloc(length);
    if (items == NULL) {
        return NULL;
    }

    memcpy(items, input, length);

    tuple = malloc(sizeof(Tuple));
    if (tuple == NULL) {
        free(items);
        return NULL;
    }

    tuple->items = items;
    tuple->count = length;
    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if (tuple == NULL) {
        return;
    }

    printf("(");
    for (i = 0; i < tuple->count; i++) {
        printf("'%c'", tuple->items[i]);
        if (i + 1 < tuple->count) {
            printf(", ");
        }
    }
    printf(")\n");
}

int main(void)
{
    const char *inputs[] = { "python3.0", "item1", "15.10" };
    size_t i;
    size_t n = sizeof(inputs) / sizeof(inputs[0]);

    for (i = 0; i < n; i++) {
        Tuple *tuple = string_to_tuple(inputs[i]);

        if (tuple == NULL) {
            fprintf(stderr, "Failed to convert string to tuple: \"%s\"\n",
                    inputs[i]);
            return EXIT_FAILURE;
        }

        printf("Input string: \"%s\"\n", inputs[i]);
        printf("Tuple: ");
        print_tuple(tuple);

        free_tuple(tuple);
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the failing test cases: the expected outputs (`['p','y','t','h','o','n','3','.','0']`, `['i','t','e','m','1']`, `['1','5','.','1','0']`) show the function is supposed to convert a string into a tuple of its individual characters, not parse comma-separated integers. The original implementation returned `NULL` for all three inputs because it rejected any token that failed integer parsing.

Additional fixes for the flagged weaknesses:

- **CWE-120 (strcpy)**: Replaced `strcpy` with a bounded `memcpy` using a length computed once, eliminating the unbounded-copy warning.
- **CWE-126 (strlen over-read)**: `strlen` is still required for a C string API, but the length is computed exactly once and used consistently for allocation and copy, so allocation and copy can never disagree.
- The integer parsing, `strtok_r`, and realloc-growth logic were removed entirely since character-wise conversion needs no tokenization, which also removes the `SIZE_MAX` overflow-check path and the missing `<stdint.h>` dependency in the original.
