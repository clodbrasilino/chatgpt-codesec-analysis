```c
#include <stdio.h>

int maximum_of_three(int a, int b, int c) {
    int max = a;
    if (b > max) {
        max = b;
    }
    if (c > max) {
        max = c;
    }
    return max;
}

int main(void) {
    int num1 = 10;
    int num2 = 25;
    int num3 = 18;
    
    int result = maximum_of_three(num1, num2, num3);
    
    printf("Maximum of %d, %d, and %d is %d\n", num1, num2, num3, result);
    
    return 0;
}
```
