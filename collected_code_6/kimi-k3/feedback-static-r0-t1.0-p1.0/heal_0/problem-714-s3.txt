#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <stdbool.h>

int count_distinct_prime_powers(unsigned long long n) {
    if (n <= 1) {
        return 0;
    }
    
    int count = 0;
    
    if (n % 2 == 0) {
        count++;
        while (n % 2 == 0) {
            n /= 2;
        }
    }
    
    for (unsigned long long i = 3; i <= n / i; i += 2) {
        if (n % i == 0) {
            count++;
            while (n % i == 0) {
                n /= i;
            }
        }
    }
    
    if (n > 1) {
        count++;
    }
    
    return count;
}

int main(void) {
    unsigned long long num;
    
    if (scanf("%llu", &num) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    int result = count_distinct_prime_powers(num);
    printf("%d\n", result);
    
    return EXIT_SUCCESS;
}
