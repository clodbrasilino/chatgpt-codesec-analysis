```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define BUFFER_SIZE 256

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }
    return i;
}

static void toggle_case(char *str, size_t max_len)
{
    size_t i;
    size_t len;

    if (str == NULL || max_len == 0) {
        return;
    }

    len = bounded_strlen(str, max_len);
    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        if (isupper(c)) {
            str[i] = (char)tolower(c);
        } else if (islower(c)) {
            str[i] = (char)toupper(c);
        }
    }
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char *newline;

    memset(buffer, 0, sizeof(buffer));

    if (printf("Enter a string: ") < 0) {
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    newline = strchr(buffer, '\n');
    if (newline != NULL) {
        *newline = '\0';
    }

    toggle_case(buffer, sizeof(buffer));

    if (printf("Toggled string: %s\n", buffer) < 0) {
        fprintf(stderr, "Error writing output\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code already mitigates the Flawfinder-flagged weakness (CWE-119/CWE-120) around the statically-sized `buffer`:

1. **Bounded input** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits reads to the buffer size, preventing the classic `gets()`-style overflow.
2. **Guaranteed termination** — `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant after `fgets`, but harmless and robust against future refactoring).
3. **Bounded string operations** — `toggle_case()` takes an explicit `max_len` and uses `bounded_strlen()` rather than raw `strlen()`, so it never scans past the buffer even if termination were missing.
4. **Correct `ctype.h` usage** — casting to `unsigned char` before `isupper()`/`islower()`/`toupper()`/`tolower()` avoids undefined behavior with negative `char` values (e.g., high-bit characters on platforms with signed `char`).
5. **Zero-initialization** — `memset` ensures no uninitialized reads.
6. **Checked I/O** — return values of `printf`, `fflush`, and `fgets` are all verified.

The only change made in the rewrite is giving `toggle_case` internal linkage (`static`), which is good hygiene for a single-translation-unit program. No functional fixes were required — the flagged risk was already properly addressed.
