```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int max_subarray_sum(const int *arr, size_t n, int *result)
{
    size_t i;
    long long current;
    long long best;

    if (arr == NULL || n == 0 || result == NULL) {
        return -1;
    }

    current = arr[0];
    best = arr[0];

    for (i = 1; i < n; i++) {
        if (current > 0) {
            current = current + (long long)arr[i];
        } else {
            current = arr[i];
        }
        if (current > best) {
            best = current;
        }
    }

    if (best > INT_MAX || best < INT_MIN) {
        return -1;
    }

    *result = (int)best;
    return 0;
}

int main(void)
{
    int arr[] = { -2, 1, -3, 4, -1, 2, 1, -5, 4 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int result = 0;

    if (max_subarray_sum(arr, n, &result) != 0) {
        fprintf(stderr, "Error: could not compute maximum subarray sum\n");
        return EXIT_FAILURE;
    }

    if (printf("Largest sum of contiguous subarray: %d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Key change:** The problematic condition `current + arr[i] > arr[i]` was replaced with the mathematically equivalent `current > 0`. The original form is a classic invalid overflow test — since signed integer overflow is undefined behavior in C, optimizing compilers are permitted to assume it never happens and will simplify `current + arr[i] > arr[i]` to `current > 0` anyway, silently removing any intended overflow "protection." Writing `current > 0` directly expresses the actual Kadane's algorithm decision (extend the running sum only if it's positive) without relying on undefined behavior.

Additionally, the accumulation now casts `arr[i]` to `long long` explicitly before addition. Since `current` is already `long long` and `n` is bounded well below the point where `n * INT_MAX` could exceed `LLONG_MAX` in any realistic scenario, the arithmetic is safe, and the final range check against `INT_MAX`/`INT_MIN` before narrowing back to `int` remains a correct guard against truncation.
