```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 1024

static unsigned long long count_substrings(const char *str, size_t max_len)
{
    unsigned long long n;

    if (str == NULL)
    {
        return 0ULL;
    }

    n = (unsigned long long)strnlen(str, max_len);
    return (n * (n + 1ULL)) / 2ULL;
}

int main(void)
{
    char *buffer;
    unsigned long long result;
    size_t len;

    buffer = calloc(BUFFER_SIZE, sizeof(char));
    if (buffer == NULL)
    {
        fprintf(stderr, "Error: memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, BUFFER_SIZE, stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    buffer[BUFFER_SIZE - 1] = '\0';

    len = strnlen(buffer, BUFFER_SIZE);
    if (len > 0 && buffer[len - 1] == '\n')
    {
        buffer[len - 1] = '\0';
    }

    result = count_substrings(buffer, BUFFER_SIZE);

    if (printf("%llu\n", result) < 0)
    {
        free(buffer);
        return EXIT_FAILURE;
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

**Analysis of the original source:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120):

1. **Bounded read:** `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow.
2. **Guaranteed termination:** `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant since `fgets` always null-terminates on success, but harmless).
3. **Bounded length calculation:** `strnlen` is used instead of `strlen`, so even a non-terminated buffer can't cause an out-of-bounds read.
4. **NULL check:** `count_substrings` guards against a NULL pointer.
5. **Overflow-safe arithmetic:** the substring count `n*(n+1)/2` uses `unsigned long long`; with `n ≤ 1023` there is no risk of wraparound.
6. **Error handling:** both `fgets` and `printf` return values are checked.

The rewrite replaces the statically-sized stack array with a heap allocation via `calloc` (zero-initialized, with allocation failure checked and memory freed on all exit paths), which directly addresses the Flawfinder "statically-sized array" flag while preserving all the existing bounds-checking discipline.
