```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

#define ZODIAC_COUNT 12
#define MIN_YEAR 1
#define MAX_YEAR 9999

static const char *get_chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return NULL;
    }
    return signs[index];
}

static int display_zodiac(int year)
{
    const char *sign;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        fprintf(stderr, "Error: year must be between %d and %d.\n", MIN_YEAR, MAX_YEAR);
        return -1;
    }
    sign = get_chinese_zodiac(year);
    if (sign == NULL) {
        fprintf(stderr, "Error: could not determine zodiac sign.\n");
        return -1;
    }
    if (printf("%s\n", sign) < 0) {
        return -1;
    }
    if (fflush(stdout) != 0) {
        return -1;
    }
    return 0;
}

static int parse_year(const char *text, int *out_year)
{
    char *endptr;
    long value;

    if (text == NULL || out_year == NULL) {
        return -1;
    }
    errno = 0;
    value = strtol(text, &endptr, 10);
    if (errno != 0 || endptr == text || *endptr != '\0') {
        return -1;
    }
    if (value < INT_MIN || value > INT_MAX) {
        return -1;
    }
    *out_year = (int)value;
    return 0;
}

int main(int argc, char *argv[])
{
    int year;
    int result;

    if (argc > 1) {
        if (parse_year(argv[1], &year) != 0) {
            fprintf(stderr, "Error: invalid input, expected an integer year.\n");
            return EXIT_FAILURE;
        }
    } else {
        result = scanf("%d", &year);
        if (result != 1) {
            fprintf(stderr, "Error: invalid input, expected an integer year.\n");
            return EXIT_FAILURE;
        }
    }
    if (display_zodiac(year) != 0) {
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **Test failures (`<no output>` for Ox, Tiger, Dog):** The tests expected just the zodiac sign name, but the original program printed a full sentence and required an interactive prompt. The rewrite outputs only the sign followed by a newline, and removes the `"Enter a year: "` prompt that polluted stdout.

2. **Input flexibility:** The program now accepts the year either as a command-line argument or from stdin, so automated test harnesses can invoke it either way — a likely cause of the missing output.

3. **CWE-119/CWE-120 (buffer bounds):** The `signs` array indexing was already defensively bounds-checked; this is preserved. The array is `static const char *const`, immutable and never written to, and the index is normalized for negative modulo results before the explicit range check.

4. **Robust argument parsing:** `strtol` with `errno`, endpointer, and `INT_MIN`/`INT_MAX` range validation replaces any unsafe conversion, rejecting trailing garbage and overflow.

5. **Output error handling:** Return values of `printf` and `fflush` are still checked so I/O failures propagate to a non-zero exit status.
