#include <stdio.h>
#include <string.h>

int find_last_char(const char *str, char ch) {
    int index = -1;
    for (int i = 0; str[i] != '\0'; ++i) {
        if (str[i] == ch) {
            index = i;
        }
    }
    return index;
}

int main() {
    const char *str = "Hello, World!";
    char ch = 'o';
    int index = find_last_char(str, ch);

    if (index != -1) {
        printf("Last occurrence of '%c' is at index: %d\n", ch, index);
    } else {
        printf("Character '%c' not found in the string.\n", ch);
    }

    return 0;
}
