```c
#include <stdio.h>

int count_equal_element_pairs(int arr[], int len) {
    int i, j, count = 0;

    for(i=0; i<len; i++) {
        for(j=i+1; j<len; j++) {
            if(arr[i] == arr[j]) {
                count++;
            }
        }
    }

    return count;
}

int main() {
    int arr[] = {1, 2, 3, 2, 1, 2, 3, 2, 1};
    int len = sizeof(arr) / sizeof(arr[0]);

    printf("The number of equal element pairs is: %d\n", count_equal_element_pairs(arr, len));

    return 0;
}
```
