#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stddef.h>
#include <limits.h>
#include <stdint.h>

static char** allocate_result_array(char** result, int* capacity, int count) {
    if (count >= *capacity) {
        int new_capacity = *capacity == 0 ? 4 : *capacity * 2;
        if (new_capacity > INT_MAX / (int)sizeof(char*)) {
            return NULL;
        }
        char** new_result = realloc(result, (size_t)new_capacity * sizeof(char*));
        if (new_result == NULL) {
            free(result);
            return NULL;
        }
        *capacity = new_capacity;
        return new_result;
    }
    return result;
}

static void cleanup_on_error(char** result, int count, regex_t* regex) {
    if (result != NULL) {
        for (int i = 0; i < count; i++) {
            free(result[i]);
        }
        free(result);
    }
    regfree(regex);
}

char** split_string(const char* input, const char* pattern, int* count) {
    if (input == NULL || pattern == NULL || count == NULL) {
        return NULL;
    }

    regex_t regex;
    int ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    char** result = NULL;
    int capacity = 0;
    *count = 0;

    const char* current = input;
    const char* search_start = input;
    regmatch_t match;
    size_t input_len = strnlen(input, SIZE_MAX / 2);

    while (1) {
        int match_result = regexec(&regex, search_start, 1, &match, 0);
        
        if (match_result == REG_NOMATCH) {
            ptrdiff_t offset = current - input;
            if (offset < (ptrdiff_t)input_len) {
                size_t remaining_len = input_len - (size_t)offset;
                if (remaining_len > SIZE_MAX - 1) {
                    cleanup_on_error(result, *count, &regex);
                    return NULL;
                }
                
                char** new_result = allocate_result_array(result, &capacity, *count);
                if (new_result == NULL) {
                    cleanup_on_error(result, *count, &regex);
                    return NULL;
                }
                result = new_result;
                
                result[*count] = malloc(remaining_len + 1);
                if (result[*count] == NULL) {
                    cleanup_on_error(result, *count, &regex);
                    return NULL;
                }
                
                if (remaining_len > 0) {
                    if (remaining_len + 1 < remaining_len) {
                        cleanup_on_error(result, *count, &regex);
                        return NULL;
                    }
                    if (remaining_len > SIZE_MAX) {
                        cleanup_on_error(result, *count, &regex);
                        return NULL;
                    }
                    memcpy(result[*count], current, remaining_len);
                }
                result[*count][remaining_len] = '\0';
                (*count)++;
            }
            break;
        } else if (match_result != 0) {
            cleanup_on_error(result, *count, &regex);
            return NULL;
        }

        size_t search_offset = (size_t)(search_start - current);
        if (search_offset > SIZE_MAX - (size_t)match.rm_so) {
            cleanup_on_error(result, *count, &regex);
            return NULL;
        }
        size_t token_len = search_offset + (size_t)match.rm_so;
        
        if (token_len > 0) {
            if (token_len > SIZE_MAX - 1) {
                cleanup_on_error(result, *count, &regex);
                return NULL;
            }
            
            char** new_result = allocate_result_array(result, &capacity, *count);
            if (new_result == NULL) {
                cleanup_on_error(result, *count, &regex);
                return NULL;
            }
            result = new_result;
            
            char* token = malloc(token_len + 1);
            if (token == NULL) {
                cleanup_on_error(result, *count, &regex);
                return NULL;
            }
            
            if (token_len + 1 < token_len) {
                free(token);
                cleanup_on_error(result, *count, &regex);
                return NULL;
            }
            if (token_len > SIZE_MAX) {
                free(token);
                cleanup_on_error(result, *count, &regex);
                return NULL;
            }
            memcpy(token, current, token_len);
            token[token_len] = '\0';
            result[*count] = token;
            (*count)++;
        }

        current = search_start + match.rm_eo;
        search_start = current;

        if (*current == '\0') {
            break;
        }
    }

    regfree(&regex);
    return result;
}

void free_split_result(char** result, int count) {
    if (result == NULL) {
        return;
    }
    for (int i = 0; i < count; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void) {
    const char* input = "Hello,world;this:is|a test";
    const char* pattern = "[,;:| ]+";
    int count = 0;
    
    char** tokens = split_string(input, pattern, &count);
    
    if (tokens == NULL) {
        printf("Failed to split string\n");
        return 1;
    }
    
    printf("Split into %d tokens:\n", count);
    for (int i = 0; i < count; i++) {
        printf("Token %d: '%s'\n", i, tokens[i]);
    }
    
    free_split_result(tokens, count);
    return 0;
}
