```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} Entry;

typedef struct {
    Entry *entries;
    size_t size;
    size_t capacity;
} Dict;

static Dict *dict_create(void)
{
    Dict *d = malloc(sizeof(Dict));
    if (d == NULL) {
        return NULL;
    }
    d->size = 0;
    d->capacity = 8;
    d->entries = malloc(d->capacity * sizeof(Entry));
    if (d->entries == NULL) {
        free(d);
        return NULL;
    }
    return d;
}

static void dict_destroy(Dict *d)
{
    if (d == NULL) {
        return;
    }
    for (size_t i = 0; i < d->size; i++) {
        free(d->entries[i].key);
        free(d->entries[i].value);
    }
    free(d->entries);
    free(d);
}

static char *dup_string(const char *s)
{
    size_t len = strlen(s) + 1;
    char *copy = malloc(len);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, s, len);
    return copy;
}

static int dict_set(Dict *d, const char *key, const char *value)
{
    if (d == NULL || key == NULL || value == NULL) {
        return -1;
    }
    for (size_t i = 0; i < d->size; i++) {
        if (strcmp(d->entries[i].key, key) == 0) {
            char *new_value = dup_string(value);
            if (new_value == NULL) {
                return -1;
            }
            free(d->entries[i].value);
            d->entries[i].value = new_value;
            return 0;
        }
    }
    if (d->size == d->capacity) {
        if (d->capacity > SIZE_MAX / 2 / sizeof(Entry)) {
            return -1;
        }
        size_t new_capacity = d->capacity * 2;
        Entry *tmp = realloc(d->entries, new_capacity * sizeof(Entry));
        if (tmp == NULL) {
            return -1;
        }
        d->entries = tmp;
        d->capacity = new_capacity;
    }
    char *k = dup_string(key);
    if (k == NULL) {
        return -1;
    }
    char *v = dup_string(value);
    if (v == NULL) {
        free(k);
        return -1;
    }
    d->entries[d->size].key = k;
    d->entries[d->size].value = v;
    d->size++;
    return 0;
}

static Dict *dict_merge(const Dict *a, const Dict *b)
{
    if (a == NULL || b == NULL) {
        return NULL;
    }
    Dict *merged = dict_create();
    if (merged == NULL) {
        return NULL;
    }
    for (size_t i = 0; i < a->size; i++) {
        if (dict_set(merged, a->entries[i].key, a->entries[i].value) != 0) {
            dict_destroy(merged);
            return NULL;
        }
    }
    for (size_t i = 0; i < b->size; i++) {
        if (dict_set(merged, b->entries[i].key, b->entries[i].value) != 0) {
            dict_destroy(merged);
            return NULL;
        }
    }
    return merged;
}

static void dict_print(const Dict *d)
{
    if (d == NULL) {
        return;
    }
    printf("{");
    for (size_t i = 0; i < d->size; i++) {
        printf("\"%s\": \"%s\"", d->entries[i].key, d->entries[i].value);
        if (i + 1 < d->size) {
            printf(", ");
        }
    }
    printf("}\n");
}

int main(void)
{
    Dict *d1 = dict_create();
    if (d1 == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        return EXIT_FAILURE;
    }
    Dict *d2 = dict_create();
    if (d2 == NULL) {
        fprintf(stderr, "Failed to create dictionary\n");
        dict_destroy(d1);
        return EXIT_FAILURE;
    }

    if (dict_set(d1, "apple", "1") != 0 ||
        dict_set(d1, "banana", "2") != 0 ||
        dict_set(d1, "cherry", "3") != 0) {
        fprintf(stderr, "Failed to populate dictionary 1\n");
        dict_destroy(d1);
        dict_destroy(d2);
        return EXIT_FAILURE;
    }

    if (dict_set(d2, "banana", "20") != 0 ||
        dict_set(d2, "date", "4") != 0) {
        fprintf(stderr, "Failed to populate dictionary 2\n");
        dict_destroy(d1);
        dict_destroy(d2);
        return EXIT_FAILURE;
    }

    printf("Dictionary 1: ");
    dict_print(d1);
    printf("Dictionary 2: ");
    dict_print(d2);

    Dict *merged = dict_merge(d1, d2);
    if (merged == NULL) {
        fprintf(stderr, "Failed to merge dictionaries\n");
        dict_destroy(d1);
        dict_destroy(d2);
        return EXIT_FAILURE;
    }

    printf("Merged: ");
    dict_print(merged);

    dict_destroy(d1);
    dict_destroy(d2);
    dict_destroy(merged);
    return EXIT_SUCCESS;
}
```

Note: add `#include <stdint.h>` if your compiler does not define `SIZE_MAX` via `<stdlib.h>`; strictly speaking it lives in `<stdint.h>`, so the portable version of the includes is:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
```
