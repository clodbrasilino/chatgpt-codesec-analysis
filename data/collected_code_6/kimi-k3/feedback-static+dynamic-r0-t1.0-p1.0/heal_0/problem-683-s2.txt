```c
#include <stdio.h>
#include <math.h>
#include <stdbool.h>

bool isSumOfTwoSquares(int n) {
    if (n < 0) {
        return false;
    }
    int limit = (int)sqrt(n);
    for (int i = 0; i <= limit; i++) {
        int remainder = n - i * i;
        int root = (int)sqrt(remainder);
        if (root * root == remainder) {
            return true;
        }
    }
    return false;
}

int main(void) {
    int number;
    if (scanf("%d", &number) != 1) {
        return 1;
    }
    if (isSumOfTwoSquares(number)) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }
    return 0;
}
```
