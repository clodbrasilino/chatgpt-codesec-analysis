#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char* abbreviate_road(const char* input) {
    if (input == NULL) {
        return NULL;
    }

    size_t len = strlen(input);
    char* result = (char*)malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    size_t i = 0;
    size_t j = 0;

    while (i < len) {
        if (i + 3 < len && 
            tolower((unsigned char)input[i]) == 'r' &&
            tolower((unsigned char)input[i+1]) == 'o' &&
            tolower((unsigned char)input[i+2]) == 'a' &&
            tolower((unsigned char)input[i+3]) == 'd' &&
            (i == 0 || !isalpha((unsigned char)input[i-1])) &&
            (i + 4 == len || !isalpha((unsigned char)input[i+4]))) {
            
            result[j++] = 'r';
            result[j++] = 'd';
            result[j++] = '.';
            i += 4;
        } else {
            result[j++] = input[i++];
        }
    }
    result[j] = '\0';

    char* shrunk_result = (char*)realloc(result, j + 1);
    if (shrunk_result != NULL) {
        result = shrunk_result;
    }

    return result;
}

int main(void) {
    const char* test_strings[] = {
        "The road to success",
        "Abbey Road",
        "Broad street",
        "road",
        "roadrunner",
        "Off-road vehicle",
        NULL
    };

    for (int i = 0; test_strings[i] != NULL; i++) {
        char* abbreviated = abbreviate_road(test_strings[i]);
        if (abbreviated != NULL) {
            printf("Original: '%s'\nAbbreviated: '%s'\n\n", test_strings[i], abbreviated);
            free(abbreviated);
        } else {
            fprintf(stderr, "Memory allocation failed for string: %s\n", test_strings[i]);
        }
    }

    return 0;
}
