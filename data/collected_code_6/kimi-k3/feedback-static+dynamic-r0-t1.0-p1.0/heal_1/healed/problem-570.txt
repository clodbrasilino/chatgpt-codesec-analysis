#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int remove_words_containing(char **list, int size, const char *substring) {
    int i, j;
    int removed = 0;
    
    if (list == NULL || substring == NULL || size < 0) {
        return -1;
    }
    
    for (i = 0, j = 0; i < size; i++) {
        if (list[i] == NULL) {
            continue;
        }
        if (strstr(list[i], substring) != NULL) {
            free(list[i]);
            list[i] = NULL;
            removed++;
        } else {
            list[j] = list[i];
            if (i != j) {
                list[i] = NULL;
            }
            j++;
        }
    }
    
    return j;
}
