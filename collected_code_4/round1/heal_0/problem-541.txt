```C
#include <stdio.h>

int checkAbundant(int num) {
  int sum = 0;

  for (int i = 1; i <= num / 2; i++) {
    if (num % i == 0) {
      sum += i;
    }
  }

  if (sum > num) {
    return 1; //number is abundant
  } else {
    return 0; //number is not abundant
  }
}

int main() {
  int num;
  
  printf("Enter a number: ");
  scanf("%d", &num);
  
  if (checkAbundant(num) == 1) {
    printf("%d is an abundant number.\n", num);
  } else {
    printf("%d is not an abundant number.\n", num);
  }
  
  return 0;
}
```
