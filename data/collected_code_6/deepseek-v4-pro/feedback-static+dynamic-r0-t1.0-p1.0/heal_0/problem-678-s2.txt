```c
#include <stdio.h>
#include <string.h>

void remove_spaces(char *str) {
    if (str == NULL) {
        return;
    }
    
    int i = 0, j = 0;
    while (str[i] != '\0') {
        if (str[i] != ' ') {
            str[j++] = str[i];
        }
        i++;
    }
    str[j] = '\0';
}

int main(void) {
    char str[] = "Hello World This is C Programming";
    
    printf("Original string: \"%s\"\n", str);
    remove_spaces(str);
    printf("After removing spaces: \"%s\"\n", str);
    
    return 0;
}
```
