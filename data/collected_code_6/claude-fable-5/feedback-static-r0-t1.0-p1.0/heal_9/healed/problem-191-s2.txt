#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static bool normalize_month(const char *month, char *out, size_t out_size)
{
    size_t len;
    size_t i;

    if (month == NULL || out == NULL || out_size == 0) {
        return false;
    }

    len = strnlen(month, out_size);
    if (len == 0 || len >= out_size) {
        return false;
    }

    for (i = 0; i < len; i++) {
        out[i] = (char)tolower((unsigned char)month[i]);
    }
    out[len] = '\0';

    return true;
}

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static bool is_valid_month(const char *month)
{
    static const char *all_months[] = {
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    };
    char normalized[MAX_INPUT_LEN];
    size_t i;

    if (!normalize_month(month, normalized, sizeof(normalized))) {
        return false;
    }

    for (i = 0; i < sizeof(all_months) / sizeof(all_months[0]); i++) {
        if (strcmp(normalized, all_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        discard_remaining_input();
    }

    if (!is_valid_month(input)) {
        fprintf(stderr, "Invalid month name: %s\n", input);
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}

1. **Statically-sized `char` arrays (CWE-119/CWE-120)** — All three flagged buffers (`normalized` in both helper functions and `input` in `main`) are already properly bounds-checked. `fgets` is called with `sizeof(input)`, which guarantees no overflow, and a null terminator is forced at `input[sizeof(input) - 1]`. The `normalize_month` function receives the destination size explicitly, uses `strnlen` bounded by `out_size`, and rejects any string that would not fit (`len >= out_size`), so writes to `normalized` can never exceed its bounds.

2. **`getchar` in a loop (CWE-120/CWE-20)** — This is a false positive. `discard_remaining_input` reads and discards characters into a single `int`; no buffer is written, so no overflow is possible. The `EOF` check also prevents an infinite loop on closed input streams.

3. **Input validation (CWE-20)** — Input is validated against a whitelist of month names before use, `tolower` is called with the argument cast to `unsigned char` (avoiding undefined behavior on negative `char` values), and overlong input lines are drained via `discard_remaining_input` so stray characters don't affect subsequent reads.
