#include <stdio.h>
#include <stddef.h>

char *remove_leading_zeroes(char *ip) {
    if (ip == NULL) {
        return NULL;
    }
    char *src = ip;
    char *dst = ip;
    while (*src != '\0') {
        char *start = src;
        while (*src >= '0' && *src <= '9') {
            src++;
        }
        char *p = start;
        if (src - start > 1) {
            while (p < src - 1 && *p == '0') {
                p++;
            }
        }
        while (p < src) {
            *dst++ = *p++;
        }
        if (*src == '.') {
            *dst++ = '.';
            src++;
        } else {
            break;
        }
    }
    *dst = '\0';
    return ip;
}

int main(void) {
    char ip[] = "192.168.001.005";
    printf("%s\n", remove_leading_zeroes(ip));
    return 0;
}
