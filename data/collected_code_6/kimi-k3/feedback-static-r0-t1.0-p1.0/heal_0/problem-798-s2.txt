#include <stdio.h>
#include <stddef.h>
#include <limits.h>

int sum_array(const int *arr, size_t size, int *result) {
    size_t i;
    int sum = 0;

    if (arr == NULL || result == NULL) {
        return -1;
    }

    if (size == 0) {
        *result = 0;
        return 0;
    }

    for (i = 0; i < size; i++) {
        if ((arr[i] > 0 && sum > INT_MAX - arr[i]) ||
            (arr[i] < 0 && sum < INT_MIN - arr[i])) {
            return -1;
        }
        sum += arr[i];
    }

    *result = sum;
    return 0;
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int result;

    if (sum_array(arr, size, &result) == 0) {
        printf("Sum: %d\n", result);
    } else {
        printf("Error occurred\n");
    }

    return 0;
}
