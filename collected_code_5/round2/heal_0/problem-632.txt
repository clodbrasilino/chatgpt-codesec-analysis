```C
#include <stdio.h>

void moveZeroesToEnd(int arr[], int n) {
    int count = 0; 

    for (int i = 0; i < n; i++) {
        if (arr[i] != 0) {
            arr[count++] = arr[i];
        }
    }
  
    while (count < n) {
        arr[count++] = 0;
    }
}

int main() {
    int arr[] = {0, 1, 9, 8, 0, 0, 2, 3, 0, 6};
    int n = sizeof(arr) / sizeof(arr[0]);
    moveZeroesToEnd(arr, n);
    printf("Array after pushing all zeros to end: ");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}
```
