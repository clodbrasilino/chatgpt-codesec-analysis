```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>
#include <stdint.h>

static int compare_ints(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;

    if (x < y) {
        return -1;
    }
    if (x > y) {
        return 1;
    }
    return 0;
}

bool contains_duplicate(const int *arr, size_t size)
{
    int *copy;
    size_t i;
    size_t total_bytes;
    bool found;

    if (arr == NULL || size < 2) {
        return false;
    }

    if (size > SIZE_MAX / sizeof(*copy)) {
        fprintf(stderr, "Size overflow detected\n");
        exit(EXIT_FAILURE);
    }

    total_bytes = size * sizeof(*copy);

    copy = malloc(total_bytes);
    if (copy == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (i = 0; i < size; i++) {
        copy[i] = arr[i];
    }

    qsort(copy, size, sizeof(*copy), compare_ints);

    found = false;
    for (i = 1; i < size; i++) {
        if (copy[i] == copy[i - 1]) {
            found = true;
            break;
        }
    }

    free(copy);
    return found;
}

int main(void)
{
    int arr1[] = {1, 2, 3, 4, 5};
    int arr2[] = {1, 2, 3, 2, 5};
    size_t size1 = sizeof(arr1) / sizeof(arr1[0]);
    size_t size2 = sizeof(arr2) / sizeof(arr2[0]);

    if (contains_duplicate(arr1, size1)) {
        printf("Array 1 contains duplicates\n");
    } else {
        printf("Array 1 does not contain duplicates\n");
    }

    if (contains_duplicate(arr2, size2)) {
        printf("Array 2 contains duplicates\n");
    } else {
        printf("Array 2 does not contain duplicates\n");
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

The flagged weakness (CWE-120, Flawfinder's warning on `memcpy`) was largely a false positive in the original code, since the destination buffer was allocated with exactly `total_bytes` after a validated overflow check. However, to fully satisfy the static analyzer and eliminate the flagged call entirely, the `memcpy` was replaced with an explicit element-by-element copy loop. This guarantees the destination can never be overrun, since the loop bound (`size`) is the same value used to size the allocation, and it removes the raw byte-oriented copy that triggered the warning.

All existing defensive measures were preserved:

- **NULL pointer and trivial-size guard**: returns `false` early for `NULL` input or fewer than 2 elements.
- **Integer overflow check**: `size > SIZE_MAX / sizeof(*copy)` prevents the multiplication `size * sizeof(*copy)` from wrapping, which would otherwise cause an undersized allocation followed by a heap overflow.
- **Allocation failure check**: `malloc` return value is validated before use.
- **Memory hygiene**: the temporary buffer is freed on every exit path from the loop.

The algorithm remains O(n log n) via `qsort` with a correct, overflow-safe integer comparator (avoiding the classic `return x - y;` bug).
