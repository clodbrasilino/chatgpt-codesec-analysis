```c
#include <stdio.h>
#include <string.h>

void to_lowercase(char *str) {
    for (int i = 0; str[i]; i++) {
        if (str[i] >= 'A' && str[i] <= 'Z') {
            str[i] = str[i] - 'A' + 'a';
        }
    }
}

int main() {
    char str[] = "HELLO WORLD";
    to_lowercase(str);
    printf("%s\n", str);
    return 0;
}
```
