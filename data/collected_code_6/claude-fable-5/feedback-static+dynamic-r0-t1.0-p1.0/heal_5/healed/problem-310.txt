```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096

typedef struct {
    char *items;
    size_t count;
} Tuple;

static size_t bounded_length(const char *input, size_t max_length)
{
    size_t i = 0;

    while (i < max_length && input[i] != '\0') {
        i++;
    }

    return i;
}

Tuple *string_to_tuple(const char *input)
{
    Tuple *tuple = NULL;
    size_t length = 0;
    char *items = NULL;

    if (input == NULL) {
        return NULL;
    }

    length = bounded_length(input, MAX_INPUT_LENGTH);
    if (length == 0 || length >= MAX_INPUT_LENGTH) {
        return NULL;
    }

    items = malloc(length + 1);
    if (items == NULL) {
        return NULL;
    }

    memcpy(items, input, length);
    items[length] = '\0';

    tuple = malloc(sizeof(Tuple));
    if (tuple == NULL) {
        free(items);
        return NULL;
    }

    tuple->items = items;
    tuple->count = length;
    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        tuple->items = NULL;
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if (tuple == NULL || tuple->items == NULL) {
        return;
    }

    printf("[");
    for (i = 0; i < tuple->count; i++) {
        printf("'%c'", tuple->items[i]);
        if (i + 1 < tuple->count) {
            printf(", ");
        }
    }
    printf("]\n");
    fflush(stdout);
}

int main(void)
{
    const char *inputs[] = { "python3.0", "item1", "15.10" };
    size_t i;
    size_t n = sizeof(inputs) / sizeof(inputs[0]);

    for (i = 0; i < n; i++) {
        Tuple *tuple = string_to_tuple(inputs[i]);

        if (tuple == NULL) {
            fprintf(stderr, "Failed to convert string to tuple: \"%s\"\n",
                    inputs[i]);
            return EXIT_FAILURE;
        }

        print_tuple(tuple);

        free_tuple(tuple);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

1. **`strnlen` portability (likely root cause of "no output"):** `strnlen` is a POSIX function, not part of strict ISO C90/C99. Compiling with `-std=c99` (without `_POSIX_C_SOURCE` defined) can cause an implicit-declaration warning or outright build failure, which explains all three test cases producing *no output at all* — the binary likely never built or linked. I replaced it with a portable `bounded_length` helper that performs the same bounded scan.

2. **Flawfinder `memcpy` warning (CWE-120):** This was a false positive in practice — the destination is allocated as `length + 1` and exactly `length` bytes are copied. The bounded length computation guarantees the destination can always hold the source, so the `memcpy` is retained but now fed by a fully portable, verifiably bounded length.

3. **Defensive hardening in `free_tuple`:** Setting `tuple->items = NULL` before freeing the struct guards against accidental use-after-free of the inner pointer if the code is later modified.

4. **Everything else preserved:** allocation-failure paths, the `length == 0 || length >= MAX_INPUT_LENGTH` guard (rejecting unterminated/oversized input), the exact output format expected by the tests (`['p', 'y', ...]`), and `fflush(stdout)` to ensure output is not lost in buffered/piped test harnesses.
