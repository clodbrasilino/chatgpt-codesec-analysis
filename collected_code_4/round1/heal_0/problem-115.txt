Since C is a low-level language, the concept of a dictionary (as known in Python or other high-level languages) does not exist implicitly. Therefore, we need to create our own structure to mimic this type of data structure (a dictionary is actually an associative array that works with key-value pairs). 

In this example, let's implement a very basic version of a dictionary, which handles string keys and integer values. 

Furthermore, keep in mind that C does not support dynamic resizing of arrays, so you need to manually allocate memory for the array of dictionaries.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DICTIONARY_SIZE 20
#define KEY_SIZE 100
#define DICTIONARY_LIST_SIZE 10

struct dictionary {
    char keys[DICTIONARY_SIZE][KEY_SIZE];
    int values[DICTIONARY_SIZE];
    int size;
};

void initDictionary(struct dictionary *dict) {
    dict->size = 0;
}

void addElement(struct dictionary *dict, char* key, int value) {
    strcpy(dict->keys[dict->size], key);
    dict->values[dict->size] = value;
    dict->size++;
}

int isEmpty(struct dictionary *dict) {
    return dict->size == 0;
}

int areAllDictionariesEmpty(struct dictionary *dictList) {
    for (int i = 0; i < DICTIONARY_LIST_SIZE; i++) {
        if (!isEmpty(&dictList[i])) {
            return 0;
        }
    }
    return 1;
}

int main() {
    struct dictionary dictList[DICTIONARY_LIST_SIZE];

    for (int i = 0; i < DICTIONARY_LIST_SIZE; i++) {
        initDictionary(&dictList[i]);
    }

    if (areAllDictionariesEmpty(dictList)) {
        printf("All dictionaries are empty\n");
    } else {
        printf("Not all dictionaries are empty\n");
    }

    addElement(&dictList[0], "Hello", 1);

    if (areAllDictionariesEmpty(dictList)) {
        printf("All dictionaries are empty\n");
    } else {
        printf("Not all dictionaries are empty\n");
    }

    return 0;
}
```

This program starts by initializing an array of dictionaries. Then it checks if all dictionaries are empty (they should be at this point), adding an element to the first dictionary and finally it checks again if all dictionaries are empty.
