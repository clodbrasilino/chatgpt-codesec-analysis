#include <stdio.h>
#include <stdlib.h>

int max_sublist_length(int *arr, int n) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    int max_len = 1;
    int current_len = 1;

    for (int i = 1; i < n; i++) {
        if (arr[i] >= arr[i - 1]) {
            current_len++;
            if (current_len > max_len) {
                max_len = current_len;
            }
        } else {
            current_len = 1;
        }
    }

    return max_len;
}

int main(void) {
    int arr1[] = {1, 2, 3, 2, 5, 7, 8};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    printf("%d\n", max_sublist_length(arr1, n1));

    int arr2[] = {5, 4, 3, 2, 1};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    printf("%d\n", max_sublist_length(arr2, n2));

    int arr3[] = {1};
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    printf("%d\n", max_sublist_length(arr3, n3));

    int arr4[] = {1, 1, 1, 1};
    int n4 = sizeof(arr4) / sizeof(arr4[0]);
    printf("%d\n", max_sublist_length(arr4, n4));

    printf("%d\n", max_sublist_length(NULL, 5));

    return 0;
}
