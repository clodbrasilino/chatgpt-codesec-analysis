#include <stdio.h>
#include <limits.h>

int find_most_significant_set_bit(unsigned int num) {
    if (num == 0) return -1;
    int msb = 0;
    while (num != 0) {
        num >>= 1;
        msb++;
    }
    return msb - 1;
}

int main(void) {
    unsigned int num = 18;
    int result = find_most_significant_set_bit(num);
    if (result != -1) {
        printf("The most significant set bit is at position: %d\n", result);
    } else {
        printf("No set bits found.\n");
    }
    return 0;
}
