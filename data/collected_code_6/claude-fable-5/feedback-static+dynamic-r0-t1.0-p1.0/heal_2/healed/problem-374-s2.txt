```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 64U

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void permute(char *str, size_t left, size_t right)
{
    size_t i;

    if (left == right) {
        if (puts(str) == EOF) {
            exit(EXIT_FAILURE);
        }
        return;
    }

    for (i = left; i <= right; i++) {
        swap_chars(&str[left], &str[i]);
        permute(str, left + 1U, right);
        swap_chars(&str[left], &str[i]);
    }
}

static int print_permutations(const char *input)
{
    char *buffer;
    size_t len;
    size_t buffer_size;

    if (input == NULL) {
        return -1;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len == 0U || len > MAX_INPUT_LEN) {
        return -1;
    }

    buffer_size = len + 1U;
    buffer = (char *)malloc(buffer_size);
    if (buffer == NULL) {
        return -1;
    }

    if (len >= buffer_size) {
        free(buffer);
        return -1;
    }

    memcpy(buffer, input, len);
    buffer[len] = '\0';

    permute(buffer, 0U, len - 1U);
    free(buffer);

    return 0;
}

int main(void)
{
    const char *test = "ABC";

    if (print_permutations(test) != 0) {
        (void)fprintf(stderr, "Failed to print permutations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already largely safe — the Flawfinder warning on `memcpy` was a static-analysis heuristic rather than an actual defect, since the destination buffer was allocated with `len + 1U` bytes and exactly `len` bytes were copied, followed by explicit NUL termination.

The rewritten version makes the safety property *explicit and verifiable*:

1. **Explicit buffer size tracking** — `buffer_size` is computed once and used for both the allocation and the bounds check, so the invariant `len < buffer_size` is checked in code rather than assumed by inspection. This satisfies the CWE-120 mitigation requirement ("make sure destination can always hold the source data") in a way static analyzers can follow.

2. **Retained defenses from the original:**
   - `strnlen` with `MAX_INPUT_LEN + 1U` bounds the length scan and detects over-long input without reading unbounded memory.
   - Rejection of empty strings prevents the `len - 1U` underflow (which would wrap `size_t` to `SIZE_MAX` in the `permute` call).
   - `malloc` return value is checked; `buffer` is freed on all paths after allocation.
   - Manual NUL termination guarantees `puts` operates on a valid C string.

3. **No overflow in `len + 1U`** — since `len <= MAX_INPUT_LEN` (64), the addition cannot wrap.

The recursion depth in `permute` is bounded by `MAX_INPUT_LEN`, so stack exhaustion is not a concern, though note that runtime is factorial in input length — 64 characters is astronomically infeasible to fully enumerate, which is a practical (not security) limitation.
