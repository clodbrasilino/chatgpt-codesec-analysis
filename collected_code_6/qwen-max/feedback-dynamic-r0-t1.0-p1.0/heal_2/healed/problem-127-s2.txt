#include <stdio.h>
#include <limits.h>

int multiply(int x, int y) {
    long long result = 0;
    if (y < 0) {
        y = -y;
        x = -x;
    }
    while (y > 0) {
        if (y & 1) {
            if (result > INT_MAX || result < INT_MIN) {
                return 0;
            }
            result += x;
        }
        if (x > 0 && (x > INT_MAX / 2)) {
            return 0;
        }
        if (x < 0 && (x < INT_MIN / 2)) {
            return 0;
        }
        x = (long long)x << 1;
        y >>= 1;
    }
    if (result > INT_MAX || result < INT_MIN) {
        return 0;
    }
    return (int)result;
}

int main() {
    int a, b;
    printf("Enter two integers: ");
    scanf("%d %d", &a, &b);
    int product = multiply(a, b);
    printf("Product: %d\n", product);
    return 0;
}
