#include <stdio.h>

int find_minimum(int a, int b, int c) {
    int min = a;
    if (b < min) {
        min = b;
    }
    if (c < min) {
        min = c;
    }
    return min;
}

int main(void) {
    int x = 0, y = 0, z = 0;
    if (scanf("%d %d %d", &x, &y, &z) == 3) {
        printf("%d\n", find_minimum(x, y, z));
        return 0;
    }
    return 1;
}
