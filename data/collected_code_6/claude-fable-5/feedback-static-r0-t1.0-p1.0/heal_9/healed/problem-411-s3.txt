```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

static char *cleanup_fail(regex_t *regex, char *buffer)
{
    if (regex != NULL) {
        regfree(regex);
    }
    free(buffer);
    return NULL;
}

static int safe_copy(char *dest, size_t dest_size, size_t dest_pos,
                     const char *src, size_t copy_len)
{
    if (dest == NULL || src == NULL) {
        return -1;
    }
    if (dest_pos >= dest_size) {
        return -1;
    }
    if (copy_len > dest_size - dest_pos) {
        return -1;
    }
    if (copy_len >= dest_size - dest_pos) {
        return -1;
    }
    memcpy(&dest[dest_pos], src, copy_len);
    return 0;
}

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t buf_size;
    size_t out_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        size_t prefix_len;
        size_t remaining;

        if (match[0].rm_so < 0 || match[1].rm_so < 0 ||
            match[0].rm_eo < match[0].rm_so) {
            return cleanup_fail(&regex, result);
        }

        prefix_len = (size_t)match[0].rm_so;

        if (out_pos >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        remaining = buf_size - out_pos;

        if (prefix_len >= remaining || (remaining - prefix_len) < 2U) {
            return cleanup_fail(&regex, result);
        }

        if (prefix_len > strnlen(cursor, prefix_len)) {
            return cleanup_fail(&regex, result);
        }

        if (prefix_len > 0U) {
            if (safe_copy(result, buf_size, out_pos, cursor, prefix_len) != 0) {
                return cleanup_fail(&regex, result);
            }
            out_pos += prefix_len;
        }

        if (out_pos + 1U >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        result[out_pos] = (char)toupper((unsigned char)cursor[match[1].rm_so]);
        out_pos += 1U;

        cursor += match[0].rm_eo;
    }

    {
        size_t rest_len;
        size_t remaining;

        if (out_pos >= buf_size) {
            return cleanup_fail(&regex, result);
        }

        remaining = buf_size - out_pos;
        rest_len = strnlen(cursor, remaining);

        if (rest_len >= remaining) {
            return cleanup_fail(&regex, result);
        }

        if (rest_len > 0U) {
            if (safe_copy(result, buf_size, out_pos, cursor, rest_len) != 0) {
                return cleanup_fail(&regex, result);
            }
            out_pos += rest_len;
        }
    }

    if (out_pos >= buf_size) {
        return cleanup_fail(&regex, result);
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s -> %s\n", tests[i], converted);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already defensively written — both `memcpy` calls flagged by Flawfinder (CWE-120) were preceded by explicit bounds checks. The rewrite consolidates those guards into a single `safe_copy` helper that validates the destination pointer, source pointer, current write position, and copy length against the allocated buffer size before every copy. This centralizes the overflow protection, eliminates duplicated guard logic, and makes it impossible to invoke `memcpy` on an unchecked path, directly addressing the Flawfinder findings.

Key safety properties preserved and reinforced:

1. **Input length capping** via `strnlen` with `MAX_INPUT_LEN + 1U` prevents unbounded reads on non-terminated input.
2. **Buffer sizing invariant**: the output can never exceed the input length (camelCase conversion only removes underscores), so `len + 1U` is always sufficient — every copy is still verified at runtime regardless.
3. **Regex match sanity checks** guard against negative or inverted offsets before any pointer arithmetic.
4. **All error paths** release both the compiled regex and the heap buffer through `cleanup_fail`, avoiding leaks.
5. **Null-termination** is verified in-bounds before writing the final `'\0'`.
