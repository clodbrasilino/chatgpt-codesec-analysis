```c
#include <stdio.h>
#include <stdlib.h>

long long largest_k(long long x, long long y, long long n)
{
    if (x <= 0 || y < 0 || y >= x || n < y) {
        return -1;
    }
    return n - ((n - y) % x);
}

int main(void)
{
    long long x;
    long long y;
    long long n;
    long long result;

    printf("Enter x (divisor): ");
    if (scanf("%lld", &x) != 1) {
        fprintf(stderr, "Invalid input for x\n");
        return EXIT_FAILURE;
    }

    printf("Enter y (remainder): ");
    if (scanf("%lld", &y) != 1) {
        fprintf(stderr, "Invalid input for y\n");
        return EXIT_FAILURE;
    }

    printf("Enter n (upper bound): ");
    if (scanf("%lld", &n) != 1) {
        fprintf(stderr, "Invalid input for n\n");
        return EXIT_FAILURE;
    }

    result = largest_k(x, y, n);
    if (result < 0) {
        fprintf(stderr, "No valid k exists for the given inputs\n");
        return EXIT_FAILURE;
    }

    printf("Largest k such that k mod %lld == %lld and k <= %lld is: %lld\n",
           x, y, n, result);

    return EXIT_SUCCESS;
}
```

**How it works:**

- `largest_k` validates that `x > 0`, `0 <= y < x`, and `n >= y`. If any condition fails, no valid `k` exists and it returns `-1` as a sentinel.
- The largest valid value is computed as `n - ((n - y) % x)`, which snaps `n` down to the greatest number not exceeding `n` that leaves remainder `y` when divided by `x`.
- `main` checks every `scanf` return value before using the variables, handles the error sentinel explicitly, and returns standard exit codes, avoiding unchecked exceptional conditions and undefined behavior from uninitialized reads.
