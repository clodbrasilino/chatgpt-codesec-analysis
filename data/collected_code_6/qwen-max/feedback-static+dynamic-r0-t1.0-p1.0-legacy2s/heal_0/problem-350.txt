```c
#include <stdio.h>
#include <string.h>

char* minimize_string(char *str, char ch) {
    int i, j;
    for (i = 0, j = 0; str[i] != '\0'; ++i) {
        if (str[i] != ch) {
            str[j++] = str[i];
        }
    }
    str[j] = '\0';
    return str;
}

int main() {
    char input[100] = "example string with some char";
    char ch = 'e';
    printf("Original string: %s\n", input);
    minimize_string(input, ch);
    printf("Minimized string: %s\n", input);
    return 0;
}
```
