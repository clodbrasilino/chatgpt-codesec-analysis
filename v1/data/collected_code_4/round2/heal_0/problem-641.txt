Sure, here's a simple C program that calculates the nth nonagonal number. Note that a nonagonal number for a term n is given by the formula n*(7*n-5)/2.

```c
#include <stdio.h>

long long nonagonalNumber(int n) {
    return n * ((7 * n) - 5) / 2;
}

int main() {
    int n;
    scanf("%d", &n);
    
    if(n < 1) {
        printf("Invalid input. Please enter a positive integer.\n");
        return 1;
    }
  
    printf("%lld\n", nonagonalNumber(n));
  
    return 0;
}
```
