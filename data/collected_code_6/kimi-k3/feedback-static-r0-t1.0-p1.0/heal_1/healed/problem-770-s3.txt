#include <stdio.h>

unsigned long long sumFourthPowerOdd(int n) {
    unsigned long long sum = 0;
    int i;
    
    if (n <= 0) {
        return 0;
    }
    
    for (i = 0; i < n; i++) {
        unsigned long long odd = 2ULL * i + 1ULL;
        sum += odd * odd * odd * odd;
    }
    
    return sum;
}

int main(void) {
    int n;
    unsigned long long result;
    
    printf("Enter the value of n: ");
    
    if (scanf("%d", &n) != 1) {
        printf("Invalid input\n");
        return 1;
    }
    
    if (n < 0) {
        printf("Please enter a non-negative integer\n");
        return 1;
    }
    
    result = sumFourthPowerOdd(n);
    printf("Sum of fourth power of first %d odd natural numbers: %llu\n", n, result);
    
    return 0;
}
