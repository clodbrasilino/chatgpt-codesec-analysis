```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int get_max(const int *arr, size_t n)
{
    size_t i;
    int max_val = arr[0];

    for (i = 1; i < n; i++) {
        if (arr[i] > max_val) {
            max_val = arr[i];
        }
    }
    return max_val;
}

static int counting_sort_by_digit(int *arr, size_t n, int exp)
{
    int *output;
    size_t count[10];
    size_t i;
    size_t j;
    size_t bytes;

    if (arr == NULL || n == 0 || exp <= 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(*output)) {
        return -1;
    }

    bytes = n * sizeof(*output);

    output = malloc(bytes);
    if (output == NULL) {
        return -1;
    }

    memset(count, 0, sizeof(count));

    for (i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }

    for (j = 1; j < 10; j++) {
        count[j] += count[j - 1];
    }

    i = n;
    while (i > 0) {
        i--;
        output[count[(arr[i] / exp) % 10] - 1] = arr[i];
        count[(arr[i] / exp) % 10]--;
    }

    memcpy(arr, output, bytes);
    free(output);
    return 0;
}

static int radix_sort(int *arr, size_t n)
{
    int max_val;
    int exp;

    if (arr == NULL || n == 0) {
        return -1;
    }

    max_val = get_max(arr, n);
    if (max_val < 0) {
        return -1;
    }

    for (exp = 1; max_val / exp > 0; exp *= 10) {
        if (counting_sort_by_digit(arr, n, exp) != 0) {
            return -1;
        }
        if (exp > max_val / 10) {
            break;
        }
    }
    return 0;
}

int main(void)
{
    int data[] = { 170, 45, 75, 90, 802, 24, 2, 66 };
    size_t n = sizeof(data) / sizeof(data[0]);
    size_t i;

    if (radix_sort(data, n) != 0) {
        fprintf(stderr, "Sorting failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", data[i]) < 0) {
            return EXIT_FAILURE;
        }
    }
    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The flagged `memcpy` was already safe in the original code — `output` is allocated with exactly `n * sizeof(*output)` bytes, and the copy writes exactly `n * sizeof(*arr)` bytes into `arr`, which the caller guarantees holds `n` elements. The overflow check `n > SIZE_MAX / sizeof(*output)` before `malloc` already prevents integer-overflow-based undersized allocations.

In the rewrite, the allocation size is computed once into `bytes` and reused for both `malloc` and `memcpy`, making it structurally impossible for the copy size to diverge from the allocated size. This satisfies the Flawfinder CWE-120 concern by construction: the destination and source sizes are provably identical.

Other defensive properties preserved from the original:

- **NULL/zero-length guards** in both `counting_sort_by_digit` and `radix_sort` (returns `-1`).
- **Negative-value rejection** in `radix_sort` (`max_val < 0`), since the digit extraction `(arr[i] / exp) % 10` would index `count[]` out of bounds for negative inputs.
- **`exp` overflow protection** via the `exp > max_val / 10` break, preventing signed integer overflow on `exp *= 10`.
- **Checked I/O** — `printf` return values are verified in `main`.
- Stable, correct backwards traversal in the counting sort using an underflow-safe `while (i > 0) { i--; ... }` idiom for `size_t`.
