char* remove_parenthesis(const char *input) {
    int len = strlen(input);
    int new_len = 0;

    for (int i = 0; i < len; ++i) {
        if (input[i] != '(' && input[i] != ')') {
            new_len++;
        }
    }

    char *output = (char *)malloc(new_len + 1);
    if (!output) return NULL;

    int j = 0;
    for (int i = 0; i < len; ++i) {
        if (input[i] != '(' && input[i] != ')') {
            output[j++] = input[i];
        }
    }
    output[j] = '\0';
    return output;
}
