```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(const char *input, char *output, size_t output_size);

int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len;
        size_t space_left;

        if ((match.rm_so < 0) || (match.rm_eo < match.rm_so)) {
            regfree(&regex);
            return -1;
        }

        prefix_len = (size_t)match.rm_so;
        space_left = output_size - out_pos;

        if (prefix_len >= space_left) {
            regfree(&regex);
            return -1;
        }

        if ((prefix_len + 1U) >= space_left) {
            regfree(&regex);
            return -1;
        }

        if (prefix_len > 0U) {
            memcpy(&output[out_pos], cursor, prefix_len);
            out_pos += prefix_len;
        }

        output[out_pos] = ' ';
        out_pos++;

        cursor += (size_t)match.rm_eo;
    }

    {
        size_t remaining = strlen(cursor);
        size_t space_left = output_size - out_pos;

        if (remaining >= space_left) {
            regfree(&regex);
            return -1;
        }

        if (remaining > 0U) {
            memcpy(&output[out_pos], cursor, remaining);
            out_pos += remaining;
        }

        output[out_pos] = '\0';
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256];

    memset(result, 0, sizeof(result));

    if (strlen(test) >= sizeof(result)) {
        (void)fprintf(stderr, "Input too long for buffer.\n");
        return EXIT_FAILURE;
    }

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and applied fixes:**

1. **CWE-120 (memcpy buffer overflow):** The original code already guarded both `memcpy` calls with bounds checks, but the checks were redundant and slightly confusing (`prefix_len >= (output_size - out_pos)` followed by an overlapping second check). The rewrite consolidates the logic using an explicit `space_left` variable, making it verifiable at a glance that `memcpy` never writes past `output_size`. Zero-length copies are also skipped.

2. **CWE-119/CWE-120 (statically-sized array in `main`):** The fixed-size `result[256]` buffer is now validated in `main` before use — the input length is checked against the buffer capacity up front, guaranteeing the destination can always hold the worst-case output (which can never exceed the input length for this transformation).

3. **`strnlen(cursor, output_size)` bug:** The original used `output_size` as the limit for the *remaining input*, which could silently truncate detection of an over-long tail and pass a misleading length to the bounds check. Replaced with `strlen`, since `cursor` points into a valid NUL-terminated string; the subsequent `remaining >= space_left` check safely rejects anything that won't fit.

4. **Regex match sanity:** Added a check for `match.rm_eo < match.rm_so` to defend against a malformed match result before advancing the cursor, avoiding any possibility of a negative/wild pointer offset.

5. **Resource management:** `regfree` is called on every exit path after `regcomp` succeeds, so there are no regex memory leaks — this property from the original is preserved.
