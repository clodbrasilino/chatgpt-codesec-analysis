```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
```

**Analysis of the original code and why this version is safe:**

The Flawfinder warning about the statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120) is a heuristic flag on any fixed-size buffer, not a confirmed vulnerability. The original code already mitigated it correctly, so the rewrite preserves those mitigations:

1. **Bounded input reading** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow. The return value is checked for `NULL` to handle EOF or read errors.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` is defensive (technically redundant since `fgets` always null-terminates on success, but harmless and explicit).

3. **Safe newline stripping** — `strcspn(buffer, "\n")` is the idiomatic, safe way to remove the trailing newline; it never indexes out of bounds since it returns at most the string length.

4. **Bounded string length** — `strnlen(str, max_len)` instead of `strlen` guarantees the scan never reads past the buffer even if the string were unterminated, and the `NULL` check guards against invalid pointers.

5. **Correct `tolower` usage** — the cast to `unsigned char` before passing to `tolower` avoids undefined behavior for negative `char` values on platforms where `char` is signed.

No functional changes were needed; the code was already free of the flagged weakness, so the rewrite simply removes the comments as requested.
