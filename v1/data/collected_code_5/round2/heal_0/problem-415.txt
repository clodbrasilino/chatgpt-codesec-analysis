```C
#include <stdio.h>
#include <limits.h>

void maxProductPair(int arr[], int n)
{
    if(n < 2)
    {
        printf("No pairs exist\n");
        return;
    }
    
    int posa = INT_MIN, posb = INT_MIN;
    int nega = INT_MIN, negb = INT_MIN;

    for(int i = 0; i < n; i++)
    {
        if(arr[i] > posa)
        {
            posb = posa;
            posa = arr[i];
        }
        else if(arr[i] > posb)
            posb = arr[i];

        if(arr[i] < 0 && abs(arr[i]) > abs(nega))
        {
            negb = nega;
            nega = arr[i];
        }
        else if(arr[i] < 0 && abs(arr[i]) > abs(negb))
            negb = arr[i];
    }

    if(nega * negb > posa * posb)
        printf("Max product pair is {%d, %d}", nega, negb);
    else
        printf("Max product pair is {%d, %d}", posa, posb);
}

int main()
{
    int arr[] = {1, 4, 3, 6, 7, 0};
    int n = sizeof(arr)/sizeof(arr[0]);
    
    maxProductPair(arr, n);

    return 0;
}
```   
This program finds the pair of numbers in an array which gives the maximum product. It handles conditions where the array has less than two elements. Also, it handles negative numbers correctly considering that multiplication of two negative numbers can give a larger product than two positive numbers. Finally, it does not use any resources improperly as it does not use any dynamic memory allocation or file handling.
