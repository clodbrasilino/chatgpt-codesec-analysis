#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void find_max_product_pair(int *arr, size_t size, int *res1, int *res2) {
    if (arr == NULL || res1 == NULL || res2 == NULL) {
        exit(EXIT_FAILURE);
    }

    if (size < 2) {
        exit(EXIT_FAILURE);
    }

    int max1 = INT_MIN;
    int max2 = INT_MIN;
    int min1 = INT_MAX;
    int min2 = INT_MAX;

    for (size_t i = 0; i < size; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    if ((long long)max1 * max2 >= (long long)min1 * min2) {
        *res1 = max1;
        *res2 = max2;
    } else {
        *res1 = min1;
        *res2 = min2;
    }
}

int main(void) {
    int arr[] = {1, 4, 3, 6, 2, -10, -9};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int res1, res2;

    find_max_product_pair(arr, size, &res1, &res2);

    printf("%d %d\n", res1, res2);

    return 0;
}
