```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

static char *snake_to_camel(const char *snake)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL) {
        return NULL;
    }

    len = strnlen(snake, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = snake[i];
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

static void strip_trailing_newlines(char *s)
{
    size_t len;

    if (s == NULL) {
        return;
    }

    len = strnlen(s, MAX_INPUT_LEN);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r')) {
        s[len - 1] = '\0';
        len--;
    }
}

int main(int argc, char *argv[])
{
    char *input;
    char *result;
    size_t arg_len;

    input = calloc(MAX_INPUT_LEN, sizeof(char));
    if (input == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (argc > 1 && argv[1] != NULL) {
        arg_len = strnlen(argv[1], MAX_INPUT_LEN);
        if (arg_len >= MAX_INPUT_LEN) {
            fprintf(stderr, "Input too long\n");
            free(input);
            return EXIT_FAILURE;
        }
        memcpy(input, argv[1], arg_len);
        input[arg_len] = '\0';
    } else {
        if (fgets(input, MAX_INPUT_LEN, stdin) == NULL) {
            fprintf(stderr, "No input provided\n");
            free(input);
            return EXIT_FAILURE;
        }
        input[MAX_INPUT_LEN - 1] = '\0';
    }

    strip_trailing_newlines(input);

    result = snake_to_camel(input);
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        free(input);
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        free(input);
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        free(result);
        free(input);
        return EXIT_FAILURE;
    }

    free(result);
    free(input);
    return EXIT_SUCCESS;
}
```

Analysis of the original code and the changes made:

**Identified issues:**

1. **CWE-120 (strncpy flaw):** The original used `strncpy`, which is a banned function because it doesn't guarantee null-termination and can silently truncate. Since the argument length was already validated with `strnlen`, the fix replaces it with `memcpy` of the exact validated length followed by explicit null-termination. This is both safer and faster (no unnecessary zero-padding of the remaining buffer).

2. **Failing test cases (`<no output>`):** The tests expected outputs like `PythonProgram`, `PythonLanguage`, and `ProgrammingLanguage`. The conversion logic itself is correct (`python_program` → `PythonProgram`). The `<no output>` symptom typically points to the program exiting on an error path before printing. The rewrite hardens the input path: the argument length is computed once, copied exactly, and terminated explicitly, eliminating any truncation/termination ambiguity that could cause the length check in `snake_to_camel` to reject the input.

3. **Defensive NULL check** added in `strip_trailing_newlines`, since it previously dereferenced its parameter unconditionally.

**What was preserved:** bounded reads via `strnlen`/`fgets`, allocation failure checks, checked `printf`/`fflush` return values, and consistent freeing of all allocations on every exit path — the original had no memory leaks or unbounded copies, which is good practice.
