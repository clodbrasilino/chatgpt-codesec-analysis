#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int** data;
    size_t rows;
    size_t* cols_per_row;
} NestedList;

int* extract_column(const NestedList* list, size_t col_index, size_t* out_size) {
    if (!list || !list->data || !list->cols_per_row || !out_size) {
        return NULL;
    }

    size_t count = 0;
    for (size_t i = 0; i < list->rows; i++) {
        if (col_index < list->cols_per_row[i]) {
            count++;
        }
    }

    *out_size = count;
    if (count == 0) {
        return NULL;
    }

    int* extracted = (int*)malloc(count * sizeof(int));
    if (!extracted) {
        *out_size = 0;
        return NULL;
    }

    size_t current = 0;
    for (size_t i = 0; i < list->rows; i++) {
        if (col_index < list->cols_per_row[i]) {
            extracted[current++] = list->data[i][col_index];
        }
    }

    return extracted;
}

int main(void) {
    NestedList list;
    list.rows = 3;
    
    list.data = (int**)malloc(list.rows * sizeof(int*));
    if (!list.data) {
        return 1;
    }

    list.cols_per_row = (size_t*)malloc(list.rows * sizeof(size_t));
    if (!list.cols_per_row) {
        free(list.data);
        return 1;
    }

    list.cols_per_row[0] = 3;
    list.cols_per_row[1] = 2;
    list.cols_per_row[2] = 4;

    for (size_t i = 0; i < list.rows; i++) {
        list.data[i] = (int*)malloc(list.cols_per_row[i] * sizeof(int));
        if (!list.data[i]) {
            for (size_t j = 0; j < i; j++) {
                free(list.data[j]);
            }
            free(list.data);
            free(list.cols_per_row);
            return 1;
        }
        for (size_t j = 0; j < list.cols_per_row[i]; j++) {
            list.data[i][j] = (int)((i + 1) * 10 + (j + 1));
        }
    }

    size_t target_col = 1;
    size_t extracted_size = 0;
    int* extracted = extract_column(&list, target_col, &extracted_size);

    if (extracted) {
        for (size_t i = 0; i < extracted_size; i++) {
            printf("%d\n", extracted[i]);
        }
        free(extracted);
    }

    for (size_t i = 0; i < list.rows; i++) {
        free(list.data[i]);
    }
    free(list.data);
    free(list.cols_per_row);

    return 0;
}
