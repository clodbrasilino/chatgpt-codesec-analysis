#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <stdint.h>

#define MAX_TEXT_LEN 1024

typedef struct {
    unsigned char element;
    size_t count;
} ElementCount;

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t i;

    if (s == NULL) {
        return 0;
    }

    for (i = 0; i < max_len; i++) {
        if (s[i] == '\0') {
            return i;
        }
    }
    return max_len;
}

int find_most_common(const char *text, size_t text_size, ElementCount *results, size_t max_results, size_t *num_results)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    size_t len;
    size_t found = 0;

    if (text == NULL || results == NULL || num_results == NULL || max_results == 0 || text_size == 0) {
        return -1;
    }

    len = bounded_strlen(text, text_size);
    if (len == text_size) {
        return -1;
    }

    if (len == 0) {
        *num_results = 0;
        return 0;
    }

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)text[i];
        if (counts[c] < SIZE_MAX) {
            counts[c]++;
        }
        if (counts[c] > max_count) {
            max_count = counts[c];
        }
    }

    for (i = 0; i <= UCHAR_MAX; i++) {
        if (counts[i] == max_count) {
            if (found < max_results) {
                results[found].element = (unsigned char)i;
                results[found].count = counts[i];
                found++;
            } else {
                break;
            }
        }
    }

    *num_results = found;
    return 0;
}

int main(void)
{
    char text[MAX_TEXT_LEN];
    ElementCount results[UCHAR_MAX + 1];
    size_t num_results = 0;
    size_t i;
    size_t len;

    memset(text, 0, sizeof(text));
    memset(results, 0, sizeof(results));

    printf("Enter text: ");
    if (fgets(text, (int)sizeof(text), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    text[sizeof(text) - 1] = '\0';

    len = bounded_strlen(text, sizeof(text));
    if (len > 0 && text[len - 1] == '\n') {
        text[len - 1] = '\0';
    }

    if (find_most_common(text, sizeof(text), results, UCHAR_MAX + 1, &num_results) != 0) {
        fprintf(stderr, "Error processing text\n");
        return EXIT_FAILURE;
    }

    if (num_results == 0) {
        printf("No elements found\n");
        return EXIT_SUCCESS;
    }

    printf("Most common element(s):\n");
    for (i = 0; i < num_results; i++) {
        printf("'%c' : %zu\n", results[i].element, results[i].count);
    }

    return EXIT_SUCCESS;
}

The original code already mitigated the flagged CWE-119/CWE-120 weakness effectively, so the rewrite preserves those defenses. Key points of the analysis:

1. **Fixed-size buffer (`char text[MAX_TEXT_LEN]`)** — the Flawfinder hit is a low-risk informational warning. The buffer is never written without bounds control: `fgets` receives `sizeof(text)` as its limit, guaranteeing no overflow, and a forced NUL terminator at `text[sizeof(text) - 1]` provides defense-in-depth even though `fgets` already terminates.

2. **`bounded_strlen`** replaces bare `strlen`, capping traversal at `max_len` so an unterminated buffer can never cause an out-of-bounds read. The `len == text_size` check in `find_most_common` explicitly rejects unterminated input.

3. **Signed-char pitfall avoided** — characters are cast to `unsigned char` before indexing `counts[]`, preventing negative array indices on platforms where `char` is signed.

4. **Counter overflow guard** — `counts[c] < SIZE_MAX` prevents wraparound, though it is unreachable given the 1024-byte input cap; it's harmless hardening.

5. **Defensive API contract** — `find_most_common` validates all pointers and sizes, writes at most `max_results` entries into `results`, and reports the actual count through `num_results`, so the caller's `results[UCHAR_MAX + 1]` array cannot be overrun.

6. **Initialization** — `memset` on both buffers ensures deterministic state before use.

No functional changes were required; the code stands as a correct, bounds-safe implementation.
