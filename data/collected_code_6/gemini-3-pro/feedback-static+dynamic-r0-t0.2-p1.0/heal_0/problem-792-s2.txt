#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

size_t count_valid_lists(Node** lists, size_t total_elements) {
    size_t count = 0;
    size_t i;

    if (lists == NULL || total_elements == 0) {
        return 0;
    }

    for (i = 0; i < total_elements; ++i) {
        if (lists[i] != NULL) {
            count++;
        }
    }

    return count;
}

Node* create_node(int data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (new_node == NULL) {
        exit(EXIT_FAILURE);
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

void free_list(Node* head) {
    Node* current = head;
    Node* next_node = NULL;

    while (current != NULL) {
        next_node = current->next;
        free(current);
        current = next_node;
    }
}

int main(void) {
    size_t total_elements = 3;
    size_t valid_count = 0;
    size_t i;
    
    Node** lists = (Node**)malloc(total_elements * sizeof(Node*));
    if (lists == NULL) {
        return EXIT_FAILURE;
    }

    lists[0] = create_node(1);
    lists[0]->next = create_node(2);

    lists[1] = NULL;

    lists[2] = create_node(3);

    valid_count = count_valid_lists(lists, total_elements);
    printf("%zu\n", valid_count);

    for (i = 0; i < total_elements; ++i) {
        if (lists[i] != NULL) {
            free_list(lists[i]);
        }
    }
    free(lists);

    return EXIT_SUCCESS;
}
