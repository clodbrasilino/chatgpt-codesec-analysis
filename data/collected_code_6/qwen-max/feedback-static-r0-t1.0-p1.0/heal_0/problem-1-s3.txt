#include <stdio.h>
#include <limits.h>

#define ROW 3
#define COL 3

int min(int x, int y, int z) {
    return (x < y) ? ((x < z) ? x : z) : ((y < z) ? y : z);
}

int findMinCostPath(int cost[ROW][COL], int m, int n) {
    int i, j;
    int tc[ROW][COL];
  
    tc[0][0] = cost[0][0];

    for (i = 1; i <= m; i++)
        tc[i][0] = tc[i-1][0] + cost[i][0];

    for (j = 1; j <= n; j++)
        tc[0][j] = tc[0][j-1] + cost[0][j];

    for (i = 1; i <= m; i++)
        for (j = 1; j <= n; j++)
            tc[i][j] = cost[i][j] + min(tc[i-1][j-1], tc[i-1][j], tc[i][j-1]);

    return tc[m][n];
}

int main() {
    int cost[ROW][COL] = { 
        {1, 3, 5},
        {2, 1, 4},
        {7, 6, 8}
    };
    
    int m = 2;
    int n = 2;

    printf("The minimum cost to reach from (0, 0) to (%d, %d) is %d\n", m, n, findMinCostPath(cost, m, n));

    return 0;
}
