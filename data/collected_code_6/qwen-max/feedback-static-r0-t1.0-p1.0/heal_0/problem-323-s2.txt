#include <stdio.h>
#include <stdlib.h>

void rearrange(int *arr, int n) {
    int temp[n], index = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] > 0 && index < n) {
            temp[index++] = arr[i];
        }
    }
    for (int i = 0; i < n; i++) {
        if (arr[i] < 0 && index < n) {
            temp[index++] = arr[i];
        }
    }
    for (int i = 0; i < n; i++) {
        arr[i] = temp[(i / 2) * 2 + i % 2];
    }
}

int main() {
    int arr[] = {1, -1, 3, -3, 5, -5, 7, -7};
    int n = sizeof(arr)/sizeof(arr[0]);
    rearrange(arr, n);
    for(int i=0; i<n; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}
