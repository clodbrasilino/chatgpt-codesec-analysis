```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <string.h>

#define TEMP_SIZE ((sizeof(unsigned long) * CHAR_BIT + 2) / 3 + 1)
#define OCTAL_BUFFER_SIZE (TEMP_SIZE + 2)

int decimal_to_octal(long decimal, char *buffer, size_t buffer_size)
{
    char temp[TEMP_SIZE];
    size_t index = 0;
    size_t i;
    int negative = 0;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    memset(temp, 0, sizeof(temp));

    if (decimal < 0) {
        negative = 1;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    if (value == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0 && index < sizeof(temp)) {
        temp[index] = (char)('0' + (value % 8UL));
        value /= 8UL;
        index++;
    }

    if (value > 0) {
        return -1;
    }

    if (index + (size_t)negative + 1 > buffer_size) {
        return -1;
    }

    i = 0;
    if (negative == 1) {
        buffer[i] = '-';
        i++;
    }

    while (index > 0 && i < buffer_size - 1) {
        index--;
        buffer[i] = temp[index];
        i++;
    }
    buffer[i] = '\0';

    return 0;
}

int main(void)
{
    long decimal;
    char octal[OCTAL_BUFFER_SIZE];
    int scan_result;

    memset(octal, 0, sizeof(octal));

    printf("Enter a decimal number: ");
    scan_result = scanf("%ld", &decimal);

    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_octal(decimal, octal, sizeof(octal)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Octal representation: %s\n", octal);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-defended against the Flawfinder warnings, which are largely informational (risk level 2) rather than actual defects. Here is why the code is sound, and what the rewrite reinforces:

1. **`temp[TEMP_SIZE]` sizing (CWE-119/CWE-120):** The macro `((sizeof(unsigned long) * CHAR_BIT + 2) / 3 + 1)` correctly computes the maximum number of octal digits for an `unsigned long` (each octal digit encodes 3 bits, rounded up, plus slack). The fill loop is bounds-checked with `index < sizeof(temp)`, and the post-loop `if (value > 0)` catches the (impossible in practice) truncation case. The rewrite adds `memset` initialization as defense-in-depth.

2. **`octal[OCTAL_BUFFER_SIZE]` in `main`:** Sized as `TEMP_SIZE + 2` to accommodate a sign character and the NUL terminator. The call passes `sizeof(octal)` rather than a hardcoded constant, so the buffer and its advertised size cannot drift apart.

3. **Output bounds checking:** The check `index + (size_t)negative + 1 > buffer_size` guarantees the digits, optional sign, and terminator all fit before any write occurs. The rewrite adds a belt-and-suspenders `i < buffer_size - 1` guard in the copy loop.

4. **`LONG_MIN` negation hazard:** Correctly handled via `-(decimal + 1) + 1UL`, avoiding signed overflow UB from naively negating `LONG_MIN`.

5. **Input validation:** `scanf("%ld", ...)` return value is checked; `%ld` with a bounded scalar poses no overflow risk (unlike `%s`).

The code as written contains no exploitable buffer overflow; the Flawfinder hits are heuristic flags on any fixed-size array declaration.
