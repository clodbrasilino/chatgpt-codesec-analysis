#include <stdio.h>
#include <stdlib.h>

void rearrange(int *arr, int n) {
    int i, j;
    for (i = 0, j = n-1; i < j; ) {
        while (i < n && arr[i] < 0) i++;
        while (j >= 0 && arr[j] > 0) j--;
        if (i < j) {
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            i++; j--;
        }
    }
}

int main() {
    int arr[] = {1, -2, 3, -4, 5, -6, 7, -8, 9};
    int n = sizeof(arr) / sizeof(arr[0]);
    rearrange(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    return 0;
}
