#include <stdio.h>
#include <limits.h>
#include <stddef.h>
#include <string.h>

#define INPUT_BUFFER_SIZE 256

int find_max_occurring_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    unsigned char max_char = 0;
    int found = 0;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
            max_char = c;
            found = 1;
        }
    }

    if (found == 0) {
        return -1;
    }

    *result = (char)max_char;
    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    char max_char;
    char *newline;

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    newline = memchr(input, '\n', sizeof(input));
    if (newline != NULL) {
        *newline = '\0';
    }

    if (find_max_occurring_char(input, &max_char) != 0) {
        fprintf(stderr, "Empty string or invalid input\n");
        return 1;
    }

    if (printf("Maximum occurring character: %c\n", max_char) < 0) {
        return 1;
    }

    return 0;
}

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning about the statically-sized array `char input[INPUT_BUFFER_SIZE]` (CWE-119/CWE-120) is a **false positive** in this context, because the code already implements all the necessary mitigations:

1. **Bounded input reading** — `fgets(input, (int)sizeof(input), stdin)` limits input to at most `sizeof(input) - 1` characters plus a null terminator, so no buffer overflow is possible. This is the correct alternative to unsafe functions like `gets()` or unbounded `scanf("%s", ...)`.

2. **Defensive null termination** — `input[sizeof(input) - 1] = '\0';` guarantees termination even in edge cases, making subsequent string iteration safe.

3. **Bounded newline search** — `memchr(input, '\n', sizeof(input))` never reads past the buffer, unlike `strchr` on a potentially unterminated string.

4. **Safe character counting** — casting to `unsigned char` before indexing `counts[]` prevents negative-index undefined behavior on platforms where `char` is signed, and the array size `UCHAR_MAX + 1` covers all possible byte values.

5. **Robust error handling** — `NULL` checks on pointers, `fgets` failure detection, empty-string detection, and even checking the return value of `printf`.

The code as written is already compliant with secure coding practices (CERT C: STR31-C, STR32-C, FIO37-C), so the rewrite preserves the logic verbatim with the analysis comments removed.
