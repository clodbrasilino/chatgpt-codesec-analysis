#include <stdio.h>
#include <stdlib.h>

double sum_of_inverse_divisors(int n) {
    if (n <= 0) {
        return 0.0;
    }

    double sum = 0.0;
    
    for (int i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            sum += 1.0 / i;
            if (i * i != n) {
                sum += 1.0 / (n / i);
            }
        }
    }
    
    return sum;
}

int main(void) {
    int num;
    
    if (scanf("%d", &num) != 1) {
        return EXIT_FAILURE;
    }
    
    if (num <= 0) {
        return EXIT_FAILURE;
    }
    
    double result = sum_of_inverse_divisors(num);
    printf("%f\n", result);
    
    return EXIT_SUCCESS;
}
