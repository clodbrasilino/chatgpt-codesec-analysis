```c
#include <stdio.h>
#include <stddef.h>

size_t count_positives(const int *arr, size_t len)
{
    size_t count = 0;
    size_t i;

    if (arr == NULL) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (arr[i] > 0) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    int numbers[] = { 3, -1, 0, 7, -5, 12, 9, -8 };
    size_t len = sizeof(numbers) / sizeof(numbers[0]);
    size_t result;

    result = count_positives(numbers, len);

    if (printf("Positive count: %zu\n", result) < 0) {
        return 1;
    }

    return 0;
}
```
