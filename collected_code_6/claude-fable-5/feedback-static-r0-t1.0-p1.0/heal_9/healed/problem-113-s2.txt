```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>
#include <stdint.h>

#define MAX_LINE_LENGTH 1048576

int is_integer(const char *str)
{
    char *endptr;
    long value;

    if (str == NULL) {
        return 0;
    }

    while (isspace((unsigned char)*str)) {
        str++;
    }

    if (*str == '\0') {
        return 0;
    }

    errno = 0;
    value = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return 0;
    }

    if (endptr == str) {
        return 0;
    }

    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }

    if (*endptr != '\0') {
        return 0;
    }

    (void)value;
    return 1;
}

char *read_line(FILE *stream)
{
    size_t capacity = 64;
    size_t length = 0;
    char *line;
    int ch;

    if (stream == NULL) {
        return NULL;
    }

    line = malloc(capacity);
    if (line == NULL) {
        return NULL;
    }

    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length >= MAX_LINE_LENGTH) {
            free(line);
            return NULL;
        }
        if (length + 1 >= capacity) {
            size_t new_capacity;
            char *tmp;

            if (capacity > SIZE_MAX / 2) {
                free(line);
                return NULL;
            }
            new_capacity = capacity * 2;
            tmp = realloc(line, new_capacity);
            if (tmp == NULL) {
                free(line);
                return NULL;
            }
            line = tmp;
            capacity = new_capacity;
        }
        line[length++] = (char)ch;
    }

    if (ch == EOF && length == 0) {
        free(line);
        return NULL;
    }

    line[length] = '\0';
    return line;
}

int main(void)
{
    char *buffer;

    printf("Enter a string: ");
    fflush(stdout);

    buffer = read_line(stdin);
    if (buffer == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (is_integer(buffer)) {
        printf("\"%s\" is a valid integer\n", buffer);
    } else {
        printf("\"%s\" is not a valid integer\n", buffer);
    }

    free(buffer);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weakness (Flawfinder: `fgetc` in a loop, CWE-120/CWE-20):**

This is a risk-level-1 informational warning, and the original code already mitigates it correctly. The rewritten code preserves those mitigations:

1. **Bounded growth (CWE-120 — buffer overflow):** Every write to `line` is guarded. The check `length + 1 >= capacity` guarantees space for both the incoming character and the eventual null terminator before `line[length++]` executes, so no out-of-bounds write is possible.

2. **Hard upper limit (CWE-20 — input validation / resource exhaustion):** `MAX_LINE_LENGTH` caps total input at 1 MiB, preventing an attacker from exhausting memory with an unbounded input stream.

3. **Overflow-safe reallocation:** The `capacity > SIZE_MAX / 2` guard prevents integer overflow in `capacity * 2`, which could otherwise wrap and cause an undersized allocation.

4. **Correct `realloc` idiom:** The result is assigned to a temporary (`tmp`), so the original pointer is not lost on failure — no memory leak, no use of a null pointer.

5. **EOF handling:** `ch` is an `int`, not a `char`, so `EOF` is distinguishable from valid byte value 0xFF. Empty EOF reads return `NULL` rather than an empty allocation.

6. **`is_integer` robustness:** Uses `strtol` with `errno`/`ERANGE` checks, `endptr` validation, and casts to `unsigned char` for `isspace`, avoiding undefined behavior on negative `char` values.

No functional changes were required — the code as written already resolves the flagged concern; the warning is a false positive given the existing boundary checks.
