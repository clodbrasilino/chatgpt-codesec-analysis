#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char *key;
    char *value;
} Dictionary;

int compare_dicts(const void *a, const void *b) {
    const Dictionary *dictA = (const Dictionary *)a;
    const Dictionary *dictB = (const Dictionary *)b;
    if (!dictA->key || !dictB->key) {
        return 0;
    }
    return strcmp(dictA->key, dictB->key);
}

void sort_dictionaries(Dictionary *list, size_t count) {
    if (list == NULL || count == 0) {
        return;
    }
    qsort(list, count, sizeof(Dictionary), compare_dicts);
}

char *safe_create_string(const char *source) {
    if (source == NULL) {
        return NULL;
    }
    
    size_t len = strlen(source);
    char *dest = (char *)malloc(len + 1);
    
    if (dest != NULL) {
        memcpy(dest, source, len + 1);
    }
    
    return dest;
}

int main(void) {
    size_t num_dicts = 3;
    Dictionary *list = (Dictionary *)calloc(num_dicts, sizeof(Dictionary));
    if (list == NULL) {
        return EXIT_FAILURE;
    }

    list[0].key = safe_create_string("cherry");
    list[0].value = safe_create_string("red");

    list[1].key = safe_create_string("apple");
    list[1].value = safe_create_string("green");

    list[2].key = safe_create_string("banana");
    list[2].value = safe_create_string("yellow");

    for (size_t i = 0; i < num_dicts; i++) {
        if (list[i].key == NULL || list[i].value == NULL) {
            for (size_t j = 0; j <= i; j++) {
                free(list[j].key);
                free(list[j].value);
            }
            free(list);
            return EXIT_FAILURE;
        }
    }

    sort_dictionaries(list, num_dicts);

    for (size_t i = 0; i < num_dicts; i++) {
        printf("%s: %s\n", list[i].key, list[i].value);
    }

    for (size_t i = 0; i < num_dicts; i++) {
        free(list[i].key);
        free(list[i].value);
    }
    
    free(list);

    return EXIT_SUCCESS;
}
