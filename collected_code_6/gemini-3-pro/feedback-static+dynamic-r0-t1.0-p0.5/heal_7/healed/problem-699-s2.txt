#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int min_swaps(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }

    size_t i = 0;
    int count_01 = 0;
    int count_10 = 0;

    while (str1[i] != '\0' && str2[i] != '\0') {
        if ((str1[i] != '0' && str1[i] != '1') || (str2[i] != '0' && str2[i] != '1')) {
            return -1;
        }

        if (str1[i] == '0' && str2[i] == '1') {
            count_01++;
        } else if (str1[i] == '1' && str2[i] == '0') {
            count_10++;
        }
        i++;
    }

    if (str1[i] != '\0' || str2[i] != '\0') {
        return -1;
    }

    if (count_01 != count_10) {
        return -1;
    }

    return count_01;
}

int main(void) {
    char *str1 = calloc(2048, sizeof(char));
    char *str2 = calloc(2048, sizeof(char));

    if (str1 == NULL || str2 == NULL) {
        free(str1);
        free(str2);
        return 1;
    }

    if (fgets(str1, 2048, stdin) == NULL) {
        free(str1);
        free(str2);
        return 0;
    }
    str1[strcspn(str1, "\r\n")] = '\0';

    if (fgets(str2, 2048, stdin) == NULL) {
        free(str1);
        free(str2);
        return 0;
    }
    str2[strcspn(str2, "\r\n")] = '\0';

    int result = min_swaps(str1, str2);

    if (result != -1) {
        printf("%d\n", result);
    } else {
        printf("Not Possible\n");
    }

    free(str1);
    free(str2);

    return 0;
}
