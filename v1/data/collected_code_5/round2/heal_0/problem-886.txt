```c
#include <stdio.h>
#include <stdlib.h>

float average(int *list, size_t length) {
    if (list == NULL || length == 0) {
        return 0;
    }

    int sum = 0;

    for (size_t i = 0; i < length; i++) {
        sum += list[i];
    }

    return (float)sum / length;
}

int main() {
    int list[] = {1, 2, 3, 4, 5};
    size_t length = sizeof(list) / sizeof(int);

    float avg = average(list, length);

    printf("Average is: %f\n", avg);

    return 0;
}
```
