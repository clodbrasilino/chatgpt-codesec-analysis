```c
#include <stdio.h>
#include <stdlib.h>

int findMaxOccurrences(const int *list, size_t size, int *result) {
    if (list == NULL || size == 0 || result == NULL) {
        return -1;
    }

    int *hashMap = (int *)calloc(100000, sizeof(int));
    if (hashMap == NULL) {
        return -1;
    }

    int maxCount = 0;
    int maxItem = list[0];

    for (size_t i = 0; i < size; ++i) {
        int currentItem = list[i];
        
        if (currentItem >= 0 && currentItem < 100000) {
            hashMap[currentItem]++;
            if (hashMap[currentItem] > maxCount) {
                maxCount = hashMap[currentItem];
                maxItem = currentItem;
            }
        }
    }

    free(hashMap);
    *result = maxItem;
    return 0;
}

int main(void) {
    int list[] = {1, 3, 2, 3, 4, 3, 5, 1, 2, 1, 1};
    size_t size = sizeof(list) / sizeof(list[0]);
    int result = 0;

    int status = findMaxOccurrences(list, size, &result);

    if (status == 0) {
        printf("Item with maximum occurrences: %d\n", result);
    } else {
        printf("Error finding maximum occurrences.\n");
    }

    return 0;
}
```
