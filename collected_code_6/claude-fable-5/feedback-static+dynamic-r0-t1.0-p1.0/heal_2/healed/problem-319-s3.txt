```c
#include <stdio.h>
#include <stdlib.h>
#include <regex.h>

#define NUM_GROUPS 3U

static void print_regex_error(int rc, const regex_t *regex)
{
    size_t needed;
    char *errbuf;

    needed = regerror(rc, regex, NULL, 0U);
    if (needed == 0U) {
        (void)fprintf(stderr, "Regex error: unknown error\n");
        return;
    }

    errbuf = (char *)malloc(needed);
    if (errbuf == NULL) {
        (void)fprintf(stderr, "Regex error: allocation failure\n");
        return;
    }

    (void)regerror(rc, regex, errbuf, needed);
    errbuf[needed - 1U] = '\0';
    (void)fprintf(stderr, "Regex error: %s\n", errbuf);
    free(errbuf);
}

static int find_five_letter_words(const char *text)
{
    regex_t regex;
    regmatch_t groups[NUM_GROUPS];
    const char *pattern = "(^|[^[:alnum:]_])([[:alpha:]]{5})([^[:alnum:]_]|$)";
    const char *cursor = text;
    int rc;
    int found = 0;

    if (text == NULL) {
        return -1;
    }

    rc = regcomp(&regex, pattern, REG_EXTENDED);
    if (rc != 0) {
        print_regex_error(rc, &regex);
        return -1;
    }

    for (;;) {
        rc = regexec(&regex, cursor, (size_t)NUM_GROUPS, groups, 0);
        if (rc == REG_NOMATCH) {
            break;
        }
        if (rc != 0) {
            print_regex_error(rc, &regex);
            regfree(&regex);
            return -1;
        }
        if ((groups[2].rm_so < 0) || (groups[2].rm_eo < groups[2].rm_so)) {
            break;
        }
        if (printf("%.*s\n", (int)(groups[2].rm_eo - groups[2].rm_so),
                   cursor + groups[2].rm_so) < 0) {
            regfree(&regex);
            return -1;
        }
        found++;
        cursor += groups[2].rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (found == 0) {
        (void)printf("No five character words found.\n");
    }

    return 0;
}

int main(void)
{
    const char *text = "There might exist words which occur about seven times daily";
    int result;

    result = find_five_letter_words(text);
    if (result != 0) {
        (void)fprintf(stderr, "Failed to process the input string.\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

This eliminates any possibility of truncation or improper size restriction, guarantees the buffer always fits the message, and the explicit null-termination plus `free()` ensure safe, leak-free handling. All other bounds-checked logic (validated `rm_so`/`rm_eo` offsets, length-limited `%.*s` printing, and cursor advancement guarded against overrun) is preserved from the original.
