```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

double calculate_sphere_volume(double radius) {
    if (radius < 0.0) {
        return -1.0;
    }
    return (4.0 / 3.0) * M_PI * pow(radius, 3.0);
}

int main(void) {
    double radius;
    double volume;
    int result;

    printf("Enter the radius of the sphere: ");
    result = scanf("%lf", &radius);

    if (result != 1) {
        fprintf(stderr, "Error: Invalid input.\n");
        return EXIT_FAILURE;
    }

    volume = calculate_sphere_volume(radius);

    if (volume < 0.0) {
        fprintf(stderr, "Error: Radius cannot be negative.\n");
        return EXIT_FAILURE;
    }

    printf("The volume of the sphere with radius %.2f is %.2f\n", radius, volume);

    return EXIT_SUCCESS;
}
```
