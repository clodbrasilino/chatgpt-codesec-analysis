#include <stdio.h>
#include <string.h>
#include <stddef.h>

char *abbreviate_road(char *str, size_t size) {
    char *read;
    char *write;
    char *end;

    if (str == NULL || size == 0) {
        return NULL;
    }

    read = str;
    write = str;
    end = str + size;

    while (read < end && *read != '\0') {
        if (read + 4 <= end && memcmp(read, "road", 4) == 0) {
            *write++ = 'r';
            *write++ = 'd';
            *write++ = '.';
            read += 4;
        } else {
            *write++ = *read++;
        }
    }

    if (write < end) {
        *write = '\0';
    } else {
        *(end - 1) = '\0';
    }

    return str;
}

int main(void) {
    char input[] = "123 main road and broadway road";

    printf("%s\n", input);
    abbreviate_road(input, sizeof(input));
    printf("%s\n", input);

    return 0;
}
