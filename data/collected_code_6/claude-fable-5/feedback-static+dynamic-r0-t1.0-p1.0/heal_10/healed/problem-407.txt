#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness and why the code is safe as written:

**The Flawfinder warning (CWE-119/CWE-120)** flags the statically-sized `char digits[DIGITS_BUF_SIZE]` array as a potential buffer overflow risk. However, this is a false positive here because the code already implements all the required mitigations:

1. **Bounded write**: `snprintf(digits, sizeof(digits), ...)` limits the write length to the buffer size, never overflowing regardless of input.

2. **Return-value validation**: The check `len < 0 || (size_t)len >= sizeof(digits)` catches both encoding errors and truncation. A truncated result would silently corrupt the algorithm, so rejecting it is correct.

3. **Adequate sizing**: `LLONG_MAX` (9223372036854775807) is 19 digits; with sign and NUL terminator the maximum needed is 21 bytes, so 32 bytes provides comfortable headroom.

4. **Zero initialization**: `memset` guarantees the buffer is NUL-terminated even in edge cases.

5. **Safe index arithmetic**: `reverse_range` is only called after confirming `i >= 0` and `j > i`, so the `size_t` casts never wrap around from negative values.

6. **Overflow-safe reconstruction**: The `result > (LLONG_MAX - d) / 10` guard prevents signed integer overflow (undefined behavior) when converting digits back to a `long long`.

Since every write into the fixed-size buffer is length-limited and validated, the code satisfies the mitigation guidance in the warning, and no structural changes are required.
