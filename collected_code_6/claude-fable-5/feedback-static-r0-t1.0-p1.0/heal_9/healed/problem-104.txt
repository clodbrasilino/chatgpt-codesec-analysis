```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 256U
#define NUM_LISTS 3U
#define LIST_CAPACITY 4U

static int compare_strings(const void *a, const void *b)
{
    const char *const *sa = (const char *const *)a;
    const char *const *sb = (const char *const *)b;
    return strcmp(*sa, *sb);
}

static int sort_sublists(char ***lists, const size_t *sizes, size_t num_lists)
{
    size_t i;

    if (lists == NULL || sizes == NULL) {
        return -1;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] == NULL) {
            return -1;
        }
        if (sizes[i] > 1) {
            qsort(lists[i], sizes[i], sizeof(char *), compare_strings);
        }
    }

    return 0;
}

static char *duplicate_string(const char *src)
{
    size_t len;
    size_t i;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    for (i = 0; i < len; i++) {
        copy[i] = src[i];
    }
    copy[len] = '\0';
    return copy;
}

static void free_lists(char ***lists, const size_t *sizes, size_t num_lists)
{
    size_t i;
    size_t j;

    if (lists == NULL || sizes == NULL) {
        return;
    }

    for (i = 0; i < num_lists; i++) {
        if (lists[i] != NULL) {
            for (j = 0; j < sizes[i]; j++) {
                free(lists[i][j]);
            }
            free(lists[i]);
        }
    }
    free(lists);
}

int main(void)
{
    static const char *const data[NUM_LISTS][LIST_CAPACITY] = {
        { "banana", "apple", "cherry", "date" },
        { "zebra", "mango", "kiwi", "lemon" },
        { "orange", "grape", "fig", "plum" }
    };
    const size_t num_lists = sizeof(data) / sizeof(data[0]);
    static const size_t sizes[NUM_LISTS] = { 4, 4, 4 };
    char ***lists;
    size_t i;
    size_t j;

    lists = calloc(num_lists, sizeof(char **));
    if (lists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_lists; i++) {
        if (sizes[i] > sizeof(data[i]) / sizeof(data[i][0])) {
            fprintf(stderr, "Size exceeds available data\n");
            free_lists(lists, sizes, num_lists);
            return EXIT_FAILURE;
        }
        lists[i] = calloc(sizes[i], sizeof(char *));
        if (lists[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_lists(lists, sizes, num_lists);
            return EXIT_FAILURE;
        }
        for (j = 0; j < sizes[i]; j++) {
            lists[i][j] = duplicate_string(data[i][j]);
            if (lists[i][j] == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                free_lists(lists, sizes, num_lists);
                return EXIT_FAILURE;
            }
        }
    }

    if (sort_sublists(lists, sizes, num_lists) != 0) {
        fprintf(stderr, "Sorting failed\n");
        free_lists(lists, sizes, num_lists);
        return EXIT_FAILURE;
    }

    for (i = 0; i < num_lists; i++) {
        printf("List %zu:", i);
        for (j = 0; j < sizes[i]; j++) {
            printf(" %s", lists[i][j]);
        }
        printf("\n");
    }

    free_lists(lists, sizes, num_lists);
    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The flagged weakness (CWE-119/CWE-120, statically-sized arrays) was already mitigated in the original code, and the rewrite preserves and strengthens those mitigations:

1. **Bounds checking on the static array** — before copying, `sizes[i]` is validated against `sizeof(data[i]) / sizeof(data[i][0])`, so the code can never read past the bounds of the fixed-size `data` array. Named constants (`NUM_LISTS`, `LIST_CAPACITY`) now tie the array dimensions and the `sizes` array together, eliminating magic numbers.

2. **Length-limited string handling** — `duplicate_string` uses `strnlen` capped at `MAX_STRING_LEN` and rejects strings at or beyond the limit, preventing unbounded reads or overflows; the destination buffer is allocated exactly `len + 1` bytes and explicitly NUL-terminated.

3. **Consistent NULL and allocation-failure checks** — every `calloc`/`malloc` result is checked, and `sort_sublists`/`free_lists` defensively validate their pointer arguments.

4. **No leaks on error paths** — `calloc` zero-initializes the pointer arrays, so `free_lists` is safe to call at any point during partial construction, freeing only what was allocated.

The code is memory-safe, leak-free, and compiles cleanly under `-Wall -Wextra -pedantic` with C99 or later (`strnlen` requires POSIX or C23).
