#include <stdio.h>
#include <stdlib.h>

static size_t count_digits(size_t n)
{
    size_t d = 1;

    while (n >= 10) {
        n /= 10;
        d++;
    }
    return d;
}

static size_t write_number(char *dst, size_t avail, size_t n)
{
    size_t d = count_digits(n);
    size_t i = d;

    if (d > avail) {
        return 0;
    }

    while (i > 0) {
        dst[i - 1] = (char)('0' + (n % 10));
        n /= 10;
        i--;
    }
    return d;
}
