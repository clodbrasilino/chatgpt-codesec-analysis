```c
#define _POSIX_C_SOURCE 200809L

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 256

static int is_vowel(char c)
{
    char lower = (char)tolower((unsigned char)c);
    return (lower == 'a' || lower == 'e' || lower == 'i' ||
            lower == 'o' || lower == 'u');
}

int count_chars_with_vowel_neighbors(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    int count = 0;

    if (str == NULL) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        if (i > 0 && is_vowel(str[i - 1])) {
            count++;
            continue;
        }
        if (i + 1 < len && is_vowel(str[i + 1])) {
            count++;
        }
    }

    return count;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    int result;

    printf("Enter a string: ");
    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output.\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';
    buffer[strcspn(buffer, "\n")] = '\0';

    result = count_chars_with_vowel_neighbors(buffer, sizeof(buffer));
    printf("Number of characters with vowel neighbors: %d\n", result);

    return 0;
}
```

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning about the statically-sized `char buffer[MAX_INPUT_LEN]` (CWE-119/CWE-120) was already effectively mitigated in the original source, and those mitigations are preserved in the rewrite:

1. **Bounded input reading** — `fgets(buffer, (int)sizeof(buffer), stdin)` limits the number of bytes read to the buffer size, preventing a classic `gets()`-style overflow. Using `sizeof(buffer)` rather than a hardcoded constant keeps the bound correct if `MAX_INPUT_LEN` changes.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0';` is defensive (technically redundant, since `fgets` always null-terminates on success), but harmless and protects against future refactoring to unsafe read functions.

3. **Safe newline stripping** — `buffer[strcspn(buffer, "\n")] = '\0';` correctly handles the case where no newline exists (input truncated), unlike the common buggy idiom `buffer[strlen(buffer) - 1] = '\0'`, which underflows on an empty string.

4. **Bounded string traversal** — `count_chars_with_vowel_neighbors` uses `strnlen` with an explicit `max_len` cap rather than `strlen`, guarding against reads past the buffer even if a non-terminated string were passed. It also null-checks `str` before use.

5. **Safe `ctype` usage** — `tolower((unsigned char)c)` correctly casts to `unsigned char` first, avoiding undefined behavior when `char` is signed and holds negative values (e.g., extended characters).

The only minor change made in the rewrite is checking the return value of `fflush(stdout)`, which is a trivial robustness improvement. The code is otherwise free of buffer-overflow risks: all reads and writes are bounds-checked, and the loop indices (`i - 1`, `i + 1`) are guarded against underflow/overflow of the valid range.
