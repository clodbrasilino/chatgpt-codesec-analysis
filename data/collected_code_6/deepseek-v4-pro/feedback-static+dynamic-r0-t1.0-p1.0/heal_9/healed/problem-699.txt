#include <stdio.h>
#include <string.h>

int min_swaps(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }

    size_t len1 = strnlen(str1, 101);
    size_t len2 = strnlen(str2, 101);
    
    if (len1 != len2 || len1 > 100) {
        return -1;
    }
    
    size_t len = len1;
    if (len == 0) {
        return 0;
    }
    
    int count1 = 0, count2 = 0;
    for (size_t i = 0; i < len; i++) {
        if (str1[i] == '1') count1++;
        if (str2[i] == '1') count2++;
    }
    
    if (count1 != count2) {
        return -1;
    }
    
    int mismatch = 0;
    for (size_t i = 0; i < len; i++) {
        if (str1[i] != str2[i]) {
            mismatch++;
        }
    }
    
    return mismatch / 2;
}
