```C
#include <stdio.h>

int sumSquares(int n) {
    int i, sum = 0;
    for(i = 1; i <= n; i++) {
        sum += ((2*i-1) * (2*i-1));
    }
    return sum;
}

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);
    if(n<0) {
        printf("Invalid input. Please enter a positive integer.\n");
        return 1;
    }
    printf("Sum of squares of first %d odd natural numbers is: %d\n", n, sumSquares(n));
    return 0;
}
```
