#include <stdio.h>
#include <stdlib.h>

int triangle_perimeter(int a, int b, int c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return -1;
    }
    if (a + b <= c || a + c <= b || b + c <= a) {
        return -1;
    }
    return a + b + c;
}

int main(void) {
    int a, b, c;
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    int perimeter = triangle_perimeter(a, b, c);
    if (perimeter == -1) {
        fprintf(stderr, "Invalid triangle sides\n");
        return EXIT_FAILURE;
    }
    printf("%d\n", perimeter);
    return EXIT_SUCCESS;
}
