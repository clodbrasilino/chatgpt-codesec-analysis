#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long find_product_of_non_repeated_elements(const int arr[], int n) {
    long long product = 1;
    int found = 0;
    int i, j;
    
    if (n <= 0) {
        return 0;
    }
    
    for (i = 0; i < n; i++) {
        int count = 0;
        for (j = 0; j < n; j++) {
            if (arr[i] == arr[j]) {
                count++;
            }
        }
        if (count == 1) {
            if (arr[i] != 0 && llabs(product) > LLONG_MAX / llabs((long long)arr[i])) {
                return 0;
            }
            product *= arr[i];
            found = 1;
        }
    }
    
    if (!found) {
        return 0;
    }
    
    return product;
}

int main(void) {
    int n, i;
    int *arr;
    long long result;
    
    printf("Enter the number of elements: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (n <= 0) {
        fprintf(stderr, "Invalid array size\n");
        return EXIT_FAILURE;
    }
    
    arr = (int *)malloc((size_t)n * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++) {
        if (scanf("%d", &arr[i]) != 1) {
            fprintf(stderr, "Invalid input\n");
            free(arr);
            return EXIT_FAILURE;
        }
    }
    
    result = find_product_of_non_repeated_elements(arr, n);
    printf("Product of non-repeated elements: %lld\n", result);
    
    free(arr);
    
    return EXIT_SUCCESS;
}
