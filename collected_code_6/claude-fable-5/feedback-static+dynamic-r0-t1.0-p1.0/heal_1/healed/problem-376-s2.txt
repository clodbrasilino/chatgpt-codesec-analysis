```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stddef.h>

int replace_duplicates(int *arr, size_t size, int custom_value)
{
    size_t i;
    size_t j;

    if (arr == NULL) {
        return -1;
    }

    if (size == 0U) {
        return 0;
    }

    for (i = 1U; i < size; i++) {
        bool found = false;
        for (j = 0U; j < i; j++) {
            if (arr[j] == arr[i]) {
                found = true;
                break;
            }
        }
        if (found) {
            arr[i] = custom_value;
        }
    }

    return 0;
}

int print_array(const int *arr, size_t size)
{
    size_t i;

    if (arr == NULL) {
        return -1;
    }

    if (printf("(") < 0) {
        return -1;
    }

    for (i = 0U; i < size; i++) {
        if (i > 0U) {
            if (printf(", ") < 0) {
                return -1;
            }
        }
        if (printf("%d", arr[i]) < 0) {
            return -1;
        }
    }

    if (printf(")\n") < 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    int tuple[] = { 1, 3, 5, 3, 7, 1, 9, 5, 11 };
    size_t size = sizeof(tuple) / sizeof(tuple[0]);
    int custom_value = -1;

    if (print_array(tuple, size) != 0) {
        return EXIT_FAILURE;
    }

    if (replace_duplicates(tuple, size, custom_value) != 0) {
        return EXIT_FAILURE;
    }

    if (print_array(tuple, size) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The only change required was addressing the `[variableScope]` finding: the `bool found` declaration was moved from function scope into the body of the outer `for` loop in `replace_duplicates`, since it is only used within a single iteration. This reduces its scope to the minimum necessary, improves readability, and ensures it is freshly initialized each iteration by its declaration rather than by a separate assignment. The rest of the code was already defensive (NULL checks, checked `printf` return values, unsigned literals for `size_t` comparisons) and needed no modification.
