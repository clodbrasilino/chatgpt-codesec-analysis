#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long maxSubArraySumRepeated(const int *arr, int n, int k) {
    if (arr == NULL || n <= 0 || k <= 0) {
        return 0;
    }

    long long max_ending_here = 0;
    long long max_so_far = LLONG_MIN;
    long long total_sum = 0;

    for (int i = 0; i < n; i++) {
        total_sum += arr[i];
        max_ending_here += arr[i];
        if (max_ending_here < 0) {
            max_ending_here = 0;
        }
        if (max_so_far < max_ending_here) {
            max_so_far = max_ending_here;
        }
    }

    if (max_so_far == 0) {
        max_so_far = LLONG_MIN;
        for (int i = 0; i < n; i++) {
            if (arr[i] > max_so_far) {
                max_so_far = arr[i];
            }
        }
    }

    if (k == 1) {
        return max_so_far;
    }

    long long max_prefix = LLONG_MIN;
    long long current = 0;
    for (int i = 0; i < n; i++) {
        current += arr[i];
        if (current > max_prefix) {
            max_prefix = current;
        }
    }

    long long max_suffix = LLONG_MIN;
    current = 0;
    for (int i = n - 1; i >= 0; i--) {
        current += arr[i];
        if (current > max_suffix) {
            max_suffix = current;
        }
    }

    long long result = max_so_far;
    long long boundary_sum = max_prefix + max_suffix;
    
    if (boundary_sum > result) {
        result = boundary_sum;
    }

    if (total_sum > 0 && k > 2) {
        long long extended = boundary_sum + (long long)(k - 2) * total_sum;
        if (extended > result) {
            result = extended;
        }
    }

    return result;
}

int main(void) {
    int arr[] = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 3;

    long long result = maxSubArraySumRepeated(arr, n, k);
    printf("%lld\n", result);

    return 0;
}
