```c
#include <stdio.h>
#include <string.h>

#define BUFFER_SIZE 1024

void remove_multiple_spaces(char *str)
{
    size_t read_idx = 0;
    size_t write_idx = 0;
    int prev_space = 0;

    if (str == NULL) {
        return;
    }

    while (str[read_idx] != '\0') {
        if (str[read_idx] == ' ') {
            if (!prev_space) {
                str[write_idx] = str[read_idx];
                write_idx++;
            }
            prev_space = 1;
        } else {
            str[write_idx] = str[read_idx];
            write_idx++;
            prev_space = 0;
        }
        read_idx++;
    }
    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    remove_multiple_spaces(buffer);

    if (printf("Result: %s\n", buffer) < 0) {
        fprintf(stderr, "Error writing output\n");
        return 1;
    }

    return 0;
}
```

The key changes addressing the flagged weaknesses:

1. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is now defined as a named constant `BUFFER_SIZE`, and `fgets` continues to use `sizeof(buffer)` to strictly limit input length, guaranteeing no overflow. An explicit forced null termination at `buffer[sizeof(buffer) - 1]` adds defense in depth.

2. **CWE-126 (strlen over-read)**: Replaced `strlen` with `strnlen`, which bounds the scan to `sizeof(buffer)`, so even a hypothetically non-terminated buffer cannot cause an over-read.

Note that the original code was already functionally safe — `fgets` always null-terminates on success — so these changes are hardening measures that satisfy static analyzers without altering behavior.
