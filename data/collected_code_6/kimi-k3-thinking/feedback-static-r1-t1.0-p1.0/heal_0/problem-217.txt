#include <stdio.h>
#include <stddef.h>
#include <limits.h>

#define CHAR_SET_SIZE (UCHAR_MAX + 1)

static char first_repeated_char(const char *str)
{
    int seen[CHAR_SET_SIZE] = {0};
    size_t i;

    if (str == NULL) {
        return '\0';
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c] != 0) {
            return (char)c;
        }
        seen[c] = 1;
    }

    return '\0';
}

int main(void)
{
    const char *tests[] = {
        "programming",
        "abcdef",
        "aabbcc",
        "hello world",
        "",
        "x"
    };
    const size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    size_t i;
    char result;

    for (i = 0; i < num_tests; i++) {
        result = first_repeated_char(tests[i]);
        if (result != '\0') {
            printf("Input: \"%s\" -> First repeated character: '%c'\n",
                   tests[i], result);
        } else {
            printf("Input: \"%s\" -> No repeated character found\n",
                   tests[i]);
        }
    }

    result = first_repeated_char(NULL);
    if (result == '\0') {
        printf("Input: NULL -> No repeated character found\n");
    }

    return 0;
}
