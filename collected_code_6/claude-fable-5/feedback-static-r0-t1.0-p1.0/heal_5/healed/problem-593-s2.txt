```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_IP_LEN 16

static int remove_leading_zeroes(const char *input, char *output, size_t output_size)
{
    unsigned int octets[4];
    size_t pos = 0;
    int octet_index;
    int written;

    if (input == NULL || output == NULL || output_size < MAX_IP_LEN) {
        return -1;
    }

    for (octet_index = 0; octet_index < 4; octet_index++) {
        unsigned long value = 0;
        int digit_count = 0;

        while (input[pos] != '\0' && isdigit((unsigned char)input[pos])) {
            value = value * 10 + (unsigned long)(input[pos] - '0');
            digit_count++;
            pos++;
            if (digit_count > 3) {
                return -1;
            }
        }

        if (digit_count == 0 || value > 255) {
            return -1;
        }

        octets[octet_index] = (unsigned int)value;

        if (octet_index < 3) {
            if (input[pos] != '.') {
                return -1;
            }
            pos++;
        }
    }

    if (input[pos] != '\0') {
        return -1;
    }

    written = snprintf(output, output_size, "%u.%u.%u.%u",
                       octets[0], octets[1], octets[2], octets[3]);

    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *tests[] = {
        "192.068.001.010",
        "001.002.003.004",
        "255.255.255.255",
        "000.000.000.000",
        "10.0.0.1",
        "256.1.1.1",
        "1.2.3",
        "abc.def.ghi.jkl"
    };
    char result[MAX_IP_LEN];
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    memset(result, 0, sizeof(result));

    for (i = 0; i < count; i++) {
        if (remove_leading_zeroes(tests[i], result, sizeof(result)) == 0) {
            if (printf("%s -> %s\n", tests[i], result) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Invalid IP address: %s\n", tests[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

The code is already properly hardened against this weakness, so the rewrite preserves the existing mitigations rather than restructuring them:

1. **Buffer size is provably sufficient**: The maximum canonical IPv4 string is `"255.255.255.255"` — 15 characters plus a NUL terminator, exactly `MAX_IP_LEN` (16). Since each octet is validated to be ≤ 255 before formatting, the output can never exceed the buffer.

2. **Defensive size parameter**: The buffer size is passed explicitly (`sizeof(result)`) rather than assumed, and the function rejects any caller-supplied buffer smaller than `MAX_IP_LEN` up front.

3. **Length-limited formatting**: `snprintf` is used instead of `sprintf`, so even if the size reasoning were wrong, no overflow could occur. The return value is checked for both encoding errors (`written < 0`) and truncation (`written >= output_size`).

4. **Input parsing is bounds-safe**: The parser never indexes past the NUL terminator, caps digits per octet at 3 (preventing `unsigned long` accumulation issues), validates each octet ≤ 255, and rejects trailing garbage.

5. **Initialization**: `memset` zeroes the buffer before use, so no uninitialized reads occur even on early failure paths.
