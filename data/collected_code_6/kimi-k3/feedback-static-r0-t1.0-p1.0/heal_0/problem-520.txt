#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static long long gcd_ll(long long a, long long b)
{
    if (a < 0) {
        if (a == LLONG_MIN) {
            return -1;
        }
        a = -a;
    }
    if (b < 0) {
        if (b == LLONG_MIN) {
            return -1;
        }
        b = -b;
    }

    while (b != 0) {
        long long t = a % b;
        a = b;
        b = t;
    }

    return a;
}

int lcm_array(const long long *arr, size_t n, long long *result)
{
    if (result == NULL) {
        return -1;
    }

    if (arr == NULL || n == 0U) {
        return -2;
    }

    long long lcm = 1;

    for (size_t i = 0; i < n; i++) {
        if (arr[i] == 0) {
            *result = 0;
            return 0;
        }

        long long g = gcd_ll(lcm, arr[i]);
        if (g <= 0) {
            return -3;
        }

        long long reduced = lcm / g;

        long long abs_val;
        if (arr[i] == LLONG_MIN) {
            return -4;
        }
        abs_val = (arr[i] < 0) ? -arr[i] : arr[i];

        if (reduced > LLONG_MAX / abs_val) {
            return -5;
        }

        lcm = reduced * abs_val;
    }

    *result = lcm;
    return 0;
}

int main(void)
{
    long long arr[] = {4, 6, 8, 10, 12};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    long long result = 0;

    int status = lcm_array(arr, n, &result);

    if (status == 0) {
        printf("LCM = %lld\n", result);
    } else {
        fprintf(stderr, "Error computing LCM (code %d)\n", status);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
