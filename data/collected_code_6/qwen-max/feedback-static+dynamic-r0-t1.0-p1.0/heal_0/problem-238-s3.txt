```c
#include <stdio.h>
#include <string.h>

long countNonEmptySubstrings(const char *str) {
    long n = strlen(str);
    return (n * (n + 1)) / 2;
}

int main() {
    const char *input = "test";
    long result = countNonEmptySubstrings(input);
    printf("Number of non-empty substrings: %ld\n", result);
    return 0;
}
```
