#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void find_words(const char *text) {
    if (text == NULL) {
        return;
    }

    regex_t regex;
    int reti;
    regmatch_t pmatch[1];
    const char *p = text;

    reti = regcomp(&regex, "\\b[a-zA-Z0-9_]{4,}\\b", REG_EXTENDED);
    if (reti != 0) {
        return;
    }

    while (1) {
        reti = regexec(&regex, p, 1, pmatch, 0);
        if (reti != 0) {
            break;
        }

        regoff_t start = pmatch[0].rm_so;
        regoff_t end = pmatch[0].rm_eo;
        regoff_t len = end - start;

        if (len > 0) {
            char *word = malloc(len + 1);
            if (word != NULL) {
                snprintf(word, len + 1, "%.*s", (int)len, p + start);
                printf("%s\n", word);
                free(word);
            }
        }

        p += end;
    }

    regfree(&regex);
}

int main(void) {
    const char *sample_text = "This is a sample string with some short and longer words.";
    find_words(sample_text);
    return 0;
}
