#include <stdio.h>
#include <limits.h>

int numWays(int n, int k) {
    if (n == 0) return 0;
    if (n == 1) return k;

    int same = 0, diff = k;

    for (int i = 2; i <= n; i++) {
        long sum = (long)same + (long)diff;
        if (sum > INT_MAX) return -1;
        
        same = diff;
        long new_diff = (long)(k - 1) * (long)sum;
        if (new_diff > INT_MAX) return -1;
        
        diff = (int)new_diff;
    }

    long result = (long)same + (long)diff;
    if (result > INT_MAX) return -1;
    
    return (int)result;
}

int main() {
    int n, k;
    scanf("%d%d", &n, &k);
    printf("%d\n", numWays(n, k));
    return 0;
}
