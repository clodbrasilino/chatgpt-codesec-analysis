```c
#include <stdio.h>
#include <stddef.h>

void sortArray(int arr[], size_t size) {
    if (arr == NULL || size == 0) {
        return;
    }
    
    size_t count0 = 0, count1 = 0, count2 = 0;
    
    for (size_t i = 0; i < size; i++) {
        if (arr[i] == 0) {
            count0++;
        } else if (arr[i] == 1) {
            count1++;
        } else if (arr[i] == 2) {
            count2++;
        }
    }
    
    size_t index = 0;
    
    while (count0 > 0) {
        arr[index++] = 0;
        count0--;
    }
    
    while (count1 > 0) {
        arr[index++] = 1;
        count1--;
    }
    
    while (count2 > 0) {
        arr[index++] = 2;
        count2--;
    }
}

int main(void) {
    int arr[] = {2, 0, 1, 2, 1, 0, 0, 2, 1};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    
    sortArray(arr, size);
    
    for (size_t i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
```
