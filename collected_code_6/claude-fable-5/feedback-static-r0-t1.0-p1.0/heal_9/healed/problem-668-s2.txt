```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 256U

void squeeze_char(char *str, size_t max_len, char ch);
static void discard_remaining_input(void);

static void discard_remaining_input(void)
{
    int input_char;

    do {
        input_char = getchar();
    } while ((input_char != '\n') && (input_char != EOF));
}

void squeeze_char(char *str, size_t max_len, char ch)
{
    size_t read_idx;
    size_t write_idx;
    size_t len;

    if ((str == NULL) || (max_len == 0U)) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }

    write_idx = 0U;

    for (read_idx = 0U; read_idx < len; read_idx++) {
        if ((str[read_idx] == ch) &&
            (read_idx > 0U) &&
            (str[read_idx - 1U] == ch)) {
            continue;
        }
        str[write_idx] = str[read_idx];
        write_idx++;
    }

    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char target;
    int input_char;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    } else if (len == (sizeof(buffer) - 1U)) {
        discard_remaining_input();
    }

    printf("Enter character to squeeze: ");
    input_char = getchar();
    if (input_char == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }

    target = (char)input_char;

    if (input_char != '\n') {
        discard_remaining_input();
    }

    squeeze_char(buffer, sizeof(buffer), target);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The Flawfinder warnings here are false positives in practice. `getchar()` returns an `int` stored in `input_char` (correctly typed as `int`, not `char`, so `EOF` detection works reliably). The loop in `discard_remaining_input()` never writes to a buffer — it only consumes characters — so no overflow is possible. The termination condition checks both `'\n'` and `EOF`, preventing infinite loops on closed streams.

2. **Statically-sized `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** This is properly mitigated. Input is read via `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces the bound. A defensive `buffer[sizeof(buffer) - 1U] = '\0'` guarantees termination, and `strnlen` (rather than `strlen`) bounds all length computations. Overlong input lines are safely drained via `discard_remaining_input()` so leftover characters don't pollute the next read.

3. **`squeeze_char()` robustness:** The function validates against `NULL` and zero-length buffers, clamps `len` to `max_len - 1`, and the write index can never exceed the read index, so the in-place compaction and final null-termination are always within bounds.

The code as written already addresses all identified weaknesses; the rewrite above preserves those mitigations with the comments removed.
