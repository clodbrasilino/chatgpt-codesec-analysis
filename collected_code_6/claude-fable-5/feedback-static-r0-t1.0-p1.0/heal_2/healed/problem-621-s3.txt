#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include <errno.h>

static int append_chunk(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || *buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n > SIZE_MAX - 1 || *len > SIZE_MAX - n - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t needed = *len + n + 1;
        size_t newcap = *cap;
        if (newcap == 0) {
            newcap = 1;
        }
        while (newcap < needed) {
            if (newcap > SIZE_MAX / 2) {
                return -1;
            }
            newcap *= 2;
        }
        char *tmp = realloc(*buf, newcap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = newcap;
    }
    if (*len + n + 1 > *cap) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

char *increment_numbers(const char *input, long k)
{
    size_t cap = 64;
    size_t len = 0;
    size_t i = 0;
    char *out;

    if (input == NULL) {
        return NULL;
    }

    out = malloc(cap);
    if (out == NULL) {
        return NULL;
    }
    out[0] = '\0';

    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            char *end = NULL;
            long val;
            long newval;
            char numbuf[32];
            int written;

            errno = 0;
            val = strtol(&input[i], &end, 10);
            if (errno == ERANGE || end == NULL || end == &input[i]) {
                free(out);
                return NULL;
            }
            if ((k > 0 && val > LONG_MAX - k) || (k < 0 && val < LONG_MIN - k)) {
                free(out);
                return NULL;
            }
            newval = val + k;

            written = snprintf(numbuf, sizeof numbuf, "%ld", newval);
            if (written < 0 || (size_t)written >= sizeof numbuf) {
                free(out);
                return NULL;
            }
            if (append_chunk(&out, &len, &cap, numbuf, (size_t)written) != 0) {
                free(out);
                return NULL;
            }
            i = (size_t)(end - input);
        } else {
            if (append_chunk(&out, &len, &cap, &input[i], 1) != 0) {
                free(out);
                return NULL;
            }
            i++;
        }
    }
    return out;
}

int main(void)
{
    const char *tests[] = {
        "abc 5 def 10",
        "no numbers here",
        "1,2,3",
        "value=99end",
        "0 and 100"
    };
    long k = 5;
    size_t i;
    size_t count = sizeof tests / sizeof tests[0];

    for (i = 0; i < count; i++) {
        char *result = increment_numbers(tests[i], k);
        if (result == NULL) {
            fprintf(stderr, "Failed to process string: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        if (printf("Input:  %s\nOutput: %s\n\n", tests[i], result) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
        free(result);
    }
    return EXIT_SUCCESS;
}

**Analysis of the flagged weaknesses and their fixes:**

1. **`memcpy` (CWE-120, risk 2):** The Flawfinder warning is a false positive in the original code, but the rewrite hardens it further. Before every `memcpy`, the buffer is guaranteed to hold `*len + n + 1` bytes: the growth loop expands `cap` with overflow guards (`newcap > SIZE_MAX / 2`), and a redundant post-condition check (`*len + n + 1 > *cap`) acts as a defensive fail-safe. I also added an explicit `n > SIZE_MAX - 1` guard so the arithmetic in the overflow check itself cannot wrap.

2. **`char numbuf[32]` (CWE-119/CWE-120, risk 2):** The statically sized array is safe because it is only written via `snprintf(numbuf, sizeof numbuf, ...)`, which enforces the bound, and the return value is checked for both negative (encoding error) and truncation (`written >= sizeof numbuf`) before the buffer is used. A 32-byte buffer comfortably holds any 64-bit `long` (max 20 digits plus sign and NUL).

Additional hardening in the rewrite: a `NULL` check on `end` after `strtol`, a guard against `cap == 0` in the doubling loop, and the existing signed-overflow check on `val + k` is retained. All allocation failures free `out` before returning, so there are no leaks on error paths.
