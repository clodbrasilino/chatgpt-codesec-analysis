#include <stdio.h>
#include <stdlib.h>

int is_scalene(int a, int b, int c) {
    int max_side = a;
    if (b > max_side) max_side = b;
    if (c > max_side) max_side = c;

    if (max_side >= (a + b + c - max_side)) {
        return 0;
    }

    if (a <= 0 || b <= 0 || c <= 0) {
        return 0;
    }

    if (a != b && b != c && a != c) {
        return 1;
    }

    return 0;
}

int main(void) {
    int a, b, c;

    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        return EXIT_FAILURE;
    }

    if (is_scalene(a, b, c)) {
        printf("Scalene\n");
    } else {
        printf("Not Scalene\n");
    }

    return EXIT_SUCCESS;
}
