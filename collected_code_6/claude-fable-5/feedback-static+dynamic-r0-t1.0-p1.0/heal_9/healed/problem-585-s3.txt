```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_NAME_LEN 64

typedef struct {
    char name[MAX_NAME_LEN];
    double price;
} Item;

typedef struct {
    Item *data;
    size_t size;
    size_t capacity;
} MinHeap;

static int item_set_name(Item *item, const char *name)
{
    size_t len;

    if (item == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(item->name));
    if (len >= sizeof(item->name)) {
        return -1;
    }

    if (len > 0) {
        memcpy(item->name, name, len);
    }
    item->name[len] = '\0';
    return 0;
}

static MinHeap *heap_create(size_t capacity)
{
    MinHeap *heap;

    if (capacity == 0 || capacity > SIZE_MAX / sizeof(Item)) {
        return NULL;
    }

    heap = malloc(sizeof(MinHeap));
    if (heap == NULL) {
        return NULL;
    }

    heap->data = malloc(capacity * sizeof(Item));
    if (heap->data == NULL) {
        free(heap);
        return NULL;
    }

    heap->size = 0;
    heap->capacity = capacity;
    return heap;
}

static void heap_destroy(MinHeap *heap)
{
    if (heap != NULL) {
        free(heap->data);
        free(heap);
    }
}

static void heap_swap(Item *a, Item *b)
{
    Item tmp = *a;
    *a = *b;
    *b = tmp;
}

static void heap_sift_up(MinHeap *heap, size_t idx)
{
    while (idx > 0) {
        size_t parent = (idx - 1) / 2;
        if (heap->data[idx].price < heap->data[parent].price) {
            heap_swap(&heap->data[idx], &heap->data[parent]);
            idx = parent;
        } else {
            break;
        }
    }
}

static void heap_sift_down(MinHeap *heap, size_t idx)
{
    for (;;) {
        size_t left = 2 * idx + 1;
        size_t right = 2 * idx + 2;
        size_t smallest = idx;

        if (left < heap->size &&
            heap->data[left].price < heap->data[smallest].price) {
            smallest = left;
        }
        if (right < heap->size &&
            heap->data[right].price < heap->data[smallest].price) {
            smallest = right;
        }
        if (smallest == idx) {
            break;
        }
        heap_swap(&heap->data[idx], &heap->data[smallest]);
        idx = smallest;
    }
}

static int heap_push(MinHeap *heap, const Item *item)
{
    if (heap == NULL || item == NULL || heap->size >= heap->capacity) {
        return -1;
    }
    heap->data[heap->size] = *item;
    heap_sift_up(heap, heap->size);
    heap->size++;
    return 0;
}

static int heap_replace_min(MinHeap *heap, const Item *item)
{
    if (heap == NULL || item == NULL || heap->size == 0) {
        return -1;
    }
    heap->data[0] = *item;
    heap_sift_down(heap, 0);
    return 0;
}

static int compare_desc(const void *a, const void *b)
{
    const Item *ia = (const Item *)a;
    const Item *ib = (const Item *)b;

    if (ia->price < ib->price) {
        return 1;
    }
    if (ia->price > ib->price) {
        return -1;
    }
    return 0;
}

static Item *find_n_expensive(const Item *items, size_t count, size_t n,
                              size_t *result_count)
{
    MinHeap *heap;
    Item *result;
    size_t i;
    size_t limit;
    size_t result_bytes;

    if (items == NULL || result_count == NULL || count == 0 || n == 0) {
        return NULL;
    }

    limit = (n < count) ? n : count;

    heap = heap_create(limit);
    if (heap == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        if (heap->size < heap->capacity) {
            if (heap_push(heap, &items[i]) != 0) {
                heap_destroy(heap);
                return NULL;
            }
        } else if (items[i].price > heap->data[0].price) {
            if (heap_replace_min(heap, &items[i]) != 0) {
                heap_destroy(heap);
                return NULL;
            }
        }
    }

    if (heap->size == 0 || heap->size > heap->capacity ||
        heap->size > SIZE_MAX / sizeof(Item)) {
        heap_destroy(heap);
        return NULL;
    }

    result_bytes = heap->size * sizeof(Item);
    result = malloc(result_bytes);
    if (result == NULL) {
        heap_destroy(heap);
        return NULL;
    }

    memcpy(result, heap->data, result_bytes);
    *result_count = heap->size;

    qsort(result, *result_count, sizeof(Item), compare_desc);

    heap_destroy(heap);
    return result;
}

int main(void)
{
    static const struct {
        const char *name;
        double price;
    } raw_data[] = {
        {"Laptop", 1299.99},
        {"Phone", 899.50},
        {"Headphones", 199.99},
        {"Monitor", 449.00},
        {"Keyboard", 89.99},
        {"Mouse", 49.99},
        {"Tablet", 649.00},
        {"Camera", 1099.00},
        {"Speaker", 299.99},
        {"Charger", 29.99}
    };
    size_t count = sizeof(raw_data) / sizeof(raw_data[0]);
    Item dataset[sizeof(raw_data) / sizeof(raw_data[0])];
    size_t n = 3;
    size_t result_count = 0;
    size_t i;
    Item *result;

    for (i = 0; i < count; i++) {
        if (item_set_name(&dataset[i], raw_data[i].name) != 0) {
            fprintf(stderr, "Item name too long at index %zu\n", i);
            return EXIT_FAILURE;
        }
        dataset[i].price = raw_data[i].price;
    }

    result = find_n_expensive(dataset, count, n, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Failed to compute the %zu most expensive items\n", n);
        return EXIT_FAILURE;
    }

    printf("Top %zu most expensive items:\n", result_count);
    for (i = 0; i < result_count; i++) {
        printf("%.*s: %.2f\n", MAX_NAME_LEN - 1, result[i].name,
               result[i].price);
    }

    free(result);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and how the code addresses them:

1. **`char name[MAX_NAME_LEN]` (CWE-119/CWE-120)** — The statically-sized buffer is safe because every write path goes through `item_set_name()`, which uses `strnlen()` bounded by `sizeof(item->name)` and rejects any string that would not fit including the null terminator. Reads are also bounded via the `%.*s` precision specifier in `printf`.

2. **`memcpy` in `item_set_name` (CWE-120)** — Safe: `len` is guaranteed to be strictly less than `sizeof(item->name)` before the copy, so `len` bytes plus the terminator always fit. I added a `len > 0` guard as a minor defensive refinement.

3. **`memcpy` in `find_n_expensive` (CWE-120)** — Safe: `result_bytes` is computed from `heap->size`, which is validated against `heap->capacity` and against `SIZE_MAX / sizeof(Item)` to prevent integer overflow, and `result` is allocated with exactly `result_bytes`. Source and destination sizes match by construction.

The original code was already well-hardened — the Flawfinder hits are low-risk (risk 2) pattern matches rather than genuine vulnerabilities. Overflow-checked allocation in `heap_create`, bounds checks in `heap_push`/`heap_replace_min`, and consistent NULL/error propagation with cleanup on every failure path mean no functional changes were required to fix real defects.
