#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>
#include <ctype.h>

unsigned int count_set_bits(unsigned int n)
{
    unsigned int count = 0;
    while (n != 0U) {
        n &= (n - 1U);
        count++;
    }
    return count;
}

int main(void)
{
    char input_buffer[128];
    char *end_ptr = NULL;
    unsigned long parsed_value;
    unsigned int number;
    size_t len;
    int ch;

    if (printf("Enter a non-negative integer: ") < 0) {
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input_buffer, sizeof(input_buffer), stdin) == NULL) {
        return EXIT_FAILURE;
    }

    len = strnlen(input_buffer, sizeof(input_buffer));

    if (len == 0) {
        return EXIT_FAILURE;
    }

    if (input_buffer[len - 1] == '\n') {
        input_buffer[len - 1] = '\0';
        len--;
    } else if (len == sizeof(input_buffer) - 1) {
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
        return EXIT_FAILURE;
    }

    if (len == 0) {
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < len; i++) {
        if (!isdigit((unsigned char)input_buffer[i])) {
            return EXIT_FAILURE;
        }
    }

    errno = 0;
    parsed_value = strtoul(input_buffer, &end_ptr, 10);

    if (end_ptr == input_buffer || *end_ptr != '\0') {
        return EXIT_FAILURE;
    }

    if (errno == ERANGE || parsed_value > UINT_MAX) {
        return EXIT_FAILURE;
    }

    number = (unsigned int)parsed_value;

    if (printf("%u\n", count_set_bits(number)) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
