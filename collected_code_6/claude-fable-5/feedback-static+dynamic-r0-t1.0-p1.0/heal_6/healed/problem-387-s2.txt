```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 128
#define MAX_FLUSH_ITERATIONS 4096

int is_valid_hex(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    size_t start = 0;

    if (str == NULL || max_len == 0) {
        return 0;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len >= max_len) {
        return 0;
    }

    if (len > 2 && str[0] == '0' && (str[1] == 'x' || str[1] == 'X')) {
        start = 2;
    }

    if (start >= len) {
        return 0;
    }

    for (i = start; i < len; i++) {
        if (!isxdigit((unsigned char)str[i])) {
            return 0;
        }
    }

    return 1;
}

int hex_is_even(const char *hex_str, size_t max_len)
{
    size_t len;
    char last;
    int digit_value;

    if (hex_str == NULL || max_len == 0) {
        return 0;
    }

    len = strnlen(hex_str, max_len);
    if (len == 0 || len >= max_len) {
        return 0;
    }

    last = hex_str[len - 1];

    if (last >= '0' && last <= '9') {
        digit_value = last - '0';
    } else if (last >= 'a' && last <= 'f') {
        digit_value = last - 'a' + 10;
    } else if (last >= 'A' && last <= 'F') {
        digit_value = last - 'A' + 10;
    } else {
        return 0;
    }

    return (digit_value % 2) == 0;
}

static void flush_stdin(void)
{
    int ch;
    size_t iterations = 0;

    do {
        ch = getchar();
        iterations++;
    } while (ch != '\n' && ch != EOF && iterations < MAX_FLUSH_ITERATIONS);
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a hexadecimal number: ");
    fflush(stdout);

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        flush_stdin();
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (!is_valid_hex(buffer, sizeof(buffer))) {
        fprintf(stderr, "Invalid hexadecimal number\n");
        return EXIT_FAILURE;
    }

    if (hex_is_even(buffer, sizeof(buffer))) {
        printf("%s is even\n", buffer);
    } else {
        printf("%s is odd\n", buffer);
    }

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `getchar` warning (CWE-120, CWE-20) in `flush_stdin`:** This is a false positive in practice. The `getchar()` result is never written into a buffer — it is only used for loop control. The loop also has a hard iteration cap (`MAX_FLUSH_ITERATIONS`) preventing unbounded looping on a stream that never delivers `\n` or `EOF`, and it correctly checks for `EOF` using an `int` return type rather than `char`.

**2. Flawfinder `char buffer[MAX_INPUT_LEN]` warning (CWE-119/CWE-120):** The statically-sized array is properly protected:
- Input is read with `fgets(buffer, sizeof(buffer), stdin)`, which limits the write to the buffer size.
- The buffer is zero-initialized with `memset` and explicitly NUL-terminated at `sizeof(buffer) - 1` as a defensive measure.
- Over-long input (no newline found at max fill) is detected, the remainder of the line is drained via `flush_stdin`, and the program exits with an error instead of processing truncated data.
- All subsequent string operations use `strnlen` with an explicit bound rather than `strlen`, and both `is_valid_hex` and `hex_is_even` validate `NULL`, zero-length, and unterminated (`len >= max_len`) cases before touching the data.

Additional good practices already present: `isxdigit` receives an `unsigned char` cast to avoid undefined behavior with negative `char` values, and output uses `printf("%s", buffer)` with a fixed format string, avoiding format-string injection (CWE-134). No changes to the logic were required.
