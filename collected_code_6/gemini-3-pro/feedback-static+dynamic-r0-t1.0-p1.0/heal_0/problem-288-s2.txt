#include <stdio.h>
#include <stdlib.h>

int count_self_inverses(const int *arr, size_t size, int p) {
    if (arr == NULL || p <= 1) {
        return 0;
    }

    int count = 0;
    for (size_t i = 0; i < size; ++i) {
        long long val = (long long)arr[i];
        
        long long mod_val = val % p;
        if (mod_val < 0) {
            mod_val += p;
        }

        if (mod_val == 0) {
            continue;
        }

        if ((mod_val * mod_val) % p == 1) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int arr[] = {1, 2, 3, 6, 8, 10, -1};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    int p = 7;

    int result = count_self_inverses(arr, size, p);
    printf("%d\n", result);

    return 0;
}
