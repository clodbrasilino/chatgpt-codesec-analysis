```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[32];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

The original code was already largely safe, but the Flawfinder warning about the statically-sized `digits[32]` buffer (CWE-119/CWE-120) prompted these hardening measures:

1. **Buffer bounds** — The buffer size of 32 bytes comfortably exceeds the maximum possible length of a `long long` decimal representation (19 digits plus sign and NUL terminator, i.e. 21 bytes max). The existing `snprintf` return-value check (`len < 0 || (size_t)len >= sizeof(digits)`) already guarantees no truncation or overflow occurred; this is retained as the primary defense.

2. **Buffer initialization** — Added `memset` to zero the buffer before use, ensuring no uninitialized memory is ever read even if logic changes later.

3. **Bounded search loop** — The original `while (digits[j] <= digits[i]) j--;` could theoretically underrun the buffer if invariants were violated. Adding the `j > i` guard and the subsequent `j <= i` check makes the loop provably bounded regardless of input, directly addressing the out-of-bounds read risk flagged under CWE-119.

4. **Digit validation** — Added a `d < 0 || d > 9` sanity check before arithmetic, so any unexpected non-digit byte in the buffer causes a clean failure rather than corrupt output.

5. **Overflow protection** — The existing `LLONG_MAX` pre-check before `result * 10 + d` is preserved, preventing signed integer overflow (undefined behavior) during reconstruction.

The algorithm itself (next lexicographic permutation of digits) is unchanged and correct.
