```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_BINARY_DIGITS 64
#define INPUT_BUFFER_SIZE 128

static int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value;

    if (binary == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0 || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    value = 0;
    for (i = 0; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int read_line(char *buffer, size_t size)
{
    size_t len;
    int ch;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    memset(buffer, 0, size);

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = strnlen(buffer, size);
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
        return 0;
    }

    if (len == size - 1) {
        do {
            ch = getchar();
        } while (ch != '\n' && ch != EOF);
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal;

    printf("Enter a binary number: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (read_line(input, sizeof(input)) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long\n");
        return EXIT_FAILURE;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fixes applied:**

The Flawfinder warning about the statically-sized `char input[INPUT_BUFFER_SIZE]` buffer (CWE-119/CWE-120) was already largely mitigated in the original code, since `read_line` receives the buffer size via `sizeof(input)`, uses `fgets` with the exact size, forces null termination, and uses `strnlen` instead of `strlen`. The rewrite preserves all of those defenses and hardens the code further:

1. **Size validation in `read_line`:** The guard is tightened to `size < 2` (a 1-byte buffer can never hold a valid line) and `size > INT_MAX` is rejected before the cast to `int` for `fgets`, eliminating a potential signed-overflow/truncation issue on the cast.

2. **Stdin drain on overlong input:** When input exceeds the buffer, the leftover characters are now consumed from `stdin`. In the original, an overly long line left residue in the stream, which could corrupt subsequent reads in a larger program.

3. **`max_len == 0` check in `binary_to_decimal`:** Prevents `strnlen` misuse if the function is ever called with a zero length.

4. **`fflush(stdout)` after the prompt:** Guarantees the prompt is visible before blocking on input when stdout is fully buffered (e.g., piped).

5. **Retained overflow protection:** The `value > (ULONG_MAX >> 1)` check before each shift and the `MAX_BINARY_DIGITS` cap correctly prevent unsigned wraparound during conversion, and the strict `'0'`/`'1'` validation rejects malformed input.
