```c
#include <stdio.h>
#include <stdbool.h>

bool is_even(int num) {
    return num % 2 == 0;
}

int main(void) {
    int num;
    if (scanf("%d", &num) == 1) {
        if (is_even(num)) {
            printf("true\n");
        } else {
            printf("false\n");
        }
    } else {
        return 1;
    }
    return 0;
}
```
