#include <stdio.h>
#include <stdlib.h>

unsigned long long pell_number(int n) {
    if (n < 0) {
        fprintf(stderr, "Error: negative input\n");
        exit(EXIT_FAILURE);
    }
    if (n == 0) return 0;
    if (n == 1) return 1;

    unsigned long long a = 0;
    unsigned long long b = 1;
    unsigned long long c;

    for (int i = 2; i <= n; ++i) {
        c = 2 * b + a;
        a = b;
        b = c;
    }
    return b;
}

int main(void) {
    int n;
    printf("Enter n: ");
    if (scanf("%d", &n) != 1) {
        fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    unsigned long long result = pell_number(n);
    printf("Pell number P%d = %llu\n", n, result);

    return EXIT_SUCCESS;
}
