```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <errno.h>
#include <string.h>

#define INPUT_BUFFER_SIZE 64
#define DISCARD_GUARD_LIMIT 4096

static int closest_smaller(int n, int *result)
{
    if (result == NULL) {
        return -1;
    }
    if (n == INT_MIN) {
        return -1;
    }
    *result = n - 1;
    return 0;
}

static int parse_int(const char *str, int *value)
{
    char *endptr = NULL;
    long parsed;

    if (str == NULL || value == NULL) {
        return -1;
    }

    errno = 0;
    parsed = strtol(str, &endptr, 10);

    if (errno == ERANGE) {
        return -1;
    }
    if (endptr == str) {
        return -1;
    }
    if (*endptr != '\0' && *endptr != '\n') {
        return -1;
    }
    if (parsed > INT_MAX || parsed < INT_MIN) {
        return -1;
    }

    *value = (int)parsed;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;
    size_t guard = 0;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < (size_t)DISCARD_GUARD_LIMIT);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }

    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }

    buffer[size - 1] = '\0';

    len = strnlen(buffer, size);
    if (len > 0 && buffer[len - 1] != '\n' && len == size - 1) {
        discard_remaining_input();
        return -1;
    }

    return 0;
}

int main(void)
{
    char buffer[INPUT_BUFFER_SIZE];
    int n = 0;
    int result = 0;

    memset(buffer, 0, sizeof(buffer));

    if (read_line(buffer, sizeof(buffer)) != 0) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    if (parse_int(buffer, &n) != 0) {
        fprintf(stderr, "Error: invalid integer input\n");
        return EXIT_FAILURE;
    }

    if (closest_smaller(n, &result) != 0) {
        fprintf(stderr, "Error: no smaller integer representable\n");
        return EXIT_FAILURE;
    }

    if (printf("%d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **`getchar()` in a loop (CWE-120, CWE-20):** The Flawfinder warning in `discard_remaining_input()` is a false positive in practice here. The loop never writes to a buffer — it only reads and discards characters. Additionally, the `guard` counter with `DISCARD_GUARD_LIMIT` bounds the iteration count, preventing an unbounded loop on a malicious or malformed stream that never delivers `'\n'` or `EOF`.

2. **Statically-sized `char` array (CWE-119/CWE-120):** The `buffer[INPUT_BUFFER_SIZE]` warning is mitigated because:
   - `read_line()` passes `sizeof(buffer)` and uses `fgets()`, which limits writes to `size - 1` characters plus a null terminator.
   - The buffer is explicitly null-terminated at `size - 1` as a belt-and-suspenders measure.
   - `strnlen()` is used instead of `strlen()`, so length computation is bounded even if termination somehow failed.
   - Truncated input (line longer than the buffer) is detected and rejected, with the rest of the stdin line drained to avoid leaving stale data for subsequent reads.

3. **Integer handling:** `parse_int()` uses `strtol()` with full error checking (`ERANGE`, no-conversion, trailing garbage, and range narrowing to `int`), and `closest_smaller()` guards against `INT_MIN - 1` underflow (undefined behavior in C).

The code as written already addresses all identified weaknesses; no functional changes were required beyond removing the comments.
