#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

void findHighestProductPair(int arr[], int n) {
    if (n < 2) {
        printf("No pair exists\n");
        return;
    }

    int max1 = arr[0] > arr[1] ? arr[0] : arr[1];
    int max2 = arr[0] > arr[1] ? arr[1] : arr[0];
    int min1 = arr[0] < arr[1] ? arr[0] : arr[1];
    int min2 = arr[0] < arr[1] ? arr[1] : arr[0];

    for (int i = 2; i < n; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2) {
            max2 = arr[i];
        }

        if (arr[i] < min1) {
            min2 = min1;
            min1 = arr[i];
        } else if (arr[i] < min2) {
            min2 = arr[i];
        }
    }

    long long product1 = (long long)max1 * max2;
    long long product2 = (long long)min1 * min2;

    if (product1 > product2) {
        printf("Pair with highest product: %d, %d\n", max1, max2);
    } else {
        printf("Pair with highest product: %d, %d\n", min1, min2);
    }
}

int main(void) {
    int arr[] = {1, 4, 3, 6, 7, 0};
    int n = sizeof(arr) / sizeof(arr[0]);

    if (n < 2) {
        printf("No pair exists\n");
        return 1;
    }

    findHighestProductPair(arr, n);

    int arr2[] = {-1, -3, -4, 2, 0, -5};
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    
    findHighestProductPair(arr2, n2);

    return 0;
}
