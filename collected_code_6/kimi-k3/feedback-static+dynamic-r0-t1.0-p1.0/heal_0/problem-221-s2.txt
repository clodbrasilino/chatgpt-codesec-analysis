```c
#include <stdio.h>

int find_first_even(const int *numbers, size_t size, int *result)
{
    size_t i;

    if (numbers == NULL || result == NULL || size == 0)
    {
        return 0;
    }

    for (i = 0; i < size; i++)
    {
        if (numbers[i] % 2 == 0)
        {
            *result = numbers[i];
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    int numbers[] = {1, 3, 5, 7, 8, 10, 12};
    size_t size = sizeof(numbers) / sizeof(numbers[0]);
    int result = 0;

    if (find_first_even(numbers, size, &result))
    {
        printf("First even number: %d\n", result);
    }
    else
    {
        printf("No even number found.\n");
    }

    return 0;
}
```
