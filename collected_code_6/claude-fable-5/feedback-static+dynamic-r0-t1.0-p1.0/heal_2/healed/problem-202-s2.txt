#include <stdio.h>

#include <string.h>

#define MAX_INPUT_LEN 255U

char *remove_even_chars(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len / 2 + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (i % 2 != 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN + 1];
    char *output;
    size_t len;

    printf("Enter a string: ");

        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    output = remove_even_chars(buffer);
    if (output == NULL) {
        fprintf(stderr, "Error processing string\n");
        return EXIT_FAILURE;
    }

    printf("Result: %s\n", output);

The original code was already well-defended against the flagged weakness (CWE-119/CWE-120). Here's my analysis of why the statically-sized `buffer` is safe as written:

1. **Bounded read**: `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to `sizeof(buffer) - 1` characters plus the null terminator, so no overflow can occur on read.

2. **Guaranteed termination**: `buffer[sizeof(buffer) - 1] = '\0';` is a defensive belt-and-suspenders measure ensuring null termination even in edge cases.

3. **Bounded length calculation**: `strnlen(buffer, sizeof(buffer))` never scans past the buffer, unlike plain `strlen` on a potentially unterminated string.

4. **Safe callee**: `remove_even_chars` validates for `NULL`, re-checks the length cap with `strnlen(input, MAX_INPUT_LEN + 1)`, checks the `malloc` return value, and allocates exactly `len / 2 + 1` bytes — sufficient since the loop writes at most `len / 2` characters plus the terminator (`j` can never exceed `len / 2` because only odd indices are copied).

5. **No leaks**: the allocated `output` is freed on the success path, and the error paths exit before allocation succeeds or after it fails.
