#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char **split_at_uppercase(const char *input, size_t *count)
{
    size_t len;
    size_t i;
    size_t start;
    size_t parts;
    size_t idx;
    char **result;

    if (input == NULL || count == NULL) {
        return NULL;
    }

    *count = 0;
    len = strlen(input);
    if (len == 0) {
        return NULL;
    }

    parts = 1;
    for (i = 1; i < len; i++) {
        if (isupper((unsigned char)input[i])) {
            parts++;
        }
    }

    result = malloc(parts * sizeof(char *));
    if (result == NULL) {
        return NULL;
    }

    idx = 0;
    start = 0;
    for (i = 1; i <= len; i++) {
        if (i == len || isupper((unsigned char)input[i])) {
            size_t seg_len = i - start;
            char *segment = malloc(seg_len + 1);
            if (segment == NULL) {
                size_t j;
                for (j = 0; j < idx; j++) {
                    free(result[j]);
                }
                free(result);
                return NULL;
            }
            memcpy(segment, input + start, seg_len);
            segment[seg_len] = '\0';
            result[idx] = segment;
            idx++;
            start = i;
        }
    }

    *count = idx;
    return result;
}

void free_split(char **parts, size_t count)
{
    size_t i;

    if (parts == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(parts[i]);
    }
    free(parts);
}

int main(void)
{
    const char *text = "helloWorldThisIsATest";
    size_t count;
    size_t i;
    char **parts;

    parts = split_at_uppercase(text, &count);
    if (parts == NULL) {
        fprintf(stderr, "Failed to split string\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", parts[i]) < 0) {
            free_split(parts, count);
            return EXIT_FAILURE;
        }
    }

    free_split(parts, count);
    return EXIT_SUCCESS;
}
