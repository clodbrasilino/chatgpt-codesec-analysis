```c
#include <stdio.h>
#include <string.h>

#define BUFFER_SIZE 1024

void remove_multiple_spaces(char *str)
{
    size_t read_idx = 0;
    size_t write_idx = 0;
    int prev_space = 0;

    if (str == NULL) {
        return;
    }

    while (str[read_idx] != '\0') {
        if (str[read_idx] == ' ') {
            if (!prev_space) {
                str[write_idx] = str[read_idx];
                write_idx++;
            }
            prev_space = 1;
        } else {
            str[write_idx] = str[read_idx];
            write_idx++;
            prev_space = 0;
        }
        read_idx++;
    }
    str[write_idx] = '\0';
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    if (printf("Enter a string: ") < 0) {
        fprintf(stderr, "Error writing output\n");
        return 1;
    }

    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error flushing output\n");
        return 1;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    remove_multiple_spaces(buffer);

    if (printf("Result: %s\n", buffer) < 0) {
        fprintf(stderr, "Error writing output\n");
        return 1;
    }

    return 0;
}
```

**Analysis of the fixes applied:**

1. **CWE-119/CWE-120 (statically-sized buffer):** The original code already mitigated this correctly — `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to the buffer size, and `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination as defense-in-depth. This is retained.

2. **CWE-126 (`strlen` over-read on non-terminated strings):** Replaced `strlen` with `strnlen(buffer, sizeof(buffer))`, which bounds the scan to the buffer size and cannot read past it even if termination were somehow missing.

3. **In-place compaction safety:** `remove_multiple_spaces` is safe by construction since `write_idx <= read_idx` always holds, so it never writes beyond the original string length. The `NULL` check is preserved.

4. **I/O error handling:** All return values of `printf`, `fflush`, and `fgets` are checked, which is good defensive practice and was kept intact.
