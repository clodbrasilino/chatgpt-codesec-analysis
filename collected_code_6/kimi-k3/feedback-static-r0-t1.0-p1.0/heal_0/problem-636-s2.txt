#include <stdio.h>
#include <math.h>

int are_roots_reciprocal(double a, double b, double c) {
    if (fabs(a) < 1e-9) {
        return 0;
    }
    double discriminant = b * b - 4 * a * c;
    if (discriminant < 0) {
        return 0;
    }
    return fabs(c / a - 1.0) < 1e-9;
}

int main(void) {
    double a, b, c;
    if (scanf("%lf %lf %lf", &a, &b, &c) != 3) {
        return 1;
    }
    if (are_roots_reciprocal(a, b, c)) {
        printf("Roots are reciprocal\n");
    } else {
        printf("Roots are not reciprocal\n");
    }
    return 0;
}
