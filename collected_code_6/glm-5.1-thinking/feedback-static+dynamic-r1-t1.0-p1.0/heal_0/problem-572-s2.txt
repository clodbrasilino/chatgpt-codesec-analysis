#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int value;
    size_t original_index;
} Element;

int cmp_value(const void *a, const void *b) {
    int va = ((const Element*)a)->value;
    int vb = ((const Element*)b)->value;
    if (va < vb) return -1;
    if (va > vb) return 1;
    return 0;
}

size_t remove_two_duplicates(int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return 0;
    }
    
    Element *elements = malloc(size * sizeof(Element));
    if (elements == NULL) {
        return 0;
    }
    
    for (size_t i = 0; i < size; i++) {
        elements[i].value = arr[i];
        elements[i].original_index = i;
    }
    
    qsort(elements, size, sizeof(Element), cmp_value);
    
    int *keep = malloc(size * sizeof(int));
    if (keep == NULL) {
        free(elements);
        return 0;
    }
    
    memset(keep, 0, size * sizeof(int));
    
    size_t i = 0;
    while (i < size) {
        size_t j = i;
        while (j < size && elements[j].value == elements[i].value) {
            j++;
        }
        size_t count = j - i;
        size_t keep_count = count > 2 ? 2 : count;
        for (size_t k = i; k < i + keep_count; k++) {
            keep[elements[k].original_index] = 1;
        }
        i = j;
    }
    
    free(elements);
    
    size_t new_size = 0;
    for (size_t k = 0; k < size; k++) {
        if (keep[k]) {
            arr[new_size++] = arr[k];
        }
    }
    
    free(keep);
    return new_size;
}

int main(void) {
    int arr[] = {5, 2, 2, 3, 3, 3, 4, 1, 1};
    size_t size = sizeof(arr) / sizeof(arr[0]);
    size_t new_size = remove_two_duplicates(arr, size);
    
    for (size_t i = 0; i < new_size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
