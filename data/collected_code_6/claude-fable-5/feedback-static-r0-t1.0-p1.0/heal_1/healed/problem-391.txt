```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LEN 4096U

typedef struct NestedEntry {
    char *outer_key;
    char *inner_key;
    int value;
} NestedEntry;

typedef struct NestedDict {
    NestedEntry *entries;
    size_t count;
} NestedDict;

static char *duplicate_string(const char *src)
{
    if (src == NULL) {
        return NULL;
    }

    size_t len = strnlen(src, MAX_STRING_LEN);
    if (len >= MAX_STRING_LEN) {
        return NULL;
    }

    char *copy = malloc(len + 1U);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

void free_nested_dict(NestedDict *dict)
{
    if (dict == NULL) {
        return;
    }
    if (dict->entries != NULL) {
        for (size_t i = 0; i < dict->count; i++) {
            free(dict->entries[i].outer_key);
            free(dict->entries[i].inner_key);
        }
        free(dict->entries);
    }
    free(dict);
}

NestedDict *convert_lists_to_nested_dict(const char **keys,
                                         const char **subkeys,
                                         const int *values,
                                         size_t count)
{
    if (keys == NULL || subkeys == NULL || values == NULL || count == 0U) {
        return NULL;
    }

    NestedDict *dict = malloc(sizeof(NestedDict));
    if (dict == NULL) {
        return NULL;
    }

    dict->entries = calloc(count, sizeof(NestedEntry));
    if (dict->entries == NULL) {
        free(dict);
        return NULL;
    }
    dict->count = count;

    for (size_t i = 0; i < count; i++) {
        dict->entries[i].outer_key = duplicate_string(keys[i]);
        dict->entries[i].inner_key = duplicate_string(subkeys[i]);
        dict->entries[i].value = values[i];
        if (dict->entries[i].outer_key == NULL ||
            dict->entries[i].inner_key == NULL) {
            free_nested_dict(dict);
            return NULL;
        }
    }

    return dict;
}

void print_nested_dict(const NestedDict *dict)
{
    if (dict == NULL || dict->entries == NULL) {
        return;
    }

    printf("[");
    for (size_t i = 0; i < dict->count; i++) {
        printf("{\"%s\": {\"%s\": %d}}",
               dict->entries[i].outer_key,
               dict->entries[i].inner_key,
               dict->entries[i].value);
        if (i + 1U < dict->count) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main(void)
{
    const char *ids[] = { "S001", "S002", "S003", "S004" };
    const char *names[] = { "Adina Park", "Leyton Marsh",
                            "Duncan Boyle", "Saim Richards" };
    const int scores[] = { 85, 98, 89, 92 };
    size_t count = sizeof(scores) / sizeof(scores[0]);

    NestedDict *dict = convert_lists_to_nested_dict(ids, names, scores, count);
    if (dict == NULL) {
        fprintf(stderr, "Failed to build nested dictionary\n");
        return EXIT_FAILURE;
    }

    print_nested_dict(dict);
    free_nested_dict(dict);

    return EXIT_SUCCESS;
}
```

Key changes addressing the flagged weaknesses:

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen` bounded by `MAX_STRING_LEN`, so a non-terminated string can no longer cause an unbounded over-read. Inputs at or beyond the bound are rejected.

2. **CWE-120 (`memcpy` overflow):** The destination buffer is now allocated as `len + 1U` where `len` is the validated, bounded length; `memcpy` copies exactly `len` bytes and the terminator is written explicitly, guaranteeing the destination always holds the source data.

3. **Variable scope (`cppcheck` warning):** Loop counters (`i`) are now declared inside the `for` statements (C99 style), and other locals (`len`, `copy`, `dict`) are declared at first use, minimizing their scope.

The memory-management structure is preserved: `free_nested_dict` remains safe to call on partially constructed dictionaries since `calloc` zero-initializes all pointers, making the cleanup path on allocation failure correct.
