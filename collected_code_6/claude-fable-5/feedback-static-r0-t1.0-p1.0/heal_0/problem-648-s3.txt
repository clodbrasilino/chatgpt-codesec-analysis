#include <stdio.h>
#include <stdlib.h>

int swap_pairs(int *list, size_t length)
{
    size_t i;

    if (list == NULL) {
        return -1;
    }

    for (i = 0; i + 1 < length; i += 2) {
        int temp = list[i];
        list[i] = list[i + 1];
        list[i + 1] = temp;
    }

    return 0;
}

int print_list(const int *list, size_t length)
{
    size_t i;

    if (list == NULL) {
        return -1;
    }

    for (i = 0; i < length; i++) {
        if (printf("%d ", list[i]) < 0) {
            return -1;
        }
    }

    if (printf("\n") < 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    int values[] = {1, 2, 3, 4, 5, 6, 7};
    size_t length = sizeof(values) / sizeof(values[0]);

    if (print_list(values, length) != 0) {
        return EXIT_FAILURE;
    }

    if (swap_pairs(values, length) != 0) {
        return EXIT_FAILURE;
    }

    if (print_list(values, length) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
