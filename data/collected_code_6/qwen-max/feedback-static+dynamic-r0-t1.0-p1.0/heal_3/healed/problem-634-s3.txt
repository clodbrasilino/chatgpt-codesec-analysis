#include <stdio.h>
#include <stdint.h>
#include <inttypes.h>

long long sum_of_fourth_power_even(int n) {
    long long sum = 0;
    for (int i = 1; i <= n; ++i) {
        int64_t even = 2 * (int64_t)i;
        int64_t fourth_power = even * even * even * even;
        if (__builtin_add_overflow(sum, fourth_power, &sum)) {
            return -1;
        }
    }
    return sum;
}

int main() {
    int n;
    printf("Enter the number of even natural numbers: ");
    if (scanf("%d", &n) != 1 || n < 0) {
        printf("Invalid input.\n");
        return 1;
    }
    long long result = sum_of_fourth_power_even(n);
    if (result == -1) {
        printf("Overflow occurred.\n");
    } else {
        printf("The sum of the fourth power of the first %d even natural numbers is: %lld\n", n, result);
    }
    return 0;
}
