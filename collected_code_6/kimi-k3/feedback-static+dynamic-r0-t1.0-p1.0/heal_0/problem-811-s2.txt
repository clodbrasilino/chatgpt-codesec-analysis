#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int x;
    int y;
} Tuple;

int are_identical(const Tuple *list1, size_t size1, const Tuple *list2, size_t size2) {
    size_t i;
    
    if (list1 == NULL || list2 == NULL) {
        return 0;
    }
    
    if (size1 != size2) {
        return 0;
    }
    
    for (i = 0; i < size1; i++) {
        if (list1[i].x != list2[i].x || list1[i].y != list2[i].y) {
            return 0;
        }
    }
    
    return 1;
}

int main(void) {
    Tuple list1[] = {{1, 2}, {3, 4}, {5, 6}};
    Tuple list2[] = {{1, 2}, {3, 4}, {5, 6}};
    Tuple list3[] = {{1, 2}, {3, 4}, {5, 7}};
    Tuple list4[] = {{1, 2}, {3, 4}};
    size_t size1 = sizeof(list1) / sizeof(list1[0]);
    size_t size2 = sizeof(list2) / sizeof(list2[0]);
    size_t size3 = sizeof(list3) / sizeof(list3[0]);
    size_t size4 = sizeof(list4) / sizeof(list4[0]);
    int result;
    
    result = are_identical(list1, size1, list2, size2);
    printf("List1 vs List2: %s\n", result ? "Identical" : "Not identical");
    
    result = are_identical(list1, size1, list3, size3);
    printf("List1 vs List3: %s\n", result ? "Identical" : "Not identical");
    
    result = are_identical(list1, size1, list4, size4);
    printf("List1 vs List4: %s\n", result ? "Identical" : "Not identical");
    
    result = are_identical(NULL, 0, list1, size1);
    printf("NULL vs List1: %s\n", result ? "Identical" : "Not identical");
    
    return 0;
}
