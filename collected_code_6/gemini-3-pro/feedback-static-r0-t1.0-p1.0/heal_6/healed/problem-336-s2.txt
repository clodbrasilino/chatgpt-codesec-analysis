#include <stdio.h>
#include <ctype.h>
#include <stdbool.h>

bool has_28_days(const char *month) {
    if (month == NULL) {
        return false;
    }

    const char *full = "february";
    const char *short_name = "feb";
    size_t i = 0;

    while (full[i] != '\0' && tolower((unsigned char)month[i]) == full[i]) {
        i++;
    }
    if (full[i] == '\0' && month[i] == '\0') {
        return true;
    }

    i = 0;
    while (short_name[i] != '\0' && tolower((unsigned char)month[i]) == short_name[i]) {
        i++;
    }
    if (short_name[i] == '\0' && month[i] == '\0') {
        return true;
    }

    return false;
}

int main(void) {
    const char *test_months[] = {"February", "feb", "JANUARY", "April", "Invalid", NULL};
    
    for (size_t i = 0; i < sizeof(test_months) / sizeof(test_months[0]); i++) {
        if (test_months[i] != NULL) {
            bool result = has_28_days(test_months[i]);
            printf("%s: %s\n", test_months[i], result ? "Yes" : "No");
        } else {
            bool result = has_28_days(test_months[i]);
            printf("NULL: %s\n", result ? "Yes" : "No");
        }
    }
    
    return 0;
}
