#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

static bool is_palindrome(int num)
{
    char buf[32];
    int len;
    int i;
    int j;

    if (num < 0) {
        return false;
    }

    len = snprintf(buf, sizeof(buf), "%d", num);
    if (len < 0 || (size_t)len >= sizeof(buf)) {
        return false;
    }

    i = 0;
    j = len - 1;
    while (i < j) {
        if (buf[i] != buf[j]) {
            return false;
        }
        i++;
        j--;
    }
    return true;
}

static bool find_largest_palindrome(const int *arr, size_t size, int *result)
{
    size_t i;
    bool found = false;
    int largest = 0;

    if (arr == NULL || result == NULL || size == 0) {
        return false;
    }

    for (i = 0; i < size; i++) {
        if (is_palindrome(arr[i])) {
            if (!found || arr[i] > largest) {
                largest = arr[i];
                found = true;
            }
        }
    }

    if (found) {
        *result = largest;
    }
    return found;
}

int main(void)
{
    int numbers[] = { 123, 121, 45654, 78, 909, 2, 10 };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int largest;

    if (find_largest_palindrome(numbers, count, &largest)) {
        if (printf("Largest palindromic number: %d\n", largest) < 0) {
            return 1;
        }
    } else {
        if (printf("No palindromic number found.\n") < 0) {
            return 1;
        }
    }

    return 0;
}

Analysis notes on the flagged weakness (CWE-119/CWE-120, statically-sized buffer):

- The buffer `buf[32]` is written exclusively via `snprintf` with `sizeof(buf)` as the limit, which guarantees no overflow and null-termination.
- The return value of `snprintf` is checked both for encoding errors (`len < 0`) and truncation (`(size_t)len >= sizeof(buf)`), so the subsequent index arithmetic (`j = len - 1`) always operates within valid, written bounds.
- A 32-byte buffer comfortably exceeds the maximum decimal representation of any `int` (11 characters plus terminator for 32-bit, 20 plus terminator even for 64-bit), so truncation cannot occur in practice; the check is defense-in-depth.
- Negative inputs are rejected before formatting, so no `'-'` sign ever complicates the palindrome comparison.
- `find_largest_palindrome` validates all pointers and the size before use, and `main` checks `printf` return values.
