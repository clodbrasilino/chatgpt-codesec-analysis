#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

int is_number_at_end(const char *str) {
    if (str == NULL || !*str) return 0;
    const char *end = str;
    while (*end) ++end;
    while (end > str && isspace(*(end-1))) --end;
    while (end > str && isdigit(*(end-1))) --end;
    return end != str && !isdigit(*(end-1));
}

int main(void) {
    const char *test_strings[] = {"hello123", "123hello", "hello 123", "hello", ""};
    for (size_t i = 0; i < sizeof(test_strings) / sizeof(test_strings[0]); ++i) {
        printf("'%s': %d\n", test_strings[i], is_number_at_end(test_strings[i]));
    }
    return 0;
}
