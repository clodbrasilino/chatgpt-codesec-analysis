```c
#include <stdio.h>

int isValidTriangle(int a, int b, int c) {
    if (a <= 0 || b <= 0 || c <= 0) {
        return 0;
    }
    if ((a + b > c) && (b + c > a) && (c + a > b)) {
        return 1;
    }
    return 0;
}

int main(void) {
    int a, b, c;
    if (scanf("%d %d %d", &a, &b, &c) != 3) {
        return 1;
    }
    if (isValidTriangle(a, b, c)) {
        printf("Valid\n");
    } else {
        printf("Invalid\n");
    }
    return 0;
}
```
