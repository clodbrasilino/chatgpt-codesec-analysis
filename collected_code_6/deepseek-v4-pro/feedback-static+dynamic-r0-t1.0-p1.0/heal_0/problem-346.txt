#include <stdlib.h>
#include <stdio.h>

unsigned long long entringer(int n, int k) {
    if (n < 0 || k < 0 || k > n) {
        return 0;
    }

    unsigned long long *prev = (unsigned long long *)calloc(n + 2, sizeof(unsigned long long));
    unsigned long long *curr = (unsigned long long *)calloc(n + 2, sizeof(unsigned long long));
    if (prev == NULL || curr == NULL) {
        free(prev);
        free(curr);
        return 0;
    }

    prev[0] = 1;

    for (int i = 1; i <= n; i++) {
        if (i % 2 == 1) {
            curr[0] = 0;
            for (int j = 1; j <= i; j++) {
                curr[j] = curr[j - 1] + prev[j - 1];
            }
        } else {
            curr[i] = 0;
            for (int j = i - 1; j >= 0; j--) {
                curr[j] = curr[j + 1] + prev[j];
            }
        }

        unsigned long long *tmp = prev;
        prev = curr;
        curr = tmp;

        for (int j = 0; j <= n + 1; j++) {
            curr[j] = 0;
        }
    }

    unsigned long long result = prev[k];

    free(prev);
    free(curr);

    return result;
}

int main(void) {
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 1;
    }

    printf("%llu\n", entringer(n, k));

    return 0;
}
