#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long sum_even_at_even_positions(const int *arr, size_t size, int *status)
{
    long long sum = 0;
    size_t i;

    if (arr == NULL || status == NULL) {
        if (status != NULL) {
            *status = -1;
        }
        return 0;
    }

    *status = 0;

    for (i = 0; i < size; i += 2) {
        if (arr[i] % 2 == 0) {
            if ((arr[i] > 0 && sum > LLONG_MAX - arr[i]) ||
                (arr[i] < 0 && sum < LLONG_MIN - arr[i])) {
                *status = -2;
                return 0;
            }
            sum += arr[i];
        }
    }

    return sum;
}

int main(void)
{
    int numbers[] = {2, 5, 4, 7, 6, 9, 8, 11, 10, 3};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int status = 0;
    long long result;

    result = sum_even_at_even_positions(numbers, count, &status);

    if (status != 0) {
        fprintf(stderr, "Error computing sum: code %d\n", status);
        return EXIT_FAILURE;
    }

    if (printf("Sum of even numbers at even positions: %lld\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
