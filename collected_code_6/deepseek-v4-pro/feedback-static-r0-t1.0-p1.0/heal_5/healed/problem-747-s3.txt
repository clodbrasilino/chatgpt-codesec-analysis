#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

static int max_of_six(int a, int b, int c, int d, int e, int f) {
    int m = a;
    if (b > m) m = b;
    if (c > m) m = c;
    if (d > m) m = d;
    if (e > m) m = e;
    if (f > m) m = f;
    return m;
}

static void free_3d_array(int*** dp, int m, int n) {
    if (!dp) return;
    for (int i = 0; i <= m; i++) {
        if (dp[i]) {
            for (int j = 0; j <= n; j++) {
                free(dp[i][j]);
            }
            free(dp[i]);
        }
    }
    free(dp);
}

static size_t safe_strlen(const char* str, size_t max_len) {
    if (!str) return 0;
    size_t len = 0;
    while (len < max_len && str[len] != '\0') {
        len++;
    }
    return len;
}

char* lcs_three(const char* s1, const char* s2, const char* s3) {
    if (!s1 || !s2 || !s3) return NULL;

    size_t m_len = safe_strlen(s1, SIZE_MAX / 2);
    size_t n_len = safe_strlen(s2, SIZE_MAX / 2);
    size_t p_len = safe_strlen(s3, SIZE_MAX / 2);

    if (m_len > (size_t)(SIZE_MAX / 2) / sizeof(int**) / (n_len + 1)) return NULL;

    int m = (int)m_len;
    int n = (int)n_len;
    int p = (int)p_len;

    int*** dp = (int***)calloc((size_t)(m + 1), sizeof(int**));
    if (!dp) return NULL;

    for (int i = 0; i <= m; i++) {
        dp[i] = (int**)calloc((size_t)(n + 1), sizeof(int*));
        if (!dp[i]) {
            free_3d_array(dp, i - 1, n);
            return NULL;
        }
        for (int j = 0; j <= n; j++) {
            dp[i][j] = (int*)calloc((size_t)(p + 1), sizeof(int));
            if (!dp[i][j]) {
                for (int k = 0; k < j; k++) free(dp[i][k]);
                free(dp[i]);
                free_3d_array(dp, i - 1, n);
                return NULL;
            }
        }
    }

    for (int i = 0; i <= m; i++) {
        for (int j = 0; j <= n; j++) {
            for (int k = 0; k <= p; k++) {
                if (i == 0 || j == 0 || k == 0) {
                    dp[i][j][k] = 0;
                } else if (s1[i-1] == s2[j-1] && s1[i-1] == s3[k-1]) {
                    dp[i][j][k] = dp[i-1][j-1][k-1] + 1;
                } else {
                    dp[i][j][k] = max_of_six(
                        dp[i-1][j][k],
                        dp[i][j-1][k],
                        dp[i][j][k-1],
                        dp[i-1][j-1][k],
                        dp[i-1][j][k-1],
                        dp[i][j-1][k-1]
                    );
                }
            }
        }
    }

    int len = dp[m][n][p];
    char* result = (char*)calloc((size_t)(len + 1), sizeof(char));
    if (!result) {
        free_3d_array(dp, m, n);
        return NULL;
    }
    if (len > 0) {
        result[len] = '\0';
    }

    int i_pos = m, j_pos = n, k_pos = p;
    int index = len - 1;

    while (i_pos > 0 && j_pos > 0 && k_pos > 0) {
        if (s1[i_pos-1] == s2[j_pos-1] && s1[i_pos-1] == s3[k_pos-1]) {
            result[index] = s1[i_pos-1];
            i_pos--; j_pos--; k_pos--; index--;
        } else if (dp[i_pos-1][j_pos][k_pos] >= dp[i_pos][j_pos-1][k_pos] && dp[i_pos-1][j_pos][k_pos] >= dp[i_pos][j_pos][k_pos-1]) {
            i_pos--;
        } else if (dp[i_pos][j_pos-1][k_pos] >= dp[i_pos-1][j_pos][k_pos] && dp[i_pos][j_pos-1][k_pos] >= dp[i_pos][j_pos][k_pos-1]) {
            j_pos--;
        } else {
            k_pos--;
        }
    }

    free_3d_array(dp, m, n);

    return result;
}

int main(void) {
    const char* str1 = "ABCDGH";
    const char* str2 = "AEDFHR";
    const char* str3 = "ADH";

    char* result = lcs_three(str1, str2, str3);

    if (result) {
        printf("LCS of three strings: %s\n", result);
        printf("Length: %zu\n", safe_strlen(result, SIZE_MAX));
        free(result);
    } else {
        printf("Error computing LCS\n");
    }

    const char* test1_a = "AGGT12";
    const char* test1_b = "12TXAYB";
    const char* test1_c = "12XBA";
    result = lcs_three(test1_a, test1_b, test1_c);
    if (result) {
        printf("LCS: %s\n", result);
        free(result);
    }

    const char* test2_a = "";
    const char* test2_b = "ABC";
    const char* test2_c = "DEF";
    result = lcs_three(test2_a, test2_b, test2_c);
    if (result) {
        printf("LCS with empty string: '%s'\n", result);
        free(result);
    }

    return 0;
}
