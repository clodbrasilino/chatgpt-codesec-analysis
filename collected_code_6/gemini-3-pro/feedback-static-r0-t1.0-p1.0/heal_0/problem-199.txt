#include <stdio.h>
#include <stdint.h>

uint64_t highest_power_of_2(uint64_t n) {
    if (n == 0) {
        return 0;
    }
    
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    n |= n >> 32;
    
    return n - (n >> 1);
}

int main(void) {
    uint64_t test_values[] = {0, 1, 2, 3, 4, 10, 15, 16, 127, 128, 1000};
    size_t num_tests = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;
    
    for (i = 0; i < num_tests; i++) {
        uint64_t val = test_values[i];
        uint64_t res = highest_power_of_2(val);
        printf("%llu -> %llu\n", (unsigned long long)val, (unsigned long long)res);
    }
    
    return 0;
}
