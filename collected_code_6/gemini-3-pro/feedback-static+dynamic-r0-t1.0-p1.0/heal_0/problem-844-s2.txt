#include <stdio.h>
#include <stdlib.h>

int findKthElement(int* arr, int size, int k, int* result) {
    if (arr == NULL || k < 1 || k > size || result == NULL) {
        return -1;
    }

    int left = 0;
    int right = size - 1;

    while (left < right) {
        if (arr[left] % 2 != 0) {
            left++;
        } else if (arr[right] % 2 == 0) {
            right--;
        } else {
            int temp = arr[left];
            arr[left] = arr[right];
            arr[right] = temp;
            left++;
            right--;
        }
    }

    *result = arr[k - 1];
    return 0;
}

int main() {
    int arr[] = {2, 4, 1, 6, 3, 5, 8, 7};
    int size = sizeof(arr) / sizeof(arr[0]);
    int k = 4;
    int result = 0;

    int status = findKthElement(arr, size, k, &result);

    if (status == 0) {
        printf("%d\n", result);
    } else {
        printf("Error\n");
    }

    return 0;
}
