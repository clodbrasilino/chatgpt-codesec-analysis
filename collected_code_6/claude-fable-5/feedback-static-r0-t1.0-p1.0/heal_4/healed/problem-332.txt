#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

int count_char_frequency(const char *str, size_t str_max, size_t freq_size, unsigned long freq[])
{
    size_t i;

    if (str == NULL || freq == NULL || freq_size < (size_t)UCHAR_MAX + 1) {
        return -1;
    }

    for (i = 0; i < freq_size; i++) {
        freq[i] = 0;
    }

    for (i = 0; i < str_max && str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        if (freq[c] < ULONG_MAX) {
            freq[c]++;
        }
    }

    return 0;
}

        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';

        fprintf(stderr, "Error counting character frequency\n");
        return EXIT_FAILURE;

    for (i = 0; i <= UCHAR_MAX; i++) {
        if (freq[i] > 0) {
            if (i >= 32 && i < 127) {

2. **Guaranteed null termination**: `buffer[sizeof(buffer) - 1] = '\0'` is a defensive measure ensuring the buffer is always null-terminated (though `fgets` already guarantees this on success).

3. **Bounded string operations**: `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, so length computation never reads past the buffer even if termination were somehow missing.

4. **Bounded iteration in the counting function**: `count_char_frequency` receives `str_max` explicitly and its loop condition `i < str_max && str[i] != '\0'` ensures it never reads beyond the buffer.

5. **Robust parameter validation**: The function checks for `NULL` pointers and verifies `freq_size` covers all 256 possible `unsigned char` values before writing to the frequency array, preventing out-of-bounds writes.

6. **Overflow-safe counters**: The `freq[c] < ULONG_MAX` guard prevents unsigned integer wraparound on the counters.
