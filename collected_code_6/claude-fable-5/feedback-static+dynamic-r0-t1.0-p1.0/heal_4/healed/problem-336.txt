#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MONTH_BUF_SIZE 32
#define INPUT_BUF_SIZE 64

static int has_28_days(const char *month)
{
    char normalized[MONTH_BUF_SIZE];
    size_t len;
    size_t i;

    if (month == NULL) {
        return 0;
    }

    len = strnlen(month, sizeof(normalized));
    if (len == 0 || len >= sizeof(normalized)) {
        return 0;
    }

    for (i = 0; i < len; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);
}

int main(void)
{
    char input[INPUT_BUF_SIZE];
    size_t len;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
    }

    if (has_28_days(input)) {
        printf("%s has 28 days.\n", input);
    } else {
        printf("%s does not have exactly 28 days.\n", input);
    }

    return EXIT_SUCCESS;
}

The original code was already well-defended against the weaknesses Flawfinder flagged — those warnings are heuristic hits on `char` arrays and `getchar()`, not confirmed vulnerabilities. Here is why each is mitigated:

1. **`char normalized[MONTH_BUF_SIZE]` (CWE-119/CWE-120):** Protected by `strnlen(month, sizeof(normalized))` with an explicit rejection when `len >= sizeof(normalized)`, guaranteeing room for the NUL terminator. The copy loop is bounded by `len`, so no overflow is possible.

2. **`char input[INPUT_BUF_SIZE]` (CWE-119/CWE-120):** `fgets` is given exactly `sizeof(input)`, which never writes beyond the buffer. The explicit `input[sizeof(input) - 1] = '\0'` is defensive redundancy (fgets already NUL-terminates), and `strnlen` is used instead of `strlen` for bounded length computation.

3. **`getchar()` in a loop (CWE-120/CWE-20):** The loop writes nothing to any buffer — it only discards excess characters until newline or `EOF`, and correctly checks `EOF` to avoid an infinite loop on closed input streams.

Additional good practices already present: `tolower((unsigned char)...)` avoids undefined behavior for negative `char` values, `fflush` return value is checked, and `fgets` failure is handled. No changes to logic were required; the code is a correct, bounded, overflow-free implementation.
