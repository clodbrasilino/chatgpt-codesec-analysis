#include <stdio.h>
#include <stdlib.h>

int sum_positive(const int *numbers, size_t count) {
    int sum = 0;
    if (numbers == NULL) {
        return 0;
    }
    for (size_t i = 0; i < count; i++) {
        if (numbers[i] > 0) {
            sum += numbers[i];
        }
    }
    return sum;
}

int main(void) {
    int data[] = {1, -2, 3, -4, 5, 0, -6, 7};
    size_t count = sizeof(data) / sizeof(data[0]);
    
    int result = sum_positive(data, count);
    
    printf("%d\n", result);
    
    return 0;
}
