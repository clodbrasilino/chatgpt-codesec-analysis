#include <stdio.h>
#include <stdbool.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

int min(int a, int b) {
    return (a < b) ? a : b;
}

int getMedian(int arr1[], int arr2[], int n) {
    if (n <= 0) return -1;
    if (n == 1) return (arr1[0] + arr2[0]) / 2;
    if (n == 2) return (max(arr1[0], arr2[0]) + min(arr1[1], arr2[1])) / 2;

    int m1 = arr1[n/2-1];
    int m2 = arr2[n/2-1];

    if (m1 == m2) return m1;

    if (m1 < m2) {
        if (n % 2 == 0)
            return getMedian(arr1 + n/2 - 1, arr2, n - n/2 + 1);
        else
            return getMedian(arr1 + n/2, arr2, n - n/2);
    } else {
        if (n % 2 == 0)
            return getMedian(arr1, arr2 + n/2 - 1, n - n/2 + 1);
        else
            return getMedian(arr1, arr2 + n/2, n - n/2);
    }
}

int main() {
    int arr1[] = {1, 3, 5, 7, 9};
    int arr2[] = {2, 4, 6, 8, 10};
    int n = sizeof(arr1) / sizeof(arr1[0]);
    int median = getMedian(arr1, arr2, n);
    printf("Median: %d\n", median);
    return 0;
}
