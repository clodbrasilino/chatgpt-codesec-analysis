#include <stdio.h>
#include <stdlib.h>
#include <regex.h>
#include <limits.h>

void extract_date_from_url(const char *url, int *year, int *month, int *day) {
    regex_t regex;
    regmatch_t match[4];
    const char *pattern = "([0-9]{4})/([0-9]{2})/([0-9]{2})";

    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        fprintf(stderr, "Could not compile regex\n");
        exit(EXIT_FAILURE);
    }

    if (regexec(&regex, url, 4, match, 0) == 0) {
        long y = strtol(url + match[1].rm_so, NULL, 10);
        long m = strtol(url + match[2].rm_so, NULL, 10);
        long d = strtol(url + match[3].rm_so, NULL, 10);

        if (y >= INT_MIN && y <= INT_MAX && m >= 1 && m <= 12 && d >= 1 && d <= 31) {
            *year = y;
            *month = m;
            *day = d;
        } else {
            *year = -1;
            *month = -1;
            *day = -1;
        }
    } else {
        *year = -1;
        *month = -1;
        *day = -1;
    }

    regfree(&regex);
}

int main() {
    const char *url = "https://example.com/2023/10/05/some-article";
    int year, month, day;

    extract_date_from_url(url, &year, &month, &day);

    if (year != -1 && month != -1 && day != -1) {
        printf("Year: %d, Month: %d, Day: %d\n", year, month, day);
    } else {
        printf("Date not found in URL\n");
    }

    return 0;
}
