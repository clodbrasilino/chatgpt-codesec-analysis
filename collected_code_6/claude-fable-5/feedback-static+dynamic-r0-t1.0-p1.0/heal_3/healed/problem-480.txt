```c
#include <stdio.h>
#include <limits.h>
#include <stddef.h>
#include <string.h>

#define INPUT_SIZE 256

int find_max_occurring_char(const char *str, char *result)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    unsigned char max_char = 0;
    int found = 0;

    if (str == NULL || result == NULL) {
        return -1;
    }

    for (i = 0; str[i] != '\0'; i++) {
        unsigned char c = (unsigned char)str[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
            max_char = c;
            found = 1;
        }
    }

    if (found == 0) {
        return -1;
    }

    *result = (char)max_char;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    char max_char;
    size_t len;

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_max_occurring_char(input, &max_char) != 0) {
        fprintf(stderr, "Empty string or invalid input\n");
        return 1;
    }

    if (printf("Maximum occurring character: %c\n", max_char) < 0) {
        return 1;
    }

    return 0;
}
```

**Analysis notes:**

The flawfinder warning about the statically-sized `char input[INPUT_SIZE]` buffer (CWE-119/CWE-120) is a false positive in this code, and the existing mitigations remain in place:

1. **Bounded read:** `fgets(input, (int)sizeof(input), stdin)` limits input to at most `INPUT_SIZE - 1` characters plus the null terminator, so no overflow is possible.
2. **Defensive null termination:** `input[sizeof(input) - 1] = '\0'` guarantees termination even though `fgets` already terminates on success.
3. **Bounded length calculation:** `strnlen(input, sizeof(input))` is used instead of `strlen`, preventing reads past the buffer even if termination were somehow missing.
4. **Safe counting array:** `counts[UCHAR_MAX + 1]` indexed by `unsigned char` guarantees the index can never exceed `UCHAR_MAX`, avoiding out-of-bounds access (a common bug when plain `char` is signed and used as an array index).
5. **Null-pointer checks and error propagation:** `find_max_occurring_char` validates its arguments and signals failure via return code; `main` checks the return values of `fgets` and `printf`.
