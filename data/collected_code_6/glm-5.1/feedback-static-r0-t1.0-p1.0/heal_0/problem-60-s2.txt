#include <stdio.h>
#include <stdlib.h>

int max_length_subsequence(int *arr, int n, int k) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    int *dp = (int *)malloc(n * sizeof(int));
    if (dp == NULL) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        dp[i] = 1;
    }

    int max_len = 1;

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (abs(arr[i] - arr[j]) <= k) {
                if (dp[j] + 1 > dp[i]) {
                    dp[i] = dp[j] + 1;
                }
            }
        }
        if (dp[i] > max_len) {
            max_len = dp[i];
        }
    }

    free(dp);

    return max_len;
}

int main(void) {
    int arr[] = {1, 4, 2, 6, 3, 8, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 2;

    int result = max_length_subsequence(arr, n, k);

    printf("%d\n", result);

    return 0;
}
