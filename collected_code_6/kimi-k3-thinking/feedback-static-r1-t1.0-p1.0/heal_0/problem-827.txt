#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

bool sum_column(int const *const *matrix, size_t rows, size_t cols, size_t column, long *result)
{
    size_t i;
    long sum;
    int value;

    if (matrix == NULL || result == NULL || cols == 0 || column >= cols)
    {
        return false;
    }

    sum = 0;
    for (i = 0; i < rows; i++)
    {
        if (matrix[i] == NULL)
        {
            return false;
        }
        value = matrix[i][column];
        if ((value > 0 && sum > LONG_MAX - value) || (value < 0 && sum < LONG_MIN - value))
        {
            return false;
        }
        sum += value;
    }

    *result = sum;
    return true;
}

int main(void)
{
    size_t rows = 3;
    size_t cols = 4;
    size_t column = 2;
    int const **matrix;
    long result;
    size_t i;
    size_t j;
    int *row;
    bool ok;

    matrix = malloc(rows * sizeof(*matrix));
    if (matrix == NULL)
    {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < rows; i++)
    {
        matrix[i] = NULL;
    }

    ok = true;
    for (i = 0; i < rows && ok; i++)
    {
        row = malloc(cols * sizeof(*row));
        if (row == NULL)
        {
            fprintf(stderr, "Memory allocation failed\n");
            ok = false;
        }
        else
        {
            for (j = 0; j < cols; j++)
            {
                row[j] = (int)(i * cols + j) + 1;
            }
            matrix[i] = row;
        }
    }

    if (ok)
    {
        if (sum_column(matrix, rows, cols, column, &result))
        {
            printf("Sum of column %zu: %ld\n", column, result);
        }
        else
        {
            fprintf(stderr, "Unable to sum column %zu\n", column);
            ok = false;
        }
    }

    for (i = 0; i < rows; i++)
    {
        free((void *)matrix[i]);
    }
    free(matrix);

    return ok ? EXIT_SUCCESS : EXIT_FAILURE;
}
