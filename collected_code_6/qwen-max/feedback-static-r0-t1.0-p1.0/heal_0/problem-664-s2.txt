```c
#include <stdio.h>

double average_of_evens(int n) {
    if (n <= 0 || n % 2 != 0) return 0.0;

    int sum = 0;
    int count = 0;
    for (int i = 2; i <= n; i += 2) {
        sum += i;
        count++;
    }
    return (double)sum / count;
}

int main() {
    int n;
    printf("Enter an even number: ");
    scanf("%d", &n);
    double avg = average_of_evens(n);
    printf("The average of even numbers up to %d is: %.2f\n", n, avg);
    return 0;
}
```
