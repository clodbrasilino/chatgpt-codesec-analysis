#include <stdio.h>
#include <string.h>

#define BUFFER_SIZE 256

int find_first_repeated(const char *str, size_t max_len, char *result)
{
    int seen[256] = {0};
    size_t i;
    size_t len;

    if (str == NULL || result == NULL || max_len == 0) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == max_len) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)str[i];
        if (seen[c]) {
            *result = str[i];
            return 0;
        }
        seen[c] = 1;
    }

    return 1;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char repeated;
    int status;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    status = find_first_repeated(buffer, sizeof(buffer), &repeated);
    if (status == 0) {
        printf("First repeated character: %c\n", repeated);
    } else if (status == 1) {
        printf("No repeated character found\n");
    } else {
        fprintf(stderr, "Invalid input\n");
        return 1;
    }

    return 0;
}

**Analysis summary:**

The original code already implements the appropriate mitigations for the CWE-119/CWE-120 concern flagged by Flawfinder regarding the statically-sized `buffer` array:

1. **Bounded input reading** — `fgets(buffer, sizeof(buffer), stdin)` limits input to the buffer size, preventing overflow. It never writes more than `BUFFER_SIZE - 1` characters plus a null terminator.

2. **Guaranteed null termination** — `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders measure (fgets already null-terminates, but this protects against future refactoring).

3. **Bounded length calculation** — `strnlen()` is used instead of `strlen()`, so even a non-terminated buffer cannot cause an out-of-bounds read.

4. **Defensive API design** — `find_first_repeated()` validates all pointers, rejects `max_len == 0`, treats a string filling the entire buffer (no terminator found within `max_len`) as invalid, and casts to `unsigned char` before indexing `seen[]`, avoiding undefined behavior with negative `char` values on platforms where `char` is signed.

5. **Safe newline stripping** — the `len > 0` check prevents an out-of-bounds access at `buffer[-1]` on empty input.
