```c
#include <stdlib.h>
#include <string.h>

typedef struct {
    int a;
    int b;
} Tuple;

void clear_tuples(Tuple *tuples, size_t count) {
    if (tuples == NULL) {
        return;
    }
    memset(tuples, 0, count * sizeof(Tuple));
}

int main(void) {
    Tuple *t = malloc(3 * sizeof(Tuple));
    if (t == NULL) {
        return 1;
    }
    
    for (size_t i = 0; i < 3; i++) {
        t[i].a = i + 1;
        t[i].b = (i + 1) * 10;
    }
    
    clear_tuples(t, 3);
    
    free(t);
    return 0;
}
```
