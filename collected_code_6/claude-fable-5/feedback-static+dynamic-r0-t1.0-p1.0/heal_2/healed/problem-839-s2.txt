```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, MAX_NAME_LEN);
    if (len >= MAX_NAME_LEN) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, MAX_NAME_LEN);
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;
    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis of the fixes applied:**

1. **`strlen` (CWE-126, over-read on non-terminated strings):** Replaced with `strnlen(name, MAX_NAME_LEN)`, which bounds the scan to `MAX_NAME_LEN` bytes. Even if the input is not NUL-terminated, the read cannot exceed the buffer limit, and the subsequent `len >= MAX_NAME_LEN` check rejects such inputs safely.

2. **`memcpy` (CWE-120, unchecked destination size):** The copy remains safe because `len` is strictly validated (`len >= MAX_NAME_LEN` returns `-1`) before the copy, guaranteeing `len <= MAX_NAME_LEN - 1` bytes are copied plus one byte for the explicit NUL terminator — always within the 64-byte destination.

3. **Statically-sized `char first[MAX_NAME_LEN]` (CWE-119/CWE-120):** All accesses are bounds-checked: writes go through the validated `set_tuple`, comparisons use `strncmp(..., MAX_NAME_LEN)`, and printing uses the `%.*s` precision specifier so `printf` never reads past the buffer even if the terminator were missing.

4. **General robustness:** NULL-pointer checks in `set_tuple`, `sort_tuples`, and `print_tuples` are retained, return codes are checked in `main`, and `printf`/`fprintf` results are explicitly cast to `(void)` to signal intentional discard.

Note: `strnlen` is POSIX.1-2008 / C23; compile with `-D_POSIX_C_SOURCE=200809L` (or equivalent) if targeting strict C99/C11 environments.
