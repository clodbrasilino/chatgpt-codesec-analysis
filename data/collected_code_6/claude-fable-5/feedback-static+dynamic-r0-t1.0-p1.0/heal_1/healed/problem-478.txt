```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test1 = "HeLLoWoRLd";
    const char *test2 = "abcDEFghi123";
    const char *test3 = "alllowercase";
    char *output;

    output = remove_lowercase(test1);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test1, output);
    free(output);

    output = remove_lowercase(test2);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test2, output);
    free(output);

    output = remove_lowercase(test3);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test3, output);
    free(output);

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder CWE-126 warning: `strlen` was replaced with `strnlen` bounded by `MAX_INPUT_LEN`, which guarantees the scan stops after a fixed maximum even if the input buffer is not null-terminated, preventing a potential buffer over-read. The rest of the logic was already safe: the `malloc` result is checked, the `islower` argument is correctly cast to `unsigned char` to avoid undefined behavior with negative `char` values, the output buffer is always null-terminated, and every allocation in `main` is freed with its return value checked.
