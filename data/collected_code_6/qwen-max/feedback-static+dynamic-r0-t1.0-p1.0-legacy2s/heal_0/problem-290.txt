#include <stdio.h>
#include <stdlib.h>

typedef struct List {
    int *data;
    size_t length;
} List;

List* findMaxLists(List lists[], size_t count) {
    if (count == 0) return NULL;

    size_t max_length = 0;
    for (size_t i = 0; i < count; ++i) {
        if (lists[i].length > max_length) {
            max_length = lists[i].length;
        }
    }

    size_t max_count = 0;
    for (size_t i = 0; i < count; ++i) {
        if (lists[i].length == max_length) {
            ++max_count;
        }
    }

    List *result = (List *)malloc(max_count * sizeof(List));
    if (result == NULL) {
        exit(EXIT_FAILURE);
    }

    size_t result_index = 0;
    for (size_t i = 0; i < count; ++i) {
        if (lists[i].length == max_length) {
            result[result_index++] = lists[i];
        }
    }

    return result;
}

int main() {
    List lists[] = {
        { .data = (int []){1, 2, 3}, .length = 3 },
        { .data = (int []){4, 5}, .length = 2 },
        { .data = (int []){6, 7, 8, 9}, .length = 4 },
        { .data = (int []){10, 11, 12, 13, 14}, .length = 5 },
        { .data = (int []){15, 16, 17, 18, 19, 20}, .length = 6 },
        { .data = (int []){21, 22, 23, 24, 25, 26, 27}, .length = 7 },
        { .data = (int []){28, 29, 30, 31, 32, 33, 34}, .length = 7 },
    };

    size_t count = sizeof(lists) / sizeof(lists[0]);

    List *max_lists = findMaxLists(lists, count);

    for (size_t i = 0; i < 2; ++i) {
        if (max_lists[i].length > 0) {
            for (size_t j = 0; j < max_lists[i].length; ++j) {
                printf("%d ", max_lists[i].data[j]);
            }
            printf("\n");
        }
    }

    free(max_lists);
    return 0;
}
