#include <stdio.h>
#include <stdlib.h>

int findSmallestMissing(const int arr[], int n) {
    int i;
    int *temp;
    int result;

    if (arr == NULL || n <= 0) {
        return 1;
    }

    temp = (int *)calloc(n, sizeof(int));
    if (temp == NULL) {
        return -1;
    }

    for (i = 0; i < n; i++) {
        if (arr[i] > 0 && arr[i] <= n) {
            temp[arr[i] - 1] = 1;
        }
    }

    result = n + 1;
    for (i = 0; i < n; i++) {
        if (temp[i] == 0) {
            result = i + 1;
            break;
        }
    }

    free(temp);
    temp = NULL;

    return result;
}

int main(void) {
    int arr1[] = {2, 3, 7, 6, 8, -1, -10, 15};
    int arr2[] = {2, 3, -7, 6, 8, 1, -10, 15};
    int arr3[] = {1, 1, 0, -1, -2};
    int n1 = sizeof(arr1) / sizeof(arr1[0]);
    int n2 = sizeof(arr2) / sizeof(arr2[0]);
    int n3 = sizeof(arr3) / sizeof(arr3[0]);
    int result;

    result = findSmallestMissing(arr1, n1);
    if (result != -1) {
        printf("Smallest missing number: %d\n", result);
    }

    result = findSmallestMissing(arr2, n2);
    if (result != -1) {
        printf("Smallest missing number: %d\n", result);
    }

    result = findSmallestMissing(arr3, n3);
    if (result != -1) {
        printf("Smallest missing number: %d\n", result);
    }

    return 0;
}
