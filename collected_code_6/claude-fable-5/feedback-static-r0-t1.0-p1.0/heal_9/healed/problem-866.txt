```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_MONTH_LEN 32

static int has_31_days(const char *month)
{
    static const char *months_31[] = {
        "january", "march", "may", "july",
        "august", "october", "december"
    };
    char normalized[MAX_MONTH_LEN];
    size_t i;
    size_t len;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_MONTH_LEN);
    if (len == 0 || len >= MAX_MONTH_LEN) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        if (!isalpha((unsigned char)month[i])) {
            return -1;
        }
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    for (i = 0; i < sizeof(months_31) / sizeof(months_31[0]); i++) {
        if (strcmp(normalized, months_31[i]) == 0) {
            return 1;
        }
    }

    return 0;
}

int main(void)
{
    char input[MAX_MONTH_LEN];
    int result;

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    if (strchr(input, '\n') == NULL && !feof(stdin)) {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
            ;
        }
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\r\n")] = '\0';

    result = has_31_days(input);

    if (result == -1) {
        fprintf(stderr, "Invalid month name provided\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s has 31 days\n", input);
    } else {
        printf("%s does not have 31 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened, and the Flawfinder hits are informational rather than actual vulnerabilities:

1. **`char normalized[MAX_MONTH_LEN]` (CWE-119/CWE-120)** — Mitigated. `strnlen` bounds the length check to `MAX_MONTH_LEN`, and the `len >= MAX_MONTH_LEN` guard guarantees `normalized[len] = '\0'` never writes out of bounds. The copy loop is strictly bounded by `len`.

2. **`char input[MAX_MONTH_LEN]` (CWE-119/CWE-120)** — Mitigated. `fgets` with `sizeof(input)` cannot overflow the buffer, and truncated input is detected (no `\n` present and not at EOF), reported as an error, and the remainder of the line is drained.

3. **`getchar` in a loop (CWE-120/CWE-20)** — Not a real issue. The loop only discards characters and writes nothing to any buffer; it correctly terminates on `\n` or `EOF`.

Other correct practices already present: `NULL` check before dereference, casting to `unsigned char` before `isalpha`/`tolower` (avoids UB on negative `char` values), `strcspn` to strip both `\r` and `\n`, checked return values for `fgets` and `fflush`, and a distinct error path (`-1`) for invalid input. No changes were required beyond removing the comments as requested.
