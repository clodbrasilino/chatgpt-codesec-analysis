#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

void reverse_string(char *str)
{
    size_t len;
    size_t i;

    if (str == NULL) {
        return;
    }

    len = strnlen(str, MAX_STRING_LENGTH);
    if (len < 2) {
        return;
    }

    for (i = 0; i < len / 2; i++) {
        char temp = str[i];
        str[i] = str[len - 1 - i];
        str[len - 1 - i] = temp;
    }
}

int reverse_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return -1;
    }

    for (i = 0; i < count; i++) {
        if (list[i] == NULL) {
            return -1;
        }
        reverse_string(list[i]);
    }

    return 0;
}

static void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char *duplicate_string(const char *src)
{
    size_t len;
    size_t dest_size;
    char *dest;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LENGTH);
    if (len >= MAX_STRING_LENGTH) {
        return NULL;
    }

    dest_size = len + 1;
    if (dest_size < len) {
        return NULL;
    }

    dest = malloc(dest_size);
    if (dest == NULL) {
        return NULL;
    }

    if (len > 0 && len <= dest_size - 1) {
        memcpy(dest, src, len);
    }
    dest[len] = '\0';

    return dest;
}

int main(void)
{
    const char *source[] = { "hello", "world", "example", "C" };
    size_t count = sizeof(source) / sizeof(source[0]);
    char **list;
    size_t i;

    list = calloc(count, sizeof(char *));
    if (list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        list[i] = duplicate_string(source[i]);
        if (list[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_list(list, count);
            return EXIT_FAILURE;
        }
    }

    if (reverse_string_list(list, count) != 0) {
        fprintf(stderr, "Failed to reverse string list\n");
        free_list(list, count);
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("%s\n", list[i]) < 0) {
            fprintf(stderr, "Output error\n");
            free_list(list, count);
            return EXIT_FAILURE;
        }
    }

    free_list(list, count);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The flagged weakness (CWE-120, unchecked `memcpy` in `duplicate_string`) was already effectively mitigated in the original code: `len` is bounded by `strnlen(src, MAX_STRING_LENGTH)`, the `len >= MAX_STRING_LENGTH` guard rejects unterminated or oversized input, and `dest` is allocated with exactly `len + 1` bytes, so the copy of `len` bytes plus the null terminator can never overflow the destination.

The rewrite hardens it further by:

1. **Explicit overflow check** on `dest_size = len + 1` — defensive against integer wraparound if `MAX_STRING_LENGTH` were ever raised near `SIZE_MAX`.
2. **Explicit bounds verification** (`len <= dest_size - 1`) immediately before the `memcpy`, making the destination-capacity guarantee locally provable to static analyzers rather than relying on reasoning across statements.
3. **Skipping the copy entirely for zero-length input**, avoiding a technically-legal-but-pointless `memcpy` of 0 bytes.

The rest of the program was already sound: null checks throughout, `calloc` for pointer-array initialization (safe with `free_list` on partial failure since unset slots are `NULL` and `free(NULL)` is a no-op), correct cleanup on every error path, and checked `printf` return values.
