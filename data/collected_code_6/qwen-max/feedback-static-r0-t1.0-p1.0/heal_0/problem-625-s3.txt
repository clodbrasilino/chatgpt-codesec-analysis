#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        return NULL;
    }
    newNode->data = data;
    newNode->next = NULL;
    return newNode;
}

void swapFirstLast(Node** head) {
    if (*head == NULL || (*head)->next == NULL) {
        return;
    }

    Node *first = *head, *lastPrev = *head, *last = *head;
    while (last->next != NULL) {
        lastPrev = last;
        last = last->next;
    }

    int temp = first->data;
    first->data = last->data;
    last->data = temp;

    if (first == lastPrev) {
        lastPrev->next = first;
        *head = last;
    } else {
        lastPrev->next = first;
        first->next = last->next;
        last->next = (*head)->next;
        *head = last;
    }
}

void printList(Node* node) {
    while (node != NULL) {
        printf("%d -> ", node->data);
        node = node->next;
    }
    printf("NULL\n");
}

int main() {
    Node* head = createNode(1);
    head->next = createNode(2);
    head->next->next = createNode(3);
    head->next->next->next = createNode(4);

    printf("Original List: ");
    printList(head);

    swapFirstLast(&head);

    printf("List after swapping first and last elements: ");
    printList(head);

    return 0;
}
