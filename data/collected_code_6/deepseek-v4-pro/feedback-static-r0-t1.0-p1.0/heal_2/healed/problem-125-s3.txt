#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int maxDifference(const char *str) {
    if (str == NULL) return 0;
    
    int len = 0;
    while (str[len] != '\0') {
        if (len >= 1000) return -1;
        len++;
    }
    
    if (len == 0) return 0;
    
    int current_sum = 0;
    int max_sum = -1;
    int zeros = 0;
    int ones = 0;
    
    for (int i = 0; i < len; i++) {
        if (str[i] == '0') {
            current_sum++;
            zeros++;
        } else if (str[i] == '1') {
            current_sum--;
            ones++;
        } else {
            return -1;
        }
        
        if (current_sum > max_sum) {
            max_sum = current_sum;
        }
        
        if (current_sum < 0) {
            current_sum = 0;
        }
    }
    
    if (ones == len && zeros == 0) {
        return -1;
    }
    
    if (ones < len) {
        return max_sum;
    }
    
    return -1;
}

int main() {
    char *input = NULL;
    size_t bufsize = 0;
    ssize_t characters;
    
    printf("Enter a binary string: ");
    characters = getline(&input, &bufsize, stdin);
    
    if (characters == -1) {
        free(input);
        return 1;
    }
    
    if (characters > 0 && input[characters - 1] == '\n') {
        input[characters - 1] = '\0';
        characters--;
    }
    
    if (characters > 1000) {
        printf("Invalid input or no sub-string with positive difference\n");
        free(input);
        return 0;
    }
    
    if (characters == 0) {
        printf("Maximum difference: 0\n");
        free(input);
        return 0;
    }
    
    int result = maxDifference(input);
    
    if (result == -1) {
        printf("Invalid input or no sub-string with positive difference\n");
    } else {
        printf("Maximum difference: %d\n", result);
    }
    
    free(input);
    return 0;
}
