#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i || j >= len || i >= len) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and the fix applied:**

The Flawfinder warning flagged the statically-sized `char digits[DIGITS_BUF_SIZE]` buffer (CWE-119/CWE-120). In practice, the original code already mitigated this well:

1. **Bounded write:** `snprintf(digits, sizeof(digits), ...)` limits output to the buffer size and null-terminates. A 64-bit `long long` maximum (`9223372036854775807`) needs 20 characters plus the terminator, so a 32-byte buffer is sufficient.
2. **Truncation detection:** The check `len < 0 || (size_t)len >= sizeof(digits)` correctly rejects encoding errors and truncated output, so all subsequent indexing is confined within initialized, in-bounds memory.
3. **Overflow-safe reconstruction:** The `result > (LLONG_MAX - d) / 10` guard prevents signed integer overflow when rebuilding the number (e.g., permutations of `LLONG_MAX` digits that exceed the type's range).

The rewrite hardens the index arithmetic slightly by adding explicit `i >= len` / `j >= len` guards before the swap, making the bounds invariants locally verifiable rather than relying solely on loop reasoning — a defense-in-depth measure against future refactoring errors. The negative-input rejection, `memset` initialization, and digit-validation loop (`d < 0 || d > 9`) are retained, ensuring no out-of-bounds read or write path exists in the function.
