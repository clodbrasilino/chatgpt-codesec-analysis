#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define BUFFER_SIZE 256

int contains_interior_z(const char *word, size_t len);
int find_word_with_interior_z(const char *text, char *result, size_t result_size);

int contains_interior_z(const char *word, size_t len)
{
    size_t i;

    if (word == NULL || len < 3) {
        return 0;
    }

    for (i = 1; i < len - 1; i++) {
        if (word[i] == 'z' || word[i] == 'Z') {
            return 1;
        }
    }

    return 0;
}

int find_word_with_interior_z(const char *text, char *result, size_t result_size)
{
    size_t i = 0;

    if (text == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    result[0] = '\0';

    while (text[i] != '\0') {
        size_t start;
        size_t len;

        while (text[i] != '\0' && !isalpha((unsigned char)text[i])) {
            i++;
        }

        start = i;

        while (text[i] != '\0' && isalpha((unsigned char)text[i])) {
            i++;
        }

        len = i - start;

        if (len > 0 && contains_interior_z(text + start, len)) {
            if (len >= result_size) {
                return -1;
            }

            memcpy(result, text + start, len);
            result[len] = '\0';
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    const char *tests[] = {
        "the lazy dog jumped",
        "zebra runs fast",
        "jazz music is nice",
        "no matching words here",
        "amazing puzzle pieces"
    };
    size_t num_tests = sizeof(tests) / sizeof(tests[0]);
    char buffer[BUFFER_SIZE];
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < num_tests; i++) {
        int status = find_word_with_interior_z(tests[i], buffer, sizeof(buffer));
        if (status == 0) {
            printf("Input: \"%s\" -> Match: \"%s\"\n", tests[i], buffer);
        } else if (status == 1) {
            printf("Input: \"%s\" -> No match found\n", tests[i]);
        } else {
            fprintf(stderr, "Error processing input: \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `memcpy` warning (CWE-120) in `find_word_with_interior_z`:**
This is a false positive. The code performs an explicit bounds check before the copy: `if (len >= result_size) { return -1; }`. This guarantees that `len` bytes plus the null terminator (`len + 1` total) always fit within `result_size`. The `memcpy` can never overflow the destination buffer.

**2. Flawfinder `char buffer[BUFFER_SIZE]` warning (CWE-119/CWE-120) in `main`:**
Also a false positive. The statically-sized buffer is only ever written to through `find_word_with_interior_z`, which receives the correct size via `sizeof(buffer)` and enforces it internally. The `memset` initialization also uses `sizeof(buffer)`, so no size mismatch is possible.

Additional defensive qualities already present in the code:

- **NULL and zero-size guards:** Both functions validate their pointer arguments and reject a zero-sized result buffer before any writes occur.
- **Correct `isalpha` usage:** Characters are cast to `unsigned char` before being passed to `isalpha`, avoiding undefined behavior with negative `char` values on platforms where `char` is signed.
- **Underflow protection:** `contains_interior_z` checks `len < 3` before computing `len - 1`, preventing a `size_t` wraparound if `len` were 0.
- **Guaranteed termination:** `result[len] = '\0'` only executes after the bounds check, so the terminator write is also in-bounds.
