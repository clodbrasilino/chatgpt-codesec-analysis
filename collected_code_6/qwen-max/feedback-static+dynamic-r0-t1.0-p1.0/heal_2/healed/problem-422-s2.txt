```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>

double average_of_cubes(int n) {
    if (n <= 0) return 0.0;

    int64_t sum = 0;
    for (int i = 1; i <= n; ++i) {
        int64_t cube = (int64_t)i * (int64_t)i * (int64_t)i;
        if (__builtin_add_overflow(sum, cube, &sum)) {
            fprintf(stderr, "Integer overflow detected\n");
            exit(EXIT_FAILURE);
        }
    }
    return (double)sum / n;
}

int main() {
    int n;
    printf("Enter the value of n: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    double result = average_of_cubes(n);
    printf("The average of cubes of first %d natural numbers is: %.2f\n", n, result);

    return EXIT_SUCCESS;
}
```
