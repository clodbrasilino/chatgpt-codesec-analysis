```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

#define MAX_INPUT_SIZE 256U

static int find_min_rotations(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    size_t doubled_size;
    char *doubled;
    int result;

    if (str == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == 0U || len == max_len) {
        return -1;
    }

    if (len > (SIZE_MAX - 1U) / 2U) {
        return -1;
    }

    if (len > (size_t)INT_MAX) {
        return -1;
    }

    doubled_size = (2U * len) + 1U;

    doubled = calloc(doubled_size, sizeof(char));
    if (doubled == NULL) {
        return -1;
    }

    if (len > doubled_size - 1U) {
        free(doubled);
        return -1;
    }
    memcpy(doubled, str, len);

    if (len > doubled_size - len - 1U) {
        free(doubled);
        return -1;
    }
    memcpy(doubled + len, str, len);

    doubled[2U * len] = '\0';

    result = -1;
    for (i = 1U; i <= len; i++) {
        if (strncmp(doubled + i, str, len) == 0) {
            result = (int)i;
            break;
        }
    }

    free(doubled);
    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    size_t len;
    int rotations;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
        len--;
    }

    if (len == 0U) {
        fprintf(stderr, "Empty string provided\n");
        return EXIT_FAILURE;
    }

    rotations = find_min_rotations(buffer, sizeof(buffer));
    if (rotations < 0) {
        fprintf(stderr, "Failed to compute rotations\n");
        return EXIT_FAILURE;
    }

    if (printf("Minimum rotations required: %d\n", rotations) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and applied fixes:**

1. **`memcpy` (CWE-120, both occurrences):** The original code was already functionally safe because `doubled` is allocated with `doubled_size = 2*len + 1` bytes, guaranteeing room for both copies plus the NUL terminator. To satisfy the static-analysis finding, I added explicit bounds checks before each `memcpy` that verify the destination region can hold `len` bytes (`len <= doubled_size - 1` for the first copy, `len <= doubled_size - len - 1` for the second). These checks are provably always true given the allocation, but they make the safety invariant explicit and defend against future refactoring errors.

2. **Statically-sized `char buffer[MAX_INPUT_SIZE]` (CWE-119/CWE-120):** The original code already mitigated this correctly: `fgets` is bounded by `sizeof(buffer)`, a forced NUL terminator is written at the last index, `strnlen` is used rather than `strlen`, and the length is validated before use. No change was required beyond keeping those mitigations intact.

3. **Pre-existing sound defenses retained:** NULL checks, `strnlen` with an upper bound, overflow guards against `SIZE_MAX` and `INT_MAX` before the `2*len + 1` computation, `calloc` failure handling, and freeing the buffer on every exit path.

The code has no memory leaks, no unbounded string operations, and all arithmetic on `len` is guarded against overflow before allocation.
