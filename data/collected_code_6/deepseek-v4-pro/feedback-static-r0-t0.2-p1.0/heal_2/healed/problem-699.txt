#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stddef.h>

static size_t safe_strlen(const char *s, size_t maxlen) {
    size_t len = 0;
    while (len < maxlen && s[len] != '\0') {
        len++;
    }
    return len;
}

int min_swaps_to_convert(const char *s1, const char *s2) {
    if (s1 == NULL || s2 == NULL) {
        return -1;
    }

    size_t len1 = safe_strlen(s1, 1001);
    size_t len2 = safe_strlen(s2, 1001);

    if (len1 != len2 || len1 == 0 || len1 >= 1001) {
        return -1;
    }

    int count1 = 0, count2 = 0;
    int mismatch_positions = 0;
    int ones_in_mismatch = 0;

    for (size_t i = 0; i < len1; i++) {
        if (s1[i] != '0' && s1[i] != '1') {
            return -1;
        }
        if (s2[i] != '0' && s2[i] != '1') {
            return -1;
        }

        if (s1[i] == '1') count1++;
        if (s2[i] == '1') count2++;

        if (s1[i] != s2[i]) {
            mismatch_positions++;
            if (s1[i] == '1') {
                ones_in_mismatch++;
            }
        }
    }

    if (count1 != count2) {
        return -1;
    }

    return ones_in_mismatch;
}

int main(void) {

    char *str2 = NULL;

    size_t len1, len2;

        fprintf(stderr, "Memory allocation failed\n");
        free(str1);
        free(str2);
        return 1;
    }

    printf("Enter first binary string: ");
    if (fgets(str1, (int)bufsize, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        free(str1);
        free(str2);
        return 1;
    }
    len1 = safe_strlen(str1, bufsize);
    if (len1 > 0 && str1[len1 - 1] == '\n') {
        str1[len1 - 1] = '\0';
        len1--;
    }

    printf("Enter second binary string: ");
    if (fgets(str2, (int)bufsize, stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");

        return 1;

    if (len1 >= 1001 || len2 >= 1001) {
        fprintf(stderr, "Input exceeds maximum allowed length\n");
        free(str1);
        free(str2);
