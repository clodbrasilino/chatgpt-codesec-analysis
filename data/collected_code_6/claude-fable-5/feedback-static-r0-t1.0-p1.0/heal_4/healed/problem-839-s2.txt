```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 64

typedef struct {
    char first[MAX_NAME_LEN];
    int second;
} Tuple;

static int set_tuple(Tuple *t, const char *name, int value)
{
    size_t len;

    if (t == NULL || name == NULL) {
        return -1;
    }

    len = strnlen(name, sizeof(t->first));
    if (len >= sizeof(t->first)) {
        return -1;
    }

    memcpy(t->first, name, len);
    t->first[len] = '\0';
    t->second = value;
    return 0;
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    return strncmp(ta->first, tb->first, sizeof(ta->first));
}

static int sort_tuples(Tuple *tuples, size_t count)
{
    if (tuples == NULL || count == 0U) {
        return -1;
    }
    qsort(tuples, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *tuples, size_t count)
{
    size_t i;

    if (tuples == NULL) {
        return;
    }
    for (i = 0U; i < count; i++) {
        (void)printf("(%.*s, %d)\n", MAX_NAME_LEN, tuples[i].first, tuples[i].second);
    }
}

int main(void)
{
    Tuple tuples[5];
    size_t count = sizeof(tuples) / sizeof(tuples[0]);

    if (set_tuple(&tuples[0], "banana", 3) != 0 ||
        set_tuple(&tuples[1], "apple", 5) != 0 ||
        set_tuple(&tuples[2], "cherry", 1) != 0 ||
        set_tuple(&tuples[3], "date", 7) != 0 ||
        set_tuple(&tuples[4], "apricot", 2) != 0) {
        (void)fprintf(stderr, "Error: failed to initialize tuples\n");
        return EXIT_FAILURE;
    }

    if (sort_tuples(tuples, count) != 0) {
        (void)fprintf(stderr, "Error: failed to sort tuples\n");
        return EXIT_FAILURE;
    }

    print_tuples(tuples, count);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weaknesses:**

The original code was already well-defended against both Flawfinder findings — they were false positives, but worth verifying:

1. **CWE-119/CWE-120 (statically-sized array `first`):** The array is only ever written through `set_tuple`, which validates length before any copy. All reads are bounded (`strncmp` with a length limit, `printf` with `%.*s` precision). No unbounded string functions (`strcpy`, `sprintf`, `gets`) are used anywhere.

2. **CWE-120 (`memcpy`):** The copy is guarded by `strnlen(name, sizeof(t->first))` followed by a rejection when `len >= sizeof(t->first)`, guaranteeing room for `len` bytes plus the null terminator. In the rewrite I tightened this by deriving the bound directly from `sizeof(t->first)` rather than the macro, and removed the redundant second check — the single guard is now provably tied to the actual destination buffer, so the invariant survives even if the struct definition changes independently of `MAX_NAME_LEN`.

Other correctness points preserved: `strnlen` (not `strlen`) prevents over-reads on non-terminated input, all pointer parameters are NULL-checked, `qsort`'s comparator uses bounded comparison, and return codes are checked in `main`.
