```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int key;
    int value;
} Tuple;

int rearrange_tuples(Tuple *tuples, size_t num_tuples, const int *order, size_t order_size) {
    if (tuples == NULL || order == NULL) {
        return -1;
    }
    if (num_tuples != order_size) {
        return -1;
    }
    if (num_tuples == 0) {
        return 0;
    }

    Tuple *temp = malloc(num_tuples * sizeof(Tuple));
    if (temp == NULL) {
        return -1;
    }

    int *used = calloc(num_tuples, sizeof(int));
    if (used == NULL) {
        free(temp);
        return -1;
    }

    for (size_t i = 0; i < order_size; ++i) {
        int found = 0;
        for (size_t j = 0; j < num_tuples; ++j) {
            if (tuples[j].key == order[i] && used[j] == 0) {
                temp[i] = tuples[j];
                used[j] = 1;
                found = 1;
                break;
            }
        }
        if (found == 0) {
            free(temp);
            free(used);
            return -1;
        }
    }

    memcpy(tuples, temp, num_tuples * sizeof(Tuple));
    free(temp);
    free(used);

    return 0;
}

int main(void) {
    Tuple tuples[] = {
        {1, 100},
        {2, 200},
        {3, 300}
    };
    int order[] = {3, 1, 2};
    size_t num_tuples = sizeof(tuples) / sizeof(tuples[0]);
    size_t order_size = sizeof(order) / sizeof(order[0]);

    if (rearrange_tuples(tuples, num_tuples, order, order_size) == 0) {
        for (size_t i = 0; i < num_tuples; ++i) {
            printf("(%d, %d)\n", tuples[i].key, tuples[i].value);
        }
    } else {
        printf("Rearrangement failed.\n");
    }

    return 0;
}
```
