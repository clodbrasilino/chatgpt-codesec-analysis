#include <stdio.h>
#include <math.h>

double calculate_wind_chill(double temperature, double wind_speed) {
    return 13.12 + 0.6215 * temperature - 11.37 * pow(wind_speed, 0.16) + 0.3965 * temperature * pow(wind_speed, 0.16);
}

int main(void) {
    double temperature = 0.0;
    double wind_speed = 0.0;
    double wind_chill = 0.0;

    printf("Enter temperature (Celsius): ");
    if (scanf("%lf", &temperature) != 1) {
        fprintf(stderr, "Invalid temperature input\n");
        return 1;
    }

    printf("Enter wind speed (km/h): ");
    if (scanf("%lf", &wind_speed) != 1) {
        fprintf(stderr, "Invalid wind speed input\n");
        return 1;
    }

    if (wind_speed < 0.0) {
        fprintf(stderr, "Wind speed cannot be negative\n");
        return 1;
    }

    if (temperature > 50.0 || temperature < -50.0) {
        fprintf(stderr, "Temperature out of valid range\n");
        return 1;
    }

    wind_chill = calculate_wind_chill(temperature, wind_speed);
    printf("Wind chill index: %.2f\n", wind_chill);

    return 0;
}
