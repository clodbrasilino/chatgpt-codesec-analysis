```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LEN 1024U

long long count_substrings(const char *s, size_t max_len);
static int read_line(char *buf, size_t buf_size);

long long count_substrings(const char *s, size_t max_len)
{
    size_t n;
    size_t i;
    size_t table_size;
    long long *freq;
    long long prefix_sum;
    long long offset;
    long long count;

    if (s == NULL) {
        return -1;
    }

    n = strnlen(s, max_len);
    if (n == max_len) {
        return -1;
    }

    if (n == 0U) {
        return 0;
    }

    for (i = 0U; i < n; i++) {
        if (!isdigit((unsigned char)s[i])) {
            return -1;
        }
    }

    if (n > ((SIZE_MAX - 1U) / 10U)) {
        return -1;
    }

    table_size = (10U * n) + 1U;
    if (table_size > (SIZE_MAX / sizeof *freq)) {
        return -1;
    }

    freq = calloc(table_size, sizeof *freq);
    if (freq == NULL) {
        return -1;
    }

    prefix_sum = 0;
    offset = (long long)n;
    count = 0;
    freq[offset] = 1;

    for (i = 0U; i < n; i++) {
        long long idx;

        prefix_sum += (long long)(s[i] - '0') - 1;
        idx = prefix_sum + offset;
        if (idx < 0 || (size_t)idx >= table_size) {
            free(freq);
            return -1;
        }
        count += freq[idx];
        freq[idx]++;
    }

    free(freq);
    return count;
}

static int read_line(char *buf, size_t buf_size)
{
    size_t len;

    if (buf == NULL || buf_size < 2U || buf_size > (size_t)INT32_MAX) {
        return -1;
    }

    if (fgets(buf, (int)buf_size, stdin) == NULL) {
        return -1;
    }

    buf[buf_size - 1U] = '\0';

    len = strcspn(buf, "\r\n");
    if (buf[len] == '\0' && len == buf_size - 1U) {
        int c;
        do {
            c = getchar();
        } while (c != '\n' && c != EOF);
        return -1;
    }

    buf[len] = '\0';
    return 0;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    long long result;

    memset(buffer, 0, sizeof buffer);

    printf("Enter a digit string: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (read_line(buffer, sizeof buffer) != 0) {
        fprintf(stderr, "Error: failed to read input or input too long.\n");
        return EXIT_FAILURE;
    }

    result = count_substrings(buffer, sizeof buffer);
    if (result < 0) {
        fprintf(stderr, "Error: invalid input or internal failure.\n");
        return EXIT_FAILURE;
    }

    printf("Number of substrings: %lld\n", result);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is already safe:

**1. Flawfinder `getchar` warning (CWE-120, CWE-20) in `read_line`:**
The `getchar()` loop is used only to drain excess characters from stdin when the input exceeds the buffer. It never writes to a buffer, so no overflow is possible. The loop terminates correctly on both `'\n'` and `EOF`, avoiding an infinite loop when stdin closes. This is a false positive — the pattern is the standard, correct way to flush an overlong line.

**2. Flawfinder `char buffer[MAX_INPUT_LEN]` warning (CWE-119/CWE-120) in `main`:**
The statically-sized array is properly bounds-protected:
- `read_line` receives `sizeof buffer` and passes it to `fgets`, which never writes more than `buf_size - 1` characters plus a null terminator.
- The buffer is explicitly null-terminated at `buf_size - 1U` as defense in depth.
- Overlong input is detected (`len == buf_size - 1U` with no newline found) and rejected rather than silently truncated.
- `count_substrings` uses `strnlen` with the buffer size, so it can never read past the buffer even if termination were somehow missing.

**Additional robustness already present:**
- Overflow guards before computing `table_size` (`n > (SIZE_MAX - 1U) / 10U`) and before the `calloc` multiplication.
- `calloc` return value checked; `freq` freed on every exit path (no leaks).
- Defensive index check on `idx` before array access.
- `isdigit` called with an `unsigned char` cast, avoiding UB for negative `char` values.
- `fflush` and `fgets` return values checked.

No changes were required; the code as written already mitigates both reported findings.
