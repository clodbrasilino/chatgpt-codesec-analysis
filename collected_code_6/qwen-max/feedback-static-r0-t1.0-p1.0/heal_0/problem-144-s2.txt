```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long long sum_of_abs_diff(int *arr, int n) {
    long long sum = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            sum += llabs((long long)arr[i] - (long long)arr[j]);
        }
    }
    return sum * 2;
}

int main() {
    int arr[] = {1, 2, 3, 4};
    int n = sizeof(arr) / sizeof(arr[0]);
    printf("%lld\n", sum_of_abs_diff(arr, n));
    return 0;
}
```
