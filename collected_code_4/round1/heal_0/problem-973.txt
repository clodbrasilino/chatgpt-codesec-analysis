```C
#include <stdio.h>
#include <string.h>

char *rotate_string_left(char *str, size_t n) {
    size_t str_length = strlen(str);
    char *result = malloc((str_length + 1) * sizeof(char));

    if (result == NULL) {
        return NULL;
    }

    for (size_t i = 0; i < str_length; ++i) {
        size_t new_idx = (i + n) % str_length;
        result[i] = str[new_idx];
    }

    result[str_length] = '\0';

    return result;
}

int main(void) {
    char str[] = "Hello, world!";
    size_t n = 2;

    printf("Original string: %s\n", str);

    char *rotated = rotate_string_left(str, n);

    if (rotated == NULL) {
        return 1;
    }

    printf("Rotated string: %s\n", rotated);

    free(rotated);

    return 0;
}
```
This code follows general coding standards, checks correctly for exceptional conditions (like `malloc` failing), and manages memory correctly, avoiding CWE-664: Improper Control of a Resource Through its Lifetime. The function `rotate_string_left` creates a new string which is the original string rotated `n` positions to the left. It works by modulating the index into the source string by the string length and copying the character at the new index to the result string. The new string is null-terminated.
