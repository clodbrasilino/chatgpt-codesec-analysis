#include <stdio.h>
#include <stdlib.h>
#include <stddef.h>

typedef enum {
    TYPE_INT,
    TYPE_TUPLE
} ElementType;

struct Element;

typedef struct {
    struct Element* elements;
    size_t count;
} Tuple;

typedef struct Element {
    ElementType type;
    union {
        int i_val;
        Tuple t_val;
    } data;
} Element;

typedef struct {
    int* array;
    size_t size;
    size_t capacity;
} IntArray;

void init_int_array(IntArray* arr) {
    if (!arr) return;
    arr->size = 0;
    arr->capacity = 10;
    arr->array = (int*)malloc(arr->capacity * sizeof(int));
    if (!arr->array) {
        exit(EXIT_FAILURE);
    }
}

void append_int_array(IntArray* arr, int val) {
    if (!arr || !arr->array) return;
    if (arr->size >= arr->capacity) {
        size_t new_cap = arr->capacity * 2;
        int* new_arr = (int*)realloc(arr->array, new_cap * sizeof(int));
        if (!new_arr) {
            free(arr->array);
            exit(EXIT_FAILURE);
        }
        arr->array = new_arr;
        arr->capacity = new_cap;
    }
    arr->array[arr->size++] = val;
}

void destroy_int_array(IntArray* arr) {
    if (arr) {
        free(arr->array);
        arr->array = NULL;
        arr->size = 0;
        arr->capacity = 0;
    }
}

void extract_evens(const Element* elem, IntArray* evens) {
    if (!elem || !evens) return;

    if (elem->type == TYPE_INT) {
        if (elem->data.i_val % 2 == 0) {
            append_int_array(evens, elem->data.i_val);
        }
    } else if (elem->type == TYPE_TUPLE) {
        for (size_t i = 0; i < elem->data.t_val.count; ++i) {
            extract_evens(&elem->data.t_val.elements[i], evens);
        }
    }
}

int main(void) {
    Element e1 = { TYPE_INT, { .i_val = 1 } };
    Element e2 = { TYPE_INT, { .i_val = 2 } };
    Element e3 = { TYPE_INT, { .i_val = 3 } };
    Element e4 = { TYPE_INT, { .i_val = 4 } };

    Element inner_elems[] = {e2, e3};
    Element inner_tuple = { TYPE_TUPLE, { .t_val = { inner_elems, 2 } } };

    Element outer_elems[] = {e1, inner_tuple, e4};
    Element outer_tuple = { TYPE_TUPLE, { .t_val = { outer_elems, 3 } } };

    IntArray evens;
    init_int_array(&evens);

    extract_evens(&outer_tuple, &evens);

    for (size_t i = 0; i < evens.size; ++i) {
        printf("%d ", evens.array[i]);
    }
    printf("\n");

    destroy_int_array(&evens);

    return 0;
}
