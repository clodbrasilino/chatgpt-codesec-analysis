#include <stdio.h>
#include <stdlib.h>

long long product_of_unique(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return 0;
    }

    long long product = 1;
    int has_unique = 0;

    for (size_t i = 0; i < size; ++i) {
        int is_unique = 1;
        for (size_t j = 0; j < i; ++j) {
            if (arr[i] == arr[j]) {
                is_unique = 0;
                break;
            }
        }
        if (is_unique) {
            product *= arr[i];
            has_unique = 1;
        }
    }

    return has_unique ? product : 0;
}

int main(void) {
    int arr[] = {2, 3, 4, 3, 5, 2, 6};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    long long result = product_of_unique(arr, size);

    printf("%lld\n", result);

    return 0;
}
