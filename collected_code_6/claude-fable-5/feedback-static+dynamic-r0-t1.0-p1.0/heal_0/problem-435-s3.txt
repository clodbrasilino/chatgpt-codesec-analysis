#include <stdio.h>
#include <stdlib.h>

int last_digit(long long number)
{
    long long digit = number % 10;
    if (digit < 0)
    {
        digit = -digit;
    }
    return (int)digit;
}

int main(void)
{
    long long number = 0;

    printf("Enter a number: ");
    if (scanf("%lld", &number) != 1)
    {
        fprintf(stderr, "Invalid input.\n");
        return EXIT_FAILURE;
    }

    printf("The last digit of %lld is %d\n", number, last_digit(number));

    return EXIT_SUCCESS;
}
