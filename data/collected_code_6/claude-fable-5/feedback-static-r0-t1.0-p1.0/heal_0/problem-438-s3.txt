#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int first;
    int second;
} Tuple;

int count_bidirectional_pairs(const Tuple *tuples, size_t n)
{
    if (tuples == NULL || n == 0U) {
        return 0;
    }

    bool *used = calloc(n, sizeof(bool));
    if (used == NULL) {
        return -1;
    }

    int count = 0;

    for (size_t i = 0U; i < n; i++) {
        if (used[i]) {
            continue;
        }
        for (size_t j = i + 1U; j < n; j++) {
            if (used[j]) {
                continue;
            }
            if ((tuples[i].first == tuples[j].second) &&
                (tuples[i].second == tuples[j].first)) {
                used[i] = true;
                used[j] = true;
                count++;
                break;
            }
        }
    }

    free(used);
    return count;
}

int main(void)
{
    Tuple data[] = {
        {5, 6},
        {1, 2},
        {6, 5},
        {9, 1},
        {6, 5},
        {2, 1}
    };
    size_t n = sizeof(data) / sizeof(data[0]);

    int result = count_bidirectional_pairs(data, n);
    if (result < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("Bidirectional pairs: %d\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
