#include <stdio.h>
#include <stdlib.h>

long long *memo;
size_t memo_size;

long long max(long long a, long long b) {
    return a > b ? a : b;
}

long long f(long long n) {
    if (n < 12) {
        return n;
    }

    if (n < (long long)memo_size && memo[n] != -1) {
        return memo[n];
    }

    long long result = max(n, f(n / 2) + f(n / 3) + f(n / 4) + f(n / 5));

    if (n < (long long)memo_size) {
        memo[n] = result;
    }

    return result;
}

int main(void) {
    long long n;
    printf("Enter n: ");
    if (scanf("%lld", &n) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (n < 0) {
        fprintf(stderr, "n must be non-negative\n");
        return EXIT_FAILURE;
    }

    memo_size = (size_t)(n < 1000000 ? n + 1 : 1000000);
    memo = (long long *)malloc(memo_size * sizeof(long long));
    if (memo == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (size_t i = 0; i < memo_size; i++) {
        memo[i] = -1;
    }

    long long result = f(n);
    printf("Maximum sum: %lld\n", result);

    free(memo);
    return EXIT_SUCCESS;
}
