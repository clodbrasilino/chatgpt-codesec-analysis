```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool calculate_loss(double cost_price, double selling_price, double *loss);

bool calculate_loss(double cost_price, double selling_price, double *loss)
{
    if (loss == NULL) {
        return false;
    }

    if (cost_price > selling_price) {
        *loss = cost_price - selling_price;
        return true;
    }

    *loss = 0.0;
    return false;
}

int main(void)
{
    double cost_price = 0.0;
    double selling_price = 0.0;
    double loss = 0.0;

    printf("Enter cost price: ");
    if (scanf("%lf", &cost_price) != 1) {
        fprintf(stderr, "Invalid input for cost price\n");
        return EXIT_FAILURE;
    }

    printf("Enter selling price: ");
    if (scanf("%lf", &selling_price) != 1) {
        fprintf(stderr, "Invalid input for selling price\n");
        return EXIT_FAILURE;
    }

    if (cost_price < 0.0 || selling_price < 0.0) {
        fprintf(stderr, "Prices must be non-negative\n");
        return EXIT_FAILURE;
    }

    if (calculate_loss(cost_price, selling_price, &loss)) {
        printf("Loss amount: %.2f\n", loss);
    } else {
        printf("None\n");
    }

    return EXIT_SUCCESS;
}
```
