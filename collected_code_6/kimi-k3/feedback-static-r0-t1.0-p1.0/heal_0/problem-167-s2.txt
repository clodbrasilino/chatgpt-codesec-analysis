#include <stdio.h>
#include <stdint.h>
#include <limits.h>

static uint64_t smallest_power_of_two(uint64_t n)
{
    uint64_t power = 1;
    
    if (n == 0) {
        return 1;
    }
    
    if (n > (UINT64_MAX >> 1)) {
        return 0;
    }
    
    while (power < n) {
        power <<= 1;
    }
    
    return power;
}

int main(void)
{
    uint64_t n;
    uint64_t result;
    
    if (scanf("%llu", (unsigned long long *)&n) != 1) {
        return 1;
    }
    
    result = smallest_power_of_two(n);
    
    if (result == 0) {
        printf("Overflow\n");
    } else {
        printf("%llu\n", (unsigned long long)result);
    }
    
    return 0;
}
