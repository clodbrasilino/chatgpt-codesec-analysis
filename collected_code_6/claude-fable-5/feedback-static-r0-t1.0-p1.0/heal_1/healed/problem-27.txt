```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdint.h>

#define MAX_INPUT_LENGTH 4096

char *remove_digits_from_string(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LENGTH);
    if (len == MAX_INPUT_LENGTH && input[len - 1] != '\0') {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!isdigit((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

char **remove_digits_from_list(const char **list, size_t count)
{
    char **output;
    size_t i;
    size_t k;

    if (list == NULL || count == 0) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(char *)) {
        return NULL;
    }

    output = malloc(count * sizeof(char *));
    if (output == NULL) {
        return NULL;
    }

    for (i = 0; i < count; i++) {
        output[i] = remove_digits_from_string(list[i]);
        if (output[i] == NULL) {
            for (k = 0; k < i; k++) {
                free(output[k]);
            }
            free(output);
            return NULL;
        }
    }

    return output;
}

void free_string_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *strings[] = {
        "hello123world",
        "abc456def",
        "2024report",
        "no_digits_here",
        "999"
    };
    size_t count = sizeof(strings) / sizeof(strings[0]);
    char **cleaned;
    size_t i;

    cleaned = remove_digits_from_list(strings, count);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to process string list\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < count; i++) {
        if (printf("\"%s\" -> \"%s\"\n", strings[i], cleaned[i]) < 0) {
            free_string_list(cleaned, count);
            return EXIT_FAILURE;
        }
    }

    free_string_list(cleaned, count);
    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes made:**

The original code was already well-written — it had proper NULL checks, correct `isdigit` usage with an `unsigned char` cast (avoiding undefined behavior on negative `char` values), careful cleanup on partial allocation failure, and no memory leaks.

The Flawfinder warning (CWE-126) concerned `strlen` performing an over-read on non-NUL-terminated input. The fix replaces `strlen` with `strnlen` bounded by `MAX_INPUT_LENGTH`, rejecting strings that reach the bound without termination. This caps the maximum read and prevents an unbounded over-read from a malformed input string.

Additionally, a defensive overflow check was added before `malloc(count * sizeof(char *))` — on platforms where `count` could be attacker-influenced, `count * sizeof(char *)` could wrap around `SIZE_MAX`, producing an undersized allocation followed by heap overflow. Guarding with `count > SIZE_MAX / sizeof(char *)` eliminates that risk (CWE-190/CWE-131).
