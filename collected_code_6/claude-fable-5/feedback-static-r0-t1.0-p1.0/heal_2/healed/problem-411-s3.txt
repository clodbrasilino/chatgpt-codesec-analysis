```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <regex.h>

#define MAX_INPUT_LEN 4096U

char *snake_to_camel(const char *input)
{
    regex_t regex;
    regmatch_t match[2];
    char *result;
    size_t len;
    size_t buf_size;
    size_t out_pos;
    const char *cursor;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len > MAX_INPUT_LEN) {
        return NULL;
    }

    buf_size = len + 1U;
    result = malloc(buf_size);
    if (result == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "_([a-zA-Z0-9])", REG_EXTENDED);
    if (ret != 0) {
        free(result);
        return NULL;
    }

    cursor = input;
    out_pos = 0U;

    while (regexec(&regex, cursor, 2U, match, 0) == 0) {
        size_t prefix_len;

        if (match[0].rm_so < 0 || match[1].rm_so < 0) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        prefix_len = (size_t)match[0].rm_so;

        if (prefix_len >= buf_size ||
            out_pos >= buf_size ||
            out_pos + prefix_len + 1U >= buf_size) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, prefix_len);
        out_pos += prefix_len;

        result[out_pos] = (char)toupper((unsigned char)cursor[match[1].rm_so]);
        out_pos += 1U;

        cursor += match[0].rm_eo;
    }

    {
        size_t rest_len = strnlen(cursor, buf_size);

        if (rest_len >= buf_size ||
            out_pos >= buf_size ||
            out_pos + rest_len >= buf_size) {
            regfree(&regex);
            free(result);
            return NULL;
        }

        memcpy(&result[out_pos], cursor, rest_len);
        out_pos += rest_len;
    }

    result[out_pos] = '\0';
    regfree(&regex);

    return result;
}

int main(void)
{
    const char *tests[] = {
        "hello_world",
        "convert_snake_case_to_camel_case",
        "already",
        "multiple__underscores",
        "trailing_"
    };
    size_t i;
    size_t count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0U; i < count; i++) {
        char *converted = snake_to_camel(tests[i]);
        if (converted == NULL) {
            fprintf(stderr, "Conversion failed for: %s\n", tests[i]);
            return EXIT_FAILURE;
        }
        printf("%s -> %s\n", tests[i], converted);
        free(converted);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already fundamentally sound regarding the two Flawfinder `memcpy` warnings (CWE-120) — both were false positives in the strict sense, since each copy was preceded by an explicit bounds check against `buf_size`. However, the rewrite hardens the code further:

1. **First `memcpy` (prefix copy):** Added defensive checks that `match[0].rm_so` and `match[1].rm_so` are non-negative before casting to `size_t` — a negative `regoff_t` cast to `size_t` would produce a huge value, and while `out_pos + prefix_len + 1U` could then wrap around, the individual guards (`prefix_len >= buf_size`, `out_pos >= buf_size`) now prevent any integer-overflow bypass of the combined check.

2. **Second `memcpy` (tail copy):** Same pattern — individual bounds checks on `rest_len` and `out_pos` before the combined sum check, eliminating any theoretical unsigned wraparound path.

3. **Invariants preserved:** The allocation of `len + 1U` bytes remains correct because camelCase output is always ≤ input length (each `_X` pair collapses to one character), so the checks should never fire in practice — they exist purely as defense-in-depth.

4. **Resource management:** All error paths correctly release both the compiled regex (`regfree`) and the heap buffer (`free`), so no leaks are introduced.
