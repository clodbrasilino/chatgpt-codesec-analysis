#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

char *remove_multiple_spaces(const char *input)
{
    regex_t regex;
    regmatch_t match;
    char *output;
    const char *cursor;
    size_t input_len;
    size_t out_len;
    size_t remainder_len;
    size_t output_size;
    int ret;

    if (input == NULL) {
        return NULL;
    }

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return NULL;
    }

    input_len = strnlen(input, (size_t)-1 / 2U);

    if (input_len >= ((size_t)-1 / 2U)) {
        regfree(&regex);
        return NULL;
    }

    output_size = input_len + 1U;

    output = malloc(output_size);
    if (output == NULL) {
        regfree(&regex);
        return NULL;
    }

    out_len = 0U;
    cursor = input;

    while ((ret = regexec(&regex, cursor, 1U, &match, 0)) == 0) {
        const size_t prefix_len = (size_t)match.rm_so;

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            free(output);
            regfree(&regex);
            return NULL;
        }

        if (prefix_len > (output_size - 1U) ||
            out_len > ((output_size - 1U) - prefix_len)) {
            free(output);
            regfree(&regex);
            return NULL;
        }

        memcpy(output + out_len, cursor, prefix_len);
        out_len += prefix_len;
        output[out_len] = ' ';
        out_len += 1U;
        cursor += match.rm_eo;
    }

    if (ret != REG_NOMATCH) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    if ((size_t)(cursor - input) > input_len) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    remainder_len = strnlen(cursor, input_len - (size_t)(cursor - input));

    if (remainder_len > (output_size - 1U) ||
        out_len > ((output_size - 1U) - remainder_len)) {
        free(output);
        regfree(&regex);
        return NULL;
    }

    memcpy(output + out_len, cursor, remainder_len);
    out_len += remainder_len;
    output[out_len] = '\0';

    regfree(&regex);
    return output;
}

int main(void)
{
    const char *test = "This   is  a    string   with     multiple spaces.";
    char *result;

    result = remove_multiple_spaces(test);
    if (result == NULL) {
        fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already largely defensive, but the flagged `memcpy` calls warranted stronger guarantees. The rewritten version addresses the CWE-120 concerns as follows:

1. **Explicit buffer-capacity checks before each `memcpy`**: Both bounds checks are now expressed relative to `output_size - 1U` (the usable capacity excluding the NUL terminator) and written in overflow-safe form — `out_len > ((output_size - 1U) - prefix_len)` instead of `(out_len + prefix_len + 1U) > input_len`, avoiding any possibility of unsigned wraparound in the check itself.

2. **Validation of regex match offsets**: `match.rm_so` and `match.rm_eo` are checked for sanity (`rm_so >= 0`, `rm_eo >= rm_so`) before being cast to `size_t`, since a negative `regoff_t` cast to `size_t` would produce a huge length and defeat the subsequent bounds check.

3. **Cursor position validation**: Before computing `remainder_len`, the code verifies `(cursor - input) <= input_len`, ensuring the subtraction passed to `strnlen` cannot underflow.

4. **Guard against pathological input length**: An explicit check rejects inputs at the `strnlen` truncation limit, ensuring `input_len + 1U` cannot overflow in the `malloc` size computation.

The logic remains correct: since each replacement writes at most as many bytes as it consumes (a run of N ≥ 2 spaces becomes 1 space), the output can never exceed the input length, and the checks now prove this invariant to both static analyzers and human reviewers.
