```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_WORD_LENGTH 4096

size_t remove_words_containing(char **words, size_t count, const char *pattern);
char *duplicate_string(const char *source);
void free_word_list(char **words, size_t count);

size_t remove_words_containing(char **words, size_t count, const char *pattern)
{
    size_t write_index = 0;
    size_t read_index;

    if (words == NULL || pattern == NULL) {
        return count;
    }

    for (read_index = 0; read_index < count; read_index++) {
        if (words[read_index] == NULL) {
            continue;
        }
        if (strstr(words[read_index], pattern) != NULL) {
            free(words[read_index]);
            words[read_index] = NULL;
        } else {
            words[write_index] = words[read_index];
            if (write_index != read_index) {
                words[read_index] = NULL;
            }
            write_index++;
        }
    }

    return write_index;
}

char *duplicate_string(const char *source)
{
    size_t length;
    size_t buffer_size;
    char *copy;

    if (source == NULL) {
        return NULL;
    }

    length = strnlen(source, MAX_WORD_LENGTH);
    if (length >= MAX_WORD_LENGTH) {
        return NULL;
    }

    buffer_size = length + 1;
    copy = malloc(buffer_size);
    if (copy == NULL) {
        return NULL;
    }

    if (length > 0 && length < buffer_size) {
        memcpy(copy, source, length);
    }
    copy[length] = '\0';
    return copy;
}

void free_word_list(char **words, size_t count)
{
    size_t i;

    if (words == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(words[i]);
        words[i] = NULL;
    }
    free(words);
}

int main(void)
{
    const char *initial_words[] = {
        "apple", "banana", "cherry", "grape", "pineapple", "mango"
    };
    const char *pattern = "ap";
    size_t total = sizeof(initial_words) / sizeof(initial_words[0]);
    size_t i;
    size_t remaining;
    char **words;

    words = malloc(total * sizeof(char *));
    if (words == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < total; i++) {
        words[i] = duplicate_string(initial_words[i]);
        if (words[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_word_list(words, i);
            return EXIT_FAILURE;
        }
    }

    printf("Original list:\n");
    for (i = 0; i < total; i++) {
        printf("  %s\n", words[i]);
    }

    remaining = remove_words_containing(words, total, pattern);

    printf("After removing words containing \"%s\":\n", pattern);
    for (i = 0; i < remaining; i++) {
        printf("  %s\n", words[i]);
    }

    free_word_list(words, remaining);

    return EXIT_SUCCESS;
}
```

**Analysis notes on the flagged weakness:**

The Flawfinder warning on `memcpy` (CWE-120) was a false positive in the original code, but the rewrite hardens it further:

1. **Destination sizing is provably correct**: The buffer is allocated with `malloc(length + 1)`, and exactly `length` bytes are copied, leaving room for the null terminator. The added explicit guard `length < buffer_size` makes the bounds relationship visible to both static analyzers and reviewers.

2. **Length is bounded before allocation**: `strnlen(source, MAX_WORD_LENGTH)` guarantees `length` never exceeds `MAX_WORD_LENGTH`, and the `length >= MAX_WORD_LENGTH` rejection prevents unterminated-input edge cases and integer-overflow concerns in `length + 1`.

3. **Null termination is unconditional**: `copy[length] = '\0'` executes regardless of the copy branch, so the returned string is always properly terminated.

Other defensive properties preserved from the original: `remove_words_containing` compacts in place without double-frees (slots are NULLed after move/free), `free_word_list` tolerates NULL input and NULLs pointers after freeing, and `main` correctly frees only `i` entries on partial allocation failure.
