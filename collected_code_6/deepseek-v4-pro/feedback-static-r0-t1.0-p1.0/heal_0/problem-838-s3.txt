#include <stdio.h>
#include <string.h>

int minSwaps(char *s1, char *s2) {
    int len = strlen(s1);
    int count1 = 0, count2 = 0;
    int diff = 0;

    for (int i = 0; i < len; i++) {
        if (s1[i] == '1') count1++;
        if (s2[i] == '1') count2++;
    }

    if (count1 != count2) {
        return -1;
    }

    int pos1 = 0, pos2 = 0;
    for (int i = 0; i < len; i++) {
        if (s1[i] != s2[i]) {
            if (s1[i] == '1') pos1++;
            else pos2++;
        }
    }

    if (pos1 % 2 == 0 && pos2 % 2 == 0) {
        return pos1 / 2 + pos2 / 2;
    } else if (pos1 % 2 == 1 && pos2 % 2 == 1) {
        return pos1 / 2 + pos2 / 2 + 2;
    }

    return -1;
}

int main() {
    char s1[100], s2[100];

    printf("Enter first binary string: ");
    scanf("%99s", s1);
    printf("Enter second binary string: ");
    scanf("%99s", s2);

    if (strlen(s1) != strlen(s2)) {
        printf("-1\n");
        return 0;
    }

    int result = minSwaps(s1, s2);
    printf("%d\n", result);

    return 0;
}
