#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void remove_even_characters(char *str, size_t max_size) {
    if (str == NULL || max_size == 0) {
        return;
    }

    size_t len = 0;
    while (len < max_size && str[len] != '\0') {
        len++;
    }

    if (len == max_size) {
        len = max_size - 1;
    }

    size_t j = 0;
    for (size_t i = 0; i < len; i++) {
        if (i % 2 != 0) {
            str[j++] = str[i];
        }
    }
    str[j] = '\0';
}

int main(void) {
    size_t buffer_size = 256;
    char *str = malloc(buffer_size * sizeof(char));
    if (str == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    snprintf(str, buffer_size, "Hello World!");
    
    remove_even_characters(str, buffer_size);
    
    printf("%s\n", str);
    
    free(str);
    
    return 0;
}
