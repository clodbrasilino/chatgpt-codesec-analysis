#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 9
#define BUFFER_SIZE 64

static int find_demlo(const char *input, size_t input_max, char *output, size_t output_size);

static int find_demlo(const char *input, size_t input_max, char *output, size_t output_size)
{
    size_t len;
    size_t pos;
    size_t i;

    if (input == NULL || output == NULL || output_size == 0U)
    {
        return -1;
    }

    len = strnlen(input, input_max);

    if (len == 0U || len >= input_max || len > (size_t)MAX_INPUT_LEN)
    {
        return -1;
    }

    for (i = 0U; i < len; i++)
    {
        if (input[i] != '1')
        {
            return -1;
        }
    }

    if (output_size < (2U * len))
    {
        return -1;
    }

    pos = 0U;

    for (i = 1U; i <= len; i++)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
    }

    i = len - 1U;

    while (i >= 1U)
    {
        if (pos >= (output_size - 1U))
        {
            return -1;
        }
        output[pos] = (char)('0' + (char)i);
        pos++;
        i--;
    }

    output[pos] = '\0';

    return 0;
}

int main(void)
{
    char input[BUFFER_SIZE];
    char result[BUFFER_SIZE];
    size_t len;

    memset(input, 0, sizeof(input));
    memset(result, 0, sizeof(result));

    printf("Enter a repunit number (only 1s, max %d digits): ", MAX_INPUT_LEN);

    if (fgets(input, (int)sizeof(input), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));

    if (len > 0U && input[len - 1U] == '\n')
    {
        input[len - 1U] = '\0';
    }
    else
    {
        int c;

        do
        {
            c = getchar();
        } while (c != '\n' && c != EOF);
    }

    if (find_demlo(input, sizeof(input), result, sizeof(result)) != 0)
    {
        fprintf(stderr, "Error: invalid input, expected 1 to %d ones\n", MAX_INPUT_LEN);
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", result);

    return EXIT_SUCCESS;
}

Analysis of the original code and the fixes applied:

**Original weaknesses assessed:**

1. **Statically-sized buffers `input` and `result` (CWE-119/CWE-120):** These were flagged by Flawfinder as risk-2 items. In practice, the original code already mitigated them reasonably well — `fgets` is bounded by `sizeof(input)`, both buffers are zeroed with `memset`, and a forced NUL terminator is written at `input[sizeof(input) - 1]`. The rewrite retains all of these mitigations.

2. **Missing per-write bounds checking in `find_demlo`:** The original relied solely on the up-front check `output_size < (2U * len)`. While mathematically sufficient (the function writes exactly `2*len - 1` characters plus the terminator), defense-in-depth was added: each write into `output` is now guarded by `pos >= (output_size - 1U)`, guaranteeing the loop can never write past the buffer or overwrite the space reserved for the NUL terminator, even if the length logic were ever modified incorrectly in the future.

3. **`getchar()` in a loop (CWE-120/CWE-20, risk 1):** This usage is safe — the loop only drains stdin until newline or EOF and never stores the characters, so no buffer boundary is involved. It is retained as-is since it correctly handles the case where input exceeded the buffer.

4. **Input validation (CWE-20):** The function rejects NULL pointers, zero-sized outputs, empty input, unterminated/truncated input (`len >= input_max`), oversized input (`len > MAX_INPUT_LEN`), and any non-`'1'` character — so a repunit is strictly enforced before computing the Demlo number (e.g., `111` → `12321`). All these checks are preserved.

The resulting code is bounds-safe on every write, handles EOF and oversized lines correctly, and cannot overflow either buffer.
