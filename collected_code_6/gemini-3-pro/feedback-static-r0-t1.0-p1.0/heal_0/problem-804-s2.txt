#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool is_product_even(const int *numbers, size_t count) {
    if (numbers == NULL || count == 0) {
        return false; 
    }

    for (size_t i = 0; i < count; i++) {
        if (numbers[i] % 2 == 0) {
            return true;
        }
    }

    return false;
}

int main(void) {
    int nums1[] = {1, 3, 5, 7};
    size_t count1 = sizeof(nums1) / sizeof(nums1[0]);

    if (is_product_even(nums1, count1)) {
        printf("Product of nums1 is even.\n");
    } else {
        printf("Product of nums1 is odd.\n");
    }

    int nums2[] = {1, 3, 4, 7};
    size_t count2 = sizeof(nums2) / sizeof(nums2[0]);

    if (is_product_even(nums2, count2)) {
        printf("Product of nums2 is even.\n");
    } else {
        printf("Product of nums2 is odd.\n");
    }

    return 0;
}
