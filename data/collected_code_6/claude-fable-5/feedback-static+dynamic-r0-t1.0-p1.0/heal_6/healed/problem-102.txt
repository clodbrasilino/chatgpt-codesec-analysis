```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_SIZE 1024

static char *snake_to_camel(const char *snake, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    char *camel;
    int capitalize_next;

    if (snake == NULL || max_len == 0) {
        return NULL;
    }

    len = strnlen(snake, max_len);
    if (len >= max_len) {
        return NULL;
    }

    camel = malloc(len + 1);
    if (camel == NULL) {
        return NULL;
    }

    capitalize_next = 1;
    j = 0;
    for (i = 0; i < len; i++) {
        if (snake[i] == '_') {
            capitalize_next = 1;
        } else {
            if (capitalize_next != 0) {
                camel[j] = (char)toupper((unsigned char)snake[i]);
                capitalize_next = 0;
            } else {
                camel[j] = (char)tolower((unsigned char)snake[i]);
            }
            j++;
        }
    }
    camel[j] = '\0';

    return camel;
}

int main(void)
{
    char input[MAX_INPUT_SIZE];
    char *result;
    size_t len;

    memset(input, 0, sizeof(input));

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "No input provided\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    while (len > 0 && (input[len - 1] == '\n' || input[len - 1] == '\r')) {
        input[len - 1] = '\0';
        len--;
    }

    result = snake_to_camel(input, sizeof(input));
    if (result == NULL) {
        fprintf(stderr, "Conversion failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", result) < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    if (fflush(stdout) != 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the flagged weakness:**

The Flawfinder warning (CWE-119/CWE-120) refers to the statically-sized `char input[MAX_INPUT_SIZE]` buffer. This is a low-severity, informational finding — static analyzers flag every fixed-size array because they *can* be misused, not because this one *is*. On review, the original code already mitigates every path to overflow, so the rewrite preserves those defenses rather than restructuring:

1. **Bounded read:** `fgets(input, (int)sizeof(input), stdin)` limits input to the buffer size and always NUL-terminates within bounds. This is the correct fix for the CWE-120 class (unlike `gets` or unbounded `scanf("%s", ...)`).

2. **Defensive termination:** `input[sizeof(input) - 1] = '\0'` is redundant after `fgets` combined with `memset`, but harmless belt-and-suspenders hardening.

3. **Bounded length calculation:** `strnlen` with `sizeof(input)` is used instead of `strlen`, so even a hypothetically unterminated buffer cannot cause an over-read.

4. **Safe allocation in `snake_to_camel`:** The output buffer is `malloc(len + 1)`, and since the camelCase result can only be shorter than or equal to the input (underscores are dropped), `j` can never exceed `len`. The `len >= max_len` check rejects unterminated input, and NULL/zero-length arguments are handled.

5. **Correct `ctype` usage:** Casting to `unsigned char` before `toupper`/`tolower` avoids undefined behavior for negative `char` values.

6. **Resource handling:** `free(result)` is called on every exit path after allocation, and I/O return values (`printf`, `fflush`) are checked.

No functional changes were required — the code is a correct, defensively written implementation, and the Flawfinder hit is a false positive properly mitigated by the existing bounds checking.
