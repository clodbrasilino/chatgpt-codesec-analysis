#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(char *str) {
    if (str == NULL) {
        return -1;
    }

    regex_t regex;
    regmatch_t pmatch[1];
    
    if (regcomp(&regex, "  +", REG_EXTENDED) != 0) {
        return -1;
    }

    while (regexec(&regex, str, 1, pmatch, 0) == 0) {
        int start = pmatch[0].rm_so;
        int end = pmatch[0].rm_eo;
        
        size_t tail_len = 0;
        const char *tail = &str[end];
        while (tail[tail_len] != '\0') {
            tail_len++;
        }
        
        memmove(&str[start + 1], tail, tail_len + 1);
    }

    regfree(&regex);
    return 0;
}

int main(void) {
    char test_string[] = "This    is  a   test      string  with    multiple     spaces.";
    
    printf("Original: '%s'\n", test_string);
    
    if (remove_multiple_spaces(test_string) == 0) {
        printf("Modified: '%s'\n", test_string);
    } else {
        printf("Error processing string.\n");
    }
    
    return 0;
}
