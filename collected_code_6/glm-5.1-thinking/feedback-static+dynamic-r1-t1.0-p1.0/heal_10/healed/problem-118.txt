#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *value;
    struct Node *next;
} Node;

void free_list(Node *head) {
    Node *current = head;
    Node *next;
    while (current != NULL) {
        next = current->next;
        free(current->value);
        free(current);
        current = next;
    }
}

void print_list(Node *head) {
    if (head != NULL) {
        printf("[");
        Node *current = head;
        while (current != NULL) {
            printf("'%s'", current->value);
            if (current->next != NULL) {
                printf(", ");
            }
            current = current->next;
        }
        printf("]\n");
    } else {
        printf("[]\n");
    }
    fflush(stdout);
}

Node *string_to_list(const char *str, const char *delim) {
    if (str == NULL || delim == NULL) {
        return NULL;
    }

    Node *head = NULL;
    Node *tail = NULL;
    const char *p = str;

    while (*p) {
        p += strspn(p, delim);
        if (!*p) {
            break;
        }

        size_t len = strcspn(p, delim);
        char *value = malloc(len + 1);
        if (value == NULL) {
            free_list(head);
            return NULL;
        }
        memcpy(value, p, len);
        value[len] = '\0';

        Node *node = malloc(sizeof(Node));
        if (node == NULL) {
            free(value);
            free_list(head);
            return NULL;
        }
        node->value = value;
        node->next = NULL;

        if (head == NULL) {
            head = node;
            tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
        p += len;
    }

    return head;
}

int main(int argc, const char *const argv[]) {
    if (argc > 1) {
        for (int i = 1; i < argc; i++) {
            if (argv[i] != NULL) {
                Node *head = string_to_list(argv[i], ", ");
                print_list(head);
                free_list(head);
            }
        }
    } else {
        char text[1024];
        while (fgets(text, sizeof(text), stdin)) {
            size_t len = strlen(text);
            while (len > 0 && (text[len - 1] == '\n' || text[len - 1] == '\r')) {
                text[--len] = '\0';
            }
            Node *head = string_to_list(text, ", ");
            print_list(head);
            free_list(head);
        }
    }
    return 0;
}
