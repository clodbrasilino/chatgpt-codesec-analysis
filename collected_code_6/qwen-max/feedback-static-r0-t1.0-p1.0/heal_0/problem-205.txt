#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
} Tuple;

int compare(const void *a, const void *b) {
    return ((Tuple *)a)->first - ((Tuple *)b)->first;
}

int* countInversions(Tuple *tuples, int n, int *resultSize) {
    *resultSize = 0;
    if (n <= 1) {
        return NULL;
    }

    qsort(tuples, n, sizeof(Tuple), compare);
    int *inversions = (int *)malloc(n * sizeof(int));
    for (int i = 0; i < n; ++i) {
        inversions[i] = 0;
    }
    
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (tuples[j].second < tuples[i].second) {
                inversions[tuples[i].first]++;
                inversions[tuples[j].first]++;
            }
        }
    }
    *resultSize = n;
    return inversions;
}

int main() {
    Tuple tuples[] = {{0, 5}, {1, 3}, {2, 7}, {3, 2}};
    int n = 4;
    int resultSize;

    int *inversions = countInversions(tuples, n, &resultSize);
    if (inversions != NULL) {
        for (int i = 0; i < resultSize; ++i) {
            printf("Inversions for tuple %d: %d\n", i, inversions[i]);
        }
        free(inversions);
    }

    return 0;
}
