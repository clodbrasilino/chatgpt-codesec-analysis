```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void count_char_freq(const char *str, int *freq) {
    for (int i = 0; i < 256; i++) {
        freq[i] = 0;
    }
    while (*str) {
        freq[(unsigned char)*str]++;
        str++;
    }
}

int main() {
    const char *input = "hello world";
    int *freq = (int *)calloc(256, sizeof(int));
    if (freq == NULL) {
        return -1;
    }
    count_char_freq(input, freq);
    for (int i = 0; i < 256; i++) {
        if (freq[i] > 0) {
            printf("%c: %d\n", i, freq[i]);
        }
    }
    free(freq);
    return 0;
}
```
