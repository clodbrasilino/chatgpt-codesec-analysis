#include <stdio.h>
#include <string.h>

#define BUFFER_SIZE 256

void remove_first_last(char *str, size_t max_size, char ch)
{
    size_t len;
    char *first;
    char *last;
    size_t pos;

    if (str == NULL || max_size == 0 || ch == '\0') {
        return;
    }

    len = strnlen(str, max_size);
    if (len == 0 || len >= max_size) {
        return;
    }

    first = memchr(str, (unsigned char)ch, len);
    if (first == NULL) {
        return;
    }

    last = strrchr(str, ch);

    if (last != NULL && last != first) {
        pos = (size_t)(last - str);
        memmove(str + pos, str + pos + 1, len - pos);
        len--;
    }

    pos = (size_t)(first - str);
    memmove(str + pos, str + pos + 1, len - pos);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return 1;
    }
    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == sizeof(buffer) - 1) {
        do {
            c = getchar();
        } while (c != EOF && c != '\n');
    }

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return 1;
    }
    ch = (char)c;

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return 0;
}

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **Statically-sized buffer (CWE-119/CWE-120):** The `buffer` array is only ever written to by `fgets(buffer, sizeof(buffer), stdin)`, which strictly bounds writes to `BUFFER_SIZE` bytes including the null terminator. As defense-in-depth, `buffer[sizeof(buffer) - 1] = '\0'` guarantees termination, and all subsequent length calculations use `strnlen` with an explicit bound rather than `strlen`. No unbounded copy functions (`strcpy`, `gets`, `sprintf`) are used, so the CWE-120 risk is mitigated.

2. **`getchar()` in a loop (CWE-120/CWE-20):** The first loop only *discards* excess input characters after a truncated line — nothing is written to any buffer, so no overflow is possible. The loop terminates correctly on both `'\n'` and `EOF`, avoiding an infinite loop on closed stdin. The second `getchar()` reads a single character into an `int`, checks for `EOF` before use, and only then narrows to `char` — proper input validation per CWE-20.

3. **`remove_first_last` bounds safety:** The function validates all parameters, rejects unterminated strings (`len >= max_size`), uses `memchr` bounded by `len`, and the `memmove` size arguments (`len - pos`) correctly include the null terminator without reading or writing past the buffer.
