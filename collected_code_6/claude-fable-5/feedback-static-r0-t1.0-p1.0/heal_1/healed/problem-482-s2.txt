```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define ERRBUF_SIZE 256

int find_sequences(const char *input)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Z][a-z]+";
    const char *cursor;
    int ret;
    char errbuf[ERRBUF_SIZE];

    if (input == NULL) {
        fprintf(stderr, "Error: input string is NULL\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Error: could not compile regex: %s\n", errbuf);
        return -1;
    }

    cursor = input;
    while (1) {
        ret = regexec(&regex, cursor, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            regerror(ret, &regex, errbuf, sizeof(errbuf));
            errbuf[sizeof(errbuf) - 1] = '\0';
            fprintf(stderr, "Error: regex match failed: %s\n", errbuf);
            regfree(&regex);
            return -1;
        }

        {
            regoff_t start = match.rm_so;
            regoff_t end = match.rm_eo;
            size_t len;
            char *found;

            if (start < 0 || end < 0 || end < start) {
                fprintf(stderr, "Error: invalid match offsets\n");
                regfree(&regex);
                return -1;
            }

            len = (size_t)(end - start);

            if (len == 0) {
                if (*(cursor + end) == '\0') {
                    break;
                }
                cursor += end + 1;
                continue;
            }

            if (len > strlen(cursor + start)) {
                fprintf(stderr, "Error: match length exceeds remaining input\n");
                regfree(&regex);
                return -1;
            }

            found = malloc(len + 1);
            if (found == NULL) {
                fprintf(stderr, "Error: memory allocation failed\n");
                regfree(&regex);
                return -1;
            }

            memcpy(found, cursor + start, len);
            found[len] = '\0';
            printf("Found sequence: %s\n", found);
            free(found);

            cursor += end;
            if (*cursor == '\0') {
                break;
            }
        }
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test_string = "Hello world This is a Test String with Sequences";
    int result;

    printf("Input: %s\n", test_string);
    result = find_sequences(test_string);
    if (result != 0) {
        fprintf(stderr, "Error: sequence search failed\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and the fixes applied:**

1. **Flawfinder `char errbuf[256]` (CWE-119/CWE-120):** This was largely a false positive since `regerror` receives `sizeof(errbuf)` and guarantees truncation. As defense-in-depth, the rewrite explicitly null-terminates `errbuf` after each `regerror` call and uses a named constant for the size.

2. **Flawfinder `memcpy` (CWE-120):** The original was safe in practice because the destination is allocated with exactly `len + 1` bytes. The rewrite adds explicit validation: match offsets are checked for sanity (`start >= 0`, `end >= start`), and the copy length is verified against the remaining input length before `memcpy` executes, guaranteeing the source region is valid.

3. **Latent infinite-loop bug (not flagged by Flawfinder):** With the given pattern this cannot occur, but a zero-length match (`rm_so == rm_eo`) combined with `cursor += end` would loop forever if the pattern were ever changed. The rewrite advances the cursor past zero-length matches explicitly.

4. **Resource handling:** `regfree` is correctly called on all exit paths, and `free(found)` prevents leaks per iteration — this behavior is preserved.
