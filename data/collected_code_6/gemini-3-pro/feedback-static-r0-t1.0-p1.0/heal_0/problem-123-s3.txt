#include <stdio.h>
#include <stdlib.h>

long sum_of_divisors(int n) {
    long sum = 1;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i * i != n) {
                sum += n / i;
            }
        }
    }
    return sum;
}

long sum_amicable_numbers(int limit) {
    if (limit < 1) {
        return 0;
    }

    long sum = 0;
    long *div_sums = (long *)calloc(limit + 1, sizeof(long));

    if (div_sums == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (int i = 2; i <= limit; i++) {
        div_sums[i] = sum_of_divisors(i);
    }

    for (int i = 2; i <= limit; i++) {
        long a = div_sums[i];
        if (a > i && a <= limit && div_sums[a] == i) {
            sum += i + a;
        }
    }

    free(div_sums);
    return sum;
}

int main(void) {
    int limit;

    if (scanf("%d", &limit) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    long result = sum_amicable_numbers(limit);
    printf("%ld\n", result);

    return EXIT_SUCCESS;
}
