```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define BINARY_BUFFER_SIZE (CHAR_BIT * sizeof(long) + 2)

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size);

static int decimal_to_binary(long decimal, char *buffer, size_t buffer_size)
{
    size_t index;
    size_t start;
    size_t end;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    index = 0;

    if (decimal < 0) {
        buffer[index] = '-';
        index++;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    start = index;

    if (value == 0UL) {
        if (index + 1 >= buffer_size) {
            return -1;
        }
        buffer[index] = '0';
        index++;
    } else {
        while (value > 0UL) {
            if (index + 1 >= buffer_size) {
                return -1;
            }
            buffer[index] = (char)('0' + (char)(value % 2UL));
            index++;
            value /= 2UL;
        }
    }

    if (index >= buffer_size) {
        return -1;
    }

    buffer[index] = '\0';

    end = index - 1;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    long decimal = 0;
    char binary[BINARY_BUFFER_SIZE] = {0};
    int scan_result;

    printf("Enter a decimal number: ");

    scan_result = scanf("%ld", &decimal);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(decimal, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary representation: %s\n", binary);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning (CWE-119/CWE-120) about the statically-sized `char` array is a low-risk, heuristic-based flag rather than an actual vulnerability in this code. The original source already mitigates it correctly, and the rewrite preserves those mitigations:

1. **Buffer sizing is provably sufficient**: `BINARY_BUFFER_SIZE` is `CHAR_BIT * sizeof(long) + 2`, which accounts for every bit of a `long` (worst case: `LONG_MIN` produces `sizeof(long) * CHAR_BIT` binary digits), plus one byte for the `'-'` sign and one for the null terminator.

2. **Explicit bounds checking**: Every write into the buffer inside `decimal_to_binary` is guarded by `index + 1 >= buffer_size` (or `index >= buffer_size` for the terminator), so no overflow is possible even if the buffer were undersized — the function fails safely with `-1` instead.

3. **The buffer size is passed explicitly** via `sizeof(binary)` at the call site, avoiding any mismatch between allocation and usage.

4. **`LONG_MIN` overflow avoided**: The negation trick `(unsigned long)(-(decimal + 1)) + 1UL` safely computes the magnitude of `LONG_MIN` without invoking undefined behavior from `-LONG_MIN`.

5. **Input validation**: `scanf`'s return value is checked, so uninitialized or garbage values never reach the conversion routine.

The code is already correct and defensively written; no functional changes were required.
