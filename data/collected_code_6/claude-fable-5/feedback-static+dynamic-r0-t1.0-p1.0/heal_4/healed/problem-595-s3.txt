```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_LEN 4096

static size_t bounded_len(const char *s, size_t max)
{
    size_t n = 0U;

    while (n < max && s[n] != '\0') {
        n++;
    }

    return n;
}

long min_swaps(const char *s1, const char *s2)
{
    size_t len1;
    size_t len2;
    size_t i;
    long count01;
    long count10;

    if (s1 == NULL || s2 == NULL) {
        return -1L;
    }

    len1 = bounded_len(s1, (size_t)MAX_LEN + 1U);
    len2 = bounded_len(s2, (size_t)MAX_LEN + 1U);

    if (len1 > (size_t)MAX_LEN || len2 > (size_t)MAX_LEN) {
        return -1L;
    }

    if (len1 != len2 || len1 == 0U) {
        return -1L;
    }

    count01 = 0L;
    count10 = 0L;

    for (i = 0U; i < len1; i++) {
        if ((s1[i] != '0' && s1[i] != '1') ||
            (s2[i] != '0' && s2[i] != '1')) {
            return -1L;
        }
        if (s1[i] == '0' && s2[i] == '1') {
            count01++;
        } else if (s1[i] == '1' && s2[i] == '0') {
            count10++;
        }
    }

    if (count01 != count10) {
        return -1L;
    }

    return count01;
}

static int read_line(char *buf, size_t size)
{
    size_t len;

    if (buf == NULL || size < 2U) {
        return -1;
    }

    if (fgets(buf, (int)size, stdin) == NULL) {
        return -1;
    }

    buf[size - 1U] = '\0';

    len = bounded_len(buf, size);
    while (len > 0U && (buf[len - 1U] == '\n' || buf[len - 1U] == '\r')) {
        buf[len - 1U] = '\0';
        len--;
    }

    if (len == 0U) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char a[MAX_LEN + 2];
    char b[MAX_LEN + 2];
    long result;
    int status;

    memset(a, 0, sizeof(a));
    memset(b, 0, sizeof(b));

    status = EXIT_SUCCESS;

    if (read_line(a, sizeof(a)) != 0 || read_line(b, sizeof(b)) != 0) {
        if (printf("Not Possible\n") < 0) {
            status = EXIT_FAILURE;
        }
        if (fflush(stdout) != 0) {
            status = EXIT_FAILURE;
        }
        return status;
    }

    result = min_swaps(a, b);

    if (result < 0L) {
        if (printf("Not Possible\n") < 0) {
            status = EXIT_FAILURE;
        }
    } else {
        if (printf("%ld\n", result) < 0) {
            status = EXIT_FAILURE;
        }
    }

    if (fflush(stdout) != 0) {
        status = EXIT_FAILURE;
    }

    return status;
}
```

Key changes made and rationale:

1. **Removed `strnlen`** — this was the most likely cause of the `<no output>` test failures. `strnlen` is a POSIX extension, not part of ISO C (C89/C99), so on a strictly conforming compiler or a non-POSIX toolchain the program fails to compile or link, producing no output at all for every test case. It is replaced with a small, bounds-checked `bounded_len` helper that performs the same job using only standard C.

2. **Bounds safety preserved (CWE-119/CWE-120)** — the Flawfinder warnings on the fixed-size buffers are mitigated as before: `fgets` is given the exact buffer size, the buffer is explicitly NUL-terminated at `size - 1`, all length scans are capped (`bounded_len`), and `min_swaps` rejects any input exceeding `MAX_LEN`, so no read or write can go out of bounds.

3. **Behavior otherwise unchanged** — the algorithm (mismatch counting: swaps possible iff `count01 == count10`, answer is `count01`), the `"Not Possible"` handling, the I/O error checking on `printf`/`fflush`, and the exit status logic are all retained, so test case 0 now yields `1` and the impossible cases correctly print `Not Possible`.
