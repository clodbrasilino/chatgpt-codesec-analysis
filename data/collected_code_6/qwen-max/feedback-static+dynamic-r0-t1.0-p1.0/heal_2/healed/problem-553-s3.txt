#include <stdio.h>
#include <stdlib.h>

double tuple_to_float(int *tuple, int size) {
    if (size != 2) {
        return 0.0;
    }
    double result = (double)tuple[0] + (double)(tuple[1] % 100) / 100.0;
    return result;
}

int main() {
    int test_cases[][2] = {{4, 56}, {7, 256}, {8, 123}};
    for (int i = 0; i < 3; i++) {
        int size = 2;
        double float_value = tuple_to_float(test_cases[i], size);
        printf("%.3f\n", float_value);
    }
    return 0;
}
