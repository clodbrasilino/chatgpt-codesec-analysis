#include <stdio.h>
#include <stdlib.h>

int find_smallest_missing(const int *arr, size_t n, int *result)
{
    unsigned char *seen;
    size_t i;
    int candidate;

    if (arr == NULL || result == NULL || n == 0U) {
        return -1;
    }

    seen = calloc(n + 1U, sizeof(unsigned char));
    if (seen == NULL) {
        return -1;
    }

    for (i = 0U; i < n; i++) {
        if (arr[i] >= 0 && (size_t)arr[i] <= n) {
            seen[arr[i]] = 1U;
        }
    }

    candidate = 0;
    for (i = 0U; i <= n; i++) {
        if (seen[i] == 0U) {
            candidate = (int)i;
            break;
        }
    }

    free(seen);
    *result = candidate;
    return 0;
}

int main(void)
{
    int arr[] = {0, 1, 2, 6, 9, 11, 15};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int missing = 0;

    if (find_smallest_missing(arr, n, &missing) != 0) {
        fprintf(stderr, "Error: could not compute smallest missing number\n");
        return EXIT_FAILURE;
    }

    if (printf("Smallest missing number: %d\n", missing) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
