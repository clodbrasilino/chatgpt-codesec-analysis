```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int min_of_two(int a, int b) {
    return (a < b) ? a : b;
}

int main(void) {
    int num1 = 0;
    int num2 = 0;
    int result = 0;
    int scan_result = 0;
    
    printf("Enter two integers: ");
    scan_result = scanf("%d %d", &num1, &num2);
    
    if (scan_result != 2) {
        fprintf(stderr, "Error: Invalid input\n");
        return EXIT_FAILURE;
    }
    
    result = min_of_two(num1, num2);
    printf("Minimum: %d\n", result);
    
    return EXIT_SUCCESS;
}
```
