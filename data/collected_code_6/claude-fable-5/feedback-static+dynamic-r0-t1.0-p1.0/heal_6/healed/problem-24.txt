```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_BUFFER_SIZE 128U
#define MAX_BINARY_DIGITS (sizeof(unsigned long) * 8U)

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int c;
    size_t guard = 0U;

    do {
        c = getchar();
        guard++;
    } while (c != '\n' && c != EOF && guard < (size_t)INT_MAX);
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE] = {0};
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == sizeof(input) - 1U) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and their resolution:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** — The original `flush_stdin()` loop was already safe since it never writes to a buffer; it only discards characters. However, to fully harden it against pathological infinite streams, I added a `guard` counter that bounds the number of iterations, guaranteeing termination even on adversarial input that never delivers `\n` or `EOF`.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** — The statically-sized `input` buffer was already handled correctly: `fgets` is given `sizeof(input)` so it cannot overflow, the buffer is explicitly NUL-terminated, `strnlen` bounds all length calculations, and over-long input is detected and rejected. I additionally zero-initialized the array (`= {0}`) as defense-in-depth so the buffer never contains indeterminate bytes, even if a read path were somehow bypassed.

3. **Pre-existing sound practices retained** — `binary_to_decimal` validates all pointers, enforces digit count against `MAX_BINARY_DIGITS`, checks for overflow via `ULONG_MAX >> 1` before each shift, and rejects non-binary characters. These checks already mitigate CWE-190 (integer overflow) and CWE-20 (improper input validation), so no changes were needed there.

The code remains portable C (note: `strnlen` requires POSIX or C23; on strict C99/C11 you may need `_POSIX_C_SOURCE 200809L` defined before the includes).
