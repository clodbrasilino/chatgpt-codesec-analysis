```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT 256
#define MAX_MATCHES 64

typedef struct {
    char text[MAX_INPUT];
} Match;

int find_sequences(const char *input, Match *matches, size_t max_matches, size_t *count)
{
    size_t i;
    size_t len;
    size_t found;

    if (input == NULL || matches == NULL || count == NULL || max_matches == 0) {
        return -1;
    }

    *count = 0;
    found = 0;
    len = strnlen(input, MAX_INPUT);
    i = 0;

    while (i < len && found < max_matches) {
        if (isupper((unsigned char)input[i]) &&
            (i + 1 < len) &&
            islower((unsigned char)input[i + 1])) {
            size_t start = i;
            size_t j = i + 1;
            size_t match_len;
            size_t dest_size;

            while (j < len && islower((unsigned char)input[j])) {
                j++;
            }

            match_len = j - start;
            dest_size = sizeof(matches[found].text);

            if (match_len >= dest_size) {
                match_len = dest_size - 1;
            }

            if (match_len > 0 && start + match_len <= len) {
                memcpy(matches[found].text, &input[start], match_len);
                matches[found].text[match_len] = '\0';
                found++;
            }

            i = j;
        } else {
            i++;
        }
    }

    *count = found;
    return 0;
}

int main(void)
{
    char input[MAX_INPUT];
    Match matches[MAX_MATCHES];
    size_t count;
    size_t len;

    printf("Enter a string: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    }

    if (find_sequences(input, matches, MAX_MATCHES, &count) != 0) {
        fprintf(stderr, "Error processing input\n");
        return EXIT_FAILURE;
    }

    if (count == 0) {
        printf("No matches found\n");
    } else {
        printf("Found %zu match(es):\n", count);
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", matches[i].text);
        }
    }

    return EXIT_SUCCESS;
}
```

Key points about the rewrite:

1. **`char text[MAX_INPUT]` array (CWE-119/CWE-120)**: The statically-sized buffers are retained, but every write into them is strictly bounds-checked. The destination size is taken via `sizeof(matches[found].text)` rather than a macro, so the check cannot drift out of sync with the actual buffer size.

2. **`memcpy` (CWE-120)**: The copy length is clamped to `dest_size - 1` before the call, guaranteeing room for the null terminator. An additional guard (`start + match_len <= len`) ensures the source range never reads past the input's measured length, and zero-length copies are skipped.

3. **Input handling in `main`**: `fgets` limits the read to `sizeof(input)`, the buffer is explicitly null-terminated, and `strnlen` (rather than `strlen`) bounds the length computation. The trailing newline is stripped safely.

4. **Defensive additions**: `max_matches == 0` is rejected up front to prevent any write into `matches[0]` when the caller provides no capacity, and `fflush(stdout)` ensures the prompt appears before blocking on input.

The original code was already largely safe; the changes harden the invariants that the static analyzer could not verify on its own.
