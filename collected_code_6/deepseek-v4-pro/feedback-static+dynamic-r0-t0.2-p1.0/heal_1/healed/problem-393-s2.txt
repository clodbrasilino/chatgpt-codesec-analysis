#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct List {
    Node *head;
    int length;
} List;

int get_list_length(const List *list) {
    if (list == NULL) {
        return 0;
    }
    return list->length;
}

List *find_max_length_list(List **lists, int num_lists) {
    if (lists == NULL || num_lists <= 0) {
        return NULL;
    }

    List *max_list = lists[0];
    int max_length = get_list_length(max_list);

    for (int i = 1; i < num_lists; i++) {
        int current_length = get_list_length(lists[i]);
        if (current_length > max_length) {
            max_length = current_length;
            max_list = lists[i];
        }
    }

    return max_list;
}

Node *create_node(int data) {
    Node *new_node = (Node *)malloc(sizeof(Node));
    if (new_node == NULL) {
        return NULL;
    }
    new_node->data = data;
    new_node->next = NULL;
    return new_node;
}

List *create_list() {
    List *list = (List *)malloc(sizeof(List));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->length = 0;
    return list;
}

int append_node(List *list, int data) {
    if (list == NULL) {
        return -1;
    }

    Node *new_node = create_node(data);
    if (new_node == NULL) {
        return -1;
    }

    if (list->head == NULL) {
        list->head = new_node;
    } else {
        Node *current = list->head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = new_node;
    }

    list->length++;
    return 0;
}

void free_list(List *list) {
    if (list == NULL) {
        return;
    }

    Node *current = list->head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }

    free(list);
}

int main(void) {
    List *lists[3];
    int num_lists = 3;

    lists[0] = create_list();
    lists[1] = create_list();
    lists[2] = create_list();

    if (lists[0] == NULL || lists[1] == NULL || lists[2] == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        for (int i = 0; i < num_lists; i++) {
            if (lists[i] != NULL) {
                free_list(lists[i]);
            }
        }
        return 1;
    }

    append_node(lists[0], 1);
    append_node(lists[0], 2);
    append_node(lists[0], 3);

    append_node(lists[1], 10);
    append_node(lists[1], 20);
    append_node(lists[1], 30);
    append_node(lists[1], 40);
    append_node(lists[1], 50);

    append_node(lists[2], 100);
    append_node(lists[2], 200);

    const List *max_list = find_max_length_list(lists, num_lists);

    if (max_list != NULL) {
        printf("List with maximum length has %d nodes\n", max_list->length);
    } else {
        printf("No lists found\n");
    }

    for (int i = 0; i < num_lists; i++) {
        free_list(lists[i]);
    }

    return 0;
}
