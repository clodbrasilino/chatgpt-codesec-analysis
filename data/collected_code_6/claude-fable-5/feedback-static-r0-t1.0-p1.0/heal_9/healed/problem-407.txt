```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define DIGITS_BUF_SIZE 32

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void reverse_range(char *s, size_t lo, size_t hi)
{
    while (lo < hi) {
        swap_chars(&s[lo], &s[hi]);
        lo++;
        hi--;
    }
}

long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }

    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }

    if (j <= i || j >= len || i >= len) {
        return -1;
    }

    swap_chars(&digits[i], &digits[j]);
    reverse_range(digits, (size_t)(i + 1), (size_t)(len - 1));

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = { 12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL };
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t k;

    for (k = 0; k < count; k++) {
        long long res = next_bigger(inputs[k]);
        if (res < 0) {
            printf("next_bigger(%lld) = no larger number possible\n", inputs[k]);
        } else {
            printf("next_bigger(%lld) = %lld\n", inputs[k], res);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-hardened against the Flawfinder warning (CWE-119/CWE-120) regarding the statically-sized `digits` buffer. The rewritten version preserves all the existing mitigations, which are sound:

1. **Bounded write via `snprintf`**: The conversion of `n` to a string uses `snprintf(digits, sizeof(digits), ...)`, which guarantees no write beyond the buffer. A 64-bit `long long` needs at most 20 characters plus the NUL terminator, so `DIGITS_BUF_SIZE` of 32 is comfortably sufficient.

2. **Truncation detection**: The check `len < 0 || (size_t)len >= sizeof(digits)` catches both encoding errors and truncation, so the code never operates on a partially written buffer.

3. **Defensive zeroing**: `memset` before use ensures no reads of uninitialized memory.

4. **Index bounds validation**: The redundant guard `j <= i || j >= len || i >= len` before `swap_chars` prevents out-of-range access even if the scan logic were modified incorrectly later.

5. **Overflow-safe reconstruction**: The check `result > (LLONG_MAX - d) / 10` prevents signed integer overflow (undefined behavior) when reassembling the permuted digits — important for inputs near `LLONG_MAX` whose next permutation exceeds the representable range.

6. **Digit validation**: The `d < 0 || d > 9` check guards against any corruption of the buffer contents before arithmetic use.

No functional or security changes were required; the code as written already performs the bounds checking the Flawfinder comment requests.
