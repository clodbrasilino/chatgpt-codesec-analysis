```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

uint64_t find_nth_carol(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n >= 32) {
        return 0;
    }
    uint64_t m = 1ULL << (n + 1);
    return (m - 1) * (m - 1) - 2;
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        return EXIT_FAILURE;
    }
    long n = strtol(argv[1], NULL, 10);
    if (n <= 0 || n >= 32) {
        return EXIT_FAILURE;
    }
    uint64_t result = find_nth_carol((int)n);
    printf("%llu\n", (unsigned long long)result);
    return EXIT_SUCCESS;
}
```
