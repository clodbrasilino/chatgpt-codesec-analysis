```c
#include <stdio.h>
#include <limits.h>
#include <errno.h>

unsigned long long fibonacci(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }
    
    unsigned long long a = 0;
    unsigned long long b = 1;
    unsigned long long result = 0;
    
    for (int i = 2; i <= n; i++) {
        result = a + b;
        a = b;
        b = result;
    }
    
    return result;
}

int main(void) {
    int n;
    
    if (scanf("%d", &n) != 1) {
        return 1;
    }
    
    if (n < 0 || n > 93) {
        return 1;
    }
    
    printf("%llu\n", fibonacci(n));
    
    return 0;
}
```
