```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

char *remove_parentheses(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    int depth;
    char *result;
    char *shrunk;

    if (input == NULL) {
        return NULL;
    }

    len = 0;
    while (len < MAX_INPUT_LEN && input[len] != '\0') {
        len++;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    depth = 0;
    j = 0;
    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    shrunk = realloc(result, j + 1);
    if (shrunk != NULL) {
        result = shrunk;
    }

    return result;
}

int main(void)
{
    const char *tests[] = {
        "Hello (remove me) World",
        "Nested (outer (inner) text) example",
        "No parentheses here"
    };
    size_t i;
    size_t count;

    count = sizeof(tests) / sizeof(tests[0]);

    for (i = 0; i < count; i++) {
        char *output = remove_parentheses(tests[i]);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        printf("%s\n", output);
        free(output);
    }

    return EXIT_SUCCESS;
}
```

The key fixes applied:

1. **Garbage output (test case failures):** The most likely culprit was `strnlen()`. It is a POSIX function, not standard C89/C90/C99, so without defining `_POSIX_C_SOURCE` (or `_GNU_SOURCE`) before the includes, the compiler may treat it as an implicitly declared function returning `int`. On 64-bit platforms, that truncates/corrupts the returned `size_t` length, leading to a bogus `len`, an undersized or oversized allocation, and garbage strings being returned. I replaced it with a portable, bounded manual scan that behaves identically but relies only on standard C.

2. **Variable scope (`variableScope` warning):** `output` is now declared inside the loop body, reducing its scope to where it is actually used and eliminating the need for the defensive `output = NULL;` reset after `free()`.

All other logic (depth tracking for nested parentheses, null-termination, defensive `realloc` shrink that keeps the original buffer on failure, and per-iteration `free`) was already correct and is preserved.
