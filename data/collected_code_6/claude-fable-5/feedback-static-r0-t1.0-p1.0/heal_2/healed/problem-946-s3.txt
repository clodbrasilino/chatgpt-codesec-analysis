```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <ctype.h>

#define MAX_TEXT_LEN 1024

typedef struct {
    unsigned char element;
    size_t count;
} ElementCount;

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t i = 0;

    while (i < max_len && s[i] != '\0') {
        i++;
    }
    return i;
}

int find_most_common(const char *text, size_t text_size, ElementCount *results, size_t max_results, size_t *num_results)
{
    size_t counts[UCHAR_MAX + 1] = {0};
    size_t max_count = 0;
    size_t i;
    size_t len;
    size_t found = 0;

    if (text == NULL || results == NULL || num_results == NULL || max_results == 0 || text_size == 0) {
        return -1;
    }

    len = bounded_strlen(text, text_size);
    if (len == text_size) {
        return -1;
    }
    if (len == 0) {
        *num_results = 0;
        return 0;
    }

    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)text[i];
        counts[c]++;
        if (counts[c] > max_count) {
            max_count = counts[c];
        }
    }

    for (i = 0; i <= UCHAR_MAX; i++) {
        if (counts[i] == max_count) {
            if (found < max_results) {
                results[found].element = (unsigned char)i;
                results[found].count = counts[i];
                found++;
            } else {
                break;
            }
        }
    }

    *num_results = found;
    return 0;
}

int main(void)
{
    char text[MAX_TEXT_LEN];
    ElementCount results[UCHAR_MAX + 1];
    size_t num_results = 0;
    size_t i;
    size_t len;

    printf("Enter text: ");
    if (fgets(text, sizeof(text), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }
    text[sizeof(text) - 1] = '\0';

    len = bounded_strlen(text, sizeof(text));
    if (len > 0 && text[len - 1] == '\n') {
        text[len - 1] = '\0';
    }

    if (find_most_common(text, sizeof(text), results, UCHAR_MAX + 1, &num_results) != 0) {
        fprintf(stderr, "Error processing text\n");
        return EXIT_FAILURE;
    }

    if (num_results == 0) {
        printf("No elements found\n");
        return EXIT_SUCCESS;
    }

    printf("Most common element(s):\n");
    for (i = 0; i < num_results; i++) {
        if (isprint(results[i].element)) {
            printf("'%c' : %zu\n", results[i].element, results[i].count);
        } else {
            printf("0x%02X : %zu\n", results[i].element, results[i].count);
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the original code and fixes applied:**

The Flawfinder warning (CWE-119/CWE-120) about the statically-sized `char text[MAX_TEXT_LEN]` array was already mitigated in the original code through several defensive measures, which I've preserved:

1. **Bounded input reading**: `fgets(text, sizeof(text), stdin)` limits input to the buffer size, preventing buffer overflow on read.
2. **Guaranteed null termination**: `text[sizeof(text) - 1] = '\0'` is defensive redundancy (fgets already null-terminates, but this protects against future refactoring).
3. **Bounded length calculation**: `bounded_strlen()` never reads past `max_len`, unlike bare `strlen()` which could run off the end of a non-terminated buffer.
4. **Size propagation**: `find_most_common()` receives `text_size` and validates that the string is terminated within bounds (`len == text_size` returns an error).
5. **Result buffer bounds**: The `found < max_results` check prevents writing past the `results` array.

**Improvement added**: The output loop now checks `isprint()` before printing the element as a character. In the original, a most-common non-printable byte (e.g., a tab or control character in pasted input) would be printed raw via `%c`, which can corrupt terminal output or enable terminal escape injection. Non-printable bytes are now displayed as hex.

The overall structure — parameter validation, `size_t` counters avoiding signed overflow, `unsigned char` indexing into a `UCHAR_MAX + 1` table to avoid negative-index UB with `char` — was already sound and remains unchanged.
