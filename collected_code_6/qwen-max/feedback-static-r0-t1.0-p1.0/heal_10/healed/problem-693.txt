#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

void remove_multiple_spaces(char *str) {
    size_t str_len = strlen(str);
    regex_t reg;
    const char *pattern = "\\s+";
    const char *replacement = " ";
    size_t replacement_length = strlen(replacement);

    if (regcomp(&reg, pattern, REG_EXTENDED) != 0) {
        fprintf(stderr, "Regex compilation failed\n");
        exit(EXIT_FAILURE);
    }

    int nmatch = 1;
    regmatch_t pmatch[nmatch];

    while (regexec(&reg, str, nmatch, pmatch, 0) == 0) {
        size_t length_before = pmatch[0].rm_so;
        size_t remaining_length = pmatch[0].rm_eo - pmatch[0].rm_so + (str_len - pmatch[0].rm_eo);

        memmove(str + length_before + replacement_length, str + pmatch[0].rm_eo, remaining_length + 1);
        memcpy(str + length_before, replacement, replacement_length);
    }

    regfree(&reg);
}

int main() {
    char *input = strdup("This   is  an example   text  with  multiple   spaces.");
    if (input == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    remove_multiple_spaces(input);
    printf("%s\n", input);
    free(input);
    return 0;
}
