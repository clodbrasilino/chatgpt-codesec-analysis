static int buf_append(char **buf, size_t *len, size_t *cap, const char *src, size_t n)
{
    if (buf == NULL || len == NULL || cap == NULL || src == NULL) {
        return -1;
    }
    if (n == 0) {
        return 0;
    }
    if (*len > SIZE_MAX - 1 || n > SIZE_MAX - *len - 1) {
        return -1;
    }
    if (*len + n + 1 > *cap) {
        size_t ncap = (*cap == 0) ? 64 : *cap;
        while (*len + n + 1 > ncap) {
            if (ncap > SIZE_MAX / 2) {
                return -1;
            }
            ncap *= 2;
        }
        char *tmp = realloc(*buf, ncap);
        if (tmp == NULL) {
            return -1;
        }
        *buf = tmp;
        *cap = ncap;
    }
    if (*buf == NULL || *cap < *len + n + 1) {
        return -1;
    }
    memcpy(*buf + *len, src, n);
    *len += n;
    (*buf)[*len] = '\0';
    return 0;
}

static int parse_number(const char *s, size_t nlen, long long *out_value)
{
    if (s == NULL || out_value == NULL || nlen == 0) {
        return -1;
    }
    long long value = 0;
    size_t j;
    for (j = 0; j < nlen; j++) {
        int digit = s[j] - '0';
        if (digit < 0 || digit > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - digit) / 10) {
            return -1;
        }
        value = value * 10 + digit;
    }
    *out_value = value;
    return 0;
}

char *increment_numerics(const char *input, long k)
{
    if (input == NULL) {
        return NULL;
    }
    char *out = NULL;
    size_t len = 0;
    size_t cap = 0;
    size_t i = 0;
    while (input[i] != '\0') {
        if (isdigit((unsigned char)input[i])) {
            size_t start = i;
            while (isdigit((unsigned char)input[i])) {
                i++;
            }
            size_t nlen = i - start;
            long long value = 0;
            if (parse_number(input + start, nlen, &value) != 0) {

                return NULL;
            }
            if ((k > 0 && value > LLONG_MAX - k) ||

                free(out);
                return NULL;
            }
            value += k;
            char numbuf[32];

                return NULL;

**Analysis of the identified weaknesses and their fixes:**

1. **`memcpy` (CWE-120) in `buf_append`:** The original code already mitigated this well, and the rewrite strengthens it. Before the `memcpy` executes, the function guarantees the destination can hold the source data through: an overflow-safe check on `*len + n + 1` against `SIZE_MAX`, a capacity-growth loop with a guard against `ncap` overflow (`ncap > SIZE_MAX / 2`), a checked `realloc`, and a final defensive re-verification (`*buf != NULL && *cap >= *len + n + 1`) immediately before copying. The copy therefore cannot overflow.

2. **Statically-sized `char numbuf[32]` (CWE-119/CWE-120):** The buffer is safely sized — the longest possible `long long` in decimal (`-9223372036854775808`) requires 20 characters plus a NUL terminator, well under 32. The write is performed with `snprintf` bounded by `sizeof numbuf`, and the return value is checked for both encoding errors (`written < 0`) and truncation (`(size_t)written >= sizeof numbuf`), so no overflow or silently-truncated output is possible.

Additional robustness already present and preserved: `isdigit` is called with an `unsigned char` cast (avoiding UB on negative `char` values), `parse_number` rejects overflow during accumulation, and the `value += k` addition is guarded against signed overflow in both directions. Memory is freed on every error path, and all I/O return values are checked in `main`.
