```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

#define MAX_INPUT_LEN 32

static bool has_thirty_days(const char *month)
{
    static const char *thirty_day_months[] = {
        "april", "june", "september", "november"
    };
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return false;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return false;
    }

    for (i = 0; i < len; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[len] = '\0';

    for (i = 0; i < sizeof(thirty_day_months) / sizeof(thirty_day_months[0]); i++) {
        if (strcmp(normalized, thirty_day_months[i]) == 0) {
            return true;
        }
    }

    return false;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else if (len == sizeof(input) - 1) {
        discard_remaining_input();
        fprintf(stderr, "Input too long.\n");
        return EXIT_FAILURE;
    }

    if (input[0] == '\0') {
        fprintf(stderr, "No month name provided.\n");
        return EXIT_FAILURE;
    }

    if (has_thirty_days(input)) {
        printf("%s has 30 days.\n", input);
    } else {
        printf("%s does not have 30 days.\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **`char normalized[MAX_INPUT_LEN]` (CWE-119/CWE-120)** — The Flawfinder warning is a false positive in practice. All writes are bounds-checked: `strnlen(month, MAX_INPUT_LEN)` caps the length, and the guard `len >= MAX_INPUT_LEN` rejects any input that would leave no room for the null terminator. The copy loop writes at most `MAX_INPUT_LEN - 1` characters plus the terminator, so no overflow is possible.

2. **`char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120)** — Input is read with `fgets(input, sizeof(input), stdin)`, which never writes beyond the buffer. The explicit `input[sizeof(input) - 1] = '\0'` guarantees termination, and oversized input (no newline found with a full buffer) is detected, drained via `discard_remaining_input()`, and rejected — preventing truncated-input logic errors and stale stdin data.

3. **`getchar()` in a loop (CWE-120/CWE-20)** — No buffer is written here; characters are read and discarded. The loop terminates on both `'\n'` and `EOF`, so it cannot spin forever on a closed stream. This is the standard safe idiom for flushing a line from stdin.

The only substantive change is replacing the `= {0}` initializers with `memset` in `main` (functionally equivalent, explicit) and removing the zero-initialization of `normalized`, which is unnecessary since every byte used is explicitly written before being read. The original code was already well-hardened; the Flawfinder hits are informational and mitigated by the existing bounds checks.
