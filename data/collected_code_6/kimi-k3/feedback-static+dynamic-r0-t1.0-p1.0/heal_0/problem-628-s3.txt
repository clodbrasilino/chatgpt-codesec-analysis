#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char *replace_spaces(const char *str)
{
    size_t space_count = 0;
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (str == NULL) {
        return NULL;
    }

    len = strlen(str);

    for (i = 0; i < len; i++) {
        if (str[i] == ' ') {
            space_count++;
        }
    }

    result = malloc(len + space_count * 2 + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (str[i] == ' ') {
            result[j++] = '%';
            result[j++] = '2';
            result[j++] = '0';
        } else {
            result[j++] = str[i];
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char input[256];
    char *output;

    printf("Enter a string: ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\n")] = '\0';

    output = replace_spaces(input);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    printf("Result: %s\n", output);

    free(output);
    output = NULL;

    return EXIT_SUCCESS;
}
