```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

static int find_min_rotations(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    char *doubled;
    int result;

    if (str == NULL) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len >= max_len) {
        return -1;
    }

    if (len > (SIZE_MAX - 1U) / 2U) {
        return -1;
    }

    if (len > (size_t)INT_MAX) {
        return -1;
    }

    doubled = malloc((2U * len) + 1U);
    if (doubled == NULL) {
        return -1;
    }

    memcpy(doubled, str, len);
    memcpy(doubled + len, str, len);
    doubled[2U * len] = '\0';

    result = -1;
    for (i = 1U; i <= len; i++) {
        if (strncmp(doubled + i, str, len) == 0) {
            result = (int)i;
            break;
        }
    }

    free(doubled);
    return result;
}

int main(void)
{
    char buffer[256];
    size_t len;
    int rotations;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
        len--;
    }

    if (len == 0U) {
        fprintf(stderr, "Empty string provided\n");
        return EXIT_FAILURE;
    }

    rotations = find_min_rotations(buffer, sizeof(buffer));
    if (rotations < 0) {
        fprintf(stderr, "Failed to compute rotations\n");
        return EXIT_FAILURE;
    }

    if (printf("Minimum rotations required: %d\n", rotations) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the flagged weaknesses:

1. **CWE-126 (`strlen` over-read):** Replaced both `strlen` calls with `strnlen`, bounding the scan by a known maximum length (`sizeof(buffer)` in `main`, and a `max_len` parameter passed into `find_min_rotations`). The buffer is also zero-initialized with `memset` and explicitly NUL-terminated after `fgets` as defense-in-depth, guaranteeing termination even in pathological cases.

2. **CWE-120 (`memcpy` overflow):** The destination size is computed from the same `len` used in both `memcpy` calls (`2*len + 1`), with an overflow guard against `SIZE_MAX` before the multiplication, so the destination provably holds the source data. This was already correct in the original; the rewrite preserves and tightens it with the additional `len >= max_len` and `INT_MAX` checks (the latter prevents the `(int)i` cast from overflowing).

3. **CWE-119/CWE-120 (statically-sized array):** All reads into `buffer` remain bounded by `fgets(buffer, sizeof(buffer), ...)`, and every subsequent operation on it is length-limited via `strnlen` and the forced terminator, so no index can exceed the array bounds.

4. **Latent bug fix:** the original code used `SIZE_MAX` without including `<stdint.h>`, which is now included so the code compiles portably under C99/C11.
