```c
#include <stdio.h>
#include <stdlib.h>

int compare(const void *a, const void *b) {
    return (*(int *)b - *(int *)a);
}

int maxPairSum(int arr[], int n, int k) {
    qsort(arr, n, sizeof(int), compare);
    int sum = 0;
    for (int i = 0; i < 2 * k && i < n; i += 2) {
        sum += arr[i];
    }
    return sum;
}

int main() {
    int arr[] = {1, 3, 8, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int k = 2;
    printf("%d\n", maxPairSum(arr, n, k));
    return 0;
}
```
