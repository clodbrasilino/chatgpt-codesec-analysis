```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

long long sum_of_absolute_differences(int* arr, int n) {
    if (arr == NULL || n <= 0) {
        return 0;
    }

    long long sum = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            sum += llabs((long long)arr[i] - (long long)arr[j]);
        }
    }
    return sum;
}

int main() {
    int arr[] = {1, 2, 3, 4};
    int n = sizeof(arr) / sizeof(arr[0]);

    long long result = sum_of_absolute_differences(arr, n);
    printf("Sum of absolute differences: %lld\n", result);

    return 0;
}
```
