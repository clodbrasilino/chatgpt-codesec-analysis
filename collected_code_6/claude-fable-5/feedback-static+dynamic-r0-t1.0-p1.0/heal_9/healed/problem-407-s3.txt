#include <stdlib.h>
#include <string.h>
#include <limits.h>
#include <errno.h>

#define DIGITS_BUF_SIZE 32

static long long next_bigger(long long n)
{
    char digits[DIGITS_BUF_SIZE];
    int len;
    int i;
    int j;
    long long result;
    char tmp;
    char *endptr;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    i = len - 2;
    while (i >= 0 && digits[i] >= digits[i + 1]) {
        i--;
    }
    if (i < 0) {
        return -1;
    }

    j = len - 1;
    while (j > i && digits[j] <= digits[i]) {
        j--;
    }
    if (j <= i || j >= len) {
        return -1;
    }

    tmp = digits[i];
    digits[i] = digits[j];
    digits[j] = tmp;

    for (int a = i + 1, b = len - 1; a < b && a >= 0 && b < len; a++, b--) {
        tmp = digits[a];
        digits[a] = digits[b];
        digits[b] = tmp;
    }

    if ((size_t)len < sizeof(digits)) {
        digits[len] = '\0';
    } else {
        return -1;
    }

    errno = 0;
    result = strtoll(digits, &endptr, 10);
    if (errno == ERANGE || endptr == digits || *endptr != '\0') {
        return -1;
    }

    if (result <= n) {
        return -1;
    }

    return result;
}

{

            if (printf("next_bigger(%lld) = no bigger number exists\n", inputs[k]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("next_bigger(%lld) = %lld\n", inputs[k], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}

**Analysis of the flagged weakness (CWE-119/CWE-120):**

The Flawfinder warning on the statically-sized `digits` buffer is a low-risk, largely informational finding here, and the code already contains all the mitigations required to neutralize it:

1. **Bounded write via `snprintf`** — the buffer is only ever written through `snprintf(digits, sizeof(digits), ...)`, which cannot overflow. The return value is validated both for negative (encoding error) and for truncation (`len >= sizeof(digits)`), rejecting any input that would not fit.

2. **Sufficient buffer size** — a `long long` is at most 64 bits; its decimal representation needs at most 20 characters plus a NUL. `DIGITS_BUF_SIZE` of 32 comfortably exceeds the worst case.

3. **Index bounds checking** — every loop index (`i`, `j`, `a`, `b`) is constrained within `[0, len)` before any array access, and the NUL terminator write at `digits[len]` is guarded by an explicit bounds check.

4. **Defensive initialization** — `memset` zeroes the buffer before use, guaranteeing NUL termination even in unexpected paths.

5. **Safe parsing** — `strtoll` is used with `errno`/`endptr` validation instead of `atoll`, catching overflow and malformed content.

Since the write is length-limited, the size is provably larger than the maximum possible output, and all indices are bounds-checked, the code is already compliant with the recommended remediations; no functional changes were needed beyond preserving those checks.
