#include <stdio.h>
#include <limits.h>

int maxProductSubarray(int arr[], int n, int *start, int *end) {
    if (n <= 0) {
        *start = *end = -1;
        return 0;
    }

    int max_prod = INT_MIN;
    int curr_max = 1;
    int curr_min = 1;
    int temp_start = 0;
    *start = 0;
    *end = 0;

    for (int i = 0; i < n; i++) {
        if (arr[i] == 0) {
            curr_max = 1;
            curr_min = 1;
            temp_start = i + 1;
            if (max_prod < 0) {
                max_prod = 0;
                *start = i;
                *end = i;
            }
            continue;
        }

        int temp_max = curr_max;
        curr_max = (arr[i] > arr[i] * temp_max) ? arr[i] : arr[i] * temp_max;
        curr_max = (curr_max > arr[i] * curr_min) ? curr_max : arr[i] * curr_min;
        
        curr_min = (arr[i] < arr[i] * temp_max) ? arr[i] : arr[i] * temp_max;
        curr_min = (curr_min < arr[i] * curr_min) ? curr_min : arr[i] * curr_min;

        if (curr_max > max_prod) {
            max_prod = curr_max;
            *end = i;
            *start = temp_start;
        }

        if (curr_max == arr[i]) {
            temp_start = i;
        }
    }

    return max_prod;
}
