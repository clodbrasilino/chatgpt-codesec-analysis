#include <stdio.h>
#include <limits.h>
#include <stddef.h>

int find_min(const int *arr, size_t size) {
    if (arr == NULL || size == 0) {
        return INT_MIN;
    }

    int min_val = arr[0];
    for (size_t i = 1; i < size; ++i) {
        if (arr[i] < min_val) {
            min_val = arr[i];
        }
    }

    return min_val;
}

int main(void) {
    int list[] = {34, 15, 88, 2, 19, 5};
    size_t size = sizeof(list) / sizeof(list[0]);

    int result = find_min(list, size);

    printf("%d\n", result);

    return 0;
}
