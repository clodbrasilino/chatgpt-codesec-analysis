```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_MATCH_GROUPS 2
#define MAX_INPUT_LEN 4096
#define ERRBUF_LEN 256

static void print_regex_error(int ret, const regex_t *regex, const char *context)
{
    char errbuf[ERRBUF_LEN];
    size_t needed;

    memset(errbuf, 0, sizeof(errbuf));
    needed = regerror(ret, regex, errbuf, sizeof(errbuf) - 1U);
    if (needed >= sizeof(errbuf)) {
        errbuf[sizeof(errbuf) - 1U] = '\0';
    }
    errbuf[sizeof(errbuf) - 1U] = '\0';
    fprintf(stderr, "%s: %s\n", context, errbuf);
}

static char *extract_quoted_value(const regex_t *regex, const char *input,
                                  size_t input_len, size_t *offset)
{
    regmatch_t matches[MAX_MATCH_GROUPS];
    char *result = NULL;
    int ret;

    if (regex == NULL || input == NULL || offset == NULL) {
        return NULL;
    }

    if (input_len >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (*offset >= input_len) {
        return NULL;
    }

    ret = regexec(regex, input + *offset, MAX_MATCH_GROUPS, matches, 0);
    if (ret == 0) {
        regoff_t start = matches[1].rm_so;
        regoff_t end = matches[1].rm_eo;

        if (start >= 0 && end >= start &&
            (size_t)end <= input_len - *offset) {
            size_t len = (size_t)(end - start);

            if (len < MAX_INPUT_LEN) {
                result = malloc(len + 1U);
                if (result != NULL) {
                    if (len > 0U) {
                        memcpy(result, input + *offset + (size_t)start, len);
                    }
                    result[len] = '\0';
                    if (matches[0].rm_eo >= 0 &&
                        (size_t)matches[0].rm_eo <= input_len - *offset) {
                        *offset += (size_t)matches[0].rm_eo;
                    } else {
                        *offset = input_len;
                    }
                } else {
                    fprintf(stderr, "Memory allocation failed\n");
                }
            }
        }
    } else if (ret != REG_NOMATCH) {
        print_regex_error(ret, regex, "Regex execution failed");
    }

    return result;
}

int main(void)
{
    const char *text = "Name: \"Alice\", City: \"Paris\", Job: \"Engineer\"";
    size_t offset = 0;
    size_t input_len;
    regex_t regex;
    int ret;

    input_len = strnlen(text, MAX_INPUT_LEN);
    if (input_len >= MAX_INPUT_LEN) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    ret = regcomp(&regex, "\"([^\"]*)\"", REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return EXIT_FAILURE;
    }

    printf("Input: %s\n", text);

    for (;;) {
        char *value = extract_quoted_value(&regex, text, input_len, &offset);
        if (value == NULL) {
            break;
        }
        printf("Extracted: %s\n", value);
        free(value);
    }

    regfree(&regex);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and how they are addressed:**

1. **`errbuf` static array (CWE-119/CWE-120):** The buffer is zero-initialized with `memset`, `regerror` is called with `sizeof(errbuf) - 1U` to reserve room for the terminator, and NUL-termination is explicitly enforced afterward. The return value of `regerror` is now checked to detect truncation rather than being discarded. There is no path where the buffer can overflow.

2. **`memcpy` (CWE-120):** The copy is safe by construction:
   - `len` is derived from validated match offsets (`start >= 0`, `end >= start`).
   - The bound check `(size_t)end <= input_len - *offset` guarantees the source region lies within the input string.
   - The destination is allocated as `malloc(len + 1U)` and the allocation is checked for `NULL` before use, so the destination always holds exactly `len` bytes plus the terminator.
   - `len < MAX_INPUT_LEN` caps the allocation size, preventing pathological allocations.

3. **Additional hardening applied:** an explicit `matches[0].rm_eo >= 0` check was added before the cast to `size_t`, closing a subtle signed-to-unsigned conversion issue where a negative `rm_eo` (theoretically possible with a corrupted match structure) would wrap to a huge unsigned value and incorrectly pass the bounds test.

The code correctly uses `strnlen` to bound input measurement, validates all pointer parameters, frees each extracted value, and releases the compiled regex with `regfree` — no leaks or unbounded operations remain.
