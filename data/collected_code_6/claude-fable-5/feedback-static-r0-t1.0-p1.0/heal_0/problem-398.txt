#include <stdio.h>
#include <stdlib.h>

unsigned int digit_sum(int number)
{
    unsigned int magnitude;
    unsigned int sum = 0U;

    if (number < 0) {
        magnitude = (unsigned int)(-(number + 1)) + 1U;
    } else {
        magnitude = (unsigned int)number;
    }

    do {
        sum += magnitude % 10U;
        magnitude /= 10U;
    } while (magnitude > 0U);

    return sum;
}

int compute_digit_sums(const int *numbers, size_t count, unsigned int *results)
{
    size_t i;

    if ((numbers == NULL) || (results == NULL)) {
        return -1;
    }

    for (i = 0U; i < count; i++) {
        results[i] = digit_sum(numbers[i]);
    }

    return 0;
}

int main(void)
{
    int numbers[] = { 123, -456, 0, 9875, 42 };
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    unsigned int *results;
    size_t i;
    int status;

    results = (unsigned int *)malloc(count * sizeof(unsigned int));
    if (results == NULL) {
        (void)fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    status = compute_digit_sums(numbers, count, results);
    if (status != 0) {
        (void)fprintf(stderr, "Failed to compute digit sums\n");
        free(results);
        return EXIT_FAILURE;
    }

    for (i = 0U; i < count; i++) {
        if (printf("Digit sum of %d is %u\n", numbers[i], results[i]) < 0) {
            free(results);
            return EXIT_FAILURE;
        }
    }

    free(results);
    return EXIT_SUCCESS;
}
