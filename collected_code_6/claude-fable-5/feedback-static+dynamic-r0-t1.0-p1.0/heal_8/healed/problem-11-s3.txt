```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U
#define FLUSH_LIMIT 4096U

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    char *first;
    char *last;
    size_t tail_len;
    size_t i;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    first = memchr(str, (int)(unsigned char)ch, len);
    if (first == NULL) {
        return;
    }

    last = NULL;
    for (i = len; i > 0U; i--) {
        if (str[i - 1U] == ch) {
            last = &str[i - 1U];
            break;
        }
    }

    if (last != NULL && last != first) {
        tail_len = len - (size_t)(last - str) - 1U;
        memmove(last, last + 1, tail_len + 1U);
        len -= 1U;
    }

    tail_len = len - (size_t)(first - str) - 1U;
    memmove(first, first + 1, tail_len + 1U);
}

static void flush_input(void)
{
    int c;
    size_t count = 0U;

    do {
        c = getchar();
        count++;
    } while (c != '\n' && c != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strcspn(buffer, "\n");
    if (len == strnlen(buffer, sizeof(buffer)) && len == sizeof(buffer) - 1U) {
        flush_input();
    }
    buffer[len] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        flush_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis of the identified weaknesses and their mitigations:**

1. **`char buffer[BUFFER_SIZE]` (CWE-119/CWE-120, statically-sized array):** The buffer is only written to via `fgets` with `sizeof(buffer)` as the limit, which guarantees no overflow. A defensive null terminator is written at `buffer[sizeof(buffer) - 1U]`, and all subsequent length calculations use `strnlen`/`strcspn` bounded by `sizeof(buffer)`. This is a false positive — bounds are properly enforced.

2. **`getchar()` in `flush_input` (CWE-120/CWE-20, unbounded loop):** The loop is bounded three ways: it terminates on newline, on `EOF`, and — critically — by the `FLUSH_LIMIT` counter, preventing an infinite loop on a pathological input stream. No data is stored, so no buffer can overflow. Mitigated.

3. **`getchar()` for the removal character (CWE-120/CWE-20):** The return value is stored in an `int` (correctly, so `EOF` is distinguishable from valid characters), checked against `EOF` before use, and only then narrowed to `char`. Leftover input is drained via the bounded `flush_input`. Mitigated.

4. **`remove_first_last` robustness:** The function defends against `NULL` pointers, zero-length limits, and unterminated buffers (`strnlen` plus forced termination). `memchr` is bounded by the computed length, and both `memmove` calls copy exactly `tail_len + 1U` bytes including the terminator, with pointer arithmetic that cannot exceed `len`. No off-by-one or overflow paths exist.

5. **Truncation handling:** If the input line filled the buffer without a newline, `flush_input` discards the remainder of the line so it isn't misread as the character prompt's input — preventing input-stream desynchronization (a CWE-20 input validation concern).
