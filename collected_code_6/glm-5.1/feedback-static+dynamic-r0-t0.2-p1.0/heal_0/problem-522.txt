#include <stdio.h>
#include <stdlib.h>

int longestBitonicSubsequence(int *arr, int n) {
    if (n == 0) {
        return 0;
    }

    int *lis = (int *)malloc(n * sizeof(int));
    int *lds = (int *)malloc(n * sizeof(int));

    if (lis == NULL || lds == NULL) {
        free(lis);
        free(lds);
        return 0;
    }

    for (int i = 0; i < n; i++) {
        lis[i] = 1;
        for (int j = 0; j < i; j++) {
            if (arr[j] < arr[i] && lis[j] + 1 > lis[i]) {
                lis[i] = lis[j] + 1;
            }
        }
    }

    for (int i = n - 1; i >= 0; i--) {
        lds[i] = 1;
        for (int j = n - 1; j > i; j--) {
            if (arr[j] < arr[i] && lds[j] + 1 > lds[i]) {
                lds[i] = lds[j] + 1;
            }
        }
    }

    int max_len = 1;
    for (int i = 0; i < n; i++) {
        int current_len = lis[i] + lds[i] - 1;
        if (current_len > max_len) {
            max_len = current_len;
        }
    }

    free(lis);
    free(lds);

    return max_len;
}

int main() {
    int arr[] = {12, 11, 40, 5, 3, 1};
    int n = sizeof(arr) / sizeof(arr[0]);

    int result = longestBitonicSubsequence(arr, n);

    printf("%d\n", result);

    return 0;
}
