```c
#include <stdio.h>
#include <string.h>
#include <limits.h>
#include <stddef.h>

#define INPUT_BUFFER_SIZE 128U
#define MAX_BINARY_DIGITS (sizeof(unsigned long) * 8U)
#define FLUSH_GUARD_LIMIT 4096U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > MAX_BINARY_DIGITS) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static void flush_stdin(void)
{
    int c;
    size_t guard = 0U;

    do {
        c = getchar();
        guard++;
    } while (c != '\n' && c != EOF && guard < FLUSH_GUARD_LIMIT);
}

int main(void)
{
    char input[INPUT_BUFFER_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else if (len == sizeof(input) - 1U) {
        flush_stdin();
        fprintf(stderr, "Error: input too long.\n");
        return 1;
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

**Analysis of the flagged weaknesses and how the code addresses them:**

1. **Flawfinder `getchar` warning (CWE-120, CWE-20)** — The `flush_stdin()` function already mitigates this. The `getchar()` return value is stored in an `int` (correctly, so `EOF` is distinguishable from valid characters), no data is written into any buffer, and a `guard` counter bounded by `FLUSH_GUARD_LIMIT` prevents an unbounded loop on a pathological or malicious input stream. This is a false positive in practice; the guard makes it defensively robust.

2. **Flawfinder `char` array warning (CWE-119/CWE-120)** — The statically sized `input` buffer is safely handled:
   - `fgets()` is called with `sizeof(input)`, guaranteeing no overflow.
   - Explicit null termination at `input[sizeof(input) - 1U]` is a belt-and-braces measure.
   - `strnlen()` is used instead of `strlen()`, bounding all length computations.
   - Overlong input (no newline found and buffer full) is detected, the remaining stream is drained, and the program exits with an error rather than silently truncating.

3. **Conversion safety** — `binary_to_decimal()` validates all pointers, rejects empty/oversized/non-terminated input, restricts characters to `'0'`/`'1'`, and checks `value > (ULONG_MAX >> 1)` before shifting to prevent unsigned overflow past the representable range.
