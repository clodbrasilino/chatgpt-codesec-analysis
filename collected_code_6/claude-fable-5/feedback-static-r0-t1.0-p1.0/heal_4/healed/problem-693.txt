```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int remove_multiple_spaces(const char *input, char *output, size_t output_size);

int remove_multiple_spaces(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    size_t out_pos = 0U;
    size_t remaining;
    int ret;

    if ((input == NULL) || (output == NULL) || (output_size == 0U)) {
        return -1;
    }

    output[0] = '\0';

    ret = regcomp(&regex, "  +", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t prefix_len;
        size_t space_left;

        if ((match.rm_so < 0) || (match.rm_eo <= match.rm_so)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        prefix_len = (size_t)match.rm_so;

        if (out_pos >= (output_size - 1U)) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        space_left = output_size - 1U - out_pos;

        if (prefix_len >= space_left) {
            regfree(&regex);
            output[0] = '\0';
            return -1;
        }

        memcpy(&output[out_pos], cursor, prefix_len);
        out_pos += prefix_len;
        output[out_pos] = ' ';
        out_pos++;

        cursor += match.rm_eo;
    }

    if (out_pos >= output_size) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    remaining = strnlen(cursor, output_size);

    if (remaining > (output_size - 1U - out_pos)) {
        regfree(&regex);
        output[0] = '\0';
        return -1;
    }

    memcpy(&output[out_pos], cursor, remaining);
    out_pos += remaining;
    output[out_pos] = '\0';

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *test = "This   is  a    string   with  multiple     spaces.";
    char result[256] = {0};

    if (remove_multiple_spaces(test, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Failed to process the string.\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        return EXIT_FAILURE;
    }

    if (printf("Cleaned:  %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and how they are addressed:**

1. **CWE-120 (first `memcpy`)**: Before this copy, the code verifies `out_pos < output_size - 1` and `prefix_len < space_left`, guaranteeing room for both the prefix bytes and the single replacement space plus the eventual NUL terminator. The copy is bounds-safe. I additionally added a defensive check on `match.rm_so`/`match.rm_eo` to reject degenerate match offsets before casting to `size_t`, which hardens the arithmetic that feeds the copy.

2. **CWE-120 (second `memcpy`)**: The length is measured with `strnlen` (bounded) rather than `strlen`, and the guard `remaining > output_size - 1 - out_pos` rejects anything that would not fit alongside the terminator. I moved the `out_pos >= output_size` check before computing `space` arithmetic so the subtraction `output_size - 1U - out_pos` can never underflow — in the original, both conditions were evaluated in one expression, which was safe only due to short-circuit ordering; splitting them makes the intent explicit and robust against refactoring.

3. **CWE-119/CWE-120 (`char result[256]`)**: The statically sized buffer is acceptable because its size is always passed via `sizeof(result)` and every write path in `remove_multiple_spaces` is bounds-checked against `output_size`, with the failure paths clearing the buffer and returning an error rather than truncating silently.

The remaining logic (NULL/zero-size parameter validation, `regcomp` failure handling, `regfree` on every exit path, and NUL-termination on both success and failure) was already correct and is preserved.
