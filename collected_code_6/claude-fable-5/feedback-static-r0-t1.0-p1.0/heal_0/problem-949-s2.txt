```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    long first;
    long second;
} Tuple;

static int count_digits(long value)
{
    int count = 0;

    if (value < 0) {
        value = -value;
    }

    if (value == 0) {
        return 1;
    }

    while (value > 0) {
        count++;
        value /= 10;
    }

    return count;
}

static int total_digits(const Tuple *t)
{
    return count_digits(t->first) + count_digits(t->second);
}

static int compare_tuples(const void *a, const void *b)
{
    const Tuple *ta = (const Tuple *)a;
    const Tuple *tb = (const Tuple *)b;
    int da = total_digits(ta);
    int db = total_digits(tb);

    if (da < db) {
        return -1;
    }
    if (da > db) {
        return 1;
    }
    return 0;
}

static int sort_tuples_by_digits(Tuple *list, size_t count)
{
    if (list == NULL || count == 0) {
        return -1;
    }

    qsort(list, count, sizeof(Tuple), compare_tuples);
    return 0;
}

static void print_tuples(const Tuple *list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        printf("(%ld, %ld)\n", list[i].first, list[i].second);
    }
}

int main(void)
{
    Tuple data[] = {
        {1234, 5678},
        {1, 2},
        {100, 200},
        {12, 3456},
        {0, 99999}
    };
    size_t count = sizeof(data) / sizeof(data[0]);

    printf("Before sorting:\n");
    print_tuples(data, count);

    if (sort_tuples_by_digits(data, count) != 0) {
        fprintf(stderr, "Error: invalid input to sort function\n");
        return EXIT_FAILURE;
    }

    printf("After sorting by total digits:\n");
    print_tuples(data, count);

    return EXIT_SUCCESS;
}
```
