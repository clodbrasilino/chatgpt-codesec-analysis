```c
#include <stdio.h>
#include <math.h>

double cylinder_surface_area(double radius, double height) {
    return 2 * M_PI * radius * (radius + height);
}

int main() {
    double radius, height;
    printf("Enter the radius and height of the cylinder: ");
    if (scanf("%lf %lf", &radius, &height) == 2) {
        if (radius > 0 && height > 0) {
            double area = cylinder_surface_area(radius, height);
            printf("The surface area of the cylinder is: %.2f\n", area);
        } else {
            printf("Radius and height must be positive.\n");
        }
    } else {
        printf("Invalid input. Please enter numerical values for radius and height.\n");
    }
    return 0;
}
```
