#include <stdio.h>
#include <stdlib.h>

typedef enum ElementType {
    ELEMENT_INT,
    ELEMENT_TUPLE
} ElementType;

typedef struct Tuple {
    int first;
    int second;
} Tuple;

typedef struct Node {
    ElementType type;
    union {
        int value;
        Tuple tuple;
    } data;
    struct Node *next;
} Node;

static Node *create_int_node(int value)
{
    Node *node = malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->type = ELEMENT_INT;
    node->data.value = value;
    node->next = NULL;
    return node;
}

static Node *create_tuple_node(int first, int second)
{
    Node *node = malloc(sizeof(Node));
    if (node == NULL) {
        return NULL;
    }
    node->type = ELEMENT_TUPLE;
    node->data.tuple.first = first;
    node->data.tuple.second = second;
    node->next = NULL;
    return node;
}

static int append_node(Node **head, Node *node)
{
    Node *current;

    if (head == NULL || node == NULL) {
        return -1;
    }
    if (*head == NULL) {
        *head = node;
        return 0;
    }
    current = *head;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = node;
    return 0;
}

static size_t count_until_tuple(const Node *head)
{
    size_t count = 0;
    const Node *current = head;

    while (current != NULL) {
        if (current->type == ELEMENT_TUPLE) {
            break;
        }
        count++;
        current = current->next;
    }
    return count;
}

static void free_list(Node *head)
{
    Node *current = head;
    Node *next;

    while (current != NULL) {
        next = current->next;
        free(current);
        current = next;
    }
}

int main(void)
{
    Node *head = NULL;
    Node *node;
    size_t count;
    int i;
    const int values[] = {10, 20, 30};

    for (i = 0; i < 3; i++) {
        node = create_int_node(values[i]);
        if (node == NULL) {
            free_list(head);
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }
        if (append_node(&head, node) != 0) {
            free(node);
            free_list(head);
            fprintf(stderr, "Failed to append node\n");
            return EXIT_FAILURE;
        }
    }

    node = create_tuple_node(1, 2);
    if (node == NULL) {
        free_list(head);
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    if (append_node(&head, node) != 0) {
        free(node);
        free_list(head);
        fprintf(stderr, "Failed to append node\n");
        return EXIT_FAILURE;
    }

    node = create_int_node(40);
    if (node == NULL) {
        free_list(head);
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    if (append_node(&head, node) != 0) {
        free(node);
        free_list(head);
        fprintf(stderr, "Failed to append node\n");
        return EXIT_FAILURE;
    }

    count = count_until_tuple(head);
    if (printf("Count until tuple: %zu\n", count) < 0) {
        free_list(head);
        return EXIT_FAILURE;
    }

    free_list(head);
    return EXIT_SUCCESS;
}
