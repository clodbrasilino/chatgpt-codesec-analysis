```c
#include <stdio.h>

int findKthElement(const int arr[], int n, int k) {
    if (k < 1 || k > n) {
        return -1;
    }
    return arr[k-1];
}

int main() {
    const int arr[] = {3, 8, 36};
    int n = sizeof(arr) / sizeof(arr[0]);

    for (int k = 1; k <= n; k++) {
        int result = findKthElement(arr, n, k);
        if (result != -1) {
            printf("The %dth element is: %d\n", k, result);
        } else {
            printf("Invalid k value.\n");
        }
    }

    return 0;
}
```
