```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_WORD_LEN 4096U

static char *duplicate_string(const char *src)
{
    size_t len;
    size_t alloc_size;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_WORD_LEN);
    if (len >= MAX_WORD_LEN) {
        return NULL;
    }

    alloc_size = len + 1U;
    copy = malloc(alloc_size);
    if (copy == NULL) {
        return NULL;
    }

    if (len > 0U) {
        memcpy(copy, src, len);
    }
    copy[len] = '\0';
    return copy;
}

static void free_word_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }

    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

static char **remove_words(const char *const *words, size_t count,
                           const char *pattern, size_t *out_count)
{
    char **result;
    size_t kept;
    size_t i;

    if (out_count == NULL) {
        return NULL;
    }

    *out_count = 0;

    if (words == NULL || pattern == NULL || count == 0U) {
        return NULL;
    }

    if (count > SIZE_MAX / sizeof(*result)) {
        return NULL;
    }

    result = malloc(count * sizeof(*result));
    if (result == NULL) {
        return NULL;
    }

    kept = 0;
    for (i = 0; i < count; i++) {
        if (words[i] == NULL) {
            continue;
        }
        if (strnlen(words[i], MAX_WORD_LEN) >= MAX_WORD_LEN) {
            free_word_list(result, kept);
            return NULL;
        }
        if (strstr(words[i], pattern) == NULL) {
            result[kept] = duplicate_string(words[i]);
            if (result[kept] == NULL) {
                free_word_list(result, kept);
                return NULL;
            }
            kept++;
        }
    }

    if (kept == 0U) {
        free(result);
        return NULL;
    }

    if (kept < count) {
        char **shrunk = realloc(result, kept * sizeof(*shrunk));
        if (shrunk != NULL) {
            result = shrunk;
        }
    }

    *out_count = kept;
    return result;
}

int main(void)
{
    const char *const words[] = {
        "apple", "banana", "cherry", "grape", "apricot", "melon"
    };
    size_t count = sizeof(words) / sizeof(words[0]);
    const char *pattern = "ap";
    char **filtered;
    size_t filtered_count;
    size_t i;

    filtered = remove_words(words, count, pattern, &filtered_count);
    if (filtered == NULL) {
        if (fprintf(stderr, "No words remain or an error occurred.\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    for (i = 0; i < filtered_count; i++) {
        if (printf("%s\n", filtered[i]) < 0) {
            free_word_list(filtered, filtered_count);
            return EXIT_FAILURE;
        }
    }

    free_word_list(filtered, filtered_count);
    return EXIT_SUCCESS;
}
```

Key changes made during the rewrite:

1. **Removed the dead check `if (len >= alloc_size)`** — the static analyzer correctly flagged this as always false. Since `alloc_size = len + 1U` and `len < MAX_WORD_LEN` guarantees no overflow of `size_t`, `len >= alloc_size` can never be true. Dead code removed.

2. **The `memcpy` (CWE-120) finding is a false positive, but the safety is now explicit** — the destination buffer is allocated as exactly `len + 1U` bytes, and exactly `len` bytes are copied, followed by manual NUL-termination at `copy[len]`. The bound was already established by `strnlen(src, MAX_WORD_LEN)` before allocation, so the copy can never overflow.

3. **The malloc NULL-check remains** — the analyzer's "assuming condition is false" note is just path-analysis commentary, not a defect; the check is correct and retained.

Everything else (overflow guard on `count * sizeof(*result)`, `strnlen` bounding, cleanup on all failure paths, tolerating `realloc` shrink failure) was already sound and preserved.
