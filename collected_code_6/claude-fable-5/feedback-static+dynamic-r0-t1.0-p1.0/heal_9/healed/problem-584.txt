```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define MAX_ERROR_MSG 128
#define MAX_SENTENCE_LEN 4096

int find_adverbs(const char *sentence);

int find_adverbs(const char *sentence)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[A-Za-z]+ly";
    char error_buf[MAX_ERROR_MSG];
    int ret;
    size_t offset = 0;
    size_t len;

    if (sentence == NULL) {
        fprintf(stderr, "Invalid input: sentence is NULL\n");
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len >= MAX_SENTENCE_LEN) {
        fprintf(stderr, "Invalid input: sentence too long or not null-terminated\n");
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        regerror(ret, &regex, error_buf, sizeof(error_buf));
        error_buf[sizeof(error_buf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", error_buf);
        return -1;
    }

    while (offset < len) {
        ret = regexec(&regex, sentence + offset, 1, &match, 0);
        if (ret == REG_NOMATCH) {
            break;
        }
        if (ret != 0) {
            regerror(ret, &regex, error_buf, sizeof(error_buf));
            error_buf[sizeof(error_buf) - 1] = '\0';
            fprintf(stderr, "Regex execution failed: %s\n", error_buf);
            regfree(&regex);
            return -1;
        }

        if (match.rm_so < 0 || match.rm_eo < match.rm_so) {
            fprintf(stderr, "Invalid match offsets\n");
            regfree(&regex);
            return -1;
        }

        {
            size_t start = offset + (size_t)match.rm_so;
            size_t end = offset + (size_t)match.rm_eo;
            size_t word_len;
            char *word;

            if (end > len || start > end) {
                fprintf(stderr, "Match out of bounds\n");
                regfree(&regex);
                return -1;
            }

            word_len = end - start;

            if (word_len >= MAX_SENTENCE_LEN) {
                fprintf(stderr, "Match length exceeds maximum allowed size\n");
                regfree(&regex);
                return -1;
            }

            word = malloc(word_len + 1);
            if (word == NULL) {
                fprintf(stderr, "Memory allocation failed\n");
                regfree(&regex);
                return -1;
            }

            if (word_len > 0 && start + word_len <= len) {
                memcpy(word, sentence + start, word_len);
            }
            word[word_len] = '\0';

            printf("Adverb: %-15s Position: %zu-%zu\n", word, start,
                   (word_len > 0) ? end - 1 : start);
            free(word);

            offset = end;
        }

        if (match.rm_eo == match.rm_so) {
            offset++;
        }
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *sentence = "She quickly ran to the store and quietly bought a surprisingly cheap gift";

    printf("Sentence: %s\n\n", sentence);

    if (find_adverbs(sentence) != 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes made:**

1. **Flawfinder `char error_buf[MAX_ERROR_MSG]` (CWE-119/CWE-120):** The original code was already largely safe because `regerror()` accepts `sizeof(error_buf)` and truncates internally. To fully harden it, I added an explicit null-terminator assignment (`error_buf[sizeof(error_buf) - 1] = '\0'`) after each `regerror()` call as a defense-in-depth measure against any non-conforming implementation.

2. **Flawfinder `memcpy` (CWE-120):** The original code already validated that `end <= len` and `start <= end` before computing `word_len`, and allocated exactly `word_len + 1` bytes, so the destination could always hold the source. I strengthened this with two additional guards: an upper-bound sanity check on `word_len` against `MAX_SENTENCE_LEN` before allocation, and a redundant `start + word_len <= len` check directly at the copy site so the bounds proof is local to the `memcpy` call.

3. **Existing sound practices preserved:** `strnlen` with a hard cap prevents reading unterminated input, `regfree` is called on every exit path after successful `regcomp` (no resource leak), `malloc` failure is handled, zero-length match handling prevents infinite loops, and all offset arithmetic is validated before use.
