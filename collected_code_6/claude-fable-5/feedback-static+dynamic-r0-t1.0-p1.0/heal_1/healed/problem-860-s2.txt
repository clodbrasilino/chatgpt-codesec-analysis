```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int ends_with_alphanumeric(const char *str)
{
    regex_t regex;
    int ret;

    if (str == NULL) {
        return 0;
    }

    ret = regcomp(&regex, "[a-zA-Z0-9]$", REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        size_t errlen = regerror(ret, &regex, NULL, 0);
        char *errbuf = malloc(errlen);
        if (errbuf != NULL) {
            regerror(ret, &regex, errbuf, errlen);
            fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
            free(errbuf);
        } else {
            fprintf(stderr, "Regex compilation failed\n");
        }
        return -1;
    }

    ret = regexec(&regex, str, 0, NULL, 0);
    regfree(&regex);

    if (ret == 0) {
        return 1;
    } else if (ret == REG_NOMATCH) {
        return 0;
    } else {
        fprintf(stderr, "Regex execution failed\n");
        return -1;
    }
}

int main(void)
{
    const char *tests[] = {
        "hello123",
        "world!",
        "test_string9",
        "endswithspace ",
        "abcXYZ",
        ""
    };
    size_t count = sizeof(tests) / sizeof(tests[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = ends_with_alphanumeric(tests[i]);
        if (result == 1) {
            printf("\"%s\" ends with an alphanumeric character.\n", tests[i]);
        } else if (result == 0) {
            printf("\"%s\" does not end with an alphanumeric character.\n", tests[i]);
        } else {
            fprintf(stderr, "Error processing \"%s\"\n", tests[i]);
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

The two flagged weaknesses were addressed as follows:

1. **Statically-sized `errbuf` (CWE-119/CWE-120)** — Instead of a fixed 128-byte stack buffer, the code now calls `regerror(ret, &regex, NULL, 0)` first to obtain the exact required buffer size, then allocates it dynamically. This guarantees the message can never be truncated or overflow, and the allocation failure path is handled gracefully. The buffer is freed immediately after use.

2. **Variable scope of `result`** — The declaration of `result` was moved from function scope in `main` into the body of the `for` loop, reducing its scope to the smallest region where it is actually used, as recommended by the cppcheck `variableScope` diagnostic.
