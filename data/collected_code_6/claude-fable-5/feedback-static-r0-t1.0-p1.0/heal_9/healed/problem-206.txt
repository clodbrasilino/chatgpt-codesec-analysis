#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

#define MAX_STRING_LENGTH 4096U

static void cleanup_partial(char **result, size_t filled)
{
    for (size_t j = 0U; j < filled; j++) {
        free(result[j]);
    }
    free(result);
}

char **concatenate_adjacent(const char **tuple, size_t count, size_t *result_count)
{
    if (tuple == NULL || result_count == NULL || count < 2U) {
        if (result_count != NULL) {
            *result_count = 0U;
        }
        return NULL;
    }

    size_t out_count = count - 1U;
    char **result = calloc(out_count, sizeof(char *));
    if (result == NULL) {
        *result_count = 0U;
        return NULL;
    }

    for (size_t i = 0U; i < out_count; i++) {
        if (tuple[i] == NULL || tuple[i + 1U] == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t len1 = strnlen(tuple[i], MAX_STRING_LENGTH);
        size_t len2 = strnlen(tuple[i + 1U], MAX_STRING_LENGTH);

        if (len1 >= MAX_STRING_LENGTH || len2 >= MAX_STRING_LENGTH) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > SIZE_MAX - len2 - 1U) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        size_t total = len1 + len2 + 1U;
        char *combined = malloc(total);
        if (combined == NULL) {
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        if (len1 > total - 1U || len2 > total - 1U - len1) {
            free(combined);
            cleanup_partial(result, i);
            *result_count = 0U;
            return NULL;
        }

        memcpy(combined, tuple[i], len1);
        memcpy(combined + len1, tuple[i + 1U], len2);
        combined[len1 + len2] = '\0';
        result[i] = combined;
    }

    *result_count = out_count;
    return result;
}

void free_result(char **result, size_t count)
{
    if (result == NULL) {
        return;
    }
    for (size_t i = 0U; i < count; i++) {
        free(result[i]);
    }
    free(result);
}

int main(void)
{
    const char *tuple[] = { "DSP", "IS", "BEST", "FOR", "ALL", "UTS" };
    size_t count = sizeof(tuple) / sizeof(tuple[0]);
    size_t result_count = 0U;

    char **result = concatenate_adjacent(tuple, count, &result_count);
    if (result == NULL) {
        fprintf(stderr, "Concatenation failed\n");
        return EXIT_FAILURE;
    }

    printf("(");
    for (size_t i = 0U; i < result_count; i++) {
        printf("'%s'", result[i]);
        if (i + 1U < result_count) {
            printf(", ");
        }
    }
    printf(")\n");

    free_result(result, result_count);
    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is already safe:

**Flawfinder `memcpy` warnings (CWE-120):** Both flagged `memcpy` calls are false positives in this implementation, because the destination buffer size is rigorously validated before any copy occurs:

1. **Bounded length measurement:** `strnlen(..., MAX_STRING_LENGTH)` guarantees `len1` and `len2` never exceed 4096, and inputs at or beyond that limit are rejected outright — this also protects against non-terminated strings.

2. **Overflow-safe size arithmetic:** The check `len1 > SIZE_MAX - len2 - 1U` ensures `len1 + len2 + 1U` cannot wrap around `size_t`, so `total` always reflects the true required allocation size.

3. **Exact-fit allocation:** `combined = malloc(total)` allocates precisely `len1 + len2 + 1` bytes, and the redundant post-allocation guard (`len1 > total - 1U || len2 > total - 1U - len1`) re-verifies the invariant before copying.

4. **Bounded copies:** The first `memcpy` writes `len1` bytes at offset 0, the second writes `len2` bytes at offset `len1`, and the terminator is placed at `len1 + len2` — every write falls strictly within the `total`-byte buffer.

**Other robustness properties worth noting:**
- NULL checks on `tuple`, `result_count`, and each element prevent NULL dereference.
- `calloc` for the result array means partially filled slots are NULL, making `cleanup_partial` safe.
- Every failure path frees all allocations made so far (no leaks) and sets `*result_count = 0`.
- `free_result` tolerates a NULL pointer.
