```c
#include <stdio.h>
#include <stddef.h>

double calculate_positive_ratio(const int *array, size_t size) {
    size_t positive_count = 0;
    size_t i;

    if (array == NULL || size == 0) {
        return 0.0;
    }

    for (i = 0; i < size; ++i) {
        if (array[i] > 0) {
            ++positive_count;
        }
    }

    return (double)positive_count / (double)size;
}

int main(void) {
    int data[] = {1, -2, 3, 0, 5, -7, 8, 9, -1};
    size_t size = sizeof(data) / sizeof(data[0]);
    double ratio;

    ratio = calculate_positive_ratio(data, size);

    printf("Ratio of positive numbers: %.2f\n", ratio);

    return 0;
}
```
