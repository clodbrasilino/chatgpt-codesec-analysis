#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int convert_list_to_integer(const int *list, size_t size, int *result) {
    if (list == NULL || result == NULL || size == 0) {
        return -1;
    }

    long long value = 0;
    for (size_t i = 0; i < size; i++) {
        if (list[i] < 0 || list[i] > 9) {
            return -1;
        }
        if (value > (LLONG_MAX - list[i]) / 10) {
            return -1;
        }
        value = value * 10 + list[i];
    }

    if (value > INT_MAX) {
        return -1;
    }

    *result = (int)value;
    return 0;
}

int main(void) {
    int list[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(list) / sizeof(list[0]);
    int result = 0;

    if (convert_list_to_integer(list, size, &result) == 0) {
        printf("Result: %d\n", result);
    } else {
        printf("Conversion failed\n");
    }

    return 0;
}
