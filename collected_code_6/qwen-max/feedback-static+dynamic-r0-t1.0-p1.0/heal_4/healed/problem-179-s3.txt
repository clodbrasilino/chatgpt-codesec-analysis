#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

bool is_keith_number(long n) {
    if (n <= 0) return false;

    long t = n;
    int digits[100] = {0};
    int k = 0;
    while (t > 0 && k < 100) {
        digits[k++] = t % 10;
        t /= 10;
    }

    long sum = 0;
    while (sum < n || (sum == n && k > 1)) {
        sum = 0;
        for (int j = 0; j < k; ++j) {
            sum += digits[j];
        }
        if (sum == n) return true;

        for (int i = 0; i < k - 1; ++i) {
            digits[i] = digits[i + 1];
        }
        digits[k - 1] = sum;
    }

    return false;
}

int main() {
    long number;

    printf("Enter a number: ");
    if (scanf("%ld", &number) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (is_keith_number(number)) {
        printf("%ld is a Keith number.\n", number);
    } else {
        printf("%ld is not a Keith number.\n", number);
    }

    return EXIT_SUCCESS;
}
