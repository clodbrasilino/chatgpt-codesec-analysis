#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

int is_woodall(long long num)
{
    long long n;
    long long power;

    if (num < 1)
    {
        return 0;
    }

    n = 1;
    power = 2;

    while (1)
    {
        long long woodall;

        if (power > LLONG_MAX / n)
        {
            break;
        }

        woodall = n * power - 1;

        if (woodall == num)
        {
            return 1;
        }

        if (woodall > num)
        {
            break;
        }

        if (power > LLONG_MAX / 2)
        {
            break;
        }

        power = power * 2;
        n = n + 1;
    }

    return 0;
}

int main(void)
{
    char buffer[64];
    char *endptr;
    long long num;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL)
    {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    errno = 0;
    num = strtoll(buffer, &endptr, 10);

    {
        fprintf(stderr, "Error: invalid input.\n");
        return 1;
    }

    if (*endptr != '\n' && *endptr != '\0')
    {
        fprintf(stderr, "Error: invalid input.\n");
        return 1;
    }

    if (is_woodall(num))
    {
        printf("True\n");
    }
    else
    {
        printf("False\n");
    }

    return 0;
}
