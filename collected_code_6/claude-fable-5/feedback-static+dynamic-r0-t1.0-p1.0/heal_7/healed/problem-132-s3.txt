```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TUPLE_STR_LEN 32

struct Tuple {
    int first;
    double second;
    char third[TUPLE_STR_LEN];
};

static int tuple_set_third(struct Tuple *t, const char *value)
{
    size_t len;

    if (t == NULL || value == NULL) {
        return -1;
    }

    len = strnlen(value, sizeof(t->third));
    if (len >= sizeof(t->third)) {
        return -1;
    }

    if (len > 0U) {
        memcpy(t->third, value, len);
    }
    t->third[len] = '\0';

    return 0;
}

static char *tuple_to_string(const struct Tuple *t)
{
    char *result;
    int needed;
    int written;
    size_t size;

    if (t == NULL) {
        return NULL;
    }

    needed = snprintf(NULL, 0, "(%d, %.2f, %s)", t->first, t->second, t->third);
    if (needed < 0) {
        return NULL;
    }

    size = (size_t)needed + 1U;

    result = malloc(size);
    if (result == NULL) {
        return NULL;
    }

    written = snprintf(result, size, "(%d, %.2f, %s)",
                       t->first, t->second, t->third);
    if (written < 0 || (size_t)written >= size) {
        free(result);
        return NULL;
    }

    return result;
}

int main(void)
{
    struct Tuple t;
    char *str;

    memset(&t, 0, sizeof(t));

    t.first = 42;
    t.second = 3.14;

    if (tuple_set_third(&t, "hello") != 0) {
        (void)fprintf(stderr, "Failed to set tuple string field\n");
        return EXIT_FAILURE;
    }

    str = tuple_to_string(&t);
    if (str == NULL) {
        (void)fprintf(stderr, "Failed to convert tuple to string\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", str) < 0) {
        free(str);
        return EXIT_FAILURE;
    }

    free(str);
    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and how the code addresses them:

**1. Statically-sized array `char third[TUPLE_STR_LEN]` (CWE-119/CWE-120):** The array itself is not a defect — the risk is unchecked writes into it. All writes to `third` go through `tuple_set_third()`, which uses `strnlen()` bounded by `sizeof(t->third)` and rejects any input that would not fit including the NUL terminator (`len >= sizeof(t->third)`). No unbounded functions like `strcpy` or `sprintf` ever touch this buffer.

**2. `memcpy` without bounds check (CWE-120):** This is a false positive in practice, since the length is validated before the copy. The rewrite keeps the validation and adds an explicit guard so `memcpy` is only called with a strictly positive, verified length, and the terminator write `t->third[len] = '\0'` is provably in bounds because `len < sizeof(t->third)`.

**Additional hardening in the rewrite:**
- The `snprintf` truncation check was tightened from `written > needed` to `(size_t)written >= size`, which is the canonical idiom for detecting truncation.
- The allocation size is computed once into `size`, avoiding a repeated and potentially inconsistent expression.
- All allocations are checked, all error paths free owned memory, and `main` returns proper `EXIT_*` codes — no leaks, no double frees, no use of uninitialized memory (the struct is zeroed with `memset` before use).

The code is portable C99+ (note: `strnlen` requires POSIX.1-2008 or `_GNU_SOURCE` on some older toolchains; on strictly ISO C targets you could replace it with a small local helper).
