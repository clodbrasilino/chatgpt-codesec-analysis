```c
#include <stdio.h>
#include <stdint.h>

uint32_t toggle_odd_bits(uint32_t num) {
    uint32_t mask = 0xAAAAAAAA;
    return num ^ mask;
}

int main() {
    uint32_t number = 0x12345678;
    uint32_t result = toggle_odd_bits(number);
    printf("Original: %08X, Toggled: %08X\n", number, result);
    return 0;
}
```
