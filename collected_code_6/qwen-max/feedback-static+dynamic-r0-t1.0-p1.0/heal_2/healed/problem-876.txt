```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int gcd(int a, int b) {
    if (a < 0 || b < 0) return -1;
    while (b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    return a;
}

int lcm(int a, int b) {
    if (a < 0 || b < 0) return -1;
    int g = gcd(a, b);
    if (g < 0) return -1;
    if (a > INT_MAX / b) return -1;
    return (a / g) * b;
}

int main() {
    int num1, num2;
    printf("Enter two positive integers: ");
    if (scanf("%d %d", &num1, &num2) != 2) {
        fprintf(stderr, "Invalid input. Please enter two integers.\n");
        return EXIT_FAILURE;
    }
    if (num1 <= 0 || num2 <= 0) {
        fprintf(stderr, "Both numbers must be positive.\n");
        return EXIT_FAILURE;
    }
    int result = lcm(num1, num2);
    if (result < 0) {
        fprintf(stderr, "Error in computation.\n");
        return EXIT_FAILURE;
    }
    printf("LCM of %d and %d is %d\n", num1, num2, result);
    return 0;
}
```
