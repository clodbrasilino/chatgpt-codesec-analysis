```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_LEN 11
#define INPUT_BUF_SIZE 64

static int is_valid_format(const char *date)
{
    size_t i;
    size_t len;

    if (date == NULL) {
        return 0;
    }

    len = strnlen(date, DATE_LEN);

    if (len != DATE_LEN - 1) {
        return 0;
    }

    for (i = 0; i < DATE_LEN - 1; i++) {
        if (i == 4 || i == 7) {
            if (date[i] != '-') {
                return 0;
            }
        } else {
            if (!isdigit((unsigned char)date[i])) {
                return 0;
            }
        }
    }

    return 1;
}

static int is_valid_date(int year, int month, int day)
{
    int days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (year < 1 || month < 1 || month > 12 || day < 1) {
        return 0;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        days_in_month[1] = 29;
    }

    if (day > days_in_month[month - 1]) {
        return 0;
    }

    return 1;
}

static int convert_date(const char *input, char *output, size_t output_size)
{
    int year;
    int month;
    int day;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_LEN) {
        return -1;
    }

    if (!is_valid_format(input)) {
        return -1;
    }

    if (sscanf(input, "%4d-%2d-%2d", &year, &month, &day) != 3) {
        return -1;
    }

    if (!is_valid_date(year, month, day)) {
        return -1;
    }

    written = snprintf(output, output_size, "%02d-%02d-%04d", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_SIZE];
    char output[DATE_LEN];
    size_t len;

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strcspn(input, "\n");
    if (len < sizeof(input)) {
        input[len] = '\0';
    }

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date: %s\n", input);
        return EXIT_FAILURE;
    }

    printf("Converted date: %s\n", output);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

The Flawfinder warnings on `input[INPUT_BUF_SIZE]` and `output[DATE_LEN]` are low-risk, informational hits (risk level 2) triggered simply because statically-sized arrays exist. Reviewing the code confirms the mitigations are already in place:

1. **`input` buffer (CWE-119/CWE-120):** Reading is done with `fgets(input, sizeof(input), stdin)`, which is length-bounded and always NUL-terminates within the buffer. The explicit `input[sizeof(input) - 1] = '\0'` is defensive redundancy, and `strcspn` combined with the `len < sizeof(input)` guard safely strips the trailing newline without any out-of-bounds write.

2. **`output` buffer (CWE-119/CWE-120):** All writes go through `snprintf(output, output_size, ...)` with the size passed explicitly from `sizeof(output)`. The return value is checked for both encoding errors (`written < 0`) and truncation (`written >= output_size`), so the buffer can never overflow and the result is guaranteed complete.

3. **Additional hardening already present:** `strnlen` bounds the format check, `isdigit` receives an `unsigned char` cast avoiding UB on negative `char` values, `sscanf` uses width-limited specifiers (`%4d`, `%2d`), and all pointers plus `output_size` are validated before use.

The code as written contains no exploitable buffer weaknesses; the warnings are false positives that are properly mitigated, so the rewrite preserves the original logic unchanged.
