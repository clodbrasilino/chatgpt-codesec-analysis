```C
#include <stdio.h>

int extractSum(int tuples[][2], int length){
    int sum = 0;

    if(tuples == NULL || length <= 0)
        return -1;

    for(int i = 0; i < length; i++){
        if(i % 2 == 0){
            for(int j = 0; j < 2; j++){
               sum += tuples[i][j];
            }
        }
    }

    return sum;
}

int main(){
    int tuples[4][2] = {{1,2}, {3,4}, {5,6}, {7,8}};
    int length = sizeof(tuples) / sizeof(tuples[0]);

    int sum = extractSum(tuples, length);
    if(sum >= 0)
        printf("The sum of alternate chains of tuples = %d\n", sum);
    else
        printf("Invalid input\n");

    return 0;
}
```
