```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

#define MAX_INPUT_LEN 128U

static int binary_to_decimal(const char *binary, unsigned long long *result)
{
    size_t len;
    size_t i;
    unsigned long long value = 0ULL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, MAX_INPUT_LEN);
    if (len == 0U || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULLONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

static int flush_stdin_check_truncation(void)
{
    int truncated = 0;
    unsigned long guard = 0UL;

    do {
        int ch = fgetc(stdin);
        if (ch == '\n' || ch == EOF) {
            break;
        }
        truncated = 1;
        guard++;
    } while (guard < ULONG_MAX);

    return truncated;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    unsigned long long decimal = 0ULL;
    size_t len;

    memset(input, 0, sizeof(input));

    printf("Enter a binary number: ");

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    } else {
        if (flush_stdin_check_truncation() == 1) {
            fprintf(stderr, "Error: input too long.\n");
            return EXIT_FAILURE;
        }
    }

    if (binary_to_decimal(input, &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return EXIT_FAILURE;
    }

    printf("Decimal equivalent: %llu\n", decimal);
    return EXIT_SUCCESS;
}
```

The changes made to address the identified weaknesses:

1. **`[variableScope]` on `ch`**: The variable `ch` was declared at function scope in `flush_stdin_check_truncation()` but only used inside the loop body. It is now declared inside the `do`/`while` block, reducing its scope to the minimum required.

2. **Flawfinder `fgetc` (CWE-120, CWE-20)**: This was a low-risk informational finding. The loop never stores `fgetc()` results into a buffer, so no overflow is possible. The existing `guard` counter already bounds the loop defensively against pathological non-terminating streams, so the logic is retained as-is.

3. **Flawfinder `char` array (CWE-119/CWE-120)**: The statically-sized buffer is already handled safely: `fgets()` is called with `sizeof(input)` limiting the read length, the buffer is explicitly NUL-terminated afterward, `strnlen()` is used instead of `strlen()` to bound length computation, and overlong input is detected and rejected via `flush_stdin_check_truncation()`. No further change was required.

The core conversion logic in `binary_to_decimal()` was already robust — it validates pointers, rejects empty or oversized input, validates each character, and checks for overflow of `unsigned long long` before each shift — so it remains unchanged.
