```c
#include <stdio.h>
#include <stdlib.h>

static int find_last_removed_position(const int arr[], size_t n, int m)
{
    size_t i;
    size_t last_index;
    long max_cycles;
    long cycles;

    if (arr == NULL || n == 0U || m <= 0) {
        return -1;
    }

    last_index = 0U;
    max_cycles = -1L;

    for (i = 0U; i < n; i++) {
        if (arr[i] < 0) {
            return -1;
        }
        cycles = ((long)arr[i] + (long)m - 1L) / (long)m;
        if (cycles >= max_cycles) {
            max_cycles = cycles;
            last_index = i;
        }
    }

    return (int)(last_index + 1U);
}

int main(void)
{
    int arr[] = { 31, 41, 59, 26, 53, 58, 97 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    int m = 5;
    int position;

    position = find_last_removed_position(arr, n, m);

    if (position < 0) {
        (void)fprintf(stderr, "Error: invalid input parameters\n");
        return EXIT_FAILURE;
    }

    if (printf("Position of last removed element: %d\n", position) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

This solves the classic problem: given an array where every element is repeatedly decreased by `m`, and an element is removed once it drops to zero or below, find the 1-based position of the last element to be removed. The element requiring the most subtraction cycles (`ceil(arr[i] / m)`) is removed last; ties are resolved in favor of the later position.

Safety measures taken:
- Null pointer, zero length, non-positive divisor, and negative element checks (CWE-703)
- Intermediate arithmetic in `long` to prevent overflow in `arr[i] + m - 1` (CWE-664)
- No dynamic allocation, so no lifetime management issues; return values of I/O functions are checked, `main(void)` signature, explicit casts, and `EXIT_SUCCESS`/`EXIT_FAILURE` used for standards adherence (CWE-710)
