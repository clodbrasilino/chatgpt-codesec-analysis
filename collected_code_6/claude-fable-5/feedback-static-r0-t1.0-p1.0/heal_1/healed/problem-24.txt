```c
#include <stdio.h>
#include <string.h>
#include <limits.h>

#define INPUT_SIZE 128U

int binary_to_decimal(const char *binary, size_t max_len, unsigned long *result)
{
    size_t len;
    size_t i;
    unsigned long value = 0UL;

    if (binary == NULL || result == NULL) {
        return -1;
    }

    len = strnlen(binary, max_len);
    if (len == 0U || len >= max_len || len > (sizeof(unsigned long) * 8U)) {
        return -1;
    }

    for (i = 0U; i < len; i++) {
        if (binary[i] != '0' && binary[i] != '1') {
            return -1;
        }
        if (value > (ULONG_MAX >> 1)) {
            return -1;
        }
        value = (value << 1) | (unsigned long)(binary[i] - '0');
    }

    *result = value;
    return 0;
}

int main(void)
{
    char input[INPUT_SIZE];
    unsigned long decimal = 0UL;
    size_t len;

    printf("Enter a binary number: ");
    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input.\n");
        return 1;
    }

    input[sizeof(input) - 1U] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0U && input[len - 1U] == '\n') {
        input[len - 1U] = '\0';
    }

    if (binary_to_decimal(input, sizeof(input), &decimal) != 0) {
        fprintf(stderr, "Error: invalid binary number.\n");
        return 1;
    }

    printf("Decimal equivalent: %lu\n", decimal);
    return 0;
}
```

Key changes addressing the flagged weaknesses:

1. **CWE-126 (`strlen` over-read)**: Both calls to `strlen` were replaced with `strnlen`, which takes an explicit maximum length and cannot read past the buffer boundary even if the string is not null-terminated. The `binary_to_decimal` function now accepts a `max_len` parameter so it never trusts the caller's string to be terminated.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is defined once as `INPUT_SIZE` and consistently referenced via `sizeof(input)`. `fgets` already bounds the read, and an explicit `input[sizeof(input) - 1U] = '\0'` guarantees null termination as defense-in-depth before any string operation.

3. **Existing protections preserved**: the overflow guard (`value > (ULONG_MAX >> 1)`), the bit-width limit check, character validation, and all NULL-pointer checks remain intact.
