```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <ctype.h>

#define ZODIAC_COUNT 12
#define MIN_YEAR 1
#define MAX_YEAR 9999
#define INPUT_BUFFER_SIZE 64

static const char *get_chinese_zodiac(int year)
{
    static const char *const signs[ZODIAC_COUNT] = {
        "Monkey", "Rooster", "Dog", "Pig",
        "Rat", "Ox", "Tiger", "Rabbit",
        "Dragon", "Snake", "Horse", "Goat"
    };
    int index;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        return NULL;
    }
    index = year % ZODIAC_COUNT;
    if (index < 0) {
        index += ZODIAC_COUNT;
    }
    if (index < 0 || index >= ZODIAC_COUNT) {
        return NULL;
    }
    return signs[index];
}

static int display_zodiac(int year)
{
    const char *sign;

    if (year < MIN_YEAR || year > MAX_YEAR) {
        fprintf(stderr, "Error: year must be between %d and %d.\n", MIN_YEAR, MAX_YEAR);
        return -1;
    }
    sign = get_chinese_zodiac(year);
    if (sign == NULL) {
        fprintf(stderr, "Error: could not determine zodiac sign.\n");
        return -1;
    }
    if (printf("%s\n", sign) < 0) {
        return -1;
    }
    if (fflush(stdout) != 0) {
        return -1;
    }
    return 0;
}

static int parse_year(const char *text, int *out_year)
{
    char *endptr;
    long value;

    if (text == NULL || out_year == NULL) {
        return -1;
    }
    while (isspace((unsigned char)*text)) {
        text++;
    }
    if (*text == '\0') {
        return -1;
    }
    errno = 0;
    value = strtol(text, &endptr, 10);
    if (errno == ERANGE || endptr == text) {
        return -1;
    }
    while (isspace((unsigned char)*endptr)) {
        endptr++;
    }
    if (*endptr != '\0') {
        return -1;
    }
    if (value < (long)INT_MIN || value > (long)INT_MAX) {
        return -1;
    }
    *out_year = (int)value;
    return 0;
}

static void discard_remaining_input(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

static int read_line(char *buffer, size_t size)
{
    size_t len;

    if (buffer == NULL || size < 2 || size > (size_t)INT_MAX) {
        return -1;
    }
    if (fgets(buffer, (int)size, stdin) == NULL) {
        return -1;
    }
    buffer[size - 1] = '\0';
    len = strlen(buffer);
    if (len >= size) {
        return -1;
    }
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    } else if (len == size - 1) {
        discard_remaining_input();
    }
    return 0;
}

int main(int argc, const char *argv[])
{
    int year;

    if (argc > 1 && argv[1] != NULL) {
        if (parse_year(argv[1], &year) != 0) {
            fprintf(stderr, "Error: invalid input, expected an integer year.\n");
            return EXIT_FAILURE;
        }
    } else {
        char buffer[INPUT_BUFFER_SIZE];

        memset(buffer, 0, sizeof(buffer));
        if (read_line(buffer, sizeof(buffer)) != 0) {
            fprintf(stderr, "Error: invalid input, expected an integer year.\n");
            return EXIT_FAILURE;
        }
        if (parse_year(buffer, &year) != 0) {
            fprintf(stderr, "Error: invalid input, expected an integer year.\n");
            return EXIT_FAILURE;
        }
    }
    if (display_zodiac(year) != 0) {
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
```

Key changes and analysis:

1. **Root cause of the failing test cases (`<no output>`)**: The original code used `strnlen()` in `read_line()`. `strnlen` is a POSIX/C23 function, not part of C89/C99, so on a strictly conforming toolchain the program fails to compile or link — producing no output at all for every test. I replaced it with standard `strlen()`, which is safe here because `fgets` guarantees NUL-termination within the buffer and we additionally force `buffer[size - 1] = '\0'` before measuring.

2. **Flawfinder buffer warnings (CWE-119/CWE-120)**: These are informational hits on statically sized arrays. Both are already properly bounded — the `signs` array is only accessed after a range-normalized modulo with an explicit bounds check, and `buffer` is only written through `fgets` with `sizeof(buffer)` passed as the limit, plus explicit NUL-termination. No overflow is possible.

3. **Flawfinder `getchar` warning (CWE-120/CWE-20)**: `discard_remaining_input()` reads into no buffer; it only drains characters until newline or EOF, terminating on `EOF`, so it cannot loop unboundedly on a closed stream.

4. **Input validation** remains robust: `strtol` with `errno`/`endptr` checks, trailing-whitespace tolerance, range clamping to `int`, and year range enforcement in both `display_zodiac` and `get_chinese_zodiac`.
