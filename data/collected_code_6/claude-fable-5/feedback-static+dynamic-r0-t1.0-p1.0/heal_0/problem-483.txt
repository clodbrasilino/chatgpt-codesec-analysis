```c
#include <stdio.h>
#include <stdlib.h>

static long long first_factorial_divisible_by(long long x)
{
    long long remainder;
    long long n;

    if (x <= 0LL) {
        return -1LL;
    }

    if (x == 1LL) {
        return 1LL;
    }

    remainder = 1LL;
    for (n = 1LL; n <= x; n++) {
        remainder = (remainder * (n % x)) % x;
        if (remainder == 0LL) {
            return n;
        }
    }

    return -1LL;
}

int main(void)
{
    long long x;
    long long result;
    int scan_status;

    printf("Enter a positive integer: ");
    scan_status = scanf("%lld", &x);
    if (scan_status != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (x <= 0LL) {
        fprintf(stderr, "Error: input must be a positive integer.\n");
        return EXIT_FAILURE;
    }

    result = first_factorial_divisible_by(x);
    if (result < 0LL) {
        fprintf(stderr, "Error: could not compute result.\n");
        return EXIT_FAILURE;
    }

    printf("The first natural number whose factorial is divisible by %lld is %lld\n", x, result);

    return EXIT_SUCCESS;
}
```

Key design decisions worth noting:

- **Overflow avoidance**: Instead of computing the full factorial (which overflows `long long` at just 21!), the function tracks the factorial *modulo x*. Since we only care whether `n! % x == 0`, this is mathematically equivalent and works for arbitrarily large valid inputs.
- **Guaranteed termination**: The loop is bounded by `x` because `x!` is always divisible by `x`, so the answer always exists within that range.
- **Input validation**: The return value of `scanf` is checked, and non-positive inputs are rejected with a clear diagnostic to `stderr`, returning `EXIT_FAILURE` rather than proceeding with invalid state.
- **Defensive error path**: Even though `first_factorial_divisible_by` cannot fail for validated input, its error return is still checked in `main` rather than assumed.
