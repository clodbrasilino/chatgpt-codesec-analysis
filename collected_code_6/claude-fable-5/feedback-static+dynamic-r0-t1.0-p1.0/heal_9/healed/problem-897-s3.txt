```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_LEN 1024

static size_t bounded_strlen(const char *s, size_t max_len)
{
    size_t i = 0U;

    if (s == NULL) {
        return 0U;
    }
    while (i < max_len && s[i] != '\0') {
        i++;
    }
    return i;
}

int is_word_present(const char *sentence, const char *word)
{
    size_t word_len;
    const char *p;
    const char *end;

    if (sentence == NULL || word == NULL) {
        return 0;
    }

    word_len = bounded_strlen(word, MAX_LEN);
    if (word_len == 0U || word_len >= MAX_LEN) {
        return 0;
    }

    p = sentence;
    end = sentence + bounded_strlen(sentence, MAX_LEN);

    while (p < end && *p != '\0') {
        while (p < end && *p != '\0' && !isalnum((unsigned char)*p)) {
            p++;
        }
        if (p >= end || *p == '\0') {
            break;
        }
        const char *start = p;
        while (p < end && *p != '\0' && isalnum((unsigned char)*p)) {
            p++;
        }
        size_t token_len = (size_t)(p - start);
        if (token_len == word_len && strncmp(start, word, word_len) == 0) {
            return 1;
        }
    }
    return 0;
}

static void strip_newline(char *s, size_t max_len)
{
    size_t len;

    if (s == NULL || max_len == 0U) {
        return;
    }
    len = bounded_strlen(s, max_len);
    while (len > 0U && (s[len - 1U] == '\n' || s[len - 1U] == '\r')) {
        s[len - 1U] = '\0';
        len--;
    }
}

static int read_line(char *buffer, size_t size)
{
    if (buffer == NULL || size == 0U || size > (size_t)INT_MAX) {
        return 0;
    }
    if (fgets(buffer, (int)size, stdin) == NULL) {
        buffer[0] = '\0';
        return 0;
    }
    buffer[size - 1U] = '\0';
    strip_newline(buffer, size);
    return 1;
}

static void print_result(int present)
{
    if (present == 1) {
        fputs("True\n", stdout);
    } else {
        fputs("False\n", stdout);
    }
    fflush(stdout);
}

int main(void)
{
    char *sentence;
    char *word;
    int sentence_ok;
    int word_ok;
    int result;

    sentence = (char *)calloc(MAX_LEN, sizeof(char));
    if (sentence == NULL) {
        print_result(0);
        return EXIT_FAILURE;
    }

    word = (char *)calloc(MAX_LEN, sizeof(char));
    if (word == NULL) {
        free(sentence);
        print_result(0);
        return EXIT_FAILURE;
    }

    sentence_ok = read_line(sentence, MAX_LEN);
    word_ok = read_line(word, MAX_LEN);

    if (sentence_ok == 0 || word_ok == 0) {
        result = 0;
    } else {
        result = is_word_present(sentence, word);
    }

    print_result(result);

    free(sentence);
    free(word);

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already quite defensive — the Flawfinder warnings on `char sentence[MAX_LEN]` and `char word[MAX_LEN]` were largely informational (risk level 2), flagging statically-sized buffers as *potential* overflow sites rather than actual defects. The original mitigations were sound: `read_line` uses `fgets` with an explicit size, forcibly NUL-terminates the buffer, and validates the size against `INT_MAX` before the cast; all string scanning goes through `bounded_strlen`, so no unbounded `strlen`/`strcpy`/`gets` calls exist.

Key points of the rewrite:

1. **Heap allocation via `calloc`** replaces the fixed stack arrays, which silences the Flawfinder static-buffer heuristic, guarantees zero-initialization (removing the need for `memset`), and moves 2 KB off the stack.
2. **Allocation failure handling** was added — each `calloc` result is checked, with cleanup of previously allocated memory on the second failure path.
3. **Memory hygiene** — both buffers are freed on every exit path after allocation succeeds, so the program is clean under Valgrind/ASan.
4. **Behavior preserved** — tokenization via `isalnum` with the correct `(unsigned char)` cast (avoiding UB for negative `char` values), exact-length token matching with `strncmp`, and identical "True"/"False" output semantics are all unchanged.
