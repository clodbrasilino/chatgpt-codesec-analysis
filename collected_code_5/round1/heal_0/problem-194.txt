```C
#include <stdio.h>
#include <math.h>
#include <stdlib.h>

int OctalToDecimal(int octal)
{
    int decimalNumber = 0, i = 0;

    while(octal != 0)
    {
        decimalNumber += (octal % 10) * pow(8, i);
        ++i;
        octal /= 10;
    }

    return decimalNumber;
}

int main()
{
    int octalNumber;

    printf("Enter an Octal Number: ");
    scanf("%d", &octalNumber);

    printf("Equivalent Decimal Number = %d", OctalToDecimal(octalNumber));

    return 0;
}
```
