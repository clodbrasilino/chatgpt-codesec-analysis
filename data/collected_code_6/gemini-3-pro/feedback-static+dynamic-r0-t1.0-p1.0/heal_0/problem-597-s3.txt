#include <stdio.h>
#include <stdlib.h>

int findKthElement(int arr1[], int m, int arr2[], int n, int k) {
    if (k < 1 || k > m + n) {
        return -1; 
    }

    if (m > n) {
        return findKthElement(arr2, n, arr1, m, k);
    }

    if (m == 0) {
        return arr2[k - 1];
    }

    if (k == 1) {
        return arr1[0] < arr2[0] ? arr1[0] : arr2[0];
    }

    int i = (m < k / 2) ? m : k / 2;
    int j = (n < k / 2) ? n : k / 2;

    if (k - i > n) {
        i = k - n;
        j = n;
    } else if (k - j > m) {
        j = k - m;
        i = m;
    } else {
        j = k - i;
    }

    if (arr1[i - 1] < arr2[j - 1]) {
        return findKthElement(arr1 + i, m - i, arr2, n, k - i);
    } else {
        return findKthElement(arr1, m, arr2 + j, n - j, k - j);
    }
}

int main() {
    int arr1[] = {2, 3, 6, 7, 9};
    int m = sizeof(arr1) / sizeof(arr1[0]);

    int arr2[] = {1, 4, 8, 10};
    int n = sizeof(arr2) / sizeof(arr2[0]);

    int k = 5;

    int result = findKthElement(arr1, m, arr2, n, k);

    if (result != -1) {
        printf("The %d-th element is %d\n", k, result);
    } else {
        printf("Invalid value of k\n");
    }

    return 0;
}
