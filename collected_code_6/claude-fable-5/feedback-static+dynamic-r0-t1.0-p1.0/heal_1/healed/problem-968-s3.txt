```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define SAMPLES 1000000L

static double periodic_function(double x)
{
    return sin(x) + cos(2.0 * x);
}

static int find_maximum(double (*f)(double), double period, long samples, double *result)
{
    long i;
    double step;
    double max_val;

    if (f == NULL || result == NULL) {
        return -1;
    }
    if (samples <= 0L || period <= 0.0 || !isfinite(period)) {
        return -1;
    }

    step = period / (double)samples;
    max_val = f(0.0);

    if (!isfinite(max_val)) {
        return -1;
    }

    for (i = 1L; i <= samples; i++) {
        const double x = step * (double)i;
        const double y = f(x);
        if (!isfinite(y)) {
            return -1;
        }
        if (y > max_val) {
            max_val = y;
        }
    }

    *result = max_val;
    return 0;
}

int main(void)
{
    double period;
    double maximum;
    int status;

    period = 2.0 * acos(-1.0);
    maximum = 0.0;

    status = find_maximum(periodic_function, period, SAMPLES, &maximum);
    if (status != 0) {
        if (fprintf(stderr, "Error: failed to compute maximum value\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Maximum value of the periodic function over one period: %.6f\n", maximum) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

The key change addresses the `[variableScope]` warnings: the variables `x` and `y` were previously declared at function scope in `find_maximum`, but they are only used inside the loop body. Moving their declarations into the loop body reduces their scope to the minimum necessary, and declaring them `const` makes the intent explicit that each iteration computes fresh, immutable values. This improves readability, helps the compiler with optimization opportunities, and eliminates any risk of stale values leaking between iterations or being accidentally used after the loop.
