```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

#define DATE_OUT_LEN 11

int convert_date(const char *input, char *output, size_t output_size);

int convert_date(const char *input, char *output, size_t output_size)
{
    regex_t regex;
    regmatch_t matches[4];
    int ret;
    size_t year_len;
    size_t month_len;
    size_t day_len;

    if (input == NULL || output == NULL || output_size < DATE_OUT_LEN) {
        return -1;
    }

    ret = regcomp(&regex, "^([0-9]{4})-([0-9]{2})-([0-9]{2})$", REG_EXTENDED);
    if (ret != 0) {
        return -1;
    }

    ret = regexec(&regex, input, 4, matches, 0);
    if (ret != 0) {
        regfree(&regex);
        return -1;
    }

    if (matches[1].rm_so < 0 || matches[2].rm_so < 0 || matches[3].rm_so < 0) {
        regfree(&regex);
        return -1;
    }

    year_len = (size_t)(matches[1].rm_eo - matches[1].rm_so);
    month_len = (size_t)(matches[2].rm_eo - matches[2].rm_so);
    day_len = (size_t)(matches[3].rm_eo - matches[3].rm_so);

    if (year_len != 4 || month_len != 2 || day_len != 2) {
        regfree(&regex);
        return -1;
    }

    ret = snprintf(output, output_size, "%.2s-%.2s-%.4s",
                   input + matches[3].rm_so,
                   input + matches[2].rm_so,
                   input + matches[1].rm_so);

    regfree(&regex);

    if (ret < 0 || (size_t)ret >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *input_date = "2024-05-17";
    char output_date[DATE_OUT_LEN];

    memset(output_date, 0, sizeof(output_date));

    if (convert_date(input_date, output_date, sizeof(output_date)) != 0) {
        fprintf(stderr, "Failed to convert date: %s\n", input_date);
        return EXIT_FAILURE;
    }

    output_date[sizeof(output_date) - 1] = '\0';

    if (printf("Input: %s\nOutput: %s\n", input_date, output_date) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code was already well-defended against the flagged weakness (CWE-119/CWE-120). The Flawfinder warning on the statically-sized `char output_date[DATE_OUT_LEN]` array is essentially a false positive here, because:

1. **Size is passed and validated** — `convert_date` receives `output_size` and rejects anything smaller than `DATE_OUT_LEN`, so the callee can never assume more space than the caller provides.
2. **Bounded write** — `snprintf` is used with `output_size`, which never writes past the buffer and always NUL-terminates.
3. **Truncation detection** — the return value of `snprintf` is checked (`ret < 0 || (size_t)ret >= output_size`), so silent truncation is treated as an error rather than ignored.
4. **Match-length validation** — the regex capture group lengths are verified (4-2-2) before formatting, and I added an extra guard that the `rm_so` offsets are non-negative before using them as pointer arithmetic operands.
5. **Defensive termination** — a belt-and-braces NUL terminator write in `main` guarantees the buffer is a valid string before `printf`, even if the API contract were ever violated.

`DATE_OUT_LEN` of 11 correctly accounts for `DD-MM-YYYY` (10 characters) plus the terminating NUL, and all `regfree` calls are paired on every exit path, so there are no resource leaks.
