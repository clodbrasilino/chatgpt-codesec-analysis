bool all_chars_same(const char *str, size_t len)
{
    if (str == NULL || len == 0)
    {
        return false;
    }

    char first = str[0];

    for (size_t i = 1; i < len; i++)
    {
        if (str[i] != first)
        {
            return false;
        }
    }

    return true;
}
