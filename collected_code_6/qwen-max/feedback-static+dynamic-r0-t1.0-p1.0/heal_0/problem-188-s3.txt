#include <stdio.h>
#include <math.h>

int isProductOfTwoSquares(int n) {
    for (int i = 1; i * i <= n; ++i) {
        int j = n / (i * i);
        if (j * i * i == n && sqrt(j) == (int)sqrt(j)) {
            return 1;
        }
    }
    return 0;
}

int main() {
    int number;
    printf("Enter a number: ");
    scanf("%d", &number);
    if (isProductOfTwoSquares(number)) {
        printf("The number can be represented as the product of two squares.\n");
    } else {
        printf("The number cannot be represented as the product of two squares.\n");
    }
    return 0;
}
