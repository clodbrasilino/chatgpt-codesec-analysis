```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

size_t longest_word_length(const char *str)
{
    size_t max_len = 0;
    size_t current_len = 0;
    size_t i;

    if (str == NULL) {
        return 0;
    }

    for (i = 0; str[i] != '\0'; i++) {
        if (isalpha((unsigned char)str[i])) {
            current_len++;
            if (current_len > max_len) {
                max_len = current_len;
            }
        } else {
            current_len = 0;
        }
    }

    return max_len;
}

int main(void)
{
    char buffer[1024];
    size_t result;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    result = longest_word_length(buffer);

    if (printf("Longest word length: %zu\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning about the statically-sized `char buffer[1024]` (CWE-119/CWE-120) is a low-risk, informational flag rather than an actual vulnerability in this code. The original source already mitigates the concern correctly:

1. **Bounded input:** `fgets(buffer, sizeof(buffer), stdin)` limits reads to at most 1023 characters plus the null terminator, using `sizeof(buffer)` rather than a hardcoded magic number, so the bound stays correct even if the buffer size changes.

2. **NULL check:** The return value of `fgets` is checked, handling EOF and read errors gracefully.

3. **Safe newline stripping:** `buffer[strcspn(buffer, "\n")] = '\0'` is the idiomatic, safe way to remove a trailing newline — it never indexes out of bounds since `strcspn` returns at most the string length, and `fgets` guarantees null termination.

4. **Correct `ctype` usage:** The cast to `(unsigned char)` before `isalpha()` avoids undefined behavior with negative `char` values on platforms where `char` is signed.

5. **Defensive function design:** `longest_word_length` guards against a `NULL` pointer and uses `size_t` consistently, matching the `%zu` format specifier.
