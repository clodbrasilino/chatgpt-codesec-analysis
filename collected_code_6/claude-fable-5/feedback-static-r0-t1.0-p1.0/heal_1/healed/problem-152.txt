```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

static int merge(int *arr, size_t left, size_t mid, size_t right)
{
    size_t n1 = mid - left + 1;
    size_t n2 = right - mid;
    size_t i, j, k;
    int *L;
    int *R;

    if (n1 > SIZE_MAX / sizeof(int) || n2 > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    L = malloc(n1 * sizeof(int));
    if (L == NULL) {
        return -1;
    }

    R = malloc(n2 * sizeof(int));
    if (R == NULL) {
        free(L);
        return -1;
    }

    for (i = 0; i < n1; i++) {
        L[i] = arr[left + i];
    }

    for (j = 0; j < n2; j++) {
        R[j] = arr[mid + 1 + j];
    }

    i = 0;
    j = 0;
    k = left;

    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k] = L[i];
            i++;
        } else {
            arr[k] = R[j];
            j++;
        }
        k++;
    }

    while (i < n1) {
        arr[k] = L[i];
        i++;
        k++;
    }

    while (j < n2) {
        arr[k] = R[j];
        j++;
        k++;
    }

    free(L);
    free(R);
    return 0;
}

static int merge_sort(int *arr, size_t left, size_t right)
{
    if (arr == NULL) {
        return -1;
    }

    if (left < right) {
        size_t mid = left + (right - left) / 2;

        if (merge_sort(arr, left, mid) != 0) {
            return -1;
        }
        if (merge_sort(arr, mid + 1, right) != 0) {
            return -1;
        }
        if (merge(arr, left, mid, right) != 0) {
            return -1;
        }
    }

    return 0;
}

int main(void)
{
    int arr[] = { 38, 27, 43, 3, 9, 82, 10 };
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t i;

    if (merge_sort(arr, 0, n - 1) != 0) {
        fprintf(stderr, "Error: merge sort failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Summary of changes made to address the identified weaknesses:**

1. **Flawfinder `memcpy` warnings (CWE-120):** Replaced both `memcpy` calls with explicit element-by-element copy loops. The loop bounds are directly tied to the allocation sizes (`n1`, `n2`), making it statically evident that no buffer overflow can occur, and eliminating the flagged function entirely. This also removed the need for `<string.h>`.

2. **Integer overflow hardening:** Added a check against `SIZE_MAX / sizeof(int)` before the `malloc` calls to guard against multiplication overflow in the allocation size computation (requires `<stdint.h>`).

3. **`variableScope` (cppcheck):** Moved the declaration of `mid` inside the `if (left < right)` block in `merge_sort`, reducing its scope to where it is actually used.

4. **`knownConditionTrueFalse` (cppcheck):** Removed the redundant `if (n > 0)` guard in `main`. Since `arr` is a fixed-size, non-empty array, `n` is a compile-time constant equal to 7, so the condition was always true; calling `merge_sort(arr, 0, n - 1)` directly is safe.

The original error handling (checking `malloc`, `printf` return values, and propagating failures via return codes) was preserved, as it was already correct.
