#include <stdio.h>
#include <stdlib.h>

int bisect_left(const int arr[], int n, int value) {
    int left = 0;
    int right = n;
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] < value) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }
    return left;
}

int main() {
    const int arr[] = {1, 2, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int value = 3;
    int index = bisect_left(arr, n, value);
    printf("The left insertion point for %d is at index %d\n", value, index);
    return 0;
}
