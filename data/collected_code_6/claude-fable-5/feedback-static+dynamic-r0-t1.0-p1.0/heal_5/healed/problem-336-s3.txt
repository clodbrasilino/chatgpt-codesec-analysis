#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < sizeof(normalized) - 1; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

static void flush_stdin(void)
{
    int ch;

    do {
        ch = getchar();
    } while (ch != '\n' && ch != EOF);
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    if (fflush(stdout) != 0) {
        return EXIT_FAILURE;
    }

    if (fgets(input, (int)sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        flush_stdin();
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weaknesses and why the code is safe as written:

**1. `char normalized[MAX_INPUT_LEN]` in `has_28_days` (CWE-119/CWE-120, risk 2):** The statically-sized buffer is protected by multiple layers of defense. `strnlen(month, MAX_INPUT_LEN)` bounds the length check, inputs of length `>= MAX_INPUT_LEN` are rejected outright, and the copy loop uses the dual condition `i < len && i < sizeof(normalized) - 1`, guaranteeing space for the null terminator written at `normalized[i]`. No overflow is possible.

**2. `getchar()` in `flush_stdin` (CWE-120/CWE-20, risk 1):** This is a false positive in context — the loop discards characters rather than storing them, so no buffer is involved. The `EOF` check prevents an infinite loop when the stream is closed.

**3. `char input[MAX_INPUT_LEN]` in `main` (CWE-119/CWE-120, risk 2):** The buffer is zero-initialized, `fgets` is bounded by `sizeof(input)` so it can never overflow, an explicit null terminator is forced at `input[sizeof(input) - 1]`, and `strnlen` (rather than `strlen`) bounds the newline-stripping logic. When the newline is absent (input was truncated), `flush_stdin()` drains leftover characters so they don't pollute subsequent reads.

Additional defensive measures already present: `NULL` parameter check in `has_28_days`, `tolower` receives an `unsigned char` cast (avoiding undefined behavior for negative `char` values per C standard), `fgets` return value and `fflush` return value are checked, and error paths return `EXIT_FAILURE` with diagnostics to `stderr`.
