```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

long long next_bigger(long long n)
{
    char digits[32];
    int len;
    int i;
    int pivot;
    int swap_idx;
    char tmp;
    int lo;
    int hi;
    long long result;

    if (n < 0) {
        return -1;
    }

    memset(digits, 0, sizeof(digits));

    len = snprintf(digits, sizeof(digits), "%lld", n);
    if (len < 0 || (size_t)len >= sizeof(digits)) {
        return -1;
    }

    pivot = -1;
    for (i = len - 2; i >= 0; i--) {
        if (digits[i] < digits[i + 1]) {
            pivot = i;
            break;
        }
    }

    if (pivot == -1) {
        return -1;
    }

    swap_idx = -1;
    for (i = len - 1; i > pivot; i--) {
        if (digits[i] > digits[pivot]) {
            swap_idx = i;
            break;
        }
    }

    if (swap_idx == -1) {
        return -1;
    }

    tmp = digits[pivot];
    digits[pivot] = digits[swap_idx];
    digits[swap_idx] = tmp;

    lo = pivot + 1;
    hi = len - 1;
    while (lo < hi) {
        tmp = digits[lo];
        digits[lo] = digits[hi];
        digits[hi] = tmp;
        lo++;
        hi--;
    }

    result = 0;
    for (i = 0; i < len; i++) {
        int d = digits[i] - '0';
        if (d < 0 || d > 9) {
            return -1;
        }
        if (result > (LLONG_MAX - d) / 10) {
            return -1;
        }
        result = result * 10 + d;
    }

    return result;
}

int main(void)
{
    long long inputs[] = {12LL, 513LL, 2017LL, 414LL, 144LL, 9LL, 111LL, 531LL};
    size_t count = sizeof(inputs) / sizeof(inputs[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        long long res = next_bigger(inputs[i]);
        if (res == -1) {
            if (printf("%lld -> no bigger number possible\n", inputs[i]) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (printf("%lld -> %lld\n", inputs[i], res) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary of the changes:**

1. **CWE-119/CWE-120 (buffer weakness on `digits[32]`):** The original code already mitigated this correctly — `snprintf` with `sizeof(digits)` guarantees no overflow, and the return value is checked against both negative values (encoding error) and truncation (`len >= sizeof(digits)`). Since a `long long` can be at most 20 characters plus sign and NUL, the 32-byte buffer is amply sized. I hardened it further by:
   - Zero-initializing the buffer with `memset`, ensuring no uninitialized bytes are ever read even if logic changes later.
   - Casting `len` to `size_t` for a type-correct comparison against `sizeof(digits)`.

2. **Defensive digit validation:** Added a `d < 0 || d > 9` check when reconstructing the number, guaranteeing the parsing loop can never operate on non-digit characters, which protects the overflow check invariant.

3. **Existing safe patterns preserved:** the `LLONG_MAX` overflow guard on reconstruction, the negative-input rejection, and all `printf` return-value checks remain intact.
