```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (isupper((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];

    while (fgets(buffer, (int)sizeof(buffer), stdin) != NULL) {
        size_t len = strnlen(buffer, sizeof(buffer));

        while (len > 0 && (buffer[len - 1] == '\n' || buffer[len - 1] == '\r')) {
            buffer[len - 1] = '\0';
            len--;
        }

        char *output = remove_lowercase(buffer);
        if (output == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        printf("%s\n", output);
        fflush(stdout);
        free(output);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes made:**

The key defect explaining the failing test cases was in the filtering predicate. The original code used `!islower(...)`, which retains *everything that is not a lowercase letter* — including digits, spaces, punctuation, and control characters. The test expectations (`PYTH`, `FID`, `STRG`) indicate the requirement is to output **only uppercase letters**. Switching the condition to `isupper((unsigned char)input[i])` satisfies that contract: inputs such as `PYThon 3` now yield `PYTH` rather than `PYTH 3`.

Regarding the Flawfinder warning on the statically-sized `buffer` (CWE-119/CWE-120): the existing code already mitigates it correctly —

- `fgets(buffer, (int)sizeof(buffer), stdin)` bounds the read to the buffer size and guarantees NUL termination.
- `strnlen(buffer, sizeof(buffer))` is used instead of `strlen`, capping the scan length.
- The newline/CR stripping loop checks `len > 0` before indexing, avoiding underflow.
- `remove_lowercase` allocates `len + 1` bytes and writes at most `len` characters plus the terminator, so no overflow is possible; the `malloc` return value and the `NULL` input are both checked.

The cast to `unsigned char` before calling `isupper` is retained, which is required to avoid undefined behavior when `char` is signed and the input contains bytes ≥ 0x80.
