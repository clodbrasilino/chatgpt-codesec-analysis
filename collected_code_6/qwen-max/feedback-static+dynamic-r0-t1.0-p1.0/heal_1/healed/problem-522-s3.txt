#include <stdio.h>
#include <stdlib.h>

int *longestBitonicSubsequence(const int arr[], int n) {
    int *lis = (int *)malloc(n * sizeof(int));
    int *lds = (int *)malloc(n * sizeof(int));
    int *result = (int *)malloc(n * sizeof(int));

    for (int i = 0; i < n; i++) {
        lis[i] = 1;
        lds[i] = 1;
    }

    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (arr[i] > arr[j] && lis[i] < lis[j] + 1) {
                lis[i] = lis[j] + 1;
            }
        }
    }

    for (int i = n - 2; i >= 0; i--) {
        for (int j = n - 1; j > i; j--) {
            if (arr[i] > arr[j] && lds[i] < lds[j] + 1) {
                lds[i] = lds[j] + 1;
            }
        }
    }

    int maxLength = 0;
    for (int i = 0; i < n; i++) {
        if (lis[i] + lds[i] - 1 > maxLength) {
            maxLength = lis[i] + lds[i] - 1;
        }
    }

    int index = 0;
    for (int i = 0; i < n; i++) {
        if (lis[i] + lds[i] - 1 == maxLength) {
            result[index++] = arr[i];
        }
    }

    free(lis);
    free(lds);

    return result;
}

int main() {
    const int arr[] = {1, 11, 2, 10, 4, 5, 2, 1};
    int n = sizeof(arr) / sizeof(arr[0]);

    int *result = longestBitonicSubsequence(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%d ", result[i]);
    }
    free(result);

    return 0;
}
