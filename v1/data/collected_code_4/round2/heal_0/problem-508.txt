```C
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool checkOrder(int *list1, int size1, int *list2, int size2) {
    int i, j, prev = -1;

    for(i = 0; i < size1; i++){
        for(j = 0; j < size2; j++){
            if(list1[i] == list2[j]){
                if(j < prev)
                    return false;
                prev = j;
                break;
            }
        }
    }
    return true;
}

int main() {
    int list1[] = {1, 3, 5, 7, 9};
    int list2[] = {2, 4, 6, 8, 10, 1, 3, 5, 7, 9};
    int size1 = sizeof(list1) / sizeof(list1[0]);
    int size2 = sizeof(list2) / sizeof(list2[0]);

    bool isSameOrder = checkOrder(list1, size1, list2, size2);

    if (isSameOrder) {
        printf("Common elements are in the same order\n");
    } else {
        printf("Common elements are not in the same order\n");
    }

    return 0;
}
```
