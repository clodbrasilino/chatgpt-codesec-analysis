```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;
    size_t total_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    total_bytes = n * sizeof(int);

    k = k % n;
    if (k == 0) {
        return 0;
    }

    if (k > n || (n - k) > n) {
        return -1;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    if (head_bytes > total_bytes || tail_bytes > total_bytes) {
        return -1;
    }

    if (head_bytes + tail_bytes != total_bytes) {
        return -1;
    }

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    if (head_bytes > total_bytes) {
        free(temp);
        return -1;
    }
    memcpy(temp, arr, head_bytes);

    if (tail_bytes > total_bytes) {
        free(temp);
        return -1;
    }
    memmove(arr, arr + k, tail_bytes);

    if ((n - k) > n || head_bytes > total_bytes - tail_bytes) {
        free(temp);
        return -1;
    }
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Error: rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis summary:**

The original code was already largely defensive, and the flawfinder warnings on `memcpy` (CWE-120) are addressed as follows:

1. **First `memcpy(temp, arr, head_bytes)`** — The destination `temp` is allocated with exactly `head_bytes`, so the copy can never overflow it. The source `arr` is guaranteed to hold at least `head_bytes` because `k < n` after the modulo, hence `head_bytes < total_bytes`. The guard `head_bytes > total_bytes` before the call makes this explicit.

2. **Second `memcpy(arr + (n - k), temp, head_bytes)`** — The destination region starting at `arr + (n - k)` has exactly `k * sizeof(int) == head_bytes` bytes available. The check `head_bytes > total_bytes - tail_bytes` verifies the destination capacity precisely, since `total_bytes - tail_bytes` equals the remaining space at the destination offset. I also added `(n - k) > n` as an underflow sanity guard.

3. **Overflow protection** — The `n > SIZE_MAX / sizeof(int)` check prevents integer overflow in `n * sizeof(int)`, which is the classic precursor to heap overflows via undersized `malloc`.

4. **`memmove` usage** — Correctly used instead of `memcpy` for the overlapping shift of the tail, with `tail_bytes` bounded by `total_bytes`.

The remaining checks are provably redundant given `k = k % n` and the early return on `k == 0`, but they serve as belt-and-suspenders assertions that satisfy static analyzers and protect against future refactoring mistakes at negligible runtime cost.
