#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct Sublist {
    Node* head;
    struct Sublist* next;
} Sublist;

int* get_last_elements(Sublist* list, int* count) {
    if (!list || !count) {
        if (count) *count = 0;
        return NULL;
    }

    int list_count = 0;
    Sublist* current_sublist = list;
    while (current_sublist) {
        list_count++;
        current_sublist = current_sublist->next;
    }

    int* last_elements = (int*)malloc(list_count * sizeof(int));
    if (!last_elements) {
        *count = 0;
        return NULL;
    }

    current_sublist = list;
    int index = 0;
    while (current_sublist) {
        Node* current_node = current_sublist->head;
        if (!current_node) {
            free(last_elements);
            *count = 0;
            return NULL;
        }
        while (current_node->next) {
            current_node = current_node->next;
        }
        last_elements[index++] = current_node->data;
        current_sublist = current_sublist->next;
    }

    *count = list_count;
    return last_elements;
}

void free_sublists(Sublist* list) {
    while (list) {
        Node* current_node = list->head;
        while (current_node) {
            Node* temp_node = current_node;
            current_node = current_node->next;
            free(temp_node);
        }
        Sublist* temp_sublist = list;
        list = list->next;
        free(temp_sublist);
    }
}

int main() {
    Sublist* list = (Sublist*)malloc(sizeof(Sublist));
    if (!list) return 1;
    
    list->head = (Node*)malloc(sizeof(Node));
    if (!list->head) {
        free(list);
        return 1;
    }
    list->head->data = 1;
    list->head->next = (Node*)malloc(sizeof(Node));
    if (!list->head->next) {
        free(list->head);
        free(list);
        return 1;
    }
    list->head->next->data = 2;
    list->head->next->next = NULL;

    list->next = (Sublist*)malloc(sizeof(Sublist));
    if (!list->next) {
        free_sublists(list);
        return 1;
    }
    list->next->head = (Node*)malloc(sizeof(Node));
    if (!list->next->head) {
        free_sublists(list);
        return 1;
    }
    list->next->head->data = 3;
    list->next->head->next = (Node*)malloc(sizeof(Node));
    if (!list->next->head->next) {
        free_sublists(list);
        return 1;
    }
    list->next->head->next->data = 4;
    list->next->head->next->next = NULL;
    list->next->next = NULL;

    int count = 0;
    int* last_elements = get_last_elements(list, &count);

    if (last_elements) {
        for (int i = 0; i < count; i++) {
            printf("%d\n", last_elements[i]);
        }
        free(last_elements);
    }

    free_sublists(list);

    return 0;
}
