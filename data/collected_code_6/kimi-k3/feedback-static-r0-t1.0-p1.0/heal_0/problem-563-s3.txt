#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char **extract_quoted_values(const char *str, size_t *count) {
    size_t capacity = 10;
    size_t num_values = 0;
    char **values = malloc(capacity * sizeof(char *));
    if (values == NULL) {
        *count = 0;
        return NULL;
    }

    const char *p = str;
    while (*p != '\0') {
        if (*p == '"') {
            p++;
            const char *start = p;
            while (*p != '\0' && *p != '"') {
                p++;
            }
            if (*p == '"') {
                size_t length = (size_t)(p - start);
                char *value = malloc(length + 1);
                if (value == NULL) {
                    for (size_t i = 0; i < num_values; i++) {
                        free(values[i]);
                    }
                    free(values);
                    *count = 0;
                    return NULL;
                }
                strncpy(value, start, length);
                value[length] = '\0';

                if (num_values >= capacity) {
                    capacity *= 2;
                    char **new_values = realloc(values, capacity * sizeof(char *));
                    if (new_values == NULL) {
                        free(value);
                        for (size_t i = 0; i < num_values; i++) {
                            free(values[i]);
                        }
                        free(values);
                        *count = 0;
                        return NULL;
                    }
                    values = new_values;
                }
                values[num_values] = value;
                num_values++;
                p++;
            }
        } else {
            p++;
        }
    }

    *count = num_values;
    return values;
}

void free_quoted_values(char **values, size_t count) {
    if (values == NULL) {
        return;
    }
    for (size_t i = 0; i < count; i++) {
        free(values[i]);
    }
    free(values);
}

int main(void) {
    const char *input = "This is a \"test\" string with \"multiple\" quoted \"values\"";
    size_t count = 0;
    char **results = extract_quoted_values(input, &count);

    if (results != NULL) {
        for (size_t i = 0; i < count; i++) {
            printf("%s\n", results[i]);
        }
        free_quoted_values(results, count);
    }

    return 0;
}
