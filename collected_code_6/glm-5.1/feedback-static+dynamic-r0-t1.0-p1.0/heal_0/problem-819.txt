#include <stdio.h>
#include <stdlib.h>

int count_consecutive_duplicates(const int *array, size_t size) {
    if (array == NULL || size == 0) {
        return 0;
    }

    int count = 0;
    for (size_t i = 1; i < size; i++) {
        if (array[i] == array[i - 1]) {
            count++;
        }
    }
    return count;
}

int main(void) {
    int data[] = {1, 2, 2, 3, 3, 3, 4, 4, 5, 1, 1};
    size_t n = sizeof(data) / sizeof(data[0]);

    int frequency = count_consecutive_duplicates(data, n);
    printf("%d\n", frequency);

    return 0;
}
