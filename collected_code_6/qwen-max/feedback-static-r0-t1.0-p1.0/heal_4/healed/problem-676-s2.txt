#include <stdio.h>
#include <string.h>
#include <regex.h>

void clean_string(char *str) {
    regex_t regex;
    regmatch_t match[1];
    char result[1024] = "";
    size_t pos = 0;

    if (regcomp(&regex, "[^a-zA-Z0-9]", 0)) {
        return;
    }

    while (regexec(&regex, str + pos, 1, match, 0) == 0) {
        if (match[0].rm_so > 0) {
            size_t len = match[0].rm_so;
            size_t current_len = strnlen(result, sizeof(result));
            if (current_len + len < sizeof(result) - 1) {
                strncat(result, str + pos, len);
            }
        }
        pos += match[0].rm_eo;
    }

    size_t remaining_len = strnlen(str + pos, sizeof(str) - pos);
    size_t current_len = strnlen(result, sizeof(result));
    if (pos < strlen(str) && current_len + remaining_len < sizeof(result) - 1) {
        strncat(result, str + pos, remaining_len);
    }

    strncpy(str, result, sizeof(result) - 1);
    str[sizeof(result) - 1] = '\0';
    regfree(&regex);
}

int main() {
    char input[] = "Hello, World! 123";
    clean_string(input);
    printf("%s\n", input);
    return 0;
}
