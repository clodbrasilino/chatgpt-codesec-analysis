#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 255U

char *remove_even_chars(const char *input, size_t max_len);

char *remove_even_chars(const char *input, size_t max_len)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, max_len);

    if (len == SIZE_MAX) {
        return NULL;
    }

    result = malloc(len + 1U);
    if (result == NULL) {
        return NULL;
    }

    j = 0U;
    for (i = 0U; i < len; i++) {
        if ((i % 2U) != 0U) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN + 1U];
    char *output;
    size_t len;

    (void)memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        (void)fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if ((len > 0U) && (buffer[len - 1U] == '\n')) {
        buffer[len - 1U] = '\0';
    }

    output = remove_even_chars(buffer, sizeof(buffer));
    if (output == NULL) {
        (void)fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", output) < 0) {
        free(output);
        return EXIT_FAILURE;
    }

    free(output);
    return EXIT_SUCCESS;
}

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen`, which bounds the scan to a maximum length, preventing over-reads on non-null-terminated strings. The `remove_even_chars` function now takes an explicit `max_len` parameter so it never reads beyond the caller's buffer.

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is now defined via a named constant (`MAX_INPUT_LEN`), the buffer is zero-initialized with `memset` before use, and null termination is explicitly enforced after `fgets` as a defensive measure. `fgets` already receives `sizeof(buffer)` to guarantee bounds checking.

3. **Additional hardening**: The `int` cast on `sizeof(buffer)` for `fgets` avoids implicit conversion warnings, and return values of `fprintf`/`memset` are explicitly voided to satisfy strict static analyzers (e.g., MISRA-style rules).

Note: `strnlen` requires POSIX (`<string.h>` with `_POSIX_C_SOURCE >= 200809L`) or C23; if targeting strict C89/C99 portability, a small local bounded-length helper would be the equivalent fix.
