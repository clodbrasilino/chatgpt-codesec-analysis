```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SENTENCE_LEN 4096

int find_first_max_even_word(const char *sentence, char *result, size_t result_size)
{
    size_t i = 0;
    size_t len = 0;
    size_t max_len = 0;
    size_t best_start = 0;
    int found = 0;

    if (sentence == NULL || result == NULL || result_size == 0) {
        return -1;
    }

    len = strnlen(sentence, MAX_SENTENCE_LEN);
    if (len == MAX_SENTENCE_LEN) {
        return -1;
    }

    while (i < len) {
        while (i < len && sentence[i] == ' ') {
            i++;
        }
        if (i < len) {
            size_t start = i;
            size_t word_len = 0;
            while (i < len && sentence[i] != ' ') {
                i++;
            }
            word_len = i - start;
            if ((word_len % 2 == 0) && (word_len > max_len)) {
                max_len = word_len;
                best_start = start;
                found = 1;
            }
        }
    }

    if (!found) {
        return -1;
    }

    if (max_len >= result_size) {
        return -1;
    }

    memcpy(result, sentence + best_start, max_len);
    result[max_len] = '\0';
    return 0;
}

int main(void)
{
    const char *sentence = "It is a pleasant day today";
    char result[256] = {0};

    if (find_first_max_even_word(sentence, result, sizeof(result)) == 0) {
        if (printf("First maximum length even word: %s\n", result) < 0) {
            return EXIT_FAILURE;
        }
    } else {
        if (printf("No even length word found\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **`memcpy` (CWE-120):** This is a false positive in the original code. The copy is already properly guarded — `max_len >= result_size` is checked immediately before the call, guaranteeing that `max_len` bytes plus the null terminator fit within `result_size`. No change is required; the bounds check constitutes the fix.

2. **Statically-sized `char result[256]` (CWE-119/CWE-120):** Also a false positive. The buffer size is passed to the function via `sizeof(result)`, and the function refuses to write beyond `result_size`. The array is zero-initialized and all writes are bounds-checked.

**Additional defensive properties already present:**

- `NULL` and zero-size parameter validation at function entry.
- `strnlen` with `MAX_SENTENCE_LEN` prevents unbounded reads on non-terminated input; input at or beyond the cap is rejected.
- The return value of `printf` is checked, and `EXIT_SUCCESS`/`EXIT_FAILURE` are used for portable exit codes.

The code is safe as written; the rewrite preserves the existing bounds checks, which are the correct mitigations for both flagged CWEs.
