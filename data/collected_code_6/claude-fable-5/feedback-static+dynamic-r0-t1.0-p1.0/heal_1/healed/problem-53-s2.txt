```c
#include <stdio.h>
#include <string.h>
#include <stdbool.h>

#define BUFFER_SIZE 256

bool first_last_equal(const char *str, size_t max_len)
{
    size_t len;

    if (str == NULL) {
        return false;
    }

    len = strnlen(str, max_len);
    if (len == 0 || len == max_len) {
        return false;
    }

    return str[0] == str[len - 1];
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    if (buffer[0] == '\0') {
        printf("Empty string provided\n");
        return 1;
    }

    if (first_last_equal(buffer, sizeof(buffer))) {
        printf("The first and last characters are equal\n");
    } else {
        printf("The first and last characters are not equal\n");
    }

    return 0;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (over-read via `strlen`)**: Replaced both calls to `strlen` with `strnlen`, which takes an explicit maximum length and cannot read past the buffer even if the string is not null-terminated. The `first_last_equal` function now receives the buffer's capacity as a parameter and rejects input where `len == max_len` (indicating no terminator was found within bounds).

2. **CWE-119/CWE-120 (statically-sized array)**: The buffer size is defined as a named constant (`BUFFER_SIZE`), `fgets` is bounded by `sizeof(buffer)` (which it already was — this is the correct pattern), and an explicit `buffer[sizeof(buffer) - 1] = '\0'` guarantees null-termination as defense-in-depth before any string operation touches the buffer.

3. **Defensive guarantees preserved**: The existing `NULL` check, empty-string check, and `fgets` error handling remain intact, so all original behavior is retained while the flagged risks are mitigated.
