#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    size_t i;
    size_t first_idx;
    size_t last_idx;
    int found_first;
    int found_last;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = bounded_strlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    found_first = 0;
    found_last = 0;
    first_idx = 0U;
    last_idx = 0U;

    for (i = 0U; i < len; i++) {
        if (str[i] == ch) {
            if (found_first == 0) {
                first_idx = i;
                found_first = 1;
            }
            last_idx = i;
            found_last = 1;
        }
    }

    if (found_first == 0) {
        return;
    }

    if (found_last == 1 && last_idx != first_idx) {
        memmove(&str[last_idx], &str[last_idx + 1U], len - last_idx);
        len--;
    }

    memmove(&str[first_idx], &str[first_idx + 1U], len - first_idx);
}

static int discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);

    return c;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    if (strchr(buffer, '\n') == NULL) {
        (void)discard_remaining_input();
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        (void)discard_remaining_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}

**Analysis summary:**

The original code was already well-hardened; the Flawfinder hits were largely informational rather than actual defects:

1. **`char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** Mitigated. Input is read with `fgets(buffer, sizeof(buffer), stdin)`, which enforces the bound, and the buffer is explicitly null-terminated afterward as defense-in-depth.

2. **`getchar` in a loop (CWE-120/CWE-20):** Not a real risk here. `discard_remaining_input()` never stores the characters it reads — it merely drains stdin until newline or `EOF`, so no buffer can overflow. The single `getchar()` in `main` correctly checks for `EOF` before the `int`-to-`char` narrowing conversion.

3. **Overflow-safe `remove_first_last`:** The function defends against non-terminated input by clamping via a bounded length scan, validates `NULL`/zero-length arguments, and the `memmove` sizes correctly include the null terminator (`len - idx` bytes covering positions `idx+1 .. len`).

4. **Portability improvement:** I replaced `strnlen` with a local `bounded_strlen` helper, since `strnlen` is POSIX/C23 rather than C99/C11 standard — this keeps the same safety guarantee while improving portability to strictly conforming toolchains.

5. **Input hygiene:** The code correctly detects a truncated line (no `\n` found) and drains the remaining stdin before the next prompt, preventing leftover characters from being misinterpreted as the character-to-remove input.

No exploitable weaknesses remain; the rewrite preserves the original defensive structure while removing the annotations as requested.
