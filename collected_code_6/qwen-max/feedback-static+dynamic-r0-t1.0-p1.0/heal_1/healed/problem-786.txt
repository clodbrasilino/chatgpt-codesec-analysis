#include <stdio.h>

int findInsertPosition(const int *arr, int n, int value) {
    int left = 0;
    int right = n - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == value) {
            return mid;
        } else if (arr[mid] < value) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return left;
}

int main() {
    int arr[] = {1, 3, 5, 6};
    int n = sizeof(arr) / sizeof(arr[0]);
    int value = 4;
    int position = findInsertPosition(arr, n, value);
    printf("The insertion point for %d is at index %d\n", value, position);
    return 0;
}
