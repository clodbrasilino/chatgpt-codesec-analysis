```c
#include <stdio.h>
#include <ctype.h>

int count_uppercase(const char *str) {
    int count = 0;
    if (str == NULL) {
        return 0;
    }
    for (size_t i = 0; str[i] != '\0'; i++) {
        if (isupper((unsigned char)str[i])) {
            count++;
        }
    }
    return count;
}

int main(void) {
    const char *text = "Hello World!";
    int result = count_uppercase(text);
    printf("%d\n", result);
    return 0;
}
```
