```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096U

static char *longest_common_subsequence(const char *a, const char *b)
{
    size_t m;
    size_t n;
    size_t i;
    size_t j;
    size_t idx;
    int **dp;
    char *result;

    if (a == NULL || b == NULL) {
        return NULL;
    }

    m = strnlen(a, MAX_INPUT_LEN);
    n = strnlen(b, MAX_INPUT_LEN);

    if (m >= MAX_INPUT_LEN || n >= MAX_INPUT_LEN) {
        return NULL;
    }

    if (m + 1 > SIZE_MAX / sizeof(int *) || n + 1 > SIZE_MAX / sizeof(int)) {
        return NULL;
    }

    dp = malloc((m + 1) * sizeof(int *));
    if (dp == NULL) {
        return NULL;
    }

    for (i = 0; i <= m; i++) {
        dp[i] = calloc(n + 1, sizeof(int));
        if (dp[i] == NULL) {
            while (i > 0) {
                i--;
                free(dp[i]);
            }
            free(dp);
            return NULL;
        }
    }

    for (i = 1; i <= m; i++) {
        for (j = 1; j <= n; j++) {
            if (a[i - 1] == b[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else if (dp[i - 1][j] >= dp[i][j - 1]) {
                dp[i][j] = dp[i - 1][j];
            } else {
                dp[i][j] = dp[i][j - 1];
            }
        }
    }

    idx = (size_t)dp[m][n];
    result = malloc(idx + 1);
    if (result == NULL) {
        for (i = 0; i <= m; i++) {
            free(dp[i]);
        }
        free(dp);
        return NULL;
    }

    result[idx] = '\0';
    i = m;
    j = n;
    while (i > 0 && j > 0) {
        if (a[i - 1] == b[j - 1]) {
            idx--;
            result[idx] = a[i - 1];
            i--;
            j--;
        } else if (dp[i - 1][j] >= dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }

    for (i = 0; i <= m; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    const char *seq1 = "AGGTAB";
    const char *seq2 = "GXTXAYB";
    char *lcs;

    lcs = longest_common_subsequence(seq1, seq2);
    if (lcs == NULL) {
        fprintf(stderr, "Failed to compute LCS\n");
        return EXIT_FAILURE;
    }

    if (printf("Sequence 1: %s\n", seq1) < 0 ||
        printf("Sequence 2: %s\n", seq2) < 0 ||
        printf("LCS: %s\n", lcs) < 0 ||
        printf("Length: %zu\n", strnlen(lcs, MAX_INPUT_LEN)) < 0) {
        free(lcs);
        return EXIT_FAILURE;
    }

    free(lcs);
    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (buffer over-read via `strlen`)**: Replaced all three `strlen` calls with `strnlen` bounded by `MAX_INPUT_LEN`. This guarantees the scan never reads past a fixed limit, even if a non-NUL-terminated buffer were passed in. If the string hits the bound (i.e., no terminator was found within the limit), the function rejects the input and returns `NULL` rather than proceeding with potentially truncated/invalid data.

2. **Defensive integer-overflow checks**: Added guards on `(m + 1) * sizeof(int *)` and `(n + 1) * sizeof(int)` against `SIZE_MAX` before allocation. With the length cap these can't realistically overflow, but the checks make the allocation sizes provably safe. Note that `calloc` already performs its own internal overflow check on `count * size`, which is another reason it was kept for the row allocations.

3. **Preserved existing correct behavior**: The original code already handled the harder problems correctly — `NULL` argument checks, full cleanup on partial allocation failure (the backwards `free` loop), `calloc` zero-initialization for the DP base cases, checked `printf` return values, and no memory leaks on any exit path — so that structure was retained unchanged.

Note: `strnlen` requires `_POSIX_C_SOURCE >= 200809L` or a C23-conforming library; if you need strict C89/C99 portability, a small static bounded-length helper would be the substitute.
