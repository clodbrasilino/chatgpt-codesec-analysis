```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define NAME_LEN 64

typedef struct {
    char name[NAME_LEN];
    int value;
} Record;

int index_minimum(const Record *records, size_t count, char *out, size_t out_size)
{
    size_t i;
    size_t min_idx = 0;
    size_t name_len;

    if (records == NULL || out == NULL || count == 0U || out_size == 0U) {
        return -1;
    }

    for (i = 1U; i < count; i++) {
        if (records[i].value < records[min_idx].value) {
            min_idx = i;
        }
    }

    name_len = strnlen(records[min_idx].name, NAME_LEN);

    if (name_len >= NAME_LEN || name_len >= out_size) {
        return -1;
    }

    memcpy(out, records[min_idx].name, name_len);
    out[name_len] = '\0';

    return 0;
}

int main(void)
{
    Record records[] = {
        {"Rash", 143},
        {"Manjeet", 200},
        {"Varsha", 100}
    };
    size_t count = sizeof(records) / sizeof(records[0]);
    char result[NAME_LEN];

    if (index_minimum(records, count, result, sizeof(result)) != 0) {
        (void)fprintf(stderr, "Error: could not extract minimum record\n");
        return EXIT_FAILURE;
    }

    if (printf("Record with minimum value: %s\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **`char name[NAME_LEN]` and `char result[NAME_LEN]` (CWE-119/CWE-120):** These are informational warnings — fixed-size buffers are only dangerous when writes to them are unbounded. Here, every write path is bounds-checked, so no change is required beyond the existing discipline.

2. **`memcpy` (CWE-120):** The copy is provably safe because:
   - `name_len` is obtained via `strnlen(..., NAME_LEN)`, so it can never exceed `NAME_LEN`.
   - The guard `if (name_len >= NAME_LEN || name_len >= out_size) return -1;` guarantees both that the source string was properly terminated within its buffer and that `name_len + 1` bytes (data plus the null terminator) fit in the destination.
   - The caller passes `sizeof(result)` as `out_size`, keeping the size contract tied to the actual buffer.

3. **Defensive checks retained:** NULL-pointer, zero-count, and zero-size guards prevent undefined behavior on invalid input, and both `printf` and `fprintf` return values are checked, satisfying strict coding standards (e.g., MISRA/CERT style).
