#include <stdio.h>
#include <limits.h>

#define max(a, b) ((a > b)? a : b) 
#define minJumps(n, arr) minJumpsUtil(arr, n) 

int minJumpsUtil(int arr[], int n) 
{ 
    if (n <= 1) 
        return 0; 
  
    if (arr[0] == 0) 
        return -1; 
  
    int maxReach = arr[0]; 
    int step = arr[0]; 
    int jump = 1; 
  
    for (int i = 1; i < n; i++) 
    { 
        if (i == n-1) 
            return jump; 
  
        maxReach = max(maxReach, i + arr[i]); 
  
        step--; 
  
        if (step == 0) 
        { 
            jump++; 
  
            if(i >= maxReach) 
                return -1; 
  
            step = maxReach - i; 
        } 
    } 
  
    return -1; 
} 
  
int main() 
{ 
    int arr[30], len;
    printf("Enter the number of elements: ");
    scanf("%d", &len);

    printf("Enter the array elements: ");
    for (int i = 0; i < len; ++i)
        scanf("%d", &arr[i]);
  
    printf("Minimum number of jumps to reach end is: %d", minJumps(len, arr)); 
    return 0; 
}
