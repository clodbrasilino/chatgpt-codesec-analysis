```c
#include <stdio.h>
#include <math.h>
#include <errno.h>
#include <stdlib.h>
#include <fenv.h>

double radians_to_degrees(double radians)
{
    return radians * (180.0 / M_PI);
}

int main(void)
{
    double radians;
    
    if (printf("Enter radians: ") < 0) {
        return EXIT_FAILURE;
    }
    
    if (scanf("%lf", &radians) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    
    if (isnan(radians) || isinf(radians)) {
        fprintf(stderr, "Invalid numeric value\n");
        return EXIT_FAILURE;
    }
    
    errno = 0;
    feclearexcept(FE_ALL_EXCEPT);
    
    double degrees = radians_to_degrees(radians);
    
    if (errno != 0 || fetestexcept(FE_OVERFLOW | FE_UNDERFLOW)) {
        fprintf(stderr, "Conversion error\n");
        return EXIT_FAILURE;
    }
    
    if (printf("%f radians = %f degrees\n", radians, degrees) < 0) {
        return EXIT_FAILURE;
    }
    
    return EXIT_SUCCESS;
}
```
