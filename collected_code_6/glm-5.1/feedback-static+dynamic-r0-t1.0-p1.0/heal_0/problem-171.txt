```c
#include <stdio.h>
#include <stdlib.h>

double calculate_pentagon_perimeter(double side_length) {
    if (side_length < 0.0) {
        return -1.0;
    }
    return 5.0 * side_length;
}

int main(int argc, char *argv[]) {
    double side = 5.0;
    double perimeter = calculate_pentagon_perimeter(side);
    if (perimeter < 0.0) {
        return EXIT_FAILURE;
    }
    printf("%f\n", perimeter);
    return EXIT_SUCCESS;
}
```
