```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX_BITS (sizeof(unsigned int) * CHAR_BIT)

static int decimal_to_binary(unsigned int number, char *buffer, size_t buffer_size)
{
    size_t index = 0U;
    size_t start;
    size_t end;

    if (buffer == NULL || buffer_size < 2U) {
        return -1;
    }

    if (number == 0U) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (number > 0U && index < (buffer_size - 1U)) {
        buffer[index] = (char)('0' + (number % 2U));
        number /= 2U;
        index++;
    }

    if (number > 0U) {
        return -1;
    }

    buffer[index] = '\0';

    start = 0U;
    end = index - 1U;
    while (start < end) {
        char temp = buffer[start];
        buffer[start] = buffer[end];
        buffer[end] = temp;
        start++;
        end--;
    }

    return 0;
}

int main(void)
{
    unsigned int value = 0U;
    char binary[MAX_BITS + 1U] = {0};
    int scan_result;

    printf("Enter a non-negative decimal number: ");

    scan_result = scanf("%u", &value);
    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_binary(value, binary, sizeof(binary)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Binary equivalent: %s\n", binary);

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The Flawfinder warning (CWE-119/CWE-120) on the `binary` buffer is a **false positive** in this code. The original implementation already contains the correct mitigations, which I have preserved:

1. **Correct buffer sizing** — `MAX_BITS + 1U` guarantees room for every bit of an `unsigned int` plus the NUL terminator, regardless of platform (it's derived from `sizeof(unsigned int) * CHAR_BIT` rather than a hard-coded 32).

2. **Bounds checking in the conversion loop** — the condition `index < (buffer_size - 1U)` prevents writing past the buffer, and the post-loop check `if (number > 0U)` detects truncation and reports failure rather than silently producing a wrong result.

3. **Size passed explicitly** — `sizeof(binary)` is passed to the function, so the callee never assumes a buffer length.

4. **Defensive parameter validation** — `buffer == NULL || buffer_size < 2U` guards against degenerate inputs before any write occurs.

5. **Safe input parsing** — `scanf("%u", &value)` writes to a fixed-size integer (no string format specifier without a width limit), and its return value is checked.

No functional or security changes were required; the code is already robust against the flagged weakness.
