#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int min_swaps_to_convert(const char *str1, const char *str2) {
    if (str1 == NULL || str2 == NULL) {
        return -1;
    }
    
    size_t len1 = strlen(str1);
    size_t len2 = strlen(str2);
    
    if (len1 != len2) {
        return -1;
    }
    
    int count1_0 = 0, count1_1 = 0;
    int count2_0 = 0, count2_1 = 0;
    
    for (size_t i = 0; i < len1; i++) {
        if (str1[i] == '0') count1_0++;
        else if (str1[i] == '1') count1_1++;
        else return -1;
        
        if (str2[i] == '0') count2_0++;
        else if (str2[i] == '1') count2_1++;
        else return -1;
    }
    
    if (count1_0 != count2_0 || count1_1 != count2_1) {
        return -1;
    }
    
    int mismatches = 0;
    for (size_t i = 0; i < len1; i++) {
        if (str1[i] != str2[i]) {
            mismatches++;
        }
    }
    
    return mismatches / 2;
}

int main(void) {
    char str1[1000];
    char str2[1000];
    
    printf("Enter first binary string: ");
    if (fgets(str1, sizeof(str1), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }
    str1[strcspn(str1, "\n")] = '\0';
    
    printf("Enter second binary string: ");
    if (fgets(str2, sizeof(str2), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }
    str2[strcspn(str2, "\n")] = '\0';
    
    int result = min_swaps_to_convert(str1, str2);
    
    if (result == -1) {
        printf("Cannot convert strings (invalid input or different character counts)\n");
        return 1;
    }
    
    printf("Minimum swaps required: %d\n", result);
    
    return 0;
}
