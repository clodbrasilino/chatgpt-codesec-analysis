```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int first;
    int second;
} Tuple;

typedef struct {
    Tuple *data;
    int size;
} TupleList;

int compare_tuples(const void *a, const void *b) {
    const Tuple *tuple_a = (const Tuple *)a;
    const Tuple *tuple_b = (const Tuple *)b;
    return tuple_a->first - tuple_b->first;
}

void sort_tuples(TupleList *list) {
    if (list == NULL || list->data == NULL || list->size <= 0) {
        return;
    }
    qsort(list->data, list->size, sizeof(Tuple), compare_tuples);
}

int main(void) {
    Tuple tuples[] = {
        {3, 10},
        {1, 20},
        {2, 30},
        {1, 40},
        {3, 50}
    };
    int size = sizeof(tuples) / sizeof(tuples[0]);
    TupleList list;
    int i;

    list.data = malloc((size_t)size * sizeof(Tuple));
    if (list.data == NULL) {
        return 1;
    }
    for (i = 0; i < size; i++) {
        list.data[i] = tuples[i];
    }
    list.size = size;

    sort_tuples(&list);

    for (i = 0; i < list.size; i++) {
        printf("(%d, %d) ", list.data[i].first, list.data[i].second);
    }
    printf("\n");

    free(list.data);
    list.data = NULL;
    list.size = 0;

    return 0;
}
```
