```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LEN 100

struct entry {
    char str[MAX_LEN];
    int count;
};

static const char *second_most_frequent(const char *arr[], size_t n,
                                        struct entry *table, size_t *table_size)
{
    size_t i;
    size_t j;
    size_t unique = 0;
    size_t first_idx;
    size_t second_idx;
    int found_second = 0;

    if (arr == NULL || table == NULL || table_size == NULL || n == 0) {
        return NULL;
    }

    for (i = 0; i < n; i++) {
        int found = 0;
        size_t len;

        if (arr[i] == NULL) {
            return NULL;
        }

        len = strnlen(arr[i], MAX_LEN);
        if (len >= MAX_LEN) {
            return NULL;
        }

        for (j = 0; j < unique; j++) {
            if (strncmp(table[j].str, arr[i], MAX_LEN) == 0) {
                table[j].count++;
                found = 1;
                break;
            }
        }

        if (found == 0) {
            if (len >= sizeof(table[unique].str)) {
                return NULL;
            }
            memcpy(table[unique].str, arr[i], len);
            table[unique].str[len] = '\0';
            table[unique].count = 1;
            unique++;
        }
    }

    *table_size = unique;

    if (unique < 2) {
        return NULL;
    }

    first_idx = 0;
    second_idx = 0;

    for (i = 1; i < unique; i++) {
        if (table[i].count > table[first_idx].count) {
            first_idx = i;
        }
    }

    for (i = 0; i < unique; i++) {
        if (i == first_idx) {
            continue;
        }
        if (found_second == 0 || table[i].count > table[second_idx].count) {
            second_idx = i;
            found_second = 1;
        }
    }

    if (found_second == 0) {
        return NULL;
    }

    return table[second_idx].str;
}

int main(void)
{
    const char *sequence[] = {
        "apple", "banana", "apple", "orange",
        "banana", "apple", "orange", "banana", "banana"
    };
    size_t n = sizeof(sequence) / sizeof(sequence[0]);
    struct entry *table;
    size_t table_size = 0;
    const char *result;

    table = calloc(n, sizeof(struct entry));
    if (table == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    result = second_most_frequent(sequence, n, table, &table_size);

    if (result == NULL) {
        fprintf(stderr, "No second most frequent string found\n");
        free(table);
        return EXIT_FAILURE;
    }

    if (printf("Second most frequent string: %s\n", result) < 0) {
        free(table);
        return EXIT_FAILURE;
    }

    free(table);
    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is safe:**

1. **Flawfinder `char str[MAX_LEN]` (CWE-119/CWE-120):** The statically-sized buffer is safe here because every write path into it is bounds-checked. Input strings are measured with `strnlen(arr[i], MAX_LEN)`, and any string of length `>= MAX_LEN` is rejected before any copy occurs, guaranteeing space for the NUL terminator.

2. **Flawfinder `memcpy` (CWE-120):** The copy is guarded twice — first by the `len >= MAX_LEN` early rejection, and again by the explicit `len >= sizeof(table[unique].str)` check immediately before the `memcpy`. The destination can therefore always hold `len` bytes plus the manually appended `'\0'`. This is actually preferable to `strcpy`/`strncpy`: it avoids unbounded copies and avoids `strncpy`'s non-termination pitfall.

Additional defensive properties already present and preserved:

- `strncmp(..., MAX_LEN)` bounds all comparisons.
- All pointer parameters (`arr`, `table`, `table_size`, individual `arr[i]`) are NULL-checked.
- `unique` can never exceed `n`, and the caller allocates `n` entries via `calloc`, so `table[unique]` never goes out of bounds.
- `calloc` zero-initializes the table, so no uninitialized reads occur.
- All allocation paths free `table` before returning, so there are no leaks, and `printf`'s return value is checked.

The Flawfinder hits are false positives given the surrounding validation; no functional changes were required, only confirmation that the mitigations are correct and complete.
