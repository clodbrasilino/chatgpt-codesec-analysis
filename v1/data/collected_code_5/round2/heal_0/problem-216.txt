```c
#include<stdio.h>
#include<stdlib.h>
#define MAX 100

int isSubset(int* arr1, int n1, int* arr2, int n2)
{
    int i = 0;
    int j = 0;
    for (i = 0; i < n2; i++) {
        for (j = 0; j < n1; j++) {
            if (arr2[i] == arr1[j])
                break;
        }

        if (j == n1)
            return 0;
    }
    return 1;
}

int main()
{
    int arr1[MAX], arr2[MAX];
    int n1, n2, i;

    printf("Enter number of elements in first set: ");
    scanf("%d",&n1);
    printf("Enter elements of first set: ");
    for(i = 0; i < n1; i++)
        scanf("%d",&arr1[i]);

    printf("Enter number of elements in second set: ");
    scanf("%d",&n2);
    printf("Enter elements of second set: ");
    for(i = 0; i < n2; i++)
        scanf("%d",&arr2[i]);

    if(isSubset(arr1, n1, arr2, n2))
        printf("Second set is a subset of the first set\n");
    else
        printf("Second set is not a subset of the first set\n");

    return 0;
}
```
This code includes a function isSubset() to check if one array is a subset of another. It loops over each element of the second array (arr2) and compares it against all elements of the first array (arr1), using a nested for loop. If a match is found, it moves to the next element in arr2. If it is unable to find a match for any element in arr2 within arr1, it returns 0, indicating that arr2 is not a subset of arr1. If all elements in arr2 are found within arr1, it returns 1, indicating that arr2 is indeed a subset of arr1. 

The main function prompts the user to enter the elements of the two arrays and uses the isSubset() function to determine whether the second array is a subset of the first. The result is printed to the console.
