```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <limits.h>
#include <regex.h>

int extract_max_number(const char *input, long *result)
{
    regex_t regex;
    regmatch_t match;
    const char *cursor;
    char buffer[64];
    long max_value = LONG_MIN;
    int found = 0;
    int rc;

    if (input == NULL || result == NULL) {
        return -1;
    }

    rc = regcomp(&regex, "-?[0-9]+", REG_EXTENDED);
    if (rc != 0) {
        return -1;
    }

    cursor = input;
    while (regexec(&regex, cursor, 1, &match, 0) == 0) {
        size_t len = (size_t)(match.rm_eo - match.rm_so);
        long value;
        char *endptr;

        if (len >= sizeof(buffer)) {
            regfree(&regex);
            return -1;
        }

        memcpy(buffer, cursor + match.rm_so, len);
        buffer[len] = '\0';

        errno = 0;
        value = strtol(buffer, &endptr, 10);
        if (errno == ERANGE || endptr == buffer || *endptr != '\0') {
            regfree(&regex);
            return -1;
        }

        if (!found || value > max_value) {
            max_value = value;
            found = 1;
        }

        cursor += match.rm_eo;
        if (*cursor == '\0') {
            break;
        }
    }

    regfree(&regex);

    if (!found) {
        return -1;
    }

    *result = max_value;
    return 0;
}

int main(void)
{
    const char *test1 = "abc 42 def -17 ghi 305 jkl 7";
    const char *test2 = "no numbers here";
    long max;

    if (extract_max_number(test1, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test1, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test1);
    }

    if (extract_max_number(test2, &max) == 0) {
        printf("Max value in \"%s\" is %ld\n", test2, max);
    } else {
        fprintf(stderr, "Failed to extract number from \"%s\"\n", test2);
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

1. **Variable scope (`value`, `len`, `endptr`)** — These variables were declared at function scope but only used inside the `while` loop. They are now declared inside the loop body, which limits their lifetime, improves readability, and eliminates the cppcheck `[variableScope]` warnings.

2. **Statically-sized buffer / `memcpy` (CWE-119/CWE-120)** — The Flawfinder warnings are effectively false positives here, but the mitigation is retained and made explicit: before copying, `len >= sizeof(buffer)` is checked, guaranteeing room for both the matched substring and the null terminator. The `memcpy` can therefore never overflow `buffer`. Any match longer than 63 characters (which would overflow `long` anyway) is rejected safely.

3. **Existing robustness preserved** — The code already handles the important correctness concerns: `NULL` argument checks, `regcomp` failure handling, `strtol` overflow detection via `errno == ERANGE`, full-token validation via `endptr`, `regfree` on every exit path (no resource leak), and loop termination since the pattern cannot match an empty string (`rm_eo > 0` always advances the cursor).
