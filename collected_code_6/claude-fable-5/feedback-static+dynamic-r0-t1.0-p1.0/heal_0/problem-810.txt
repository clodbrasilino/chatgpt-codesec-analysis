#include <stdio.h>
#include <stdlib.h>

struct element {
    int value;
    size_t count;
};

int repeat_elements(const struct element *elements, size_t num_elements,
                    int **out_result, size_t *out_size)
{
    size_t total = 0;
    size_t i;
    size_t j;
    size_t pos = 0;
    int *result = NULL;

    if (elements == NULL || out_result == NULL || out_size == NULL) {
        return -1;
    }

    for (i = 0; i < num_elements; i++) {
        if (elements[i].count > SIZE_MAX - total) {
            return -1;
        }
        total += elements[i].count;
    }

    if (total == 0) {
        *out_result = NULL;
        *out_size = 0;
        return 0;
    }

    if (total > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    result = malloc(total * sizeof(int));
    if (result == NULL) {
        return -1;
    }

    for (i = 0; i < num_elements; i++) {
        for (j = 0; j < elements[i].count; j++) {
            result[pos] = elements[i].value;
            pos++;
        }
    }

    *out_result = result;
    *out_size = total;
    return 0;
}

int main(void)
{
    struct element elements[] = {
        { 4, 3 },
        { 7, 1 },
        { 2, 4 },
        { 9, 2 }
    };
    size_t num_elements = sizeof(elements) / sizeof(elements[0]);
    int *result = NULL;
    size_t size = 0;
    size_t i;

    if (repeat_elements(elements, num_elements, &result, &size) != 0) {
        fprintf(stderr, "Failed to expand elements\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < size; i++) {
        if (printf("%d ", result[i]) < 0) {
            free(result);
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        free(result);
        return EXIT_FAILURE;
    }

    free(result);
    return EXIT_SUCCESS;
}
