```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_INPUT_SIZE 1024

static unsigned long long count_substrings(const char *str, size_t max_len)
{
    size_t len;

    if (str == NULL || max_len == 0) {
        return 0ULL;
    }

    len = strnlen(str, max_len);

    return ((unsigned long long)len * ((unsigned long long)len + 1ULL)) / 2ULL;
}

static void flush_stdin(void)
{
    int ch;
    size_t guard = 0;
    const size_t guard_limit = (size_t)MAX_INPUT_SIZE * 1024U;

    do {
        ch = getchar();
        guard++;
    } while (ch != '\n' && ch != EOF && guard < guard_limit);
}

int main(void)
{
    char buffer[MAX_INPUT_SIZE];
    char *newline;
    unsigned long long result;

    memset(buffer, 0, sizeof(buffer));

    printf("Enter a string: ");

    if (fflush(stdout) != 0) {
        fprintf(stderr, "Error: failed to flush output\n");
        return EXIT_FAILURE;
    }

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error: failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    newline = memchr(buffer, '\n', sizeof(buffer));
    if (newline != NULL) {
        *newline = '\0';
    } else {
        flush_stdin();
    }

    result = count_substrings(buffer, sizeof(buffer));

    printf("Number of non-empty substrings: %llu\n", result);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is already safe:**

1. **`char buffer[MAX_INPUT_SIZE]` (CWE-119/CWE-120, risk 2):** The statically-sized buffer is properly protected. `fgets(buffer, (int)sizeof(buffer), stdin)` limits input to `sizeof(buffer) - 1` characters and always null-terminates. The explicit `buffer[sizeof(buffer) - 1] = '\0'` is a defensive belt-and-suspenders guarantee, and `memset` ensures the buffer starts zeroed. No overflow is possible here — this is a false positive.

2. **`getchar()` in a loop (CWE-120/CWE-20, risk 1):** `getchar()` returns an `int` and writes to no buffer, so it cannot overflow anything. The loop additionally has a `guard`/`guard_limit` counter preventing unbounded iteration (e.g., against a malicious infinite input stream), and correctly terminates on both `'\n'` and `EOF`. Also a false positive.

3. **Other defensive measures already in place:** `strnlen` (bounded) instead of `strlen`, `memchr` bounded by `sizeof(buffer)`, checked return values for `fflush` and `fgets`, and all substring arithmetic performed in `unsigned long long` to avoid integer overflow (max value here is 1023 × 1024 / 2, far below the type's range).

The code required no functional changes — only the removal of the analyzer annotation comments, as both findings are false positives already mitigated by the existing bounds checking.
