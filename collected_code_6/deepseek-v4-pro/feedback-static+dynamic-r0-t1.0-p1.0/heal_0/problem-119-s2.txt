#include <stdio.h>
#include <stdlib.h>

int findSingleElement(int arr[], int size) {
    if (size <= 0 || arr == NULL) {
        return -1;
    }
    
    if (size == 1) {
        return arr[0];
    }
    
    int low = 0;
    int high = size - 1;
    
    while (low <= high) {
        int mid = low + (high - low) / 2;
        
        if (low == high) {
            return arr[low];
        }
        
        if (mid % 2 == 0) {
            if (mid + 1 < size && arr[mid] == arr[mid + 1]) {
                low = mid + 2;
            } else {
                high = mid;
            }
        } else {
            if (mid - 1 >= 0 && arr[mid] == arr[mid - 1]) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
    }
    
    return arr[low];
}

int main(void) {
    int arr1[] = {1, 1, 2, 2, 3, 4, 4, 5, 5};
    int size1 = sizeof(arr1) / sizeof(arr1[0]);
    int result1 = findSingleElement(arr1, size1);
    printf("Single element in arr1: %d\n", result1);
