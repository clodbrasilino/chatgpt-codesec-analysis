#include <stdio.h>
#include <stdlib.h>

int calculate_loss(double cost_price, double selling_price, double *loss_amount)
{
    if (loss_amount == NULL) {
        return -1;
    }
    if (cost_price < 0.0 || selling_price < 0.0) {
        return -1;
    }
    if (selling_price < cost_price) {
        *loss_amount = cost_price - selling_price;
        return 1;
    }
    *loss_amount = 0.0;
    return 0;
}

int main(void)
{
    double cost_price = 0.0;
    double selling_price = 0.0;
    double loss = 0.0;
    int result = 0;

    printf("Enter cost price: ");
    if (scanf("%lf", &cost_price) != 1) {
        fprintf(stderr, "Invalid input for cost price\n");
        return EXIT_FAILURE;
    }

    printf("Enter selling price: ");
    if (scanf("%lf", &selling_price) != 1) {
        fprintf(stderr, "Invalid input for selling price\n");
        return EXIT_FAILURE;
    }

    result = calculate_loss(cost_price, selling_price, &loss);

    if (result < 0) {
        fprintf(stderr, "Invalid values provided\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("Loss amount: %.2f\n", loss);
    } else {
        printf("None\n");
    }

    return EXIT_SUCCESS;
}
