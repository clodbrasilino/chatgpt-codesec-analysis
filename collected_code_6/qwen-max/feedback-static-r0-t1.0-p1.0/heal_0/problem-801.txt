```c
#include <stdio.h>

int countEqualNumbers(int a, int b, int c) {
    int count = 0;
    if (a == b) count++;
    if (b == c) count++;
    if (a == c && a != b) count++;
    return count;
}

int main() {
    int x = 5, y = 5, z = 10;
    int result = countEqualNumbers(x, y, z);
    printf("Number of equal numbers: %d\n", result);
    return 0;
}
```
