```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LENGTH 4096U

typedef struct {
    char *items;
    size_t count;
} Tuple;

static size_t bounded_length(const char *str, size_t max_length)
{
    size_t len = 0U;

    while ((len < max_length) && (str[len] != '\0')) {
        len++;
    }

    return len;
}

Tuple *string_to_tuple(const char *str)
{
    Tuple *tuple = NULL;
    size_t length;
    size_t i;

    if ((str == NULL) || (str[0] == '\0')) {
        return NULL;
    }

    length = bounded_length(str, MAX_INPUT_LENGTH);
    if (length == 0U) {
        return NULL;
    }

    tuple = (Tuple *)malloc(sizeof(Tuple));
    if (tuple == NULL) {
        return NULL;
    }

    tuple->items = (char *)malloc((length + 1U) * sizeof(char));
    if (tuple->items == NULL) {
        free(tuple);
        return NULL;
    }

    for (i = 0U; i < length; i++) {
        tuple->items[i] = str[i];
    }
    tuple->items[length] = '\0';
    tuple->count = length;

    return tuple;
}

void free_tuple(Tuple *tuple)
{
    if (tuple != NULL) {
        free(tuple->items);
        tuple->items = NULL;
        tuple->count = 0U;
        free(tuple);
    }
}

void print_tuple(const Tuple *tuple)
{
    size_t i;

    if ((tuple == NULL) || (tuple->items == NULL) || (tuple->count == 0U)) {
        return;
    }

    (void)printf("[");
    for (i = 0U; i < tuple->count; i++) {
        (void)printf("'%c'", tuple->items[i]);
        if (i < (tuple->count - 1U)) {
            (void)printf(", ");
        }
    }
    (void)printf("]\n");
    (void)fflush(stdout);
}

static void strip_newline(char *str)
{
    size_t len;

    if (str == NULL) {
        return;
    }

    len = bounded_length(str, MAX_INPUT_LENGTH);
    while ((len > 0U) && ((str[len - 1U] == '\n') || (str[len - 1U] == '\r'))) {
        str[len - 1U] = '\0';
        len--;
    }
}

int main(void)
{
    char buffer[MAX_INPUT_LENGTH];

    (void)setvbuf(stdout, NULL, _IONBF, 0);

    while (fgets(buffer, (int)sizeof(buffer), stdin) != NULL) {
        Tuple *tuple = NULL;

        buffer[sizeof(buffer) - 1U] = '\0';
        strip_newline(buffer);

        if (buffer[0] == '\0') {
            continue;
        }

        tuple = string_to_tuple(buffer);
        if (tuple == NULL) {
            (void)fprintf(stderr, "Failed to convert string to tuple\n");
            return EXIT_FAILURE;
        }

        print_tuple(tuple);
        free_tuple(tuple);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original logic for `string_to_tuple`, `print_tuple`, and memory management was actually correct — the `<no output>` test failures were almost certainly caused by output buffering when `stdout` is connected to a pipe (as test harnesses typically do). In that mode, `stdout` is fully buffered rather than line buffered, so if the process is inspected before the buffer flushes, the harness sees no output.

Fixes applied:

1. **`setvbuf(stdout, NULL, _IONBF, 0)`** — disables output buffering so every `printf` reaches the test harness immediately. This directly addresses the three failing test cases.
2. **Explicit NUL termination of `buffer`** after `fgets` — `fgets` already guarantees termination, but forcing `buffer[sizeof(buffer) - 1] = '\0'` provides defense-in-depth against the Flawfinder CWE-119/CWE-120 warning about statically-sized arrays, guaranteeing `bounded_length` and `strip_newline` can never read past the array.
3. **Retained the existing bounds discipline** — `bounded_length` caps all scans at `MAX_INPUT_LENGTH`, `fgets` is limited by `sizeof(buffer)`, and all allocations check for `NULL` with proper cleanup on the partial-allocation failure path, so there are no leaks or overflows.
