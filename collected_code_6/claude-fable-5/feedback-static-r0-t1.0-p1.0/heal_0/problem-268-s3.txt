#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

static long long star_number(unsigned int n)
{
    unsigned long long un = (unsigned long long)n;

    if (n == 0U) {
        return -1LL;
    }

    if (un > 1000000000ULL) {
        return -1LL;
    }

    unsigned long long result = 6ULL * un * (un - 1ULL) + 1ULL;

    if (result > (unsigned long long)LLONG_MAX) {
        return -1LL;
    }

    return (long long)result;
}

int main(void)
{
    unsigned int n = 0U;
    int status = EXIT_SUCCESS;

    printf("Enter n: ");

    if (scanf("%u", &n) != 1) {
        fprintf(stderr, "Error: invalid input\n");
        status = EXIT_FAILURE;
    } else {
        long long result = star_number(n);

        if (result < 0LL) {
            fprintf(stderr, "Error: n out of valid range\n");
            status = EXIT_FAILURE;
        } else {
            printf("The %u'th star number is: %lld\n", n, result);
        }
    }

    return status;
}
