#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>
#include <stdint.h>

char *camel_to_snake(const char *input) {
    if (input == NULL) {
        return NULL;
    }

    regex_t regex;
    const char *pattern = "([a-z0-9])([A-Z])";
    
    if (regcomp(&regex, pattern, REG_EXTENDED) != 0) {
        return NULL;
    }

    size_t input_len = strlen(input);
    char *result = (char *)malloc(input_len * 2 + 1);
    if (result == NULL) {
        regfree(&regex);
        return NULL;
    }

    size_t src_pos = 0;
    size_t dst_pos = 0;
    const char *remaining = input;
    regmatch_t matches[3];

    while (src_pos < input_len && regexec(&regex, remaining, 3, matches, 0) == 0) {
        size_t match_start = matches[0].rm_so;
        size_t match_end = matches[0].rm_eo;
        size_t group1_start = matches[1].rm_so;
        size_t group1_end = matches[1].rm_eo;
        size_t group2_start = matches[2].rm_so;
        size_t group2_end = matches[2].rm_eo;

        if (dst_pos + match_start > input_len * 2) {
            break;
        }
        memcpy(result + dst_pos, remaining, match_start);
        dst_pos += match_start;

        if (group1_start != (size_t)-1 && group1_end != (size_t)-1) {
            size_t group1_len = group1_end - group1_start;
            if (dst_pos + group1_len > input_len * 2) {
                break;
            }
            memcpy(result + dst_pos, remaining + group1_start, group1_len);
            dst_pos += group1_len;
        }

        if (dst_pos >= input_len * 2) {
            break;
        }
        result[dst_pos++] = '_';

        if (group2_start != (size_t)-1 && group2_end != (size_t)-1) {
            if (dst_pos >= input_len * 2) {
                break;
            }
            char lower_char = remaining[group2_start];
            if (lower_char >= 'A' && lower_char <= 'Z') {
                lower_char = lower_char + ('a' - 'A');
            }
            result[dst_pos++] = lower_char;
        }

        src_pos += match_end;
        remaining += match_end;
    }

    if (src_pos < input_len) {
        size_t remaining_len = input_len - src_pos;
        if (dst_pos + remaining_len <= input_len * 2) {
            memcpy(result + dst_pos, remaining, remaining_len);
            dst_pos += remaining_len;
        }
    }

    if (dst_pos <= input_len * 2) {
        result[dst_pos] = '\0';
    } else {
        result[input_len * 2] = '\0';
    }

    regfree(&regex);
    return result;
}

int main(void) {
    const char *test_cases[] = {
        "googleAssistant",
        "chromeCast",
        "quadCore",
        "camelCaseString",
        "simple",
        "already_snake_case",
        "with123Numbers",
        "HTTPResponseCode",
        "a",
        "",
        "ABC",
        NULL
    };

    for (int i = 0; test_cases[i] != NULL; i++) {
        char *converted = camel_to_snake(test_cases[i]);
        if (converted != NULL) {
            printf("%s\n", converted);
            free(converted);
        }
    }

    return 0;
}
