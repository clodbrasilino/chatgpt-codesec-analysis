```c
#include <stdio.h>
#include <stdlib.h>

static double largest_triangle_area(double radius)
{
    if (radius < 0.0) {
        return -1.0;
    }
    return radius * radius;
}

int main(void)
{
    double radius = 0.0;
    double area = 0.0;
    int result = 0;

    printf("Enter the radius of the semicircle: ");
    result = scanf("%lf", &radius);
    if (result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    area = largest_triangle_area(radius);
    if (area < 0.0) {
        fprintf(stderr, "Error: radius must be non-negative.\n");
        return EXIT_FAILURE;
    }

    printf("The area of the largest inscribed triangle is: %.6f\n", area);
    return EXIT_SUCCESS;
}
```

The largest triangle inscribed in a semicircle of radius `r` has its base along the diameter (length `2r`) and its apex at the top of the arc (height `r`), giving an area of `(1/2) * 2r * r = r²`.
