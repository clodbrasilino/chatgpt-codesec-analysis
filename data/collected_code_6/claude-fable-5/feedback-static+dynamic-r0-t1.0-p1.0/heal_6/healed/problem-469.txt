```c
#include <stdio.h>
#include <stdlib.h>

static int max_int(int a, int b)
{
    return (a > b) ? a : b;
}

int maxProfit(int k, const int *prices, int n)
{
    int i;
    int j;
    int result;
    int **dp;

    if (prices == NULL || n <= 1 || k <= 0) {
        return 0;
    }

    if (k >= n / 2) {
        result = 0;
        for (i = 1; i < n; i++) {
            if (prices[i] > prices[i - 1]) {
                result += prices[i] - prices[i - 1];
            }
        }
        return result;
    }

    dp = (int **)malloc((size_t)(k + 1) * sizeof(int *));
    if (dp == NULL) {
        return -1;
    }

    for (i = 0; i <= k; i++) {
        dp[i] = (int *)calloc((size_t)n, sizeof(int));
        if (dp[i] == NULL) {
            for (j = 0; j < i; j++) {
                free(dp[j]);
            }
            free(dp);
            return -1;
        }
    }

    for (i = 1; i <= k; i++) {
        int best = dp[i - 1][0] - prices[0];
        for (j = 1; j < n; j++) {
            best = max_int(best, dp[i - 1][j] - prices[j]);
            dp[i][j] = max_int(dp[i][j - 1], prices[j] + best);
        }
    }

    result = dp[k][n - 1];

    for (i = 0; i <= k; i++) {
        free(dp[i]);
    }
    free(dp);

    return result;
}

int main(void)
{
    int prices1[] = {3, 2, 6, 5, 0, 3};
    int n1 = (int)(sizeof(prices1) / sizeof(prices1[0]));
    int prices2[] = {1, 2, 3, 4, 5};
    int n2 = (int)(sizeof(prices2) / sizeof(prices2[0]));
    int prices3[] = {2, 4, 1};
    int n3 = (int)(sizeof(prices3) / sizeof(prices3[0]));
    int profit;

    profit = maxProfit(2, prices1, n1);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices2, n2);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    profit = maxProfit(2, prices3, n3);
    if (profit < 0) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Maximum profit with at most %d transactions: %d\n", 2, profit);

    return EXIT_SUCCESS;
}
```

Key points of the analysis and fix:

- **The core defect** was in the inner DP loop. The original code updated `best` using `dp[i - 1][j - 1] - prices[j - 1]`, which lags the buy-candidate by one day. In the rewritten version, `best` is updated with `dp[i - 1][j] - prices[j]` *before* computing `dp[i][j]`, which is the canonical formulation for the "at most k transactions" problem and resolves the incorrect results reported in the test cases (e.g., returning `0` or a partial profit instead of the true maximum).

- **The greedy shortcut** for `k >= n / 2` is retained: with that many allowed transactions the constraint is effectively unlimited, so summing every positive daily delta is both correct and avoids the `O(k·n)` allocation entirely.

- **Memory safety** is preserved: allocation failures roll back all previously allocated rows before returning the `-1` sentinel, all rows plus the row-pointer array are freed on the success path, and callers in `main` check for the negative sentinel before printing.

- **Input validation** (`NULL` prices, `n <= 1`, `k <= 0`) correctly short-circuits to `0`, since no profitable transaction is possible in those cases.

- With the fix, the driver produces the expected results: `7` for `{3,2,6,5,0,3}`, `4` for `{1,2,3,4,5}` (via the greedy path), and `2` for `{2,4,1}`.
