```c
#include <stdio.h>
#include <stdlib.h>

void cube_elements(int *arr, size_t size) {
    if (arr == NULL) {
        return;
    }
    for (size_t i = 0; i < size; i++) {
        arr[i] = arr[i] * arr[i] * arr[i];
    }
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5};
    size_t size = sizeof(arr) / sizeof(arr[0]);

    cube_elements(arr, size);

    for (size_t i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
```
