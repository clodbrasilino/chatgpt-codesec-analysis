```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 16
#define MIN_ROMAN_VALUE 1
#define MAX_ROMAN_VALUE 3999

int int_to_roman(int number, char *buffer, size_t buffer_size);

int int_to_roman(int number, char *buffer, size_t buffer_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (buffer == NULL || buffer_size == 0) {
        return -1;
    }

    if (number < MIN_ROMAN_VALUE || number > MAX_ROMAN_VALUE) {
        return -1;
    }

    buffer[0] = '\0';

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t sym_len = symbol_lengths[i];

            if (sym_len >= buffer_size || used > buffer_size - sym_len - 1) {
                buffer[0] = '\0';
                return -1;
            }

            memcpy(buffer + used, symbols[i], sym_len);
            used += sym_len;
            buffer[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char roman[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t i;
    int input;

    memset(roman, 0, sizeof(roman));

    for (i = 0; i < sizeof(test_values) / sizeof(test_values[0]); i++) {
        if (int_to_roman(test_values[i], roman, sizeof(roman)) == 0) {
            printf("%d -> %s\n", test_values[i], roman);
        } else {
            fprintf(stderr, "Conversion failed for %d\n", test_values[i]);
            return EXIT_FAILURE;
        }
    }

    printf("Enter an integer (1-3999): ");
    if (scanf("%d", &input) != 1) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }

    if (int_to_roman(input, roman, sizeof(roman)) != 0) {
        fprintf(stderr, "Value out of range (1-3999)\n");
        return EXIT_FAILURE;
    }

    printf("%d -> %s\n", input, roman);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses:**

1. **`memcpy` (CWE-120):** This is a false positive in practice. The guard condition `sym_len >= buffer_size || used > buffer_size - sym_len - 1` executes before every copy and mathematically guarantees that `used + sym_len + 1 <= buffer_size`, so the copy plus the null terminator always fits. The subtraction `buffer_size - sym_len - 1` cannot underflow because the first clause of the `||` already rejects the case `sym_len >= buffer_size`. The check is performed *inside* the loop on every iteration, so incremental growth of `used` is always validated.

2. **Statically-sized array `roman` (CWE-119/CWE-120):** Also safe. `ROMAN_BUFFER_SIZE` is 16, and the longest possible Roman numeral in range (3888 → `MMMDCCCLXXXVIII`) is 15 characters plus the null terminator — exactly 16 bytes. Every call passes `sizeof(roman)` rather than a hardcoded constant, so the callee's bounds checking always receives the true capacity. The `memset` before use ensures the buffer is initialized even on error paths.

3. **Input handling:** `scanf("%d", ...)` return value is checked, and range validation is delegated to `int_to_roman`, which rejects values outside 1–3999. On overflow of `int` from user input the behavior of `scanf` is technically undefined per the standard; for hardened production code you could switch to `fgets` + `strtol` with `errno`/`ERANGE` checking, but the flagged findings themselves are already mitigated by the existing bounds checks.

The code compiles cleanly under `-Wall -Wextra -pedantic` and both Flawfinder hits are mitigated by the explicit size checks; no functional changes were required.
