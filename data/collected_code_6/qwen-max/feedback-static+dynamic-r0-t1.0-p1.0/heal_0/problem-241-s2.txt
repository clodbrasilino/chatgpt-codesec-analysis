#include <stdio.h>
#include <stdlib.h>

char ***create_3d_array(int depth, int rows, int cols) {
    char ***array = (char ***)malloc(depth * sizeof(char **));
    if (array == NULL) {
        return NULL;
    }
    for (int i = 0; i < depth; i++) {
        array[i] = (char **)malloc(rows * sizeof(char *));
        if (array[i] == NULL) {
            for (int j = 0; j < i; j++) {
                free(array[j]);
            }
            free(array);
            return NULL;
        }
        for (int j = 0; j < rows; j++) {
            array[i][j] = (char *)malloc(cols * sizeof(char));
            if (array[i][j] == NULL) {
                for (int k = 0; k < j; k++) {
                    free(array[i][k]);
                }
                for (int l = 0; l < i; l++) {
                    free(array[l]);
                }
                free(array);
                return NULL;
            }
            for (int k = 0; k < cols; k++) {
                array[i][j][k] = '*';
            }
        }
    }
    return array;
}

void free_3d_array(char ***array, int depth, int rows) {
    for (int i = 0; i < depth; i++) {
        for (int j = 0; j < rows; j++) {
            free(array[i][j]);
        }
        free(array[i]);
    }
    free(array);
}

int main() {
    int depth = 2, rows = 3, cols = 4;
    char ***array = create_3d_array(depth, rows, cols);

    if (array != NULL) {
        for (int i = 0; i < depth; i++) {
            for (int j = 0; j < rows; j++) {
                for (int k = 0; k < cols; k++) {
                    printf("%c ", array[i][j][k]);
                }
                printf("\n");
            }
            printf("\n");
        }

        free_3d_array(array, depth, rows);
    } else {
        printf("Memory allocation failed.\n");
    }

    return 0;
}
