#include <stdio.h>
#include <stdlib.h>

char* intToRoman(int num) {
    if (num <= 0 || num > 3999) {
        return NULL;
    }

    const int values[] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    const char* const symbols[] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    
    char* roman = (char*)malloc(16 * sizeof(char));
    if (!roman) {
        return NULL;
    }

    int idx = 0;
    for (int i = 0; i < 13; i++) {
        while (num >= values[i]) {
            num -= values[i];
            const char* sym = symbols[i];
            int j = 0;
            while (sym[j] != '\0' && idx < 15) {
                roman[idx++] = sym[j++];
            }
        }
    }
    roman[idx] = '\0';

    return roman;
}

int main(void) {
    int num;
    while (scanf("%d", &num) == 1) {
        char* roman = intToRoman(num);
        if (roman) {
            printf("%s\n", roman);
            free(roman);
        }
    }
    return 0;
}
