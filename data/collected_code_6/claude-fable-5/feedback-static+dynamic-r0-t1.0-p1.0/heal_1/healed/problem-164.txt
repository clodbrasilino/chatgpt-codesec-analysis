#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long sum_of_divisors(long n)
{
    long total = 0;
    long i;

    if (n <= 0)
    {
        return 0;
    }

    for (i = 1; i <= n / i; i++)
    {
        if (n % i == 0)
        {
            if (total > LONG_MAX - i)
            {
                return -1;
            }
            total += i;

            if (i != n / i)
            {
                long other = n / i;
                if (total > LONG_MAX - other)
                {
                    return -1;
                }
                total += other;
            }
        }
    }

    return total;
}

int are_divisor_sums_equal(long a, long b)
{
    long sum_a = sum_of_divisors(a);
    long sum_b = sum_of_divisors(b);

    if (sum_a < 0 || sum_b < 0)
    {
        return -1;
    }

    return sum_a == sum_b;
}

int main(void)
{
    long num1 = 0;
    long num2 = 0;
    int result;

    printf("Enter first number: ");
    if (scanf("%ld", &num1) != 1)
    {
        fprintf(stderr, "Invalid input for first number.\n");
        return EXIT_FAILURE;
    }

    printf("Enter second number: ");
    if (scanf("%ld", &num2) != 1)
    {
        fprintf(stderr, "Invalid input for second number.\n");
        return EXIT_FAILURE;
    }

    if (num1 <= 0 || num2 <= 0)
    {
        fprintf(stderr, "Numbers must be positive.\n");
        return EXIT_FAILURE;
    }

    result = are_divisor_sums_equal(num1, num2);

    if (result < 0)
    {
        fprintf(stderr, "Computation overflow: numbers too large.\n");
        return EXIT_FAILURE;
    }

    if (result)
    {
        printf("The sum of divisors of %ld and %ld are the same.\n", num1, num2);
    }
    else
    {
        printf("The sum of divisors of %ld and %ld are not the same.\n", num1, num2);
    }

    return EXIT_SUCCESS;
}
