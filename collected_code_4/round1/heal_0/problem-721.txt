#include <stdio.h>
#include <stdlib.h>
#include <limits.h>
#include <float.h>

double maxAvgRec(int matrix[][100], int n, int i, int j, double sum, int len){

    if(i == n-1 && j == n-1){
        return (sum + matrix[i][j]) / (len + 1);
    }
    if(i < 0 || j < 0 || i >= n || j >= n){
        return -DBL_MAX;
    }
    else {
            double down = maxAvgRec(matrix, n, i+1, j, sum + matrix[i][j], len+1);
            double right = maxAvgRec(matrix, n, i, j+1, sum + matrix[i][j], len+1);
            return (down > right)?down:right;
    }
}

void findMaxAverage(int matrix[][100], int n){

    printf("\n Maximum average is %lf ", maxAvgRec(matrix, n, 0, 0, 0, 0));
} 

int main(){
    int n;
    printf("Enter the value of n: ");
    scanf("%d", &n);
    int matrix[100][100];
    printf("Enter matrix elements:\n");
    for(int i = 0; i < n; i++){
        for(int j = 0; j < n; j++){
            scanf("%d", &matrix[i][j]);
        }
    }   
    findMaxAverage(matrix,n);
    return 0;
}
