```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096U

char *move_numbers_to_end(const char *input)
{
    size_t len;
    size_t i;
    size_t pos;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1U);
    if (result == NULL) {
        return NULL;
    }

    pos = 0U;
    for (i = 0U; i < len; i++) {
        if (isdigit((unsigned char)input[i]) == 0) {
            result[pos] = input[i];
            pos++;
        }
    }

    for (i = 0U; i < len; i++) {
        if (isdigit((unsigned char)input[i]) != 0) {
            result[pos] = input[i];
            pos++;
        }
    }

    result[pos] = '\0';
    return result;
}

int main(void)
{
    const char *test = "a1b2c3d4";
    char *moved;

    moved = move_numbers_to_end(test);
    if (moved == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("Original: %s\n", test) < 0) {
        free(moved);
        return EXIT_FAILURE;
    }

    if (printf("Result:   %s\n", moved) < 0) {
        free(moved);
        return EXIT_FAILURE;
    }

    free(moved);
    moved = NULL;

    return EXIT_SUCCESS;
}
```

The key change addresses the Flawfinder finding (CWE-126): `strlen` was replaced with `strnlen` bounded by `MAX_INPUT_LEN`, ensuring that a non-null-terminated string cannot cause an unbounded over-read. If the input reaches the maximum bound (meaning no terminator was found within the limit), the function safely rejects it by returning `NULL`. All other defensive measures from the original — null-pointer checks, `malloc` failure handling, casting to `unsigned char` for `isdigit`, checking `printf` return values, and freeing/nulling the pointer — are preserved.
