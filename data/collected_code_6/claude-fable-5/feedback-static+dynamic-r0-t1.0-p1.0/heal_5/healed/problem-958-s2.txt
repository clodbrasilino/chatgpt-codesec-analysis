#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ROMAN_BUFFER_SIZE 32
#define ROMAN_MIN 1
#define ROMAN_MAX 3999

int int_to_roman(int number, char *output, size_t output_size);

int int_to_roman(int number, char *output, size_t output_size)
{
    static const int values[] = {
        1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
    };
    static const char symbols[][3] = {
        "M", "CM", "D", "CD", "C", "XC", "L", "XL",
        "X", "IX", "V", "IV", "I"
    };
    static const size_t symbol_lengths[] = {
        1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1, 2, 1
    };
    size_t used = 0;
    size_t i;

    if (output == NULL || output_size == 0) {
        return -1;
    }

    output[0] = '\0';

    if (number < ROMAN_MIN || number > ROMAN_MAX) {
        return -1;
    }

    for (i = 0; i < sizeof(values) / sizeof(values[0]); i++) {
        while (number >= values[i]) {
            size_t symbol_len = symbol_lengths[i];
            size_t j;

            if (used >= output_size || symbol_len >= output_size - used) {
                output[0] = '\0';
                return -1;
            }

            for (j = 0; j < symbol_len; j++) {
                output[used + j] = symbols[i][j];
            }
            used += symbol_len;

            if (used >= output_size) {
                output[0] = '\0';
                return -1;
            }

            output[used] = '\0';
            number -= values[i];
        }
    }

    return 0;
}

int main(void)
{
    char buffer[ROMAN_BUFFER_SIZE];
    int test_values[] = { 1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999 };
    size_t count = sizeof(test_values) / sizeof(test_values[0]);
    size_t i;

    memset(buffer, 0, sizeof(buffer));

    for (i = 0; i < count; i++) {
        if (int_to_roman(test_values[i], buffer, sizeof(buffer)) == 0) {
            if (printf("%d -> %s\n", test_values[i], buffer) < 0) {
                return EXIT_FAILURE;
            }
        } else {
            if (fprintf(stderr, "Conversion failed for %d\n",
                        test_values[i]) < 0) {
                return EXIT_FAILURE;
            }
        }
    }

    if (int_to_roman(0, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 0\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    if (int_to_roman(4000, buffer, sizeof(buffer)) != 0) {
        if (printf("Correctly rejected out-of-range value 4000\n") < 0) {
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}

Analysis of the flagged weakness (CWE-119/CWE-120, statically-sized `buffer`):

The Flawfinder hit on `char buffer[ROMAN_BUFFER_SIZE]` is a low-risk informational warning — statically-sized arrays are only dangerous when writes to them are unbounded. In this code the risk is already fully mitigated, so the rewrite preserves the original logic and simply removes the comments:

1. **Size is always passed explicitly**: every call site uses `sizeof(buffer)`, so `int_to_roman` never has to guess the capacity.
2. **All writes are bounds-checked**: before copying each symbol, the function verifies `used < output_size` and `symbol_len < output_size - used` (written to avoid unsigned underflow), and re-checks before writing the NUL terminator. On any potential overflow it truncates to an empty string and returns `-1`.
3. **Output is always NUL-terminated**: `output[0] = '\0'` is set immediately after validating the pointer and size, and the terminator is maintained after every append, so the buffer can never be read as an unterminated string.
4. **Buffer is adequately sized**: the longest Roman numeral in range (3888 → "MMMDCCCLXXXVIII", 15 chars + NUL) fits comfortably in 32 bytes.
5. **Input range is validated**: values outside `[1, 3999]` are rejected before any writing occurs, and `memset` in `main` guarantees a defined initial state.

No unsafe functions (`strcpy`, `sprintf`, `gets`, etc.) are used; all copying is done via explicitly bounded loops. The code is therefore already safe against the flagged buffer-overflow class of weaknesses.
