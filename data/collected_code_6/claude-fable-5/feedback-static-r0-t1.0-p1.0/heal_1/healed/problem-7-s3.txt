```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

static void print_regex_error(int errcode, const regex_t *regex, const char *context)
{
    size_t needed;
    char *errbuf;

    needed = regerror(errcode, regex, NULL, 0);
    errbuf = malloc(needed);
    if (errbuf == NULL) {
        fprintf(stderr, "%s: unable to allocate error buffer\n", context);
        return;
    }

    regerror(errcode, regex, errbuf, needed);
    fprintf(stderr, "%s: %s\n", context, errbuf);
    free(errbuf);
}

int find_long_words(const char *text)
{
    regex_t regex;
    regmatch_t match;
    const char *pattern = "[[:alpha:]]{4,}";
    const char *cursor;
    int ret;

    if (text == NULL) {
        return -1;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED);
    if (ret != 0) {
        print_regex_error(ret, &regex, "Regex compilation failed");
        return -1;
    }

    cursor = text;
    while ((ret = regexec(&regex, cursor, 1, &match, 0)) == 0) {
        regoff_t start = match.rm_so;
        regoff_t end = match.rm_eo;
        regoff_t length = end - start;

        if (length <= 0) {
            break;
        }

        if (printf("%.*s\n", (int)length, cursor + start) < 0) {
            regfree(&regex);
            return -1;
        }

        cursor += end;
        if (*cursor == '\0') {
            break;
        }
    }

    if (ret != 0 && ret != REG_NOMATCH) {
        print_regex_error(ret, &regex, "Regex execution failed");
        regfree(&regex);
        return -1;
    }

    regfree(&regex);
    return 0;
}

int main(void)
{
    const char *text = "The quick brown fox jumps over a lazy dog near the riverbank";

    if (find_long_words(text) != 0) {
        fprintf(stderr, "Failed to process text\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The two Flawfinder warnings flagged the statically-sized `char errbuf[256]` arrays (CWE-119/CWE-120). In the original code these were actually *low-risk false positives*, because `regerror()` was correctly called with `sizeof(errbuf)`, which guarantees bounds-checked, null-terminated output — no overflow was possible.

That said, the rewrite eliminates the warning entirely and improves robustness by:

1. **Dynamic sizing**: Calling `regerror(errcode, regex, NULL, 0)` first to obtain the exact buffer size required, then allocating it with `malloc`. This avoids both the fixed-size array and any possibility of truncated error messages longer than 256 bytes.
2. **Allocation failure handling**: If `malloc` fails, a fallback message is printed rather than dereferencing a NULL pointer.
3. **Deduplication**: The repeated error-reporting logic was factored into a single `print_regex_error()` helper, reducing the surface area for future mistakes.

The rest of the logic was already sound: `regfree()` is called on all exit paths after successful compilation, `printf` return values are checked, the loop guards against zero-length matches (preventing infinite loops), and pointer advancement checks for end-of-string.
