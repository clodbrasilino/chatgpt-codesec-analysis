```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 64U

static void swap_chars(char *a, char *b)
{
    char tmp = *a;
    *a = *b;
    *b = tmp;
}

static void permute(char *str, size_t left, size_t right)
{
    size_t i;

    if (left == right) {
        if (puts(str) == EOF) {
            exit(EXIT_FAILURE);
        }
        return;
    }

    for (i = left; i <= right; i++) {
        swap_chars(&str[left], &str[i]);
        permute(str, left + 1U, right);
        swap_chars(&str[left], &str[i]);
    }
}

static int print_permutations(const char *input)
{
    char *buffer;
    size_t len;

    if (input == NULL) {
        return -1;
    }

    len = strnlen(input, MAX_INPUT_LEN + 1U);
    if (len == 0U || len > MAX_INPUT_LEN) {
        return -1;
    }

    buffer = (char *)malloc(len + 1U);
    if (buffer == NULL) {
        return -1;
    }

    memcpy(buffer, input, len);
    buffer[len] = '\0';

    permute(buffer, 0U, len - 1U);
    free(buffer);

    return 0;
}

int main(void)
{
    const char *test = "ABC";

    if (print_permutations(test) != 0) {
        (void)fprintf(stderr, "Failed to print permutations\n");
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (`strlen` over-read):** Replaced `strlen` with `strnlen`, bounded by `MAX_INPUT_LEN + 1U`. If the input is not NUL-terminated within that limit, the function rejects it (`len > MAX_INPUT_LEN`), so the read is always bounded and unterminated strings can no longer cause an over-read.

2. **CWE-120 (`memcpy` overflow):** The copy is now exactly `len` bytes into a buffer of `len + 1U` bytes, with the NUL terminator written explicitly (`buffer[len] = '\0'`). The destination size is guaranteed by the immediately preceding `malloc(len + 1U)` and its NULL check, so the destination can always hold the source data.

A bounded input length also has the practical benefit of capping the factorial explosion of the recursion (64! permutations would be infeasible anyway, but the check formalizes the limit).
