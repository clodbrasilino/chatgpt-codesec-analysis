#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <limits.h>

bool is_keith_number(long num) {
    long digits[50], sum = 0, count = 0;
    if (num <= 0) return false;
    long temp = num;
    while (temp > 0) {
        if (count >= 50) return false;
        digits[count++] = temp % 10;
        temp /= 10;
    }
    for (long i = count; i < num; i++) {
        for (long j = 1; j < count; j++) {
            sum += digits[j];
            if (sum > num || sum < 0) break;
        }
        if (sum == num) return true;
        for (long j = count - 1; j > 0; j--) {
            digits[j] = digits[j - 1];
        }
        digits[0] = sum;
        sum = 0;
    }
    return false;
}

int main() {
    long number;
    printf("Enter a number: ");
    if (scanf("%ld", &number) != 1) {
        printf("Invalid input.\n");
        return 1;
    }
    if (is_keith_number(number)) {
        printf("%ld is a Keith number.\n", number);
    } else {
        printf("%ld is not a Keith number.\n", number);
    }
    return 0;
}
