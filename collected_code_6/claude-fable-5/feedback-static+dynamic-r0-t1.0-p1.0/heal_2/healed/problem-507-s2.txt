```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_STRING_LENGTH 4096

size_t remove_words(char **list, size_t count, const char **words_to_remove, size_t remove_count);
char *duplicate_string(const char *src);
void free_list(char **list, size_t count);

size_t remove_words(char **list, size_t count, const char **words_to_remove, size_t remove_count)
{
    size_t write_index = 0;
    size_t i;
    size_t j;

    if (list == NULL || words_to_remove == NULL) {
        return count;
    }

    for (i = 0; i < count; i++) {
        int match = 0;
        if (list[i] != NULL) {
            for (j = 0; j < remove_count; j++) {
                if (words_to_remove[j] != NULL &&
                    strncmp(list[i], words_to_remove[j], MAX_STRING_LENGTH) == 0) {
                    match = 1;
                    break;
                }
            }
        }
        if (match) {
            free(list[i]);
            list[i] = NULL;
        } else {
            list[write_index] = list[i];
            if (write_index != i) {
                list[i] = NULL;
            }
            write_index++;
        }
    }

    return write_index;
}

char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strnlen(src, MAX_STRING_LENGTH);
    if (len >= MAX_STRING_LENGTH) {
        return NULL;
    }

    copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len] = '\0';
    return copy;
}

void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *initial_words[] = { "apple", "banana", "cherry", "date", "banana", "fig" };
    const char *words_to_remove[] = { "banana", "date" };
    size_t initial_count = sizeof(initial_words) / sizeof(initial_words[0]);
    size_t remove_count = sizeof(words_to_remove) / sizeof(words_to_remove[0]);
    char **list;
    size_t i;
    size_t new_count;

    list = malloc(initial_count * sizeof(char *));
    if (list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < initial_count; i++) {
        list[i] = duplicate_string(initial_words[i]);
        if (list[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_list(list, i);
            return EXIT_FAILURE;
        }
    }

    printf("Original list:\n");
    for (i = 0; i < initial_count; i++) {
        printf("  %s\n", list[i]);
    }

    new_count = remove_words(list, initial_count, words_to_remove, remove_count);

    printf("List after removal:\n");
    for (i = 0; i < new_count; i++) {
        printf("  %s\n", list[i]);
    }

    free_list(list, new_count);

    return EXIT_SUCCESS;
}
```

The key changes address the identified weaknesses:

1. **`strlen` → `strnlen` (CWE-126):** The unbounded `strlen` could over-read if given a non-terminated string. Using `strnlen` with a `MAX_STRING_LENGTH` cap bounds the read, and inputs at or beyond the limit are rejected. This also replaces the nonsensical `len == (size_t)-1` check, which could never meaningfully trigger since `strlen` returning `SIZE_MAX` would already imply an over-read occurred.

2. **`memcpy` (CWE-120):** The destination buffer is now explicitly allocated as `len + 1` and exactly `len` bytes are copied, followed by explicit null termination at `copy[len]`. The allocation size and copy size are derived from the same bounded `len`, guaranteeing the destination can always hold the source data.

3. **`strcmp` → `strncmp`:** As a defensive measure consistent with the bounded-string approach, comparisons are also length-limited so a corrupted or non-terminated string cannot cause an unbounded read during comparison.

The overall program logic — compaction via `write_index`, ownership transfer of pointers, and cleanup paths in `main` — remains unchanged and leak-free.
