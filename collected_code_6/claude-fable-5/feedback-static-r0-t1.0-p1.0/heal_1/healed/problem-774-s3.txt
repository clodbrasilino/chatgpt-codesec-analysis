```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <regex.h>

int is_valid_email(const char *email)
{
    regex_t regex;
    int ret;
    const char *pattern = "^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$";

    if (email == NULL) {
        return 0;
    }

    ret = regcomp(&regex, pattern, REG_EXTENDED | REG_NOSUB);
    if (ret != 0) {
        char errbuf[256];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        fprintf(stderr, "Regex compilation failed: %s\n", errbuf);
        return -1;
    }

    ret = regexec(&regex, email, 0, NULL, 0);

    if (ret == 0) {
        regfree(&regex);
        return 1;
    } else if (ret == REG_NOMATCH) {
        regfree(&regex);
        return 0;
    } else {
        char errbuf[256];
        errbuf[0] = '\0';
        regerror(ret, &regex, errbuf, sizeof(errbuf));
        errbuf[sizeof(errbuf) - 1] = '\0';
        regfree(&regex);
        fprintf(stderr, "Regex execution failed: %s\n", errbuf);
        return -1;
    }
}

int main(void)
{
    const char *test_emails[] = {
        "user@example.com",
        "john.doe@company.co.uk",
        "invalid.email",
        "@missing-local.org",
        "no-at-sign.com",
        "user@domain",
        "user+tag@example.io"
    };
    size_t count = sizeof(test_emails) / sizeof(test_emails[0]);
    size_t i;

    for (i = 0; i < count; i++) {
        int result = is_valid_email(test_emails[i]);
        if (result == 1) {
            printf("%s : valid\n", test_emails[i]);
        } else if (result == 0) {
            printf("%s : invalid\n", test_emails[i]);
        } else {
            fprintf(stderr, "Error validating: %s\n", test_emails[i]);
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
```

Key changes made:

1. **Use-after-free fix (the real bug):** In the original error path after `regexec`, `regfree(&regex)` was called *before* `regerror(ret, &regex, ...)`, meaning `regerror` was reading a freed/invalidated `regex_t`. The rewrite calls `regerror` first, then `regfree`, and ensures `regfree` is called on every exit path after successful compilation.

2. **Buffer hardening (CWE-119/CWE-120):** While `regerror` respects the size limit passed via `sizeof(errbuf)`, the buffer is now explicitly initialized and forcibly NUL-terminated after the call, guaranteeing a valid string even in edge cases with unusual libc implementations.

3. **Variable scope reduction:** `result` is now declared inside the `for` loop body, addressing the cppcheck `[variableScope]` finding and preventing accidental reuse of stale values across iterations.
