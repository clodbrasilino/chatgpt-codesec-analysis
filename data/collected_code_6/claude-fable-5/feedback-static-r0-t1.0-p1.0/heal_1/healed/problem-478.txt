```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_lowercase(const char *input)
{
    size_t len;
    size_t i;
    size_t j;
    char *result;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len == MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    j = 0;
    for (i = 0; i < len; i++) {
        if (!islower((unsigned char)input[i])) {
            result[j] = input[i];
            j++;
        }
    }
    result[j] = '\0';

    return result;
}

int main(void)
{
    const char *test1 = "HeLLoWoRLd";
    const char *test2 = "abcDEFghi123";
    const char *test3 = "alllowercase";
    char *output;

    output = remove_lowercase(test1);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test1, output);
    free(output);

    output = remove_lowercase(test2);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test2, output);
    free(output);

    output = remove_lowercase(test3);
    if (output == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }
    printf("Input: %s -> Output: %s\n", test3, output);
    free(output);

    return EXIT_SUCCESS;
}
```

**Key change addressing the identified weakness:**

The Flawfinder warning (CWE-126, buffer over-read) concerned `strlen()` reading past the end of a buffer if the string is not NUL-terminated. The fix replaces `strlen()` with `strnlen()`, which bounds the scan to `MAX_INPUT_LEN` bytes. If no terminator is found within that limit (i.e., `strnlen` returns the maximum), the function safely rejects the input by returning `NULL` rather than risking an over-read.

Everything else in the original code was already sound: the `(unsigned char)` cast before `islower()` correctly avoids undefined behavior for negative `char` values, the `malloc` return is checked, the output buffer cannot overflow since `j <= i < len`, and every allocation in `main` is freed.

Note: `strnlen()` is POSIX.1-2008; compile with `-D_POSIX_C_SOURCE=200809L` if your environment requires it.
