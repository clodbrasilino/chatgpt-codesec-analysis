```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INPUT_LEN 4096

int remove_parentheses(const char *input, char *output, size_t output_size)
{
    size_t i;
    size_t j;
    int depth;
    size_t len;

    if (input == NULL || output == NULL || output_size == 0) {
        return -1;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return -1;
    }

    j = 0;
    depth = 0;

    for (i = 0; i < len; i++) {
        if (input[i] == '(') {
            depth++;
        } else if (input[i] == ')') {
            if (depth > 0) {
                depth--;
            }
        } else if (depth == 0) {
            if (j + 1 >= output_size) {
                output[j] = '\0';
                return -1;
            }
            output[j] = input[i];
            j++;
        }
    }

    output[j] = '\0';

    if (depth != 0) {
        return -1;
    }

    return 0;
}

int main(void)
{
    const char *test_strings[] = {
        "Hello (world) example",
        "No parentheses here",
        "(remove this) keep this",
        "Nested (outer (inner) more) text",
        "Multiple (one) and (two) groups"
    };
    size_t count;
    size_t i;

    count = sizeof(test_strings) / sizeof(test_strings[0]);

    for (i = 0; i < count; i++) {
        size_t buf_size;
        char *buffer;

        buf_size = strnlen(test_strings[i], MAX_INPUT_LEN) + 1;
        buffer = (char *)malloc(buf_size);
        if (buffer == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            return EXIT_FAILURE;
        }

        if (remove_parentheses(test_strings[i], buffer, buf_size) == 0) {
            printf("Input:  \"%s\"\n", test_strings[i]);
            printf("Output: \"%s\"\n\n", buffer);
        } else {
            fprintf(stderr, "Failed to process: \"%s\"\n", test_strings[i]);
        }

        free(buffer);
        buffer = NULL;
    }

    return EXIT_SUCCESS;
}
```

Key changes made to address the identified weaknesses:

1. **CWE-126 (strlen over-read)**: Replaced both `strlen` calls with `strnlen` bounded by a `MAX_INPUT_LEN` constant, so a non-null-terminated string cannot cause an unbounded over-read. In `remove_parentheses`, the function now rejects inputs that reach the bound.

2. **[variableScope] on `buffer` and `buf_size`**: Both variables were moved inside the `for` loop body in `main`, reducing their scope to where they are actually used.

The existing safe practices were preserved: NULL/size validation, bounds checking before every write to `output`, guaranteed null-termination on all exit paths, malloc failure handling, and freeing plus nulling the buffer each iteration.
