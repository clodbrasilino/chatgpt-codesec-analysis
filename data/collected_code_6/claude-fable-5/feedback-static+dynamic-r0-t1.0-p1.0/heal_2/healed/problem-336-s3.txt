```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define MAX_INPUT_LEN 32

static int has_28_days(const char *month)
{
    char normalized[MAX_INPUT_LEN];
    size_t len;
    size_t i;

    if (month == NULL) {
        return -1;
    }

    len = strnlen(month, MAX_INPUT_LEN);
    if (len == 0 || len >= MAX_INPUT_LEN) {
        return -1;
    }

    for (i = 0; i < len && i < MAX_INPUT_LEN - 1; i++) {
        normalized[i] = (char)tolower((unsigned char)month[i]);
    }
    normalized[i] = '\0';

    if (strcmp(normalized, "february") == 0) {
        return 1;
    }

    return 0;
}

int main(void)
{
    char input[MAX_INPUT_LEN];
    size_t len;
    int result;

    memset(input, 0, sizeof(input));

    printf("Enter a month name: ");
    fflush(stdout);

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';

    len = strnlen(input, sizeof(input));
    if (len > 0 && input[len - 1] == '\n') {
        input[len - 1] = '\0';
    } else {
        int ch;
        while ((ch = getchar()) != '\n' && ch != EOF) {
        }
    }

    result = has_28_days(input);

    if (result < 0) {
        fprintf(stderr, "Invalid month name\n");
        return EXIT_FAILURE;
    }

    if (result == 1) {
        printf("%s contains 28 days\n", input);
    } else {
        printf("%s does not contain 28 days\n", input);
    }

    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code already mitigates the weaknesses flagged by Flawfinder, so the rewrite preserves those defenses rather than changing logic:

- **CWE-119/CWE-120 (statically-sized arrays):** Both `normalized` and `input` are fixed-size buffers, but every write into them is bounds-checked. `fgets(input, sizeof(input), stdin)` limits reads to the buffer size, `strnlen` is used instead of `strlen` to cap length scans at `MAX_INPUT_LEN`, and the copy loop in `has_28_days` enforces `i < MAX_INPUT_LEN - 1` before null-terminating at index `i`. Explicit termination via `input[sizeof(input) - 1] = '\0'` guarantees a valid C string even in edge cases.

- **CWE-120/CWE-20 (getchar in a loop):** The `getchar()` loop never writes to any buffer — it only discards excess characters left in `stdin` when the input line exceeded the buffer, terminating on `'\n'` or `EOF`. This is a safe input-flushing idiom, not an overflow risk.

- **NULL and empty-input handling:** `has_28_days` rejects `NULL` and zero-length/oversized input with `-1`, and `main` checks the `fgets` return value, so all failure paths are handled with `EXIT_FAILURE`.

The code is portable C (C99+ for `strnlen` via POSIX), uses `(unsigned char)` casting before `tolower` to avoid undefined behavior on negative `char` values, and requires no functional changes to be secure.
