```c
#include <stdio.h>
#include <string.h>

const int ASCII_SIZE = 256;

char getMinOccurringChar(char *str){
    int count[ASCII_SIZE] = {0};
    int n = strlen(str);
    for (int i = 0; i < n; i++)
        count[str[i]]++;
    int min = ASCII_SIZE, c;
    for (int i = 0; i < n; i++) {
        if (count[str[i]] < min) {
            min = count[str[i]], c = str[i];
        }
    }

    return c;
}

void removeChar(char *s, int c){
    int j = 0;
    for (int i = 0; s[i]; i++)
        if (s[i] != c)
            s[j++] = s[i];

    s[j] = '\0';
}

int main(){
    char str[] = "hello world";
    char minOccChar = getMinOccurringChar(str);
    removeChar(str, minOccChar);
    printf("%s\n", str);
    return 0;
}
```
