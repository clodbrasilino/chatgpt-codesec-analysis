```c
#include <stdio.h>
#include <stdlib.h>

double calculate_perimeter(double side_a, double side_b);

int main(void)
{
    double side_a = 0.0;
    double side_b = 0.0;
    double perimeter = 0.0;
    int result = 0;

    printf("Enter the length of side a: ");
    result = scanf("%lf", &side_a);
    if (result != 1 || side_a <= 0.0)
    {
        fprintf(stderr, "Invalid input for side a.\n");
        return EXIT_FAILURE;
    }

    printf("Enter the length of side b: ");
    result = scanf("%lf", &side_b);
    if (result != 1 || side_b <= 0.0)
    {
        fprintf(stderr, "Invalid input for side b.\n");
        return EXIT_FAILURE;
    }

    perimeter = calculate_perimeter(side_a, side_b);
    printf("Perimeter of the parallelogram: %.2f\n", perimeter);

    return EXIT_SUCCESS;
}

double calculate_perimeter(double side_a, double side_b)
{
    return 2.0 * (side_a + side_b);
}
```
