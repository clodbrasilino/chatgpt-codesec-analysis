#include <stdio.h>
#include <stdlib.h>

int find_min_sublist_length(const size_t *lengths, size_t count, size_t *result)
{
    size_t i;
    size_t min_len;

    if (lengths == NULL || result == NULL || count == 0U) {
        return -1;
    }

    min_len = lengths[0];
    for (i = 1U; i < count; i++) {
        if (lengths[i] < min_len) {
            min_len = lengths[i];
        }
    }

    *result = min_len;
    return 0;
}

int main(void)
{
    int list1[] = {1};
    int list2[] = {1, 2};
    int list3[] = {1, 2, 3};
    int *lists[3];
    size_t lengths[3];
    size_t min_len = 0U;
    size_t i;
    size_t j;
    int status;

    lists[0] = list1;
    lists[1] = list2;
    lists[2] = list3;

    lengths[0] = sizeof(list1) / sizeof(list1[0]);
    lengths[1] = sizeof(list2) / sizeof(list2[0]);
    lengths[2] = sizeof(list3) / sizeof(list3[0]);

    status = find_min_sublist_length(lengths, 3U, &min_len);
    if (status != 0) {
        (void)fprintf(stderr, "Error: invalid input\n");
        return EXIT_FAILURE;
    }

    for (i = 0U; i < 3U; i++) {
        (void)printf("Sublist %zu: ", i);
        for (j = 0U; j < lengths[i]; j++) {
            (void)printf("%d ", lists[i][j]);
        }
        (void)printf("\n");
    }

    (void)printf("Minimum sublist length: %zu\n", min_len);

    return EXIT_SUCCESS;
}
