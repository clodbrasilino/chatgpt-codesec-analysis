```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_INPUT_LEN 4096

char *remove_uppercase(const char *input)
{
    char *result;
    size_t len;
    size_t i;
    size_t out;

    if (input == NULL) {
        return NULL;
    }

    len = strnlen(input, MAX_INPUT_LEN);
    if (len >= MAX_INPUT_LEN) {
        return NULL;
    }

    result = malloc(len + 1);
    if (result == NULL) {
        return NULL;
    }

    out = 0;
    for (i = 0; i < len; i++) {
        unsigned char c = (unsigned char)input[i];
        if (!isupper(c)) {
            result[out] = (char)c;
            out++;
        }
    }

    result[out] = '\0';
    return result;
}

int main(void)
{
    char buffer[MAX_INPUT_LEN];
    char *cleaned;
    size_t len;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Failed to read input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len > 0 && buffer[len - 1] == '\n') {
        buffer[len - 1] = '\0';
    }

    cleaned = remove_uppercase(buffer);
    if (cleaned == NULL) {
        fprintf(stderr, "Failed to process string\n");
        return EXIT_FAILURE;
    }

    if (printf("%s\n", cleaned) < 0) {
        free(cleaned);
        return EXIT_FAILURE;
    }

    free(cleaned);
    return EXIT_SUCCESS;
}
```

**Analysis notes:**

The original code failed all test cases (`<no output>`) because of the regex-based approach combined with overly aggressive and partly incorrect bounds arithmetic. In particular, checks like `out > len - chunk` and `pos > len - chunk` mix up which quantities they are guarding, and can trigger spurious `NULL` returns on valid input (unsigned underflow-style logic errors when `chunk` relates to `out` rather than `pos`). When `remove_uppercase` returns `NULL`, `main` prints to `stderr` and exits, producing no stdout output — exactly matching the failing test symptoms.

Key improvements in the rewrite:

1. **Eliminated the regex engine entirely** — filtering uppercase characters is a linear scan; `regcomp`/`regexec` added complexity, allocation churn, and the fragile offset bookkeeping that caused the failures. `isupper()` on an `unsigned char` cast (avoiding UB for negative `char` values) does the job correctly.

2. **CWE-120 (memcpy buffer overflow) resolved by construction** — the destination is `len + 1` bytes and at most `len` characters plus a terminator are written; no `memcpy` with computed offsets remains, so the Flawfinder findings are moot.

3. **CWE-119/CWE-120 (statically sized buffer)** — retained the fixed buffer but kept the mitigations: `fgets` bounded by `sizeof(buffer)`, forced NUL termination, and `strnlen` for length computation, so no unbounded read or scan is possible.

4. **Simplified control flow** — a single loop with one obvious invariant (`out <= i <= len`) replaces the multi-branch error paths, making the correctness auditable at a glance and removing every early-return failure mode that broke the tests.
