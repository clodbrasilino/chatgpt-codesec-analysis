#include <stdio.h>
#include <string.h>
#include <regex.h>

void remove_lowercase_substrings(char *str) {
    regex_t reg;
    regmatch_t pmatch[1];

    if (regcomp(&reg, "[a-z]+", REG_EXTENDED)) {

    }

    while (regexec(&reg, str, 1, pmatch, 0) == 0) {

    }

    regfree(&reg);
}

int main() {
