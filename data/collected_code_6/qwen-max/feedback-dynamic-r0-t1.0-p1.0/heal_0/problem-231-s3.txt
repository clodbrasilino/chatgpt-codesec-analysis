#include <stdio.h>
#include <stdlib.h>

int** allocateTriangle(int n) {
    int **triangle = (int**)malloc(n * sizeof(int*));
    for (int i = 0; i < n; ++i) {
        triangle[i] = (int*)malloc((i + 1) * sizeof(int));
    }
    return triangle;
}

void freeTriangle(int** triangle, int n) {
    for (int i = 0; i < n; ++i) {
        free(triangle[i]);
    }
    free(triangle);
}

int findMaxSumInTriangle(int** triangle, int n) {
    for (int i = n - 2; i >= 0; --i) {
        for (int j = 0; j <= i; ++j) {
            triangle[i][j] += (triangle[i + 1][j] > triangle[i + 1][j + 1] ? triangle[i + 1][j] : triangle[i + 1][j + 1]);
        }
    }
    return triangle[0][0];
}

int main() {
    int n;
    printf("Enter the number of rows: ");
    scanf("%d", &n);

    int** triangle = allocateTriangle(n);

    printf("Enter %d elements:\n", n * (n + 1) / 2);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= i; ++j) {
            scanf("%d", &triangle[i][j]);
        }
    }

    int maxSum = findMaxSumInTriangle(triangle, n);
    printf("Maximum sum: %d\n", maxSum);

    freeTriangle(triangle, n);
    return 0;
}
