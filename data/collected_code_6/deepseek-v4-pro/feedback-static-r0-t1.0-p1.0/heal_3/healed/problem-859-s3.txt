#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

typedef struct {
    int* items;
    int length;
} Sublist;

void print_sublists(const Sublist* sublists, int count) {
    for (int i = 0; i < count; i++) {
        printf("[");
        for (int j = 0; j < sublists[i].length; j++) {
            printf("%d", sublists[i].items[j]);
            if (j < sublists[i].length - 1) {
                printf(", ");
            }
        }
        printf("]\n");
    }
}

Sublist* generate_sublists(const int* list, int size, int* out_count) {
    if (list == NULL && size != 0) {
        *out_count = 0;
        return NULL;
    }

    if (size < 0 || size >= (int)(sizeof(size_t) * 8)) {
        *out_count = 0;
        return NULL;
    }

    size_t total_sublists = (size_t)1 << size;
    if (total_sublists > SIZE_MAX / sizeof(Sublist)) {
        *out_count = 0;
        return NULL;
    }

    Sublist* sublists = (Sublist*)malloc(total_sublists * sizeof(Sublist));
    if (sublists == NULL) {
        *out_count = 0;
        return NULL;
    }

    for (size_t i = 0; i < total_sublists; i++) {
        int bit_count = 0;
        for (int j = 0; j < size; j++) {
            if (i & ((size_t)1 << j)) {
                bit_count++;
            }
        }

        sublists[i].items = (int*)malloc((size_t)bit_count * sizeof(int));
        if (sublists[i].items == NULL && bit_count > 0) {
            for (size_t k = 0; k < i; k++) {
                free(sublists[k].items);
            }
            free(sublists);
            *out_count = 0;
            return NULL;
        }

        sublists[i].length = bit_count;
        int idx = 0;
        for (int j = 0; j < size; j++) {
            if (i & ((size_t)1 << j)) {
                sublists[i].items[idx++] = list[j];
            }
        }
    }

    *out_count = (int)total_sublists;
    return sublists;
}

void free_sublists(Sublist* sublists, int count) {
    if (sublists == NULL) {
        return;
    }
    for (int i = 0; i < count; i++) {
        free(sublists[i].items);
    }
    free(sublists);
}

int main(void) {
    int list[] = {1, 2, 3};
    int size = (int)(sizeof(list) / sizeof(list[0]));
    int count = 0;

    Sublist* sublists = generate_sublists(list, size, &count);
    if (sublists == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1;
    }

    print_sublists(sublists, count);
    free_sublists(sublists, count);

    return 0;
}
