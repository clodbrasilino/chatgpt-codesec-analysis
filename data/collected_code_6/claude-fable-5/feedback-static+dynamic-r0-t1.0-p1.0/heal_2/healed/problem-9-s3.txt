#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <limits.h>

static int find_min_rotations(const char *str, size_t max_len)
{
    size_t len;
    size_t i;
    size_t doubled_size;
    char *doubled;
    int result;

    if (str == NULL || max_len == 0U) {
        return -1;
    }

    len = strnlen(str, max_len);
    if (len == 0U || len == max_len) {
        return -1;
    }

    if (len > (SIZE_MAX - 1U) / 2U) {
        return -1;
    }

    if (len > (size_t)INT_MAX) {
        return -1;
    }

    doubled_size = (2U * len) + 1U;

    doubled = malloc(doubled_size);
    if (doubled == NULL) {
        return -1;
    }

    if (len >= doubled_size || (2U * len) >= doubled_size) {
        free(doubled);
        return -1;
    }

    memcpy(doubled, str, len);
    memcpy(doubled + len, str, len);
    doubled[2U * len] = '\0';

    result = -1;
    for (i = 1U; i <= len; i++) {
        if (strncmp(doubled + i, str, len) == 0) {
            result = (int)i;
            break;
        }
    }

    free(doubled);
    return result;
}

int main(void)
{
    char buffer[256];
    size_t len;
    int rotations;

    memset(buffer, 0, sizeof(buffer));

    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strnlen(buffer, sizeof(buffer));
    if (len >= sizeof(buffer)) {
        fprintf(stderr, "Input too long\n");
        return EXIT_FAILURE;
    }

    if (len > 0U && buffer[len - 1U] == '\n') {
        buffer[len - 1U] = '\0';
        len--;
    }

    if (len == 0U) {
        fprintf(stderr, "Empty string provided\n");
        return EXIT_FAILURE;
    }

    rotations = find_min_rotations(buffer, sizeof(buffer));
    if (rotations < 0) {
        fprintf(stderr, "Failed to compute rotations\n");
        return EXIT_FAILURE;
    }

    if (printf("Minimum rotations required: %d\n", rotations) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

**Analysis of the original code and the fixes applied:**

The original code was already largely defensive, and the Flawfinder hits were mostly false positives — but here is how each flagged weakness is addressed:

1. **`memcpy` (CWE-120, both instances):** The original code was actually safe because `doubled` is allocated as `2*len + 1` bytes, with overflow guarded by the `len > (SIZE_MAX - 1U)/2U` check. In the rewrite, an explicit post-allocation bounds verification (`len >= doubled_size`, `2U * len >= doubled_size`) was added so the invariant "destination can always hold the source data" is provably checked immediately before the copies, silencing the CWE-120 concern with a real runtime guard rather than an implicit one.

2. **Statically-sized `char buffer[256]` (CWE-119/CWE-120):** The buffer is now zero-initialized with `memset` before use, `fgets` is bounded by `sizeof(buffer)`, forced NUL-termination is retained, and an explicit `len >= sizeof(buffer)` sanity check rejects any pathological unterminated input, guaranteeing all subsequent indexing (`buffer[len - 1U]`) stays in bounds.

3. **Retained hardening from the original:** NULL checks, `strnlen` instead of `strlen`, integer-overflow guard before `malloc`, `INT_MAX` guard before the `(int)` cast, `malloc` failure handling, `free` on all paths, and checking the return value of `printf` — all preserved, plus an added `max_len == 0U` guard in `find_min_rotations`.
