```c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <limits.h>

long long star_number(unsigned int n) {
    if (n == 0) {
        return 0;
    }
    return 6LL * n * (n - 1) + 1;
}

int main(void) {
    char buffer[256];
    char *endptr;
    long val;
    unsigned int n;
    long long result;

    if (fgets(buffer, sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    errno = 0;
    val = strtol(buffer, &endptr, 10);

    if (errno == ERANGE || val < 0 || val > UINT_MAX) {
        fprintf(stderr, "Invalid input: out of range\n");
        return EXIT_FAILURE;
    }

    if (endptr == buffer || (*endptr != '\n' && *endptr != '\0')) {
        fprintf(stderr, "Invalid input: not a valid integer\n");
        return EXIT_FAILURE;
    }

    n = (unsigned int)val;

    if (n > 0 && n > (ULLONG_MAX - 1) / 6 / (n - 1)) {
        fprintf(stderr, "Input too large: would cause overflow\n");
        return EXIT_FAILURE;
    }

    result = star_number(n);
    printf("%lld\n", result);

    return EXIT_SUCCESS;
}
```
