```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

int split_and_rotate(int *arr, size_t n, size_t k)
{
    int *temp;
    size_t head_bytes;
    size_t tail_bytes;

    if (arr == NULL || n == 0) {
        return -1;
    }

    if (n > SIZE_MAX / sizeof(int)) {
        return -1;
    }

    k = k % n;
    if (k == 0) {
        return 0;
    }

    head_bytes = k * sizeof(int);
    tail_bytes = (n - k) * sizeof(int);

    temp = malloc(head_bytes);
    if (temp == NULL) {
        return -1;
    }

    memcpy(temp, arr, head_bytes);
    memmove(arr, arr + k, tail_bytes);
    memcpy(arr + (n - k), temp, head_bytes);

    free(temp);
    return 0;
}

int main(void)
{
    int arr[] = {12, 10, 5, 6, 52, 36};
    size_t n = sizeof(arr) / sizeof(arr[0]);
    size_t k = 2;
    size_t i;

    if (split_and_rotate(arr, n, k) != 0) {
        fprintf(stderr, "Rotation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < n; i++) {
        if (printf("%d ", arr[i]) < 0) {
            return EXIT_FAILURE;
        }
    }

    if (printf("\n") < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

**Analysis of the changes:**

1. **Removed redundant conditions flagged as `knownConditionTrueFalse`:** After `k = k % n`, it is mathematically guaranteed that `k < n`. Therefore `head_bytes <= total_bytes`, `tail_bytes <= total_bytes`, `k <= n`, and `(n - k) <= n` are always true, and `head_bytes + tail_bytes == total_bytes` always holds. These checks were dead code and have been removed.

2. **Retained the essential overflow guard:** The check `n > SIZE_MAX / sizeof(int)` remains, since it prevents integer overflow in the byte-size calculations. This is the only arithmetic check that can actually fail.

3. **Addressed the Flawfinder `memcpy` warnings (CWE-120):** The warnings are heuristic. The rewritten code guarantees safety by construction:
   - `temp` is allocated with exactly `head_bytes`, and exactly `head_bytes` are copied into it.
   - `arr` holds `n * sizeof(int)` bytes; `memmove` writes `tail_bytes` (`(n-k)*sizeof(int)`) starting at index 0, and the final `memcpy` writes `head_bytes` starting at index `n-k`, together filling exactly the buffer with no overflow, since `k < n` is invariant after the modulo.

4. **Kept the `NULL`/zero-length guard and the `malloc` failure check**, as those are genuine runtime failure paths.

5. **`free(temp)`** occurs exactly once on the success path; since the redundant error branches were removed, there are no leak paths remaining.
