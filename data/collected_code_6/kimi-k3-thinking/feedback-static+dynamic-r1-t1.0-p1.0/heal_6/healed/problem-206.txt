#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <ctype.h>

#define MAX_ELEMENT_LENGTH (16777216U)

static void free_string_array(char **strings, size_t count)
{
    if (strings == NULL)
    {
        return;
    }

    for (size_t i = 0U; i < count; i++)
    {
        free(strings[i]);
    }
    free(strings);
}

static size_t bounded_length(const char *s, int *terminated)
{
    size_t len = 0U;

    *terminated = 0;

    if (s == NULL)
    {
        return 0U;
    }

    while (len < MAX_ELEMENT_LENGTH)
    {
        if (s[len] == '\0')
        {
            *terminated = 1;
            return len;
        }
        len++;
    }

    return len;
}

static char **concat_adjacent_elements(const char *const *elements, size_t count, size_t *out_count)
{
    if (out_count == NULL)
    {
        return NULL;
    }
    *out_count = 0U;

    if (elements == NULL || count < 2U)
    {
        return NULL;
    }

    if ((count - 1U) > (SIZE_MAX / sizeof(char *)))
    {
        return NULL;
    }

    char **result = malloc((count - 1U) * sizeof(*result));
    if (result == NULL)
    {
        return NULL;
    }

    for (size_t i = 0U; i < count - 1U; i++)
    {
        const char *left = elements[i];
        const char *right = elements[i + 1U];

        if (left == NULL || right == NULL)
        {
            free_string_array(result, i);
            return NULL;
        }

        int left_ok;
        int right_ok;
        size_t left_len = bounded_length(left, &left_ok);
        size_t right_len = bounded_length(right, &right_ok);

        if (!left_ok || !right_ok)
        {
            free_string_array(result, i);
            return NULL;
        }

        if (right_len > (SIZE_MAX - left_len - 1U))
        {
            free_string_array(result, i);
            return NULL;
        }

        size_t total_len = left_len + right_len;
        size_t capacity = total_len + 1U;

        result[i] = malloc(capacity);
        if (result[i] == NULL)
        {
            free_string_array(result, i);
            return NULL;
        }

        size_t written = 0U;
        for (size_t j = 0U; (j < left_len) && (written < (capacity - 1U)); j++)
        {
            result[i][written] = left[j];
            written++;
        }
        for (size_t j = 0U; (j < right_len) && (written < (capacity - 1U)); j++)
        {
            result[i][written] = right[j];
            written++;
        }
        result[i][written] = '\0';
    }

    *out_count = count - 1U;
    return result;
}

static int add_token(char ***tokens, size_t *count, size_t *capacity, char *token)
{
    if (token == NULL)
    {
        return -1;
    }

    if (*count == *capacity)
    {
        size_t new_capacity = (*capacity == 0U) ? 8U : (*capacity * 2U);

        if ((new_capacity < *capacity) || (new_capacity > (SIZE_MAX / sizeof(**tokens))))
        {
            free(token);
            return -1;
        }

        char **grown = realloc(*tokens, new_capacity * sizeof(**tokens));
        if (grown == NULL)
        {
            free(token);
            return -1;
        }

        *tokens = grown;
        *capacity = new_capacity;
    }

    (*tokens)[*count] = token;
    *count += 1U;
    return 0;
}

static int split_and_add(const char *text, char ***tokens, size_t *count, size_t *capacity)
{
    const char *cursor = text;

    if (cursor == NULL)
    {
        return -1;
    }

    while (*cursor != '\0')
    {
        while ((*cursor != '\0') && isspace((unsigned char)*cursor))
        {
            cursor++;
        }
        if (*cursor == '\0')
        {
            break;
        }

        const char *start = cursor;
        while ((*cursor != '\0') && !isspace((unsigned char)*cursor))
        {
            cursor++;
        }

        size_t length = (size_t)(cursor - start);
        if ((length == 0U) || (length == SIZE_MAX))
        {
            return -1;
        }

        char *token = malloc(length + 1U);
        if (token == NULL)
        {
            return -1;
        }

        for (size_t j = 0U; j < length; j++)
        {
            token[j] = start[j];
        }
        token[length] = '\0';

        if (add_token(tokens, count, capacity, token) != 0)
        {
            return -1;
        }
    }

    return 0;
}

static char *read_token(FILE *stream)
{
    size_t capacity = 32U;
    size_t length = 0U;
    int c;

    do
    {
        c = fgetc(stream);
    } while ((c != EOF) && isspace((unsigned char)c));

    if (c == EOF)
    {
        return NULL;
    }

    char *buffer = malloc(capacity);
    if (buffer == NULL)
    {
        return NULL;
    }

    while ((c != EOF) && !isspace((unsigned char)c))
    {
        if ((length + 1U) >= capacity)
        {
            if (capacity > (SIZE_MAX / 2U))
            {
                free(buffer);
                return NULL;
            }

            size_t new_capacity = capacity * 2U;
            char *grown = realloc(buffer, new_capacity);
            if (grown == NULL)
            {
                free(buffer);
                return NULL;
            }
            buffer = grown;
            capacity = new_capacity;
        }

        buffer[length] = (char)c;
        length++;
        c = fgetc(stream);
    }

    buffer[length] = '\0';
    return buffer;
}

int main(int argc, char **argv)
{
    static const char default_input[] = "DSP IS BEST FOR ALL UTS";
    char **tokens = NULL;
    size_t token_count = 0U;
    size_t token_capacity = 0U;
    int exit_code = EXIT_FAILURE;

    if (argc > 1)
    {
        for (int arg_index = 1; arg_index < argc; arg_index++)
        {
            if (split_and_add(argv[arg_index], &tokens, &token_count, &token_capacity) != 0)
            {
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
        }
    }
    else
    {
        char *token;

        while ((token = read_token(stdin)) != NULL)
        {
            if (add_token(&tokens, &token_count, &token_capacity, token) != 0)
            {
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }
        }
    }

    if (token_count == 0U)
    {
        if (split_and_add(default_input, &tokens, &token_count, &token_capacity) != 0)
        {
            free_string_array(tokens, token_count);
            return EXIT_FAILURE;
        }
    }

    if (token_count >= 2U)
    {
        char **elements = calloc(token_count, sizeof(*elements));
        if (elements == NULL)
        {
            free_string_array(tokens, token_count);
            return EXIT_FAILURE;
        }

        for (size_t i = 0U; i < token_count; i++)
        {
            int terminated;
            size_t length = bounded_length(tokens[i], &terminated);
            if (!terminated)
            {
                free_string_array(elements, token_count);
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }

            int needs_space = ((i + 1U) < token_count);
            size_t total = length + (needs_space ? 1U : 0U);

            elements[i] = malloc(total + 1U);
            if (elements[i] == NULL)
            {
                free_string_array(elements, token_count);
                free_string_array(tokens, token_count);
                return EXIT_FAILURE;
            }

            for (size_t j = 0U; j < length; j++)
            {
                elements[i][j] = tokens[i][j];
            }
            if (needs_space)
            {
                elements[i][length] = ' ';
            }
            elements[i][total] = '\0';
        }

        size_t result_count = 0U;
        char **result = concat_adjacent_elements((const char *const *)elements, token_count, &result_count);
        if (result == NULL)
        {
            fprintf(stderr, "Error: failed to concatenate adjacent elements.\n");
        }
        else
        {
            for (size_t i = 0U; i < result_count; i++)
            {
                printf("%s\n", result[i]);
            }
            free_string_array(result, result_count);
            exit_code = EXIT_SUCCESS;
        }

        free_string_array(elements, token_count);
    }
    else
    {
        exit_code = EXIT_SUCCESS;
    }

    free_string_array(tokens, token_count);
    return exit_code;
}
