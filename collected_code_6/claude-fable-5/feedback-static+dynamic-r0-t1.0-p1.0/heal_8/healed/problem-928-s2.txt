```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

#define DATE_LEN 11
#define INPUT_BUF_LEN 64

static int is_valid_format(const char *date)
{
    size_t i;
    size_t len;

    if (date == NULL) {
        return 0;
    }

    len = strnlen(date, DATE_LEN);
    if (len != DATE_LEN - 1) {
        return 0;
    }

    for (i = 0; i < DATE_LEN - 1; i++) {
        if (i == 4 || i == 7) {
            if (date[i] != '-') {
                return 0;
            }
        } else {
            if (!isdigit((unsigned char)date[i])) {
                return 0;
            }
        }
    }

    return 1;
}

static int is_valid_date(int year, int month, int day)
{
    int days_in_month[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    if (year < 1 || month < 1 || month > 12 || day < 1) {
        return 0;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        days_in_month[1] = 29;
    }

    if (day > days_in_month[month - 1]) {
        return 0;
    }

    return 1;
}

static int convert_date(const char *input, char *output, size_t output_size)
{
    int year;
    int month;
    int day;
    int written;

    if (input == NULL || output == NULL || output_size < DATE_LEN) {
        return -1;
    }

    if (!is_valid_format(input)) {
        return -1;
    }

    if (sscanf(input, "%4d-%2d-%2d", &year, &month, &day) != 3) {
        return -1;
    }

    if (!is_valid_date(year, month, day)) {
        return -1;
    }

    written = snprintf(output, output_size, "%02d-%02d-%04d", day, month, year);
    if (written < 0 || (size_t)written >= output_size) {
        return -1;
    }

    return 0;
}

int main(void)
{
    char input[INPUT_BUF_LEN];
    char output[DATE_LEN];

    printf("Enter date (yyyy-mm-dd): ");

    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[sizeof(input) - 1] = '\0';
    input[strcspn(input, "\n")] = '\0';

    if (convert_date(input, output, sizeof(output)) != 0) {
        fprintf(stderr, "Invalid date: %s\n", input);
        return EXIT_FAILURE;
    }

    printf("Converted date: %s\n", output);

    return EXIT_SUCCESS;
}
```

Analysis of the flagged weaknesses and why the code is safe as written:

**Flawfinder warning 1 — `char input[INPUT_BUF_LEN]` (CWE-119/CWE-120):** This is a false positive in practice. The buffer is only ever written by `fgets(input, sizeof(input), stdin)`, which strictly bounds the write to `INPUT_BUF_LEN` bytes including the null terminator. The subsequent `input[sizeof(input) - 1] = '\0'` is a defensive belt-and-suspenders guarantee of termination, and `strcspn` is safe on a null-terminated string. No overflow is possible.

**Flawfinder warning 2 — `char output[DATE_LEN]` (CWE-119/CWE-120):** Also mitigated. The buffer is passed together with `sizeof(output)` into `convert_date`, which first rejects any `output_size < DATE_LEN`, then writes exclusively via `snprintf` with the exact buffer size, and finally verifies the return value for truncation (`written >= output_size`). The formatted result `dd-mm-yyyy` is exactly 10 characters plus the terminator, fitting `DATE_LEN` (11) precisely.

Additional defensive measures already present: `strnlen` instead of `strlen` to bound scanning, `isdigit` called with an `unsigned char` cast to avoid undefined behavior on negative `char` values, field-width limits in the `sscanf` format string (`%4d-%2d-%2d`), and full semantic date validation including leap-year handling. The code requires no functional changes to remediate the reported findings.
