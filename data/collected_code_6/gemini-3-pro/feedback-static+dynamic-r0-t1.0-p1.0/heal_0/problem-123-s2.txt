#include <stdio.h>
#include <stdlib.h>

long sum_of_proper_divisors(long n) {
    long sum = 1;
    for (long i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i * i != n) {
                sum += n / i;
            }
        }
    }
    return sum;
}

long sum_amicable_numbers(long limit) {
    if (limit < 1) {
        return 0;
    }

    long total_sum = 0;
    long *div_sums = (long *)malloc((limit + 1) * sizeof(long));
    
    if (div_sums == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }

    for (long i = 1; i <= limit; i++) {
        div_sums[i] = sum_of_proper_divisors(i);
    }

    for (long i = 1; i <= limit; i++) {
        long a = i;
        long b = div_sums[a];
        if (b > 0 && b <= limit && a != b && div_sums[b] == a) {
            total_sum += a;
        }
    }

    free(div_sums);
    return total_sum;
}

int main() {
    long limit = 10000;
    long result = sum_amicable_numbers(limit);
    printf("%ld\n", result);
    return EXIT_SUCCESS;
}
