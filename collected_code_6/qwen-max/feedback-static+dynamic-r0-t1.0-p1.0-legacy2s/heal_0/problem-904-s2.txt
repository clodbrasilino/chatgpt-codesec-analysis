```c
#include <stdbool.h>
#include <stdio.h>

bool is_even(int num) {
    return (num % 2) == 0;
}

int main() {
    int number = 4; // Example number
    if (is_even(number)) {
        printf("%d is even.\n", number);
    } else {
        printf("%d is not even.\n", number);
    }
    return 0;
}
```
