bool is_woodall(unsigned long long n) {
    if (n == 0) return false;
    
    unsigned long long m = 1;
    int k = 1;
    
    while (1) {
        if (k > 0 && m > (ULLONG_MAX + 1ULL) / (unsigned long long)k) break;
        
        if (k > 0 && m > (n + 1ULL) / (unsigned long long)k) break;
        
        unsigned long long woodall = (unsigned long long)k * m - 1ULL;
        if (woodall == n) return true;
        if (woodall > n) break;
        
        k++;
        if (m > ULLONG_MAX / 2ULL) break;
        m *= 2ULL;
    }
    
    return false;
}

int main(void) {
    unsigned long long num;
