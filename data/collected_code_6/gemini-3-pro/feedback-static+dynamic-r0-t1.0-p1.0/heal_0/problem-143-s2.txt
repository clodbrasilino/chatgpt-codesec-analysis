#include <stdio.h>

int count_lists_in_tuple(const char *tuple_str) {
    if (tuple_str == NULL) {
        return 0;
    }

    int count = 0;
    for (int i = 0; tuple_str[i] != '\0'; i++) {
        if (tuple_str[i] == '[') {
            count++;
        }
    }
    
    return count;
}

int main(void) {
    const char *sample_tuple = "(1, [2, 3], 4, [5, 6], 'test', [])";
    int list_count;

    list_count = count_lists_in_tuple(sample_tuple);
    printf("%d\n", list_count);

    return 0;
}
