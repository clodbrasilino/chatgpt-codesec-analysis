#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <stdbool.h>

bool is_hex_even(const char *hex_str) {
    if (hex_str == NULL || hex_str[0] == '\0') {
        return false;
    }

    size_t i = 0;
    if (hex_str[0] == '0' && (hex_str[1] == 'x' || hex_str[1] == 'X')) {
        i = 2;
    }

    if (hex_str[i] == '\0') {
        return false;
    }

    char last_char = '\0';
    
    while (hex_str[i] != '\0') {
        if (!isxdigit((unsigned char)hex_str[i])) {
            return false;
        }
        last_char = tolower((unsigned char)hex_str[i]);
        i++;
    }

    if (last_char == '0' || last_char == '2' || last_char == '4' || 
        last_char == '6' || last_char == '8' || last_char == 'a' || 
        last_char == 'c' || last_char == 'e') {
        return true;
    }

    return false;
}

int main(int argc, const char * const argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <hexadecimal_number>\n", argv[0]);
        return EXIT_FAILURE;
    }

    if (is_hex_even(argv[1])) {
        printf("Even\n");
    } else {
        printf("Odd or Invalid\n");
    }

    return EXIT_SUCCESS;
}
