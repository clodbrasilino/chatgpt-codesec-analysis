```c
#include <stdio.h>

int min_of_three(int a, int b, int c) {
    int min = a;
    
    if (b < min) {
        min = b;
    }
    if (c < min) {
        min = c;
    }
    
    return min;
}

int main(void) {
    int num1 = 10, num2 = 5, num3 = 8;
    
    printf("Minimum of %d, %d, and %d is %d\n", 
           num1, num2, num3, min_of_three(num1, num2, num3));
    
    return 0;
}
```
