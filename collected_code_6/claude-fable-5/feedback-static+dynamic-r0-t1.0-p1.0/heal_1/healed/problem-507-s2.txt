```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

size_t remove_words(char **list, size_t count, const char **words_to_remove, size_t remove_count)
{
    size_t write_index = 0;
    size_t i;
    size_t j;

    if (list == NULL || words_to_remove == NULL) {
        return count;
    }

    for (i = 0; i < count; i++) {
        int match = 0;
        if (list[i] != NULL) {
            for (j = 0; j < remove_count; j++) {
                if (words_to_remove[j] != NULL &&
                    strcmp(list[i], words_to_remove[j]) == 0) {
                    match = 1;
                    break;
                }
            }
        }
        if (match) {
            free(list[i]);
            list[i] = NULL;
        } else {
            list[write_index] = list[i];
            if (write_index != i) {
                list[i] = NULL;
            }
            write_index++;
        }
    }

    return write_index;
}

char *duplicate_string(const char *src)
{
    size_t len;
    char *copy;

    if (src == NULL) {
        return NULL;
    }

    len = strlen(src);
    if (len == (size_t)-1) {
        return NULL;
    }
    len += 1;

    copy = malloc(len);
    if (copy == NULL) {
        return NULL;
    }

    memcpy(copy, src, len);
    copy[len - 1] = '\0';
    return copy;
}

void free_list(char **list, size_t count)
{
    size_t i;

    if (list == NULL) {
        return;
    }
    for (i = 0; i < count; i++) {
        free(list[i]);
    }
    free(list);
}

int main(void)
{
    const char *initial_words[] = { "apple", "banana", "cherry", "date", "banana", "fig" };
    const char *words_to_remove[] = { "banana", "date" };
    size_t initial_count = sizeof(initial_words) / sizeof(initial_words[0]);
    size_t remove_count = sizeof(words_to_remove) / sizeof(words_to_remove[0]);
    char **list;
    size_t i;
    size_t new_count;

    list = malloc(initial_count * sizeof(char *));
    if (list == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return EXIT_FAILURE;
    }

    for (i = 0; i < initial_count; i++) {
        list[i] = duplicate_string(initial_words[i]);
        if (list[i] == NULL) {
            fprintf(stderr, "Memory allocation failed\n");
            free_list(list, i);
            return EXIT_FAILURE;
        }
    }

    printf("Original list:\n");
    for (i = 0; i < initial_count; i++) {
        printf("  %s\n", list[i]);
    }

    new_count = remove_words(list, initial_count, words_to_remove, remove_count);

    printf("List after removal:\n");
    for (i = 0; i < new_count; i++) {
        printf("  %s\n", list[i]);
    }

    free_list(list, new_count);

    return EXIT_SUCCESS;
}
```

**Summary of fixes applied:**

1. **`[variableScope]` on `match`** — The declaration of `match` was moved from function scope into the body of the outer `for` loop in `remove_words`, reducing its scope to exactly where it is used and guaranteeing re-initialization on every iteration.

2. **Flawfinder `strlen` (CWE-126, over-read on unterminated strings)** — The function already guards against `NULL` input; since all call sites pass string literals (guaranteed NUL-terminated), the risk is inherently mitigated. A defensive check against `size_t` overflow before the `+ 1` increment was added to prevent a zero-byte allocation wrap-around.

3. **Flawfinder `memcpy` (CWE-120, buffer overflow)** — The destination buffer is allocated with exactly `len` bytes immediately before the copy of `len` bytes, so the destination is guaranteed to hold the source. An explicit `copy[len - 1] = '\0'` was added as a belt-and-braces guarantee that the result is always NUL-terminated.

The program's behavior is unchanged: it prints the original list, removes `"banana"` and `"date"`, prints the compacted list, and frees all memory with no leaks (removed entries are freed inside `remove_words`, survivors via `free_list`).
