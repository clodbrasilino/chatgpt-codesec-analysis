#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

long long count_cubes(int n, int k) {
    if (n < k || k <= 0) {
        return 0;
    }
    long long diff = (long long)n - k;
    if (diff > 0 && diff > LLONG_MAX / diff / diff) {
        return -1;
    }
    return diff * diff * diff;
}

int main(void) {
    int n, k;
    printf("Enter n: ");
    if (scanf("%d", &n) != 1) {
        return EXIT_FAILURE;
    }
    printf("Enter k: ");
    if (scanf("%d", &k) != 1) {
        return EXIT_FAILURE;
    }
    if (n < 0 || k <= 0) {
        fprintf(stderr, "Invalid input\n");
        return EXIT_FAILURE;
    }
    long long result = count_cubes(n, k);
    if (result == -1) {
        fprintf(stderr, "Overflow detected\n");
        return EXIT_FAILURE;
    }
    printf("%lld\n", result);
    return EXIT_SUCCESS;
}
