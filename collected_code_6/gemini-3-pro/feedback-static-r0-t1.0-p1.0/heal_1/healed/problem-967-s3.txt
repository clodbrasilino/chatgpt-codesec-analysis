#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <stdbool.h>

bool has_all_vowels(const char *str) {
    if (str == NULL) {
        return false;
    }

    bool found_a = false;
    bool found_e = false;
    bool found_i = false;
    bool found_o = false;
    bool found_u = false;

    for (size_t i = 0; str[i] != '\0'; ++i) {
        char ch = (char)tolower((unsigned char)str[i]);
        if (ch == 'a') found_a = true;
        else if (ch == 'e') found_e = true;
        else if (ch == 'i') found_i = true;
        else if (ch == 'o') found_o = true;
        else if (ch == 'u') found_u = true;
        
        if (found_a && found_e && found_i && found_o && found_u) {
            return true;
        }
    }

    return false;
}

int main(void) {
    char *buffer = NULL;
    size_t bufsize = 0;
    ssize_t len;

    len = getline(&buffer, &bufsize, stdin);

    if (len != -1) {
        if (len > 0 && buffer[len - 1] == '\n') {
            buffer[len - 1] = '\0';
        }

        if (has_all_vowels(buffer)) {
            printf("Accepted\n");
        } else {
            printf("Rejected\n");
        }
        free(buffer);
    } else {
        printf("Error reading input\n");
        free(buffer);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
