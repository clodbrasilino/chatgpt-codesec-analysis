```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U

static size_t bounded_strlen(const char *str, size_t max_len)
{
    size_t i;

    for (i = 0U; i < max_len; i++) {
        if (str[i] == '\0') {
            break;
        }
    }

    return i;
}

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    size_t i;
    size_t first_idx;
    size_t last_idx;
    int found_first;
    int found_last;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = bounded_strlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    found_first = 0;
    found_last = 0;
    first_idx = 0U;
    last_idx = 0U;

    for (i = 0U; i < len; i++) {
        if (str[i] == ch) {
            if (found_first == 0) {
                first_idx = i;
                found_first = 1;
            }
            last_idx = i;
            found_last = 1;
        }
    }

    if (found_first == 0) {
        return;
    }

    if (found_last == 1 && last_idx != first_idx) {
        memmove(&str[last_idx], &str[last_idx + 1U], len - last_idx);
        len--;
    }

    memmove(&str[first_idx], &str[first_idx + 1U], len - first_idx);
}

static int discard_remaining_input(void)
{
    int c;

    do {
        c = getchar();
    } while (c != '\n' && c != EOF);

    return c;
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    if (strchr(buffer, '\n') == NULL) {
        (void)discard_remaining_input();
    }

    buffer[strcspn(buffer, "\n")] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        (void)discard_remaining_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why the code is safe:**

1. **Flawfinder `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120):** This is already mitigated. The buffer is filled exclusively via `fgets(buffer, (int)sizeof(buffer), stdin)`, which enforces the length limit. A forced null terminator is written at `buffer[sizeof(buffer) - 1U]` as defense-in-depth, and `remove_first_last` receives the buffer size explicitly and re-validates length with `bounded_strlen`, so no unbounded string operations occur.

2. **Flawfinder `getchar` in a loop (CWE-120/CWE-20):** This is a false positive here. `discard_remaining_input` never stores the read characters into any buffer — it only consumes them until `'\n'` or `EOF`, so no overflow is possible. The single `getchar` in `main` is checked against `EOF` before the value is used.

3. **Input validation (CWE-20):** Return values of `fgets` and `getchar` are checked; leftover stdin data is flushed when the input line exceeds the buffer, preventing stale bytes from bleeding into the second prompt; the newline is stripped with `strcspn` rather than an unchecked index.

4. **`remove_first_last` internal safety:** The `memmove` calls are bounds-correct — after removing the last occurrence, the move copies `len - last_idx` bytes (which includes the terminating null), and `len` is decremented before the second move, so both operations stay within the string plus its terminator.

The code as flagged is already defensively written; the warnings are static-analysis heuristics rather than actual exploitable defects, so the rewrite preserves the logic with the comments removed.
