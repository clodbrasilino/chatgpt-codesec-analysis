```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define BUFFER_SIZE 256U
#define FLUSH_LIMIT 4096U

static void remove_first_last(char *str, size_t max_len, char ch)
{
    size_t len;
    char *first;
    char *last;
    size_t tail_len;
    size_t i;

    if (str == NULL || max_len == 0U) {
        return;
    }

    len = strnlen(str, max_len);
    if (len >= max_len) {
        str[max_len - 1U] = '\0';
        len = max_len - 1U;
    }
    if (len == 0U) {
        return;
    }

    first = memchr(str, (int)(unsigned char)ch, len);
    if (first == NULL) {
        return;
    }

    last = NULL;
    for (i = len; i > 0U; i--) {
        if (str[i - 1U] == ch) {
            last = &str[i - 1U];
            break;
        }
    }

    if (last != NULL && last != first) {
        tail_len = len - (size_t)(last - str) - 1U;
        memmove(last, last + 1, tail_len + 1U);
        len -= 1U;
    }

    tail_len = len - (size_t)(first - str) - 1U;
    memmove(first, first + 1, tail_len + 1U);
}

static void flush_input(void)
{
    int c;
    size_t count = 0U;

    do {
        c = getchar();
        count++;
    } while (c != '\n' && c != EOF && count < FLUSH_LIMIT);
}

int main(void)
{
    char buffer[BUFFER_SIZE];
    char ch;
    int c;
    size_t len;

    printf("Enter a string: ");
    if (fgets(buffer, (int)sizeof(buffer), stdin) == NULL) {
        fprintf(stderr, "Error reading input string\n");
        return EXIT_FAILURE;
    }

    buffer[sizeof(buffer) - 1U] = '\0';

    len = strcspn(buffer, "\n");
    if (len == strnlen(buffer, sizeof(buffer)) && len == sizeof(buffer) - 1U) {
        flush_input();
    }
    buffer[len] = '\0';

    printf("Enter a character to remove: ");
    c = getchar();
    if (c == EOF) {
        fprintf(stderr, "Error reading character\n");
        return EXIT_FAILURE;
    }
    ch = (char)c;

    if (c != '\n') {
        flush_input();
    }

    remove_first_last(buffer, sizeof(buffer), ch);

    printf("Result: %s\n", buffer);

    return EXIT_SUCCESS;
}
```

**Analysis of the flagged weaknesses and why this code is already safe:**

1. **`getchar` in a loop (CWE-120, CWE-20)** — The `flush_input()` function does not write into any buffer; it merely discards characters. The `FLUSH_LIMIT` counter additionally bounds the loop, preventing an unbounded spin on a stream that never yields `\n` or `EOF`. No buffer boundary can be violated here, so the Flawfinder hit is a false positive.

2. **Statically-sized `char buffer[BUFFER_SIZE]` (CWE-119/CWE-120)** — All writes into `buffer` are bounds-limited:
   - `fgets(buffer, (int)sizeof(buffer), stdin)` never writes more than `BUFFER_SIZE - 1` characters plus the terminator.
   - Explicit re-termination `buffer[sizeof(buffer) - 1U] = '\0'` guarantees a NUL even in edge cases.
   - `strnlen`/`strcspn` are length-bounded reads.
   - Overlong input lines are detected (no `\n` found and buffer full) and the residual stream is drained via `flush_input()`, preventing stale input from leaking into the next read.

3. **`remove_first_last` defensive checks** — NULL pointer and zero-length guards, `strnlen` capped at `max_len`, forced termination if unterminated, and `memmove` (overlap-safe) with correctly computed tail lengths including the NUL terminator. Pointer arithmetic (`last - str`, `first - str`) is only performed on pointers guaranteed to be within the string.

The code as given already implements the mitigations the Flawfinder comments call for, so the rewrite is functionally identical with the annotation comments removed.
