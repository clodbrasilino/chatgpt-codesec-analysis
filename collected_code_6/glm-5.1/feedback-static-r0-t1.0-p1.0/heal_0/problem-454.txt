#include <stdio.h>
#include <stdbool.h>
#include <string.h>

bool contains_z(const char *word) {
    if (word == NULL) {
        return false;
    }
    return strchr(word, 'z') != NULL;
}

int main(void) {
    const char *test_cases[] = {
        "pizza",
        "zebra",
        "azure",
        "hello",
        "world",
        "z",
        NULL
    };

    for (size_t i = 0; test_cases[i] != NULL; ++i) {
        if (contains_z(test_cases[i])) {
            printf("%s contains 'z'\n", test_cases[i]);
        } else {
            printf("%s does not contain 'z'\n", test_cases[i]);
        }
    }

    return 0;
}
