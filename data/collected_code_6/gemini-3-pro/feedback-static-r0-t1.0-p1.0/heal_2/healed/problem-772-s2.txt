#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void remove_words_of_length_k(char *str, size_t max_len, int k) {
    if (str == NULL || k <= 0 || max_len == 0) {
        return;
    }

    size_t len = 0;
    while (len < max_len && str[len] != '\0') {
        len++;
    }

    char *result = (char *)malloc(len + 1);
    if (result == NULL) {
        return;
    }

    size_t i = 0;
    size_t j = 0;
    int word_start = -1;
    int word_len = 0;

    while (i <= len) {
        if (i < len && str[i] != '\0' && !isspace((unsigned char)str[i])) {
            if (word_start == -1) {
                word_start = (int)i;
            }
            word_len++;
        } else {
            if (word_start != -1) {
                if (word_len != k) {
                    for (size_t x = (size_t)word_start; x < i; x++) {
                        result[j++] = str[x];
                    }
                }
                word_start = -1;
                word_len = 0;
            }
            if (i == len || str[i] == '\0') {
                break;
            }
            if (j > 0 && result[j-1] != ' ' && str[i] == ' ') {
                result[j++] = str[i];
            } else if (str[i] != ' ') {
                result[j++] = str[i];
            }
        }
        i++;
    }

    if (j > 0 && isspace((unsigned char)result[j - 1])) {
        j--;
    }
    
    result[j] = '\0';
    
    size_t copy_len = j + 1;
    if (copy_len > max_len) {
        copy_len = max_len;
    }
    
    for (size_t idx = 0; idx < copy_len; idx++) {
        str[idx] = result[idx];
    }
    str[max_len - 1] = '\0';
    
    free(result);
}

int main(void) {
    size_t buffer_size = 1024;
    char *str = (char *)malloc(buffer_size);
    int k;

    if (str == NULL) {
        return EXIT_FAILURE;
    }

    if (fgets(str, (int)buffer_size, stdin) != NULL) {
        str[strcspn(str, "\n")] = '\0';

        if (scanf("%d", &k) == 1) {
            remove_words_of_length_k(str, buffer_size, k);
            printf("%s\n", str);
        }
    }

    free(str);
    return EXIT_SUCCESS;
}
