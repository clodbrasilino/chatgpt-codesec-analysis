#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX_WORDS 100
#define MAX_WORD_LEN 50

typedef struct {
    char word[MAX_WORD_LEN];
    int position;
} AdverbInfo;

static int is_adverb(const char *word, size_t word_len) {
    if (word_len < 2) return 0;
    return (word[word_len - 2] == 'l' && word[word_len - 1] == 'y');
}

static void safe_str_copy(char *dest, const char *src, size_t dest_size) {
    if (dest == NULL || src == NULL || dest_size == 0) {
        return;
    }
    
    size_t i;
    for (i = 0; i < dest_size - 1 && src[i] != '\0'; i++) {
        dest[i] = src[i];
    }
    dest[i] = '\0';
}

int find_adverbs(const char *sentence, AdverbInfo *adverbs, int max_adverbs) {
    if (sentence == NULL || adverbs == NULL || max_adverbs <= 0) {
        return -1;
    }

    int count = 0;
    int pos = 0;
    size_t len = 0;
    const char *ptr = sentence;
    
    while (*ptr != '\0' && len < SIZE_MAX) {
        len++;
        ptr++;
    }
    
    if (len == SIZE_MAX) {
        return -1;
    }
    
    char *buffer = (char *)malloc(len + 1);
    if (buffer == NULL) {
        return -1;
    }
    
    for (size_t i = 0; i <= len; i++) {
        buffer[i] = sentence[i];
    }
    buffer[len] = '\0';

    const char *delimiters = " ,.!?;:\"()[]{}";
    char *saveptr = NULL;
    char *token = strtok_r(buffer, delimiters, &saveptr);
    
    while (token != NULL && count < max_adverbs) {
        size_t token_len = 0;
        const char *t = token;
        
        while (*t != '\0' && token_len < SIZE_MAX) {
            token_len++;
            t++;
        }
        
        if (token_len == SIZE_MAX) {
            break;
        }
        
        if (token_len > 0 && is_adverb(token, token_len)) {
            safe_str_copy(adverbs[count].word, token, MAX_WORD_LEN);
            adverbs[count].position = pos;
            count++;
        }
        pos++;
        token = strtok_r(NULL, delimiters, &saveptr);
    }

    free(buffer);
    return count;
}

int main(void) {
    char sentence[500];
    AdverbInfo adverbs[MAX_WORDS];

    printf("Enter a sentence: ");
    if (fgets(sentence, sizeof(sentence), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return 1;
    }

    size_t input_len = 0;
    char *p = sentence;
    while (*p != '\0' && input_len < sizeof(sentence)) {
        input_len++;
        p++;
    }
    
    if (input_len > 0 && input_len <= sizeof(sentence) - 1 && sentence[input_len - 1] == '\n') {
        sentence[input_len - 1] = '\0';
        input_len--;
    }
    
    if (input_len >= sizeof(sentence)) {
        sentence[sizeof(sentence) - 1] = '\0';
    }

    int result = find_adverbs(sentence, adverbs, MAX_WORDS);
    if (result < 0) {
        fprintf(stderr, "Error processing sentence\n");
        return 1;
    }

    if (result == 0) {
        printf("No adverbs found.\n");
    } else {
        printf("Adverbs found:\n");
        for (int i = 0; i < result; i++) {
            printf("Word: %s, Position: %d\n", adverbs[i].word, adverbs[i].position);
        }
    }

    return 0;
}
