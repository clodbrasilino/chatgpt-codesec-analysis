#include <stdio.h>
#include <stdlib.h>

typedef struct sublist_node {
    int data;
    struct sublist_node *next;
} sublist_node_t;

typedef struct list_node {
    sublist_node_t *head;
    struct list_node *next;
} list_node_t;

void free_sublist(sublist_node_t *head) {
    sublist_node_t *current = head;
    while (current != NULL) {
        sublist_node_t *next = current->next;
        free(current);
        current = next;
    }
}

void free_list(list_node_t *head) {
    list_node_t *current = head;
    while (current != NULL) {
        list_node_t *next = current->next;
        free_sublist(current->head);
        free(current);
        current = next;
    }
}

size_t get_sublist_length(const sublist_node_t *head) {
    size_t length = 0;
    const sublist_node_t *current = head;
    while (current != NULL) {
        length++;
        current = current->next;
    }
    return length;
}

void filter_sublists(list_node_t **head_ref, size_t min_len, size_t max_len) {
    if (head_ref == NULL) {
        return;
    }

    list_node_t **current_ptr = head_ref;
    while (*current_ptr != NULL) {
        list_node_t *entry = *current_ptr;
        size_t len = get_sublist_length(entry->head);
        
        if (len < min_len || len > max_len) {
            *current_ptr = entry->next;
            free_sublist(entry->head);
            free(entry);
        } else {
            current_ptr = &(entry->next);
        }
    }
}

sublist_node_t *create_sublist_node(int data) {
    sublist_node_t *node = (sublist_node_t *)malloc(sizeof(sublist_node_t));
    if (node == NULL) {
        return NULL;
    }
    node->data = data;
    node->next = NULL;
    return node;
}

int append_to_sublist(sublist_node_t **head_ref, int data) {
    if (head_ref == NULL) {
        return -1;
    }
    
    sublist_node_t *new_node = create_sublist_node(data);
    if (new_node == NULL) {
        return -1;
    }
    
    if (*head_ref == NULL) {
        *head_ref = new_node;
        return 0;
    }
    
    sublist_node_t *current = *head_ref;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = new_node;
    return 0;
}

list_node_t *create_list_node(sublist_node_t *sublist_head) {
    list_node_t *node = (list_node_t *)malloc(sizeof(list_node_t));
    if (node == NULL) {
        return NULL;
    }
    node->head = sublist_head;
    node->next = NULL;
    return node;
}

int append_to_list(list_node_t **head_ref, sublist_node_t *sublist_head) {
    if (head_ref == NULL) {
        return -1;
    }
    
    list_node_t *new_node = create_list_node(sublist_head);
    if (new_node == NULL) {
        return -1;
    }
    
    if (*head_ref == NULL) {
        *head_ref = new_node;
        return 0;
    }
    
    list_node_t *current = *head_ref;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = new_node;
    return 0;
}

void print_list(const list_node_t *head) {
    const list_node_t *current = head;
    while (current != NULL) {
        const sublist_node_t *sub_current = current->head;
        while (sub_current != NULL) {
            printf("%d ", sub_current->data);
            sub_current = sub_current->next;
        }
        printf("\n");
        current = current->next;
    }
}

int main(void) {
    list_node_t *list = NULL;
    sublist_node_t *sub1 = NULL;
    sublist_node_t *sub2 = NULL;
    sublist_node_t *sub3 = NULL;

    if (append_to_sublist(&sub1, 1) != 0 || 
        append_to_sublist(&sub1, 2) != 0 || 
        append_to_sublist(&sub1, 3) != 0) {
        free_sublist(sub1);
        return EXIT_FAILURE;
    }

    if (append_to_list(&list, sub1) != 0) {
        free_sublist(sub1);
        free_list(list);
        return EXIT_FAILURE;
    }

    if (append_to_sublist(&sub2, 4) != 0 || 
        append_to_sublist(&sub2, 5) != 0) {
        free_sublist(sub2);
        free_list(list);
        return EXIT_FAILURE;
    }

    if (append_to_list(&list, sub2) != 0) {
        free_sublist(sub2);
        free_list(list);
        return EXIT_FAILURE;
    }

    if (append_to_sublist(&sub3, 6) != 0) {
        free_sublist(sub3);
        free_list(list);
        return EXIT_FAILURE;
    }

    if (append_to_list(&list, sub3) != 0) {
        free_sublist(sub3);
        free_list(list);
        return EXIT_FAILURE;
    }

    filter_sublists(&list, 2, 2);

    print_list(list);

    free_list(list);

    return EXIT_SUCCESS;
}
