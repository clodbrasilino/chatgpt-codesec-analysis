```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

typedef struct Tuple {
    int *elements;
    size_t count;
} Tuple;

Node* createNode(int value) {
    Node *newNode = (Node *)malloc(sizeof(Node));
    if (newNode == NULL) exit(EXIT_FAILURE);
    newNode->value = value;
    newNode->next = NULL;
    return newNode;
}

void appendToList(Node **head, int value) {
    Node *newNode = createNode(value);
    if (*head == NULL) {
        *head = newNode;
    } else {
        Node *current = *head;
        while (current->next != NULL) {
            current = current->next;
        }
        current->next = newNode;
    }
}

Tuple listToTuple(Node *head) {
    if (head == NULL) {
        Tuple t = {NULL, 0};
        return t;
    }

    Node *current = head;
    size_t count = 0;
    while (current != NULL) {
        count++;
        current = current->next;
    }

    int *elements = (int *)malloc(count * sizeof(int));
    if (elements == NULL) exit(EXIT_FAILURE);

    current = head;
    for (size_t i = 0; i < count; i++) {
        elements[i] = current->value;
        current = current->next;
    }

    Tuple t = {elements, count};
    return t;
}

void freeList(Node *head) {
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

void freeTuple(Tuple t) {
    if (t.elements != NULL) {
        free(t.elements);
    }
}

int main() {
    Node *list = NULL;
    appendToList(&list, 1);
    appendToList(&list, 2);
    appendToList(&list, 3);

    Tuple tuple = listToTuple(list);
    for (size_t i = 0; i < tuple.count; i++) {
        printf("%d ", tuple.elements[i]);
    }
    printf("\n");

    freeList(list);
    freeTuple(tuple);

    return 0;
}
```
