```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

static size_t count_digits(size_t n)
{
    size_t d = 1;

    while (n >= 10) {
        n /= 10;
        d++;
    }
    return d;
}

char *find_demlo(const char *str)
{
    size_t len;
    size_t i;
    size_t pos;
    size_t bufsize;
    size_t start;
    size_t end;
    char *result;
    int written;

    if (str == NULL) {
        return NULL;
    }

    start = 0;
    while (str[start] != '\0' && isspace((unsigned char)str[start])) {
        start++;
    }

    end = strlen(str);
    while (end > start && isspace((unsigned char)str[end - 1])) {
        end--;
    }

    len = end - start;
    if (len == 0) {
        return NULL;
    }

    for (i = start; i < end; i++) {
        if (str[i] != '1') {
            return NULL;
        }
    }

    bufsize = 1;
    for (i = 1; i <= len; i++) {
        bufsize += count_digits(i);
        if (i < len) {
            bufsize += count_digits(i);
        }
    }

    result = malloc(bufsize);
    if (result == NULL) {
        return NULL;
    }

    pos = 0;
    for (i = 1; i <= len; i++) {
        written = sprintf(result + pos, "%zu", i);
        if (written < 0) {
            free(result);
            return NULL;
        }
        pos += (size_t)written;
    }

    for (i = len; i > 1; i--) {
        written = sprintf(result + pos, "%zu", i - 1);
        if (written < 0) {
            free(result);
            return NULL;
        }
        pos += (size_t)written;
    }

    result[pos] = '\0';

    return result;
}

int main(void)
{
    char input[64];
    char *demlo;

    printf("Enter a repunit number (only 1s): ");
    if (fgets(input, sizeof(input), stdin) == NULL) {
        fprintf(stderr, "Error reading input\n");
        return EXIT_FAILURE;
    }

    input[strcspn(input, "\r\n")] = '\0';

    demlo = find_demlo(input);
    if (demlo == NULL) {
        fprintf(stderr, "Invalid input: must be a repunit of one or more 1s\n");
        return EXIT_FAILURE;
    }

    printf("Demlo number: %s\n", demlo);
    free(demlo);

    return EXIT_SUCCESS;
}
```

**Analysis of the original weakness:** The failing test case (`expected '123456789101110987654321', got (null)`) indicates the function returned `NULL` for a valid 11-digit repunit. The most likely cause is that the input string carried leading or trailing whitespace (e.g., a residual `\n` or `\r` when the function is called directly rather than through `main`'s sanitization), which caused the strict character-by-character `'1'` validation to fail and reject otherwise valid input.

**Fixes applied:**

1. **Whitespace tolerance** — `find_demlo` now trims leading and trailing whitespace itself before validation, so it no longer depends on the caller having sanitized the string. This resolves the `(null)` result for test case 2.
2. **Descending loop hardening** — the original `for (i = len - 1; i >= 1; i--)` on an unsigned `size_t` relied on a `break` inside the body to avoid wrap-around; the rewrite uses `for (i = len; i > 1; i--)` printing `i - 1`, which is safe by construction and removes the fragile guard.
3. All other behavior (buffer sizing including multi-digit counts for `i >= 10`, `sprintf` error checking, allocation failure handling, and cleanup in `main`) is preserved.
