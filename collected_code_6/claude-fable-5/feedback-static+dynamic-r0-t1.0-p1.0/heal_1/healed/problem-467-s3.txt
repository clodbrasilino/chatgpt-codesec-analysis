```c
#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define TEMP_SIZE ((sizeof(unsigned long) * CHAR_BIT + 2) / 3 + 1)

int decimal_to_octal(long decimal, char *buffer, size_t buffer_size)
{
    char temp[TEMP_SIZE];
    size_t index = 0;
    size_t i;
    int negative = 0;
    unsigned long value;

    if (buffer == NULL || buffer_size < 2) {
        return -1;
    }

    if (decimal < 0) {
        negative = 1;
        value = (unsigned long)(-(decimal + 1)) + 1UL;
    } else {
        value = (unsigned long)decimal;
    }

    if (value == 0) {
        buffer[0] = '0';
        buffer[1] = '\0';
        return 0;
    }

    while (value > 0 && index < sizeof(temp)) {
        temp[index] = (char)('0' + (value % 8UL));
        value /= 8UL;
        index++;
    }

    if (value > 0) {
        return -1;
    }

    if (index + (size_t)negative + 1 > buffer_size) {
        return -1;
    }

    i = 0;
    if (negative == 1) {
        buffer[i] = '-';
        i++;
    }

    while (index > 0) {
        index--;
        buffer[i] = temp[index];
        i++;
    }
    buffer[i] = '\0';

    return 0;
}

int main(void)
{
    long decimal;
    char octal[64];
    int scan_result;

    printf("Enter a decimal number: ");
    scan_result = scanf("%ld", &decimal);

    if (scan_result != 1) {
        fprintf(stderr, "Error: invalid input.\n");
        return EXIT_FAILURE;
    }

    if (decimal_to_octal(decimal, octal, sizeof(octal)) != 0) {
        fprintf(stderr, "Error: conversion failed.\n");
        return EXIT_FAILURE;
    }

    printf("Octal representation: %s\n", octal);

    return EXIT_SUCCESS;
}
```

Analysis of the changes made:

1. **`temp` buffer sizing (CWE-119/CWE-120):** The fixed `char temp[32]` was replaced with a size computed from the platform's `unsigned long` width (`TEMP_SIZE`). Each octal digit encodes 3 bits, so this guarantees the buffer is always large enough for the maximum possible value on any platform, rather than relying on an assumption that 32 digits suffice.

2. **Silent truncation check:** In the original code, if `index` reached `sizeof(temp)` while `value` was still non-zero, the loop exited and produced a silently truncated (incorrect) result. The rewrite adds an explicit `if (value > 0) return -1;` check after the loop so truncation is reported as an error instead of producing wrong output.

3. **Existing mitigations preserved:** The original code already handled the important cases correctly, and these were kept intact:
   - `LONG_MIN` negation is done safely via `-(decimal + 1) + 1UL` in unsigned arithmetic, avoiding signed overflow (undefined behavior).
   - Output bounds checking (`index + negative + 1 > buffer_size`) prevents overflow of the caller's buffer.
   - `NULL` and minimum-size validation on `buffer`.
   - `scanf` return value is checked in `main`, and `%ld` with a `long*` poses no buffer risk.

The `char octal[64]` in `main` is safe as-is because the conversion function performs strict bounds checking against the passed `sizeof(octal)`, so no change was needed there.
