#include <stdio.h>

int find_first_occurrence(int arr[], int size, int target) {
    int left = 0;
    int right = size - 1;
    int result = -1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target) {
            result = mid;
            right = mid - 1;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return result;
}

int main(void) {
    int arr[] = {1, 2, 2, 2, 3, 4, 5, 5, 6};
    int size = sizeof(arr) / sizeof(arr[0]);
    int target1 = 2;
    int target2 = 5;
    int target3 = 7;
    
    int index1 = find_first_occurrence(arr, size, target1);
    if (index1 != -1) {
        printf("First occurrence of %d is at index %d\n", target1, index1);
    } else {
        printf("%d not found in array\n", target1);
    }
    
    int index2 = find_first_occurrence(arr, size, target2);
    if (index2 != -1) {
        printf("First occurrence of %d is at index %d\n", target2, index2);
    } else {
        printf("%d not found in array\n", target2);
    }
    
    int index3 = find_first_occurrence(arr, size, target3);
    if (index3 != -1) {
        printf("First occurrence of %d is at index %d\n", target3, index3);
    } else {
        printf("%d not found in array\n", target3);
    }
    
    return 0;
}
