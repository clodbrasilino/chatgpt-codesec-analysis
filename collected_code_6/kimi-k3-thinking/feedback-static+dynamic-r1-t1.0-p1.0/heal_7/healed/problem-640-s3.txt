#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <ctype.h>

char *remove_parenthesis_area(const char *str)
{
    if (str == NULL) {
        return NULL;
    }

    size_t len = strlen(str);

    if (len == SIZE_MAX) {
        return NULL;
    }

    char *buffer = malloc(len + 1);
    if (buffer == NULL) {
        return NULL;
    }

    size_t depth = 0;
    size_t j = 0;

    for (size_t i = 0; i < len; i++) {
        unsigned char ch = (unsigned char)str[i];

        if (ch == '(') {
            depth++;
        } else if (ch == ')') {
            if (depth > 0) {
                depth--;
            } else {
                buffer[j++] = (char)ch;
            }
        } else if (depth == 0) {
            buffer[j++] = (char)ch;
        }
    }

    char *result = malloc(j + 1);
    if (result == NULL) {
        free(buffer);
        return NULL;
    }

    size_t w = 0;
    size_t r = 0;
    int pending_space = 0;

    while (r < j && isspace((unsigned char)buffer[r])) {
        r++;
    }

    for (; r < j; r++) {
        unsigned char ch = (unsigned char)buffer[r];
        if (isspace(ch)) {
            pending_space = 1;
        } else {
            if (pending_space && w > 0) {
                result[w++] = ' ';
            }
            result[w++] = (char)ch;
            pending_space = 0;
        }
    }

    result[w] = '\0';

    free(buffer);
    return result;
}

static char *read_line(FILE *stream)
{
    size_t capacity = 128;
    size_t length = 0;

    char *buffer = malloc(capacity);
    if (buffer == NULL) {
        return NULL;
    }

    int ch;
    while ((ch = fgetc(stream)) != EOF && ch != '\n') {
        if (length + 1 >= capacity) {
            if (capacity > SIZE_MAX / 2) {
                free(buffer);
                return NULL;
            }
            capacity *= 2;
            char *resized = realloc(buffer, capacity);
            if (resized == NULL) {
                free(buffer);
                return NULL;
            }
            buffer = resized;
        }
        buffer[length++] = (char)ch;
    }

    if (ferror(stream) != 0) {
        free(buffer);
        return NULL;
    }

    if (ch == EOF && length == 0) {
        free(buffer);
        return NULL;
    }

    if (length > 0 && buffer[length - 1] == '\r') {
        length--;
    }

    buffer[length] = '\0';
    return buffer;
}

int main(int argc, char *argv[])
{
    char *input = NULL;
    const char *source;

    if (argc > 1) {
        source = argv[1];
    } else {
        input = read_line(stdin);
        if (input == NULL) {
            fprintf(stderr, "Error reading input\n");
            return EXIT_FAILURE;
        }
        source = input;
    }

    char *cleaned = remove_parenthesis_area(source);
    if (cleaned == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        free(input);
        return EXIT_FAILURE;
    }

    printf("%s\n", cleaned);

    free(cleaned);
    free(input);

    return EXIT_SUCCESS;
}
