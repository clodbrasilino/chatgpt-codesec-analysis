#include <stdio.h>
#include <stdlib.h>

double average(const int *list, size_t length, int *error)
{
    double sum = 0.0;
    size_t i;

    if (list == NULL || length == 0U) {
        if (error != NULL) {
            *error = 1;
        }
        return 0.0;
    }

    for (i = 0U; i < length; i++) {
        sum += (double)list[i];
    }

    if (error != NULL) {
        *error = 0;
    }

    return sum / (double)length;
}

int main(void)
{
    int numbers[] = {10, 20, 30, 40, 50};
    size_t count = sizeof(numbers) / sizeof(numbers[0]);
    int error = 0;
    double result;

    result = average(numbers, count, &error);

    if (error != 0) {
        if (fprintf(stderr, "Error: invalid list\n") < 0) {
            return EXIT_FAILURE;
        }
        return EXIT_FAILURE;
    }

    if (printf("Average: %.2f\n", result) < 0) {
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
